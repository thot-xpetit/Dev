    PROCEDURE UpdateInventoryLookup@1100409000();
    VAR
      TSUtil@1100409002 : Codeunit 99008909;
      optionStr@1100409001 : Text[250];
      result@1100409000 : Integer;
      PROMPT_INVUPDATE@1100409004 : TextConst 'ENU=Update the POS Inventory Lookup table for all Product Groups?,Update the POS Inventory Lookup table for %1 only?';
      LOOKUP_DISABLED@1100409003 : TextConst 'ENU=POS Inventory Lookup is not enabled for this Product Group';
    BEGIN
      //LS
      //
      // Updates the POS Inventory Lookup table for either the selected Product Group or all
      // Product Groups.

      optionStr := STRSUBSTNO(PROMPT_INVUPDATE,Code);
      result := STRMENU(optionStr,2);

      IF result = 0 THEN
        EXIT;

      IF result = 1 THEN
        TSUtil.UpdateLookupTable('','')
      ELSE BEGIN
        IF NOT "POS Inventory Lookup" THEN
          ERROR(LOOKUP_DISABLED);
        TSUtil.UpdateLookupTable('',Code);
      END;
    END;
