    PROCEDURE CheckQtyToAssign@8053006();

    PROCEDURE CheckAssignment@8053007(P_TourNo@8053000 : Code[20]);
    VAR
      L_recOutstandingLineSrc@8053001 : Record 8053044;
      L_recWarehouseShipmentHeader@8053002 : Record 7320;
      L_recWarehouseShipmentLine@8053003 : Record 7321;
      L_recItem@8053004 : Record 27;
    BEGIN
      // ---------------------------------------------------------------
      // VERIFICATION LORS DE L'ASSIGNATION ( FONCTION F9 )
      // ---------------------------------------------------------------

      // ---------------------------------------------------------------
      // Statut du document = Lanc'
      // ---------------------------------------------------------------
      IF L_recOutstandingLineSrc.GET("Source Document Type", "Source Document No.", "Source Document Line No.", 0, 0) THEN
        IF L_recOutstandingLineSrc."Document Status" <> L_recOutstandingLineSrc."Document Status"::Released THEN
          ERROR(Text8053000);

      // ---------------------------------------------------------------
      // Statut de la tourn'e = Ouvert
      // ---------------------------------------------------------------
      L_recWarehouseShipmentHeader.GET(P_TourNo);
      L_recWarehouseShipmentHeader.TESTFIELD(Status, L_recWarehouseShipmentHeader.Status::Open);

      // ---------------------------------------------------------------
      // Stock magasin suffisant ( Non bloquant )
      // ---------------------------------------------------------------
      L_recItem.RESET;
      L_recItem.SETRANGE("No.", "Item No.");
      L_recItem.SETFILTER("Location Filter", '%1', "Departure Location Code");
      L_recItem.SETFILTER("Variant Filter", '%1', "Variant Code");
      IF L_recItem.FINDFIRST THEN;
      L_recItem.CALCFIELDS(Inventory);

      L_recWarehouseShipmentLine.RESET;
      L_recWarehouseShipmentLine.SETCURRENTKEY("Item No.","Location Code","Variant Code");
      L_recWarehouseShipmentLine.SETRANGE("Item No.", "Item No.");
      L_recWarehouseShipmentLine.SETRANGE("Location Code", "Departure Location Code");
      L_recWarehouseShipmentLine.SETRANGE("Variant Code", "Variant Code");
      L_recWarehouseShipmentLine.CALCSUMS("Qty. to Ship");

      IF (L_recItem.Inventory - L_recWarehouseShipmentLine."Qty. to Ship") < "Qty. to Assign" THEN

      IF NOT CONFIRM(Text8053004, FALSE, "Departure Location Code",
        L_recItem.Inventory - L_recWarehouseShipmentLine."Qty. to Ship", L_recItem."No.")
      THEN
          ERROR('')
      // --> 20140716-TER01.01.00-RMZ01 Begin
      ELSE
        // --> 20171203-WI00-ADV01.01.00-3286-FBE01
        IF NOT "Build Kit" THEN
        // <-- 20171203-WI00-ADV01.01.00-3286-FBE01
          VerifyItemSubstitution("Item No.",P_TourNo);
      // <-- 20140716-TER01.01.00-RMZ01 End
    END;

    PROCEDURE CheckADRExempt@8053011() R_IsExempt : Boolean;
    VAR
      L_recADR@8053000 : Record 8053054;
      L_recTransCategory@8053001 : Record 8053055;
      L_recItem@8053002 : Record 27;
      L_TotalADR@8053003 : Decimal;
      L_MaxiWeight@8053004 : Decimal;
      L_TotalWeight@8053005 : Decimal;
    BEGIN
      IF L_recItem.GET("Item No.") THEN
       IF L_recADR.GET(L_recItem."ADR Code") THEN
        IF L_recTransCategory.GET(L_recADR."Transport Category") THEN
         BEGIN
          L_TotalADR += "Qty. to Assign"*L_recItem."Gross Weight"*L_recTransCategory.Coefficient;
          L_MaxiWeight := L_recTransCategory."Maximum Quantity";
          L_TotalWeight += "Qty. to Assign"*L_recItem."Gross Weight";
         END;
      IF (L_TotalWeight = 0) AND (L_MaxiWeight = 0) THEN
       R_IsExempt := TRUE
      ELSE
       BEGIN
        IF L_TotalWeight <= L_MaxiWeight THEN
         R_IsExempt := TRUE
        ELSE
         R_IsExempt := FALSE;
       END;
      EXIT(R_IsExempt);
    END;

    PROCEDURE ItemAvailability@12(AvailabilityType@1000 : 'Date,Variant,Location,Bin');
    VAR
      OutstandingOrderLine_L@8053000 : Record 8053044;
      SalesLine_L@8053001 : Record 37;
      SalesHeader_L@8053004 : Record 36;
      WhRequest_L@8053002 : Record 5765;
      WhseSalesRelease_L@8053003 : Codeunit 5771;
    BEGIN
      TESTFIELD("Item No.");
      Item_G.RESET;
      Item_G.GET("Item No.");
      Item_G.SETRANGE("No.","Item No.");
      Item_G.SETRANGE("Date Filter",0D,"Promised Delivery Date");

      CASE AvailabilityType OF
        AvailabilityType::Location:
          BEGIN
            Item_G.SETRANGE("Variant Filter","Variant Code");
            CLEAR(ItemAvailByLoc);
            ItemAvailByLoc.LOOKUPMODE(TRUE);
            ItemAvailByLoc.SETRECORD(Item_G);
            ItemAvailByLoc.SETTABLEVIEW(Item_G);
            OutstandingOrderLine_L.GET("Source Document Type","Source Document No.","Source Document Line No.",0,0);
            IF (ItemAvailByLoc.RUNMODAL = ACTION::LookupOK) AND
              (OutstandingOrderLine_L."Source Document Type"=OutstandingOrderLine_L."Source Document Type"::"Sales Order")
            THEN BEGIN
              IF "Departure Location Code" <> ItemAvailByLoc.GetLastLocation THEN
                IF CONFIRM(
                     Text8053006,TRUE,FIELDCAPTION("Departure Location Code"),"Departure Location Code",
                     ItemAvailByLoc.GetLastLocation)
                THEN BEGIN
                  //update ligne vente
                  SalesLine_L.GET(OutstandingOrderLine_L."Source Document Type",OutstandingOrderLine_L."Source Document No.",
                     OutstandingOrderLine_L."Source Document Line No.");
                  SalesLine_L."Location Code":=ItemAvailByLoc.GetLastLocation;
                  SalesLine_L.MODIFY;

                  // update Warehouse Request
                  IF SalesHeader_L.GET(SalesHeader_L."Document Type"::Order,OutstandingOrderLine_L."Source Document No.")
                    THEN BEGIN
                      WhseSalesRelease_L.Reopen(SalesHeader_L);
                      WhseSalesRelease_L.Release(SalesHeader_L);
                    END;

                  IF CurrFieldNo = 0 THEN
                    xRec := Rec;
                  VALIDATE("Departure Location Code",ItemAvailByLoc.GetLastLocation);
                  //update CMC Outstanding Order Line
                  OutstandingOrderLine_L."Departure Location Code":=ItemAvailByLoc.GetLastLocation;
                  OutstandingOrderLine_L.MODIFY;
                END;
            END;
          END;
      END;
    END;
