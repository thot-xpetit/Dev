    PROCEDURE VariantDelete@1200000000();
    VAR
      ItemTranslation@1200000004 : Record 30;
      SKU@1200000002 : Record 5700;
      ItemTrans@1200000006 : Record 32;
      ISText001@1200000007 : TextConst 'ENU=Inventory transactions exist on variant %1';
      Barcodes@1200070003 : Record 99001451;
      SalesLine@1200070002 : Record 37;
      PurchaseLine@1200070001 : Record 39;
      ItemLedgerEntry@1200070000 : Record 32;
      ItemIdent@10012000 : Record 7704;
    BEGIN
      //LS
      Barcodes.RESET;
      Barcodes.SETCURRENTKEY("Item No.","Variant Code");
      Barcodes.SETRANGE("Item No.","Item No.");
      Barcodes.SETRANGE("Variant Code",Code);
      IF Barcodes.FIND('-') THEN
        ERROR(Text001,TABLECAPTION,"Item No.",Code,Barcodes."Barcode No.");

      SalesLine.RESET;
      SalesLine.SETCURRENTKEY(Type,"No.","Variant Code");
      SalesLine.SETRANGE(Type,SalesLine.Type::Item);
      SalesLine.SETRANGE("No.","Item No.");
      SalesLine.SETRANGE("Variant Code",Code);
      IF SalesLine.FIND('-') THEN
        ERROR(Text002,TABLECAPTION,"Item No.",Code,SalesLine."Document Type",SalesLine."Document No.");

      PurchaseLine.RESET;
      PurchaseLine.SETCURRENTKEY(Type,"No.","Variant Code");
      PurchaseLine.SETRANGE(Type,SalesLine.Type::Item);
      PurchaseLine.SETRANGE("No.","Item No.");
      PurchaseLine.SETRANGE("Variant Code",Code);
      IF PurchaseLine.FIND('-') THEN
        ERROR(Text003,TABLECAPTION,"Item No.",Code,PurchaseLine."Document Type",PurchaseLine."Document No.");

      ItemLedgerEntry.RESET;
      ItemLedgerEntry.SETCURRENTKEY("Item No.",Open,"Variant Code");
      ItemLedgerEntry.SETRANGE("Item No.","Item No.");
      ItemLedgerEntry.SETRANGE("Variant Code",Code);
      ItemLedgerEntry.SETRANGE(Open,TRUE);
      IF ItemLedgerEntry.FIND('-') THEN
        ERROR(Text004,TABLECAPTION,"Item No.",Code,ItemLedgerEntry.TABLECAPTION);

      ItemTrans.SETCURRENTKEY("Item No.","Entry Type","Variant Code");
      ItemTrans.SETRANGE(ItemTrans."Item No.","Item No.");
      ItemTrans.SETRANGE(ItemTrans."Variant Code",Code);
      IF ItemTrans.FIND('-') THEN
        ERROR(ISText001,Code);

      Barcodes.SETRANGE("Item No.","Item No.");
      Barcodes.SETRANGE("Variant Code",Code);
      Barcodes.DELETEALL(TRUE);

      ItemTranslation.SETRANGE("Item No.","Item No.");
      ItemTranslation.SETRANGE("Variant Code",Code);
      ItemTranslation.DELETEALL(TRUE);

      SKU.SETRANGE("Item No.","Item No.");
      SKU.SETRANGE("Variant Code",Code);
      SKU.DELETEALL(TRUE);


      ItemIdent.RESET;
      ItemIdent.SETCURRENTKEY("Item No.");
      ItemIdent.SETRANGE("Item No.","Item No.");
      ItemIdent.SETRANGE("Variant Code",Code);
      ItemIdent.DELETEALL;
    END;
