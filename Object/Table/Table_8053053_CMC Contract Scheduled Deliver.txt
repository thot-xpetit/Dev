    PROCEDURE MajDateFilter@8053000();
    BEGIN

    END;

    PROCEDURE CreateForecastDelivery@8053001(CMCContractSchedDeliver_P@8053000 : Record 8053053);
    VAR
      OutstandingQuantity_L@8053001 : Decimal;
      NbTruck_L@8053002 : Integer;
      NbTruckMax_L@8053003 : Integer;
      CMCDelivery_L@8053004 : Record 8053013;
      CMCSetup_L@8053005 : Record 8052982;
      NoSeriesMgtcdu_L@8053006 : Codeunit 396;
      CMCSalesContract_L@8053007 : Record 8053012;
      ForeCastQuantity_L@8053008 : Decimal;
    BEGIN
      // G'n'ration des bons de sorties pr'visionnelles

      OutstandingQuantity_L := CMCContractSchedDeliver_P.Quantity  - ( CMCContractSchedDeliver_P."Delivered Quantity" +
                               CMCContractSchedDeliver_P."Forecast Quantity" + CMCContractSchedDeliver_P."Planned Delivery Quantity" );
      IF OutstandingQuantity_L <= 0 THEN
       ERROR(Text8053000);

      IF CMCContractSchedDeliver_P."Truck Capacity" = 0 THEN
       ERROR(Text8053001);

      NbTruckMax_L := ROUND ( OutstandingQuantity_L / CMCContractSchedDeliver_P."Truck Capacity",1,'>');
      CMCSetup_L.GET;
      ForeCastQuantity_L := CMCContractSchedDeliver_P.Quantity;
      FOR NbTruck_L := 1 TO NbTruckMax_L
      DO BEGIN
       CMCDelivery_L.INIT;
       CMCSalesContract_L.GET("Contract No.");
       CMCDelivery_L."No." := NoSeriesMgtcdu_L.GetNextNo(CMCSetup_L."Collection Delivery Nos.",WORKDATE,TRUE);
       CMCDelivery_L.VALIDATE("No.");
       CMCDelivery_L.INSERT(TRUE);

       CMCDelivery_L."Delivery Type" := CMCDelivery_L."Delivery Type"::Delivery;
       CMCDelivery_L.VALIDATE("Location Code",CMCSalesContract_L."Default Location Code");
       CMCDelivery_L.VALIDATE("Customer No.",CMCSalesContract_L."Customer No.");
       CMCDelivery_L.VALIDATE("Species Code",CMCSalesContract_L."Species Code");
       CMCDelivery_L.VALIDATE("Contract No.",CMCSalesContract_L."No.");
       IF CMCSalesContract_L."Shipping Type" = CMCSalesContract_L."Shipping Type"::" " THEN
         CMCDelivery_L.VALIDATE("Shipping Type",CMCSalesContract_L."Shipping Type"::Customer)
       ELSE
         CMCDelivery_L.VALIDATE("Shipping Type",CMCSalesContract_L."Shipping Type"-1);
       CMCDelivery_L.VALIDATE("Transport Method",CMCSalesContract_L."Transport Method");
       CMCDelivery_L.VALIDATE("Requested Delivery Date","Ending Date");

       ForeCastQuantity_L := ForeCastQuantity_L - "Truck Capacity";

       IF ForeCastQuantity_L < 0 THEN BEGIN
        CMCDelivery_L.VALIDATE("Net Weight",ForeCastQuantity_L + "Truck Capacity");
        CMCDelivery_L."Planned Quantity" := ForeCastQuantity_L + "Truck Capacity";
       END ELSE BEGIN
       CMCDelivery_L.VALIDATE("Net Weight","Truck Capacity");
       CMCDelivery_L."Planned Quantity" := "Truck Capacity";
       END;
       CMCDelivery_L.Forecast := TRUE;
       CMCDelivery_L.MODIFY;
      END;

      MESSAGE(STRSUBSTNO(Text8053002,CMCContractSchedDeliver_P."Starting Date",CMCContractSchedDeliver_P."Ending Date"));
    END;
