    PROCEDURE SelectMixVersion@60000();

      MixBOMVersion_L@60000 : Record 60027;

      SalesLine_L@60001 : Record 37;
      MixSalesLine_L@60005 : Record 60032;
      Item_L@60002 : Record 27;
      MixBOMVersionList_L@60003 : Page 60035;

      // The objective of this function is to allow select a version of the mix for the item no.
      GetKitLine;
      TestStatusOpen;
      IF NOT QtyShippedCheckSuspend THEN
        KitLine.TESTFIELD("Quantity Shipped",0);
      TestStatusOpen;
      MixManagement.RollUpPrice(Rec,1);

      IF ("Extended Quantity" <> 0) AND ItemExists(xRec."No.") THEN
        ReserveMixSalesLine.VerifyChange(Rec,xRec);

      CLEAR(MixSalesLine_L);
      MixSalesLine_L.SETRANGE("Document Type","Document Type");
      MixSalesLine_L.SETRANGE("Document No.","Document No.");
      MixSalesLine_L.SETRANGE("Document Line No.","Document Line No.");
      IF MixSalesLine_L.FINDFIRST THEN
        IF NOT CONFIRM(Text60005,FALSE) THEN
          EXIT;


      Item_L.GET(KitLine."No.");
      CLEAR(MixBOMVersion_L);
      MixBOMVersion_L.FILTERGROUP := 2;
      MixBOMVersion_L.SETRANGE("Mix BOM No.",Item_L."Mix BOM No.");
      CLEAR(MixBOMVersionList_L);
      MixBOMVersionList_L.LOOKUPMODE := TRUE;
      MixBOMVersionList_L.SETTABLEVIEW(MixBOMVersion_L);
      IF MixBOMVersionList_L.RUNMODAL = ACTION::LookupOK THEN BEGIN
        MixBOMVersionList_L.GETRECORD(MixBOMVersion_L);

        CLEAR(MixBOMLine_L);
        MixBOMLine_L.SETRANGE("Mix BOM No.",Item_L."Mix BOM No.");
        MixBOMLine_L.SETRANGE("Version Code",MixBOMVersion_L."Version Code");
        IF NOT MixBOMLine_L.FINDFIRST THEN
          ERROR(Text60006);

        CLEAR(MixSalesLine_L);
        MixSalesLine_L.SETRANGE("Document Type","Document Type");
        MixSalesLine_L.SETRANGE("Document No.","Document No.");
        MixSalesLine_L.SETRANGE("Document Line No.","Document Line No.");
        MixSalesLine_L.DELETEALL(TRUE);

        KitLine.FillMixLines(MixBOMVersion_L."Version Code");
        // --> 20151125-APP28-TER01.01.00-2500-JCD01
        KitLine.MODIFY;
        // <-- 20151125-APP28-TER01.01.00-2500-JCD01
      END;
    END;
