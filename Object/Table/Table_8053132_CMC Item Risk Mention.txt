    PROCEDURE UpdateItemRiskPhrase@8053000();
    VAR
      RecItem_L@8053000 : Record 27;
      RecItemRiskMention_L@8053001 : Record 8053132;
      cduMCDataMgt_L@8053003 : Codeunit 8053306;

      // Appel'e dans le trigger OnClosePage de la page 8053257
      //---------------------------------------------------------------------------------------
      // Entr'es:
      //  -
      // Sorties:

      IF cduMCDataMgt_L.IsMCCompany(DATABASE::Item) THEN
        EXIT;
      IF RecItem_L.GET("Item No.") THEN BEGIN
        RecItem_L."Risque Phrase" := '';
        RecItemRiskMention_L.SETRANGE("Item No.","Item No.");

          REPEAT
            RecItem_L."Risque Phrase" += RecItemRiskMention_L.Code + '-';
          UNTIL RecItemRiskMention_L.NEXT = 0;
          RecItem_L."Risque Phrase" := COPYSTR(RecItem_L."Risque Phrase",1,STRLEN(RecItem_L."Risque Phrase") - 1);
        END;
        RecItem_L.UpdateCMR;
        RecItem_L.MODIFY(TRUE);
      END;
    END;
