            //"Calculated Reference Price" := (TotalAmount_L / quantity) - (Comission + Negociation + Compensation) -
        END;
        "Contract Type"::"Firm Purch. Price": BEGIN

    PROCEDURE CalcOpenPosition@1000000002();
    VAR
      CMCMatifContract_L@1000000001 : Record 8053065;
      FutureQty_L@1000000000 : Decimal;
    BEGIN
      // 20150422-CER121-TER01.01.00-341-RMZ01
      // This function will calc the open position for the contract
      // Input
      // Output
      IF NOT ("Contract Type" IN ["Contract Type"::"Sales Bonus","Contract Type"::"Purch. Bonus"]) THEN BEGIN
        "Open Position" := 0;
        EXIT;
      END;

      G_CMCSetup.GET;
      // --> 20150506-CER121-TER01.01.00-341-RMZ01
      // Before
      //"Open Position" := Quantity / G_CMCSetup."MATIF Weight per Lot";
      IF "Lot Number" <> 0 THEN
      // After
        "Open Position" := "Lot Number"
      ELSE
        "Open Position" := 0;
      // <-- 20150506-CER121-TER01.01.00-341-RMZ01
      FutureQty_L := 0;

      CLEAR(CMCMatifContract_L);
      CMCMatifContract_L.SETCURRENTKEY("Sales Contract Link",Type);
      CMCMatifContract_L.SETRANGE("Sales Contract Link","No.");
      CMCMatifContract_L.SETRANGE(Type,CMCMatifContract_L.Type::Future);

      CASE "Contract Type" OF
        "Contract Type"::"Sales Bonus": BEGIN
          CMCMatifContract_L.SETRANGE(Way,CMCMatifContract_L.Way::Sales);
          IF CMCMatifContract_L.FINDFIRST THEN
            REPEAT
              FutureQty_L := FutureQty_L + (CMCMatifContract_L.Quantity / G_CMCSetup."MATIF Weight per Lot");
            UNTIL CMCMatifContract_L.NEXT = 0;
          "Open Position" := "Open Position" - FutureQty_L;
        END;
        "Contract Type"::"Purch. Bonus": BEGIN
          CMCMatifContract_L.SETRANGE(Way,CMCMatifContract_L.Way::Purchase);
          IF CMCMatifContract_L.FINDFIRST THEN
            REPEAT
              FutureQty_L := FutureQty_L + (CMCMatifContract_L.Quantity / G_CMCSetup."MATIF Weight per Lot");
            UNTIL CMCMatifContract_L.NEXT = 0;
          "Open Position" := -("Open Position" - FutureQty_L);
        END;
      END;
    END;
