    PROCEDURE GetAllVariantAvailableQty@60002(ItemNo_P@1000000002 : Code[20];LocationCode_P@1000000001 : Code[10]) : Decimal;
    VAR
      RecItem_L@60002 : Record 27;
      RecWhseShipmentLine_L@60001 : Record 7321;
      ReturnQty_L@60000 : Decimal;
      RecKitSalesLine_L@60005 : Record 25000;
      KitQuantity_L@60004 : Decimal;
      QtyToShip_L@60003 : Decimal;
    BEGIN
      ReturnQty_L := 0;

      RecItem_L.RESET;
      RecItem_L.GET(ItemNo_P);
      RecItem_L.SETRANGE("Location Filter", LocationCode_P);
      RecItem_L.CALCFIELDS(Inventory);

      RecWhseShipmentLine_L.RESET;
      RecWhseShipmentLine_L.SETCURRENTKEY("Item No.","Location Code","Variant Code");
      RecWhseShipmentLine_L.SETRANGE("Item No.", ItemNo_P);
      RecWhseShipmentLine_L.SETRANGE("Location Code", LocationCode_P);
      RecWhseShipmentLine_L.SETRANGE("Variant Code");
      RecWhseShipmentLine_L.CALCSUMS("Qty. to Ship");
      QtyToShip_L := RecWhseShipmentLine_L."Qty. to Ship";

      RecWhseShipmentLine_L.RESET;
      RecKitSalesLine_L.RESET;
      RecKitSalesLine_L.SETCURRENTKEY("Document Type",Type,"No.","Variant Code","Location Code","Shipment Date");
      RecKitSalesLine_L.SETRANGE(Type, RecKitSalesLine_L.Type::Item);
      RecKitSalesLine_L.SETRANGE("No.", ItemNo_P);
      RecKitSalesLine_L.SETRANGE("Variant Code");
      RecKitSalesLine_L.SETRANGE("Location Code", LocationCode_P);
      IF RecKitSalesLine_L.FINDSET THEN REPEAT
        RecWhseShipmentLine_L.SETRANGE("Source Document" ,RecKitSalesLine_L."Document Type");
        RecWhseShipmentLine_L.SETRANGE("Source No." ,RecKitSalesLine_L."Document No.");
        RecWhseShipmentLine_L.SETRANGE("Source Line No.",RecKitSalesLine_L."Document Line No.");
        IF RecWhseShipmentLine_L.FINDSET THEN
          REPEAT
            KitQuantity_L += RecWhseShipmentLine_L."Qty. to Ship" * RecKitSalesLine_L."Quantity per";
          UNTIL RecWhseShipmentLine_L.NEXT = 0;
      UNTIL RecKitSalesLine_L.NEXT = 0;

      ReturnQty_L:= (RecItem_L.Inventory - QtyToShip_L -KitQuantity_L);

      EXIT(ReturnQty_L);
    END;
