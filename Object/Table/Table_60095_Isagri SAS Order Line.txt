    PROCEDURE TranslateItemVariant@1000000000(ItemVariantIsagri_P@1000000000 : Code[31]);
    VAR
      Position_L@1000000001 : Integer;
    BEGIN
      //fonction utilis'e pour determiner article et eventuellement variante dans Isagri
      //sachant que Isagri nous envoi ou "code article" ou "code article/code variante"
      //si "/" d'tecter on doit d'couper
      //Position := STRPOS(String, SubString)
      //
      Position_L := STRPOS(ItemVariantIsagri_P, '/');
      IF Position_L = 0 THEN BEGIN
        VALIDATE("Item No.",ItemVariantIsagri_P)
      END
      ELSE
        BEGIN
          IF Position_L >= 2 THEN
            VALIDATE("Item No.",COPYSTR(ItemVariantIsagri_P,1,Position_L - 1))
          ELSE
            ERROR(Text60000,ItemVariantIsagri_P);
        END;
    END;
