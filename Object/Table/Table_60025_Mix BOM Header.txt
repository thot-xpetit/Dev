    LOCAL PROCEDURE TestUnitsOfMeasure@1020000();
    VAR
      CMCMCItem1_L@1020000 : Record 8053079;
      CMCMCItem2_L@1020001 : Record 8053079;
    BEGIN
      {BEFORE
      CMCMCItem1_L.SETCURRENTKEY("Production BOM No.");
      CMCMCItem1_L.SETRANGE("Production BOM No.","No.");
      AFTER}
      CMCMCItem1_L.SETFILTER("Base Unit of Measure",'<>%1',"Unit of Measure Code");
      CMCMCItem2_L.COPY(CMCMCItem1_L);
      CMCMCItem1_L.SETRANGE("Kit BOM No.","No.");
      CMCMCItem2_L.SETRANGE("Kit Disassembly BOM No.","No.");
      IF CMCMCItem1_L.FINDFIRST THEN
        CMCMCItem1_L.TESTFIELD("Base Unit of Measure","Unit of Measure Code");
      IF CMCMCItem2_L.FINDFIRST THEN
        CMCMCItem2_L.TESTFIELD("Base Unit of Measure","Unit of Measure Code");
    END;
