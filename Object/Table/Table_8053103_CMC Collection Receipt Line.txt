      Result_L := FALSE;

            IF "Line No." = CollectionReceiptLine_L."Line No." THEN BEGIN
              ActualLineFinded_L := TRUE;
              Quantity_L += Quantity;
              IF LineNo_L <= "Line No." THEN BEGIN
