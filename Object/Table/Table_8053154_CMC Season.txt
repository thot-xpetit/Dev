    PROCEDURE CheckDueDates@37();
    BEGIN
      IF ("Due Date Starting Date" <> 0D) AND ("Reference Date" <> 0D) AND
         ("Due Date Starting Date" > "Reference Date") THEN
        ERROR(Text001_G,FIELDCAPTION("Reference Date"),FIELDCAPTION("Due Date Starting Date"));
      IF ("Due Date Starting Date" <> 0D) AND ("Due Date Ending Date" <> 0D) AND
         ("Due Date Starting Date" > "Due Date Ending Date") THEN
        ERROR(Text001_G,FIELDCAPTION("Due Date Ending Date"),FIELDCAPTION("Due Date Starting Date"));
      IF ("Reference Date" <> 0D) AND ("Due Date Ending Date" <> 0D) AND
         ("Reference Date" > "Due Date Ending Date") THEN
        ERROR(Text001_G,FIELDCAPTION("Reference Date"),FIELDCAPTION("Due Date Ending Date"));
    END;
