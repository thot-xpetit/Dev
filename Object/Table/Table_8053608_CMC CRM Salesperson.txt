    PROCEDURE UpdateUserName@8053600();
    VAR
      recUserLogin_L@8053601 : Record 2000000002;
      recUserWindows_L@8053600 : Record 2000000054;
      Finded_L@8053602 : Boolean;
      i_L@8053603 : Integer;

      //----------------------------------------------------------------------------------------------------------------------------------
      // Met . jour le nom de l'utilisateur
      //----------------------------------------------------------------------------------------------------------------------------------
      //  Entr'es
      //  Sorties
      //----------------------------------------------------------------------------------------------------------------------------------
      IF recUserLogin_L.GET("User ID") THEN
        "User Name" := recUserLogin_L.Name
      ELSE BEGIN
        Finded_L := FALSE;
        recUserWindows_L.RESET;
        IF recUserWindows_L.FINDSET(FALSE, FALSE) THEN
        REPEAT
          recUserWindows_L.CALCFIELDS(ID);
          IF STRPOS(UPPERCASE(recUserWindows_L.ID), UPPERCASE("User ID")) =
                                                    (STRLEN(recUserWindows_L.ID) - STRLEN("User ID")) + 1 THEN BEGIN
            recUserWindows_L.CALCFIELDS(Name);
            "User Name" := COPYSTR(recUserWindows_L.Name, 1, MAXSTRLEN("User Name"));
            Finded_L := TRUE;
          END;
        UNTIL (recUserWindows_L.NEXT = 0) OR (Finded_L = TRUE);
      END;
    END;
