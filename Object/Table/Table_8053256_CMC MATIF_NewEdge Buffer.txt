    BEGIN
      RESET;
      DELETEALL;

      LedgerMATIFBuffer_L.RESET;
      LedgerMATIFBuffer_L.SETCURRENTKEY("User ID");
      LedgerMATIFBuffer_L.SETRANGE("User ID",USERID);

      IF ReferencyMarket_P <> '' THEN
        MATIFSpecies_L.SETRANGE("Species Code",ReferencyMarket_P);

      Campaign_L.RESET;
      Campaign_L.SETRANGE("No.",CampaignNo_P);

      GLAcc_L.RESET;
      GLAcc_L.SETFILTER("Date Filter",'%1..%2',StartingDate_P,EndingDate_P);

      MATIFAcc_L.RESET;
      IF CompteNo_P <> '' THEN
        MATIFAcc_L.SETRANGE("Account No.",CompteNo_P);

      MATIFSpeciesLink_L.RESET;
      MATIFSpeciesLink_L.SETCURRENTKEY("Species Code");

      IF MATIFSpecies_L.FINDSET THEN REPEAT
        SubTotalInsertion_L := TRUE;
        NoOfEntries_L := 0;
        MATIFSpeciesLink_L.SETRANGE("Species Code",MATIFSpecies_L."Species Code");
        IF MATIFSpeciesLink_L.FINDSET THEN BEGIN
          // Insertion des 'ch'ances MATIF
          REPEAT
            MATIFTermsBuffer_L.DELETEALL;
            // Contrat MATIF Future
            MATIFContract_L.RESET;
            MATIFContract_L.SETCURRENTKEY("Sales Contract Link");
            MATIFContract_L.SETRANGE("Referency Market",MATIFSpecies_L."Species Code");
            IF Campaign_L."No." <> '' THEN
              MATIFContract_L.SETRANGE("Campaign No.",Campaign_L."No.");
            MATIFContract_L.SETRANGE(Status,MATIFContract_L.Status::Locked);
            MATIFContract_L.SETRANGE("Sign-Off Date",StartingDate_P,EndingDate_P);
            IF MATIFContract_L.FINDSET THEN REPEAT
              MATIFTerms_L.GET(MATIFContract_L."MATIF Terms Code");
              IF NOT MATIFTermsBuffer_L.GET(MATIFContract_L."MATIF Terms Code") THEN BEGIN
                MATIFTermsBuffer_L.INIT;
                MATIFTermsBuffer_L.TRANSFERFIELDS(MATIFTerms_L);
                MATIFTermsBuffer_L.INSERT;
              END;
            UNTIL MATIFContract_L.NEXT = 0;
            IF MATIFTerms_L.GET(MATIFSpeciesLink_L."MATIF Terms") THEN BEGIN
              IF MATIFAcc_L.FINDSET THEN BEGIN
                REPEAT
                  // Insertion des 'ch'ances MATIF
                  TRANSFERFIELDS(MATIFTerms_L);
                  "MATIF Species" := MATIFSpeciesLink_L."Species Code";
                  "Campain No." := Campaign_L."No.";
                  "MATIF Account" := MATIFAcc_L."Account No.";
                  EntryNo_L += 1;
                  "Entry No." := EntryNo_L;
                  "User ID Filter" := USERID;
                  "G/L Account No." := '';
                  "G/L Account Balance" := 0;
                  "Quotation Value" := GetMATIFQuotation(Rec);
                  Valorization := 0;
                  INSERT;
                  NoOfEntries_L += 1;

                  // Insertion du d'tail des 'critures
                  InsertLedgerMATIF(Rec,StartingDate_P,EndingDate_P);

                  // Mise . jour de la valorisation si par quantit'
                  CALCFIELDS("Value (Qty)","Value (Amt)");
                  IF "Value (Qty)" <> 0 THEN BEGIN
                    "Calculated Balance" := "Value (Amt)";
                    SubTotalBalance_L += "Calculated Balance";
                    MODIFY;
                  END ELSE
                    DELETE;
                UNTIL MATIFAcc_L.NEXT = 0;
              END;
            END ELSE
              IF NoOfEntries_L = 0 THEN
                SubTotalInsertion_L := FALSE
              ELSE
                SubTotalInsertion_L := TRUE;
          UNTIL MATIFSpeciesLink_L.NEXT = 0;
        END;
        IF SubTotalInsertion_L THEN BEGIN
          // Insertion du sous-total espSce
          INIT;
          "MATIF Terms" := Text8053005;
          "MATIF Species" := MATIFSpeciesLink_L."Species Code";
          "Campain No." := '';
          "MATIF Account" := '';
          "Period Start":=01019999D;

          "Entry No." := EntryNo_L;
          IF GLAcc_L.GET(MATIFSpecies_L."G/L Account No.") THEN BEGIN
            GLAcc_L.SETFILTER("Date Filter",'%1..%2',StartingDate_P,EndingDate_P);
            GLAcc_L.CALCFIELDS("Net Change");
            "G/L Account No." := GLAcc_L."No.";
            "G/L Account Balance" := GLAcc_L."Net Change";
          END;
          "Quotation Value" := 0;
          CALCFIELDS("Value (Qty)");
          "Calculated Balance" := SubTotalBalance_L;
          TotalBalance_L += SubTotalBalance_L;
          SubTotalBalance_L := 0;
          IF "Value (Qty)" <> 0 THEN
            INSERT;
        END;
      UNTIL MATIFSpecies_L.NEXT = 0;
      // Insertion de la ligne de total g'n'ral
      INIT;
      "MATIF Terms" := Text8053006;
      "MATIF Species" := '';
      "Campain No." := '';
      "MATIF Account" := '';
      "Period Start":=01019999D;
      EntryNo_L += 1;
      "Entry No." := EntryNo_L;
      "Quotation Value" := 0;
      "Calculated Balance" := TotalBalance_L;
      INSERT;

      IF FINDFIRST THEN;
    END;

    PROCEDURE InsertLedgerMATIF@8053002(CMCMatifNewEdgeBuffer_P@8053000 : TEMPORARY Record 8053256;StartingDate_P@8053008 : Date;EndingDate_P@8053007 : Date);
    VAR
      MATIFContract_L@8053002 : Record 8053065;
      SalesContract_L@8053003 : Record 8053012;
      LedgerMATIFBuffer_L@8053004 : Record 8053111;
      LedgerMATIFBufferTotal_L@8053005 : Record 8053111;
      LedgerMATIFBufferSubTotal_L@8053001 : Record 8053111;
      MATIFTerms_L@8053006 : Record 8053102;
    BEGIN
      LedgerMATIFBuffer_L.INIT;
      LedgerMATIFBufferTotal_L.INIT;
      LedgerMATIFBufferSubTotal_L.INIT;

      MATIFTerms_L.GET(CMCMatifNewEdgeBuffer_P."MATIF Terms");

      // Cr'ation de la ligne de totalisation
      // --> 20161220-WI00-ADV01.01.00-3186-JCD01
      {BEFORE
      IF NOT LedgerMATIFBufferTotal_L.GET(Text8053006,'',USERID,'','','') THEN BEGIN
      AFTER}
      IF NOT LedgerMATIFBufferTotal_L.GET(Text8053006,'',USERID,'','','',0) THEN BEGIN
      // <-- 20161220-WI00-ADV01.01.00-3186-JCD01
        LedgerMATIFBufferTotal_L."MATIF Terms" := Text8053006;
        LedgerMATIFBufferTotal_L."Contract No." := '';
        LedgerMATIFBufferTotal_L."User ID" := USERID;
        LedgerMATIFBufferTotal_L.INSERT;
      END;

      // Cr'ation de la ligne de sous-totalisation
      // --> 20161220-WI00-ADV01.01.00-3186-JCD01
      {BEFORE
      IF NOT LedgerMATIFBufferSubTotal_L.GET(Text8053005,'',USERID,CMCMatifNewEdgeBuffer_P."MATIF Species",'','') THEN BEGIN
      AFTER}
      IF NOT LedgerMATIFBufferSubTotal_L.GET(Text8053005,'',USERID,CMCMatifNewEdgeBuffer_P."MATIF Species",'','',0) THEN BEGIN
      // <-- 20161220-WI00-ADV01.01.00-3186-JCD01
        LedgerMATIFBufferSubTotal_L."MATIF Terms" := Text8053005;
        LedgerMATIFBufferSubTotal_L."MATIF Species" := CMCMatifNewEdgeBuffer_P."MATIF Species";
        LedgerMATIFBufferSubTotal_L."Contract No." := '';
        LedgerMATIFBufferSubTotal_L."User ID" := USERID;
        LedgerMATIFBufferSubTotal_L.INSERT;
      END;

      // Contrat MATIF Future
      MATIFContract_L.RESET;
      MATIFContract_L.SETCURRENTKEY("Sales Contract Link");
      MATIFContract_L.SETRANGE("Referency Market",CMCMatifNewEdgeBuffer_P."MATIF Species");
      MATIFContract_L.SETRANGE("MATIF Terms Code",CMCMatifNewEdgeBuffer_P."MATIF Terms");
      IF CMCMatifNewEdgeBuffer_P."Campain No." <> '' THEN
        MATIFContract_L.SETRANGE("Campaign No.",CMCMatifNewEdgeBuffer_P."Campain No.");
      MATIFContract_L.SETRANGE("Account No.",CMCMatifNewEdgeBuffer_P."MATIF Account");
      MATIFContract_L.SETRANGE(Status,MATIFContract_L.Status::Locked);
      MATIFContract_L.SETRANGE("Sign-Off Date",StartingDate_P,EndingDate_P);
      IF MATIFContract_L.FINDSET( FALSE , FALSE ) THEN BEGIN
        LedgerMATIFBuffer_L.INIT;
        LedgerMATIFBuffer_L."MATIF Terms" := CMCMatifNewEdgeBuffer_P."MATIF Terms";
        LedgerMATIFBuffer_L."Contract No." := '';
        LedgerMATIFBuffer_L."MATIF Species" := CMCMatifNewEdgeBuffer_P."MATIF Species";
        LedgerMATIFBuffer_L."Campaign No." := CMCMatifNewEdgeBuffer_P."Campain No.";
        LedgerMATIFBuffer_L."MATIF Account" := CMCMatifNewEdgeBuffer_P."MATIF Account";
        LedgerMATIFBuffer_L."User ID" := USERID;
        LedgerMATIFBuffer_L."Starting Date" := MATIFTerms_L."Period Start";
        // --> 20161011-WI00-ADV01.01.00-3186-NED01
        LedgerMATIFBuffer_L."Campaign No." := MATIFContract_L."Campaign No.";
        LedgerMATIFBuffer_L."MATIF Species" := CMCMatifNewEdgeBuffer_P."MATIF Species";
        // <-- 20161011-WI00-ADV01.01.00-3186-NED01
        LedgerMATIFBuffer_L.INSERT;
        REPEAT
          CASE MATIFContract_L.Way OF
            MATIFContract_L.Way::Sales: BEGIN
              LedgerMATIFBuffer_L."Value (Amt)" += (MATIFContract_L.Price - CMCMatifNewEdgeBuffer_P."Quotation Value") *
                                                   MATIFContract_L."Lot Number" * 50;
              LedgerMATIFBuffer_L."Value (Qty)" += MATIFContract_L."Lot Number";
            END;

            MATIFContract_L.Way::Purchase: BEGIN
              LedgerMATIFBuffer_L."Value (Amt)" -= (MATIFContract_L.Price - CMCMatifNewEdgeBuffer_P."Quotation Value") *
                                                   MATIFContract_L."Lot Number" * 50;
              LedgerMATIFBuffer_L."Value (Qty)" -= MATIFContract_L."Lot Number";
            END;

            MATIFContract_L.Way::AA: BEGIN
              IF SalesContract_L.GET(MATIFContract_L."Sales Contract Link") THEN BEGIN
                CASE SalesContract_L."Contract Type" OF
                  SalesContract_L."Contract Type"::"Purch. Bonus": BEGIN
                    LedgerMATIFBuffer_L."Value (Amt)" += (MATIFContract_L.Price - CMCMatifNewEdgeBuffer_P."Quotation Value") *
                                                         MATIFContract_L."Lot Number" * 50;
                    LedgerMATIFBuffer_L."Value (Qty)" += MATIFContract_L."Lot Number";
                  END;

                  SalesContract_L."Contract Type"::"Sales Bonus": BEGIN
                    LedgerMATIFBuffer_L."Value (Amt)" -= (MATIFContract_L.Price - CMCMatifNewEdgeBuffer_P."Quotation Value") *
                                                          MATIFContract_L."Lot Number" * 50;
                    LedgerMATIFBuffer_L."Value (Qty)" -= MATIFContract_L."Lot Number";
                  END;
                END;
              END;
            END;
          END;
          LedgerMATIFBuffer_L.MODIFY;
        UNTIL MATIFContract_L.NEXT = 0;
        // Mise . jour du sous-total par espSce
        LedgerMATIFBufferSubTotal_L."Value (Qty)" += LedgerMATIFBuffer_L."Value (Qty)";
        LedgerMATIFBufferSubTotal_L."Value (Amt)" += LedgerMATIFBuffer_L."Value (Amt)";
        LedgerMATIFBufferSubTotal_L.MODIFY;

        // Mise . jour du total g'n'ral
        LedgerMATIFBufferTotal_L."Value (Qty)" += LedgerMATIFBuffer_L."Value (Qty)";
        LedgerMATIFBufferTotal_L."Value (Amt)" += LedgerMATIFBuffer_L."Value (Amt)";
      END;

      // Mise . jour de la ligne de totalisation
      LedgerMATIFBufferTotal_L.MODIFY;
    END;
