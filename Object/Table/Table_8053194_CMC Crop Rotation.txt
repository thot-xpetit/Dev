      FrameContract_L@8053000 : Record 8053208;
      FrameContractLine_L@8053001 : Record 8053209;

    BEGIN
      IF Status = Status::Released THEN
        EXIT;

      CMCCropRotationLineRec_G.SETRANGE("Contact No.","Contact No.") ;
      CMCCropRotationLineRec_G.SETRANGE("Campaign No.","Campaign No.");

      FrameContract_L.RESET ;
      FrameContract_L.SETRANGE(FrameContract_L."Contact No.","Contact No.");
      FrameContract_L.SETRANGE(FrameContract_L."Crop Rotation Campaign No.","Campaign No.");
      IF  FrameContract_L.FINDFIRST() THEN BEGIN
        FrameContract_L.TESTFIELD(FrameContract_L.Status,FrameContract_L.Status::Open);
        // --> 20160516-WI00-ADV01.01.00-2925-MMI01
        IF FrameContractType_L.GET(FrameContract_L."Contract Type") THEN
          IF FrameContractType_L."Crop Rotation Mandatory" THEN BEGIN
        // <-- 20160516-WI00-ADV01.01.00-2925-MMI01
        CMCCropRotationLineRec_G.FINDSET;
        REPEAT
          IF FrameContractLine_L.GET(FrameContract_L."Contract No.",CMCCropRotationLineRec_G."Line No.") THEN BEGIN // ligne existe
                // --> 20161006-WI00-ADV01.01.00-22265-BSI01
                FrameContractLine_L.SystemCreated := ModifyAllowedFromCRM(FrameContract_L);
                // <-- 20161006-WI00-ADV01.01.00-22265-BSI01
            IF FrameContractLine_L."Area (Ha)"<CMCCropRotationLineRec_G."Area (Ha)" THEN BEGIN // quantit' augment'e
              IF FrameContractLine_L."% Rate"<>100 THEN
                FrameContractLine_L."Area (Ha)" := CMCCropRotationLineRec_G."Area (Ha)"
              ELSE
                FrameContractLine_L.VALIDATE("% Rate",100);
              FrameContractLine_L.VALIDATE(FrameContractLine_L."Area (Ha)",CMCCropRotationLineRec_G."Area (Ha)");
              FrameContractLine_L.MODIFY;
                  // --> 20161006-WI00-ADV01.01.00-22265-BSI01
                  FrameContract_L.SystemCreated := ModifyAllowedFromCRM(FrameContract_L);
                  // <-- 20161006-WI00-ADV01.01.00-22265-BSI01
              FrameContract_L.UpdateLineQty(FrameContractLine_L);
                  // --> 20170411-WI00-ADV01.01.00-26570-BSI01
                  CUSynchroCRM_L.AddJournalExport(recJrnExp_L.Type::LigneContratCadreCollect,
                  FrameContractLine_L."Contract No." +
                  CUSynchroCRM_L.GetSeparator + FORMAT(FrameContractLine_L."Line No."), TextModify, '');
                  // <-- 20170411-WI00-ADV01.01.00-26570-BSI01
            END ELSE BEGIN
              IF FrameContractLine_L."Area (Ha)">CMCCropRotationLineRec_G."Area (Ha)" THEN BEGIN // quantit' diminu'e
                FrameContractLine_L.CALCFIELDS(FrameContractLine_L."Qty. Assigned (T)");
                IF (FrameContractLine_L."Qty. Assigned (T)" = FrameContractLine_L."Quantity (T)") AND
                  (FrameContractLine_L."Quantity (T)" <> 0)
                THEN
                  ERROR(STRSUBSTNO(Text8053003,CMCCropRotationLineRec_G.Description,FrameContract_L."Contract No."))
                ELSE BEGIN
                  IF CMCCropRotationLineRec_G."Area (Ha)"=0 THEN BEGIN
                      // --> 20170411-WI00-ADV01.01.00-26570-BSI01
                      CUSynchroCRM_L.AddJournalExport(recJrnExp_L.Type::LigneContratCadreCollect,
                      FrameContractLine_L."Contract No." +
                      CUSynchroCRM_L.GetSeparator + FORMAT(FrameContractLine_L."Line No."), TextDelete, '');
                      // <-- 20170411-WI00-ADV01.01.00-26570-BSI01
                        // --> 20160520-WI00-ADV01.01.00-2980-BSI01
                        FrameContractLine_L.SuspendStatusCheck(TRUE);
                        // <-- 20160520-WI00-ADV01.01.00-2980-BSI01
                    FrameContractLine_L.DELETE(TRUE);
                  END ELSE BEGIN
                    FrameContractLine_L.VALIDATE(FrameContractLine_L."Area (Ha)",CMCCropRotationLineRec_G."Area (Ha)");
                    FrameContractLine_L.MODIFY;
                        // --> 20161006-WI00-ADV01.01.00-22265-BSI01
                        FrameContract_L.SystemCreated := ModifyAllowedFromCRM(FrameContract_L);
                        // <-- 20161006-WI00-ADV01.01.00-22265-BSI01
                    FrameContract_L.UpdateLineQty(FrameContractLine_L);
                        // --> 20170411-WI00-ADV01.01.00-26570-BSI01
                        CUSynchroCRM_L.AddJournalExport(recJrnExp_L.Type::LigneContratCadreCollect,
                        FrameContractLine_L."Contract No." +
                        CUSynchroCRM_L.GetSeparator + FORMAT(FrameContractLine_L."Line No."), TextModify, '');
                        // <-- 20170411-WI00-ADV01.01.00-26570-BSI01
                  END;
                END;
              END;
            END;
          END ELSE BEGIN // ligne . cr'er
            IF CMCCropRotationLineRec_G."Area (Ha)">0 THEN BEGIN
              Yield_L.RESET;
              Yield_L.SETRANGE("Campaign No.",CMCCropRotationLineRec_G."Campaign No.");
              Yield_L.SETRANGE("Item Funnel Code",CMCCropRotationLineRec_G.Range);
              Yield_L.SETRANGE(Yield_L."Include in Frame Contract",TRUE);
              IF Yield_L.FINDFIRST THEN BEGIN
                FrameContractLine_L.RESET;
                FrameContractLine_L.INIT;
                    // --> 20161006-WI00-ADV01.01.00-22265-BSI01
                    FrameContractLine_L.SystemCreated := ModifyAllowedFromCRM(FrameContract_L);
                    // <-- 20161006-WI00-ADV01.01.00-22265-BSI01
                FrameContractLine_L."Contract No." := FrameContract_L."Contract No.";
                FrameContractLine_L."Line No." := CMCCropRotationLineRec_G."Line No.";;
                FrameContractLine_L.Description := CMCCropRotationLineRec_G.Description;
                FrameContractLine_L.VALIDATE("Item Funnel Code",CMCCropRotationLineRec_G.Range);
                FrameContractLine_L.VALIDATE("Area (Ha)",CMCCropRotationLineRec_G."Area (Ha)");
                FrameContractLine_L.INSERT(TRUE);
                    // --> 20161006-WI00-ADV01.01.00-22265-BSI01
                    FrameContract_L.SystemCreated := ModifyAllowedFromCRM(FrameContract_L);
                    // <-- 20161006-WI00-ADV01.01.00-22265-BSI01
                FrameContract_L.UpdateLineQty(FrameContractLine_L);
                    // --> 20170411-WI00-ADV01.01.00-26570-BSI01
                    CUSynchroCRM_L.AddJournalExport(recJrnExp_L.Type::LigneContratCadreCollect,
                    FrameContractLine_L."Contract No." +
                    CUSynchroCRM_L.GetSeparator + FORMAT(FrameContractLine_L."Line No."), TextModify, '');
                    // <-- 20170411-WI00-ADV01.01.00-26570-BSI01
              END;
            END;
          END;
        UNTIL CMCCropRotationLineRec_G.NEXT=0;
         // --> 20160516-WI00-ADV01.01.00-2925-MMI01
         END;
         // <-- 20160516-WI00-ADV01.01.00-2925-MMI01
       END;

      SupFCHeader_G.RESET;
      SupFCHeader_G.SETCURRENTKEY("Contact No.","Campaign No.");
      SupFCHeader_G.SETRANGE("Contact No.","Contact No.");
      SupFCHeader_G.SETRANGE("Campaign No.","Campaign No.");

    END;
