      PreviousCustNo_L := '';
      PreviousVendNo_L := '';

      PreviousContact_L := xRec."Previous No.";
      IF ("Previous No." <> '') THEN
        PreviousContact_L := "Previous No.";

        EXIT;

        ContactBusRelation_L.RESET;
        ContactBusRelation_L.SETRANGE("MC Company Code",COMPANYNAME);
        ContactBusRelation_L.SETRANGE("Contact No.",PreviousContact_L);

          REPEAT
            CASE ContactBusRelation_L."Link to Table" OF
              ContactBusRelation_L."Link to Table"::Customer:
                PreviousCustNo_L := ContactBusRelation_L."No.";

              ContactBusRelation_L."Link to Table"::Vendor:
                PreviousVendNo_L := ContactBusRelation_L."No.";
            END;
          UNTIL ContactBusRelation_L.NEXT = 0;
        END;
      END;

      // (Re)Set Previous Customer No.
      ContactBusRelation_L.RESET;
      ContactBusRelation_L.SETRANGE("MC Company Code",COMPANYNAME);
      ContactBusRelation_L.SETRANGE("Contact No.","No.");
      ContactBusRelation_L.SETRANGE("Link to Table",ContactBusRelation_L."Link to Table"::Customer);
      IF ContactBusRelation_L.FINDFIRST THEN BEGIN
        IF Customer_L.GET(ContactBusRelation_L."No.") THEN BEGIN
          Customer_L."Previous No." := PreviousCustNo_L;
          Customer_L.MODIFY;
        END;
      END;

      // (Re)Set Previous Vendor No.
      ContactBusRelation_L.RESET;
      ContactBusRelation_L.SETRANGE("MC Company Code",COMPANYNAME);
      ContactBusRelation_L.SETRANGE("Contact No.","No.");
      ContactBusRelation_L.SETRANGE("Link to Table",ContactBusRelation_L."Link to Table"::Vendor);
      IF ContactBusRelation_L.FINDFIRST THEN BEGIN
        IF Vendor_L.GET(ContactBusRelation_L."No.") THEN BEGIN
          Vendor_L."Previous No." := PreviousVendNo_L;
          Vendor_L.MODIFY;
        END;
      END;
      // <-- 20140724-APP20-TER01.01.00-ENS01
    END;
