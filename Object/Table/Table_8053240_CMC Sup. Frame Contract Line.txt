        UNTIL Item_L.NEXT=0;

        IF ItemTMP_P.FINDFIRST THEN
          REPEAT
            FindCommonItem(ItemTMP_P,Item_L);
          UNTIL ItemTMP_P.NEXT=0;

      // --> Appliquer filtre sous-famille
      Item_L.RESET;
      ItemFunnelFilterLine_L.SETRANGE(ItemFunnelFilterLine_L."Funnel Type",ItemFunnelFilterLine_L."Funnel Type"::SubFamily);

        Item_L.SETFILTER(Item_L."Species Code",ItemFunnelFilterLine_L."Funnel Line Filter");

          UNTIL ItemTMP_P.NEXT=0;
