    PROCEDURE CheckOilseedsAnalyse@1000000000();
    VAR
      PostedDelivery_L@1000000000 : Record 8053014;
      PostedResumption_L@1000000001 : Record 8053014;
    BEGIN
      //-------------------------------------------------------------------------------------------------------------
      // Checks if analyse is related to a posted return sales associated to a posted delivery invoiced in prepayment
      //-------------------------------------------------------------------------------------------------------------
      // Entr'es:
      //     -
      // Sorties:
      //   -
      //-------------------------------------------------------------------------------------------------------------

      // --> 20160531-WI00-ADV01.01.00-2796-MMI01
      IF Type <> Type::Oilseeds THEN
        EXIT;

      CASE "Delivery Type" OF
        "Delivery Type"::Delivery:
          BEGIN
            PostedDelivery_L.GET("Delivery Type", "Delivery No.");
            IF (PostedDelivery_L."Related Posted Delivery No." <> '') AND
              (PostedDelivery_L."Posted Sales Document No" = '') AND
                (PostedDelivery_L."Posted Prepay Invoice No." <> '')
            THEN
              ERROR(Text60000, PostedDelivery_L."Related Posted Delivery No.", PostedDelivery_L."No.");
          END;
        "Delivery Type"::Resumption:
          BEGIN
            PostedResumption_L.GET("Delivery Type", "Delivery No.");
            IF PostedResumption_L."Related Posted Delivery No." <> '' THEN BEGIN
              PostedDelivery_L.GET(PostedDelivery_L."Delivery Type"::Delivery, PostedResumption_L."Related Posted Delivery No.");
              IF (PostedDelivery_L."Posted Sales Document No" = '') AND
                (PostedDelivery_L."Posted Prepay Invoice No." <> '')
              THEN
                ERROR(Text60000, PostedResumption_L."No.", PostedDelivery_L."No.");
            END;
          END;
      END;
      // <-- 20160531-WI00-ADV01.01.00-2796-MMI01
    END;
