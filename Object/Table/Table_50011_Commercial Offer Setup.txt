    PROCEDURE SetDescription@1000000001();
    VAR
      Desc@1000000000 : Text[50];
      StdText@1000000001 : Record 7;
      GLAcc@1000000002 : Record 15;
      Item@1000000003 : Record 27;
      Resource@1000000004 : Record 156;
      FixedAsset@1000000005 : Record 5600;
      ItemCharge@1000000006 : Record 5800;
    BEGIN
      Desc := '';
      CASE(Type) OF
        Type::" " : BEGIN
          IF (StdText.GET("No.")) THEN BEGIN
            Desc := StdText.Description;
          END;
        END;
        Type::"G/L Account" : BEGIN
          IF (GLAcc.GET("No.")) THEN BEGIN
            Desc := GLAcc.Name;
          END;
        END;
        Type::Item : BEGIN
          IF (Item.GET("No.")) THEN BEGIN
            Desc := Item.Description;
          END;
        END;
        Type::Resource : BEGIN
          IF (Resource.GET("No.")) THEN BEGIN
            Desc := Resource.Name;
          END;
        END;
        Type::"Fixed Asset" : BEGIN
          IF (FixedAsset.GET("No.")) THEN BEGIN
            Desc := FixedAsset.Description;
          END;
        END;
        Type::"Charge (Item)" : BEGIN
          IF (ItemCharge.GET("No.")) THEN BEGIN
            Desc := ItemCharge.Description;
          END;
        END;
      END;
      IF (Description = '') THEN
        Description := Desc;
    END;

    BEGIN
    END.
