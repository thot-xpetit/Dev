    PROCEDURE Add@8060984(MessageID_P@8060987 : GUID;ProcessusCode_P@8060986 : Code[30];TaskCode_P@8060985 : Code[30];ExecutionDescr_P@8060984 : Text[1024];LogLevel_P@8060983 : 'Information,Warning,Critical';Indentation_P@8060982 : Integer) : Integer;

      INIT;

      "Message ID" := MessageID_P;

      IF IncomingMessage_L.GET(MessageID_P) THEN BEGIN
        "Interface ID" := IncomingMessage_L."Interface ID";
        Description := IncomingMessage_L.Description;
        "Event Code" := IncomingMessage_L."Event Code";
        "Event Date" := IncomingMessage_L."Message Date";
        "Event Time" := IncomingMessage_L."Message Time";
        "Parameter 1" := IncomingMessage_L."Parameter 1";
        "Parameter 2" := IncomingMessage_L."Parameter 2";
        "Parameter 3" := IncomingMessage_L."Parameter 3";
        "Parameter 4" := IncomingMessage_L."Parameter 4";
        "Message Processing" := IncomingMessage_L.Execution;
        "User ID" := USERID;
      END;

      "Processus Code" := ProcessusCode_P;
      "Task Code" := TaskCode_P;
      "Execution Date" := TODAY;
      "Execution Time" := TIME;
      "Execution Description" := COPYSTR(ExecutionDescr_P, 1, 250);
      "Log Level" := LogLevel_P;
      Indentation := Indentation_P;

      IF "Task Code" <> '' THEN
        "Indentation Code" := "Task Code"
      ELSE IF "Processus Code" <> '' THEN
        "Indentation Code" := "Processus Code"
      ELSE
        "Indentation Code" := "Event Code";

      LOCKTABLE;
      Sequence := NewSequence(MessageID_P);
      INSERT;
      COMMIT;

      // -----------------------------------------------------------------------
      // Mail notification : uniquement Type "Avertissement" et "Critique"
      // -----------------------------------------------------------------------
      IF IncomingMessage_L.GET(MessageID_P) THEN BEGIN
        IF InterfaceID_L.GET(IncomingMessage_L."Interface ID") THEN BEGIN
          IF InterfaceID_L."Mail notification" THEN BEGIN

            CASE "Log Level" OF
              "Log Level"::Warning: BEGIN
                Subject_L := STRSUBSTNO(FORMAT(Text8060982), IncomingMessage_L."Interface ID");
              END;
              "Log Level"::Critical: BEGIN
                Subject_L := STRSUBSTNO(FORMAT(Text8060983), IncomingMessage_L."Interface ID");
              END;
            END;

            IF (((LogLevel_P = LogLevel_P::Warning)  AND (InterfaceID_L."eMail - Warning")) OR
                ((LogLevel_P = LogLevel_P::Critical) AND (InterfaceID_L."eMail - Critical"))) THEN BEGIN

               IF ISCLEAR(SMTPMail_L) THEN
                 CREATE(SMTPMail_L);

               SMTPMail_L.FromName := 'Interface ' + InterfaceID_L."Interface ID";
               SMTPMail_L.FromAddress := InterfaceID_L."SMTP - MailFrom";
               SMTPMail_L."TO" := InterfaceID_L.eMail;

               SMTPMail_L.SmtpPort := '25';
               IF InterfaceID_L."SMTP - Server port" <> 0 THEN
                 SMTPMail_L.SmtpPort := FORMAT(InterfaceID_L."SMTP - Server port");

                SMTPMail_L.Subject := Subject_L;
                SMTPMail_L.HtmlFormatted := FALSE;
                SMTPMail_L.AppendBody(STRSUBSTNO(FORMAT(Text8060984), MessageID_P));
                SMTPMail_L.AppendBody('');
                SMTPMail_L.AppendBody(ExecutionDescr_P);
                SMTPMail_L.AppendBody('');
                SMTPMail_L.AppendBody('----------------------------------------------------------------------------------');
                SMTPMail_L.AppendBody('EvSnement ......: ' +
                                      '[ ' + FORMAT(IncomingMessage_L."Message Date") + ', ' +
                                             FORMAT(IncomingMessage_L."Message Time") + ' ] - ' +
                                      IncomingMessage_L."Event Code");
                SMTPMail_L.AppendBody('  - Parametre 1 .: ' + IncomingMessage_L."Parameter 1");
                SMTPMail_L.AppendBody('  - Parametre 2 .: ' + IncomingMessage_L."Parameter 2");
                SMTPMail_L.AppendBody('  - Parametre 3 .: ' + IncomingMessage_L."Parameter 3");
                SMTPMail_L.AppendBody('  - Parametre 4 .: ' + IncomingMessage_L."Parameter 4");
                SMTPMail_L.AppendBody('Traitement ......: ' +
                                      '[ ' + FORMAT("Execution Date") + ', ' +
                                             FORMAT("Execution Time") + ' ] - ' +
                                      ProcessusCode_P + ' / ' + TaskCode_P);
                SMTPMail_L.AppendBody('----------------------------------------------------------------------------------');

                SMTPMail_L.Send(InterfaceID_L."SMTP - Server", FALSE, '', '');

            END;

          END;
        END;

      END;

      EXIT(Sequence);
    END;
