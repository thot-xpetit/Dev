      // Validates parametrization done by the user and rises errors in case of wrong parametrization.
      //--------------------------------------------------------------------------------------------------------

      // --> 20140512-App31-CMC02.02.00-2102-FEC01

      // --> 20140623-App31-CMC02.02.00-2270-FEC07
      {BEFORE

      DocumentCheck(FIELDCAPTION("Transfer Order (Outbound)"),"Transfer Order (Outbound)");
      DocumentCheck(FIELDCAPTION("Transfer Shipment"),"Transfer Shipment");
      DocumentCheck(FIELDCAPTION("Transfer Receipt"),"Transfer Receipt");

    PROCEDURE DocumentCheck@8053022(FieldName@8053000 : Text[50];FieldValue@8053001 : Boolean);
    BEGIN
      //--------------------------------------------------------------------------------------------------------
      // Rises an error if user tries to enable input document for a Sales Person, a Responsability Center,
      // a Customer or a Vendor.
      //--------------------------------------------------------------------------------------------------------
      //Input - FieldName - name of the document to be enabled for block
      //      - FieldValue - current value of the document flag


      IF FieldValue THEN BEGIN
        IF "Blocking Type" = "Blocking Type"::Salesperson THEN
          ERROR(Text002, SalesPerson_G.TABLECAPTION, FieldName);
        IF "Blocking Type" = "Blocking Type"::"Responsibility Center" THEN
          ERROR(Text002, RespCenter_G.TABLECAPTION, FieldName);
        IF "Blocking Type" = "Blocking Type"::Customer THEN
          ERROR(Text002, Customer_G.TABLECAPTION, FieldName);
        IF "Blocking Type" = "Blocking Type"::Vendor THEN
          ERROR(Text002, Vendor_G.TABLECAPTION, FieldName);
      END;
    END;
