    BEGIN

    PROCEDURE VerifyFieldsForLock@8053001();
    VAR
      L_recContrSchedDel@8053000 : Record 8053067;
    BEGIN
      TESTFIELD("Customer No.");
      TESTFIELD("Campaign No.");
      TESTFIELD("Starting Date");
      TESTFIELD("Ending Date");
      TESTFIELD("VAT Bus. Posting Group");
      TESTFIELD(Area);

      IF "Starting Date" = "Ending Date" THEN
       ERROR(Text8053000);

      TESTFIELD("Payment Method Code");
      TESTFIELD("Payment Terms Code");
      TESTFIELD("Species Code");
      TESTFIELD("Item No.");
      TESTFIELD("Committed Quantity");
      TESTFIELD("Unit of Measure Code");
      //TESTFIELD("Prime Amount");
      TESTFIELD("Shipping Type");

      IF Cadencia THEN BEGIN
        TESTFIELD("Shipping Starting Date");
        TESTFIELD("Cadencia Frequency");
      END;

      TESTFIELD("Default Location Code");

      IF "Raise Unit Amount"<>0
        THEN  TESTFIELD("Raise Starting Date");

      L_recContrSchedDel.RESET;
      L_recContrSchedDel.SETRANGE("Contract No.");
      IF Cadencia THEN BEGIN
        L_recContrSchedDel.CALCSUMS(L_recContrSchedDel.Quantity);
        IF L_recContrSchedDel.Quantity<>"Committed Quantity" THEN
          ERROR(Text8053006);
      END;
    END;
