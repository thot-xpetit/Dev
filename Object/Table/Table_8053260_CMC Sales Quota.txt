    PROCEDURE CheckNewQuotaValidity@8053011(CurrentSalesQuota_P@8053002 : Record 8053260);
    VAR
      SalesQuota_L@8053000 : Record 8053260;
      LevelType_L@8053001 : 'Company,Sales Area,Sales Zone,Salesperson';
      RealQty_L@1000000000 : Decimal;
      SalesForecastMgt_L@8053003 : Codeunit 8053200;
      DoCompanyQuotaExist_L@1000000001 : Boolean;
      DoSalesAreaQuotaExist_L@1000000002 : Boolean;
    BEGIN
      //------------------------------------------------------------------------------------------------
      // Checks if the new Quantity for the quote is not overcoming the maximum allowed quantity
      // for different levels: Sales Zone, Sales Area, Company.Also new quantity must be greater than real quantity.
      // Input:
      //   -  CurrentSalesQuota_P - the quote for which the quantity is modified
      //------------------------------------------------------------------------------------------------

      IF Quantity = 0 THEN EXIT;

      // --> 20131125-App07-CMC02.02.00-1142-BSI01
      // --> 20140325-APP07-CMC02.02.00-1889-FEC02
      {BEFORE
      RealQty_L := 0;
      RealQty_L := CalculateRealQuantity(CurrentSalesQuota_P);
      AFTER}
      RealQty_L := SalesForecastMgt_L.CalculateActualQty(Rec);
      // --> 20140325-APP07-CMC02.02.00-1889-FEC02
      IF  Quantity < RealQty_L THEN
        //-->20140114-APP07-CMC02.02.00-1479-FGS
        IF CurrFieldNo <> 0 THEN

        //<--20140114-APP07-CMC02.02.00-1479-FGS

      // <-- 20131125-App07-CMC02.02.00-1142-BSI01

      // --> 20140610-APP07-CMC02.02.00-2200-FEC01
      IF (CurrentSalesQuota_P."Sales Type" = CurrentSalesQuota_P."Sales Type"::"Sales Zone") OR
         (CurrentSalesQuota_P."Sales Type" = CurrentSalesQuota_P."Sales Type"::Salesperson) THEN BEGIN

        SalesQuota_L.RESET;
        SalesQuota_L.SETRANGE("Campaign No.",CurrentSalesQuota_P."Campaign No.");
        SalesQuota_L.SETRANGE(Type,CurrentSalesQuota_P.Type);
        SalesQuota_L.SETRANGE("No.",CurrentSalesQuota_P."No.");
        IF CurrentSalesQuota_P."Variant Code" <> '' THEN
          SalesQuota_L.SETRANGE("Variant Code", CurrentSalesQuota_P."Variant Code")
        ELSE
          SalesQuota_L.SETRANGE("Variant Code",'');
        SalesQuota_L.SETRANGE("Production Activity Code",CurrentSalesQuota_P."Production Activity Code");
        SalesQuota_L.SETRANGE("Sales Type",SalesQuota_L."Sales Type"::Company);
        //-->20150928-WI00-ADV01.01.00-2211-DGL01
        SalesQuota_L.SETFILTER("Doc. Category Code",'=%1',CurrentSalesQuota_P."Doc. Category Code");
        //<--20150928-WI00-ADV01.01.00-2211-DGL01
        SalesQuota_L.SETRANGE("Starting Date",CurrentSalesQuota_P."Starting Date");
        SalesQuota_L.SETRANGE("Ending Date",CurrentSalesQuota_P."Ending Date");
        DoCompanyQuotaExist_L := NOT SalesQuota_L.ISEMPTY;

        SalesQuota_L.SETRANGE("Sales Type",SalesQuota_L."Sales Type"::"Sales Area");
        SalesQuota_L.SETRANGE("Sales No.",CurrentSalesQuota_P."Sales Area Code");
        DoSalesAreaQuotaExist_L := NOT SalesQuota_L.ISEMPTY;

        IF (DoCompanyQuotaExist_L = FALSE) AND (DoSalesAreaQuotaExist_L = FALSE) THEN
          ERROR(Text8053010);
      END;
      // --< 20140610-APP07-CMC02.02.00-2200-FEC01

      CASE CurrentSalesQuota_P."Sales Type" OF

        "Sales Type"::Salesperson:
          BEGIN
