    PROCEDURE CalcQuantity@8053001() R_StdWeight : Decimal;
    VAR
      L_Parser@8053000 : Codeunit 8052986;
      L_RecOilseedStd@8053001 : Record 8053016;
      L_Formula@8053002 : Text[100];
      Error_L@8053004 : Text[1024];
      CMCSetup_L@8053003 : Record 8052982;
      CMCDeliveryAnalysis_L@1000000000 : Record 8053015;
    BEGIN
      // ---------CALCULATE STANDARD WEIGHT -------------------
      TESTFIELD("Species Code");
      IF "Delivery Type"<>"Delivery Type"::Transfert THEN
        IF NOT "Without Sales Contract" THEN
          TESTFIELD("Contract No.");

      R_StdWeight:="Net Weight";

      IF "Product Type"="Product Type"::Oilseed THEN BEGIN
        L_RecOilseedStd.GET("Campaign No.","Item No.");

        L_Formula :=  L_RecOilseedStd."Standard Weight Formula";
        L_Formula := EvaluateFormula(L_Formula);

        //R_StdWeight := ROUND(L_Parser.CalcDecimalExpr(L_Formula),0.0001);
        // --> 20130625-Wi00-CMC02.02.00-DRN01
        //BEFORE
        //R_StdWeight := ROUND(L_Parser.CalcDecimalExpr(L_Formula),0.01,'<');
        CMCSetup_L.GET;
        CMCSetup_L.TESTFIELD("Out. Std Weight Rounding Prec");
        CASE CMCSetup_L."Out. Std Weight Rounding Type" OF
          CMCSetup_L."Out. Std Weight Rounding Type"::Nearest :
            R_StdWeight := ROUND(L_Parser.CalcDecimalExpr(L_Formula),CMCSetup_L."Out. Std Weight Rounding Prec",'=');
          CMCSetup_L."Out. Std Weight Rounding Type"::Up :
            R_StdWeight := ROUND(L_Parser.CalcDecimalExpr(L_Formula),CMCSetup_L."Out. Std Weight Rounding Prec",'>');
          CMCSetup_L."Out. Std Weight Rounding Type"::Down :
            R_StdWeight := ROUND(L_Parser.CalcDecimalExpr(L_Formula),CMCSetup_L."Out. Std Weight Rounding Prec",'<');
        END;
        //BEFORE
        //CMCSetup_L.GET;
        // <-- 20130625-Wi00-CMC02.02.00-DRN01

        // --> 20151016-Wi00-ADV01.01.00-2162-SDE01
        "Recalc. Standard Weight" :=0;
        // <-- 20151016-Wi00-ADV01.01.00-2162-SED01
        IF CMCSetup_L."Oilseed Inventory No Effect" = TRUE THEN
        BEGIN
          IF "Standard Weight" <> R_StdWeight THEN BEGIN
            "Recalc. Standard Weight" := R_StdWeight;
            R_StdWeight := "Standard Weight";
          END;
        END;
      END
      // --> 20150626-WI00-TER02.02.00-475-SDE01
      ELSE BEGIN
        IF "Weight Standards Already" = FALSE THEN BEGIN
          CMCDeliveryAnalysis_L.RESET;
          CMCDeliveryAnalysis_L.SETRANGE("Delivery Type","Delivery Type");
          CMCDeliveryAnalysis_L.SETRANGE("Delivery No.","No.");
          IF CMCDeliveryAnalysis_L.FINDSET THEN
            REPEAT
              R_StdWeight:= CalcStandardWeight(CMCDeliveryAnalysis_L);
            UNTIL CMCDeliveryAnalysis_L.NEXT = 0;
        END ELSE
          // --> 20150731-GEN15/001-ADV01.01.00-1834-DRN01
          IF ("IC Direction" = "IC Direction"::Incoming) AND "Shared Silo" AND ("Source Standard Weight" <> 0) THEN
            R_StdWeight:= "Source Standard Weight"
          ELSE
          // <-- 20150731-GEN15/001-ADV01.01.00-1834-DRN01
          R_StdWeight:= "Net Weight";
      // <-- 20150626-WI00-TER02.02.00-475-SDE01

      END;

      EXIT(R_StdWeight);
    END;
