    PROCEDURE GetItemNo@60004(ItemID_P@60000 : Code[20]) ItemNo_Res : Code[20];
    VAR
      RecItem_L@60002 : Record 27;
      RecItemCrossRef_L@60001 : Record 5717;
    BEGIN
      ItemNo_Res := '';
      CLEAR(RecItemCrossRef_L);
      RecItemCrossRef_L.SETRANGE("Cross-Reference Type",RecItemCrossRef_L."Cross-Reference Type" :: "Bar Code");
      RecItemCrossRef_L.SETRANGE("Cross-Reference No.",ItemID_P);
      IF RecItemCrossRef_L.FINDFIRST THEN
        ItemNo_Res := RecItemCrossRef_L."Item No."
      ELSE BEGIN
        IF RecItem_L.GET(ItemID_P) THEN
          ItemNo_Res := RecItem_L."No."
        ELSE BEGIN
          IF STRPOS(ItemID_P,'#') - 1 > 1 THEN
            ItemNo_Res := COPYSTR(ItemID_P,1,STRPOS(ItemID_P,'#') - 1);
        END;
      END;
    END;
