    PROCEDURE CalcShippedSaleNotReturned@3(VAR ShippedQtyNotReturned@1003 : Decimal;VAR RevUnitCostLCY@1005 : Decimal;ExactCostReverse@1006 : Boolean);
    VAR
      ItemLedgEntry@1000 : Record 32;
      TotalCostLCY@1007 : Decimal;
      TotalQtyBase@1002 : Decimal;
    BEGIN
      ShippedQtyNotReturned := 0;
      IF (Type <> Type::Item) OR (Quantity <= 0) THEN BEGIN
        RevUnitCostLCY := "Unit Cost (LCY)";
        EXIT;
      END;

      RevUnitCostLCY := 0;
      FilterPstdDocLnItemLedgEntries(ItemLedgEntry);
      // --> 20160707-WI00-ADV01.01.00-3131-JCD01
      ItemLedgEntry.SETRANGE("Entry Type",ItemLedgEntry."Entry Type"::Sale);
      // <-- 20160707-WI00-ADV01.01.00-3131-JCD01

      IF ItemLedgEntry.FINDSET THEN
        REPEAT
          ShippedQtyNotReturned := ShippedQtyNotReturned - ItemLedgEntry."Shipped Qty. Not Returned";
          IF ExactCostReverse THEN BEGIN
            ItemLedgEntry.CALCFIELDS("Cost Amount (Expected)", "Cost Amount (Actual)");
            TotalCostLCY := TotalCostLCY + ItemLedgEntry."Cost Amount (Expected)" + ItemLedgEntry."Cost Amount (Actual)";
            TotalQtyBase := TotalQtyBase + ItemLedgEntry.Quantity;
          END;
        UNTIL ItemLedgEntry.NEXT = 0;

      IF ExactCostReverse AND (ShippedQtyNotReturned <> 0) AND (TotalQtyBase <> 0) THEN
        RevUnitCostLCY := ABS(TotalCostLCY / TotalQtyBase) * "Qty. per Unit of Measure"
      ELSE
        RevUnitCostLCY := "Unit Cost (LCY)";

      ShippedQtyNotReturned := CalcQty(ShippedQtyNotReturned);
    END;

    END;

    PROCEDURE FindShipmentLine@8053002(recSalesLine_P@8053001 : Record 37;VAR recSalesShipmentVar_P@8053000 : Record 111);
    BEGIN
      //----------------------------------------------------------------------------------------------------------------------------------
      // Recherche des lignes de livraison d'une ligne de commande de vente
      // Pour r'cup'rer le r'sultat, il faut faire un FINDSET sur recSalesShipmentVar_P
      //----------------------------------------------------------------------------------------------------------------------------------
      //  Entr'es
      //    - recSalesLine_P : ligne de commande de vente a rechercher les livraisons
      //    - recSalesShipmentVar_P : ligne de livraison en retour
      //  Sorties
      //----------------------------------------------------------------------------------------------------------------------------------
      // --> 20132810-App22-CMC02.02.00-1010-FHU01
      //1) recherche si la ligne actuelle a 1 ou plusieurs livraison . facturer
      //2) s'il n'y a qu'une ligne, on prend toutes les lignes de livraison des lignes associ'es
      //3) sinon, on ne prend que la ligne rattach'e dans la livraison

      //Etape 1 :
      recSalesShipmentVar_P.RESET;
      recSalesShipmentVar_P.SETCURRENTKEY("Order No.", "Order Line No.");
      recSalesShipmentVar_P.SETRANGE("Order No.", "Document No.");
      recSalesShipmentVar_P.SETRANGE("Order Line No.", "Line No.");
      recSalesShipmentVar_P.SETFILTER("Qty. Shipped Not Invoiced", '<>0');
      IF recSalesShipmentVar_P.COUNT > 1 THEN BEGIN
        //Etape 2 :
        recSalesShipmentVar_P.RESET;
        recSalesShipmentVar_P.SETCURRENTKEY("Order No.", "Order Line No.");
        recSalesShipmentVar_P.SETRANGE("Order No.", recSalesLine_P."Document No.");
        recSalesShipmentVar_P.SETRANGE("Order Line No.", recSalesLine_P."Line No.");
        recSalesShipmentVar_P.SETFILTER("Qty. Shipped Not Invoiced", '<>0');
      END ELSE BEGIN
        //Etape 3 :
        recSalesShipmentVar_P.RESET;
        recSalesShipmentVar_P.SETCURRENTKEY("Order No.", "Order Line No.", "Document No.");
        recSalesShipmentVar_P.SETRANGE("Order No.", recSalesLine_P."Document No.");
        recSalesShipmentVar_P.SETRANGE("Order Line No.", recSalesLine_P."Line No.");
        recSalesShipmentVar_P.SETRANGE("Document No.", "Document No.");
        recSalesShipmentVar_P.SETFILTER("Qty. Shipped Not Invoiced", '<>0');
        recSalesShipmentVar_P.SETRANGE("Attached to Line No.", "Line No.");
        IF recSalesShipmentVar_P.ISEMPTY THEN BEGIN
          recSalesShipmentVar_P.SETRANGE("Attached to Line No.", 0);
        END;
      END;
      // <-- 20132810-App22-CMC02.02.00-1010-FHU01
    END;
