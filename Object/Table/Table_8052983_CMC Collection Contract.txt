    PROCEDURE isCheckAnalysis@8053008(P_ReceiptNo@8053000 : Code[20];P_ContractNo@8053001 : Code[10]) RES_isAnalysisOk : Boolean;
    VAR
      L_recReceipt@8053002 : Record 8052993;
      L_recReceiptAnalysis@8053003 : Record 8052994;
      L_recContractQuality@8053004 : Record 8052984;
      L_recQualityStandards@8053005 : Record 8052992;
      L_recCollectionContract@8053006 : Record 8052983;
      L_AnalysisErr@8053007 : Boolean;
    BEGIN
      RES_isAnalysisOk := TRUE;

      L_AnalysisErr := TRUE;

      G_CMCSetup.GET;

      L_recCollectionContract.GET(P_ContractNo);

      IF ((P_ReceiptNo <> '') AND (G_CMCSetup."Automatic Downgrading")) THEN BEGIN
        IF L_recReceipt.GET(L_recReceipt."Receipt Type"::Receipt,P_ReceiptNo) THEN BEGIN
          L_recReceiptAnalysis.SETCURRENTKEY("Receipt Type","Receipt No.", "Analysis Code");
          L_recReceiptAnalysis.SETRANGE("Receipt Type",L_recReceipt."Receipt Type");
          L_recReceiptAnalysis.SETRANGE("Receipt No.", P_ReceiptNo);
          IF L_recReceiptAnalysis.FINDSET(FALSE, FALSE) THEN BEGIN
            REPEAT

              IF L_recContractQuality.GET(P_ContractNo, L_recReceiptAnalysis."Analysis Code") THEN BEGIN
                L_AnalysisErr := G_CMCManagement.CheckContractQuality(L_recReceiptAnalysis.Value, L_recContractQuality);
              END ELSE
                IF L_recQualityStandards.GET(L_recCollectionContract."Campaign No.", L_recCollectionContract."Item No.",
                  L_recReceiptAnalysis."Analysis Code")
                THEN
                  L_AnalysisErr := G_CMCManagement.CheckSpeciesQuality(L_recReceiptAnalysis.Value, L_recQualityStandards);
            UNTIL ((NOT L_AnalysisErr) OR (L_recReceiptAnalysis.NEXT = 0));
          END;
        END;
      END;

      IF (L_recCollectionContract."Blocking Characteristics" = TRUE) AND (L_AnalysisErr = TRUE) THEN BEGIN
        L_recReceiptAnalysis.RESET;
        L_recReceiptAnalysis.SETCURRENTKEY("Receipt Type","Receipt No.", "Analysis Code");
        L_recReceiptAnalysis.SETRANGE("Receipt Type",L_recReceipt."Receipt Type");
        L_recReceiptAnalysis.SETRANGE("Receipt No.", P_ReceiptNo);

        L_recContractQuality.RESET;
        L_recContractQuality.SETRANGE("Contract No.", P_ContractNo);
        IF L_recContractQuality.FINDSET(FALSE, FALSE) THEN
          REPEAT
            L_AnalysisErr := G_CMCManagement.CheckContractQuality(L_recReceiptAnalysis.Value, L_recContractQuality);
          UNTIL (L_recContractQuality.NEXT = 0) OR (L_AnalysisErr = TRUE);
      END;

      RES_isAnalysisOk := L_AnalysisErr;

      EXIT(RES_isAnalysisOk);
    END;

    PROCEDURE UpdateStatus@60000(QtyOnLine_P@60001 : Decimal);
    VAR
      Qty_L@60000 : Decimal;
    BEGIN
      // --> 20141203-CER03-TER01.01.00-RMZ01
      // This function will update the status of the contract.
      // If all received status = Finished. If not all received status = Locked
      // Input
      //   QtyOnLine_P - quantity from the table 8053104
      // Output
      G_CMCSetup.GET;
      IF NOT G_CMCSetup."Collect Cont. Aut.Clos." THEN
        EXIT;

      CALCFIELDS("Quantity Received");

      Qty_L := ROUND(Quantity * (1 + ("Threshold Percentage" / 100)),0.001);
      IF "Quantity Received" + QtyOnLine_P >= Qty_L THEN
        Status := Status::Finished
      ELSE
        Status := Status::Locked;

      MODIFY;
    END;
