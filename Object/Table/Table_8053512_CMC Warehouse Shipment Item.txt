    PROCEDURE DoOrderExist@1000000000();
    VAR
      SalesLine_L@1000000000 : Record 37;
      WhseShptHeader_L@1000000001 : Record 7320;
      OrderExist_L@1000000002 : Boolean;
      FIFOOrderShptMgt_L@1000000003 : Codeunit 8053065;
      OutstandingQty_L@1000000004 : Decimal;
    BEGIN
      WhseShptHeader_L.GET("No.");
      OrderExist_L := FALSE;
      OutstandingQty_L := 0;

      SalesLine_L.RESET;
      SalesLine_L.SETCURRENTKEY("Document Type","Sell-to Customer No.","Document No.",Type);
      SalesLine_L.SETRANGE("Document Type",SalesLine_L."Document Type"::Order);
      SalesLine_L.SETRANGE("Sell-to Customer No.",WhseShptHeader_L."Sell-to Customer No.");
      SalesLine_L.SETRANGE(Type,SalesLine_L.Type::Item);
      SalesLine_L.SETRANGE("No.","Item No.");
      IF "Variant Code" <> '' THEN
        SalesLine_L.SETRANGE("Variant Code","Variant Code");
      SalesLine_L.SETFILTER("Outstanding Quantity",'<>0');
      IF SalesLine_L.FINDSET THEN
        REPEAT
          IF FIFOOrderShptMgt_L.IsFIFOOrderShptLocation(SalesLine_L."Location Code") THEN BEGIN
            OrderExist_L := TRUE;
            OutstandingQty_L += SalesLine_L."Outstanding Quantity";
          END;
        UNTIL (SalesLine_L.NEXT = 0);

      "Create Adduction Order" := NOT OrderExist_L;
      "Outstanding Quantity" := OutstandingQty_L;
    END;
