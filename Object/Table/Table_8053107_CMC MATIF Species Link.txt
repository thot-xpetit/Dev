    PROCEDURE GetSpeciesActive@1000000000(MatifTerms_P@1000000002 : Code[20]) Species_R : Code[1024];
    VAR
      CMCMatifSpeciesLink_P@1000000000 : Record 8053107;
      CMCMatifSpeciesLinkTemp_P@1000000001 : TEMPORARY Record 8053107;
    BEGIN
      // --> 20150421-CER114-TER01.01.00-334-RMZ01
      // Input
      //   - MatifTerms_P
      // Output
      //   - Species_R
      // This function will return all the species active
      Species_R := '';
      CLEAR(CMCMatifSpeciesLink_P);
      IF MatifTerms_P <> '' THEN
        CMCMatifSpeciesLink_P.SETRANGE("MATIF Terms",MatifTerms_P);
      CMCMatifSpeciesLink_P.SETRANGE(Status,CMCMatifSpeciesLink_P.Status::Active);
      IF CMCMatifSpeciesLink_P.FINDFIRST THEN
        REPEAT
          CMCMatifSpeciesLinkTemp_P.RESET;
          CMCMatifSpeciesLinkTemp_P.SETRANGE("Species Code",CMCMatifSpeciesLink_P."Species Code");
          CMCMatifSpeciesLinkTemp_P.SETRANGE("MATIF Terms",CMCMatifSpeciesLink_P."MATIF Terms");
          IF NOT CMCMatifSpeciesLinkTemp_P.FINDFIRST THEN BEGIN
            CMCMatifSpeciesLinkTemp_P.INIT;
            CMCMatifSpeciesLinkTemp_P := CMCMatifSpeciesLink_P;
            CMCMatifSpeciesLinkTemp_P.INSERT;
          END;
        UNTIL CMCMatifSpeciesLink_P.NEXT = 0;

      CMCMatifSpeciesLinkTemp_P.RESET;
      IF CMCMatifSpeciesLinkTemp_P.FINDFIRST THEN
        REPEAT
          IF Species_R = '' THEN
            Species_R := CMCMatifSpeciesLinkTemp_P."Species Code"
          ELSE
            Species_R := Species_R + '|' + CMCMatifSpeciesLinkTemp_P."Species Code";
        UNTIL CMCMatifSpeciesLinkTemp_P.NEXT = 0;
      EXIT(Species_R);
    END;

    BEGIN
    {

      // --> 3LI END
    }
