    PROCEDURE UpdateTransport@60000();
    VAR
      RecPostedDelivery_L@60001 : Record 8053014;
      RecCalcRefPriceDetailed_L@60002 : Record 60045;
      RecCalcRefPrice_L@60003 : Record 60046;
    BEGIN
      // --> 20161103-CER77-TER01.01.00-3035-NED
      CLEAR(RecPostedDelivery_L);
      CASE "Source Document Type" OF
        "Source Document Type" :: Delivery :
          RecPostedDelivery_L.GET(RecPostedDelivery_L."Delivery Type" :: Delivery,"Source Document No.");
        "Source Document Type" :: Resumption :
          RecPostedDelivery_L.GET(RecPostedDelivery_L."Delivery Type" :: Resumption,"Source Document No.");
      END;
      IF RecPostedDelivery_L."Contract No." <> '' THEN BEGIN
        CLEAR(RecCalcRefPriceDetailed_L);
        IF RecCalcRefPriceDetailed_L.GET(RecPostedDelivery_L."Contract No.",
            RecCalcRefPriceDetailed_L."Document Type" :: "Transport Estimated",RecPostedDelivery_L."No.") THEN BEGIN
          RecCalcRefPriceDetailed_L.RENAME(RecPostedDelivery_L."Contract No.",
          RecCalcRefPriceDetailed_L."Document Type" :: "Transport Invoice",RecPostedDelivery_L."No.");
          // Transport Invoice
          CLEAR(RecCalcRefPriceDetailed_L);
          RecCalcRefPriceDetailed_L.SETRANGE("Contract No.",RecPostedDelivery_L."Contract No.");
          RecCalcRefPriceDetailed_L.SETRANGE("Document Type",RecCalcRefPriceDetailed_L."Document Type"::"Transport Invoice");
          RecCalcRefPriceDetailed_L.CALCSUMS("Total Amount");
          CLEAR(RecCalcRefPrice_L);
          RecCalcRefPrice_L.GET(RecPostedDelivery_L."Contract No.");
          RecCalcRefPrice_L."Transport Invoice" := -RecCalcRefPriceDetailed_L."Total Amount";

          // Transport estimated
          RecCalcRefPriceDetailed_L.SETRANGE("Document Type");
          RecCalcRefPriceDetailed_L.SETRANGE("Document Type",RecCalcRefPriceDetailed_L."Document Type"::"Transport Estimated");
          RecCalcRefPriceDetailed_L.CALCSUMS("Total Amount");
          RecCalcRefPrice_L."Transport Estimation not Inv." := -RecCalcRefPriceDetailed_L."Total Amount";
          RecCalcRefPrice_L.MODIFY;
        END;
      END;
      // <-- 20161103-CER77-TER01.01.00-3035-NED
    END;
