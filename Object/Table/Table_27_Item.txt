    PROCEDURE AddItemPrice@1000000002();
    VAR
      PriceGroup@1000000000 : Record 6;
      StoreGroup@1100409000 : Record 10000779;
      PriceNoPriceGroupLink@1100409001 : Record 10000750;
    BEGIN
      //LS
      //AddItemPrice

      // We want to insert a default item price when we create an item. The price is based on
      // the default price group in the retail setup table.

      RetailSetup.GET;
      IF RetailSetup."Default Price Group" <> '' THEN BEGIN
        ItemPrice.INIT;
        ItemPrice."Item No." := "No.";
        ItemPrice."Sales Type" := ItemPrice."Sales Type"::"Customer Price Group";
        ItemPrice.VALIDATE("Sales Code",RetailSetup."Default Price Group");
        ItemPrice.SetItemCardUpdate();
        IF ItemPrice.INSERT(TRUE) THEN;
      END;

      // We also want to create prices for the 3rd party POS automatically. Since the default price
      // group is taken from the retail setup table, we only take prices 2,3 and 4 from the
      // price group link table.

      PriceNoPriceGroupLink.SETFILTER("Price No.",'>%1',1);
      IF PriceNoPriceGroupLink.FIND('-') THEN REPEAT
        PriceGroup.GET(PriceNoPriceGroupLink."Price Group");
        ItemPrice.INIT;
        ItemPrice."Item No." := "No.";
        ItemPrice."Sales Type" := ItemPrice."Sales Type"::"Customer Price Group";
        ItemPrice.VALIDATE(ItemPrice."Sales Code",PriceGroup.Code);
        ItemPrice.SetItemCardUpdate();
        IF ItemPrice.INSERT(TRUE) THEN;
      UNTIL PriceNoPriceGroupLink.NEXT = 0;
    END;

    PROCEDURE UpdateItemPrice@1000000001();
    VAR
      PriceNoPriceGroupLink@1100409001 : Record 10000750;
      StoreGroup@1100409000 : Record 10000779;
      xItemPrice@1200070000 : Record 7002;
    BEGIN
      //LS
      RetailSetup.GET;

      ItemPrice.RESET;
      ItemPrice.SETRANGE(ItemPrice."Item No.","No.");
      IF ItemPrice.FIND('-') THEN REPEAT
        xItemPrice := ItemPrice;
        IF (RetailSetup."Default Price Group" <> '') AND
           (ItemPrice."Sales Type" = ItemPrice."Sales Type"::"Customer Price Group") AND
           (ItemPrice."Sales Code" = RetailSetup."Default Price Group")
        THEN BEGIN
          IF (ItemPrice."Price Includes VAT" = "Price Includes VAT") AND
             (ItemPrice."VAT Bus. Posting Gr. (Price)" = "VAT Bus. Posting Gr. (Price)") AND
             (ItemPrice."Starting Date" = 0D) AND
             (ItemPrice."Ending Date" = 0D) AND
             (ItemPrice."Currency Code" = '') AND
             (ItemPrice."Variant Code" = '') AND
             ((ItemPrice."Unit of Measure Code" = '') OR (ItemPrice."Unit of Measure Code" = "Sales Unit of Measure")) AND
             (ItemPrice."Minimum Quantity" = 0)
          THEN BEGIN
              ItemPrice."Unit Price" := "Unit Price";
              ItemPrice."Unit Price Including VAT" := "Unit Price Including VAT";
          END;
        END;
        ItemPrice."Calculate Prices"(Rec,0);
        IF (ItemPrice."Unit Price" <> xItemPrice."Unit Price") OR
           (ItemPrice."Unit Price Including VAT" <> xItemPrice."Unit Price Including VAT") OR
           (ItemPrice."Markup %" <> xItemPrice."Markup %") OR
           (ItemPrice."Profit %" <> xItemPrice."Profit %") OR
           (ItemPrice."Profit (LCY)" <> xItemPrice."Profit (LCY)") THEN
          BEGIN
            ItemPrice.SetItemCardUpdate();
            ItemPrice.MODIFY(TRUE);
          END;
      UNTIL ItemPrice.NEXT = 0;
    END;

    PROCEDURE ValidateItemAttributes@1100409006(pAttributeNo@1100409005 : Integer;VAR pNewValue@1100409004 : Text[30]);
    VAR
      lRetailSetup@1100409006 : Record 10000700;
      lAttributeSetup@1100409000 : Record 10000784;
      lAttributeOptionValues@1100409001 : Record 10000785;
      lAttributeValues@1100409002 : Record 10000786;
      lAttributeCode@1100409007 : Code[20];
      lTstValue@1100409012 : Decimal;
      lTstDay@1100409011 : Integer;
      lTstMon@1100409010 : Integer;
      lTstYr@1100409009 : Integer;
      lTstDate@1100409008 : Date;
    BEGIN
      //LS

      IF pNewValue = '' THEN
        EXIT;

      lRetailSetup.GET;
      CASE pAttributeNo OF
        1 : lAttributeCode := lRetailSetup."Item Attrib. 1 Code";
        2 : lAttributeCode := lRetailSetup."Item Attrib. 2 Code";
        3 : lAttributeCode := lRetailSetup."Item Attrib. 3 Code";
        4 : lAttributeCode := lRetailSetup."Item Attrib. 4 Code";
        5 : lAttributeCode := lRetailSetup."Item Attrib. 5 Code";
      END;

      IF lAttributeCode = '' THEN
        EXIT;

      IF lAttributeSetup.GET(lAttributeSetup."Attribute Type"::Item,lAttributeCode) THEN BEGIN

        lAttributeOptionValues.SETRANGE("Attribute Type",lAttributeSetup."Attribute Type");
        lAttributeOptionValues.SETRANGE("Attribute Code",lAttributeCode);

        IF lAttributeOptionValues.FINDFIRST THEN BEGIN
          lAttributeOptionValues.SETRANGE("Option Value",pNewValue);
          IF NOT lAttributeOptionValues.FINDFIRST THEN
            ERROR(Text10000700_01+' = '+pNewValue);
        END;

        CASE lAttributeSetup.Type OF
          lAttributeSetup.Type::Numeric, lAttributeSetup.Type::Amount:
            BEGIN
              IF NOT EVALUATE(lTstValue,pNewValue) THEN
                ERROR(Text10000700_02);

              IF (lAttributeSetup."Min Value" <> 0) OR
                 (lAttributeSetup."Max Value" <> 0) THEN
                 IF (lTstValue < lAttributeSetup."Min Value") OR
                     (lTstValue > lAttributeSetup."Max Value") THEN
                       ERROR(Text10000700_03,lAttributeSetup."Min Value",lAttributeSetup."Max Value");

              IF lAttributeSetup.Type = lAttributeSetup.Type::Amount THEN BEGIN
                IF lTstValue <> 0 THEN
                  pNewValue := FORMAT(lTstValue,15,3)
                ELSE
                  pNewValue := '';
              END;
            END;

          lAttributeSetup.Type::Date:
            BEGIN
              pNewValue := DELCHR(pNewValue,'=','./-');
              CASE STRLEN(pNewValue) OF
                6:
                  BEGIN
                   IF EVALUATE(lTstDay,COPYSTR(pNewValue,1,2)) THEN

    PROCEDURE UpdateCMR@8053014();
    VAR
      RecSecurityLabel_L@8053004 : Record 8053130;
      RecHealthLabel_L@8053003 : Record 8053130;
      RecNatureLabel_L@8053002 : Record 8053130;
      RecItemRiskMention_L@8053001 : Record 8053132;
      RecRiskMention_L@8053000 : Record 8053131;
      RecCategoryCode_L@8053005 : Record 8053183;
      RecTempRiskMention_L@8053006 : TEMPORARY Record 8053131;
      RecTempCategoryCode_L@8053007 : TEMPORARY Record 8053183;
    BEGIN
      //---------------------------------------------------------------------------------------
      // Met . jour le champ CMR en fonction des 'tiquettes et des phrases de risque
      //---------------------------------------------------------------------------------------
      // Entr'es:
      //  -
      // Sorties:
      //  -
      //---------------------------------------------------------------------------------------
      // --> 20121203-Gen03-CMC02.02.00-DRN01
      IF RecSecurityLabel_L.GET(RecSecurityLabel_L.Type::Security,"Security Labels") THEN;
      IF RecHealthLabel_L.GET(RecHealthLabel_L.Type::Health,"Health Labels") THEN;
      IF RecNatureLabel_L.GET(RecNatureLabel_L.Type::Nature,"Nature Labels") THEN;

      RecItemRiskMention_L.SETRANGE("Item No.","No.");
      RecTempRiskMention_L.DELETEALL;
      RecTempCategoryCode_L.DELETEALL;
      CMR := FALSE;
      IF RecItemRiskMention_L.FINDSET THEN
        REPEAT
          RecRiskMention_L.GET(RecItemRiskMention_L.Code);
          RecTempRiskMention_L.INIT;
          RecTempRiskMention_L.TRANSFERFIELDS(RecRiskMention_L);
          RecTempRiskMention_L.INSERT;
        UNTIL (RecItemRiskMention_L.NEXT = 0);
      RecTempRiskMention_L.RESET;
      RecTempRiskMention_L.SETRANGE(RecTempRiskMention_L.CMR,TRUE);
      IF RecTempRiskMention_L.FINDFIRST THEN BEGIN
        REPEAT
          RecCategoryCode_L.GET(RecTempRiskMention_L."CMR Category Code");
          RecTempCategoryCode_L.INIT;
          RecTempCategoryCode_L.TRANSFERFIELDS(RecCategoryCode_L);
          RecTempCategoryCode_L.INSERT;
        UNTIL RecTempRiskMention_L.NEXT=0;
        RecTempCategoryCode_L.RESET;
        RecTempCategoryCode_L.SETCURRENTKEY(Level);
        RecTempCategoryCode_L.ASCENDING(FALSE);
        IF (RecSecurityLabel_L.CMR OR RecHealthLabel_L.CMR OR RecNatureLabel_L.CMR) THEN BEGIN
          CMR := TRUE;
          IF RecTempCategoryCode_L.FINDFIRST THEN
            "CMR Category Code" := RecTempCategoryCode_L.Code;
        END;
      END;
      // <-- 20121203-Gen03-CMC02.02.00-DRN01
    END;
