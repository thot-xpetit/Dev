    PROCEDURE CheckValueWithStandards@8053002(P_ValueToCheck@8053004 : Decimal);
    VAR
      L_RecContractQual@8053000 : Record 8052984;
      L_RecStandardQual@8053001 : Record 8052992;

    BEGIN
      IF L_RecReceipt.GET("Receipt Type", "Receipt No.") THEN BEGIN
        IF L_RecContractQual.GET(L_RecReceipt."Contract No.", "Analysis Code") THEN BEGIN
          IF (NOT L_CMCMgt.CheckContractQuality(P_ValueToCheck,L_RecContractQual)) THEN BEGIN
            IF L_RecContractQual.Blocking THEN
              ERROR(Text8053004, P_ValueToCheck, L_RecReceipt."Contract No.")
            ELSE
              MESSAGE(Text8053004, P_ValueToCheck, L_RecReceipt."Contract No.");
          END;
        END ELSE BEGIN
          IF L_RecStandardQual.GET(L_RecReceipt."Campaign No.", L_RecReceipt."Item No.", "Analysis Code") THEN BEGIN
            IF (NOT L_CMCMgt.CheckSpeciesQuality(P_ValueToCheck, L_RecStandardQual)) THEN BEGIN
              IF L_RecStandardQual.Blocking THEN
                ERROR(Text8053005, P_ValueToCheck, L_RecReceipt."Campaign No.", L_RecReceipt."Item No.")
              ELSE
                MESSAGE(Text8053005, P_ValueToCheck, L_RecReceipt."Campaign No.", L_RecReceipt."Item No.");
            END;
          END;
        END;
      END;
    END;
