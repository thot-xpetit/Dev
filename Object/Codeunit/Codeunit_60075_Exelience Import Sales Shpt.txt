      EndLine_L := FORMAT(CharLF_L) + FORMAT(CharRC_L);

            Result_L := Text8053601;

    PROCEDURE ControlTypeExelienceHeader@1000000018();
    VAR

      LineHeaderNo_L@1000000001 : Text[100];

    BEGIN
      ExelienceFileFlow_L.RESET;
      ExelienceFileFlow_L.SETRANGE(Status,ExelienceFileFlow_L.Status::"In Process");
      ExelienceFileFlow_L.SETRANGE("Import Type", ExelienceFileFlow_L."Import Type"::SalesShpt);
      IF ExelienceFileFlow_L.FINDFIRST THEN REPEAT
        IF ExelienceFileFlow_SequenceNo_L <> ExelienceFileFlow_L."Sequence No." THEN BEGIN
          ExelienceHeader_L.RESET;
          ExelienceHeader_L.SETRANGE(Status,ExelienceHeader_L.Status::"In Process");
          ExelienceHeader_L.SETRANGE("Sequence Flow No.", ExelienceFileFlow_L."Sequence No.");
          IF ExelienceHeader_L.FINDFIRST THEN REPEAT
            WITH ExelienceHeader_L DO BEGIN
              LineFileFlowNo_L := "Sequence Flow No.";
              LineHeaderNo_L := "Sequence No.";

              IF NOT ExelienceImport_G.IsNotEmptyControl("Sales Shipment No.") THEN
                ExelienceImport_G.InsertError(STRSUBSTNO(Text60001,LineHeaderNo_L,FIELDCAPTION("Sales Shipment No."))
                            ,LineFileFlowNo_L,LineHeaderNo_L,'', "Record Order No.");
              IF NOT ExelienceImport_G.IsNotEmptyControl("Sales Shipment Date") THEN
                ExelienceImport_G.InsertError(STRSUBSTNO(Text60001,LineHeaderNo_L,FIELDCAPTION("Sales Shipment Date"))
                            ,LineFileFlowNo_L,LineHeaderNo_L,'', "Record Order No.");
              IF NOT ExelienceImport_G.IsNotEmptyControl("Delivered To") THEN
                ExelienceImport_G.InsertError(STRSUBSTNO(Text60001,LineHeaderNo_L,FIELDCAPTION("Delivered To"))
                            ,LineFileFlowNo_L,LineHeaderNo_L,'', "Record Order No.");
            END;
          UNTIL ExelienceHeader_L.NEXT = 0;
          ExelienceFileFlow_SequenceNo_L := ExelienceFileFlow_L."Sequence No.";
        END;
