    PROCEDURE CreatePeriodFormat@4(PeriodType@1000 : 'Day,Week,Month,Quarter,Year,Accounting Period,Campaign';Date@1001 : Date) : Text[10];
    BEGIN
      CASE PeriodType OF
        PeriodType::Day:
          EXIT(FORMAT(Date));
        PeriodType::Week:
          BEGIN
            IF DATE2DWY(Date,2) = 1 THEN
              Date := Date + 7 - DATE2DWY(Date,1);
            EXIT(FORMAT(Date,0,Text000));
          END;
        PeriodType::Month:
          EXIT(FORMAT(Date,0,Text10800));
        PeriodType::Quarter:
          EXIT(FORMAT(Date,0,Text002));
        PeriodType::Year:
          EXIT(FORMAT(Date,0,Text003));
        PeriodType::"Accounting Period":
          EXIT(FORMAT(Date));
        // --> 20141203-FINA12-CMC02.02.00-FEC01
        PeriodType::Campaign:
          EXIT(FORMAT(Date));
        // <-- 20141203-FINA12-CMC02.02.00-FEC01
      END;
    END;
