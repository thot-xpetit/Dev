    PROCEDURE CrtlRegistrationNo@60001(RegistrationNo_P@60000 : Text[20]);
    VAR
      Text60002@60002 : TextConst 'FRA=La longueur du N› de SIRET doit ^tre de 14 caractSres.';
      Text60003@60003 : TextConst 'FRA=Le N› de SIRET %1 n''est pas valide.';

    PROCEDURE CrtlLuhn@60004(Value_P@60000 : Text[20]) : Boolean;
    VAR
      Digit_L@60006 : ARRAY [20] OF Integer;
      Result_L@60020 : Integer;
      I_L@60024 : Integer;
    BEGIN
      FOR I_L := 1 TO 20 DO
        IF COPYSTR(Value_P,I_L,1) IN ['1','2','3','4','5','6','7','8','9'] THEN
          EVALUATE(Digit_L[21 - I_L],COPYSTR(Value_P,I_L,1));

      FOR I_L := 1 TO 10 DO
        BEGIN
          Digit_L[I_L * 2] := Digit_L[I_L * 2] * 2;
          IF Digit_L[I_L * 2] >= 10 THEN
            Digit_L[I_L * 2] := Digit_L[I_L * 2] -10 + 1;
        END;

      FOR I_L := 1 TO 20 DO
        Result_L := Result_L + Digit_L[I_L];

      IF Result_L MOD 10 <> 0 THEN
        EXIT(FALSE)
      ELSE
        EXIT(TRUE);
    END;

    PROCEDURE CheckIBAN@60000(IBANCode_P@60000 : Code[100]) : Boolean;
    VAR
      Modulus97_L@60002 : Integer;
      I_L@60001 : Integer;
    BEGIN
      IF IBANCode_P = '' THEN
        EXIT(TRUE);
      IBANCode_P := DELCHR(IBANCode_P);
      Modulus97_L := 97;
      IF (STRLEN(IBANCode_P) <= 5) OR (STRLEN(IBANCode_P) > 34) THEN
        EXIT(FALSE);
      ConvertIBAN(IBANCode_P);
      WHILE STRLEN(IBANCode_P) > 6 DO
        IBANCode_P := CalcModulus(COPYSTR(IBANCode_P,1,6),Modulus97_L) + COPYSTR(IBANCode_P,7);
      EVALUATE(I_L,IBANCode_P);
      IF (I_L MOD Modulus97_L) <> 1 THEN
        EXIT(FALSE);
      EXIT(TRUE);
    END;

    LOCAL PROCEDURE ConvertIBAN@4(VAR IBANCode@1000 : Code[100]);
    VAR
      I@1002 : Integer;
    BEGIN
      IBANCode := COPYSTR(IBANCode,5) + COPYSTR(IBANCode,1,4);
      I := 0;
      WHILE I < STRLEN(IBANCode) DO BEGIN
        I := I + 1;
        IF ConvertLetter(IBANCode,COPYSTR(IBANCode,I,1),I) THEN
          I := 0;
      END;
    END;

    LOCAL PROCEDURE CalcModulus@3(Number@1000 : Code[10];Modulus97@1001 : Integer) : Code[10];
    VAR
      I@1002 : Integer;
    BEGIN
      EVALUATE(I,Number);
      I := I MOD Modulus97;
      IF I = 0 THEN
        EXIT('');
      EXIT(FORMAT(I));
    END;
