    PROCEDURE CheckSalesQty@8060984(VAR SalesLine_P@8060982 : Record 37);
    VAR
      ItemUoM_L@8060983 : Record 5404;
      SalesSetup_L@8060984 : Record 311;
      CalcMethod_L@8060985 : 'Upward,Downward';
      UpwardCalcQty_L@8060986 : Decimal;
      DownwardCalcQty_L@8060987 : Decimal;
      DisplayedSTRMenu_L@8060988 : Text[150];
      DisplayedSTRInstr_L@8060989 : Text[150];
    BEGIN
      WITH SalesLine_P DO BEGIN

        IF NOT ("Document Type" IN ["Document Type"::Quote,"Document Type"::Order]) THEN
          EXIT;

        SalesSetup_L.GET;

        IF (Type = Type::Item) AND (Quantity <> 0) THEN
          IF ItemUoM_L.GET("No.","Unit of Measure Code") AND (ItemUoM_L."Sales Indiv. Qty" <> 0) THEN
            IF (Quantity MOD ItemUoM_L."Sales Indiv. Qty") <> 0 THEN BEGIN

              UpwardCalcQty_L := ROUND(Quantity / ItemUoM_L."Sales Indiv. Qty",1,'>') * ItemUoM_L."Sales Indiv. Qty";
              DownwardCalcQty_L := ROUND(Quantity / ItemUoM_L."Sales Indiv. Qty",1,'<') * ItemUoM_L."Sales Indiv. Qty";
              IF DownwardCalcQty_L = 0 THEN
                DownwardCalcQty_L := ItemUoM_L."Sales Indiv. Qty";

              IF SalesSetup_L."Indiv. Qty Method" = SalesSetup_L."Indiv. Qty Method"::Prompt THEN BEGIN
                DisplayedSTRMenu_L := STRSUBSTNO(Text8060984,UpwardCalcQty_L,DownwardCalcQty_L);
                DisplayedSTRInstr_L := STRSUBSTNO(Text8060986,Quantity);
                CASE STRMENU(DisplayedSTRMenu_L,1,DisplayedSTRInstr_L) OF
                  0 : ERROR(Text8060985);
                  1 : CalcMethod_L := 0;
                  2 : CalcMethod_L := 1;
                END;
              END ELSE
                CalcMethod_L := SalesSetup_L."Indiv. Qty Method";

              CASE CalcMethod_L OF
                CalcMethod_L::Upward :
                  Quantity := UpwardCalcQty_L;
                CalcMethod_L::Downward :
                  Quantity := DownwardCalcQty_L;
              END;

            END;
      END;
    END;
