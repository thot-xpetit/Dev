            //-->20130419-INT03-CMC02.02.00-ADVITAM-GMX

            END ELSE BEGIN

  CODE
  {
    VAR
      "<-- 3LI TEXT GLOBALS -->"@1000000006 : TextConst;
      Text0001@1000000010 : TextConst 'ENU=Error Table : %1 doesn''t exist for company : "%2";FRA=Erreur Table : "%1"  Inexistante pour la societe : "%2"';
      Text0002@1000000003 : TextConst 'ENU=File Name Doesn''t exist : "%1" for the company : "%2";FRA=Nom fichier a exporter : "%1" inexistant  pour la societe : "%2"';
      Text0003@1000000000 : TextConst 'ENU=Error Creating File %1 \%2.;FRA=Erreur cr'ation fichier %1 \%2.';
      Text0004@1000000013 : TextConst 'ENU=Error updating table %1;FRA=Erreur modification table "%1"';
      Text0005@1000000002 : TextConst 'ENU=See the log please;FRA=Veuillez consulter le log';
      Text0006@1000000005 : TextConst 'FRA=.ŸÎS'^%O<"--?';
      Text0007@1000000004 : TextConst 'FRA=aaceeeeiiouuu';
      "<-- 3LI GLOBALS -->"@1000000012 : Integer;
      MoteurInterface_G@1000000011 : Codeunit 8061230;
      ConnectorName_G@1000000009 : Code[20];
      ImpExpFolder_G@1000000008 : Text[250];
      LogFolder_G@1000000007 : Text[250];
      Text0011@1000000001 : TextConst 'ENU=Error backup copy.;FRA=Erreur copie de sauvegarde.';

    PROCEDURE "<-- 3LI FUNCTIONS -->"@1000000000();
    BEGIN
    END;

    PROCEDURE ExportCombination@1000000001(FileName_P@1000000000 : Text[256]) Result_L : Text[256];
    VAR
      "<-- 3LI LOCALS -->"@1000000003 : Integer;
      FileExport_L@1000000002 : File;
      FileExportOutstream_L@1000000001 : OutStream;
      ToolsFile_L@1000000004 : Codeunit 8053405;
      MoteurInterface_L@1000000013 : Codeunit 8061230;
      TableCompany_L@1000000006 : Record 2000000006;
      TableParameter_L@1000000005 : Record 8053402;
      TableCombination_L@1000000007 : Record 8061733;
      TableLedgerSetup_L@1000000009 : Record 98;
      tableGlobalDimension_L@1000000010 : Record 349;
      ConnectorName_L@1000000017 : Code[20];
      ImpExpFolder_L@1000000016 : Text[256];
      LogFolder_L@1000000015 : Text[256];
      ExtensionFile_L@1000000021 : Text[10];
      PrefixFile_L@1000000020 : Text[10];
      FileName_L@1000000014 : Text[256];
      FileLogName_L@1000000012 : Text[256];
      FilePath_L@1000000019 : Text[256];
      FileN_L@1000000018 : Text[256];
      FileTemp_L@1000000008 : Text[256];
      CodeUnit_L@1000000011 : Integer;
      FileNameOuvert_L@1000000023 : Boolean;
      NbCarEcrit_L@1000000022 : Integer;
    BEGIN
      //----------------------------------------------------------------------------------------------------------------------------------
      // Extraction des matricules dans toutes les societes pour Cr'ation du fichier Dimo : DIMMAT.txt
      //----------------------------------------------------------------------------------------------------------------------------------
      // Entr'es:
      //   - FileName_P : Nom du fichier PC . G'n'rer
      // Sorties:
      //   - Result_L   : Erreur Retourn'e, Vide si Ok
      //----------------------------------------------------------------------------------------------------------------------------------
      FileNameOuvert_L := FALSE;
      CodeUnit_L := 8053411;
      MoteurInterface_L.GetParameter(CodeUnit_L, ConnectorName_L, ImpExpFolder_L, LogFolder_L, ExtensionFile_L, PrefixFile_L);
      ToolsFile_L.LogInit(LogFolder_L, 'DIMO');
      //-->20131112-Int03-CMC02.02.00-ADVITAM-CDL
      ToolsFile_L.LogInitCodeUnit(CodeUnit_L);
      //<--20131112-Int03-CMC02.02.00-ADVITAM-CDL

      // Nom du fichier pass' en paramStre ou Nom constitu' par r'p. du Robot & Nom fic. r'cup'r' du robot
      IF FileName_P = '' THEN BEGIN
        IF (ExtensionFile_L = '') OR (PrefixFile_L = '') THEN BEGIN
          ToolsFile_L.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0002, TableParameter_L.TABLECAPTION, COMPANYNAME), 1, MAXSTRLEN(Result_L)));
        END ELSE BEGIN
          FileName_L := ImpExpFolder_L + '\' + PrefixFile_L + '.' + ExtensionFile_L;
        END;
      END ELSE BEGIN
        FileName_L := FileName_P;
      END;

      IF ToolsFile_L.LogErreurType() <> 'E' THEN BEGIN
        IF TableCompany_L.FINDSET THEN BEGIN
          ToolsFile_L.LogInitFileExecute(FileName_L);
          ToolsFile_L.SplitFilename(FileName_L, FilePath_L, FileN_L);
          FileTemp_L := FilePath_L + 'Temp_' + FileN_L;
          IF NOT FileExport_L.CREATE(FileTemp_L) THEN BEGIN
            ToolsFile_L.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0003, FileTemp_L, GETLASTERRORTEXT), 1, MAXSTRLEN(Result_L)));
            CLEARLASTERROR;
          END ELSE BEGIN
            FileExport_L.CREATEOUTSTREAM(FileExportOutstream_L);
            FileNameOuvert_L := TRUE;
            REPEAT
              TableParameter_L.CHANGECOMPANY(TableCompany_L.Name);
              TableCombination_L.CHANGECOMPANY(TableCompany_L.Name);
              TableLedgerSetup_L.CHANGECOMPANY(TableCompany_L.Name);
              tableGlobalDimension_L.CHANGECOMPANY(TableCompany_L.Name);
              IF (NOT TableParameter_L.FINDFIRST) THEN BEGIN
                ToolsFile_L.LogWriteInfo(COPYSTR(STRSUBSTNO(Text0001, TableParameter_L.TABLECAPTION, TableCompany_L.Name), 1
                  , MAXSTRLEN(Result_L)));
              END ELSE BEGIN
                IF TableParameter_L."Export Files DIMO" THEN BEGIN
                  IF TableCombination_L.FINDSET THEN BEGIN
                    // r'cup'ration du nom des 2 axes analytiques
                    IF NOT TableLedgerSetup_L.FINDFIRST THEN BEGIN
                      ToolsFile_L.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0001, TableLedgerSetup_L.TABLECAPTION, TableCompany_L.Name), 1
                        , MAXSTRLEN(Result_L)));
                    END ELSE BEGIN
                      REPEAT
                        //-->20140422-Int03-CMC02.02.00-ADVITAM-GMX
                        // BEFORE
                        IF TableCombination_L.Blocked THEN BEGIN
                        END ELSE BEGIN
                          {
                          FileExportOutstream_L.WRITETEXT(PADSTR(TableCombination_L.Code, 6, ' '), 6);
                          FileExportOutstream_L.WRITETEXT('|', 1);
                          FileExportOutstream_L.WRITETEXT(PADSTR(UPPERCASE(CONVERTSTR(TableCombination_L.Description, Text0006, Text0007))
                             30, ' '), 30);
                          }
                          // AFTER
                          FileExportOutstream_L.WRITETEXT(PADSTR(TableCombination_L.Code, 20, ' '), 20);
                          FileExportOutstream_L.WRITETEXT('|', 1);
                          FileExportOutstream_L.WRITETEXT(PADSTR(UPPERCASE(CONVERTSTR(TableCombination_L.Description, Text0006, Text0007))
      W
                             80, ' '), 80);
                          //<--20140422-Int03-CMC02.02.00-ADVITAM-GMX
                          FileExportOutstream_L.WRITETEXT('|', 1);
                          FileExportOutstream_L.WRITETEXT(PADSTR(TableCombination_L."Shortcut Dimension 1 Code", 4, ' '), 4);
                          FileExportOutstream_L.WRITETEXT('|', 1);
                          FileExportOutstream_L.WRITETEXT(PADSTR(TableCombination_L."Shortcut Dimension 2 Code", 4, ' '), 4);
                          FileExportOutstream_L.WRITETEXT('|', 1);
                          FileExportOutstream_L.WRITETEXT(PADSTR(TableParameter_L."Company Name", 3, ' '), 3);
                          FileExportOutstream_L.WRITETEXT('|', 1);
                          FileExportOutstream_L.WRITETEXT('99999999', 8);
                          FileExportOutstream_L.WRITETEXT('|', 1);

                          //CLEAR(tableGlobalDimension_L);   Ne pas faire de CLEAR quand CHANGECOMPANY
                          tableGlobalDimension_L.SETRANGE(tableGlobalDimension_L."Dimension Code",
                            TableLedgerSetup_L."Global Dimension 1 Code");
                          tableGlobalDimension_L.SETRANGE(tableGlobalDimension_L.Code, TableCombination_L."Shortcut Dimension 1 Code");
                          IF NOT tableGlobalDimension_L.FINDFIRST THEN BEGIN
                            CLEAR(tableGlobalDimension_L.Name);
                          END;
                          FileExportOutstream_L.WRITETEXT(PADSTR(UPPERCASE(CONVERTSTR(tableGlobalDimension_L.Name, Text0006, Text0007)),
                             30, ' '), 30);
                          FileExportOutstream_L.WRITETEXT('|', 1);

                          //CLEAR(tableGlobalDimension_L);   Ne pas faire de CLEAR quand CHANGECOMPANY
                          tableGlobalDimension_L.SETRANGE(tableGlobalDimension_L."Dimension Code",
                            TableLedgerSetup_L."Global Dimension 2 Code");
                          tableGlobalDimension_L.SETRANGE(tableGlobalDimension_L.Code, TableCombination_L."Shortcut Dimension 2 Code");
                          IF NOT tableGlobalDimension_L.FINDFIRST THEN BEGIN
                            CLEAR(tableGlobalDimension_L.Name);
                          END;
                          FileExportOutstream_L.WRITETEXT(PADSTR(UPPERCASE(CONVERTSTR(tableGlobalDimension_L.Name, Text0006, Text0007)),
                             30, ' '), 30);
                          FileExportOutstream_L.WRITETEXT() ;
                        END;
                      UNTIL TableCombination_L.NEXT=0;
                    END;
                  END;
                  TableParameter_L."Last Date Export DIMMAT" := CURRENTDATETIME;
                  IF NOT TableParameter_L.MODIFY(TRUE) THEN BEGIN
                    ToolsFile_L.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0004, TableParameter_L.TABLECAPTION), 1, MAXSTRLEN(Result_L)));
                  END;
                END;
              END;
            UNTIL TableCompany_L.NEXT=0;
            NbCarEcrit_L := FileExport_L.LEN;
            FileExport_L.CLOSE;
          END;
        END;
      END;

      IF (NbCarEcrit_L > 0) THEN BEGIN
        IF ToolsFile_L.LogErreurType() = 'E' THEN BEGIN
          IF FileNameOuvert_L THEN BEGIN
            FILE.ERASE(FileTemp_L);
          END;
        END ELSE BEGIN
          IF FILE.ERASE(FileName_L) THEN BEGIN
          END;
          IF FILE.RENAME(FileTemp_L, FileName_L) THEN BEGIN
          END;
          IF NOT ToolsFile_L.FileCopySav(FileName_L, 'Sav', TRUE) THEN BEGIN
            ToolsFile_L.LogWriteErreur(STRSUBSTNO(Text0011, FileName_L));
          END;
        END;
      END ELSE BEGIN
        IF FileNameOuvert_L THEN BEGIN
          FILE.ERASE(FileTemp_L);
        END;
      END;

      CLEAR(Result_L);
      FileLogName_L := ToolsFile_L.LogName;
      ToolsFile_L.LogClose();
      IF ToolsFile_L.LogErreurType() = 'E' THEN BEGIN
        Result_L := 'ERREUR ' + FileLogName_L;
        ERROR(Result_L);
      END ELSE BEGIN
        IF ToolsFile_L.LogErreurType() = 'I' THEN BEGIN
          Result_L := Text0005;
          MoteurInterface_L.InsertLog(CodeUnit_L, FileLogName_L);
        END ELSE BEGIN
          MoteurInterface_L.InsertOk(CodeUnit_L);
        END;
      END;
      //<--20130311-INT03-CMC02.02.00-ADVITAM-GMX
    END;

    BEGIN
    {
      //--> 3LI BEGIN
      20130311-INT03-CMC02.02.00-ADVITAM-GMX : Export matricule
      20131112-Int03-CMC02.02.00-ADVITAM-CDL : - Ajout appel fonction ToolsFile_L.LogInitCodeUnit.
      20140422-Int03-CMC02.02.00-ADVITAM-GMX : - Modification taille champ code mtricule et libell' matricule.
      //<-- 3LI END
    }
    END.
  }
