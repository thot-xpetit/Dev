    PROCEDURE ExportGLAccount@1000000001(FileName_P@1000000000 : Text[256]) Result_L : Text[256];
    VAR
      "<-- 3LI LOCALS -->"@1000000001 : Integer;
      FileExport_L@1000000002 : File;
      FileExportOutstream_L@1000000003 : OutStream;
      ToolsFile_L@1000000005 : Codeunit 8053405;
      MoteurInterface_L@1000000009 : Codeunit 8061230;
      TableCompany_L@1000000006 : Record 2000000006;
      TableParameter_L@1000000008 : Record 8053402;
      TableGLAccount_L@1000000011 : Record 15;
      TableLedgerSetup_L@1000000004 : Record 98;
      ConnectorName_L@1000000013 : Code[20];
      ImpExpFolder_L@1000000012 : Text[250];
      LogFolder_L@1000000010 : Text[250];
      ExtensionFile_L@1000000019 : Text[10];
      PrefixFile_L@1000000020 : Text[10];
      FileName_L@1000000014 : Text[250];
      FilePath_L@1000000016 : Text[250];
      FileN_L@1000000017 : Text[250];
      FileTemp_L@1000000018 : Text[250];
      FileLogName_L@1000000007 : Text[250];
      CodeUnit_L@1000000015 : Integer;
      FileNameOuvert_L@1000000021 : Boolean;
      NbCarEcrit_L@1000000022 : Integer;
    BEGIN
      //----------------------------------------------------------------------------------------------------------------------------------
      // Extraction des comptes g'n'raux dans toutes les societes pour Cr'ation du fichier Dimo : DIMCPTG.txt
      //----------------------------------------------------------------------------------------------------------------------------------
      // Entr'es:
      //   - FileName_P : Nom du fichier PC . G'n'rer
      // Sorties:
      //   - Result_L   : Erreur Retourn'e, Vide si Ok
      //----------------------------------------------------------------------------------------------------------------------------------
      FileNameOuvert_L := FALSE;
      CodeUnit_L := 8053406;
      MoteurInterface_L.GetParameter(CodeUnit_L, ConnectorName_L, ImpExpFolder_L, LogFolder_L, ExtensionFile_L, PrefixFile_L);
      ToolsFile_L.LogInit(LogFolder_L, 'DIMO');
      //-->20131112-Int03-CMC02.02.00-ADVITAM-CDL
      ToolsFile_L.LogInitCodeUnit(CodeUnit_L);
      //<--20131112-Int03-CMC02.02.00-ADVITAM-CDL

      // Nom du fichier pass' en paramStre ou Nom constitu' par r'p. du Robot & Nom fic. (Pr'fixe + Extension) r'cup'r' du robot
      IF FileName_P = '' THEN BEGIN
        IF (ExtensionFile_L = '') OR (PrefixFile_L = '') THEN BEGIN
          ToolsFile_L.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0002, 'ROBOT', COMPANYNAME), 1, MAXSTRLEN(Result_L)));
        END ELSE BEGIN
          FileName_L := ImpExpFolder_L + '\' + PrefixFile_L + '.' + ExtensionFile_L;
        END;
      END ELSE BEGIN
        FileName_L := FileName_P;
      END;

      IF ToolsFile_L.LogErreurType() <> 'E' THEN BEGIN
        IF TableCompany_L.FINDSET THEN BEGIN
          ToolsFile_L.LogInitFileExecute(FileName_L);
          ToolsFile_L.SplitFilename(FileName_L, FilePath_L, FileN_L);
          FileTemp_L := FilePath_L + 'Temp_' + FileN_L;
          IF NOT FileExport_L.CREATE(FileTemp_L) THEN BEGIN
            ToolsFile_L.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0003, FileTemp_L, GETLASTERRORTEXT), 1, MAXSTRLEN(Result_L)));
            CLEARLASTERROR;
          END ELSE BEGIN
            FileExport_L.CREATEOUTSTREAM(FileExportOutstream_L);
            FileNameOuvert_L := TRUE;
            REPEAT
              TableGLAccount_L.CHANGECOMPANY(TableCompany_L.Name);
              TableParameter_L.CHANGECOMPANY(TableCompany_L.Name);
              TableLedgerSetup_L.CHANGECOMPANY(TableCompany_L.Name);
              TableDftDim_G.CHANGECOMPANY(TableCompany_L.Name);
              IF (NOT TableParameter_L.FINDFIRST) THEN BEGIN
                ToolsFile_L.LogWriteInfo(COPYSTR(STRSUBSTNO(Text0001, TableParameter_L.TABLECAPTION, TableCompany_L.Name),
                  1, MAXSTRLEN(Result_L)));
              END ELSE BEGIN
                IF TableParameter_L."Export Files DIMO" THEN BEGIN
                  //CLEAR(TableGLAccount_L);  Ne pas faire de CLEAR quand CHANGECOMPANY
                  // Filtre sur compte imputable
                  TableGLAccount_L.SETRANGE(TableGLAccount_L."Account Type" , TableGLAccount_L."Account Type"::Posting);
                  IF TableGLAccount_L.FINDSET THEN BEGIN
                    // r'cup'ration du nom des 2 axes analytiques
                    IF NOT TableLedgerSetup_L.FINDFIRST THEN BEGIN
                      ToolsFile_L.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0001, TableLedgerSetup_L.TABLECAPTION, TableCompany_L.Name), 1,
                         MAXSTRLEN(Result_L)));
                    END ELSE BEGIN
                      REPEAT
                        FileExportOutstream_L.WRITETEXT(PADSTR(TableGLAccount_L."No.", 9, ' '), 9);
                        FileExportOutstream_L.WRITETEXT('|', 1);
                        FileExportOutstream_L.WRITETEXT(PADSTR(UPPERCASE(CONVERTSTR(TableGLAccount_L.Name, Text0006, Text0007)), 30,
                          ' '), 30);
                        FileExportOutstream_L.WRITETEXT('|', 1);
                        FileExportOutstream_L.WRITETEXT(PADSTR(TableParameter_L."Company Name", 3, ' '), 3);
                        FileExportOutstream_L.WRITETEXT('|', 1);
                        FileExportOutstream_L.WRITETEXT(DimensionOk(TableLedgerSetup_L."Global Dimension 1 Code",
                           TableGLAccount_L."No."), 1);
                        FileExportOutstream_L.WRITETEXT('|', 1);
                        FileExportOutstream_L.WRITETEXT(DimensionOk(TableLedgerSetup_L."Global Dimension 2 Code",
                           TableGLAccount_L."No."), 1);
                        FileExportOutstream_L.WRITETEXT();
                      UNTIL TableGLAccount_L.NEXT=0;
                    END;
                  END;
                  TableParameter_L."Last Date Export DIMCPTG" := CURRENTDATETIME;
                  IF NOT TableParameter_L.MODIFY(TRUE) THEN BEGIN
                    ToolsFile_L.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0004, TableParameter_L.TABLECAPTION), 1, MAXSTRLEN(Result_L)));
                  END;
                END;
              END;
            UNTIL TableCompany_L.NEXT=0;
            NbCarEcrit_L := FileExport_L.LEN;
            FileExport_L.CLOSE;
          END;
        END;
      END;

      IF (NbCarEcrit_L > 0) THEN BEGIN
        IF ToolsFile_L.LogErreurType() = 'E' THEN BEGIN
          IF FileNameOuvert_L THEN BEGIN
            FILE.ERASE(FileTemp_L);
          END;
        END ELSE BEGIN
          IF FILE.ERASE(FileName_L) THEN BEGIN
          END;
          IF FILE.RENAME(FileTemp_L, FileName_L) THEN BEGIN
          END;
          IF NOT ToolsFile_L.FileCopySav(FileName_L, 'Sav', TRUE) THEN BEGIN
            ToolsFile_L.LogWriteErreur(STRSUBSTNO(Text0011, FileName_L));
          END;
        END;
      END ELSE BEGIN
        IF FileNameOuvert_L THEN BEGIN
          FILE.ERASE(FileTemp_L);
        END;
      END;

      CLEAR(Result_L);
      FileLogName_L := ToolsFile_L.LogName;
      ToolsFile_L.LogClose();
      IF ToolsFile_L.LogErreurType() = 'E' THEN BEGIN
        Result_L := 'ERREUR ' + FileLogName_L;
        ERROR(Result_L);
      END ELSE BEGIN
        IF ToolsFile_L.LogErreurType() = 'I' THEN BEGIN
          Result_L := Text0005;
          MoteurInterface_L.InsertLog(CodeUnit_L, FileLogName_L);
        END ELSE BEGIN
          MoteurInterface_L.InsertOk(CodeUnit_L);
        END;
      END;
    END;

    PROCEDURE DimensionOk@1000000002(Dimension_P@1000000000 : Code[20];No_P@1000000001 : Code[20]) Result_L : Text[1];
    BEGIN
      //----------------------------------------------------------------------------------------------------------------------------------
      // Cette fonction permet de savoir si l'axe analytique est g'r' ou pas en fonction du No de compte
      //----------------------------------------------------------------------------------------------------------------------------------
      // Entr'es:
      //   - Dimension_P : Nom du l'axe analytique
      //   - No_P        : No du compte g'n'ral
      // Sorties:
      //   - Result_L    : 'N' si axe non trait', 'O' si axe trait'
      //----------------------------------------------------------------------------------------------------------------------------------

      Result_L := 'N';
      //CLEAR(TableDftDim_G); Ne pas faire de CLEAR quand CHANGECOMPANY
      TableDftDim_G.SETRANGE(TableDftDim_G."Table ID", 15);
      TableDftDim_G.SETRANGE(TableDftDim_G."No.", No_P);
      TableDftDim_G.SETRANGE(TableDftDim_G."Dimension Code", Dimension_P);
      IF TableDftDim_G.FINDFIRST THEN BEGIN
        IF ((TableDftDim_G."Value Posting" = TableDftDim_G."Value Posting"::"Code Mandatory")
         OR (TableDftDim_G."Value Posting" = TableDftDim_G."Value Posting"::"Same Code")
         OR (TableDftDim_G."Value Posting" = TableDftDim_G."Value Posting"::Always)) THEN BEGIN
          Result_L := 'O';
        END;
      END;

      //<--20130204-INT03-CMC02.02.00-ADVITAM-GMX
    END;

    BEGIN
    {
      //--> 3LI BEGIN
      20130204-INT03-CMC02.02.00-ADVITAM-GMX : Export Compte G'n'ral
      20131112-Int03-CMC02.02.00-ADVITAM-CDL : - Ajout appel fonction ToolsFile_L.LogInitCodeUnit.
      //<-- 3LI END
    }
    END.
  }
