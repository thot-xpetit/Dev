    PROCEDURE TraitementInitialisation@1000000008();
    VAR
      "<-- 3LI LOCALS -->"@1000000009 : Integer;
      TableADR_L@1000000001 : Record 8053054;
      Position_L@1000000000 : Integer;
      ZoneTemp_L@1000000002 : Text[250];
    BEGIN
      // Code ADR d'termin' sur Code ONU & Groupe emballage
      CLEAR(CodeADR_G);
      IF (TableProDat_G.CodeONU <> '') OR (TableProDat_G.GroupeEmbal <> '') THEN BEGIN
        TableADR_L.SETRANGE("UN No.", TableProDat_G.CodeONU);
        TableADR_L.SETRANGE("Packing Group", TableProDat_G.GroupeEmbal);
        IF TableADR_L.FINDFIRST THEN BEGIN
          CodeADR_G := TableADR_L.Code;
        END;
      END;

      // Conversion zone text en num'rique
      IF TableProDat_G.PointEclair = '' THEN BEGIN
        PointEclair_G := 999;
      END ELSE BEGIN
        IF NOT EVALUATE(PointEclair_G, TableProDat_G.PointEclair) THEN BEGIN
          ERROR(TextZone01, TableProDat_G.FIELDCAPTION(PointEclair), TableProDat_G.PointEclair,
            TableProDat_G."Code Ean", TableProDat_G.LibCourt);
        END;
      END;
      IF TableProDat_G.SensiChaud = '' THEN BEGIN
        SensiChaud_G := 999;
      END ELSE BEGIN
        IF NOT EVALUATE(SensiChaud_G, TableProDat_G.SensiChaud) THEN BEGIN
          ERROR(TextZone01, TableProDat_G.FIELDCAPTION(SensiChaud), TableProDat_G.SensiChaud,
            TableProDat_G."Code Ean", TableProDat_G.LibCourt);
        END;
      END;
      IF TableProDat_G.SensiFroid = '' THEN BEGIN
        SensiFroid_G := 999;
      END ELSE BEGIN
        IF NOT EVALUATE(SensiFroid_G, TableProDat_G.SensiFroid) THEN BEGIN
          ERROR(TextZone01, TableProDat_G.FIELDCAPTION(SensiFroid), TableProDat_G.SensiFroid,
            TableProDat_G."Code Ean", TableProDat_G.LibCourt);
        END;
      END;
      CASE TableProDat_G.EtatPhysique OF
        'L': BEGIN
          EtatPhysique_G := EtatPhysique_G::Liquid;
        END;
        'S': BEGIN
          EtatPhysique_G := EtatPhysique_G::Solid;
        END;
        '': BEGIN
          CLEAR(EtatPhysique_G);
        END;
        ELSE BEGIN
          ERROR(TextZone01, TableProDat_G.FIELDCAPTION(EtatPhysique), TableProDat_G.EtatPhysique,
            TableProDat_G."Code Ean", TableProDat_G.LibCourt);
        END;
      END;

      CLEAR(CodeRubICPE1_G);
      CLEAR(CodeRubICPE2_G);
      IF TableProDat_G.CodeRubICPE <> '' THEN BEGIN
        Position_L := 1;
        // ICPE D'composer sur ';'
        REPEAT
          ZoneTemp_L := ToolsFile_G.ExtractTextCsv(TableProDat_G.CodeRubICPE, Position_L, ';', 30);
          IF ZoneTemp_L <> '' THEN BEGIN
            IF CodeRubICPE1_G = '' THEN BEGIN
              CodeRubICPE1_G := ZoneTemp_L;
            END ELSE BEGIN
              IF CodeRubICPE2_G = '' THEN BEGIN
                CodeRubICPE2_G := ZoneTemp_L;
                // On ne prend que les 2 premiers
                Position_L := STRLEN(TableProDat_G.CodeRubICPE) + 1;
              END;
            END;
          END;
        UNTIL Position_L > STRLEN(TableProDat_G.CodeRubICPE);
      END;
    END;
