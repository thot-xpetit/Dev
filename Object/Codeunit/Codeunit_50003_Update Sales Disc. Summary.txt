          END;

  }
  CODE
  {
    VAR
      CMCSalesDiscSalesSummary@1000000000 : Record 8053241;

    PROCEDURE ResetSalesDiscSummary@1000000001();
    VAR
      datefilter@1000000000 : Date;
    BEGIN
      datefilter := DMY2DATE(1,1,1918);
      CMCSalesDiscSalesSummary.SETRANGE("Campaign No.", '2018');
      CMCSalesDiscSalesSummary.SETRANGE("Document Date", datefilter);
      CMCSalesDiscSalesSummary.SETRANGE("Order Date", datefilter);
      IF (CMCSalesDiscSalesSummary.FINDFIRST()) THEN BEGIN
        REPEAT
          CMCSalesDiscSalesSummary."Cust. Disc. Group Code" := '';
          CMCSalesDiscSalesSummary.MODIFY();
        UNTIL (CMCSalesDiscSalesSummary.NEXT() = 0)
      END;
      CMCSalesDiscSalesSummary.RESET();
    END;

    PROCEDURE SetSalesDiscSummary@1000000002();
    VAR
      Cust@1000000000 : Record 18;
      datefilter@1000000001 : Date;
    BEGIN
      datefilter := DMY2DATE(1,1,1918);
      Cust.SETFILTER("Customer Disc. Group", '<>%1', '');
      IF (Cust.FINDFIRST()) THEN BEGIN
        REPEAT
          CMCSalesDiscSalesSummary.SETRANGE("Campaign No.", '2018');
          CMCSalesDiscSalesSummary.SETRANGE("Document Date", datefilter);
          CMCSalesDiscSalesSummary.SETRANGE("Order Date", datefilter);
          CMCSalesDiscSalesSummary.SETRANGE("Customer No.", Cust."No.");
          IF (CMCSalesDiscSalesSummary.FINDFIRST()) THEN BEGIN
            CMCSalesDiscSalesSummary."Cust. Disc. Group Code" := Cust."Customer Disc. Group";
            CMCSalesDiscSalesSummary.MODIFY();
          END ELSE BEGIN
            AddSalesDiscSalesSummary(Cust, '2018', datefilter);
          END;
        UNTIL (Cust.NEXT() = 0);
      END;
    END;

    PROCEDURE AddSalesDiscSalesSummary@1000000004(Cust@1000000000 : Record 18;CampaignNo@1000000001 : Text[30];DateDocument@1000000002 : Date);
    VAR
      NewSummary@1000000003 : Record 8053241;
    BEGIN
      NewSummary.INIT();
      NewSummary."Document Type" := NewSummary."Document Type"::Order;
      NewSummary."Document No." := '14' + CampaignNo + Cust."No.";
      NewSummary."Campaign No." := CampaignNo;
      NewSummary."Document Date" := DateDocument;
      NewSummary."Order Date" := DateDocument;
      NewSummary."Customer No." := Cust."No.";
      NewSummary."Cust. Disc. Group Code" := Cust."Customer Disc. Group";
      NewSummary."Item Disc. Group Code" := '4';
      NewSummary."Gross Total Amount (LCY)" := 1;
      NewSummary.INSERT();
    END;

    PROCEDURE DuplicateSalesDiscSummary@1000000000();
    VAR
      datefilter@1000000000 : Date;
      NewCMCSalesDiscSalesSummary@1000000001 : Record 8053241;
    BEGIN
      CMCSalesDiscSalesSummary.RESET();
      datefilter := DMY2DATE(1,1,1918);
      CMCSalesDiscSalesSummary.SETRANGE("Campaign No.", '2018');
      CMCSalesDiscSalesSummary.SETRANGE("Document Date", datefilter);
      CMCSalesDiscSalesSummary.SETRANGE("Order Date", datefilter);
      IF (CMCSalesDiscSalesSummary.FINDFIRST()) THEN BEGIN
        REPEAT
          NewCMCSalesDiscSalesSummary.INIT();
          NewCMCSalesDiscSalesSummary.TRANSFERFIELDS(CMCSalesDiscSalesSummary);
          NewCMCSalesDiscSalesSummary."Campaign No." := '2019';
          IF (NewCMCSalesDiscSalesSummary.INSERT()) THEN BEGIN
          END;
        UNTIL (CMCSalesDiscSalesSummary.NEXT() = 0)
      END;
      CMCSalesDiscSalesSummary.RESET();
    END;

    BEGIN
    END.
  }
}
