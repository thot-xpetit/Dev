                IF FileNameOuvert_L THEN BEGIN
                  REPEAT
                    IF TableVendor_L.Blocked <> TableVendor_L.Blocked::All THEN BEGIN
                      VendorRadar_L := MoteurInterface_L.GetReferentialCorrespondance(VendorTemplateCode_L, FALSE,

    PROCEDURE GetRadarParameters@1000000006() ReturnValue : Boolean;
    VAR
      RadarSetup@1000000000 : Record 8053404;
    BEGIN
      ReturnValue := FALSE;
      TempRadarParameter_G.DELETEALL();
      IF (RadarSetup.FINDSET()) THEN BEGIN
        TempRadarParameter_G.INIT();
        TempRadarParameter_G.TRANSFERFIELDS(RadarSetup);
        IF (TempRadarParameter_G.INSERT(FALSE)) THEN BEGIN
          ReturnValue := TempRadarParameter_G.FINDSET();
        END;
      END;
    END;

    PROCEDURE SaveRadarParameters@1000000007() ReturnValue : Boolean;
    VAR
      RadarSetup@1000000000 : Record 8053404;
    BEGIN
      ReturnValue := FALSE;
      RadarSetup.LOCKTABLE(TRUE,FALSE);
      IF (RadarSetup.FINDSET()) THEN BEGIN
        RadarSetup."Last Date Export Vendor" := TempRadarParameter_G."Last Date Export Vendor";
        RadarSetup."Last Vendor File Number" := TempRadarParameter_G."Last Vendor File Number";

        ReturnValue := RadarSetup.MODIFY(TRUE);
      END
    END;
