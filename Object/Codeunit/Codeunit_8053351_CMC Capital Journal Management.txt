    BEGIN

            CapJnlTemplate.Type := PageTemplate;
            CapJnlTemplate.Name := FORMAT(CapJnlTemplate.Type,MAXSTRLEN(CapJnlTemplate.Name));
            CapJnlTemplate.Description := STRSUBSTNO(Text8053000_L,CapJnlTemplate.Type);
            CapJnlTemplate.VALIDATE(Type);
            CapJnlTemplate.INSERT;
            COMMIT;
          END;
        1:
          CapJnlTemplate.FIND('-');
         ELSE
          JnlSelected := FORM.RUNMODAL(0,CapJnlTemplate) = ACTION::LookupOK;
      END;
      IF JnlSelected THEN BEGIN
        CapJnlLine.FILTERGROUP := 2;
        CapJnlLine.SETRANGE("Capital Journal Template",CapJnlTemplate.Name);
        CapJnlLine.FILTERGROUP := 0;
        IF OpenFromBatch_G THEN BEGIN
          CapJnlLine."Capital Journal Template" := '';
          FORM.RUN(CapJnlTemplate."Page ID",CapJnlLine);
         END;
       END;
    END;
