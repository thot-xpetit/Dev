    PROCEDURE CheckFileName@100000000(FileName_P@100000000 : Text[1024]);
    VAR
      NewEdgeSAS_L@100000001 : Record 60111;
      Pos_L@100000002 : Integer;
      Found_L@100000003 : Boolean;
      FileName_L@100000004 : Text[1024];
    BEGIN
      NewEdgeSAS_L.RESET;
      NewEdgeSAS_L.SETRANGE(FileName, FileName_P);
      IF NewEdgeSAS_L.FINDFIRST THEN BEGIN
        FileName_P := DELCHR(FileName_P, '<>');
        IF (FileName_P <> '') THEN BEGIN
          Pos_L := STRLEN(FileName_P);
          REPEAT
            Found_L := (COPYSTR(FileName_P, Pos_L, 1) = '\');
            IF NOT Found_L THEN
              Pos_L := Pos_L - 1;
          UNTIL (Pos_L = 0) OR Found_L;
          IF Found_L THEN BEGIN
            FileName_L := COPYSTR(FileName_P, Pos_L+1);
          END ELSE BEGIN
            FileName_L := FileName_P;
          END;
        END;
        ERROR(STRSUBSTNO(Text60000, FileName_L));
      END;
    END;

      // Lit et lance l'action de d'coupage du fichier
      //----------------------------------------------------------------------------------------------------------------------------------

      //    - CompanyNo_P : N› de soci't' . qui appartient le fichier

      //----------------------------------------------------------------------------------------------------------------------------------

      CharLF_L := 10;
      CharRC_L := 13;
      EndLine_L := FORMAT(CharLF_L) + FORMAT(CharRC_L);

      //V'rification si le fichier exist
      IF NOT(EXISTS(Filename_P)) THEN

      //----------------------------------------------------------------------------------------------------------------------------------
      //Ouverture du fichier pour lire du texte en lecture seule
      //----------------------------------------------------------------------------------------------------------------------------------
        IF Result_L = Text8053600 THEN BEGIN
          File_L.WRITEMODE(FALSE);
          File_L.TEXTMODE(TRUE);
          IF NOT(File_L.OPEN(Filename_P)) THEN
            Result_L := Text8053601;
        END;

      //----------------------------------------------------------------------------------------------------------------------------------
      //D'coupage du fichier pour r'cup'rer chaque ligne
      //----------------------------------------------------------------------------------------------------------------------------------
        IF Result_L = Text8053600 THEN BEGIN
          LineNo_L := 1;

          File_L.CREATEINSTREAM(FileStream_L);
            REPEAT
              IF FileStream_L.READTEXT(Line_L) > 0 THEN BEGIN
                //On ne tient pas compte de la premiSre ligne comportant le nom des colonnes
                IF LineNo_L > 1 THEN BEGIN

    PROCEDURE ReadNextItem@8053600(VAR Var_Line_P@8053600 : Text[1000]) Result_L : Text[50];
    VAR
      Position_L@8053601 : Integer;
    BEGIN
      //----------------------------------------------------------------------------------------------------------------------------------
      // R'cupSre le prochain 'l'ment de la liste 'tant s'par' par des ; et l'enlSve de la liste
      //----------------------------------------------------------------------------------------------------------------------------------
      //  Entr'es
      //    - Var_Line_P : Ligne . traiter
      //  Sorties
      //    - Result_L : la prochain 'l'ment de la liste
      //----------------------------------------------------------------------------------------------------------------------------------

      Position_L := STRPOS(Var_Line_P, ',');
      IF Position_L > 0 THEN BEGIN
        Result_L := COPYSTR(Var_Line_P, 1, Position_L - 1);
        Var_Line_P := COPYSTR(Var_Line_P, Position_L + 1, STRLEN(Var_Line_P));
      END ELSE BEGIN
      //Dernier 'l'ment de la liste
        Result_L := Var_Line_P;
        Var_Line_P := '';
      END;
      // --> 20191122-WI00-ADV01.01.00-3444-MRU01
      //Suppression des "
      Result_L := DELCHR(Result_L, '=', '"');
      // <-- 20191122-WI00-ADV01.01.00-3444-MRU01
      EXIT(Result_L);
    END;

    PROCEDURE InsertCotation@100000008(Filename_P@100000000 : Text[1000]);
    VAR
      NewEdgeSAS_L@100000001 : Record 60111;
      OldPFC_L@100000003 : Text[50];
      OldPCTYM_L@100000002 : Text[50];
      NewCotation_L@100000004 : Boolean;
      UpdateMatifPriceContracts_L@100000005 : Record 60054;
      DateText_L@100000006 : Text[30];
      DueDate_L@100000007 : Date;
      CMCMatifTerms_L@100000008 : Record 8053102;
      Amount_L@100000009 : Decimal;
    BEGIN
      //Insertion des cotations
      OldPCTYM_L := '';
      OldPFC_L := '';

      WITH NewEdgeSAS_L DO BEGIN
        RESET;
        SETCURRENTKEY(PFC, PCTYM);
        SETRANGE(FileName, Filename_P);
        SETRANGE(PRECID, 'P');
        SETRANGE(PSUBTY, '');
        SETRANGE(PSTYPE, 'F');
        IF FINDFIRST THEN REPEAT
          NewCotation_L := (PCTYM <> OldPCTYM_L) OR (PFC <> OldPFC_L);
          IF NewCotation_L THEN BEGIN
            // --> 20191125-WI00-ADV01.01.00-3444-MRU01
            {BEFORE
            UpdateMatifPriceContracts_L.RESET;
            AFTER}
            CLEAR(UpdateMatifPriceContracts_L);
            // <-- 20191125-WI00-ADV01.01.00-3444-MRU01

            //Code espSce Matif : g'r' par la table de correspondance
            UpdateMatifPriceContracts_L.VALIDATE("Matif Species Code", MoteurInterface_G.Getcorrespondance(GlobalTemplateCode_G
                                                                , 'CMC MATIF Species', 'Species Code', PFC, TRUE, TRUE));

            //Pour la date d''ch'ance dans le fichier le jour n'est pas renseign' : premier jour du mois par d'faut
            DateText_L := PCTYM;
            DateText_L := '01' + COPYSTR(DateText_L,5,2) + COPYSTR(DateText_L,1,4);
            EVALUATE(DueDate_L, DateText_L);

            //Code 'ch'ance MATIF : d'termin' par les p'riodes d'finies dans la table CMC MATIF Terms
            CMCMatifTerms_L.SETFILTER("Period Start", '<=%1', DueDate_L);
            CMCMatifTerms_L.SETFILTER("Period End", '>=%1', DueDate_L);
            CMCMatifTerms_L.FINDFIRST;
            UpdateMatifPriceContracts_L.VALIDATE("MATIF Terms Code", CMCMatifTerms_L."MATIF Terms");

            EVALUATE(Amount_L, CONVERTSTR(PCLOSE, '.', ','));
            UpdateMatifPriceContracts_L.VALIDATE("Amount Daily Prices to Update", Amount_L);
            UpdateMatifPriceContracts_L."To Update" := TRUE;

            IF NOT UpdateMatifPriceContracts_L.MODIFY THEN BEGIN
              IF MsgError_G <> '' THEN
                MsgError_G += '\';
              MsgError_G += STRSUBSTNO(Text60002, UpdateMatifPriceContracts_L."Matif Species Code"
                                      , UpdateMatifPriceContracts_L."MATIF Terms Code");
            END;
          END;

          OldPCTYM_L := PCTYM;
          OldPFC_L := PFC;
        UNTIL NEXT = 0;
      END;
    END;

    PROCEDURE PostCotation@100000011();
    VAR
      UpdateMatifPriceContracts_L@100000000 : Record 60054;
      CounterFarmer_L@100000002 : Integer;
      MatifContractArbFarmer_L@100000001 : Record 60037;
      CMCMatifContract_L@100000003 : Record 8053065;
      ToLock_L@100000004 : Boolean;
    BEGIN
      WITH UpdateMatifPriceContracts_L DO BEGIN
        RESET;
        SETRANGE("INCOGRAIN Code", '');
        SETRANGE("Commercial Region Code", '');
        SETRANGE("To Update",TRUE);
        IF FINDFIRST THEN BEGIN
          CounterFarmer_L := 0;
          REPEAT
            MatifContractArbFarmer_L.RESET;
            MatifContractArbFarmer_L.SETRANGE("Referency Market", "Matif Species Code");
            MatifContractArbFarmer_L.SETRANGE("MATIF Terms Code", "MATIF Terms Code");
            MatifContractArbFarmer_L.SETFILTER(Status,'%1|%2',MatifContractArbFarmer_L.Status::Open,
                    MatifContractArbFarmer_L.Status::Locked);
            IF MatifContractArbFarmer_L.FINDFIRST THEN REPEAT
              ToLock_L := FALSE;
              IF MatifContractArbFarmer_L.Status = MatifContractArbFarmer_L.Status::Locked THEN BEGIN
                ToLock_L := TRUE;
                MatifContractArbFarmer_L.VALIDATE(Status,MatifContractArbFarmer_L.Status::Open);
                MatifContractArbFarmer_L.MODIFY;
              END;
              MatifContractArbFarmer_L.VALIDATE("Daily Price", "Amount Daily Prices to Update");
              IF ToLock_L THEN BEGIN
                MatifContractArbFarmer_L.VALIDATE(Status,MatifContractArbFarmer_L.Status::Locked);
              END;
              MatifContractArbFarmer_L.MODIFY;
              CounterFarmer_L := CounterFarmer_L + 1;
            UNTIL MatifContractArbFarmer_L.NEXT = 0;

            "To Update" := FALSE;
            "Updated Contract Number" := CounterFarmer_L;
            MODIFY;
          UNTIL NEXT = 0;
        END;
      END;
    END;

    PROCEDURE InitCotation@100000002(Filename_P@100000000 : Text[1000]);
    VAR
      NewEdgeSAS_L@100000001 : Record 60111;
      OldPFC_L@100000003 : Text[50];
      NewCotation_L@100000004 : Boolean;
      UpdateMatifPriceContracts_L@100000005 : Record 60054;
    BEGIN
      // --> 20191202-WI00-ADV01.01.00-3444-MRU01
      //R'initialisation des cotations
      OldPFC_L := '';

      WITH NewEdgeSAS_L DO BEGIN
        RESET;
        SETCURRENTKEY(PFC, PCTYM);
        SETRANGE(FileName, Filename_P);
        SETRANGE(PRECID, 'P');
        SETRANGE(PSUBTY, '');
        SETRANGE(PSTYPE, 'F');
        IF FINDFIRST THEN REPEAT
          NewCotation_L := PFC <> OldPFC_L;
          IF NewCotation_L THEN BEGIN
            //Code espSce Matif : g'r' par la table de correspondance
            UpdateMatifPriceContracts_L.RESET;
            UpdateMatifPriceContracts_L.SETRANGE("Matif Species Code", MoteurInterface_G.Getcorrespondance(GlobalTemplateCode_G
                                                                , 'CMC MATIF Species', 'Species Code', PFC, TRUE, TRUE));
            UpdateMatifPriceContracts_L.SETRANGE("INCOGRAIN Code", '');
            IF UpdateMatifPriceContracts_L.FINDSET THEN REPEAT
              UpdateMatifPriceContracts_L.VALIDATE("Amount Daily Prices to Update", 0);
              UpdateMatifPriceContracts_L.MODIFY;
            UNTIL UpdateMatifPriceContracts_L.NEXT = 0;
          END;

          OldPFC_L := PFC;
        UNTIL NEXT = 0;
      END;
      // <-- 20191202-WI00-ADV01.01.00-3444-MRU01
    END;
