      IF WorkPath = '' THEN
        EXIT(FALSE);

      PathLowerCase := LOWERCASE(WorkPath);

      // ==================================================================
      // HTTPS
      // ==================================================================
      IF STRPOS(PathLowerCase,'https:') = 1 THEN BEGIN
        WorkPath := COPYSTR(WorkPath,7);
        WorkPath := DELCHR(WorkPath,'<','/\');
        WorkPath := 'https://' + WorkPath;
        WorkPath := CONVERTSTR(WorkPath,'\','/');
        Path := WorkPath;
        EXIT(TRUE);
      END;

      // ==================================================================
      // HTTP
      // ==================================================================
      IF STRPOS(PathLowerCase,'http:') = 1 THEN BEGIN
        WorkPath := COPYSTR(WorkPath,6);

        WorkPath := 'http://' + WorkPath;
        WorkPath := CONVERTSTR(WorkPath,'\','/');
        Path := WorkPath;
        EXIT(TRUE);
      END;

      // FTP
      // ==================================================================

        WorkPath := COPYSTR(WorkPath,5);
        WorkPath := DELCHR(WorkPath,'<','/\');
        WorkPath := 'ftp://' + WorkPath;
        WorkPath := CONVERTSTR(WorkPath,'\','/');
        Path := WorkPath;
        EXIT(TRUE);
      END;

      // ==================================================================
      // FILE
      // ==================================================================
      IF STRPOS(PathLowerCase,'file:') = 1 THEN BEGIN
        WorkPath := COPYSTR(WorkPath,6);
        WorkPath := DELCHR(WorkPath,'<','/');

        IF COPYSTR(WorkPath,1,2) = '\\' THEN
          WorkPath := COPYSTR(WorkPath,3);
        WorkPath := CONVERTSTR(WorkPath,'\','/');
        Path := 'file://' + WorkPath;

      // UNC d'j. correct

      // ==================================================================

          EXIT(FALSE);
        WorkPath := CONVERTSTR(WorkPath,'/','\');
        Path := '\\' + WorkPath;
        EXIT(TRUE);
      END;

      // ==================================================================
      // URL sans protocole
      // ==================================================================
      IF (STRPOS(WorkPath,'.') > 0) AND (STRPOS(WorkPath,' ') = 0) THEN BEGIN
        Path := 'http://' + WorkPath;
        EXIT(TRUE);
      END;
