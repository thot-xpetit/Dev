OBJECT Codeunit 50057 MAJ Fournisseurs apporteurs UN
{
  OBJECT-PROPERTIES
  {
    Date=17/05/17;
    Heure=10:07:24;
    Modifi'=Oui;
    Liste versions=REP001;
  }
  PROPERTIES
  {
    OnRun=BEGIN

            //===============================================================================================================================
            //                      Mises . jour Fournisseurs
            //===============================================================================================================================

            Cpt  := 0;
            Cpt2 := 0;
            Vendor_G.RESET;

            Total := Vendor_G.COUNT;
            Fenetre.OPEN('Maj des fournisseurs          \' +
                         'Nb lecture                      #1######\' +
                         'Nb flux modifi'                 #2######\' +
                         '@4@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@',Cpt,Cpt2);

            Vendor_G.SETRANGE(Vendor_G."Contributor Vendor",TRUE);
            Vendor_G.SETFILTER(Vendor_G."Customer No.", '');

            IF Vendor_G.FIND('-') THEN
            REPEAT
              IF COMPANYNAME = 'UNEAL' THEN BEGIN
                IF Contact_G.GET (Vendor_G."No.") THEN BEGIN
                  IF Contact_G."Contact Type" = Contact_G."Contact Type"::Farm THEN BEGIN
                    IF Vendor_G."Customer No." = '' THEN BEGIN
                      Vendor_G."Customer No." :=  Vendor_G."No." ;
                      Vendor_G.MODIFY;
                      Cpt2 := Cpt2 + 1;
                    END;
                  END;
                END;
              END;

            Cpt := Cpt + 1;
            Fenetre.UPDATE;
            Fenetre.UPDATE(1,Cpt);
            Fenetre.UPDATE(2,Cpt2);
            Fenetre.UPDATE(4,ROUND(Cpt / Total * 10000, 1));

            UNTIL Vendor_G.NEXT = 0;
            Fenetre.CLOSE;
          END;

  }
  CODE
  {
    VAR
      Total@1000000004 : Integer;
      Cpt@1000000005 : Integer;
      Cpt2@1000000006 : Integer;
      Cpt3@1000000022 : Integer;
      Fenetre@1000000007 : Dialog;
      Ok_lancement@1000000013 : Boolean;
      Vendor_G@1000000002 : Record 23;
      Contact_G@1000000003 : Record 5050;

    BEGIN
    {

    }
    END.
  }
