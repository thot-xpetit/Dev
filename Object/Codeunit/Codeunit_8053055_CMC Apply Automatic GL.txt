    PROCEDURE ApplyAutoGLLedgEntryAmt@1000000002(GLAccNo_P@1000000000 : Code[20];PostApply_P@1000000001 : Boolean;DatePostDue_P@1000000002 : 'Posting Date,Due Date');
    VAR
      GLEntry_L@1000000003 : Record 17;
      GLEntryTA_L@1000000004 : Record 17;
    BEGIN
      GLEntry_L.RESET;
      GLEntry_L.SETCURRENTKEY("G/L Account No.",Applied,Amount,"Due Date");
      GLEntry_L.SETRANGE("G/L Account No.",GLAccNo_P);
      GLEntry_L.SETRANGE(Applied,FALSE);
      GLEntry_L.SETFILTER(Amount,'<0');
      GLEntry_L.SETRANGE("Applies-to ID",'');
      IF GLEntry_L.FINDFIRST THEN BEGIN
        REPEAT
          //Search only one document to apply
          GLEntryTA_L.RESET;
          GLEntryTA_L.SETCURRENTKEY("G/L Account No.",Applied,Amount,"Due Date");
          GLEntryTA_L.SETRANGE("G/L Account No.",GLAccNo_P);
          GLEntryTA_L.SETRANGE(Applied,FALSE);
          GLEntryTA_L.SETRANGE(Amount,-GLEntry_L.Amount);
          GLEntryTA_L.SETRANGE("Applies-to ID",'');
          IF GLEntryTA_L.FINDFIRST AND (GLEntryTA_L.COUNT = 1) THEN BEGIN
            //Save values on entry and entry to apply
            GLEntry_L."Applies-to ID":=FORMAT(GLEntry_L."Entry No.");
            GLEntry_L.MODIFY;

            GLEntryTA_L."Applies-to ID" := FORMAT(GLEntry_L."Entry No.");
            GLEntryTA_L.MODIFY;

            // --> 20140903-FINA23/CMC02.02.00-2478-BDN01
            IF PostApply_P THEN BEGIN
              CreateAppliedEntriesHist(0,GLEntry_L."Entry No.");
              CreateAppliedEntriesHist(0,GLEntryTA_L."Entry No.");
            END;
            // <-- 20140903-FINA23/CMC02.02.00-2478-BDN01

            //To post ?
            IF PostApply_P THEN
              GLEntriesApplication_G.Validate(GLEntryTA_L,FALSE);

          END;

    END;

    PROCEDURE ApplyAutoGLLedgEntryBal@1000000003(GLAccNo_P@1000000002 : Code[20];PostApply_P@1000000001 : Boolean;DatePostDue_P@1000000000 : 'Posting Date,Due Date');
    VAR
      GLEntry_L@1000000003 : Record 17;
      Balance_L@1000000005 : Decimal;
      EntryQty_L@1000000004 : Integer;
    BEGIN
      GLEntry_L.RESET;
      CASE DatePostDue_P OF
        DatePostDue_P::"Posting Date" : GLEntry_L.SETCURRENTKEY("G/L Account No.",Applied,"Posting Date");
        DatePostDue_P::"Due Date" : GLEntry_L.SETCURRENTKEY("G/L Account No.",Applied,"Due Date");
      END;
      GLEntry_L.SETRANGE("G/L Account No.",GLAccNo_P);
      GLEntry_L.SETRANGE(Applied,FALSE);
      GLEntry_L.SETRANGE("Applies-to ID",'');
      IF GLEntry_L.FINDFIRST THEN BEGIN
        Balance_L := 0;
        EntryQty_L := 0;
        GLEntryTMP_G.DELETEALL;
        REPEAT
          Balance_L := Balance_L + GLEntry_L.Amount;
          EntryQty_L := EntryQty_L + 1;
          GLEntryTMP_G.INIT;
          GLEntryTMP_G.TRANSFERFIELDS(GLEntry_L);
          GLEntryTMP_G.INSERT;
          IF (Balance_L = 0) AND (EntryQty_L > 1) THEN BEGIN
            ApplyGroup(PostApply_P);
            EntryQty_L := 0;
            GLEntryTMP_G.DELETEALL;
          END;
        UNTIL GLEntry_L.NEXT=0;
      END;
    END;
