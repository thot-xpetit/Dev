    VAR

    BEGIN
      WITH CompensLine_G DO
       BEGIN
        SETRANGE("Capital Journal Template","Capital Journal Template");
        SETRANGE("Capital Batch Name","Capital Batch Name");
        IF RECORDLEVELLOCKING THEN
          LOCKTABLE;
        SetOffJnlTemplate_G.GET("Capital Journal Template");
        SetOffJnlBatch_G.GET("Capital Journal Template","Capital Batch Name");

        IF NOT FIND('=><') THEN BEGIN
          "Line No." := 0;
          COMMIT;
          EXIT;
         END;
        IF NOT G_WinDialog THEN BEGIN
          IF ShowDialog() THEN BEGIN
            Window_G.OPEN(Text8053001 + Text8053002 + Text8053007);
            Window_G.UPDATE(1,"Capital Batch Name");
           END;
         END;

      // check lines
        LineCount_G := 0;
        StartLineNo_G := "Line No.";
        REPEAT
          LineCount_G := LineCount_G + 1;
          IF NOT G_WinDialog THEN
            IF ShowDialog() THEN
              Window_G.UPDATE(2,LineCount_G);
          CompensLine_G.TESTFIELD("Posting Date");
          CompensLine_G.TESTFIELD(CompensLine_G."Account Type");
          CompensLine_G.TESTFIELD(CompensLine_G."Account No.");
          CompensLine_G.TESTFIELD(CompensLine_G."Document No.");
          CheckEntryData(CompensLine_G);
          IF NEXT = 0 THEN
            FINDFIRST;
        UNTIL "Line No." = StartLineNo_G;
        NoOfRecords_G := LineCount_G;

      // Check Doc No
        LineCount_G := 0;
        FINDSET(TRUE,FALSE);
        REPEAT
          LineCount_G := LineCount_G + 1;
          IF NOT EmptyCompLine THEN
            IF (SetOffJnlBatch_G."No. Series" <> '') AND ("Document No." <> LastDocNo_G) THEN
              TESTFIELD("Document No.",NoSeriesMgt_G.GetNextNo(SetOffJnlBatch_G."No. Series","Posting Date",FALSE));
        UNTIL NEXT = 0;

        // Post lines
        LineCount_G := 0;
        LastDocNo_G := '';
        LastPostedDocNo_G := '';
        FINDSET(TRUE,FALSE);
        REPEAT
          IF NOT CompensLine_G.EmptyCompLine THEN BEGIN
            CompensLine3_G :=  CompensLine_G;
            WITH  CompensLine3_G DO
            BEGIN
              LineCount_G := LineCount_G + 1;

              IF NOT G_WinDialog THEN BEGIN
                IF ShowDialog() THEN BEGIN
                 Window_G.UPDATE(5,LineCount_G);
                 Window_G.UPDATE(6,ROUND(LineCount_G / NoOfRecords_G * 10000,1));
                END;
              END;
              CheckDocumentNo(CompensLine3_G);

              SetOffJnlPostLine_G.RUN(CompensLine3_G);
              // --> 20150429-FINA13-CMC02.02.00-2956-FHS01
              IF CompensLine3_G."Account Type" = CompensLine3_G."Account Type"::Adh'rent THEN
                IF CompensLine3_G."Account No." <> '' THEN BEGIN
                  IF NOT(recContactTmp_L.GET(CompensLine3_G."Account No.")) THEN BEGIN
                    recContactTmp_L.INIT;
                    recContactTmp_L."No." := CompensLine3_G."Account No.";
                    IF recContactTmp_L.INSERT THEN ;
                  END;
                END;
              // <-- 20150429-FINA13-CMC02.02.00-2956-FHS01
            END;
          END;
        UNTIL NEXT = 0;
        // Update/delete lines

          IF NOT RECORDLEVELLOCKING THEN
            LOCKTABLE(TRUE,TRUE);
            SetOffJnlLine2_G.COPY(CompensLine_G);
            SetOffJnlLine2_G.SETFILTER("Account No.",'<>%1','');
            IF SetOffJnlLine2_G.FINDLAST THEN; // Remember the last line
            CompensLine3_G.COPY(CompensLine_G);

            IF CompensLine3_G.FINDSET(TRUE,FALSE) THEN
              REPEAT
                   CompensLine3_G.DELETE(TRUE);
                    CompJnlLine_L.RESET;
                    CompJnlLine_L.SETRANGE("Account No.",CompensLine3_G."Account No.");
                    CompJnlLine_L.SETRANGE(Indentation,1);
                    CompJnlLine_L.SETFILTER("Line No.",'<>%1',CompensLine3_G."Line No.");
                    IF NOT CompJnlLine_L.FINDFIRST THEN BEGIN
                      CompJnlLine_L.SETRANGE(Indentation,0);
                      CompJnlLine_L.DELETEALL;
                    END
              UNTIL CompensLine3_G.NEXT = 0;
            CompensLine3_G.RESET;
            CompensLine3_G.SETRANGE("Capital Journal Template","Capital Journal Template");
            CompensLine3_G.SETRANGE("Capital Batch Name","Capital Batch Name");
            IF NOT CompensLine3_G.FINDLAST THEN
          IF INCSTR("Capital Batch Name") <> '' THEN BEGIN
                SetOffJnlBatch_G.DELETE;
                SetOffJnlBatch_G.Name := INCSTR("Capital Batch Name");
                IF SetOffJnlBatch_G.INSERT THEN;
                "Capital Batch Name" := SetOffJnlBatch_G.Name;
               END;

            CompensLine3_G.SETRANGE("Capital Batch Name","Capital Batch Name");
        IF (SetOffJnlBatch_G."No. Series" = '') AND NOT CompensLine3_G.FINDLAST THEN BEGIN
              CompensLine3_G.INIT;
              CompensLine3_G."Capital Journal Template" := "Capital Journal Template";
              CompensLine3_G."Capital Batch Name" := "Capital Batch Name";
              CompensLine3_G."Line No." := 10000;
              CompensLine3_G.INSERT;
             END;

        IF SetOffJnlBatch_G."No. Series" <> '' THEN
          NoSeriesMgt_G.SaveNoSeries;
        IF NoSeries_G.FINDSET THEN
          REPEAT
            EVALUATE(PostingNoSeriesNo_G,NoSeries_G.Description);
            NoSeriesMgt2[PostingNoSeriesNo_G].SaveNoSeries;
          UNTIL NoSeries_G.NEXT = 0;
        // --> 20150430-FINA13-CMC02.02.00-2956-FHS01
        recContactTmp_L.RESET;
        IF recContactTmp_L.FINDSET(FALSE, FALSE) THEN
          REPEAT
            cduLedgerMan_L.RefreshCVLetter3(recContactTmp_L."No.");
          UNTIL recContactTmp_L.NEXT = 0;
        recContactTmp_L.DELETEALL;
        // <-- 20150430-FINA13-CMC02.02.00-2956-FHS01
        COMMIT;

        CLEAR(SetOffJnlPostLine_G);
      END;
    END;

    PROCEDURE CheckDocumentNo@11(VAR CapJnlLine2_G@8053000 : Record 8053501);
    BEGIN
      WITH CapJnlLine2_G DO
        BEGIN
         IF "Posting No. Series" = '' THEN
            "Posting No. Series" := SetOffJnlBatch_G."No. Series"
         ELSE
          IF NOT EmptyCompLine THEN
            IF "Document No." = LastDocNo_G THEN
              "Document No." := LastPostedDocNo_G
            ELSE BEGIN
              IF NOT NoSeries_G.GET("Posting No. Series") THEN BEGIN
                NoOfPostingNoSeries_G := NoOfPostingNoSeries_G + 1;
                IF NoOfPostingNoSeries_G > ARRAYLEN(NoSeriesMgt2) THEN
                  ERROR(Text8053025, ARRAYLEN(NoSeriesMgt2));
                NoSeries_G.Code := "Posting No. Series";
                NoSeries_G.Description := FORMAT(NoOfPostingNoSeries_G);
                NoSeries_G.INSERT;
               END;
              LastDocNo_G := "Document No.";
              EVALUATE(PostingNoSeriesNo_G,NoSeries_G.Description);
              "Document No." := NoSeriesMgt2[PostingNoSeriesNo_G].GetNextNo("Posting No. Series","Posting Date",TRUE);
              LastPostedDocNo_G := "Document No.";
             END;
      END;
    END;

    PROCEDURE CheckEntryData@8053004(CompJournalLine_P@8053000 : Record 8053501);
    VAR
      CustLedgerEntry_L@8053001 : Record 21;
      VendLedgerEntry_L@8053002 : Record 25;
      RestraintEntry_L@8053003 : Record 8053374;
      CompEntryProp_L@8053004 : Record 8053041;
      TEXT0001_L@8053005 : TextConst 'ENU="The Set-Off Amount %1 is greater than the remaining amount of %2 %3 on document %4. ";FRA=Le montant de compensation %1 est plus grand que le montant ouvert %2 %3 au niveau %4';
      CompEntryProp2_L@8053007 : Record 8053041;
      Contact_L@8053006 : Record 5050;
    BEGIN
      // check  Invoice Amount
      Contact_L.GET(CompJournalLine_P."Account No.");
      VendLedgerEntry_L.RESET;
      VendLedgerEntry_L.SETCURRENTKEY("Document No.","Document Type","Vendor No.");
      VendLedgerEntry_L.SETRANGE("Document No.",CompJournalLine_P."Farmer Document No.");
      VendLedgerEntry_L.SETRANGE("Vendor No.",Contact_L."Vendor No.");
      VendLedgerEntry_L.SETRANGE(Open,TRUE);
      VendLedgerEntry_L.FINDFIRST;
      VendLedgerEntry_L.CALCFIELDS("Remaining Amount");
      IF ABS(VendLedgerEntry_L."Remaining Amount") < CompJournalLine_P."Vendor Balance" THEN
        // --> 20140911-FINA28-CMC02.02.00-BRA01
        //BEFORE
        //ERROR(TEXT0001_L,CompJournalLine_P."Vendor Balance",VendLedgerEntry_L."Entry No.", VendLedgerEntry_L.TABLECAPTION);
        //AFTER
        ERROR(TEXT0001_L,CompJournalLine_P."Vendor Balance",VendLedgerEntry_L."Entry No.", VendLedgerEntry_L.TABLECAPTION,
        CompJournalLine_P."Document No.");
        // <-- 20140911-FINA28-CMC02.02.00-BRA01

      CompEntryProp_L.RESET;
      CompEntryProp_L.SETRANGE("SettOff Type",CompEntryProp_L."SettOff Type"::Proposed);
      CompEntryProp_L.SETRANGE("Invoice No",CompJournalLine_P."Farmer Document No.");
      IF CompEntryProp_L.FINDSET THEN
        REPEAT
          CASE CompEntryProp_L."Entry Type" OF
            CompEntryProp_L."Entry Type"::Vendor :
            BEGIN
              VendLedgerEntry_L.GET(CompEntryProp_L."Entry No.");
              VendLedgerEntry_L.CALCFIELDS("Remaining Amount");
              CompEntryProp2_L.RESET;
              CompEntryProp2_L.SETCURRENTKEY("SettOff Type","Entry Type","Entry No.","Document No.","Invoice No");
              CompEntryProp2_L.SETRANGE("SettOff Type",CompEntryProp_L."SettOff Type"::Proposed);
              CompEntryProp2_L.SETRANGE("Entry Type",CompEntryProp2_L."Entry Type"::Vendor);
              CompEntryProp2_L.SETRANGE("Entry No.",CompEntryProp_L."Entry No.");
              CompEntryProp2_L.CALCSUMS(Amount);
              IF ABS(VendLedgerEntry_L."Remaining Amount") < CompEntryProp2_L.Amount THEN
                // --> 20140911-FINA28-CMC02.02.00-BRA01
                //BEFORE
                //ERROR(TEXT0001_L,CompEntryProp2_L.Amount,VendLedgerEntry_L."Remaining Amount",VendLedgerEntry_L.TABLENAME);
                //AFTER
                ERROR(TEXT0001_L,CompEntryProp2_L.Amount,VendLedgerEntry_L."Remaining Amount",VendLedgerEntry_L.TABLECAPTION,
                CompJournalLine_P."Document No.");
                // <-- 20140911-FINA28-CMC02.02.00-BRA01
            END;
            CompEntryProp_L."Entry Type"::Customer :
            BEGIN
              CustLedgerEntry_L.GET(CompEntryProp_L."Entry No.");
              CustLedgerEntry_L.CALCFIELDS("Remaining Amount");
              CompEntryProp2_L.RESET;
              CompEntryProp2_L.SETCURRENTKEY("SettOff Type","Entry Type","Entry No.","Document No.","Invoice No");
              CompEntryProp2_L.SETRANGE("SettOff Type",CompEntryProp_L."SettOff Type"::Proposed);
              CompEntryProp2_L.SETRANGE("Entry Type",CompEntryProp2_L."Entry Type"::Customer);
              CompEntryProp2_L.SETRANGE("Entry No.",CompEntryProp_L."Entry No.");
              CompEntryProp2_L.CALCSUMS(Amount);
              IF CustLedgerEntry_L."Remaining Amount" < CompEntryProp2_L.Amount THEN
                // --> 20140911-FINA28-CMC02.02.00-BRA01
                //BEFORE
                //ERROR(TEXT0001_L,CompEntryProp2_L.Amount,VendLedgerEntry_L."Remaining Amount",CustLedgerEntry_L.TABLENAME);
                //AFTER
                ERROR(TEXT0001_L,CompEntryProp2_L.Amount,CustLedgerEntry_L."Remaining Amount",CustLedgerEntry_L.TABLECAPTION,
                CompJournalLine_P."Document No.");
                // <-- 20140911-FINA28-CMC02.02.00-BRA01
            END;
            CompEntryProp_L."Entry Type"::Restraint :
            BEGIN
              RestraintEntry_L.GET(CompEntryProp_L."Entry No.");
              RestraintEntry_L.CALCFIELDS("Remaining Amount");
              CompEntryProp2_L.RESET;
              CompEntryProp2_L.SETCURRENTKEY("SettOff Type","Entry Type","Entry No.","Document No.","Invoice No");
              CompEntryProp2_L.SETRANGE("SettOff Type",CompEntryProp_L."SettOff Type"::Proposed);
              CompEntryProp2_L.SETRANGE("Entry Type",CompEntryProp2_L."Entry Type"::Restraint);
              CompEntryProp2_L.SETRANGE("Entry No.",CompEntryProp_L."Entry No.");
              CompEntryProp2_L.CALCSUMS(Amount);
              IF RestraintEntry_L."Remaining Amount" < CompEntryProp2_L.Amount THEN
                // --> 20140911-FINA28-CMC02.02.00-BRA01
                //BEFORE
                //ERROR(TEXT0001_L,CompEntryProp2_L.Amount,VendLedgerEntry_L."Remaining Amount",RestraintEntry_L.TABLENAME);
                //AFTER
                ERROR(TEXT0001_L,CompEntryProp2_L.Amount,RestraintEntry_L."Remaining Amount",RestraintEntry_L.TABLECAPTION,
                CompJournalLine_P."Document No.");
                // <-- 20140911-FINA28-CMC02.02.00-BRA01
            END;
          END;
        UNTIL CompEntryProp_L.NEXT = 0 ;
    END;
