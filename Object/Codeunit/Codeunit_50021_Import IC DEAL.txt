    PROCEDURE ProcessFile@1000000002();
    VAR
      FileName_L@1000000004 : Text[1024];
      BackupFileName_L@1000000006 : Text[1024];
      RecIcOutboxTrans_L@1000000000 : Record 414;
      RecIcOutboxTrans2_L@1000000001 : Record 414;

      Text8053001@1000000003 : TextConst 'ENU=There are errors in the IC Inbox Transaction.;FRA=Il y a des erreurs dans la boOte de r'ception IC.';
      i_L@1000000005 : Integer;
    BEGIN
      ErrorOccured_G := FALSE;
      // affectation des variables format
      TxtFormatDate_G := '<Day,2><Month,2><Year4>';
      TxtDecimalFormat_G := '<Integer><Decimals><Comma,.>';

      FileName_L := RecInterfaceEngine_G."File in use";

      IF FileImport_G.OPEN(FileName_L) THEN BEGIN
        RecICCollRecSetup_G.SETRANGE("Connector Name" ,RecInterfaceEngine_G."Connector Name");
        IF NOT RecICCollRecSetup_G.FINDFIRST THEN
          ERROR(STRSUBSTNO(Text0003,RecInterfaceEngine_G."Connector Name"));
        // Recherche les informations du partenaire emetteur
        RecICPartner_G.GET(RecICCollRecSetup_G."IC Partner");
        IF  RecICPartner_G."Cereal Flow" <> RecICPartner_G."Cereal Flow"::"File Location" THEN
          ERROR(STRSUBSTNO(Text0005, RecICCollRecSetup_G."IC Partner", RecICPartner_G."Cereal Flow"))
        ELSE BEGIN
          FileImport_G.CREATEINSTREAM(FileImportInStream_G);
          WHILE NOT FileImportInStream_G.EOS DO BEGIN
            FileImportInStream_G.READTEXT(LineRead_G);
            ProcessLineMessage(LineRead_G);
            CreateOutbox;
          END;
          FileImport_G.CLOSE;
          CreateInbox;
        END;
      END;
      // d'place le fichier dans le r'pertoire des logs
      i_L := 1;
      WHILE i_L <> 0 DO BEGIN
        BackupFileName_L := STRSUBSTNO('%1\%2_%3_%4_%5.%6',RecInterfaceEngine_G."Folder Log Error" ,
                                RecInterfaceEngine_G."File Prefix",RecICPartner_G.Code,
                                FORMAT(TODAY,0,TxtFormatDate_G), i_L ,RecInterfaceEngine_G."File extension");
        IF EXISTS(BackupFileName_L) THEN
          i_L := i_L + 1
        ELSE
          i_L := 0;
      END;
      IF FILE.COPY(FileName_L,BackupFileName_L) THEN
        FILE.ERASE(FileName_L)
      ELSE
        ERROR(Text0007, FileName_L, BackupFileName_L);
    END;

    PROCEDURE ReturnTextPos@1000000004(Text_P@1000000000 : Text[1024];Separator_P@1000000008 : Text[1];SubStringPos_P@1000000001 : Integer) : Text[1024];
    VAR
      TextFound_L@1000000002 : Text[1024];
      BeginPosition_L@1000000003 : Integer;
      EndPosition_L@1000000004 : Integer;
      BeginPosFound_L@1000000006 : Boolean;
      EndPositionFound_L@1000000005 : Boolean;
      CurrentPos_L@1000000007 : Integer;
      CurrentSubString_L@1000000009 : Integer;
    BEGIN
      //-------------------------------------------------------------------------------------------------------------------
      // renvoie le texte . la position indiqu'e par le paramStre
      //----------------------------------------------------------------------------------------------------------
      // Entr'es:
      //   - Text_P         : La chaine dans laquelle on cherche
      //   - Separator_P    : Le separateur
      //   - SubStringPos_P : La position du texte dans la chaine
      // Sorties:
      //   - TextFound_L    : le texte trouv'
      //----------------------------------------------------------------------------------------------------------
      CurrentPos_L := 1;
      CurrentSubString_L := 1;
      IF SubStringPos_P = 1 THEN BEGIN
        BeginPosition_L := 1;
        BeginPosFound_L := TRUE;
      END ELSE BEGIN
        REPEAT
          IF COPYSTR(Text_P,CurrentPos_L,1) = Separator_P THEN BEGIN
            IF CurrentSubString_L = SubStringPos_P-1 THEN BEGIN
              BeginPosition_L := CurrentPos_L+1;
              BeginPosFound_L := TRUE;
              CurrentPos_L += 1;
            END ELSE BEGIN
              CurrentSubString_L += 1;
              CurrentPos_L += 1;
            END;
          END ELSE
            CurrentPos_L += 1;
        UNTIL BeginPosFound_L OR (CurrentPos_L = STRLEN(Text_P));
      END;

      REPEAT
        IF COPYSTR(Text_P,CurrentPos_L,1) = Separator_P THEN BEGIN
          EndPosition_L := CurrentPos_L;
          EndPositionFound_L := TRUE;
        END ELSE
          CurrentPos_L += 1;
      UNTIL EndPositionFound_L OR (CurrentPos_L = STRLEN(Text_P));

      // si dernier champ ne se termine pas par le s'parateur mais qu'il existe quand meme des s'parateurs
      IF BeginPosFound_L AND NOT EndPositionFound_L AND (STRPOS(Text_P,Separator_P) <> 0) THEN BEGIN
        EndPosition_L := STRLEN(Text_P)+1;
        EndPositionFound_L := TRUE;
      END;

      IF BeginPosFound_L AND EndPositionFound_L AND (EndPosition_L > BeginPosition_L)THEN
        TextFound_L := COPYSTR(Text_P,BeginPosition_L,EndPosition_L- BeginPosition_L)
      ELSE
        TextFound_L := '';
      EXIT(TextFound_L);
    END;
