    PROCEDURE CheckIsMasterData@8053003(VAR RecRef_P@8053006 : RecordRef;VAR xRecRef_P@8053000 : RecordRef);
    VAR
      FieldRef_L@8053001 : FieldRef;
      i@8053002 : Integer;
      Field_l@8053003 : Record 2000000041;
      xFieldRef_L@8053004 : FieldRef;
      ChangeLogManagement_L@8053005 : Codeunit 423;
      RecMCDataSetup_L@8053008 : Record 8053077;
      BooControler_L@8053007 : Boolean;
    BEGIN
      //---------------------------------------------------------------------------------------
      // V'rifie si le champ standard modifi' correspond . un champ MS et retourne une erreur
      //---------------------------------------------------------------------------------------
      // Entr'es:
      //   - RecRef_P, xRecRef_P : enregistrement en cours de modification
      // Sorties:
      //   - erreur si le champ est MS
      //--------------------------------------------------------------------------------------
      IF IsMCCompany(RecRef_P.NUMBER) THEN
        IF GetMCTableID(RecRef_P.NUMBER,FALSE) <> 0 THEN BEGIN
          //Ne pas bloquer les tables MS pour le journal de modification
          BooControler_L := TRUE;
          IF RecRef_P.NUMBER = DATABASE::Contact THEN BEGIN
            RecMCDataSetup_L.GET;
            // Si gestion des tiers MS
            IF RecMCDataSetup_L."Active MC Third party" THEN BEGIN
              FieldRef_L := RecRef_P.FIELD(5050);
              IF FORMAT(FieldRef_L.VALUE) IN ['Personne','Person'] THEN BEGIN
                FieldRef_L := RecRef_P.FIELD(1);
                IF NOT IsMasterDataRelTiersMSExist(FORMAT(FieldRef_L.VALUE)) THEN
                  BooControler_L := FALSE;
              END;
            END;
            // Si gestion des contacts MS
            IF RecMCDataSetup_L."Active MC Contact" THEN
              BooControler_L := FALSE;
          END;
          IF ((RecRef_P.NUMBER) IN [DATABASE::"CMC MC Item",DATABASE::"CMC MC Location",DATABASE::"CMC MC G/L Account",
             DATABASE::"CMC MC Salesperson/Purchaser",DATABASE::"CMC MC Third party"])
          THEN
            BooControler_L := FALSE;

          IF BooControler_L THEN
            FOR i := 1 TO RecRef_P.FIELDCOUNT DO BEGIN
              FieldRef_L := RecRef_P.FIELDINDEX(i);
              xFieldRef_L := xRecRef_P.FIELDINDEX(i);
              IF IsNormalField(RecRef_P.NUMBER,FieldRef_L.NUMBER) THEN
                IF FORMAT(FieldRef_L.VALUE) <> FORMAT(xFieldRef_L.VALUE) THEN BEGIN
                  IF IsFieldActivateInMCTable(RecRef_P.NUMBER,FieldRef_L.NUMBER) THEN
                    ERROR(Text8053000,FieldRef_L.CAPTION);
                END;
            END;
        END;
    END;

    LOCAL PROCEDURE IsNormalField@8053017(TableID_P@8053000 : Integer;FieldID_P@8053001 : Integer) : Boolean;
    VAR
      Field_L@8053002 : Record 2000000041;
    BEGIN
      //---------------------------------------------------------------------------------------------------
      // V'rifie si le champ standard est de type Normal ou non pour la comparaison
      // (Exclusion des BLOB par exemple)
      //---------------------------------------------------------------------------------------------------
      // Entr'es:
      //   - TableID_P : ID table standard
      //   - FieldID_P : ID champ standard
      //
      // Sorties:
      //   - Bool'en : TRUE si le champ est de type Normal

    END;

    LOCAL PROCEDURE GetField@8053016(TableID_P@8053000 : Integer;FieldID_P@8053001 : Integer;VAR Field_P@8053002 : Record 2000000041);

      Field_L@8053003 : Record 2000000041;
    BEGIN
      IF NOT TempField_G.GET(TableID_P,FieldID_P) THEN BEGIN
        Field_L.GET(TableID_P,FieldID_P);

    END;

    LOCAL PROCEDURE IsFieldActivateInMCTable@8053015(TableID_P@8053000 : Integer;FieldID_P@8053001 : Integer) : Boolean;
    VAR
      Field_L@8053002 : Record 2000000041;
      MCTableID_L@8053003 : Integer;
    BEGIN
      //---------------------------------------------------------------------------------------
      // V'rifie si le champ est actif dans la table MS
      //---------------------------------------------------------------------------------------
      // Entr'es:
      //   - TableID_P : ID table MS
      //   - FieldID_P : ID champ MS
      // Sorties:
      //   - Bool'en : TRUE si le champ est actif
      //---------------------------------------------------------------------------------------
      MCTableID_L := GetMCTableID(TableID_P,FALSE);
      IF (MCTableID_L = DATABASE::Contact) AND (FieldID_P = 39) THEN
        EXIT(TRUE);

      IF Field_L.GET(MCTableID_L,FieldID_P) THEN BEGIN
        // Cas de la date dern. modification
        IF ((MCTableID_L = DATABASE::Contact) AND (FieldID_P = 54)) OR
          ((MCTableID_L = DATABASE::"CMC MC G/L Account") AND (FieldID_P = 26))
          //OR ((MCTableID_L = DATABASE::"CMC MC Item") AND (FieldID_P = 62)) //Le champ 62 n'est plus actif dans la table 8053079
        THEN
          EXIT(FALSE);
        GetField(MCTableID_L,FieldID_P,Field_L);
        EXIT(Field_L.Enabled = TRUE);
      END ELSE
        EXIT(FALSE);
    END;

    PROCEDURE GetMCTableID@8053014(TableID_P@8053000 : Integer;Detail_P@1000000000 : Boolean) MCTableID : Integer;
    VAR
      Rec_MCDataSetup_L@8053001 : Record 8053077;
    BEGIN
      //---------------------------------------------------------------------------------------
      // Retourne le N› de la table MS par rapport . une table
      //---------------------------------------------------------------------------------------
      // Entr'es:
      //   - TableID_P : ID table
      //   - Detail_P : True, renvoie la table 18 ou 23, False renvoie Tiers MS ou 5050
      // Sorties:
      //   - MCTableID : ID table MS
      //---------------------------------------------------------------------------------------
      CASE TableID_P OF
        13 : MCTableID := DATABASE::"CMC MC Salesperson/Purchaser";
        14 : MCTableID := DATABASE::"CMC MC Location";
        15 : MCTableID := DATABASE::"CMC MC G/L Account";
        18,23 :
        BEGIN
          Rec_MCDataSetup_L.GET;
          IF Rec_MCDataSetup_L."Active MC Third party" THEN
            IF Detail_P THEN BEGIN
              IF TableID_P = 18 THEN
                MCTableID := DATABASE::Customer
              ELSE
                MCTableID := DATABASE::Vendor;
            END ELSE
              MCTableID := DATABASE::"CMC MC Third party"
          ELSE
            MCTableID := DATABASE::Contact;
        END;
        27 : MCTableID := DATABASE::"CMC MC Item";
        287 : MCTableID := DATABASE::"CMC Contact Bank Account";
        288 : MCTableID := DATABASE::"CMC Contact Bank Account";
        5050 :
        BEGIN
          Rec_MCDataSetup_L.GET;
          IF Rec_MCDataSetup_L."Active MC Third party" THEN
            MCTableID := DATABASE::"CMC MC Third party"
          ELSE
            MCTableID := DATABASE::Contact;
        END;
        8053185 : MCTableID := DATABASE::"CMC MC Third party";
        5401 : MCTableID := DATABASE::"CMC MC Item Variant";
        5404 : MCTableID := DATABASE::"CMC MC Item Unit of Measure";
        8053024 : MCTableID := DATABASE::"CMC MC Item Act. Substance";
        8053078 : MCTableID := DATABASE::"CMC MC G/L Account";
        8053079 : MCTableID := DATABASE::"CMC MC Item";
        8053083 : MCTableID := DATABASE::"CMC MC Location";
        8053084 : MCTableID := DATABASE::"CMC MC Salesperson/Purchaser";
        8053132 : MCTableID := DATABASE::"CMC MC Item Risk Mention";
        8053206 : MCTableID := DATABASE::"CMC MC Production BOM Header";
        8053207 : MCTableID := DATABASE::"CMC MC Production BOM Line";
        99000771 : MCTableID := DATABASE::"CMC MC Production BOM Header";
        99000772 : MCTableID := DATABASE::"CMC MC Production BOM Line";
      END;

      EXIT(MCTableID);
    END;

    BEGIN
      //--------------------------------------------------------------------------------------------------------
      // Fonction de base du moteur de cr'ation des donn'es monosoci't's par rapport aux donn'es MS
      //--------------------------------------------------------------------------------------------------------
      // Entr'es:
      //   - RecRef_P : enregistrement MS . diffuser
      //   - CompanyName_P : Nom de la soci't' . diffuser
      //   - RecCompanyMC_P : Record de la table MS contenant la s'lection modSle client et fournisseur et diffuser client et fournisseur
      //
      // Sorties:
      //
      //--------------------------------------------------------------------------------------------------------
      IF RecRef_P.FINDSET THEN
        REPEAT
          FieldRef_L := RecRef_P.FIELD(1);
          CASE RecRef_P.NUMBER OF
            // Gestion de la table MC G/L Account
            DATABASE::"CMC MC G/L Account":
              BEGIN
                CLEAR(GLAccount_G);
                IF MCGLAccount_G.GET(FieldRef_L.VALUE) THEN BEGIN
                  IF CompanyName_P <> COMPANYNAME THEN
                    GLAccount_G.CHANGECOMPANY(CompanyName_P);
                  IF NOT GLAccount_G.GET(FieldRef_L.VALUE) THEN BEGIN
                    GLAccount_G.TRANSFERFIELDS(MCGLAccount_G);
                    GLAccount_G.INSERT;
                  END ELSE BEGIN
                    GLAccount_G.TRANSFERFIELDS(MCGLAccount_G);
                    GLAccount_G.MODIFY;
                  END;
                  InsertMasterDataRelation(FORMAT(FieldRef_L.VALUE),DATABASE::"CMC MC G/L Account",GetMCCompanyCode(CompanyName_P),
                    RecCompanyMS_P, '');
                END;
              END;
            // Gestion des tables MC Item + donn'es connexes
            DATABASE::"CMC MC Item":
            BEGIN
              CLEAR(Item_G);
              IF MCItem_G.GET(FieldRef_L.VALUE) THEN BEGIN
                IF CompanyName_P <> COMPANYNAME THEN
                  Item_G.CHANGECOMPANY(CompanyName_P);
                IF MCItem_G."Kit BOM No." <> '' THEN BEGIN
                  ProdBOMHeader_L.CHANGECOMPANY(CompanyName_P);
                  IF NOT ProdBOMHeader_L.GET(MCItem_G."Kit BOM No.") THEN
                    ERROR(Text8053016,MCItem_G."Kit BOM No.",CompanyName_P);
                END;
                IF MCItem_G."Kit Disassembly BOM No." <> '' THEN BEGIN
                  ProdBOMHeader_L.CHANGECOMPANY(CompanyName_P);
                  IF NOT ProdBOMHeader_L.GET(MCItem_G."Kit Disassembly BOM No.") THEN
                    ERROR(Text8053016,MCItem_G."Kit Disassembly BOM No.",CompanyName_P);
                END;
                IF NOT Item_G.GET(FieldRef_L.VALUE) THEN BEGIN
                  Item_G.TRANSFERFIELDS(MCItem_G);
                  // --> 20160630-WI00-ADV01.01.00-1761-JCD01
                  // --> 20160727-WI00-ADV01.01.00-1761-VMR01
                  //BEFORE
                  //Item_G."Blocked For Sales" := FALSE;
                  //AFTER
                  Item_G."Blocked For Sales" := TRUE;
                  // <-- 20160727-WI00-ADV01.01.00-1761-VMR01
                  // <-- 20160630-WI00-ADV01.01.00-1761-JCD01
                  Item_G.INSERT;
                END ELSE BEGIN
                  Item_G.TRANSFERFIELDS(MCItem_G);
                  Item_G.MODIFY;
                END;

                IF Item_G."Production BOM No." = '' THEN BEGIN
                  Item_G."Production BOM No." := Item_G."Kit BOM No.";

                END;

                InsertMasterDataRelation(FORMAT(FieldRef_L.VALUE),DATABASE::"CMC MC Item",GetMCCompanyCode(CompanyName_P),

                RecRef_L.SETPOSITION(Item_G.GETPOSITION);
                cduCMCMCSyncCRM_L.SyncInsertItem(RecRef_L, RecCompanyMS_P.Code, RecCompanyMS_P."Company Name");
              END;
              CLEAR(ItemActSubstance_G);
              MCItemActSubstance_G.RESET;
              MCItemActSubstance_G.SETRANGE(MCItemActSubstance_G."MC Item No.",FORMAT(FieldRef_L.VALUE));
              IF MCItemActSubstance_G.FINDSET THEN BEGIN
                REPEAT
                  IF CompanyName_P <> COMPANYNAME THEN
                    ItemActSubstance_G.CHANGECOMPANY(CompanyName_P);
                  IF NOT ItemActSubstance_G.GET(MCItemActSubstance_G."MC Item No.",MCItemActSubstance_G."Item Variant Code",
                    MCItemActSubstance_G."CAS Code")
                  THEN BEGIN
                    ItemActSubstance_G.TRANSFERFIELDS(MCItemActSubstance_G);
                    ItemActSubstance_G.INSERT;
                  END ELSE BEGIN
                    ItemActSubstance_G.TRANSFERFIELDS(MCItemActSubstance_G);
                    ItemActSubstance_G.MODIFY;
                  END;
                UNTIL MCItemActSubstance_G.NEXT=0;
              END;
              CLEAR(ItemUnitOfMeasure_G);
              MCItemUnitOfMeasure_G.RESET;
              MCItemUnitOfMeasure_G.SETRANGE(MCItemUnitOfMeasure_G."MC Item No.",FORMAT(FieldRef_L.VALUE));
              IF MCItemUnitOfMeasure_G.FINDSET THEN BEGIN
                REPEAT
                  IF CompanyName_P <> COMPANYNAME THEN
                    ItemUnitOfMeasure_G.CHANGECOMPANY(CompanyName_P);
                  IF NOT ItemUnitOfMeasure_G.GET(MCItemUnitOfMeasure_G."MC Item No.",MCItemUnitOfMeasure_G.Code) THEN BEGIN
