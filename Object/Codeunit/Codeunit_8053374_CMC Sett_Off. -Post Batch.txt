    PROCEDURE Code@8053002();
    BEGIN
      WITH CompensLine_G DO BEGIN
        SETRANGE("Capital Journal Template","Capital Journal Template");
        SETRANGE("Capital Batch Name","Capital Batch Name");
        IF RECORDLEVELLOCKING THEN
          LOCKTABLE;
        SetOffJnlTemplate_G.GET("Capital Journal Template");
        SetOffJnlBatch_G.GET("Capital Journal Template","Capital Batch Name");

        IF NOT FIND('=><') THEN BEGIN
          "Line No." := 0;
          COMMIT;
          EXIT;
        END;
        IF NOT G_WinDialog THEN BEGIN
          IF ShowDialog() THEN BEGIN
            Window_G.OPEN(Text8053001 + Text8053002 + Text8053007);
            Window_G.UPDATE(1,"Capital Batch Name");
          END;
        END;

        // check lines
        LineCount_G := 0;
        StartLineNo_G := "Line No.";
        REPEAT
          LineCount_G := LineCount_G + 1;
          IF NOT G_WinDialog THEN
            IF ShowDialog() THEN
              Window_G.UPDATE(2,LineCount_G);
          CompensLine_G.TESTFIELD("Posting Date");
          CompensLine_G.TESTFIELD(CompensLine_G."Account Type");
          CompensLine_G.TESTFIELD(CompensLine_G."Account No.");
          CompensLine_G.TESTFIELD(CompensLine_G."Document No.");
          IF NEXT = 0 THEN
            FINDFIRST;
        UNTIL "Line No." = StartLineNo_G;
        NoOfRecords_G := LineCount_G;

        // Check Doc No
        LineCount_G := 0;
        FINDSET(TRUE,FALSE);
        REPEAT
          LineCount_G := LineCount_G + 1;
          IF NOT EmptyCompLine THEN
            IF (SetOffJnlBatch_G."No. Series" <> '') AND ("Document No." <> LastDocNo_G) THEN
              TESTFIELD("Document No.",NoSeriesMgt_G.GetNextNo(SetOffJnlBatch_G."No. Series","Posting Date",FALSE));
        UNTIL NEXT = 0;

        // Post lines
        LineCount_G := 0;
        LastDocNo_G := '';
        LastPostedDocNo_G := '';
        FINDSET(TRUE,FALSE);
        REPEAT
          IF NOT CompensLine_G.EmptyCompLine THEN BEGIN
            CompensLine3_G :=  CompensLine_G;
            WITH  CompensLine3_G DO BEGIN
              LineCount_G := LineCount_G + 1;

              IF NOT G_WinDialog THEN BEGIN
                IF ShowDialog() THEN BEGIN
                  Window_G.UPDATE(5,LineCount_G);
                  Window_G.UPDATE(6,ROUND(LineCount_G / NoOfRecords_G * 10000,1));
                END;
              END;
              CheckDocumentNo(CompensLine3_G);

              SetOffJnlPostLine_G.RUN(CompensLine3_G);
            END;
          END;
        UNTIL NEXT = 0;

        // Update/delete lines

        IF NOT RECORDLEVELLOCKING THEN
          LOCKTABLE(TRUE, TRUE);
        SetOffJnlLine2_G.COPY(CompensLine_G);
        SetOffJnlLine2_G.SETFILTER("Account No.", '<>%1', '');
        IF SetOffJnlLine2_G.FINDLAST THEN; // Remember the last line
        CompensLine3_G.COPY(CompensLine_G);

        IF CompensLine3_G.FINDSET(TRUE,FALSE) THEN
          REPEAT
            DeleteSetoffIdFromEntries(CompensLine3_G);
            CompensLine3_G.DELETE;
          UNTIL CompensLine3_G.NEXT = 0;
        CompensLine3_G.RESET;
        CompensLine3_G.SETRANGE("Capital Journal Template","Capital Journal Template");
        CompensLine3_G.SETRANGE("Capital Batch Name","Capital Batch Name");
        IF NOT CompensLine3_G.FINDLAST THEN
          IF INCSTR("Capital Batch Name") <> '' THEN BEGIN
            SetOffJnlBatch_G.DELETE;
            SetOffJnlBatch_G.Name := INCSTR("Capital Batch Name");
            IF SetOffJnlBatch_G.INSERT THEN;
            "Capital Batch Name" := SetOffJnlBatch_G.Name;
          END;

        CompensLine3_G.SETRANGE("Capital Batch Name","Capital Batch Name");
        IF (SetOffJnlBatch_G."No. Series" = '') AND NOT CompensLine3_G.FINDLAST THEN BEGIN
          CompensLine3_G.INIT;
          CompensLine3_G."Capital Journal Template" := "Capital Journal Template";
          CompensLine3_G."Capital Batch Name" := "Capital Batch Name";
          CompensLine3_G."Line No." := 10000;
          CompensLine3_G.INSERT;
          CompensLine3_G.MODIFY;
        END;

        IF SetOffJnlBatch_G."No. Series" <> '' THEN
          NoSeriesMgt_G.SaveNoSeries;
        IF NoSeries_G.FINDSET THEN
          REPEAT
            EVALUATE(PostingNoSeriesNo_G,NoSeries_G.Description);
            NoSeriesMgt2[PostingNoSeriesNo_G].SaveNoSeries;
          UNTIL NoSeries_G.NEXT = 0;
        COMMIT;

        CLEAR(SetOffJnlPostLine_G);
      END;
    END;

    PROCEDURE ShowDialog@1200070000() : Boolean;

    END;

    PROCEDURE CheckDocumentNo@11(VAR CapJnlLine2_G@8053000 : Record 8053356);
    BEGIN
      WITH CapJnlLine2_G DO BEGIN
        IF "Posting No. Series" = '' THEN
          "Posting No. Series" := SetOffJnlBatch_G."No. Series"
        ELSE
          IF NOT EmptyCompLine THEN
            IF "Document No." = LastDocNo_G THEN
              "Document No." := LastPostedDocNo_G
            ELSE BEGIN
              IF NOT NoSeries_G.GET("Posting No. Series") THEN BEGIN
                NoOfPostingNoSeries_G := NoOfPostingNoSeries_G + 1;
                IF NoOfPostingNoSeries_G > ARRAYLEN(NoSeriesMgt2) THEN
                  ERROR(Text8053025, ARRAYLEN(NoSeriesMgt2));
                NoSeries_G.Code := "Posting No. Series";
                NoSeries_G.Description := FORMAT(NoOfPostingNoSeries_G);
                NoSeries_G.INSERT;
              END;
              LastDocNo_G := "Document No.";
              EVALUATE(PostingNoSeriesNo_G,NoSeries_G.Description);
              "Document No." := NoSeriesMgt2[PostingNoSeriesNo_G].GetNextNo("Posting No. Series", "Posting Date", TRUE);
              LastPostedDocNo_G := "Document No.";
            END;
      END;
    END;
