{
  OBJECT-PROPERTIES
  {
    Date=05/10/15;
    Heure=16:07:24;
    Modifi'=Oui;
    Liste versions=REP001;
  }
  PROPERTIES
  {
    Permissions=;
    OnRun=BEGIN

            //{
            VendeurInsee.RESET;
            Total := VendeurInsee.COUNT;
            Fenetre.OPEN('Maj information VendeurInsee ...      \' +
                         'Nb lecture                      #1######\' +
                         'Nb mises . jour                 #2######\' +
                         '@3@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@',Cpt,Cpt2);

            IF VendeurInsee.FIND('-') THEN
            REPEAT
            // traitement sur les PA BOVIN
                IF (VendeurInsee."Prod. Activity Code" = 'PA') AND (VendeurInsee."Sales Network Code" = 'BOVIN') THEN BEGIN
                // rattach' le TC '2B2' et le TC '3B1' sur la r'gion 'BZ1'
                  IF (VendeurInsee."Salesperson Code" = '2B2') OR (VendeurInsee."Salesperson Code" = '3B1') THEN BEGIN
                    // VendeurInsee.VALIDATE("Sales Area Code",'BZ1');
                    VendeurInsee.DELETE;
                    VendeurInsee."Sales Area Code" := 'BZ1';
                    VendeurInsee.INSERT;
                    Cpt2 := Cpt2 +1;
                  END;
                // rattach' les TC '4B1' et '4B2' sur la r'gion 'BZ4'
                  IF (VendeurInsee."Salesperson Code" = '4B1') OR (VendeurInsee."Salesperson Code" = '4B2') THEN BEGIN
                    // VendeurInsee.VALIDATE("Sales Area Code",'BZ4');
                    VendeurInsee.DELETE;
                    VendeurInsee."Sales Area Code" := 'BZ4';
                    VendeurInsee.INSERT;
                    Cpt2 := Cpt2 +1;
                  END;

                END;

              Cpt := Cpt + 1;
              Fenetre.UPDATE;
              Fenetre.UPDATE(1,Cpt);
              Fenetre.UPDATE(2,Cpt2);
              Fenetre.UPDATE(3,ROUND(Cpt / Total * 10000, 1));

            UNTIL VendeurInsee.NEXT = 0;
            Fenetre.CLOSE;
            //}
          END;

  }
  CODE
  {

      VendeurInsee@1000000000 : Record 8053161;
      Total@1000000004 : Integer;
      Cpt@1000000005 : Integer;
      Cpt2@1000000006 : Integer;
      Cpt3@1000000022 : Integer;
      Fenetre@1000000007 : Dialog;
      ToolsFile_L@1000000029 : Codeunit 8053405;
      Texte1_G@1000000030 : Text[50];
      Texte2_G@1000000031 : Text[50];

    BEGIN
    END.
