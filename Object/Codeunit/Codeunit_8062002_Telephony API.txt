          CreateContactEntry("No.", "Phone No.","Country/Region Code", FIELDCAPTION("Phone No."),
                             Name, "E-Mail", "Home Page", "Company Name");
        IF '' <> "Mobile Phone No." THEN
          CreateContactEntry("No.", "Mobile Phone No.", "Country/Region Code", FIELDCAPTION("Mobile Phone No."),
                             Name, "E-Mail", "Home Page", "Company Name");

        WITH ContactBusinessRelation DO BEGIN
          // --> 20150312-WI00-CMC02.02.00-2694-FBE
          IF CompanyName_G <> '' THEN
            CHANGECOMPANY(CompanyName_G);
          // <-- 20150312-WI00-CMC02.02.00-2694-FBE

        END;
      END;
    END;
