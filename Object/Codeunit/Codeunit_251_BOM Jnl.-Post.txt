{
  OBJECT-PROPERTIES
  {
    Date=18/12/17;
    Heure=00:00:00;
    Liste versions=NAVW13.00;
  }
  PROPERTIES
  {
    TableNo=89;
    OnRun=BEGIN
            BOMJnlLine.COPY(Rec);
            Code;
            Rec.COPY(BOMJnlLine);
          END;

  }
  CODE
  {
    VAR
      Text000@1000 : TextConst 'ENU=cannot be filtered when posting recurring journals;FRA=ne peut pas ^tre filtr'(e) lors de la validation de feuilles abonnement';
      Text001@1001 : TextConst 'ENU=Do you want to post the journal lines?;FRA=Souhaitez-vous valider les lignes de la feuille ?';
      Text002@1002 : TextConst 'ENU=There is nothing to post.;FRA=Il n''y a rien . valider.';
      Text003@1003 : TextConst 'ENU=The journal lines were successfully posted.;FRA=Les lignes de la feuille ont 't' valid'es avec succSs.';
      Text004@1004 : TextConst 'ENU="The journal lines were successfully posted. ";FRA="Les lignes de la feuille ont 't' valid'es avec succSs. "';
      Text005@1005 : TextConst 'ENU=You are now in the %1 journal.;FRA=Vous ^tes maintenant dans la feuille %1.';
      BOMJnlTemplate@1006 : Record 88;
      BOMJnlLine@1007 : Record 89;
      BOMJnlPostBatch@1008 : Codeunit 33;
      TempJnlBatchName@1009 : Code[10];
      UseSilentMode_G@60000 : Boolean;

    LOCAL PROCEDURE Code@1();
    BEGIN
      WITH BOMJnlLine DO BEGIN
        BOMJnlTemplate.GET("Journal Template Name");
        BOMJnlTemplate.TESTFIELD("Force Posting Report",FALSE);
        IF BOMJnlTemplate.Recurring AND (GETFILTER("Posting Date") <> '') THEN
          FIELDERROR("Posting Date",Text000);

        // --> 20171218-WI00-ADV01.01.00-3286-CJY01
        IF NOT UseSilentMode_G THEN
        // <-- 20171218-WI00-ADV01.01.00-3286-CJY01
        IF NOT CONFIRM(Text001,FALSE) THEN
          EXIT;

        TempJnlBatchName := "Journal Batch Name";

        BOMJnlPostBatch.RUN(BOMJnlLine);

        IF "Line No." = 0 THEN
          MESSAGE(Text002)
        ELSE
          IF TempJnlBatchName = "Journal Batch Name" THEN
            MESSAGE(Text003)
          ELSE
            MESSAGE(
              Text004 +
              Text005,
              "Journal Batch Name");

        IF NOT FIND('=><') OR (TempJnlBatchName <> "Journal Batch Name") THEN BEGIN
          RESET;
          FILTERGROUP(2);
          SETRANGE("Journal Template Name","Journal Template Name");
          SETRANGE("Journal Batch Name","Journal Batch Name");
          FILTERGROUP(0);
          "Line No." := 1;
        END;
      END;
    END;

    PROCEDURE UseSilentMode@60000();
    BEGIN
      // --> 20171218-WI00-ADV01.01.00-3286-CJY01
      UseSilentMode_G := TRUE;
      // <-- 20171218-WI00-ADV01.01.00-3286-CJY01
    END;

    BEGIN
    {
      20171218-WI00-ADV01.01.00-3286-CJY01 : ADDCODE (Gestion desassemblage lors de l'annulation d'une ligne BL)
      20171218-WI00-ADV01.01.00-3286-CJY01 : ADDFUNC (ajout fonction UseSilentMode)
    }
    END.
  }
}
