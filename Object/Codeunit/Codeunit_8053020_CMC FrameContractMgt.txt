    PROCEDURE CreateDetailContract@8053001(FrameContract_P@8053000 : Record 8053208);
    VAR
      FrameContractLine_L@8053001 : Record 8053209;
      CMCFunnelContract_L@1000000000 : Record 8053196;
    BEGIN
      FrameContract_P.ModifyAllowed();

      FrameContractLine_L.RESET ;
      FrameContractLine_L.SETRANGE(FrameContractLine_L."Contract No.",FrameContract_P."Contract No.");
      IF FrameContractLine_L.FINDSET THEN
        REPEAT
          // --> 20190527-WI00-CMC02.02.00-3397-MRU01
          CMCFunnelContract_L.GET(FrameContract_P."Crop Rotation Campaign No.",FrameContractLine_L."Item Funnel Code");
          IF NOT ((FrameContractLine_L."Area (Ha)" = FrameContractLine_L."Autoconsumed Surface (Ha)")
                  AND (CMCFunnelContract_L."Default Species Code" = '')
                  AND (CMCFunnelContract_L."Default Item No." = '')) THEN

          IF (FrameContractLine_L."Quantity (T)">0) OR ((FrameContractLine_L."Area (Ha)"=FrameContractLine_L."Autoconsumed Surface (Ha)"
      )
            AND (FrameContractLine_L."Area (Ha)">0) AND (FrameContractLine_L."% Rate"=100))
          THEN
            CreateDetailLineContract(FrameContractLine_L);
        UNTIL FrameContractLine_L.NEXT = 0;
    END;

    PROCEDURE CreateDetailLineContract@8053002(FrameContractLine_P@8053000 : Record 8053209);
    VAR
      FrameContract_L@8053001 : Record 8053208;
      FrameContractLineDetail_L@8053002 : Record 8053210;
      CommitmentContract_L@8053003 : Record 8053214;
      FunnelContract_L@8053004 : Record 8053196;
      recParcelDetail_L@8053007 : Record 8053123;
      recSpecies_L@8053008 : Record 8052988;
      EntryNo_L@8053005 : Integer;
      TotalQty_L@8053006 : Decimal;
      recTMPParcelDetail_L@8053009 : TEMPORARY Record 8053123;
      QteParcel_L@8053010 : Decimal;
      TMPSpecies_L@8053011 : TEMPORARY Record 8052988;
      RecFrameContLineAmend_L@8053012 : Record 8053498;
      RecCommitmentContract_L@8053013 : Record 8053214;
      recRef_L@8053015 : RecordRef;
      cduSynchroCRM_L@8053014 : Codeunit 8053600;
    BEGIN
      FrameContract_L.GET(FrameContractLine_P."Contract No.");
      FrameContractLineDetail_L.RESET;
      FrameContractLineDetail_L.SETRANGE("Contract No.",FrameContract_L."Contract No.");
      FrameContractLineDetail_L.SETRANGE("Contract Line No.",FrameContractLine_P."Line No.");
      IF NOT FrameContractLineDetail_L.FINDFIRST THEN BEGIN // pas de ligne
        FunnelContract_L.GET(FrameContract_L."Crop Rotation Campaign No.",FrameContractLine_P."Item Funnel Code");
        // test sur les quantit' PF

        IF (FrameContractLine_P."% Rate"=100) AND (FrameContractLine_P.UpdateQtyPF(FrameContractLine_P)>0) THEN
          ERROR(Text8053003,FrameContractLine_P.Description,FrameContractLine_P.UpdateQtyPF(FrameContractLine_P));

        IF FrameContract_L.Answer=FrameContract_L.Answer::Accepted THEN BEGIN
          FrameContractLineDetail_L.INIT;
          FrameContractLineDetail_L."Contract No." := FrameContractLine_P."Contract No.";
          FrameContractLineDetail_L."Contract Line No." := FrameContractLine_P."Line No.";
          FrameContractLineDetail_L."Line No." :=  10000;
          IF FrameContractLine_P."% Rate"=100 THEN BEGIN
            FunnelContract_L.TESTFIELD("Default Complet Contract Type");
            FrameContractLineDetail_L.VALIDATE("Contract Type",FunnelContract_L."Default Complet Contract Type")
          END ELSE BEGIN
            FunnelContract_L.TESTFIELD("Default Partiel Contract Type");
            FrameContractLineDetail_L.VALIDATE("Contract Type",FunnelContract_L."Default Partiel Contract Type");
          END;

          IF (FunnelContract_L."Default Species Code"<>'') AND (FunnelContract_L."Default Item No."<>'') THEN BEGIN
            FrameContractLineDetail_L.VALIDATE("Species Code" , FunnelContract_L."Default Species Code");
            FrameContractLineDetail_L.VALIDATE("Item No." , FunnelContract_L."Default Item No.");
            FrameContractLineDetail_L.INSERT(TRUE);

            recRef_L.GETTABLE(FrameContractLineDetail_L);
            recRef_L.SETPOSITION(FrameContractLineDetail_L.GETPOSITION);
            cduSynchroCRM_L.SyncModify(recRef_L, recRef_L);
          END ELSE BEGIN //recherche dans le parcellaire
            recParcelDetail_L.RESET;
            recParcelDetail_L.SETRANGE("Farmer No.",FrameContract_L."Contact No.");
            recParcelDetail_L.SETRANGE("Campaign No.", FrameContract_L."Campaign No.");
            IF recParcelDetail_L.FINDSET THEN BEGIN
              recTMPParcelDetail_L.DELETEALL;
              QteParcel_L:=0;
              SetFunnelFilterOnSpecie(FunnelContract_L."Item Funnel Code",TMPSpecies_L);
              REPEAT
                TMPSpecies_L.Code:=recParcelDetail_L."Species Code";
                IF TMPSpecies_L.FIND('=') THEN BEGIN
                  recTMPParcelDetail_L:=recParcelDetail_L;
                  recTMPParcelDetail_L.INSERT;
                  QteParcel_L:=QteParcel_L+recParcelDetail_L."Area (Ha)";
                END;
              UNTIL recParcelDetail_L.NEXT=0;
              IF QteParcel_L<>FrameContractLine_P."Area (Ha)" THEN
                ERROR(Text8053001,QteParcel_L,FrameContractLine_P."Area (Ha)")
              ELSE BEGIN  //cr'ation des lignes
                IF recTMPParcelDetail_L.FINDSET THEN
                  REPEAT
                    FrameContractLineDetail_L."Line No." :=EntryNo_L;
                    EntryNo_L+=10000;
                    FrameContractLineDetail_L.VALIDATE("Species Code" , recTMPParcelDetail_L."Species Code");
                    FrameContractLineDetail_L.VALIDATE("Item No." , recTMPParcelDetail_L."Item No.");
                    FrameContractLineDetail_L.INSERT(TRUE);
                    // --> 20150703-Cer16-ADV02.02.00-1872-SDE01
                    {
                    FrameContractLineDetail_L.VALIDATE("Quantity (T)",FrameContractLine_P."Quantity (T)"*
                      recTMPParcelDetail_L."Area (Ha)"/FrameContractLine_P."Area (Ha)");
                    }
                    FrameContractLineDetail_L.VALIDATE("Quantity (T)",ROUND(FrameContractLine_P."Quantity (T)"*
                      recTMPParcelDetail_L."Area (Ha)"/FrameContractLine_P."Area (Ha)",0.001));
                    // <-- 20150703-Cer16-ADV02.02.00-1872-SDE01
                    FrameContractLineDetail_L.MODIFY(TRUE);

                    recRef_L.GETTABLE(FrameContractLineDetail_L);
                    recRef_L.SETPOSITION(FrameContractLineDetail_L.GETPOSITION);
                    cduSynchroCRM_L.SyncModify(recRef_L, recRef_L);
                  UNTIL recTMPParcelDetail_L.NEXT=0;
              END;
            END ELSE // pas de parcellaire
              ERROR(Text8053000,FunnelContract_L."Campaign No.",FunnelContract_L."Item Funnel Code");
          END;
        END;
      END;
    END;

    PROCEDURE SetFunnelFilterOnSpecie@8053012(FunnelCode_P@8053000 : Code[10];VAR SpeciesTMP_P@8053010 : TEMPORARY Record 8052988);
    VAR
      ItemFunnelFilterLine_L@8053008 : Record 8053359;
      FrameContractLine_L@8053007 : Record 8053209;
      Found_L@8053006 : Boolean;
      SubFamily_L@8053005 : Record 8052987;
      Family_L@8053003 : Record 8052986;
      Item_L@8053002 : Record 27;
      Species_L@8053001 : Record 8052988;
    BEGIN
      Species_L.RESET;
      CLEAR(SpeciesTMP_P);

      Found_L := FALSE;
      ItemFunnelFilterLine_L.RESET;
      ItemFunnelFilterLine_L.SETRANGE(ItemFunnelFilterLine_L.Code,FunnelCode_P);
      ItemFunnelFilterLine_L.SETRANGE(ItemFunnelFilterLine_L."Funnel Type",ItemFunnelFilterLine_L."Funnel Type"::Species);
      IF ItemFunnelFilterLine_L.FINDFIRST THEN BEGIN
        Found_L := TRUE;
        Species_L.SETFILTER(Species_L.Code,ItemFunnelFilterLine_L."Funnel Line Filter");
        Species_L.SETRANGE(Species_L."Item Entry Typed",TRUE);
          IF Species_L.FINDSET THEN
            REPEAT
              SpeciesTMP_P.INIT;
              SpeciesTMP_P.TRANSFERFIELDS(Species_L);
              IF SpeciesTMP_P.INSERT THEN;
            UNTIL Species_L.NEXT = 0;
       END;

      IF NOT Found_L THEN BEGIN
        ItemFunnelFilterLine_L.SETRANGE(ItemFunnelFilterLine_L."Funnel Type",ItemFunnelFilterLine_L."Funnel Type"::SubFamily);
        IF ItemFunnelFilterLine_L.FINDFIRST THEN BEGIN
          Found_L := TRUE;
          Species_L.SETFILTER(Species_L."Sub-Family Code",ItemFunnelFilterLine_L."Funnel Line Filter");
          Species_L.SETRANGE(Species_L."Item Entry Typed",TRUE);
          IF Species_L.FINDSET THEN
            REPEAT
              SpeciesTMP_P.INIT;
              SpeciesTMP_P.TRANSFERFIELDS(Species_L);
              IF SpeciesTMP_P.INSERT THEN;
            UNTIL Species_L.NEXT = 0;
         END;
       END;

      IF NOT Found_L THEN BEGIN
        ItemFunnelFilterLine_L.SETRANGE(ItemFunnelFilterLine_L."Funnel Type",ItemFunnelFilterLine_L."Funnel Type"::Family);
        IF ItemFunnelFilterLine_L.FINDFIRST THEN BEGIN
          SubFamily_L.RESET;
          SubFamily_L.SETFILTER(SubFamily_L."Family Code",ItemFunnelFilterLine_L."Funnel Line Filter");
          IF SubFamily_L.FINDSET THEN
            REPEAT
             Species_L.SETRANGE(Species_L."Sub-Family Code",SubFamily_L.Code);
             Species_L.SETRANGE(Species_L."Item Entry Typed",TRUE);
             IF Species_L.FINDSET THEN
               REPEAT
                 SpeciesTMP_P.INIT;
                 SpeciesTMP_P.TRANSFERFIELDS(Species_L);
                 IF SpeciesTMP_P.INSERT THEN;
               UNTIL Species_L.NEXT =0 ;
            UNTIL SubFamily_L.NEXT=0;
          Found_L := TRUE;
         END;
       END;

      IF NOT Found_L THEN BEGIN
        ItemFunnelFilterLine_L.SETRANGE(ItemFunnelFilterLine_L."Funnel Type",ItemFunnelFilterLine_L."Funnel Type"::Activity);
        IF ItemFunnelFilterLine_L.FINDFIRST THEN BEGIN
          Family_L.RESET;
          Family_L.SETFILTER(Family_L."Activity Code",ItemFunnelFilterLine_L."Funnel Line Filter");
          IF Family_L.FINDSET THEN
            REPEAT
              SubFamily_L.RESET;
              SubFamily_L.SETFILTER(SubFamily_L."Family Code",Family_L.Code);
              IF SubFamily_L.FINDSET THEN
                REPEAT
                 Species_L.SETRANGE("Sub-Family Code",SubFamily_L.Code);
                 Species_L.SETRANGE(Species_L."Item Entry Typed",TRUE);
                 IF Species_L.FINDSET THEN
                    REPEAT
                      SpeciesTMP_P.INIT;
                      SpeciesTMP_P.TRANSFERFIELDS(Species_L);
                      IF SpeciesTMP_P.INSERT THEN;
                    UNTIL Species_L.NEXT=0
                UNTIL SubFamily_L.NEXT=0;
             UNTIL Family_L.NEXT=0;
          Found_L := TRUE;
         END;
       END;

      IF NOT Found_L THEN BEGIN
        ItemFunnelFilterLine_L.SETRANGE(ItemFunnelFilterLine_L."Funnel Type",ItemFunnelFilterLine_L."Funnel Type"::Item);
        IF ItemFunnelFilterLine_L.FINDFIRST THEN BEGIN
          Item_L.RESET;
          Item_L.SETFILTER(Item_L."No.",ItemFunnelFilterLine_L."Funnel Line Filter");
          IF Item_L.FINDSET THEN
            REPEAT
              IF Item_L."Species Code"<>'' THEN BEGIN
               Species_L.SETRANGE(Species_L."Item Entry Typed",TRUE);
               Species_L.SETRANGE(Species_L.Code,Item_L."Species Code");
               IF Species_L.FINDSET THEN
