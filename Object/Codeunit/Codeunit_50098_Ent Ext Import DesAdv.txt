    PROCEDURE Interface@1000000034(CodeUnit_P@1000000000 : Integer;Destinataire_P@1000000001 : Code[10];RecConnecteur_P@1000000002 : Record 8061230);
    VAR
      "<-- 3LI LOCALS -->"@1000000004 : Integer;
      Result_L@1000000003 : Text[255];
      FileLogName_L@1000000005 : Text[256];
    BEGIN
      MoteurInterface_G.GetParameter(CodeUnit_G, ConnectorName_G, ImpExpFolder_G, LogFolder_G, ExtensionFile_G, PrefixFile_G);
      MoteurInterface_G.GetTemplatesCode(ConnectorName_G, ItemTemplateCode_G, CustomerTemplateCode_G, VendorTemplateCode_G,
        ContactTemplateCode_G, AccountTemplateCode_G, GlobalTemplateCode_G);
      ToolsFile_G.LogInit(LogFolder_G, 'ENTEXT');
      ToolsFile_G.LogInitCodeUnit(CodeUnit_G);

      IF (RecConnecteur_P."File in use" <> '') THEN BEGIN
        ToolsFile_G.LogWriteInfo('Fichier : ' + RecConnecteur_P."File in use");

        ImportTexte2ToSas(RecConnecteur_P."File in use", Destinataire_P);
      END ELSE BEGIN
        ToolsFile_G.LogWriteInfo('Import . vide');
        // A la fin Appel . Vide : Traitement Sas Bdd non int'gr' vers Bdd NavAgri.
        // Mise . jour correspondance
        ImportRecordMajCorr(Destinataire_P);

        // Maj prix non trait' ==> Toper les enregistrements
        IF TrfTopMajPrix(Destinataire_P) THEN BEGIN
          // Erreur arr^t traitement Top enr non trait' ==> ne pas traiter "TRANSFERT DE PROPRIETE"
          // Traiter d'abord les transferts de propri't' afin d'incr'menter le Stock
          ImportRecord(Destinataire_P, 'TRANSFERT DE PROPRIETE', FALSE);
          ImportRecord(Destinataire_P, 'TRANSFERT DE PROPRIETE', TRUE);

          // Validation
          ImportRecordValid(Destinataire_P, 'TRANSFERT DE PROPRIETE');
        END;

        ImportRecord(Destinataire_P, 'AVIS EXPEDITION AGRICULTEUR', FALSE);
        ImportRecord(Destinataire_P, 'AVIS EXPEDITION AGRICULTEUR', TRUE);

        ImportRecord(Destinataire_P, 'AVIS EXPEDITION DEPOT SUR ORDRE', FALSE);

        ImportRecord(Destinataire_P, 'AVIS EXPEDITION DEPOT', FALSE);

        ImportRecordValid(Destinataire_P, 'AVIS EXPEDITION AGRICULTEUR');
        ImportRecordValid(Destinataire_P, 'AVIS EXPEDITION DEPOT SUR ORDRE');
        ImportRecordValid(Destinataire_P, 'AVIS EXPEDITION DEPOT');
        TransferMgt_G := TRUE;
        ImportRecord(Destinataire_P, 'AVIS EXPEDITION DEPOT', FALSE);
        //*------------------------------------------*
        //* Traiter les livraisons directes.         *
        //*------------------------------------------*
        TransferMgt_G := FALSE;
        // Pour g'rer le ROLLBACK.
        CLEAR(Result_L);
        FileLogName_L := ToolsFile_G.LogName;
        ToolsFile_G.LogClose();
        IF ToolsFile_G.LogErreurType() = 'E' THEN BEGIN
          Result_L := 'ERREUR ' + FileLogName_L;
          ERROR(Result_L);
        END ELSE BEGIN
          IF ToolsFile_G.LogErreurType() = 'I' THEN BEGIN
            Result_L := Text0008;
            MoteurInterface_G.InsertLog(CodeUnit_G, FileLogName_L);
          END ELSE BEGIN
            MoteurInterface_G.InsertOk(CodeUnit_G);
          END;
        END;
      END;
    END;

    PROCEDURE SearchWarehouseShipmentV2@1000000026(EntExtShipmentSas_P@1000000009 : Record 50002;RecExpedEntrLigne_P@1000000000 : Record 7321;VAR WarehouseShipmentNo@1000000007 : Code[20];VAR WareHouseShipmentLineNo@1000000008 : Integer;ItemNo_P@1000000010 : Code[20];VariantCode_P@1000000011 : Code[10];VAR FlagPack_P@1000000012 : Boolean) ReturnValue : Boolean;
    VAR
      FlagPackNo@1000000001 : Code[20];
      RecExpedEntrLigne_L@1000000002 : Record 7321;
      ProductionBomLine@1000000003 : Record 99000772;
      Item@1000000004 : Record 27;
      ShipmentQuantity@1000000005 : Decimal;
      WarehouseShipmentLine@1000000006 : Record 7321;
    BEGIN
      // On a pas trouv' avec le Code article ==> Pack
      FlagPack_P := FALSE;
      ReturnValue := FALSE;
      FlagPackNo := '';
      ToolsFile_G.LogWriteInfo('SearchWareShipmentV2 : ');
      //IF (RecExpedEntrLigne_P.FINDFIRST()) THEN BEGIN
        Item.RESET();
        IF (Item.GET(RecExpedEntrLigne_P."Item No.")) THEN BEGIN
          FlagPackNo := Item."No.";
          ShipmentQuantity := RecExpedEntrLigne_L.Quantity;
          IF (RecExpedEntrLigne_L."Item No." <> ItemNo_P) THEN BEGIN // Produit de la ligne exp <> de SicaLog ==> PAck ?
            //Get the flagPackNo and the calculated warehouse quantity
            ProductionBomLine.SETRANGE("Production BOM No.", Item."Kit BOM No.");
            ProductionBomLine.SETRANGE("Version Code", '');
            ProductionBomLine.SETRANGE(Type, ProductionBomLine.Type::Item);
            ProductionBomLine.SETRANGE("No.", ItemNo_P);
            IF (ProductionBomLine.FINDFIRST()) THEN BEGIN
              ReturnValue := TRUE;
              ShipmentQuantity := EntExtShipmentSas_P."Expedition quantity" / ProductionBomLine."Quantity per";
              WarehouseShipmentNo := RecExpedEntrLigne_P."No.";
            END;
          END ELSE BEGIN
            ReturnValue := TRUE;
          END;
          IF (ReturnValue) THEN BEGIN
            // V'rifier que la quantit' du pack correspond bien . celle de l'exp'dition.
            WarehouseShipmentLine.RESET();
            WarehouseShipmentLine.SETRANGE("Source Type", RecExpedEntrLigne_P."Source Type");
            WarehouseShipmentLine.SETRANGE("Source Subtype", RecExpedEntrLigne_P."Source Subtype");
            WarehouseShipmentLine.SETRANGE("No.", RecExpedEntrLigne_P."No.");
            WarehouseShipmentLine.SETRANGE("Line No.", RecExpedEntrLigne_P."Line No.");
            WarehouseShipmentLine.SETRANGE("Item No.", FlagPackNo);
            WarehouseShipmentLine.SETRANGE(Quantity, ShipmentQuantity);
            IF (NOT WarehouseShipmentLine.FINDFIRST()) THEN BEGIN
              ReturnValue := FALSE;
            END ELSE BEGIN
              FlagPack_P := TRUE;
              WarehouseShipmentNo := WarehouseShipmentLine."No.";
              WareHouseShipmentLineNo := WarehouseShipmentLine."Line No.";
            END;
          END;
        END;
      //END;
    END;
