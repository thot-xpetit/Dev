        IF NOT CRMSynch_2_Actif_G THEN BEGIN

          END ELSE BEGIN

              ZoneContactId_L := 'AccountId';

          ToolsFile_G.LogWriteInfo(COPYSTR(STRSUBSTNO(Text0002, ListNom_P, Count_L), 1, 250));
          IF Count_L > 0 THEN BEGIN
            TableCrmGuid_L.SETCURRENTKEY("Type Data", GUID);
            REPEAT
              accountId_L := CRMSynch_2_G.GetAttribute(EntityName_L, ZoneContactId_L);
              Erreur_L := FALSE;

              // Correspondance Guid Grc - Code Nav contact.
              CLEAR(CodeNav_L);
              TableCrmGuid_L.SETRANGE("Type Data", TableCrmGuid_L."Type Data"::Account);
              TableCrmGuid_L.SETRANGE(GUID, accountId_L);
              IF NOT TableCrmGuid_L.FINDFIRST THEN BEGIN
                // Une autre chance peut-^tre relation dans table contact ? (Florent Hugues)
                TableContact_L.SETRANGE(TableContact_L."Dynamics CRM Id", accountId_L);
                IF NOT TableContact_L.FINDFIRST() THEN BEGIN
                  Erreur_L := TRUE;
                  IF TypeMembre_P = '2' THEN BEGIN
                    Nom_L := GetContact(accountId_L);
                  END ELSE BEGIN
                    IF TypeMembre_P = '1' THEN BEGIN
                      Nom_L := GetAccount(accountId_L);
                    END ELSE BEGIN

                    END;
                  END;
                  ToolsFile_G.LogWriteInfo(COPYSTR(STRSUBSTNO(Text0001, TableCrmGuid_L."Type Data"::Account, accountId_L, Nom_L,
                    TableCrmGuid_L.TABLECAPTION), 1, 250));
                END ELSE BEGIN
                  CodeNav_L := TableContact_L."No.";
                END;
                TableContact_L.SETRANGE(TableContact_L."Dynamics CRM Id");
              END ELSE BEGIN
                CodeNav_L := TableCrmGuid_L."Code Nav";
              END;

              IF NOT Erreur_L THEN BEGIN
                // Correspondance AS/400
                ClientWMS_L := MoteurInterface_G.GetReferentialCorrespondance(CustomerTemplateCode_G, FALSE,
                  CodeNav_L, FALSE);

                IF NOT TableContactOption_L.GET(CodeNav_L, TableExtranetConn_G."Service Code",
                  TableExtranetConn_G."Option Code") THEN BEGIN

                  IF TableContact_L.GET(CodeNav_L) THEN BEGIN
                    Nom_L := TableContact_L.Name;
                  END ELSE BEGIN
                    CLEAR(Nom_L);
                  END;
                  CLEAR(TableContactOption_L);
                  TableContactOption_L.VALIDATE("Contact No.", CodeNav_L);
                  TableContactOption_L.VALIDATE("Account Name", Nom_L);
                  TableContactOption_L.VALIDATE("Service Code", TableExtranetConn_G."Service Code");
                  TableContactOption_L.VALIDATE("Option Code", TableExtranetConn_G."Option Code");
                  TableContactOption_L.VALIDATE(Active, TRUE);
                  TableContactOption_L.VALIDATE("Date 1", TODAY);
                  TableContactOption_L.VALIDATE("Text 2", ClientWMS_L);
                  IF NOT TableContactOption_L.INSERT() THEN BEGIN
                    ToolsFile_G.LogWriteInfo(COPYSTR(STRSUBSTNO(Text0007, CodeNav_L, TableExtranetConn_G."Service Code",
                      TableExtranetConn_G."Option Code", TableContactOption_L.TABLECAPTION) , 1, 250));
                  END ELSE BEGIN
                    // Cr'er le service sur le contact si n'existe pas
                    IF NOT TableContactServ_L.GET(TableContactOption_L."Contact No.", TableContactOption_L."Service Code") THEN BEGIN
                      TableContactServ_L.VALIDATE("Contact No.", TableContactOption_L."Contact No.");
                      TableContactServ_L."Service Code" := TableContactOption_L."Service Code";
                      TableContactServ_L.Active := TRUE;
                      IF NOT TableContactServ_L.INSERT() THEN BEGIN
                        ToolsFile_G.LogWriteInfo(COPYSTR(STRSUBSTNO(Text0007, CodeNav_L, TableContactServ_L."Service Code"
                          ,'', TableContactServ_L.TABLECAPTION) , 1, 250));
                      END;
                    END;
                    Ajout_L += 1;
                  END;
                END ELSE BEGIN
                  IF ((NOT TableContactOption_L.Active) OR (TableContactOption_L."Text 2" <> ClientWMS_L)) THEN BEGIN
                    TableContactOption_L.VALIDATE(Active, TRUE);
                    IF NOT TableContactOption_L.Active THEN BEGIN
                      TableContactOption_L.VALIDATE("Date 1", TODAY);
                    END;
                    TableContactOption_L.VALIDATE("Text 2", ClientWMS_L);
                    // <-- 20190121-INT-ADVITAM-XPE001
                    TableContactOption_L.VALIDATE("Date 2", 0D);
                    // --> 20190121-INT-ADVITAM-XPE001
                    IF NOT TableContactOption_L.MODIFY() THEN BEGIN
                      ToolsFile_G.LogWriteInfo(COPYSTR(STRSUBSTNO(Text0007, CodeNav_L, TableExtranetConn_G."Service Code",
                        TableExtranetConn_G."Option Code", TableContactOption_L.TABLECAPTION) , 1, 250));
                    END ELSE BEGIN
                      Modif_L += 1;
                    END;
                  END;
                END;
                TableContactOptionTEMP_L.TRANSFERFIELDS(TableContactOption_L);
                IF TableContactOptionTEMP_L.INSERT THEN;
              END;
            UNTIL NOT(CRMSynch_2_G.NextEntity(EntityName_L, 1));
          END;
        END;

        //==================================================
        // Les enregistrements n'existant plus en Grc.
        //==================================================
        TableContactOption_L.SETRANGE("Service Code", TableExtranetConn_G."Service Code");
        TableContactOption_L.SETRANGE("Option Code", TableExtranetConn_G."Option Code");
        TableContactOption_L.SETRANGE("Boolean 1", FALSE);
        IF TableContactOption_L.FINDSET THEN BEGIN
          REPEAT
            IF NOT TableContactOptionTEMP_L.GET(TableContactOption_L."Contact No.", TableContactOption_L."Service Code",
              TableContactOption_L."Option Code") THEN BEGIN

              IF TableContactOption_L.Active THEN BEGIN
                TableContactOption_L.VALIDATE(Active, FALSE);

      ;

                END;
              END;

          UNTIL TableContactOption_L.NEXT = 0;
        END;
        IF ((Ajout_L <> 0) OR (Modif_L <> 0) OR (Supp_L <> 0)) THEN BEGIN
