    PROCEDURE IsApprovInProgress@8053010(SalesLine_P@8060982 : Record 37;ApproTemplCodeFilter_P@1000000001 : Text[100]) : Boolean;
    VAR
      ApprovalEntry_L@8060983 : Record 454;
      ApprovalTemplates_L@1000000000 : Record 464;
    BEGIN
      WITH SalesLine_P DO BEGIN
        CLEAR(ApprovalTemplates_L);
        ApprovalTemplates_L.SETCURRENTKEY("Table ID","Document Type",Enabled);
        ApprovalTemplates_L.SETRANGE("Table ID",DATABASE::"Sales Header",DATABASE::"Sales Line");
        ApprovalTemplates_L.SETRANGE("Document Type","Document Type");
        ApprovalTemplates_L.SETRANGE(Enabled,TRUE);
        IF ApproTemplCodeFilter_P <> '' THEN
          ApprovalTemplates_L.SETFILTER("Approval Code",ApproTemplCodeFilter_P);
        IF ApprovalTemplates_L.FIND('-') THEN
          REPEAT
            CLEAR(ApprovalEntry_L);
            ApprovalEntry_L.SETCURRENTKEY("Approval Code");
            ApprovalEntry_L.SETRANGE("Table ID",DATABASE::"Sales Header");
            ApprovalEntry_L.SETRANGE("Document Type","Document Type");
            ApprovalEntry_L.SETRANGE("Document No.","Document No.");
            ApprovalEntry_L.SETRANGE("Document Line No.",0);
            ApprovalEntry_L.SETRANGE("Approval Code",ApprovalTemplates_L."Approval Code");
            IF ApprovalEntry_L.FINDLAST THEN
              IF (ApprovalEntry_L.Status <> ApprovalEntry_L.Status::Approved) AND
                 (ApprovalEntry_L.Status <> ApprovalEntry_L.Status::Canceled) THEN
                EXIT(TRUE);
            CLEAR(ApprovalEntry_L);
            ApprovalEntry_L.SETCURRENTKEY("Approval Code");
            ApprovalEntry_L.SETRANGE("Table ID",DATABASE::"Sales Line");
            ApprovalEntry_L.SETRANGE("Document Type","Document Type");
            ApprovalEntry_L.SETRANGE("Document No.","Document No.");
            ApprovalEntry_L.SETRANGE("Document Line No.","Line No.");
            ApprovalEntry_L.SETRANGE("Approval Code",ApprovalTemplates_L."Approval Code");
            IF ApprovalEntry_L.FINDLAST THEN
              IF (ApprovalEntry_L.Status <> ApprovalEntry_L.Status::Approved) AND
                 (ApprovalEntry_L.Status <> ApprovalEntry_L.Status::Canceled) THEN
                EXIT(TRUE);
            IF (SalesLine_P."Valued From Doc. Type" = SalesLine_P."Valued From Doc. Type"::Order) AND
               (SalesLine_P."Valued From Doc. No." <> '') THEN BEGIN
              CLEAR(ApprovalEntry_L);
              ApprovalEntry_L.SETCURRENTKEY("Approval Code");
              ApprovalEntry_L.SETRANGE("Table ID",DATABASE::"Sales Line");
              ApprovalEntry_L.SETRANGE("Document Type",ApprovalEntry_L."Document Type"::Order);
              ApprovalEntry_L.SETRANGE("Document No.",SalesLine_P."Valued From Doc. No.");
              ApprovalEntry_L.SETRANGE("Document Line No.",SalesLine_P."Valued From Doc. Line No.");
              ApprovalEntry_L.SETRANGE("Approval Code",ApprovalTemplates_L."Approval Code");
              IF ApprovalEntry_L.FINDLAST THEN
                IF (ApprovalEntry_L.Status <> ApprovalEntry_L.Status::Approved) AND
                   (ApprovalEntry_L.Status <> ApprovalEntry_L.Status::Canceled) THEN
                  EXIT(TRUE);
            END;
          UNTIL ApprovalTemplates_L.NEXT = 0
        ELSE
          EXIT(FALSE);
        EXIT(FALSE);
      END;
    END;
