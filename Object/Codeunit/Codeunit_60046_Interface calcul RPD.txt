OBJECT Codeunit 60046 Interface calcul RPD
{
  OBJECT-PROPERTIES
  {
    Date=14/10/20;
    Heure=00:00:00;
    Liste versions=SP3490;
  }
  PROPERTIES
  {
    TableNo=8061230;
    OnRun=VAR
            RecItem_L@1000000000 : Record 27;
            CalculRPD_L@1000000001 : Codeunit 60047;
            CodeUnit_L@1000000010 : Integer;
            ToolsFile_G@1000000009 : Codeunit 8053405;
            MoteurInterface_L@1000000008 : Codeunit 8061230;
            ConnectorName_L@1000000007 : Code[20];
            ImpExpFolder_L@1000000006 : Text[250];
            LogFolder_L@1000000005 : Text[250];
            FileLogName_L@1000000004 : Text[250];
            ExtensionFile_L@1000000003 : Text[10];
            PrefixFile_L@1000000002 : Text[10];
          BEGIN
            CodeUnit_L := Processing;
            MoteurInterface_L.GetParameter(CodeUnit_L, ConnectorName_L, ImpExpFolder_L, LogFolder_L, ExtensionFile_L, PrefixFile_L);
            ToolsFile_G.LogInit(LogFolder_L, 'NAVAGRI');
            ToolsFile_G.LogInitCodeUnit(CodeUnit_L);
            NbArticle_G := RecItem_L.COUNT;
            Compteur_G := 0;
            InfoArticle_G := Text60000 + Text60001;
            IF RecItem_L.FINDSET THEN
              BEGIN
              IF GUIALLOWED THEN
                Diag_G.OPEN(InfoArticle_G, Compteur_G, NbArticle_G);

                REPEAT
                  Compteur_G := Compteur_G + 1;
                  IF NOT CalculRPD_L.RUN(RecItem_L) THEN BEGIN
                    IF GUIALLOWED THEN
                      Diag_G.CLOSE;
                    ToolsFile_G.LogWriteInfo(COPYSTR(STRSUBSTNO(GETLASTERRORTEXT),1,1024));
                    COMMIT;
                  END
                  ELSE BEGIN
                    IF GUIALLOWED THEN
                      Diag_G.UPDATE();
                    COMMIT;
                  END;
                UNTIL RecItem_L.NEXT = 0;
              IF GUIALLOWED THEN
                Diag_G.CLOSE;
              END;

            FileLogName_L := ToolsFile_G.LogName;
            ToolsFile_G.LogClose();

            // Pour g'rer le ROLLBACK.
            IF ToolsFile_G.LogErreurType() = 'E' THEN BEGIN
              ERROR('ERREUR ' + FileLogName_L);
            END ELSE BEGIN
              IF ToolsFile_G.LogErreurType() = 'I' THEN BEGIN
                MoteurInterface_L.InsertLog(CodeUnit_L, FileLogName_L);
            MESSAGE('I');
              END ELSE BEGIN
            MESSAGE('OK');
                MoteurInterface_L.InsertOk(CodeUnit_L);
              END;
            END;
          END;

  }
  CODE
  {
    VAR
      Diag_G@1000000000 : Dialog;
      Compteur_G@1000000001 : Integer;
      NbArticle_G@1000000002 : Integer;
      Text60000@1000000003 : TextConst 'FRA=Article (et variantes associ'es)  #1######';
      Text60001@1000000004 : TextConst 'FRA=sur un total de #2######';
      InfoArticle_G@1000000005 : Text[200];

    BEGIN
    {
      // --> 3LI BEGIN
      20201012-WI00-ADV01.01.00-3490-CJY01 : ADDOBJECT
      // <-- 3LI END
    }
    END.
  }
}
