    Heure=00:00:00;

  }

            IF "Manual Process" THEN BEGIN
              ExportItem("File in use");
            END ELSE BEGIN
              ExportItem('');
            END;
          END;

  }
  CODE
  {
    VAR
      "<-- 3LI GLOBALS -->"@1000000001 : Integer;
      CodeUnit_G@1000000000 : Integer;
      "<-- 3LI TEXT GLOBALS -->"@1000000008 : TextConst;
      Text0001@1000000007 : TextConst 'ENU=Error Table "%1" doesn''t exist \Company : %2;FRA=Erreur Table "%1" inexistante \Societe : "%2"';
      Text0003@1000000006 : TextConst 'ENU=Error Creating File %1 \%2.;FRA=Erreur cr'ation fichier %1 \%2.';
      Text0004@1000000005 : TextConst 'ENU=Error updating table %1 \Company : %2;FRA=Erreur modification table "%1" \Societe : %2';
      Text0005@1000000004 : TextConst 'ENU=See the log please;FRA="Veuillez consulter le Log  "';
      Text0007@1000000003 : TextConst 'ENU=Error Correspondence "%1" : "%2" not parametrized for "%3";FRA=Erreur correspondance "%1" : "%2" non param'tr'e pour "%3".';
      Text0011@1000000002 : TextConst 'ENU=Error backup copy.;FRA=Erreur copie de sauvegarde.';

    PROCEDURE "<-- 3LI FUNCTIONS -->"@1000000000();
    BEGIN
    END;

    PROCEDURE ExportItem@1000000001(FileName_P@1000000000 : Text[128]) Result_L : Text[256];
    VAR
      "<-- 3LI LOCALS -->"@1000000002 : Integer;
      FileExport_L@1000000003 : File;
      FileExportOutstream_L@1000000004 : OutStream;
      ToolsFile_L@1000000007 : Codeunit 8053405;
      MoteurInterface_L@1000000014 : Codeunit 8061230;
      TableItem_L@1000000005 : Record 27;
      TableParameter_L@1000000006 : Record 8053450;
      ConnectorName_L@1000000023 : Code[20];
      ImpExpFolder_L@1000000022 : Text[256];
      LogFolder_L@1000000021 : Text[256];
      ExtensionFile_L@1000000031 : Text[10];
      PrefixFile_L@1000000030 : Text[30];
      FilePath_L@1000000026 : Text[256];
      FileN_L@1000000027 : Text[256];
      FileName_L@1000000020 : Text[256];
      FileTemp_L@1000000028 : Text[256];
      FileLogName_L@1000000019 : Text[256];
      Article_L@1000000001 : Text[30];
      ItemTemplateCode_L@1000000017 : Code[20];
      CustomerTemplateCode_L@1000000016 : Code[20];
      VendorTemplateCode_L@1000000015 : Code[20];
      ContactTemplateCode_L@1000000013 : Code[20];
      AccountTemplateCode_L@1000000012 : Code[20];
      GlobalTemplateCode_L@1000000008 : Code[20];
      FileNameOuvert_L@1000000034 : Boolean;
      NbCarEcrit_L@1000000035 : Integer;
      Filtre_L@1000000018 : Text[1024];
      Tab_L@1000000009 : Char;
    BEGIN
      //----------------------------------------------------------------------------------------------------------------------------------
      // Extraction des Articles  table "Item"   pour Cr'ation du fichier
      //----------------------------------------------------------------------------------------------------------------------------------
      // Entr'es:
      //   - FileName_P  : Nom du fichier PC . G'n'rer
      // Sorties:
      //   - Result_L    : Erreur Retourn'e, Vide si Ok
      //----------------------------------------------------------------------------------------------------------------------------------
      FileNameOuvert_L := FALSE;
      NbCarEcrit_L := 0;
      MoteurInterface_L.GetParameter(CodeUnit_G, ConnectorName_L, ImpExpFolder_L, LogFolder_L, ExtensionFile_L, PrefixFile_L);
      MoteurInterface_L.GetTemplatesCode (ConnectorName_L, ItemTemplateCode_L, CustomerTemplateCode_L, VendorTemplateCode_L,
        ContactTemplateCode_L, AccountTemplateCode_L, GlobalTemplateCode_L);
      ToolsFile_L.LogInit(LogFolder_L, 'CHAINFOOD');
      ToolsFile_L.LogInitCodeUnit(CodeUnit_G);

      IF FileName_P = '' THEN BEGIN
        IF (NOT TableParameter_L.FINDFIRST) THEN BEGIN
          ToolsFile_L.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0001, TableParameter_L.TABLECAPTION, COMPANYNAME), 1, MAXSTRLEN(Result_L)));
        END;
      END;

      IF ToolsFile_L.LogErreurType() <> 'E' THEN BEGIN
        IF NOT TableParameter_L.FINDFIRST THEN BEGIN
          ToolsFile_L.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0001, TableParameter_L.TABLECAPTION , COMPANYNAME), 1, MAXSTRLEN(Result_L)))
      ;
        END ELSE BEGIN
          IF TableParameter_L."ChainFood Export item" THEN BEGIN
            IF FileName_P <> '' THEN BEGIN
              FileName_L := FileName_P;
            END ELSE BEGIN
              FileName_L := ImpExpFolder_L + '\' + DELCHR(PrefixFile_L, '<>', ' ') + FORMAT(CURRENTDATETIME,
                0,'<Year4><Month,2><Day,2><Hours24,2><Minutes,2><Seconds,2><Second dec>') + '.' + ExtensionFile_L;
            END;
            ToolsFile_L.LogInitFileExecute(FileName_L);
            ToolsFile_L.SplitFilename(FileName_L, FilePath_L, FileN_L);
            FileTemp_L := FilePath_L + 'Temp_' + FileN_L;

            Tab_L :=9;

            TableItem_L.RESET;
            IF TableParameter_L."ChainFood Item Activity Code" = '' THEN BEGIN
              TableItem_L.SETRANGE("Activity Code");
            END ELSE BEGIN
              TableItem_L.SETFILTER("Activity Code", TableParameter_L."ChainFood Item Activity Code");
            END;
            IF TableParameter_L.ChainFoodItemExcludeFamilies = '' THEN BEGIN
              Filtre_L := '';
            END ELSE BEGIN
              Filtre_L := '<> ' + ToolsFile_L.TexteRemplaceCar(TableParameter_L.ChainFoodItemExcludeFamilies, '&', ' & <> ');
            END;
            TableItem_L.SETFILTER(TableItem_L."Family Code", Filtre_L);

            IF TableItem_L.FINDSET THEN BEGIN
              IF NOT FileNameOuvert_L THEN BEGIN
                IF NOT FileExport_L.CREATE(FileTemp_L) THEN BEGIN
                  ToolsFile_L.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0003, FileTemp_L, GETLASTERRORTEXT), 1, MAXSTRLEN(Result_L)));
                  CLEARLASTERROR;
                END ELSE BEGIN
                  FileExport_L.CREATEOUTSTREAM(FileExportOutstream_L);
                  FileNameOuvert_L := TRUE;
                END;
              END;
              IF FileNameOuvert_L THEN BEGIN
                REPEAT
                  IF NOT TableItem_L.Blocked THEN BEGIN
                    Article_L := MoteurInterface_L.GetReferentialCorrespondance(ItemTemplateCode_L, FALSE, TableItem_L."No.",
                      FALSE);
                    IF Article_L <> '$' THEN BEGIN
                      IF Article_L = '' THEN BEGIN
                        ToolsFile_L.LogWriteInfo(COPYSTR(STRSUBSTNO(Text0007, TableItem_L.FIELDCAPTION(TableItem_L."No."),
                          Article_L, TableItem_L."No."), 1, MAXSTRLEN(Result_L)));
                      END ELSE BEGIN
                        FileExportOutstream_L.WRITETEXT(Article_L);
                        FileExportOutstream_L.WRITETEXT(FORMAT(Tab_L),1);

                        FileExportOutstream_L.WRITETEXT(TableItem_L.Description);
                        FileExportOutstream_L.WRITETEXT(FORMAT(Tab_L),1);

                        FileExportOutstream_L.WRITETEXT(TableItem_L."Family Code");
                        FileExportOutstream_L.WRITETEXT(FORMAT(Tab_L),1);

                        FileExportOutstream_L.WRITETEXT();
                      END;
                    END;
                  END;
                UNTIL TableItem_L.NEXT=0;
              END;
            END;

            IF FileNameOuvert_L THEN BEGIN
              NbCarEcrit_L := FileExport_L.LEN;
              FileExport_L.CLOSE;
            END;
          END;
        END;
      END;

      IF (NbCarEcrit_L > 0) THEN BEGIN
        IF ToolsFile_L.LogErreurType() = 'E' THEN BEGIN
          IF FileNameOuvert_L THEN BEGIN
            FILE.ERASE(FileTemp_L);
          END;
        END ELSE BEGIN
          IF RENAME(FileTemp_L, FileName_L) THEN BEGIN
          END;
          IF NOT ToolsFile_L.FileCopySav(FileName_L, 'Sav', TRUE) THEN BEGIN
            ToolsFile_L.LogWriteErreur(STRSUBSTNO(Text0011, FileName_L));
          END;
        END;
      END ELSE BEGIN
        IF FileNameOuvert_L THEN BEGIN
          FILE.ERASE(FileTemp_L);
        END;
      END;

      CLEAR(Result_L);
      FileLogName_L := ToolsFile_L.LogName;
      ToolsFile_L.LogClose();
      IF ToolsFile_L.LogErreurType() = 'E' THEN BEGIN
        Result_L := 'ERREUR ' + FileLogName_L;
        ERROR(Result_L);
      END ELSE BEGIN
        IF ToolsFile_L.LogErreurType() = 'I' THEN BEGIN
          Result_L := Text0005;
          MoteurInterface_L.InsertLog(CodeUnit_G, FileLogName_L);
        END ELSE BEGIN
          MoteurInterface_L.InsertOk(CodeUnit_G);
        END;
      END;
    END;

    BEGIN
    {
      //--> 3LI BEGIN
      20140904-INT29-CMC02.02.00-ADVITAM-CDL : ChainFood : Export article
      Fichier Cr'' en Temporaire (pour ne pas ^tre consomm') puis renomm' si pas d'erreur sinon suppression.
      //<-- 3LI END
    }
    END.
  }
