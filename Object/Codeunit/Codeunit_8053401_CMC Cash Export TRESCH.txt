    PROCEDURE CashExportTRESCH@1000000002(FileName_P@1000000016 : Text[128];Directory_P@1000000015 : Text[128]) Result_L : Text[256];
    VAR
      "<-- 3LI LOCALS -->"@1000000009 : Integer;
      DernNoTrait_L@1000000025 : Integer;
      FileExport_L@1000000007 : File;
      FileExportOutStream_L@1000000006 : OutStream;
      TableCashSetup_L@1000000005 : Record 8053400;
      TableBankAccount_L@1000000004 : Record 271;
      TableBankAccountPostingGroup_L@1000000003 : Record 277;
      TableCompany_L@1000000000 : Record 2000000006;
      Sens_L@1000000008 : Text[1];
      FileName_L@1000000001 : Text[256];
      FilePath_L@1000000018 : Text[256];
      FileN_L@1000000019 : Text[256];
      FileTemp_L@1000000020 : Text[256];
      FileLogName_L@1000000022 : Text[256];
      ToolsFile_L@1000000002 : Codeunit 8053405;
      MoteurInterface_L@1000000010 : Codeunit 8061230;
      CodeUnit_L@1000000014 : Integer;
      ConnectorName_L@1000000013 : Code[20];
      ImpExpFolder_L@1000000012 : Text[256];
      LogFolder_L@1000000011 : Text[256];
      ExtensionFile_L@1000000024 : Text[10];
      PrefixFile_L@1000000023 : Text[10];
      FileNameOuvert_L@1000000017 : Boolean;
      NbCarEcrit_L@1000000021 : Integer;
    BEGIN
      //----------------------------------------------------------------------------------------------------------------------------------
      // Extraction des 'critures > au dernier ID trait' & correspondant au mode paiement param'tr' vers fichier texte longueur fixe
      // Cumul de toutes les soci't's param'tr's en cumul dans 1 fichier.
      //----------------------------------------------------------------------------------------------------------------------------------
      // Entr'es:
      //   - FileName_P : Nom du fichier PC . G'n'rer ou vide si utilisation du nom classique : 'TRESCHNA.TXT'
      //   - Directory_P : Nom du r'pertoire ou placer le fichier si FileName_P non pr'ciser ==> Nom classique
      // Sorties:
      //   - Result_L : Erreur Retourn'e, Vide si Ok
      //----------------------------------------------------------------------------------------------------------------------------------
      //-->20130521-Int02-CMC02.02.00-ADVITAM-CDL
      FileNameOuvert_L := FALSE;
      CodeUnit_L := 8053401;
      MoteurInterface_L.GetParameter(CodeUnit_L, ConnectorName_L, ImpExpFolder_L, LogFolder_L, ExtensionFile_L, PrefixFile_L);
      ToolsFile_L.LogInit(LogFolder_L, 'TRESO');
      //-->20131112-Int02-CMC02.02.00-ADVITAM-CDL
      ToolsFile_L.LogInitCodeUnit(CodeUnit_L);
      //<--20131112-Int02-CMC02.02.00-ADVITAM-CDL

      //<--20130521-Int02-CMC02.02.00-ADVITAM-CDL
      IF FileName_P <> '' THEN BEGIN
        FileName_L := FileName_P;
      END ELSE BEGIN
        IF Directory_P <> '' THEN BEGIN
          FileName_L := Directory_P + '\TRESCHNA.txt';
        END ELSE BEGIN
          FileName_L := ImpExpFolder_L + '\' + PrefixFile_L + '.' + ExtensionFile_L;
        END;
      END;
      IF TableCompany_L.FINDSET THEN BEGIN
        //-->20130521-Int02-CMC02.02.00-ADVITAM-CDL
        ToolsFile_L.LogInitFileExecute(FileName_L);
        ToolsFile_L.SplitFilename(FileName_L, FilePath_L, FileN_L);
        FileTemp_L := FilePath_L + 'Temp_' + FileN_L;
        // BEFORE
        {
        Result_L := ToolsFile_L.FileCreateNotExist(FileName_L, '.temp', FileExport_L);
        IF Result_L <> '' THEN BEGIN
          ToolsFile_L.LogWriteErreur(Result_L + '.temp');
        }
        // AFTER
        Result_L := ToolsFile_L.FileCreateNotExist(FileName_L, 'Temp_', FileExport_L);
        IF Result_L <> '' THEN BEGIN
          ToolsFile_L.LogWriteErreur(Result_L + 'Temp_');
        //<--20130521-Int02-CMC02.02.00-ADVITAM-CDL
        END ELSE BEGIN
          FileExport_L.CREATEOUTSTREAM(FileExportOutStream_L);
          FileNameOuvert_L := TRUE;

          REPEAT
            TableCashSetup_L.CHANGECOMPANY(TableCompany_L.Name);
            TableBankAccount_L.CHANGECOMPANY(TableCompany_L.Name);
            TableBankAccountPostingGroup_L.CHANGECOMPANY(TableCompany_L.Name);
            //-->20130628-Int02-CMC02.02.00-ADVITAM-CDL
            CLEAR(DernNoTrait_L);
            //<--20130628-Int02-CMC02.02.00-ADVITAM-CDL

            // Param'trage tr'sorerie existe
            IF (NOT TableCashSetup_L.FINDFIRST) OR
              ((TableCashSetup_L."Export TRESCH") AND (TableCashSetup_L."Company Name" = '')) THEN BEGIN

              ToolsFile_L.LogWriteInfo(STRSUBSTNO(Text0003, TableCompany_L.Name));
            END ELSE BEGIN
              IF TableCashSetup_L."Export TRESCH" THEN BEGIN
                // Filtre N› > au dernier trait'
                TableBankAccount_L.SETFILTER(TableBankAccount_L."Entry No.", '> %1', TableCashSetup_L."Last Entry No TRESCH");
                TableBankAccount_L.SETFILTER(TableBankAccount_L."Payment Method Code", TableCashSetup_L."Payment Method Code");

                IF TableBankAccount_L.FINDSET THEN BEGIN
                  REPEAT
                    // Sens
                    IF TableBankAccount_L."Amount (LCY)" < 0 THEN BEGIN
                      Sens_L := 'C';
                    END ELSE BEGIN
                      Sens_L := 'D';
                    END;

                    FileExportOutStream_L.WRITETEXT(PADSTR(TableCashSetup_L."Company Name", 3, ' '), 3);
                    FileExportOutStream_L.WRITETEXT(',');
                    // Code journal vide
                    FileExportOutStream_L.WRITETEXT(PADSTR('   ', 3, ' '), 3);
                    FileExportOutStream_L.WRITETEXT(',');
                    // Date op'ration
                    FileExportOutStream_L.WRITETEXT(PADSTR(FORMAT(TableBankAccount_L."Posting Date", 0,
                      '<Year4><Month,2><Day,2>'), 8, '0'), 8);
                    FileExportOutStream_L.WRITETEXT(',');
                    // Date 'ch'ance
                    FileExportOutStream_L.WRITETEXT(PADSTR(FORMAT(TableBankAccount_L."Document Date", 0,
                      '<Year4><Month,2><Day,2>'), 8, '0'), 8);
                    FileExportOutStream_L.WRITETEXT(',');
                    // Soci't' + compte du groupe
                    // Compte Zone "Bank Acc. Posting Group" Prendre la relation de la table 277 "Bank Account Posting Group"
                    IF TableBankAccountPostingGroup_L.GET(TableBankAccount_L."Bank Acc. Posting Group") THEN BEGIN
                      FileExportOutStream_L.WRITETEXT(PADSTR(TableCashSetup_L."Company Name", 3, ' ') +
                        PADSTR(TableBankAccountPostingGroup_L."G/L Bank Account No.", 9, ' '), 12)
                    END ELSE BEGIN
                      FileExportOutStream_L.WRITETEXT(PADSTR(TableCashSetup_L."Company Name", 3, ' ') +
                        PADSTR(TableBankAccount_L."Bank Acc. Posting Group", 9, ' '), 12);
                    END;
                    FileExportOutStream_L.WRITETEXT(',');
                    // Code type piSce = Code mode de rSglement
                    FileExportOutStream_L.WRITETEXT(PADSTR(TableBankAccount_L."Payment Method Code" + Sens_L, 11, ' '), 11);
                    FileExportOutStream_L.WRITETEXT(',');
                    // Libell'
                    FileExportOutStream_L.WRITETEXT(PADSTR(TableBankAccount_L.Description, 30, ' '), 30);
                    FileExportOutStream_L.WRITETEXT(',');
                    // Montant
                    FileExportOutStream_L.WRITETEXT(PADSTR(FORMAT(ABS(TableBankAccount_L."Amount (LCY)"), 0,
                      '<Integer><Decimals><Comma,.>'), 14, ' '), 14);
                    FileExportOutStream_L.WRITETEXT(',');
                    // Sens
                    FileExportOutStream_L.WRITETEXT(Sens_L, 1);

                    FileExportOutStream_L.WRITETEXT();
                    //-->20130628-Int02-CMC02.02.00-ADVITAM-CDL
                    DernNoTrait_L := TableBankAccount_L."Entry No.";
                    //<--20130628-Int02-CMC02.02.00-ADVITAM-CDL
                  UNTIL TableBankAccount_L.NEXT=0;
                  // Enregistrement dernier Id trait' et date export.
                  // <--20130628-Int02-CMC02.02.00-ADVITAM-CDL
                  IF DernNoTrait_L <> 0 THEN BEGIN
                  // -->20130628-Int02-CMC02.02.00-ADVITAM-CDL
                    TableCashSetup_L."Last Entry No TRESCH" := DernNoTrait_L;
                    TableCashSetup_L."Last Date Export TRESCH" := CURRENTDATETIME;
                    IF NOT TableCashSetup_L.MODIFY(TRUE) THEN BEGIN
                      ToolsFile_L.LogWriteErreur(STRSUBSTNO(Text0010, FileName_L));
                    END;
                  // -->20130628-Int02-CMC02.02.00-ADVITAM-CDL
                  END;
                  // <--20130628-Int02-CMC02.02.00-ADVITAM-CDL
                END;
              END;
            END;
          UNTIL TableCompany_L.NEXT = 0;
          //-->20130521-Int02-CMC02.02.00-ADVITAM-CDL
          NbCarEcrit_L := FileExport_L.LEN;
          // BEFORE
          {
          IF (FileExport_L.LEN > 0) THEN BEGIN
            FileExport_L.CLOSE;
            FILE.RENAME(FileName_L + '.temp', FileName_L);

            IF NOT ToolsFile_L.FileCopySav(FileName_L, 'Sav', TRUE) THEN BEGIN
              ToolsFile_L.LogWriteErreur(STRSUBSTNO(Text0011, FileName_L));
            END;
          END ELSE BEGIN
            FileExport_L.CLOSE;
            FILE.ERASE(FileName_L + '.temp');
          END;
          }
          // AFTER
          FileExport_L.CLOSE;
          //<--20130521-Int02-CMC02.02.00-ADVITAM-CDL
        END;
      END;
      //-->20130521-Int02-CMC02.02.00-ADVITAM-CDL
      // BEFORE
      {
      ToolsFile_L.LogClose();
      }
      // AFTER
      CLEAR(Result_L);
      FileLogName_L := ToolsFile_L.LogName;
      ToolsFile_L.LogClose();

      IF (NbCarEcrit_L > 0) THEN BEGIN
        IF RENAME(FileTemp_L, FileName_L) THEN BEGIN
        END;
        IF NOT ToolsFile_L.FileCopySav(FileName_L, 'Sav', TRUE) THEN BEGIN
          Result_L := STRSUBSTNO(Text0011, FileName_L);
          ToolsFile_L.LogWriteErreur(Result_L);
        END;
      END ELSE BEGIN
        IF FileNameOuvert_L THEN BEGIN
          FILE.ERASE(FileTemp_L);
        END;
      END;

      IF ToolsFile_L.LogErreurType() = 'E' THEN BEGIN
        Result_L := 'ERREUR ' + FileLogName_L;
        ERROR(Result_L);
      END ELSE BEGIN
        IF ToolsFile_L.LogErreurType() = 'I' THEN BEGIN
          Result_L := Text0012;
          MoteurInterface_L.InsertLog(CodeUnit_L, FileLogName_L);
        END ELSE BEGIN
          MoteurInterface_L.InsertOk(CodeUnit_L);
        END;
      END;
      //<--20130521-Int02-CMC02.02.00-ADVITAM-CDL
      //<--20130118-Int02-CMC02.02.00-ADVITAM-CDL
    END;

    BEGIN
    {
      //--> 3LI BEGIN
      20130118-Int02-CMC02.02.00-ADVITAM-CDL :
      - Fonction CashExportTRESCH : Export rapprochement TRESCH :
        + Traitement de toutes les soci't' en cumul dans 1 fichier texte.
        + Extraction des 'critures > au dernier ID trait' & correspondant au mode paiement param'tr' vers fichier texte longueur fixe.
      20130220-Int02-CMC02.02.00-ADVITAM-CDL : - Ajout zone Date 'ch'ance. Copie de sauvegarde.
      20130220-Int02-CMC02.02.00-ADVITAM-CDL : - Ajout s'parateur zone ','.
      20130220-Int02-CMC02.02.00-ADVITAM-CDL : - Modification formatage montant en Euro avec centimes s'parateur d'cimale '.'.
      20130521-Int02-CMC02.02.00-ADVITAM-CDL : - Ajout Robot & correspondance. Remplacement tout LogWriteErreur Par LogWriteErreur Sans
        utilisation de variable (2 lignes en 1 ligne)
      20130628-Int02-CMC02.02.00-ADVITAM-CDL : - ProblSme si pas d''criture dernier N› trait' remis . 0.
      20131112-Int02-CMC02.02.00-ADVITAM-CDL : - Ajout appel fonction ToolsFile_L.LogInitCodeUnit.
      //<-- 3LI END
    }
    END.
