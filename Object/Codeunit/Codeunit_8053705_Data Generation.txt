    OnRun=VAR
            DocDate_L@8053003 : Date;
            ShipDate_L@8053004 : Date;
          BEGIN
            IF DocumentDateMin_G = 0D THEN
              EXIT;

            // --> 20150409-CMCTools-FHS01
            IF TreatmentType_G = TreatmentType_G::"Create Document" THEN BEGIN
            // <-- 20150409-CMCTools-FHS01
              //Recherche des donn'es et placements des filtres initiaux
              recCust_G.GET(recCustTemplate_G."No.");
              recHeaderTemplate_G.GET(recTemplate_G."Document Type", recTemplate_G."No.");
              recLineTemplate_G.RESET;
              recLineTemplate_G.SETRANGE("Document Type", recTemplate_G."Document Type");
              recLineTemplate_G.SETRANGE("Document No.", recTemplate_G."No.");

              recAssignCharge_G.RESET;
              recAssignCharge_G.SETRANGE("Document Type", recTemplate_G."Document Type");
              recAssignCharge_G.SETRANGE("Document No.", recTemplate_G."No.");
            // --> 20150409-CMCTools-FHS01
            END;
            // <-- 20150409-CMCTools-FHS01

            //Pour chaque date de document dans l'interval, on cr'' un document
            REPEAT
              IF DocDate_L = 0D THEN
                DocDate_L := DocumentDateMin_G
              ELSE
                DocDate_L := CALCDATE('<CD+' + FORMAT(DocumentDateInterval_G) + 'D>', DocDate_L);

              IF TreatmentType_G = TreatmentType_G::"Create Document" THEN BEGIN
                //Pour chaque date de livraison dans l'interval, on cr'' un document
                REPEAT
                  IF ShipDate_L = 0D THEN
                    ShipDate_L := ShippingDateMin_G
                  ELSE
                    ShipDate_L := CALCDATE('<CD+' + FORMAT(ShippingDateInterval_G) + 'D>', ShipDate_L);
                  //Cr'ation du document
                  CreateDocument(DocDate_L, ShipDate_L);
                UNTIL ShipDate_L >= ShippingDateMax_G;
              END ELSE BEGIN
                Ship();
              END;
            UNTIL DocDate_L >= DocumentDateMax_G;
          END;

  }

    PROCEDURE CreateDocument@8053006(DocDate_P@8053000 : Date;ShippingDate_P@8053001 : Date);
    VAR
      recHeaderTemplate_L@8053004 : Record 36;
      recLineTemplate_L@8053003 : Record 37;
      recAssignCharge_L@8053002 : Record 5809;
      recLine_L@8053005 : Record 37;
      QtyRest_L@8053006 : Decimal;
      recItem_L@8053007 : Record 27;
      Reste_L@8053008 : Decimal;
      cduRelease_L@8053009 : Codeunit 414;
      recVariant_L@8053010 : Record 5401;
      IndQty_L@8053011 : Decimal;
    BEGIN
      CLEAR(recHeaderTemplate_L);
      recHeaderTemplate_L.INIT;
      recHeaderTemplate_L."Document Type" := recHeaderTemplate_G."Document Type";
      recHeaderTemplate_L.VALIDATE("No.", '');
      recHeaderTemplate_L.INSERT(TRUE);
      recHeaderTemplate_L.VALIDATE("Sell-to Customer No.", recCust_G."No.");
      recHeaderTemplate_L."External Document No." := ExternalDocNo_G;
      recHeaderTemplate_L.VALIDATE("Order Date", DocDate_P);
      recHeaderTemplate_L.VALIDATE("Requested Delivery Date", ShippingDate_P);
      recHeaderTemplate_L.VALIDATE("Payment Terms Code", recHeaderTemplate_G."Payment Terms Code");
      recHeaderTemplate_L.VALIDATE("Shipment Method Code", recHeaderTemplate_G."Shipment Method Code");
      recHeaderTemplate_L.VALIDATE("Location Code", recHeaderTemplate_G."Location Code");
      recHeaderTemplate_L.VALIDATE("Salesperson Code", recHeaderTemplate_G."Salesperson Code");
      recHeaderTemplate_L.VALIDATE("Shipping Agent Code", recHeaderTemplate_G."Shipping Agent Code");
      recHeaderTemplate_L.VALIDATE("Activity Code", recHeaderTemplate_G."Activity Code");
      recHeaderTemplate_L.VALIDATE(Adduction, recHeaderTemplate_G.Adduction);
      recHeaderTemplate_L.VALIDATE("Vetenarian Contact Code", recHeaderTemplate_G."Vetenarian Contact Code");
      recHeaderTemplate_L.VALIDATE("Doc. Category Code", recHeaderTemplate_G."Doc. Category Code");
      recHeaderTemplate_L.MODIFY(TRUE);

      IF recLineTemplate_G.FINDSET(FALSE, FALSE) THEN
        REPEAT
          IF (recLineTemplate_G."RPD Item Attached Line" = 0) AND
          (recLineTemplate_G."CVO Item Attached Line" = 0) AND
          (recLineTemplate_G."CVO Corn Item Attached Line" = 0) AND
          (recLineTemplate_G."Free Item Root Line" = 0)
          THEN BEGIN
            CLEAR(recLineTemplate_L);
            recLineTemplate_L.INIT;
            recLineTemplate_L."Document Type" := recLineTemplate_G."Document Type";
            recLineTemplate_L."Document No." := recHeaderTemplate_L."No.";
            recLineTemplate_L."Line No." := recLineTemplate_G."Line No.";
            recLineTemplate_L.VALIDATE("Sell-to Customer No.", recLineTemplate_G."Sell-to Customer No.");
            recLineTemplate_L.Type := recLineTemplate_G.Type;
            recLineTemplate_L.INSERT(TRUE);
            recLineTemplate_L.VALIDATE("No.", recLineTemplate_G."No.");
            recLineTemplate_L.VALIDATE("Variant Code", recLineTemplate_G."Variant Code");
            recLineTemplate_L.VALIDATE("Location Code", recLineTemplate_G."Location Code");
            recLineTemplate_L.VALIDATE("Unit of Measure", recLineTemplate_G."Unit of Measure");
            recLineTemplate_L.VALIDATE(Quantity, recLineTemplate_G.Quantity);
            recLineTemplate_L.VALIDATE("Negotiated Price", recLineTemplate_G."Negotiated Price");
            recLineTemplate_L.VALIDATE("Unit Price", recLineTemplate_G."Unit Price");
            recLineTemplate_L.VALIDATE("Prepayment %", recLineTemplate_G."Prepayment %");
            recLineTemplate_L.VALIDATE("Shipping Agent Code", recLineTemplate_G."Shipping Agent Code");
            recLineTemplate_L.VALIDATE("Shipping Agent Service Code", recLineTemplate_G."Shipping Agent Service Code");
            recLineTemplate_L.VALIDATE("Salesperson Code", recLineTemplate_G."Salesperson Code");
            recLineTemplate_L.VALIDATE("Truck Registration", recLineTemplate_G."Truck Registration");
            recLineTemplate_L.VALIDATE("Driver Name", recLineTemplate_G."Driver Name");
            recLineTemplate_L.VALIDATE("Shipment Method Code", recLineTemplate_G."Shipment Method Code");
            recLineTemplate_L.VALIDATE("Delivery Method Code", recLineTemplate_G."Delivery Method Code");
            // --> 20150409-CMCTools-FHS01
            recLineTemplate_L.VALIDATE("Location Code", recLineTemplate_G."Location Code");
            // <-- 20150409-CMCTools-FHS01
            recLineTemplate_L.VALIDATE("Actual Shipping Agent Code", recLineTemplate_G."Actual Shipping Agent Code");
            IF recLineTemplate_G."Attached to Line No." <> 0 THEN
              recLineTemplate_L."Attached to Line No." := recLineTemplate_G."Attached to Line No.";
            recLineTemplate_L.MODIFY(TRUE);

            IF recLineTemplate_L.Type = recLineTemplate_L.Type::"Charge (Item)" THEN BEGIN
              recAssignCharge_G.SETRANGE("Document Line No.", recLineTemplate_G."Line No.");
              IF recAssignCharge_G.FINDSET(FALSE, FALSE) THEN BEGIN
                REPEAT
                  CLEAR(recAssignCharge_L);
                  recAssignCharge_L.INIT;
                  recAssignCharge_L.TRANSFERFIELDS(recAssignCharge_G, TRUE);
                  recAssignCharge_L."Document No." := recHeaderTemplate_L."No.";
                  recAssignCharge_L."Applies-to Doc. No." := recHeaderTemplate_L."No.";
                  recAssignCharge_L.INSERT;
                UNTIL recAssignCharge_G.NEXT = 0;
              END;
            END;
            recLineTemplate_L.RESET;
            recLineTemplate_L.SETRANGE("Document Type", recLineTemplate_L."Document Type");
            recLineTemplate_L.SETRANGE("Document No.", recLineTemplate_L."Document No.");
            recLineTemplate_L.SETRANGE("Attached to Line No.", recLineTemplate_L."Line No.");
            recLineTemplate_L.DELETEALL;
          END
          ELSE BEGIN
            IF recLineTemplate_G."Free Item Root Line" <> 0 THEN BEGIN
              IF recLineTemplate_L.GET(recLineTemplate_L."Document Type", recHeaderTemplate_L."No.",
                recLineTemplate_G."Free Item Root Line")
              THEN
                recLineTemplate_L.CreateFreeProductLine;
              recLineTemplate_L.RESET;
              recLineTemplate_L.SETRANGE("Document Type", recTemplate_G."Document Type");
              recLineTemplate_L.SETRANGE("Document No.", recTemplate_G."No.");
              recLineTemplate_L.SETRANGE("Free Item Root Line", recLineTemplate_G."Free Item Root Line");
              IF recLineTemplate_L.FINDFIRST THEN BEGIN
                recLineTemplate_L.VALIDATE(Quantity, recLineTemplate_G.Quantity);
                recLineTemplate_L.MODIFY(TRUE);
              END;
            END;
          END;
        UNTIL recLineTemplate_G.NEXT = 0;
      IF recTemplate_G.Ship <> recTemplate_G.Ship::No THEN BEGIN
        recLineTemplate_L.RESET;
        recLineTemplate_L.SETRANGE("Document Type", recLineTemplate_G."Document Type");
        recLineTemplate_L.SETRANGE("Document No.", recHeaderTemplate_L."No.");
        IF recLineTemplate_L.FINDSET(FALSE, FALSE) THEN BEGIN
          REPEAT
            // --> 20150409-CMCTools-FHS01
            IF (recLineTemplate_L."RPD Item Attached Line" = 0) AND
            (recLineTemplate_L."CVO Item Attached Line" = 0) AND
            (recLineTemplate_L."CVO Corn Item Attached Line" = 0) AND
            (recLineTemplate_L."Free Item Root Line" = 0)
            THEN BEGIN
              // <-- 20150409-CMCTools-FHS01
              recLine_L.GET(recLineTemplate_L."Document Type", recLineTemplate_L."Document No.", recLineTemplate_L."Line No.");
              QtyRest_L := recLine_L.Quantity - recLine_L."Quantity Shipped";
              // --> 20150409-CMCTools-FHS01
              //BEFORE
              //IF recTemplate_G.Ship = recTemplate_G.Ship::Partial THEN
              //AFTER
              IF recTemplate_G.Ship = recTemplate_G.Ship::Partial THEN BEGIN
              // <-- 20150409-CMCTools-FHS01
                RANDOMIZE;
                QtyRest_L := QtyRest_L * (RANDOM(100) / 100);
              // --> 20150409-CMCTools-FHS01
              END;
              // <-- 20150409-CMCTools-FHS01
              IF QtyRest_L <> 0 THEN BEGIN
                IF recLine_L.Type = recLine_L.Type::Item THEN
                  // --> 20150416-CMCTools-FHS01
                  IF recLine_L."Variant Code" = '' THEN BEGIN
                  // <-- 20150416-CMCTools-FHS01
                    IF recItem_L.GET(recLine_L."No.") THEN BEGIN
                      // --> 20150409-CMCTools-FHS01
                      //BEFORE
                      {
                      Reste_L := QtyRest_L MOD recItem_L."Indiv. Base Qty";
                      IF (QtyRest_L + Reste_L) <= recLine_L.Quantity - recLine_L."Quantity Shipped" THEN
                        QtyRest_L += Reste_L
                      ELSE
                        QtyRest_L -= Reste_L;
                      }
                      //AFTER
                      Reste_L := QtyRest_L / recItem_L."Indiv. Base Qty";
                      Reste_L := ROUND(Reste_L, 1, '<');
                      QtyRest_L := Reste_L * recItem_L."Indiv. Base Qty";
                      // <-- 20150409-CMCTools-FHS01
                    END;
                  // --> 20150416-CMCTools-FHS01
                  END ELSE BEGIN
                    IF recVariant_L.GET(recLine_L."No.", recLine_L."Variant Code") THEN BEGIN
                      IndQty_L := recVariant_L."Indiv. Base Qty";
                      IF IndQty_L = 0 THEN BEGIN
                        IF recItem_L.GET(recLine_L."No.") THEN BEGIN
                          IndQty_L := recItem_L."Indiv. Base Qty";
                        END;
                      END;
                      IF IndQty_L <> 0 THEN BEGIN
                        Reste_L := QtyRest_L / IndQty_L;
                        Reste_L := ROUND(Reste_L, 1, '<');
                        QtyRest_L := Reste_L * IndQty_L;
                      END;
                    END;
                  END;
                  // <-- 20150416-CMCTools-FHS01
                recLine_L.VALIDATE("Qty. to Ship", QtyRest_L);
                recLine_L.MODIFY(TRUE);
              END;
            END;
          UNTIL recLineTemplate_L.NEXT = 0;
        END;
      END;
      IF Release_G = TRUE THEN BEGIN
        cduRelease_L.RUN(recHeaderTemplate_L);
      END;
    END;

    PROCEDURE Ship@8053001();
    VAR
      recOrder_L@8053001 : Record 36;
      cduSalesPost_L@8053002 : Codeunit 80;
    BEGIN
      recOrder_L.GET(recDocumentRelease_G."Document Type", recDocumentRelease_G."No.");
      recOrder_L.Ship := TRUE;
      recOrder_L.Invoice := FALSE;
      cduSalesPost_L.RUN(recOrder_L);
    END;

      // <-- 3LI END
    }
