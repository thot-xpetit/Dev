    PROCEDURE PostSalesInvDiscSalesSummary@8053000(SalesLine_P@8053008 : Record 37;SalesInvHeader_P@8053006 : Record 112;SalesInvLine_P@8053000 : Record 113);
    VAR
      Currency_L@8053001 : Record 4;
      Item_L@8053002 : Record 27;
      ItemDiscountGroup_L@8053005 : Record 341;
      SalesDiscSalesSummary_L@8053004 : Record 8053241;
      CurrExchRate_L@8053003 : Record 330;
      cduSalesPriceDiscMgt_L@8053007 : Codeunit 8053017;
      SalesShipLine_L@8053009 : Record 111;
      SalesLine2_L@8053010 : Record 37;
    BEGIN
      IF (SalesInvLine_P.Type = SalesInvLine_P.Type::Item) OR
        ((SalesInvLine_P.Type = SalesInvLine_P.Type::"Charge (Item)") AND (SalesInvLine_P."Item Charge Pricing Item No." <> ''))
      THEN BEGIN
        Currency_L.InitRoundingPrecision;
        IF SalesInvLine_P.Type = SalesInvLine_P.Type::Item THEN
          Item_L.GET(SalesInvLine_P."No.")
        ELSE
          Item_L.GET(SalesInvLine_P."Item Charge Pricing Item No.");
        IF Item_L."Item Disc. Group" <> '' THEN BEGIN
          IF ItemDiscountGroup_L.GET(Item_L."Item Disc. Group") THEN BEGIN
            IF (ItemDiscountGroup_L."Maintain Sales Summary") AND
               (NOT (ItemDiscountGroup_L."Exclude Sales Blanket Order" AND
               (SalesInvLine_P."Blanket Order No." <> '')))
            THEN BEGIN
              IF NOT SalesDiscSalesSummary_L.GET(SalesDiscSalesSummary_L."Document Type"::"Posted Invoice",
                SalesInvLine_P."Document No.",Item_L."Item Disc. Group",
                SalesInvLine_P."CMC Campaign No.")
              THEN BEGIN
                SalesDiscSalesSummary_L.INIT;
                SalesDiscSalesSummary_L."Document Type" := SalesDiscSalesSummary_L."Document Type"::"Posted Invoice";
                SalesDiscSalesSummary_L."Document No." := SalesInvLine_P."Document No.";
                SalesDiscSalesSummary_L."Item Disc. Group Code" := Item_L."Item Disc. Group";
                SalesDiscSalesSummary_L."Campaign No." := SalesInvLine_P."CMC Campaign No.";
                SalesDiscSalesSummary_L."Document Date" := SalesInvHeader_P."Posting Date";
                SalesDiscSalesSummary_L."Order Date" := GetOrderDate(SalesInvLine_P);
                SalesDiscSalesSummary_L."Customer No." := SalesInvHeader_P."Bill-to Customer No.";
                SalesDiscSalesSummary_L."Cust. Disc. Group Code" := SalesInvHeader_P."Customer Disc. Group";
                SalesDiscSalesSummary_L."Unit of Measure Code" := ItemDiscountGroup_L."Unit of Measure Code";
                IF NOT ItemDiscountGroup_L."Not Convert UoM" THEN
                  SalesDiscSalesSummary_L.Quantity :=
                    ROUND(SalesInvLine_P."Quantity (Base)" / cduSalesPriceDiscMgt_L.ConvertUnitofMeasure(Item_L.
                    "Base Unit of Measure",ItemDiscountGroup_L."Unit of Measure Code",Item_L."No."),0.00001);
                IF SalesInvHeader_P."Currency Code" <> '' THEN
                  SalesDiscSalesSummary_L."Gross Total Amount (LCY)" :=
                    ROUND(CurrExchRate_L.ExchangeAmtFCYToLCY(
                    SalesDiscSalesSummary_L."Document Date",SalesInvHeader_P."Currency Code",
                    SalesInvLine_P.Quantity * SalesInvLine_P."Unit Price",SalesInvHeader_P."Currency Factor"),
                    Currency_L."Amount Rounding Precision")
                ELSE
                  SalesDiscSalesSummary_L."Gross Total Amount (LCY)" :=
                    ROUND(SalesInvLine_P.Quantity * SalesInvLine_P."Unit Price",Currency_L."Amount Rounding Precision");
                SalesDiscSalesSummary_L.INSERT;
              END ELSE BEGIN
                IF NOT ItemDiscountGroup_L."Not Convert UoM" THEN
                  SalesDiscSalesSummary_L.Quantity := SalesDiscSalesSummary_L.Quantity +
                    ROUND(SalesInvLine_P."Quantity (Base)" /
                    cduSalesPriceDiscMgt_L.ConvertUnitofMeasure(Item_L."Base Unit of Measure",
                    ItemDiscountGroup_L."Unit of Measure Code",Item_L."No."),0.00001);
                IF SalesInvHeader_P."Currency Code" <> '' THEN
                  SalesDiscSalesSummary_L."Gross Total Amount (LCY)" := SalesDiscSalesSummary_L."Gross Total Amount (LCY)" +
                    ROUND(CurrExchRate_L.ExchangeAmtFCYToLCY(
                    SalesDiscSalesSummary_L."Document Date",SalesInvHeader_P."Currency Code",
                    SalesInvLine_P.Quantity * SalesInvLine_P."Unit Price",SalesInvHeader_P."Currency Factor"),
                    Currency_L."Amount Rounding Precision")
                ELSE
                  SalesDiscSalesSummary_L."Gross Total Amount (LCY)" := SalesDiscSalesSummary_L."Gross Total Amount (LCY)" +
                    ROUND(SalesInvLine_P.Quantity * SalesInvLine_P."Unit Price",Currency_L."Amount Rounding Precision");
                SalesDiscSalesSummary_L.MODIFY;
              END;

              // If sales order line is partially shipped and invoiced then the corresponding line
              // from Sales Disc. Sales Summary must be updated
              IF SalesLine_P."Document Type" = SalesLine_P."Document Type"::Order THEN BEGIN
                IF SalesLine_P."Qty. to Invoice" + SalesLine_P."Quantity Invoiced" <> SalesLine_P.Quantity THEN BEGIN
                  IF SalesDiscSalesSummary_L.GET(SalesDiscSalesSummary_L."Document Type"::Order,
                    SalesLine_P."Document No.",Item_L."Item Disc. Group",SalesLine_P."CMC Campaign No.")
                  THEN BEGIN
                    IF NOT ItemDiscountGroup_L."Not Convert UoM" THEN
                      SalesDiscSalesSummary_L.Quantity := SalesDiscSalesSummary_L.Quantity -
                        ROUND(SalesInvLine_P."Quantity (Base)" /
                        cduSalesPriceDiscMgt_L.ConvertUnitofMeasure(Item_L."Base Unit of Measure",
                        ItemDiscountGroup_L."Unit of Measure Code",Item_L."No."),0.00001);
                    IF SalesDiscSalesSummary_L.Quantity < 0 THEN
                      SalesDiscSalesSummary_L.Quantity := 0;
                    IF SalesInvHeader_P."Currency Code" <> '' THEN
                      SalesDiscSalesSummary_L."Gross Total Amount (LCY)" := SalesDiscSalesSummary_L."Gross Total Amount (LCY)"
                        - ROUND(CurrExchRate_L.ExchangeAmtFCYToLCY(
                        SalesDiscSalesSummary_L."Document Date",SalesInvHeader_P."Currency Code",
                        SalesInvLine_P.Quantity * SalesInvLine_P."Unit Price",SalesInvHeader_P."Currency Factor"),
                        Currency_L."Amount Rounding Precision")
                    ELSE
                      SalesDiscSalesSummary_L."Gross Total Amount (LCY)" := SalesDiscSalesSummary_L."Gross Total Amount (LCY)"
                        - ROUND(SalesInvLine_P.Quantity * SalesInvLine_P."Unit Price",Currency_L."Amount Rounding Precision");
                    SalesDiscSalesSummary_L.MODIFY;
                  END;
                END;
              END;

              // If the sales invoice line is get from a shipment line that come from a sales order line then
              // the line from Sales Disc. Sales Summary corresponding to sales order line must be updated
              IF SalesLine_P."Document Type" = SalesLine_P."Document Type"::Invoice THEN BEGIN
                IF (SalesLine_P."Shipment No." <> '') AND (SalesLine_P."Shipment Line No." <> 0) THEN BEGIN
                  IF SalesShipLine_L.GET(SalesLine_P."Shipment No.",SalesLine_P."Shipment Line No.") THEN BEGIN
                    IF SalesLine2_L.GET(SalesLine2_L."Document Type"::Order,SalesShipLine_L."Order No.",
                      SalesShipLine_L."Order Line No.")
                    THEN BEGIN
                      IF SalesDiscSalesSummary_L.GET(SalesDiscSalesSummary_L."Document Type"::Order,
                        SalesLine2_L."Document No.",Item_L."Item Disc. Group",SalesLine2_L."CMC Campaign No.")
                      THEN BEGIN
                        IF NOT ItemDiscountGroup_L."Not Convert UoM" THEN
                          SalesDiscSalesSummary_L.Quantity := SalesDiscSalesSummary_L.Quantity -
                            ROUND(SalesInvLine_P."Quantity (Base)" /
                            cduSalesPriceDiscMgt_L.ConvertUnitofMeasure(Item_L."Base Unit of Measure",
                            ItemDiscountGroup_L."Unit of Measure Code",Item_L."No."),0.00001);
                        IF SalesDiscSalesSummary_L.Quantity < 0 THEN
                          SalesDiscSalesSummary_L.Quantity := 0;
                        IF SalesInvHeader_P."Currency Code" <> '' THEN
                          SalesDiscSalesSummary_L."Gross Total Amount (LCY)" :=
                            SalesDiscSalesSummary_L."Gross Total Amount (LCY)" -
                              ROUND(CurrExchRate_L.ExchangeAmtFCYToLCY(
                              SalesDiscSalesSummary_L."Document Date",SalesInvHeader_P."Currency Code",
                              SalesInvLine_P.Quantity * SalesInvLine_P."Unit Price",SalesInvHeader_P."Currency Factor"),
                              Currency_L."Amount Rounding Precision")
                        ELSE
                          SalesDiscSalesSummary_L."Gross Total Amount (LCY)" :=
                            SalesDiscSalesSummary_L."Gross Total Amount (LCY)" -
                            ROUND(SalesInvLine_P.Quantity * SalesInvLine_P."Unit Price",Currency_L."Amount Rounding Precision");
                        SalesDiscSalesSummary_L.MODIFY;
                      END;
                    END;
                  END;
                END;
              END;
            END;
          END;
        END;
      END;
    END;

                      cduSalesPriceDiscMgt_L.ConvertUnitofMeasure(Item_L."Base Unit of Measure",
                      ItemDiscountGroup_L."Unit of Measure Code",Item_L."No."),0.00001);
                IF SalesCrMemoHeader_P."Currency Code" <> '' THEN
                  SalesDiscSalesSummary_L."Gross Total Amount (LCY)" :=
                    - ROUND(CurrExchRate_L.ExchangeAmtFCYToLCY(
                      SalesDiscSalesSummary_L."Document Date",SalesCrMemoHeader_P."Currency Code",
                      SalesCrMemoLine_P.Quantity * SalesCrMemoLine_P."Unit Price",SalesCrMemoHeader_P."Currency Factor"),
                      Currency_L."Amount Rounding Precision")
                ELSE
                  SalesDiscSalesSummary_L."Gross Total Amount (LCY)" :=
                    - ROUND(SalesCrMemoLine_P.Quantity * SalesCrMemoLine_P."Unit Price",Currency_L."Amount Rounding Precision");
                SalesDiscSalesSummary_L.INSERT;
              END ELSE BEGIN
                IF NOT ItemDiscountGroup_L."Not Convert UoM" THEN
                  SalesDiscSalesSummary_L.Quantity := SalesDiscSalesSummary_L.Quantity -
                    ROUND(SalesCrMemoLine_P."Quantity (Base)" /
                    cduSalesPriceDiscMgt_L.ConvertUnitofMeasure(Item_L."Base Unit of Measure",
                    ItemDiscountGroup_L."Unit of Measure Code",Item_L."No."),0.00001);
                IF SalesCrMemoHeader_P."Currency Code" <> '' THEN
                  SalesDiscSalesSummary_L."Gross Total Amount (LCY)" := SalesDiscSalesSummary_L."Gross Total Amount (LCY)" -
                    ROUND(CurrExchRate_L.ExchangeAmtFCYToLCY(
                      SalesDiscSalesSummary_L."Document Date",SalesCrMemoHeader_P."Currency Code",
                      SalesCrMemoLine_P.Quantity * SalesCrMemoLine_P."Unit Price",SalesCrMemoHeader_P."Currency Factor"),
                      Currency_L."Amount Rounding Precision")
                ELSE
                  SalesDiscSalesSummary_L."Gross Total Amount (LCY)" := SalesDiscSalesSummary_L."Gross Total Amount (LCY)" -
                    ROUND(SalesCrMemoLine_P.Quantity * SalesCrMemoLine_P."Unit Price",Currency_L."Amount Rounding Precision");
                SalesDiscSalesSummary_L.MODIFY;
              END;

              // If return order line is partially received and invoiced then the corresponding line
              // from Sales Disc. Sales Summary must be updated
              IF SalesLine_P."Document Type" = SalesLine_P."Document Type"::"Return Order" THEN BEGIN
                IF SalesLine_P."Qty. to Invoice" + SalesLine_P."Quantity Invoiced" <> SalesLine_P.Quantity THEN BEGIN
                  IF SalesDiscSalesSummary_L.GET(SalesDiscSalesSummary_L."Document Type"::"Return Order",
                    SalesLine_P."Document No.",Item_L."Item Disc. Group",SalesLine_P."CMC Campaign No.")
                  THEN BEGIN
                    IF NOT ItemDiscountGroup_L."Not Convert UoM" THEN
                      SalesDiscSalesSummary_L.Quantity := SalesDiscSalesSummary_L.Quantity +
                        ROUND(SalesCrMemoLine_P."Quantity (Base)" /
                        cduSalesPriceDiscMgt_L.ConvertUnitofMeasure(Item_L."Base Unit of Measure",
                        ItemDiscountGroup_L."Unit of Measure Code",Item_L."No."),0.00001);
                    IF SalesCrMemoHeader_P."Currency Code" <> '' THEN
                      SalesDiscSalesSummary_L."Gross Total Amount (LCY)" := SalesDiscSalesSummary_L."Gross Total Amount (LCY)"
                        + ROUND(CurrExchRate_L.ExchangeAmtFCYToLCY(
                        SalesDiscSalesSummary_L."Document Date",SalesCrMemoHeader_P."Currency Code",
                        SalesCrMemoLine_P.Quantity * SalesCrMemoLine_P."Unit Price",SalesCrMemoHeader_P."Currency Factor"),
                        Currency_L."Amount Rounding Precision")
                    ELSE
                      SalesDiscSalesSummary_L."Gross Total Amount (LCY)" := SalesDiscSalesSummary_L."Gross Total Amount (LCY)"
                        + ROUND(SalesCrMemoLine_P.Quantity * SalesCrMemoLine_P."Unit Price",Currency_L."Amount Rounding Precision");
                    SalesDiscSalesSummary_L.MODIFY;
                  END;
                END;
              END;

              // If the sales cr. memo line is get from a receipt line that come from a return order line then
              // the line from Sales Disc. Sales Summary corresponding to return order line must be updated
              IF SalesLine_P."Document Type" = SalesLine_P."Document Type"::"Credit Memo" THEN BEGIN
                IF (SalesLine_P."Return Receipt No." <> '') AND (SalesLine_P."Return Receipt Line No." <> 0) THEN BEGIN
                  IF ReturnRcptLine_L.GET(SalesLine_P."Return Receipt No.",SalesLine_P."Return Receipt Line No.") THEN BEGIN
                    IF SalesLine2_L.GET(SalesLine2_L."Document Type"::"Return Order",ReturnRcptLine_L."Return Order No.",
                      ReturnRcptLine_L."Return Order Line No.")
                    THEN BEGIN
                      IF SalesDiscSalesSummary_L.GET(SalesDiscSalesSummary_L."Document Type"::"Return Order",
                        SalesLine2_L."Document No.",Item_L."Item Disc. Group",SalesLine2_L."CMC Campaign No.")
                      THEN BEGIN
                        IF NOT ItemDiscountGroup_L."Not Convert UoM" THEN
                          SalesDiscSalesSummary_L.Quantity := SalesDiscSalesSummary_L.Quantity +
                            ROUND(SalesCrMemoLine_P."Quantity (Base)" /
                            cduSalesPriceDiscMgt_L.ConvertUnitofMeasure(Item_L."Base Unit of Measure",
                            ItemDiscountGroup_L."Unit of Measure Code",Item_L."No."),0.00001);
                        IF SalesCrMemoHeader_P."Currency Code" <> '' THEN
                          SalesDiscSalesSummary_L."Gross Total Amount (LCY)" :=
                            SalesDiscSalesSummary_L."Gross Total Amount (LCY)" +
                              ROUND(CurrExchRate_L.ExchangeAmtFCYToLCY(
                              SalesDiscSalesSummary_L."Document Date",SalesCrMemoHeader_P."Currency Code",
                              SalesCrMemoLine_P.Quantity * SalesCrMemoLine_P."Unit Price",SalesCrMemoHeader_P."Currency Factor"),
                              Currency_L."Amount Rounding Precision")
                        ELSE
                          SalesDiscSalesSummary_L."Gross Total Amount (LCY)" :=
                            SalesDiscSalesSummary_L."Gross Total Amount (LCY)" +
                            ROUND(SalesCrMemoLine_P.Quantity * SalesCrMemoLine_P."Unit Price",Currency_L."Amount Rounding Precision");
                        SalesDiscSalesSummary_L.MODIFY;
                      END;
                    END;
                  END;
                END;
              END;
            END;
          END;
        END;
      END;
    END;
