                  EVALUATE(TempBool,FORMAT(FieldRef.VALUE));

              'INTEGER': BEGIN
                  EVALUATE(TempInt,FORMAT(FieldRef.VALUE));
                  FieldRef.SETRANGE(TempInt);

                  TempOptionString := FieldRef.OPTIONSTRING;

                    OptionCount := 0;
                  END ELSE BEGIN
                    WHILE TempOptionString <> '' DO BEGIN
                      TestOptionString := COPYSTR(TempOptionString,1,Pos - 1);
                      TempOptionString := COPYSTR(TempOptionString,Pos + 1);
                      OptionCount += 1;
                      IF UPPERCASE(TestOptionString) = UPPERCASE(StyleSheetTableRelations."Constant Value") THEN
                        TempOptionString := ''
                      ELSE BEGIN
                        Pos := STRPOS(TempOptionString,',');
                        IF Pos = 0 THEN BEGIN
                          OptionCount += 1;
                          TempOptionString := '';
                        END;
                      END;
                    END;
                  END;
                  FieldRef.SETRANGE(OptionCount);
                END;
              'DATE': BEGIN
                  EVALUATE(TempDate,FORMAT(FieldRef.VALUE));
                  FieldRef.SETRANGE(TempDate);
                END ;
              ELSE
                FieldRef.SETRANGE(StyleSheetTableRelations."Constant Value");
            END
          ELSE BEGIN

                END ELSE
                  FieldFound := TRUE;
              END;
