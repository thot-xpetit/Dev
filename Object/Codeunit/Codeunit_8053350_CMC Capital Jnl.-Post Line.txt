    PROCEDURE CheckMemeberParts@8053003(CapJnlLine_L@8053000 : Record 8053356);
    VAR
      MemberCode_L@8053001 : Code[20];
      PartType_L@8053002 : Record 8053351;
      Text8053000_L@8053003 : TextConst 'ENU=You can''t handle more of %1 parts type %2 in possession of the member %3, Template %4 Batch %5 Line No. %6.;FRA=Vous ne pouvez manipuler plus que %1 parts de type %2 en possession de l''adh'rent %3, ModSle %4 feuille %5 Ligne n› %6.';
    BEGIN
      MemberCode_L := '';
      IF ((CapJnlLine_L."Account Type"<>CapJnlLine_L."Account Type"::Adh'rent) AND
        (CapJnlLine_L."Bal. Account Type"<>CapJnlLine_L."Bal. Account Type"::Adh'rent)) OR
        (CapJnlLine_L."Account No."='') OR (CapJnlLine_L."Bal. Account No."='') OR (CapJnlLine_L.Quantity=0) OR
        (CapJnlLine_L."Part Type"='')
      THEN
        EXIT;
      IF (CapJnlLine_L."Account Type" = CapJnlLine_L."Account Type"::G'n'ral) OR
        (CapJnlLine_L."Bal. Account Type" = CapJnlLine_L."Bal. Account Type"::G'n'ral)
      THEN BEGIN
        IF (CapJnlLine_L."Account Type" = CapJnlLine_L."Account Type"::Adh'rent) AND
          (CapJnlLine_L.Direction = CapJnlLine_L.Direction::Credit)
        THEN
          MemberCode_L := CapJnlLine_L."Account No.";
        IF (CapJnlLine_L."Bal. Account Type"= CapJnlLine_L."Bal. Account Type"::Adh'rent) AND
          (CapJnlLine_L.Direction = CapJnlLine_L.Direction::Debit)
        THEN
          MemberCode_L := CapJnlLine_L."Bal. Account No.";
      END ELSE
        CASE CapJnlLine_L.Direction OF
          CapJnlLine_L.Direction::Debit : MemberCode_L := CapJnlLine_L."Bal. Account No.";
          CapJnlLine_L.Direction::Credit :MemberCode_L := CapJnlLine_L."Account No.";
        END;

      IF MemberCode_L='' THEN
        EXIT;

      PartType_L.GET(CapJnlLine_L."Part Type");
      PartType_L.SETFILTER(PartType_L."Member Filter",MemberCode_L);
      PartType_L.CALCFIELDS(PartType_L."Part Type Count");
      IF CapJnlLine_L.Quantity>PartType_L."Part Type Count" THEN
        ERROR(STRSUBSTNO(Text8053000_L,PartType_L."Part Type Count",PartType_L."Part Code",MemberCode_L,
          CapJnlLine_L."Capital Journal Template",CapJnlLine_L."Capital Batch Name",CapJnlLine_L."Line No."));
    END;
