    PROCEDURE SetExternalWhseLine@1000000038();
    VAR
      WhseShptLine@1000000002 : Record 7321;
      SalesHeader@1000000001 : Record 36;
      Location@1000000000 : Record 14;
      SalesLine@1000000003 : Record 37;
      WhseShptHeader@1000000004 : Record 7320;
      AccreditationMgt@1000000006 : Codeunit 8053031;
      BlockingStatusMgt_L@1000000005 : Codeunit 8053022;
      WhseMgt@1000000007 : Codeunit 5775;
      WhseCreateSourceDocument@1000000008 : Codeunit 5750;
      Item@1000000009 : Record 27;
      DocCatMgt_L@1000000010 : Codeunit 8061331;
      WhseRqst@1000000012 : Record 5765;

      ReleaseWhseShptDoc@1000000013 : Codeunit 7310;
      TransferHeader@1000000017 : Record 5740;
      TransferLine@1000000016 : Record 5741;
      TransferRoute_L@1000000015 : Record 5742;
    BEGIN
      ErrorMessage_G := '';

      CASE DocType_G OF
        'COMMANDE':
          BEGIN
          SalesHeader.GET(SalesHeader."Document Type"::Order,NoSalesHeader_G);
          SalesLine.GET(SalesLine."Document Type"::Order,NoSalesHeader_G,NoSalesLine_G);

          SalesHeader.TESTFIELD(Status,SalesHeader.Status::Released);

          WhseShptLineCheck.SETRANGE("Source Type" , DATABASE::"Sales Line");
          WhseShptLineCheck.SETRANGE("Source Subtype" , SalesLine."Document Type");
          WhseShptLineCheck.SETRANGE("Source No." , SalesLine."Document No.");
          WhseShptLineCheck.SETRANGE("Source Line No." , SalesLine."Line No.");
          IF WhseShptLineCheck.FINDFIRST THEN
            ERROR(Text60040,NoSalesHeader_G,NoSalesLine_G);


          WhseShptHeader.SETRANGE("OTS Document No.",NoDocOTS_G);
          WhseShptHeader.FINDFIRST;

          IF SalesLine."Location Code" <> WhseShptHeader."Location Code" THEN
            ERROR(Text60041, WhseShptHeader."Location Code",SalesLine."Location Code");

          IF WhseShptHeader.Status = WhseShptHeader.Status::Released THEN
            ReleaseWhseShptDoc.Reopen(WhseShptHeader);

          CLEAR(WhseShptHeader);
          WhseShptHeader.SETRANGE("OTS Document No.",NoDocOTS_G);
          WhseShptHeader.FINDFIRST;

          //Insertion demande entrepot
          WhseRqst.SETRANGE(Type,WhseRqst.Type::Outbound);
          WhseRqst.SETRANGE("Source Type",DATABASE::"Sales Line");
          WhseRqst.SETRANGE("Source Subtype",SalesHeader."Document Type");
          WhseRqst.SETRANGE("Source No.",SalesHeader."No.");
          WhseRqst.SETRANGE("Document Status",WhseRqst."Document Status"::Released);
          WhseRqst.SETRANGE("Completely Handled",FALSE);
          //WhseRqst.SETRANGE("Location Code", LocationCode_G);
          IF WhseRqst.ISEMPTY AND (SalesHeader."Document Type" IN [SalesHeader."Document Type"::Order,
            SalesHeader."Document Type"::"Return Order"]) THEN
            CUWhseSalesRelease_L.Release(SalesHeader);

          //AccreditationMgt.VerifyWhseShipmentCreationAccr(SalesHeader);
          //BlockingStatusMgt_L.CheckBlockingSalesLine(SalesLine."No.",SalesHeader."Bill-to Customer No.",TRUE,FALSE);

          WITH WhseShptLine DO BEGIN
            RESET;
            "No." := WhseShptHeader."No.";
            SETRANGE("No.","No.");
            LOCKTABLE;
            IF FIND('+') THEN;

            INIT;
            "Line No." := "Line No." + 10000;

            NavWhseLineNo_G :=  "Line No.";
            NavWhseNo_G :=   WhseShptHeader."No.";

            "Source Type" := DATABASE::"Sales Line";
            "Source Subtype" := SalesLine."Document Type";
            "Source No." := SalesLine."Document No.";
            "Source Line No." := SalesLine."Line No.";
            WhseMgt.GetSourceDocument("Source Document","Source Type","Source Subtype");
            "Location Code" := SalesLine."Location Code";
            "Item No." := SalesLine."No.";
            "Variant Code" := SalesLine."Variant Code";
            SalesLine.TESTFIELD("Unit of Measure Code");
            "Unit of Measure Code" := SalesLine."Unit of Measure Code";
            "Qty. per Unit of Measure" := SalesLine."Qty. per Unit of Measure";
            Description := SalesLine.Description;
            "Description 2" := SalesLine."Description 2";
            "Build Kit" := SalesLine."Build Kit";
            Distributed := SalesLine.Distributed;
            VALIDATE(Quantity,ABS(SalesLine."Outstanding Quantity"));
            //VALIDATE(Quantity,SalesLine.Quantity);
            VALIDATE("Qty. to Ship",Quantity_G);
            IF SalesLine."Document Type" = SalesLine."Document Type"::Order THEN
              "Due Date" := SalesLine."Planned Shipment Date";
            IF SalesLine."Document Type" = SalesLine."Document Type"::"Return Order" THEN
              "Due Date" := WORKDATE;
            IF WhseShptHeader."Shipment Date" = 0D THEN
              "Shipment Date" := SalesLine."Shipment Date"
            ELSE
              "Shipment Date" := WhseShptHeader."Shipment Date";
            "Destination Type" := "Destination Type"::Customer;
            "Destination No." := SalesLine."Sell-to Customer No.";
            "Shipping Advice" := SalesHeader."Shipping Advice";
            "Bin Code" := SalesLine."Bin Code";

            "Shipment Method Code" := SalesLine."Shipment Method Code"  ;
            Consignment  := SalesLine.Consignment ;

            IF "Build Kit" THEN BEGIN
              Location.GET("Location Code");
              IF NOT Location."Directed Put-away and Pick" THEN
                UpdateShptLine(WhseShptLine,WhseShptHeader);
            END ELSE
              UpdateShptLine(WhseShptLine,WhseShptHeader);

            IF SalesLine.Type = SalesLine.Type::Item THEN
              IF Item.GET(SalesLine."No.") THEN
                "Cancel Remaining Qty" := Item."Cancel Remaining Qty";

            IF (SalesLine.Type = SalesLine.Type::Item) AND (SalesLine."Linked Order No."<>'')
              THEN "Cancel Remaining Qty" := TRUE;

            "Ship-to Code" := SalesHeader."Ship-to Code";

            IF "Cancel Remaining Qty" = FALSE THEN
              "Cancel Remaining Qty" := DocCatMgt_L.SetWhseShptLineRemCancelQty(WhseShptLine);

            CreateShptLine(WhseShptLine);
          END;
          ReleaseWhseShptDoc.Release(WhseShptHeader);
          END;
        'OT':
          BEGIN

          TransferHeader.GET(NoSalesHeader_G);
          TransferLine.GET(NoSalesHeader_G,NoSalesLine_G);

          TransferHeader.TESTFIELD(Status,TransferHeader.Status::Released);

          WhseShptLineCheck.SETRANGE("Source Type" , DATABASE::"Transfer Line");
          WhseShptLineCheck.SETRANGE("Source Subtype" , 0);
          WhseShptLineCheck.SETRANGE("Source No." , TransferLine."Document No.");
          WhseShptLineCheck.SETRANGE("Source Line No." , TransferLine."Line No.");
          IF WhseShptLineCheck.FINDFIRST THEN
            ERROR(Text60040,NoSalesHeader_G,NoSalesLine_G);

          WhseShptHeader.SETRANGE("OTS Document No.",NoDocOTS_G);
          WhseShptHeader.FINDFIRST;

          IF TransferLine."Transfer-from Code" <> WhseShptHeader."Location Code" THEN
            ERROR(Text60041, WhseShptHeader."Location Code",TransferLine."Transfer-from Code");

          IF WhseShptHeader.Status = WhseShptHeader.Status::Released THEN
            ReleaseWhseShptDoc.Reopen(WhseShptHeader);

          CLEAR(WhseShptHeader);
          WhseShptHeader.SETRANGE("OTS Document No.",NoDocOTS_G);
          WhseShptHeader.FINDFIRST;

            WITH WhseShptLine DO BEGIN
              RESET;
              "No." := WhseShptHeader."No.";
              SETRANGE("No.","No.");
              LOCKTABLE;
              IF FIND('+') THEN;

              INIT;
              "Line No." := "Line No." + 10000;

              NavWhseLineNo_G :=  "Line No.";
              NavWhseNo_G :=   WhseShptHeader."No.";

              "Source Type" := DATABASE::"Transfer Line";
              "Source No." := TransferLine."Document No.";
              "Source Line No." := TransferLine."Line No.";
              "Source Subtype" := 0;
              WhseMgt.GetSourceDocument("Source Document","Source Type","Source Subtype");
              "Item No." := TransferLine."Item No.";
              "Variant Code" := TransferLine."Variant Code";
              TransferLine.TESTFIELD("Unit of Measure Code");
              "Unit of Measure Code" := TransferLine."Unit of Measure Code";
              "Qty. per Unit of Measure" := TransferLine."Qty. per Unit of Measure";
              Description := TransferLine.Description;
              "Description 2" := TransferLine."Description 2";
              "Location Code" := TransferLine."Transfer-from Code";
              VALIDATE(Quantity,TransferLine."Outstanding Quantity");
              VALIDATE("Qty. to Ship",Quantity_G);
              "Due Date" := TransferLine."Shipment Date";
              IF WhseShptHeader."Shipment Date" = 0D THEN
                "Shipment Date" := WORKDATE
              ELSE
                "Shipment Date" := WhseShptHeader."Shipment Date";
              "Destination Type" := "Destination Type"::Location;
              "Destination No." := TransferLine."Transfer-to Code";
              //IF TransHeader.GET(TransLine."Document No.") THEN
              "Shipping Advice" := TransferHeader."Shipping Advice";
              "Bin Code" := TransferLine."Transfer-from Bin Code";

              "Shipment Method Code" :=  TransferLine."Shipment Method Code";
              Consignment :=TransferLine.Consignment;

              "Compensate Negative Inventory" := TransferLine."Compensate Negative Inventory";
              "Cancel Trsf-from Location Qty" := TransferLine."Cancel Trsf-from Location Qty";

              "Cancel Remaining Qty" := TransferLine."Cancel Remaining Qty";

              IF TransferRoute_L.GET(TransferHeader."Transfer-from Code",TransferHeader."Transfer-to Code") THEN
                "Inverted Weighing" := TransferRoute_L."Invert Shipment Weighing";

              UpdateShptLine(WhseShptLine,WhseShptHeader);
              CreateShptLine(WhseShptLine);
            END;
            ReleaseWhseShptDoc.Release(WhseShptHeader);
          END;
      END;

      ErrorMessage_G := '';
    END;

    LOCAL PROCEDURE CreateShptLine@3(VAR WhseShptLine@1001 : Record 7321);
    VAR

      WhseSourceType@1004 : ' ,Receipt,Shipment,Internal Put-away,Internal Pick,Production';
      WhseSNRequired@1006 : Boolean;
      WhseLNRequired@1007 : Boolean;
    BEGIN

      WITH WhseShptLine DO BEGIN
        Item."No." := "Item No.";
        Item.ItemSKUGet(Item,"Location Code","Variant Code");
        "Shelf No." := Item."Shelf No.";
        IF "Build Kit" THEN BEGIN
          Location.GET("Location Code");
          IF Location."Directed Put-away and Pick" THEN BEGIN
            Location.TESTFIELD("Outbound BOM Bin Code");
            "Zone Code" := '';
            "Bin Code" := '';
            VALIDATE("Bin Code",Location."Outbound BOM Bin Code");
          END;
        END;

        "Qty to Delete" := TRUE;

        INSERT;
        IF WhseShptHeader.GET("No.") THEN
          WhseShptHeader.SortWhseDoc;
        IF "Build Kit" THEN BEGIN
          KitSalesLine.SETRANGE("Document Type","Source Subtype");
          KitSalesLine.SETRANGE("Document No.","Source No.");
          KitSalesLine.SETRANGE("Document Line No.","Source Line No.");
          KitSalesLine.SETRANGE(Type,KitSalesLine.Type::Item);
          KitSalesLine.SETFILTER("Quantity per",'>0');
          IF KitSalesLine.FINDSET THEN
            REPEAT
              ItemTrackingMgt.CheckWhseItemTrkgSetup(KitSalesLine."No.",WhseSNRequired,WhseLNRequired,FALSE);
              IF WhseSNRequired OR WhseLNRequired THEN
                ItemTrackingMgt.InitItemTrkgForTempWkshLine(
                  WhseSourceType::Shipment,"No.",
                  WhseShptLine."Line No.",DATABASE::"Kit Sales Line",
                  WhseShptLine."Source Subtype","Source No.",
                  WhseShptLine."Source Line No.", KitSalesLine."Line No.");
            UNTIL KitSalesLine.NEXT = 0;
        END ELSE BEGIN
          ItemTrackingMgt.CheckWhseItemTrkgSetup("Item No.",WhseSNRequired,WhseLNRequired,FALSE);
          IF WhseSNRequired OR WhseLNRequired THEN
            ItemTrackingMgt.InitItemTrkgForTempWkshLine(
            WhseSourceType::Shipment,"No.",
            "Line No.","Source Type",
            "Source Subtype","Source No.",
            "Source Line No.",0);
        END;
      END;
    END;

        IF WhseShptHeader."Shipment Method Code" <> '' THEN
          "Shipment Method Code" := WhseShptHeader."Shipment Method Code";
      END;
    END;

    PROCEDURE DeleteWhseHeader@1000000034(NoDocOTS_P@1000000001 : Code[20]);
    VAR
      WarehouseShipmentHeader_L@1000000000 : Record 7320;
      ReleaseWhseShptDoc@1000000002 : Codeunit 7310;
    BEGIN

      WarehouseShipmentHeader_L.SETRANGE("OTS Document No.",NoDocOTS_P);
      IF WarehouseShipmentHeader_L.FINDFIRST THEN BEGIN
        IF WarehouseShipmentHeader_L.Status = WarehouseShipmentHeader_L.Status::Released THEN
          ReleaseWhseShptDoc.Reopen(WarehouseShipmentHeader_L);

        CLEAR(WarehouseShipmentHeader_L);
        WarehouseShipmentHeader_L.SETRANGE("OTS Document No.",NoDocOTS_P);
        WarehouseShipmentHeader_L.FINDFIRST;

        WarehouseShipmentHeader_L.SetControlOtsDelete(TRUE);
        WarehouseShipmentHeader_L.DELETE(TRUE);
      END;
    END;
