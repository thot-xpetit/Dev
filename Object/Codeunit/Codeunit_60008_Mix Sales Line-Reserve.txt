    PROCEDURE CreateReservation@1(VAR MixSalesLine_P@1000 : Record 60032;Description@1001 : Text[50];ExpectedReceiptDate@1002 : Date;Quantity@1003 : Decimal;ForSerialNo@1005 : Code[20];ForLotNo@1004 : Code[20]);
    VAR
      ShipmentDate@1007 : Date;
      OutBoundQty@1009 : Decimal;
    BEGIN
      IF SetFromType = 0 THEN
        ERROR(Text005);

      MixSalesLine_P.TESTFIELD(Type,MixSalesLine_P.Type::Item);
      MixSalesLine_P.TESTFIELD("No.");
      MixSalesLine_P.TESTFIELD("Shipment Date");
      MixSalesLine_P.CALCFIELDS("Reserved Qty. (Base)");
      IF ABS(MixSalesLine_P."Outstanding Qty. (Base)") < ABS(MixSalesLine_P."Reserved Qty. (Base)") + Quantity THEN
        ERROR(
          Text000,
          ABS(MixSalesLine_P."Outstanding Qty. (Base)") - ABS(MixSalesLine_P."Reserved Qty. (Base)"));

      MixSalesLine_P.TESTFIELD("Variant Code",SetFromVariantCode);
      MixSalesLine_P.TESTFIELD("Location Code",SetFromLocationCode);

      ShipmentDate := MixSalesLine_P."Shipment Date";

      CreateReservEntry.CreateReservEntryFor(
        DATABASE::"Mix Sales Line",MixSalesLine_P."Document Type",
        MixSalesLine_P."Document No.",'',MixSalesLine_P."Document Line No.",
        MixSalesLine_P."Line No.",MixSalesLine_P."Qty. per Unit of Measure",
        Quantity,ForSerialNo,ForLotNo);
      CreateReservEntry.CreateReservEntryFrom(
        SetFromType,SetFromSubtype,SetFromID,SetFromBatchName,SetFromProdOrderLine,SetFromRefNo,
        SetFromQtyPerUOM,SetFromSerialNo,SetFromLotNo);
      CreateReservEntry.CreateReservEntry(
        MixSalesLine_P."No.",MixSalesLine_P."Variant Code",MixSalesLine_P."Location Code",
        Description,ExpectedReceiptDate,ShipmentDate,0);

      IF Location.GET(MixSalesLine_P."Location Code") AND
         (Location."Bin Mandatory" OR Location."Require Pick")
      THEN BEGIN
        OutBoundQty :=
          CreatePick.CheckOutBound(
            DATABASE::"Mix Sales Line",MixSalesLine_P."Document Type",
            MixSalesLine_P."Document No.",MixSalesLine_P."Document Line No.",MixSalesLine_P."Line No.");
        IF OutBoundQty <> 0 THEN
          CreatePick.AdjustReservation(
            OutBoundQty,DATABASE::"Mix Sales Line",MixSalesLine_P."Document Type",
            MixSalesLine_P."Document No.",MixSalesLine_P."Document Line No.",MixSalesLine_P."Line No.",0);
      END;

      SetFromType := 0;
    END;

    PROCEDURE VerifyChange@59(VAR NewMixSalesLine_P@1000 : Record 60032;VAR OldMixSalesLine_P@1001 : Record 60032);
    VAR
      KitSalesLine@1005 : Record 25000;
      TempReservEntry@1002 : Record 337;
      ShowError@1003 : Boolean;
      HasError@1004 : Boolean;
    BEGIN
      IF (NewMixSalesLine_P.Type <> NewMixSalesLine_P.Type::Item) AND (OldMixSalesLine_P.Type <> OldMixSalesLine_P.Type::Item) THEN
        EXIT;
      IF NewMixSalesLine_P."Line No." = 0 THEN
        IF NOT NewMixSalesLine_P.GET(
                 NewMixSalesLine_P."Document Type",NewMixSalesLine_P."Document No.",
                 NewMixSalesLine_P."Document Line No.",NewMixSalesLine_P."Line No.")
        THEN
          EXIT;

      NewMixSalesLine_P.CALCFIELDS("Reserved Qty. (Base)");
      ShowError := NewMixSalesLine_P."Reserved Qty. (Base)" <> 0;

      IF (NewMixSalesLine_P."Shipment Date" = 0D) AND (OldMixSalesLine_P."Shipment Date" <> 0D) THEN
        IF ShowError THEN
          NewMixSalesLine_P.FIELDERROR("Shipment Date",Text002)
        ELSE
          HasError := TRUE;

      IF NewMixSalesLine_P."No." <> OldMixSalesLine_P."No." THEN
        IF ShowError THEN
          NewMixSalesLine_P.FIELDERROR("No.",Text004)
        ELSE
          HasError := TRUE;

      IF NewMixSalesLine_P."Variant Code" <> OldMixSalesLine_P."Variant Code" THEN
        IF ShowError THEN
          NewMixSalesLine_P.FIELDERROR("Variant Code",Text004)
        ELSE
          HasError := TRUE;

      IF NewMixSalesLine_P."Location Code" <> OldMixSalesLine_P."Location Code" THEN
        IF ShowError THEN
          NewMixSalesLine_P.FIELDERROR("Location Code",Text004)
        ELSE
          HasError := TRUE;

      IF NewMixSalesLine_P."Bin Code" <> OldMixSalesLine_P."Bin Code" THEN
        IF ShowError THEN
          NewMixSalesLine_P.FIELDERROR("Bin Code",Text004)
        ELSE
          HasError := TRUE;

      IF NewMixSalesLine_P."Line No." <> OldMixSalesLine_P."Line No." THEN
        HasError := TRUE;

      IF HasError THEN
        IF (NewMixSalesLine_P."No." <> OldMixSalesLine_P."No.") OR
            FindReservEntry(NewMixSalesLine_P,TempReservEntry)
        THEN BEGIN
          IF NewMixSalesLine_P."No." <> OldMixSalesLine_P."No." THEN BEGIN
            ReservMgt.SetMixSalesLine(OldMixSalesLine_P);
            ReservMgt.DeleteReservEntries(TRUE,0);
            ReservMgt.SetMixSalesLine(NewMixSalesLine_P);
          END ELSE BEGIN
            ReservMgt.SetMixSalesLine(NewMixSalesLine_P);
            ReservMgt.DeleteReservEntries(TRUE,0);
          END;
          ReservMgt.AutoTrack(NewMixSalesLine_P."Outstanding Qty. (Base)");
        END;
    END;

    PROCEDURE VerifyQuantity@2(VAR NewMixSalesLine_P@1000 : Record 60032;VAR OldMixSalesLine_P@1001 : Record 60032);
    VAR
      KitSalesLine@1002 : Record 25000;
      SalesLine@1480000 : Record 37;
    BEGIN
      WITH NewMixSalesLine_P DO BEGIN
        IF NOT ("Document Type" IN
                ["Document Type"::Quote,"Document Type"::Order])
        THEN BEGIN
          SalesLine.GET("Document Type","Document No.","Document Line No.");
          IF SalesLine."Shipment No." = '' THEN
            EXIT;
        END;
        IF Type <> Type::Item THEN
          EXIT;
        IF "Line No." = OldMixSalesLine_P."Line No." THEN
          IF "Extended Quantity (Base)" = OldMixSalesLine_P."Extended Quantity (Base)" THEN
            EXIT;
        IF "Line No." = 0 THEN
          IF NOT KitSalesLine.GET("Document Type","Document No.","Document Line No.","Line No.") THEN
            EXIT;
        ReservMgt.SetMixSalesLine(NewMixSalesLine_P);
        IF "Qty. per Unit of Measure" <> OldMixSalesLine_P."Qty. per Unit of Measure" THEN
          ReservMgt.ModifyUnitOfMeasure;
        IF "Outstanding Qty. (Base)" * OldMixSalesLine_P."Outstanding Qty. (Base)" < 0 THEN
          ReservMgt.DeleteReservEntries(TRUE,0)
        ELSE
          ReservMgt.DeleteReservEntries(FALSE,"Outstanding Qty. (Base)");
        ReservMgt.ClearSurplus;
        ReservMgt.AutoTrack("Outstanding Qty. (Base)");
      END;
    END;

    PROCEDURE TransfKitSalLineToKitSalesLine@31(VAR OldMixSalesLine_P@1000 : Record 60032;VAR NewMixSalesLine_P@1001 : Record 60032;TransferQty_P@1002 : Decimal);
    VAR
      OldReservEntry@1003 : Record 337;
      Status@1004 : 'Reservation,Tracking,Surplus,Prospect';
    BEGIN
      // Used for sales quote and blanket order when transferred to order
      IF NOT FindReservEntry(OldMixSalesLine_P,OldReservEntry) THEN
        EXIT;

      OldReservEntry.Lock;

      NewMixSalesLine_P.TESTFIELD("No.",OldMixSalesLine_P."No.");
      NewMixSalesLine_P.TESTFIELD("Variant Code",OldMixSalesLine_P."Variant Code");
      NewMixSalesLine_P.TESTFIELD("Location Code",OldMixSalesLine_P."Location Code");

      FOR Status := Status::Reservation TO Status::Prospect DO BEGIN
        IF TransferQty_P = 0 THEN
          EXIT;
        OldReservEntry.SETRANGE("Reservation Status",Status);
        IF OldReservEntry.FINDSET THEN
          REPEAT
            OldReservEntry.TESTFIELD("Item No.",OldMixSalesLine_P."No.");
            OldReservEntry.TESTFIELD("Variant Code",OldMixSalesLine_P."Variant Code");
            OldReservEntry.TESTFIELD("Location Code",OldMixSalesLine_P."Location Code");

            TransferQty_P := CreateReservEntry.TransferReservEntry(DATABASE::"Mix Sales Line",
              NewMixSalesLine_P."Document Type",NewMixSalesLine_P."Document No.",'',NewMixSalesLine_P."Document Line No.",
              NewMixSalesLine_P."Line No.",NewMixSalesLine_P."Qty. per Unit of Measure",OldReservEntry,TransferQty_P);

          UNTIL (OldReservEntry.NEXT = 0) OR (TransferQty_P = 0);
      END; // DO
    END;

    PROCEDURE TransfKitSaLineToKitBOMJnlLine@1480000(VAR MixSalesLine_P@1000 : Record 60032;VAR KitBOMJnlLine_P@1001 : Record 25001;TransferQty_P@1002 : Decimal) : Decimal;
    VAR
      OldReservEntry@1003 : Record 337;
    BEGIN
      IF NOT FindReservEntry(MixSalesLine_P,OldReservEntry) THEN
        EXIT(TransferQty_P);

      OldReservEntry.Lock;

      CLEAR(CreateReservEntry);
      KitBOMJnlLine_P.TESTFIELD("No.",MixSalesLine_P."No.");
      KitBOMJnlLine_P.TESTFIELD("Variant Code",MixSalesLine_P."Variant Code");
      KitBOMJnlLine_P.TESTFIELD("Location Code",MixSalesLine_P."Location Code");

      IF TransferQty_P = 0 THEN
        EXIT;

      IF ReservEngineMgt.InitRecordSet(OldReservEntry) THEN BEGIN
        REPEAT
          OldReservEntry.TESTFIELD("Item No.",MixSalesLine_P."No.");
          OldReservEntry.TESTFIELD("Variant Code",MixSalesLine_P."Variant Code");
          OldReservEntry.TESTFIELD("Location Code",MixSalesLine_P."Location Code");

          TransferQty_P := CreateReservEntry.TransferReservEntry(DATABASE::"Kit BOM Journal Line",
            0,KitBOMJnlLine_P."Journal Template Name",
            KitBOMJnlLine_P."Journal Batch Name",KitBOMJnlLine_P."Journal Line No.",KitBOMJnlLine_P."Line No.",
            KitBOMJnlLine_P."Qty. per Unit of Measure",OldReservEntry,TransferQty_P);

        UNTIL (ReservEngineMgt.NEXTRecord(OldReservEntry) = 0) OR (TransferQty_P = 0);
      END;
      EXIT(TransferQty_P);
    END;

    PROCEDURE DeleteLineConfirm@20(VAR MixSalesLine_P@1000 : Record 60032) : Boolean;
    BEGIN
      WITH MixSalesLine_P DO BEGIN
        IF NOT ReservEntryExist(MixSalesLine_P) THEN
          EXIT(TRUE);

        ReservMgt.SetMixSalesLine(MixSalesLine_P);
        IF ReservMgt.DeleteItemTrackingConfirm THEN
          DeleteItemTracking := TRUE;
      END;

      EXIT(DeleteItemTracking);
    END;

    PROCEDURE DeleteLine@3(VAR MixSalesLine_P@1000 : Record 60032);
    BEGIN
      IF NOT ReservEntryExist(MixSalesLine_P) THEN
        EXIT;

      WITH MixSalesLine_P DO BEGIN
        ReservMgt.SetMixSalesLine(MixSalesLine_P);
        ReservMgt.SetItemTrackingHandling(1); // Allow Deletion
        ReservMgt.DeleteReservEntries(TRUE,0);
        CALCFIELDS("Reserved Qty. (Base)");
      END;
    END;

    PROCEDURE CallItemTracking@6(VAR MixSalesLine_P@1000 : Record 60032);
    VAR
      TrackingSpecification@1001 : Record 336;
      ItemTrackingForm@1002 : Form 6510;
    BEGIN
      InitTrackingSpecification(MixSalesLine_P,TrackingSpecification);
      ItemTrackingForm.SetSource(TrackingSpecification,MixSalesLine_P."Shipment Date");

        (MixSalesLine_P."Document Type" IN

      ItemTrackingForm.RUNMODAL;
    END;
