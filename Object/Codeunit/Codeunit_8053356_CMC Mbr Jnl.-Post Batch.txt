    PROCEDURE Code@8053001();
    VAR
      Text8053000_L@8053000 : TextConst 'ENU=Do you want to post the journal lines?;FRA=Souhaitez-vous valider les lignes de la feuille ?';
      Text8053001_L@8053003 : TextConst 'ENU=There is nothing to post.;FRA=Il n''y a rien . valider.';
      Text8053002_L@8053002 : TextConst 'ENU=The journal lines were successfully posted.;FRA=Les lignes de la feuille ont 't' valid'es avec succSs.';
      Text8053003_L@8053001 : TextConst 'ENU=The journal lines were successfully posted. You are now in the %1 journal.;FRA=Les lignes de la feuille ont 't' valid'es avec succSs. Vous vous trouvez maintenant dans la feuille %1.';
    BEGIN
      WITH MbrJnlLine_G DO
        BEGIN
          IF NOT CONFIRM(Text8053000_L,FALSE) THEN
            EXIT;

          MbrJnlLine_G.SETRANGE(MbrJnlLine_G."Accept Action Message",TRUE);
          MbrJnlTemplate_G.GET(MbrJnlLine_G."Capital Journal Template");
          TempCapBatchName_G := "Capital Batch Name";
        IF (NOT MbrJnlLine_G.FINDFIRST) THEN BEGIN
              MESSAGE(Text8053001_L);
              EXIT;
            END;
          MbrJnlLine4_G.COPY(MbrJnlLine_G);
          PostBatch(MbrJnlLine4_G);
          IF ("Line No." = 0) THEN
            MESSAGE(Text8053001_L)
          ELSE
            IF TempCapBatchName_G = "Capital Batch Name" THEN
              MESSAGE(Text8053002_L)
            ELSE
            MESSAGE(Text8053003_L,"Capital Batch Name");
        IF NOT FIND('=><') OR (TempCapBatchName_G <> "Capital Batch Name") THEN BEGIN
             RESET;
             FILTERGROUP(2);
             SETRANGE("Capital Journal Template","Capital Journal Template");
             SETRANGE("Capital Batch Name","Capital Batch Name");
             FILTERGROUP(0);
             "Line No." := 1;
            END;
        END;
    END;

    PROCEDURE PostBatch@8053000(VAR MbrJnlLine_P@8053004 : Record 8053356);
    VAR
      Text8053000_L@8053000 : TextConst 'ENU=cannot exceed%1 characters;FRA=ne peut pas d'passer % caractSres';
      Text8053001_L@8053001 : TextConst 'ENU=Capital Batch Name    #1##########\\;FRA=Nom feuille                     #1##########\\';
      Text8053002_L@8053002 : TextConst 'ENU=Checking lines        #2######\;FRA=V'rification des lignes         #2######\';
      Text8053003_L@8053003 : TextConst 'ENU=Posting lines         #5###### @6@@@@@@@@@@@@@;FRA=Validation des lignes           #5###### @6@@@@@@@@@@@@@';
    BEGIN
      MbrSetup_G.GET;
      WITH MbrJnlLine_P DO
        BEGIN
          MbrJnlLine_P.SETRANGE("Capital Journal Template","Capital Journal Template");
          MbrJnlLine_P.SETRANGE("Capital Batch Name","Capital Batch Name");
          IF RECORDLEVELLOCKING THEN
            LOCKTABLE;
          MbrJnlTemplate_G.GET("Capital Journal Template");
          MbrJnlBatch_G.GET(MbrJnlLine_P."Capital Journal Template",MbrJnlLine_P."Capital Batch Name");
          IF STRLEN(INCSTR(MbrJnlBatch_G.Name)) > MAXSTRLEN(MbrJnlBatch_G.Name) THEN
          MbrJnlBatch_G.FIELDERROR(Name,STRSUBSTNO(Text8053000_L,MAXSTRLEN(MbrJnlBatch_G.Name)));

          IF NOT FIND('=><') THEN BEGIN
            "Line No." := 0;
            COMMIT;
            EXIT;
          END;

        IF NOT WinDialog_G THEN BEGIN
          IF ShowDialog() THEN BEGIN
            Window_G.OPEN(Text8053001_L +Text8053002_L +Text8053003_L);
                  Window_G.UPDATE(1,"Capital Batch Name");
                END;
            END;
          // check lines
          LineCount_G :=0;
          StartLineNo_G := "Line No.";
          REPEAT
            LineCount_G := LineCount_G + 1;
            IF NOT WinDialog_G THEN
              IF ShowDialog() THEN
                Window_G.UPDATE(2,LineCount_G);
            MbrJnlLine1_G.COPY(MbrJnlLine_P);
            MbrJnlCheckLine_G.RunCheck(MbrJnlLine1_G);
            IF NEXT = 0 THEN
              FINDFIRST;
          UNTIL "Line No." = StartLineNo_G;
          NoOfRecords_G := LineCount_G;

          // Post lines
          LineCount_G := 0;
          LastDocNo_G := '';
          LastPostedDocNo_G := '';
          FINDSET(TRUE,FALSE);
          REPEAT
          IF NOT WinDialog_G THEN BEGIN
            IF ShowDialog() THEN BEGIN
                    Window_G.UPDATE(5,LineCount_G);
                    Window_G.UPDATE(6,ROUND(LineCount_G / NoOfRecords_G * 10000,1));
                  END;
              END;
            CLEAR(MbrJnlCheckLine_G);
            MbrJnlCheckLine_G.RunWithoutCheck(MbrJnlLine_P);
          UNTIL NEXT=0;

          // update/delete lines
          MbrJnlLine3_G.COPY(MbrJnlLine_P);
          MbrJnlLine3_G.SETFILTER("Account No.",'<>%1','');
          IF MbrJnlLine3_G.FINDLAST THEN; // Remember the last line
          MbrJnlLine1_G.COPY(MbrJnlLine_P);
          IF MbrJnlLine1_G.FINDSET(TRUE,FALSE) THEN
            REPEAT
              MbrJnlLine1_G.DELETE;
            UNTIL MbrJnlLine1_G.NEXT = 0;
          MbrJnlLine1_G.RESET;
          MbrJnlLine1_G.SETRANGE("Capital Journal Template","Capital Journal Template");
          MbrJnlLine1_G.SETRANGE("Capital Batch Name","Capital Batch Name");
          IF NOT MbrJnlLine1_G.FINDLAST THEN
          IF INCSTR("Capital Batch Name") <> '' THEN BEGIN
                MbrJnlBatch_G.DELETE;
                MbrJnlBatch_G.Name := INCSTR("Capital Batch Name");
                IF MbrJnlBatch_G.INSERT THEN;
                "Capital Batch Name" := MbrJnlBatch_G.Name;
              END;
          MbrJnlLine1_G.SETRANGE("Capital Batch Name","Capital Batch Name");
        IF (MbrJnlBatch_G."No. Series" = '') AND NOT MbrJnlLine1_G.FINDLAST THEN BEGIN
              MbrJnlLine1_G.INIT;
              MbrJnlLine1_G."Capital Journal Template" := "Capital Journal Template";
              MbrJnlLine1_G."Capital Batch Name" := "Capital Batch Name";
              MbrJnlLine1_G."Line No." := 10000;
              MbrJnlLine1_G.INSERT;
              TempMbrJnlLine_G.DELETEALL;
              TempMbrJnlLine_G := MbrJnlLine3_G;
              MbrJnlLine1_G.SetUpNewLine(TempMbrJnlLine_G,TRUE);
              MbrJnlLine1_G.MODIFY;
            END;
        END;
      COMMIT;
    END;
