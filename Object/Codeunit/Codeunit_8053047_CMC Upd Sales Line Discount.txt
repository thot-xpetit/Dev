    PROCEDURE ItemReturnCheck@8053001(ItemNo_P@8053002 : Code[20]) : Boolean;
    VAR
      UserSetup_L@8053000 : Record 91;
      Found_L@8053003 : Boolean;
      Item_L@8053001 : Record 27;
    BEGIN
      UserSetup_L.GET(USERID);
      UserSetup_L.TESTFIELD("User Groupe Code");

      ItemReturnSetup_G.RESET;
      ItemReturnSetup_G.SETRANGE("User Group Code",UserSetup_L."User Groupe Code");
      ItemReturnSetup_G.SETRANGE(Type,ItemReturnSetup_G.Type::Item);
      ItemReturnSetup_G.SETRANGE("No.",ItemNo_P);
      Found_L := ItemReturnSetup_G.FINDFIRST;

      IF NOT Item_L.GET(ItemNo_P) THEN
       EXIT(FALSE);

      IF NOT Found_L THEN BEGIN
        ItemReturnSetup_G.RESET;
        ItemReturnSetup_G.SETRANGE("User Group Code",UserSetup_L."User Groupe Code");
        ItemReturnSetup_G.SETRANGE(Type,ItemReturnSetup_G.Type::Species);
        ItemReturnSetup_G.SETRANGE("No.",Item_L."Species Code");
        Found_L := ItemReturnSetup_G.FINDFIRST;
      END;

      IF NOT Found_L THEN BEGIN
        ItemReturnSetup_G.RESET;
        ItemReturnSetup_G.SETRANGE("User Group Code",UserSetup_L."User Groupe Code");
        ItemReturnSetup_G.SETRANGE(Type,ItemReturnSetup_G.Type::Subfamily);
        ItemReturnSetup_G.SETRANGE("No.",Item_L."Sub-Family Code");
        Found_L := ItemReturnSetup_G.FINDFIRST;
      END;

      IF NOT Found_L THEN BEGIN
        ItemReturnSetup_G.RESET;
        ItemReturnSetup_G.SETRANGE("User Group Code",UserSetup_L."User Groupe Code");
        ItemReturnSetup_G.SETRANGE(Type,ItemReturnSetup_G.Type::Family);
        ItemReturnSetup_G.SETRANGE("No.",Item_L."Family Code");
        Found_L := ItemReturnSetup_G.FINDFIRST;
      END;

      IF NOT Found_L THEN BEGIN
        ItemReturnSetup_G.RESET;
        ItemReturnSetup_G.SETRANGE("User Group Code",UserSetup_L."User Groupe Code");
        ItemReturnSetup_G.SETRANGE(Type,ItemReturnSetup_G.Type::Activity);
        ItemReturnSetup_G.SETRANGE("No.",Item_L."Activity Code");
        Found_L := ItemReturnSetup_G.FINDFIRST;
      END;

      EXIT(Found_L);
    END;
