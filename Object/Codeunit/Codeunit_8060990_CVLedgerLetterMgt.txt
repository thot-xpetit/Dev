      // --> 20141017- CMB6.00.01.01-SDE01
      FindApplnEntriesDtldtLedgEntry(CustLedgerEntry_P,ApplCustLedgEntry_P);
      // <-- 20141017- CMB6.00.01.01-SDE01

        REPEAT
          ApplCustLedgEntry_P.MARK(TRUE);

      ApplCustLedgEntry_P.SETRANGE("Closed by Entry No.");

      TMP_VendLedgerEntry_L@60001 : TEMPORARY Record 25;
      TMP_CustLedgerEntry_L@60000 : TEMPORARY Record 21;
    BEGIN
      AppliedToLetter_L := '';

      IF GenJnlLine_P."Account Type" = GenJnlLine_P."Account Type"::Customer THEN BEGIN
        IF (OldCVLedgEntryBuf_P.Letter <> '') AND (NewCVLedgEntryBuf_P.Letter <> '') AND
           (OldCVLedgEntryBuf_P.Letter <> NewCVLedgEntryBuf_P.Letter) THEN BEGIN
          CustLedgerEntry_L.RESET;
          CustLedgerEntry_L.SETCURRENTKEY("Customer No.",Letter);
          CustLedgerEntry_L.SETRANGE("Customer No.",OldCVLedgEntryBuf_P."CV No.");
          CustLedgerEntry_L.SETRANGE(Letter,OldCVLedgEntryBuf_P.Letter);
          IF NOT CustLedgerEntry_L.ISEMPTY THEN BEGIN
             CustLedgerEntry_L.FIND('-');
             TMP_CustLedgerEntry_L.DELETEALL;
            REPEAT
               IF NOT (((CustLedgerEntry_L."Applies-to ID" = NewCVLedgEntryBuf_P."Applies-to ID") AND CustLedgerEntry_L.Open) OR
                        (CustLedgerEntry_L."Entry No." = OldCVLedgEntryBuf_P."Entry No.") )
               THEN BEGIN
                  // On ne mettra pas . jour les 'critures qui sont lettr'es ainsi que l''criture en cours
                  TMP_CustLedgerEntry_L.INIT;
                  TMP_CustLedgerEntry_L.TRANSFERFIELDS(CustLedgerEntry_L);
                  TMP_CustLedgerEntry_L.INSERT;
               END;
            UNTIL CustLedgerEntry_L.NEXT = 0;
            // passage par table temporaire pour 'viter mise . jour sur la cl' active
            TMP_CustLedgerEntry_L.RESET;
            CustLedgerEntry_L.RESET;
            IF NOT TMP_CustLedgerEntry_L.ISEMPTY THEN BEGIN
            // --> 20141017- CMB6.00.01.01-SDE01
              TMP_CustLedgerEntry_L.FINDSET;
            // <-- 20141017- CMB6.00.01.01-SDE01
               REPEAT
                  IF CustLedgerEntry_L.GET(TMP_CustLedgerEntry_L."Entry No.") THEN BEGIN
                     CustLedgerEntry_L.Letter := NewCVLedgEntryBuf_P.Letter;
                     CustLedgerEntry_L.MODIFY;
                  END;
               UNTIL TMP_CustLedgerEntry_L.NEXT = 0;
            END;
          END;
        END;

        IF OldCVLedgEntryBuf_P.Letter <> '' THEN
          AppliedToLetter_L := OldCVLedgEntryBuf_P.Letter;
        IF NewCVLedgEntryBuf_P.Letter <> '' THEN
          AppliedToLetter_L := NewCVLedgEntryBuf_P.Letter;
        IF AppliedToLetter_L = '' THEN
          BEGIN
            AppliedToLetter_L := CustLastLetter(NewCVLedgEntryBuf_P."CV No.");
            LetterApplMngt_L.NextLetter(AppliedToLetter_L);
          END;
      END;

      IF GenJnlLine_P."Account Type" = GenJnlLine_P."Account Type"::Vendor THEN BEGIN
        IF (OldCVLedgEntryBuf_P.Letter <> '') AND (NewCVLedgEntryBuf_P.Letter <> '') AND
           (OldCVLedgEntryBuf_P.Letter <> NewCVLedgEntryBuf_P.Letter) THEN BEGIN
          VendLedgerEntry_L.RESET;
          VendLedgerEntry_L.SETCURRENTKEY("Vendor No.",Letter);
          VendLedgerEntry_L.SETRANGE("Vendor No.",OldCVLedgEntryBuf_P."CV No.");
          VendLedgerEntry_L.SETRANGE(Letter,OldCVLedgEntryBuf_P.Letter);
          IF NOT VendLedgerEntry_L.ISEMPTY THEN BEGIN
             VendLedgerEntry_L.FIND('-');
             TMP_VendLedgerEntry_L.DELETEALL;
             REPEAT
               IF NOT (((VendLedgerEntry_L."Applies-to ID" = NewCVLedgEntryBuf_P."Applies-to ID") AND VendLedgerEntry_L.Open ) OR
                        (VendLedgerEntry_L."Entry No." = OldCVLedgEntryBuf_P."Entry No.") )
               THEN BEGIN
                  // On ne mettra pas . jour les 'critures qui sont lettr'es ainsi que l''criture en cours
                  TMP_VendLedgerEntry_L.INIT;
                  TMP_VendLedgerEntry_L.TRANSFERFIELDS(VendLedgerEntry_L);
                  TMP_VendLedgerEntry_L.INSERT;
               END;
             UNTIL VendLedgerEntry_L.NEXT = 0;
             // passage par table temporaire pour 'viter mise . jour sur la cl' active
             TMP_VendLedgerEntry_L.RESET;
             VendLedgerEntry_L.RESET;
             IF NOT TMP_VendLedgerEntry_L.ISEMPTY THEN BEGIN
             // --> 20141017- CMB6.00.01.01-SDE01
              TMP_VendLedgerEntry_L.FINDSET;
             // --> 20141017- CMB6.00.01.01-SDE01
                REPEAT
                  IF VendLedgerEntry_L.GET(TMP_VendLedgerEntry_L."Entry No.") THEN BEGIN
                     VendLedgerEntry_L.Letter := NewCVLedgEntryBuf_P.Letter;
                     VendLedgerEntry_L.MODIFY;
                  END;
                UNTIL TMP_VendLedgerEntry_L.NEXT = 0;
             END;
          END;

        END;

        IF OldCVLedgEntryBuf_P.Letter <> '' THEN
          AppliedToLetter_L := OldCVLedgEntryBuf_P.Letter;
        IF NewCVLedgEntryBuf_P.Letter <> '' THEN
          AppliedToLetter_L := NewCVLedgEntryBuf_P.Letter;
        IF AppliedToLetter_L = '' THEN
          BEGIN
            AppliedToLetter_L := VendLastLetter(NewCVLedgEntryBuf_P."CV No.");
            LetterApplMngt_L.NextLetter(AppliedToLetter_L);
          END;
      END;

      NewCVLedgEntryBuf_P.Letter := AppliedToLetter_L;
      OldCVLedgEntryBuf_P.Letter := AppliedToLetter_L;
    END;

    PROCEDURE CustLetterApplyEntry@1000000005(CustLedgEntry_P@1000000000 : Record 21);
    VAR
      ApplCustLedgEntry_L@1000000002 : Record 21;
      AppliesToID_L@1000000001 : Text[20];
      ExistAppliedToID_L@1000000004 : Text[20];
      LetterApplMngt_L@1180250002 : Codeunit 10842;
      ApplCustLedgEntry2_L@1000000003 : Record 21;
    BEGIN
      AppliesToID_L := CustLastLetter(CustLedgEntry_P."Customer No.");

      CustLedgEntry_P.CALCFIELDS("Amount (LCY)","Remaining Amt. (LCY)");

      IF (CustLedgEntry_P."Amount (LCY)" <> CustLedgEntry_P."Remaining Amt. (LCY)") THEN
        BEGIN

          CustGetAppliedEntry(CustLedgEntry_P,ApplCustLedgEntry_L);

          IF ApplCustLedgEntry_L.FIND('-') THEN
            ExistAppliedToID_L := ApplCustLedgEntry_L.Letter;

          IF CustLedgEntry_P.Letter = '' THEN
            BEGIN
              IF ExistAppliedToID_L = '' THEN
                LetterApplMngt_L.NextLetter(AppliesToID_L)
              ELSE
                AppliesToID_L := ExistAppliedToID_L;
              CustLedgEntry_P.Letter := AppliesToID_L;
              CustLedgEntry_P.MODIFY;
            END
          ELSE
            AppliesToID_L := CustLedgEntry_P.Letter;

          // --> 20141017- CMB6.00.01.01-SDE01
          IF ApplCustLedgEntry_L.FINDSET THEN
            REPEAT
              IF (ApplCustLedgEntry_L.Letter<>'') AND (ApplCustLedgEntry_L.Letter<>AppliesToID_L)
              THEN BEGIN
                ApplCustLedgEntry2_L.RESET;
                ApplCustLedgEntry2_L.SETRANGE("Customer No.",ApplCustLedgEntry_L."Customer No.");
                ApplCustLedgEntry2_L.SETRANGE(Letter,ApplCustLedgEntry_L.Letter);
                ApplCustLedgEntry2_L.MODIFYALL(Letter,AppliesToID_L);
              END ELSE BEGIN
                ApplCustLedgEntry_L.Letter := AppliesToID_L;
                ApplCustLedgEntry_L.MODIFY;
              END;
            // <-- 20141017- CMB6.00.01.01-SDE01
            UNTIL ApplCustLedgEntry_L.NEXT = 0;

        END;
    END;

    PROCEDURE FindApplnEntriesDtldtLedgEntry@1(CreateCustLedgEntry@1000000001 : Record 21;VAR ApplCustLedgEntry_P@1000000000 : Record 21);
    VAR
      DtldCustLedgEntry1@1001 : Record 379;
      DtldCustLedgEntry2@1000 : Record 379;
    BEGIN
      // --> 20141017- CMB6.00.01.01-SDE01
      DtldCustLedgEntry1.SETCURRENTKEY("Cust. Ledger Entry No.");
      DtldCustLedgEntry1.SETRANGE("Cust. Ledger Entry No.",CreateCustLedgEntry."Entry No.");
      DtldCustLedgEntry1.SETRANGE(Unapplied,FALSE);
      IF DtldCustLedgEntry1.FIND('-') THEN BEGIN
        REPEAT
          IF DtldCustLedgEntry1."Cust. Ledger Entry No." =
            DtldCustLedgEntry1."Applied Cust. Ledger Entry No."
          THEN BEGIN
            DtldCustLedgEntry2.INIT;
            DtldCustLedgEntry2.SETCURRENTKEY("Applied Cust. Ledger Entry No.","Entry Type");
            DtldCustLedgEntry2.SETRANGE(
              "Applied Cust. Ledger Entry No.",DtldCustLedgEntry1."Applied Cust. Ledger Entry No.");
            DtldCustLedgEntry2.SETRANGE("Entry Type",DtldCustLedgEntry2."Entry Type"::Application);
            DtldCustLedgEntry2.SETRANGE(Unapplied,FALSE);
            IF DtldCustLedgEntry2.FIND('-') THEN BEGIN
              REPEAT
                IF DtldCustLedgEntry2."Cust. Ledger Entry No." <>
                  DtldCustLedgEntry2."Applied Cust. Ledger Entry No."
                THEN BEGIN
                  ApplCustLedgEntry_P.SETCURRENTKEY("Entry No.");
                  ApplCustLedgEntry_P.SETRANGE("Entry No.",DtldCustLedgEntry2."Cust. Ledger Entry No.");

                    ApplCustLedgEntry_P.MARK(TRUE);
                END;
              UNTIL DtldCustLedgEntry2.NEXT = 0;
            END;
          END ELSE BEGIN
            ApplCustLedgEntry_P.SETCURRENTKEY("Entry No.");
            ApplCustLedgEntry_P.SETRANGE("Entry No.",DtldCustLedgEntry1."Applied Cust. Ledger Entry No.");
            IF ApplCustLedgEntry_P.FIND('-') THEN
              ApplCustLedgEntry_P.MARK(TRUE);
          END;
        UNTIL DtldCustLedgEntry1.NEXT = 0;
      END;
      ApplCustLedgEntry_P.SETRANGE("Entry No.");
    END;
