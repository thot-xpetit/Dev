    PROCEDURE ImportTarifTexte@1000000001(FileName_P@1000000000 : Text[256]) Result_L : Text[256];
    VAR
      "<-- 3LI LOCALS -->"@1000000002 : Integer;
      TableSasImport_L@1000000006 : Record 8087785;
      TableParameter_L@1000000001 : Record 8053450;
      TabDatDeb_L@1000000008 : ARRAY [24] OF Date;
      TabDatFin_L@1000000009 : ARRAY [24] OF Date;
      FileLogName_L@1000000003 : Text[256];
      FileImport_L@1000000004 : File;
      FileInStream_L@1000000007 : InStream;
      TraitOK_L@1000000010 : Boolean;
      NbRecLu_L@1000000011 : Integer;
      NbCarLu_L@1000000012 : Integer;
      TexteLu_L@1000000013 : Text[1024];
      Pos_L@1000000014 : Integer;
      Texte_L@1000000015 : Text[256];
      DateText_L@1000000018 : Text[10];
      Ind_L@1000000016 : Integer;
      IndTabDate_L@1000000017 : Integer;
      Article_L@1000000019 : Code[20];
    BEGIN
      //----------------------------------------------------------------------------------------------------------------------------------
      // Importation des Tarifs TPG : Copie vers SAS Import Tarif TPG (8087785)
      //----------------------------------------------------------------------------------------------------------------------------------
      // Entr'es:
      //   - FileName_P : Nom du fichier PC . int'grer
      // Sorties:
      //   - Result_L   : Erreur Retourn'e, Vide si Ok
      //----------------------------------------------------------------------------------------------------------------------------------
      IF ((TableParameter_L.FINDFIRST) AND (TableParameter_L."Import TPG Price List")) THEN BEGIN
        ToolsFile_G.LogInitFileExecute(FileName_P);
        IF NOT EXISTS(FileName_P) THEN BEGIN
          ToolsFile_G.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0002, FileName_P), 1, MAXSTRLEN(Result_L)));
        END ELSE BEGIN
          IF NOT FileImport_L.OPEN(FileName_P) THEN BEGIN
            ToolsFile_G.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0003, FileName_P, GETLASTERRORTEXT), 1, MAXSTRLEN(Result_L)));
            CLEARLASTERROR;
          END ELSE BEGIN
            TableSasImport_L.SETRANGE(TableSasImport_L."Integration date", 0D);
            IF TableSasImport_L.FINDSET THEN BEGIN
              ToolsFile_G.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0008, FileName_P), 1, MAXSTRLEN(Result_L)));
            END ELSE BEGIN
              TableSasImport_L.SETRANGE(TableSasImport_L."Integration date");
              FileImport_L.CREATEINSTREAM(FileInStream_L);
              TraitOK_L := TRUE;
              NbRecLu_L := 0;
              WHILE NOT (FileInStream_L.EOS) DO BEGIN
                NbCarLu_L := FileInStream_L.READTEXT(TexteLu_L);
                CLEAR(TableSasImport_L);

                IF  TraitOK_L THEN BEGIN
                  Pos_L := 1;
                  NbRecLu_L += 1;

                  // Traitement de la 1e ligne d'en-t^te contenant les dates de d'but
                  IF NbRecLu_L = 1 THEN BEGIN
                    Texte_L := ToolsFile_G.ExtractTextCsv(TexteLu_L, Pos_L, ';', 20);
                    IF COPYSTR(Texte_L, 1, 4) <> 'CODE' THEN BEGIN
                      ToolsFile_G.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0020, FileName_P), 1, MAXSTRLEN(Result_L)));
                      TraitOK_L := FALSE;
                    END ELSE BEGIN
                      // D'signation
                      Texte_L := ToolsFile_G.ExtractTextCsv(TexteLu_L, Pos_L, ';', 50);
                      // Dates de d'but
                      Texte_L := ToolsFile_G.ExtractTextCsv(TexteLu_L, Pos_L, ';', 10);
                      Ind_L := 0;
                      WHILE (Ind_L < ARRAYLEN(TabDatDeb_L)) AND (COPYSTR(Texte_L, 1, 1) <> ' ') AND (STRLEN(Texte_L) <> 0) DO BEGIN
                        Ind_L += 1;
                        DateText_L := COPYSTR(Texte_L, 7, 4) + '-' + COPYSTR(Texte_L, 4, 2) + '-' + COPYSTR(Texte_L, 1, 2);
                        IF NOT EVALUATE(TabDatDeb_L[Ind_L] , DateText_L, 9) THEN BEGIN
                          ToolsFile_G.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0006, COPYSTR(DateText_L, 1, 10)), 1, MAXSTRLEN(Result_L)));
                          TraitOK_L := FALSE;
                        END;
                        Texte_L := ToolsFile_G.ExtractTextCsv(TexteLu_L, Pos_L, ';', 10);
                      END;
                      IndTabDate_L := Ind_L;
                    END;
                  END;

                  // Traitement de la 2e ligne d'en-t^te contenant les dates de fin
                  IF NbRecLu_L = 2 THEN BEGIN
                    Texte_L := ToolsFile_G.ExtractTextCsv(TexteLu_L, Pos_L, ';', 20);
                    IF COPYSTR(Texte_L, 1, 4) <> 'CODE' THEN BEGIN
                      ToolsFile_G.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0020, FileName_P), 1, MAXSTRLEN(Result_L)));
                      TraitOK_L := FALSE;
                    END ELSE BEGIN
                      // D'signation
                      Texte_L := ToolsFile_G.ExtractTextCsv(TexteLu_L, Pos_L, ';', 50);
                      // Dates de fin
                      Texte_L := ToolsFile_G.ExtractTextCsv(TexteLu_L, Pos_L, ';', 10);
                      Ind_L := 0;
                      WHILE (Ind_L < ARRAYLEN(TabDatDeb_L)) AND (COPYSTR(Texte_L, 1, 1) <> ' ') AND (STRLEN(Texte_L) <> 0) DO BEGIN
                        Ind_L += 1;
                        DateText_L := COPYSTR(Texte_L, 7, 4) + '-' + COPYSTR(Texte_L, 4, 2) + '-' + COPYSTR(Texte_L, 1, 2);
                        IF NOT EVALUATE(TabDatFin_L[Ind_L] , DateText_L, 9) THEN BEGIN
                          ToolsFile_G.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0006, COPYSTR(DateText_L, 1, 10)), 1, MAXSTRLEN(Result_L)));
                          TraitOK_L := FALSE;
                        END;
                        Texte_L := ToolsFile_G.ExtractTextCsv(TexteLu_L, Pos_L, ';', 10);
                      END;
                      IF Ind_L <> IndTabDate_L THEN BEGIN
                        ToolsFile_G.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0021, FileName_P), 1, MAXSTRLEN(Result_L)));
                        TraitOK_L := FALSE;
                      END;
                    END;
                  END;

                  // Traitement des lignes articles
                  IF NbRecLu_L > 2 THEN BEGIN
                    Article_L := ToolsFile_G.ExtractTextCsv(TexteLu_L, Pos_L, ';', 20);
                    IF (COPYSTR(Article_L, 1, 1) <> ' ') AND (STRLEN(Article_L) <> 0) THEN BEGIN
                      Texte_L := ToolsFile_G.ExtractTextCsv(TexteLu_L, Pos_L, ';', 50);
                      Ind_L := 0;
                      WHILE (Ind_L < IndTabDate_L) DO BEGIN
                        CLEAR(TableSasImport_L."Record No");
                        Ind_L += 1;
                        Texte_L := ToolsFile_G.ExtractTextCsv(TexteLu_L, Pos_L, ';', 17);
                        Texte_L := CONVERTSTR(Texte_L, '.', ',');
                        IF (COPYSTR(Texte_L, 1, 1) <> ' ') AND (STRLEN(Texte_L) <> 0) THEN BEGIN
                          IF NOT EVALUATE(TableSasImport_L.Price , Texte_L) THEN BEGIN
                            ToolsFile_G.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0022,
                              TableSasImport_L.FIELDCAPTION(TableSasImport_L.Price),
                              Texte_L, COPYSTR(TexteLu_L, 1, 20)), 1, MAXSTRLEN(Result_L)));
                          END ELSE BEGIN
                            TableSasImport_L."Item No" := Article_L;
                            TableSasImport_L."Starting Date" := TabDatDeb_L[Ind_L];
                            TableSasImport_L."Ending Date" := TabDatFin_L[Ind_L];
                            TableSasImport_L.HoroCreated := CURRENTDATETIME;
                            TableSasImport_L."Integration date" := 0D;
                            IF NOT TableSasImport_L.INSERT(TRUE) THEN BEGIN
                              ToolsFile_G.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0007, FORMAT(TableSasImport_L."Record No"))
                              , 1, MAXSTRLEN(Result_L)));
                            END;
                          END;
                        END;
                      END;
                    END;
                  END;

                END;
              END;
              FileImport_L.CLOSE();
              TableParameter_L."TPG PriceLast Date Import" := CURRENTDATETIME;
              IF NOT TableParameter_L.MODIFY(TRUE) THEN BEGIN
                ToolsFile_G.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0009, TableParameter_L.TABLECAPTION,COMPANYNAME)
                  , 1, MAXSTRLEN(Result_L)));
              END;
            END;
          END;
        END;

        // Pour g'rer le ROLLBACK.
        FileLogName_L := ToolsFile_G.LogName;
        ToolsFile_G.LogClose();

        IF ToolsFile_G.LogErreurType() = 'E' THEN BEGIN
          ERROR('ERREUR ' + FileLogName_L);
        END ELSE BEGIN
          IF NOT ToolsFile_G.FileRenameSav(FileName_P, 'Sav', TRUE) THEN BEGIN
            ToolsFile_G.LogWriteErreur(STRSUBSTNO(Text0004, FileName_P));
          END;
          IF ToolsFile_G.LogErreurType() = 'I' THEN BEGIN
            MoteurInterface_G.InsertLog(CodeUnit_G, FileLogName_L);
          END ELSE BEGIN
            MoteurInterface_G.InsertOk(CodeUnit_G);
          END;
        END;
      END;
    END;

    PROCEDURE ImportTarifRecord@1000000002() Result_L : Text[256];
    VAR
      "<-- 3LI LOCALS -->"@1000000000 : Integer;
      TableParameter_L@1000000001 : Record 8053450;
      TableSasImport_L@1000000003 : Record 8087785;
      TablePrixVente_L@1000000004 : Record 7002;
      FileLogName_L@1000000002 : Text[256];
    BEGIN
      //----------------------------------------------------------------------------------------------------------------------------------
      // Importation du tarif TPG dans Nav : copie du SAS vers Table 7002 (Sales Price)
      //----------------------------------------------------------------------------------------------------------------------------------
      // Entr'es:
      //
      // Sorties:
      //   - Result_L               : Erreur Retourn'e, Vide si Ok
      //----------------------------------------------------------------------------------------------------------------------------------
      IF ((TableParameter_L.FINDFIRST) AND (TableParameter_L."Import TPG Price List")) THEN BEGIN
        TableSasImport_L.SETRANGE(TableSasImport_L."Integration date", 0D);
        IF TableSasImport_L.FINDSET THEN BEGIN
          // Suppression des tarifs pr'c'demment charg's
          TablePrixVente_L.SETRANGE(TablePrixVente_L."Sales Type", TableParameter_L."TPG Sales Type");
          TablePrixVente_L.SETRANGE(TablePrixVente_L."Sales Code", TableParameter_L."TPG Sales Code");
          IF TablePrixVente_L.FINDSET THEN BEGIN
            TablePrixVente_L.DELETEALL(TRUE);
          END;

          REPEAT
            CLEAR(TablePrixVente_L);
            TablePrixVente_L.VALIDATE(TablePrixVente_L."Item No.", TableSasImport_L."Item No");
            TablePrixVente_L.VALIDATE(TablePrixVente_L."Sales Type", TableParameter_L."TPG Sales Type");
            TablePrixVente_L.VALIDATE(TablePrixVente_L."Sales Code", TableParameter_L."TPG Sales Code");
            TablePrixVente_L.VALIDATE(TablePrixVente_L."Starting Date", TableSasImport_L."Starting Date");
            TablePrixVente_L.VALIDATE(TablePrixVente_L."Unit Price", TableSasImport_L.Price);
            TablePrixVente_L.VALIDATE(TablePrixVente_L."Allow Invoice Disc.", TRUE);
            TablePrixVente_L.VALIDATE(TablePrixVente_L."Ending Date", TableSasImport_L."Ending Date");
            TablePrixVente_L.VALIDATE(TablePrixVente_L."Allow Line Disc.", TRUE);
            IF NOT TablePrixVente_L.INSERT(TRUE) THEN BEGIN
              ToolsFile_G.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0011, TablePrixVente_L.TABLECAPTION, TableSasImport_L."Item No"
                , TableSasImport_L."Starting Date", TableSasImport_L."Record No"), 1, MAXSTRLEN(Result_L)));
            END;

            TableSasImport_L."Integration date" := TODAY;
            IF NOT TableSasImport_L.MODIFY(TRUE) THEN BEGIN
              ToolsFile_G.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0005, TableSasImport_L."Record No"), 1, MAXSTRLEN(Result_L)));
            END;

          UNTIL TableSasImport_L.NEXT = 0;
        END;
      END;

      // Pour g'rer le ROLLBACK.
      FileLogName_L := ToolsFile_G.LogName;
      ToolsFile_G.LogClose();

      IF ToolsFile_G.LogErreurType() = 'E' THEN BEGIN
        ERROR('ERREUR ' + FileLogName_L);
      END ELSE BEGIN
        IF ToolsFile_G.LogErreurType() = 'I' THEN BEGIN
          MoteurInterface_G.InsertLog(CodeUnit_G, FileLogName_L);
        END ELSE BEGIN
          MoteurInterface_G.InsertOk(CodeUnit_G);
        END;
      END;

      //<--20140428-Int27-CMC02.02.00-ADVITAM-GMX
    END;

    BEGIN
    {
      //--> 3LI BEGIN
      20140428-Int27-CMC02.02.00-ADVITAM-GMX : Importation des tarifs TPG PA.
      //<-- 3LI END
    }
    END.
