              TableContratMarketOpen_L.TABLECAPTION), 1, MAXSTRLEN(Result_L)));

          END ELSE BEGIN

            IF TableContratMarket_L."Number Of Lots" = TableContratMarketOpen_L."Number Of Lots" THEN BEGIN
             { ToolsFile_G.LogWriteInfo('INF001: Affecation des bons du contrat collecte '+

             }

              ToolsFile_G.LogWriteInfo('INF020: Commit du contrat collecte '+

          END;

        END ELSE BEGIN

          // Sinon prendre les bons des n contrats open (n = nb lots vendus)
          IF TableContratMarket_L."Number Of Lots" = 1 THEN BEGIN
            CLEAR(ContractCollectionMNFNo_L);

            ToolsFile_G.LogWriteInfo('INF008: Recherche d''un no de contrat collecte MNF pour le contrat march' '+

          END ELSE BEGIN

                  TableCollectionContractMNF_L.TABLECAPTION), 1, MAXSTRLEN(Result_L)));
              END;

              MiseAJourBons(ContractCollectionMNFNo_L,TableContratMarket_L."Collection  Contrat No.");

              AnnulationContratCollecteMNF(ContractCollectionMNFNo_L);

              COMMIT;

              NbLotVendu_L := NbLotVendu_L-1;
            END;


          END;

        END;
      END;
    END;
