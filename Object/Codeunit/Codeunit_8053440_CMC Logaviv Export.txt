                      FORMAT(TableLocationLog_L.IdMember) + '/' + TableLocationLog_L.IdSiloExt),
                      1, MAXSTRLEN(Result_L)));

        TableLocationLog_L.SETRANGE(TableLocationLog_L.Deleted, FALSE);
        IF TableLocationLog_L.FINDSET THEN BEGIN
          REPEAT

              Deleted_L := TRUE;
            END ELSE BEGIN

                Deleted_L := TRUE;
              END;
            END;
            IF Deleted_L THEN BEGIN
              TableLocationLog_L.Deleted := TRUE;
              CLEAR(TableLocationLog_L.HoroIntegration);
              CLEAR(TableLocationLog_L.HoroIntegrationError);
              TableLocationLog_L.HoroModified := CURRENTDATETIME;
