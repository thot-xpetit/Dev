    PROCEDURE FillONIGCBuffer@8053007(StartDate_P@8053000 : Date;EndDate_P@8053001 : Date;Type_P@8053003 : 'Stock Collection,Stock at Collector';Bio_P@8053007 : Boolean);
    VAR
      ONIGCBuffer_L@8053002 : Record 8053058;
      CMCONIGCLines_L@8053004 : Record 8053057;
      Val@8053005 : Decimal;
      ind@8053006 : Integer;
    BEGIN
      // -----------------------------------------------------------------------------------------
      //                  Appel fonctions pour remplissage ONIGC Buffer
      // -----------------------------------------------------------------------------------------
      ONIGCBuffer_L.RESET;
      ONIGCBuffer_L.DELETEALL;
      Bio_G:=Bio_P;

      G_Windows.OPEN(Text8053011);

      IF Type_P = 0 THEN BEGIN
        CalcBalanceToDate(StartDate_P,FALSE,Type_P);
        CalcStockInputsCollection(StartDate_P,EndDate_P);
        CalcLine(StartDate_P,EndDate_P,'E02','CA','');
        CalcLine(StartDate_P,EndDate_P,'T20','D','>0');
        CalcLine(StartDate_P,EndDate_P,'T10','DB','>0');
        CalcTotal('CA|D|DB','DZ');
        CalcTotal('C|DZ','E');
        CalcRetrocessions;
        CalcLine(StartDate_P,EndDate_P,'S10','G','');
        CalcLine(StartDate_P,EndDate_P,'T20','H','<0');
        CalcLine(StartDate_P,EndDate_P,'S20','H','');
        CalcLine(StartDate_P,EndDate_P,'T10','HF','<0');
        CalcTotal('G|H|HF','I');
        CalcFinalBalanceByDept(StartDate_P,EndDate_P);
        CalcBalanceToDate(EndDate_P,TRUE,Type_P);
      END ELSE BEGIN
        CalcBalanceToDate(StartDate_P,FALSE,Type_P);
        CalcStockInputsAtCollector(StartDate_P,EndDate_P);
        CalcStockOutputsAtCollector(StartDate_P,EndDate_P);
        CalcStockOutputsRetakeAtCollec(StartDate_P,EndDate_P);
        CalcFinalBalanceByDept2(StartDate_P,EndDate_P);
        CalcBalanceToDate(EndDate_P,TRUE,Type_P);
      END;

      // -----------------------------------------------------------------------------------------
      //                  Calcul 'dont' pour remplissage ONIGC Buffer
      // -----------------------------------------------------------------------------------------
      CMCONIGCLines_L.SETFILTER(CMCONIGCLines_L."Page No.Link",'>%1',0);
      IF CMCONIGCLines_L.FINDSET THEN REPEAT
        IF ONIGCBuffer_L.FINDSET THEN REPEAT
          ind:=(CMCONIGCLines_L."Page No.Link"-2)*7+CMCONIGCLines_L."Column Index Link";
          Val:=GetBufferValue(ONIGCBuffer_L,CMCONIGCLines_L."Buffer Value No.");
          AssignBufferValue(ONIGCBuffer_L,ind,Val);
          ONIGCBuffer_L.MODIFY;
        UNTIL ONIGCBuffer_L.NEXT=0;
      UNTIL CMCONIGCLines_L.NEXT=0;
      G_Windows.CLOSE;
    END;

    PROCEDURE CalcStockInputsCollection@8053000(StartDate_P@8053000 : Date;EndDate_P@8053001 : Date);
    VAR
      CMCSetup_L@8053010 : Record 8052982;
      ItemLedgEntry_L@8053003 : Record 32;

      Date_L@8053012 : Date;
      vendor_l@8053013 : Record 23;
      CMCSpeciesSetup_L@1000000000 : Record 8053323;
    BEGIN
      // ---------------------------------------------------------------------------------------------------------
      // Calcul collecte par d'partements pour d'claration mensuelle ONIGC (1-ere partie)  typ's "B" et total "C"
      // 2012 : d'partement de l'apporteur et non plus du collecteur
      // ---------------------------------------------------------------------------------------------------------
      CMCSetup_L.GET;

      ItemLedgEntry_L.SETRANGE("Posting Date",StartDate_P,EndDate_P);
      ItemLedgEntry_L.SETRANGE("Movement Type",ItemLedgEntry_L."Movement Type" :: E01);
      IF ItemLedgEntry_L.FIND('-') THEN BEGIN
        REPEAT
          G_Windows.UPDATE(1,ItemLedgEntry_L."Entry No.");
          Item_L.GET(ItemLedgEntry_L."Item No.");
          CMCSpeciesSetup_L.GET(Item_L."Species Code");
          IF (Item_L."ONIGC Column Index" <> 0) AND (Bio_G = CMCSpeciesSetup_L."Biologic Species") AND
            GetPageCol( ONIGCLine_L,CMCSetup_L."Monthly ONIGC Code",Item_L)
          THEN BEGIN
            IF Location_L.GET(ItemLedgEntry_L."Location Code") THEN BEGIN
              IF StartDate_P >= 010712D THEN BEGIN
                vendor_l.GET(ItemLedgEntry_L."Source No.");
                IF (vendor_l."Country/Region Code" <> 'FR') AND (vendor_l."Country/Region Code" <> '') THEN
                  DeptCode_L:='99'
                ELSE
                  DeptCode_L := COPYSTR(vendor_l."Post Code",1,2);
              END;
              IF ONIGCBuff_L.GET(DeptCode_L,0,'B') THEN BEGIN
                AssignBufferValue(ONIGCBuff_L,ONIGCLine_L."Buffer Value No.",ItemLedgEntry_L.Quantity);
                ONIGCBuff_L.MODIFY;
              END ELSE BEGIN
                ONIGCBuff_L.INIT;
                ONIGCBuff_L."Line ID" := DeptCode_L;
                ONIGCBuff_L.Type := 0;
                ONIGCBuff_L."Calc Type" := 'B';
                ONIGCBuff_L.Description := Text805B + DeptCode_L;
                AssignBufferValue(ONIGCBuff_L,ONIGCLine_L."Buffer Value No.",ItemLedgEntry_L.Quantity);
                ONIGCBuff_L.INSERT;
              END;
              IF ONIGCBuffTotal_L.GET('',0,'C') THEN BEGIN
                AssignBufferValue(ONIGCBuffTotal_L,ONIGCLine_L."Buffer Value No.",ItemLedgEntry_L.Quantity);
                ONIGCBuffTotal_L.MODIFY;
              END ELSE BEGIN
                ONIGCBuffTotal_L.Type := 0;
                ONIGCBuffTotal_L."Calc Type" := 'C';
                ONIGCBuffTotal_L.Description := Text805C;
                AssignBufferValue(ONIGCBuffTotal_L,ONIGCLine_L."Buffer Value No.",ItemLedgEntry_L.Quantity);
                ONIGCBuffTotal_L.INSERT;
              END;
              ONIGCItemsFound_L := TRUE;
            END;
          END;
        UNTIL ItemLedgEntry_L.NEXT = 0;
      END;

      IF NOT ONIGCItemsFound_L THEN BEGIN
        ONIGCBuff_L.INIT;
        ONIGCBuff_L."Line ID" := DeptCode_L;
        ONIGCBuff_L.Type := 0;
        ONIGCBuff_L."Calc Type" := 'B';
        ONIGCBuff_L.Description := Text805B ;
        ONIGCBuff_L.INSERT;
        ONIGCBuffTotal_L.Type := 0;
        ONIGCBuffTotal_L."Calc Type" := 'C';
        ONIGCBuffTotal_L.Description := Text805C;
        IF ONIGCBuffTotal_L.INSERT THEN;
      END;
    END;

    PROCEDURE CalcLine@8053009(StartDate_P@8053000 : Date;EndDate_P@8053001 : Date;FilterMvt_P@8053015 : Text[30];Type_P@8053016 : Code[2];FilterQty_P@8053017 : Text[30]);
    VAR
      CMCSetup_L@8053013 : Record 8052982;
      ItemLedgEntry_L@8053003 : Record 32;
      Item_L@8053004 : Record 27;
      Location_L@8053005 : Record 14;
      ONIGCBuff_L@8053006 : Record 8053058;
      ONIGCBuffTotal_L@8053010 : Record 8053058;
      DeptCode_L@8053007 : Code[2];
      ONIGCLine_L@8053008 : Record 8053057;
      ONIGC_L@8053009 : Record 8053056;
      IsComm_L@8053002 : Boolean;
      Counter_L@8053011 : Integer;
      AddTotal_L@8053012 : Boolean;
      ONIGCItemsFound_L@8053014 : Boolean;
      Mvt_L@8053018 : Code[10];
      Qty_l@8053019 : Decimal;
      Cust_L@8053020 : Record 18;
      CMCSpeciesSetup_L@1000000000 : Record 8053323;
    BEGIN
      // -----------------------------------------------------------------------------------------
      // Calcul lignes pour d'claration mensuelle ONIGC (1-ere partie)
      // -----------------------------------------------------------------------------------------
      CMCSetup_L.GET;
      ItemLedgEntry_L.SETRANGE("Posting Date",StartDate_P,EndDate_P);
      IF FilterQty_P<>'' THEN ItemLedgEntry_L.SETFILTER(Quantity,FilterQty_P);
      ItemLedgEntry_L.SETFILTER("Movement Type",FilterMvt_P);

      IF ItemLedgEntry_L.FINDSET THEN BEGIN
        REPEAT
          G_Windows.UPDATE(1,ItemLedgEntry_L."Entry No.");
          Item_L.GET(ItemLedgEntry_L."Item No.");
          CMCSpeciesSetup_L.GET(Item_L."Species Code");
          IF (Item_L."ONIGC Column Index" <> 0) AND (Bio_G = CMCSpeciesSetup_L."Biologic Species")
            AND GetPageCol( ONIGCLine_L,CMCSetup_L."Monthly ONIGC Code",Item_L)
          THEN BEGIN
            Mvt_L:=FORMAT(ItemLedgEntry_L."Movement Type");
            IF ONIGCBuff_L.GET('',0,Type_P) THEN BEGIN
              IF Mvt_L IN ['S03','S04','S05','S10','S20'] THEN
                Qty_l:=-ItemLedgEntry_L.Quantity ELSE Qty_l:=ItemLedgEntry_L.Quantity ;
              IF (Qty_l < 0) AND (Type_P IN ['H','HF']) THEN Qty_l:=ABS(Qty_l);
              AssignBufferValue(ONIGCBuff_L,ONIGCLine_L."Buffer Value No.",Qty_l);
              ONIGCBuff_L.MODIFY;
            END ELSE BEGIN
              ONIGCBuff_L.INIT;
              ONIGCBuff_L.Type := 0;
              ONIGCBuff_L."Calc Type" := Type_P ;
              CASE Type_P OF
                'CA' : ONIGCBuff_L.Description := Text805CA;
                'D'  : ONIGCBuff_L.Description := Text805D;
                'DB' : ONIGCBuff_L.Description := Text805DB;
                'G'  : ONIGCBuff_L.Description := Text805G;
                'H'  : ONIGCBuff_L.Description := Text805H;
                'HF' : ONIGCBuff_L.Description := Text805HF;
              END;

              IF Mvt_L IN ['S03','S04','S05','S10','S20'] THEN
                Qty_l:=-ItemLedgEntry_L.Quantity ELSE Qty_l:=ItemLedgEntry_L.Quantity ;
              IF (Qty_l < 0) AND (Type_P IN ['H','HF']) THEN Qty_l:=ABS(Qty_l);
              AssignBufferValue(ONIGCBuff_L,ONIGCLine_L."Buffer Value No.",Qty_l);
              ONIGCBuff_L.INSERT;
            END;

            // Dont Eleveur
            IF Mvt_L='S10' THEN BEGIN
              Cust_L.GET(ItemLedgEntry_L."Source No.");

              IF (Cust_L.Breeder=TRUE) THEN BEGIN
                IF ONIGCBuff_L.GET('',0,'IE') THEN BEGIN
                  AssignBufferValue(ONIGCBuff_L,ONIGCLine_L."Buffer Value No.",Qty_l);
                  ONIGCBuff_L.MODIFY;
                END ELSE BEGIN
                  ONIGCBuff_L.INIT;
                  ONIGCBuff_L.Type := 0;
                  ONIGCBuff_L."Calc Type" := 'IE';
                  ONIGCBuff_L.Description := Text805IE;
                  AssignBufferValue(ONIGCBuff_L,ONIGCLine_L."Buffer Value No.",Qty_l);
                  ONIGCBuff_L.INSERT;
                END;
              END;
              IF (Cust_L."Food Industry"=TRUE ) AND (Bio_G=TRUE) THEN BEGIN
                IF ONIGCBuff_L.GET('',0,'II') THEN BEGIN
                  AssignBufferValue(ONIGCBuff_L,ONIGCLine_L."Buffer Value No.",Qty_l);
                  ONIGCBuff_L.MODIFY;
                END ELSE BEGIN
                  ONIGCBuff_L.INIT;
                  ONIGCBuff_L.Type := 0;
                  ONIGCBuff_L."Calc Type" := 'II';
                  ONIGCBuff_L.Description := Text805IAA;
                  AssignBufferValue(ONIGCBuff_L,ONIGCLine_L."Buffer Value No.",Qty_l);
                  ONIGCBuff_L.INSERT;
                END;
              END;
              IF (Cust_L.Breeder=TRUE) THEN BEGIN
                IF ONIGCBuff_L.GET('',0,'IJ') THEN BEGIN
                  AssignBufferValue(ONIGCBuff_L,ONIGCLine_L."Buffer Value No.",Qty_l);
                  ONIGCBuff_L.MODIFY;
                END ELSE BEGIN
                  ONIGCBuff_L.INIT;
                  ONIGCBuff_L.Type := 0;
                  ONIGCBuff_L."Calc Type" := 'IJ';
                  ONIGCBuff_L.Description := Text805IB;
                  AssignBufferValue(ONIGCBuff_L,ONIGCLine_L."Buffer Value No.",Qty_l);
                  ONIGCBuff_L.INSERT;
                END;
              END;
            END;
          END;
        UNTIL ItemLedgEntry_L.NEXT = 0;
      END;
    END;

    PROCEDURE CalcBalanceToDate@8053002(Date_P@8053001 : Date;Final_P@8053000 : Boolean;Type_P@8053014 : Option);
    VAR
      CMCSetup_L@8053011 : Record 8052982;
      ItemLedgEntry_L@8053008 : Record 32;
      Item_L@8053007 : Record 27;
      Location_L@8053006 : Record 14;
      ONIGCBuff_L@8053005 : Record 8053058;
      DeptCode_L@8053004 : Code[2];
      ONIGCLine_L@8053003 : Record 8053057;
      ONIGC_L@8053002 : Record 8053056;
      Inputs_L@8053009 : Decimal;
      Outputs_L@8053010 : Decimal;
      ONIGCItemsFound_L@8053012 : Boolean;
      Qty_L@8053013 : Dialog;
      CMCSpeciesSetup_L@1000000000 : Record 8053323;
    BEGIN
      // -----------------------------------------------------------------------------------------
      // Calcul stock pour d'claration mensuelle ONIGC (1-ere partie)  typ's "A" et "K"
      // -----------------------------------------------------------------------------------------
      CMCSetup_L.GET;

      IF NOT Final_P THEN
        ItemLedgEntry_L.SETFILTER("Posting Date",'<%1',Date_P)
      ELSE
        ItemLedgEntry_L.SETFILTER("Posting Date",'<=%1',Date_P);
      IF Type_P=0 THEN
        ItemLedgEntry_L.SETFILTER("Movement Type",'%1|%2|%3|%4|%5|%6|%7|%8', ItemLedgEntry_L."Movement Type"::E01,
          ItemLedgEntry_L."Movement Type"::E05, ItemLedgEntry_L."Movement Type"::E02, ItemLedgEntry_L."Movement Type"::T10,
          ItemLedgEntry_L."Movement Type"::T20, ItemLedgEntry_L."Movement Type"::S10,ItemLedgEntry_L."Movement Type"::S05,
          ItemLedgEntry_L."Movement Type"::S20)
      ELSE
        ItemLedgEntry_L.SETFILTER("Movement Type",'%1|%2',ItemLedgEntry_L."Movement Type"::E03,ItemLedgEntry_L."Movement Type"::S03);

      IF ItemLedgEntry_L.FINDSET THEN
        REPEAT
          G_Windows.UPDATE(1,ItemLedgEntry_L."Entry No.");
          Item_L.GET(ItemLedgEntry_L."Item No.");
          CMCSpeciesSetup_L.GET(Item_L."Species Code");
          IF (Item_L."ONIGC Column Index" <> 0) AND (Bio_G = CMCSpeciesSetup_L."Biologic Species") AND
            GetPageCol( ONIGCLine_L,CMCSetup_L."Monthly ONIGC Code",Item_L)
          THEN BEGIN
            IF NOT Final_P THEN BEGIN
              IF ONIGCBuff_L.GET('',Type_P,'A') THEN BEGIN
                AssignBufferValue(ONIGCBuff_L,ONIGCLine_L."Buffer Value No.",ItemLedgEntry_L.Quantity);
                ONIGCBuff_L.MODIFY;
              END ELSE BEGIN
                ONIGCBuff_L.INIT;
                ONIGCBuff_L.Type := Type_P;
                ONIGCBuff_L."Calc Type" := 'A';
                ONIGCBuff_L.Description := Text805A;
                AssignBufferValue(ONIGCBuff_L,ONIGCLine_L."Buffer Value No.",ItemLedgEntry_L.Quantity);
                ONIGCBuff_L.INSERT;
              END;
            END ELSE
              IF ONIGCBuff_L.GET('',Type_P,'K') THEN BEGIN
                AssignBufferValue(ONIGCBuff_L,ONIGCLine_L."Buffer Value No.",ItemLedgEntry_L.Quantity);
                ONIGCBuff_L.MODIFY;
              END ELSE BEGIN
                ONIGCBuff_L.INIT;
