    PROCEDURE GetReturn@1000000010() Result : Text[1024];
    VAR
      Pos@1000000000 : Integer;
      ReturnName@1000000001 : Text[30];
    BEGIN

      ////////////////////////////////////////////////
      // Astuce pour r'cup'rer une valeur de retour //
      // stock'e dans GETLASTERRORTEXT              //
      ////////////////////////////////////////////////

      // Initialisation du pattern de retour
      ReturnName := 'RETURN:';

      // Permet de d'tecter si GETLASTERRORTEXT contient une valeur de retour
      Pos := STRPOS(GETLASTERRORTEXT, ReturnName);

      // Si GETLASTERRORTEXT contient une valeur de retour
      IF Pos > 0 THEN BEGIN
        // On r'cupSre la valeur de retour
        Result := DELSTR(GETLASTERRORTEXT, Pos, STRLEN(ReturnName));
      END ELSE BEGIN
        // Sinon on lance une erreur
        ERROR(GETLASTERRORTEXT);
      END;
    END;
