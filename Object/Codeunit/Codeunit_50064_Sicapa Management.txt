    PROCEDURE ShowPurchReceiptLine@1000000005(SICAPAInvoiceHdr@1000000000 : Record 8087767;VAR SICAPAInvoiceLine@1000000001 : Record 8087768;SICAPASetup@1000000002 : Record 8053408) ReturnValue : Boolean;
    VAR
      PurchRcptLine@1000000003 : Record 121;
      PurchRcptHdr@1000000007 : Record 120;
      OrderNo@1000000004 : Code[20];
      LocationCode@1000000005 : Code[20];
      Position@1000000006 : Integer;
      DocumentDate@1000000008 : Date;
      PagePurchRcptLine@1000000009 : Page 50042;
      ToolsFile@1000000010 : Codeunit 8053405;
      Location@1000000011 : Record 14;
    BEGIN
      ReturnValue := FALSE;
      PurchRcptLine.RESET();
      PurchRcptLine.FILTERGROUP(10);
      PurchRcptLine.SETFILTER("Qty. Rcd. Not Invoiced", '<> 0');
      PurchRcptLine.FILTERGROUP(0);
      // Set the filter on "Purch rcpt line"
      IF (SICAPAInvoiceHdr.SUNavAgri <> '') THEN BEGIN
        PurchRcptLine.SETFILTER("Buy-from Vendor No.", '%1|%2' , SICAPAInvoiceHdr.SUNavAgri, SICAPASetup.Vendor);
      END ELSE BEGIN
        IF (SICAPAInvoiceHdr."Vendor No." <> '') THEN BEGIN
          PurchRcptLine.SETFILTER("Buy-from Vendor No.", '%1|%2' , SICAPAInvoiceHdr."Vendor No.", SICAPASetup.Vendor);
        END;
      END;
      IF (SICAPAInvoiceLine."Item No. NavAgri" <> '') THEN BEGIN
        PurchRcptLine.SETRANGE("No.", SICAPAInvoiceLine."Item No. NavAgri");
      END;
      IF (SICAPAInvoiceLine."Variant Code NavAgri" <> '') THEN BEGIN
        PurchRcptLine.SETRANGE("Variant Code", SICAPAInvoiceLine."Variant Code NavAgri");
      END;

      CLEAR(OrderNo);
      CLEAR(LocationCode);
      IF (SICAPAInvoiceLine."Source Document No." <> '') THEN BEGIN
        PurchRcptHdr.SETRANGE("Order No.", SICAPAInvoiceLine."Source Document No.");
        IF (PurchRcptHdr.COUNT() = 1) THEN BEGIN
          IF PurchRcptHdr.FINDFIRST THEN BEGIN
            PurchRcptLine.SETRANGE("Document No.", PurchRcptHdr."No.");
            IF (SICAPAInvoiceLine."Source Document Line No." > 0) THEN BEGIN
              PurchRcptLine.SETRANGE("Order Line No.", SICAPAInvoiceLine."Source Document Line No.");
            END;
          END;
        END;
        LocationCode := SICAPAInvoiceLine."Location Code";
      END ELSE BEGIN
        IF SICAPAInvoiceLine.RefCde <> '' THEN BEGIN
          Position := 1;
          OrderNo := ToolsFile.ExtractTextCsvSepChaine(SICAPAInvoiceLine.RefCde, Position, ' ', '', 20);
          PurchRcptHdr.SETRANGE("Order No.", OrderNo);
          IF (PurchRcptHdr.COUNT() = 1) THEN BEGIN
            IF PurchRcptHdr.FINDFIRST THEN BEGIN
              PurchRcptLine.SETRANGE("Document No.", PurchRcptHdr."No.");
            END;
          END;
          LocationCode := ToolsFile.ExtractTextCsvSepChaine(SICAPAInvoiceLine.RefCde, Position, ' ', '', 10);
        END;
      END;

      IF SICAPAInvoiceHdr.DPNavAgri <> '' THEN BEGIN
        PurchRcptLine.SETRANGE("Location Code", SICAPAInvoiceHdr.DPNavAgri);
      END ELSE BEGIN
        IF LocationCode <> '' THEN BEGIN
          IF (Location.GET(LocationCode)) THEN BEGIN
            PurchRcptLine.SETRANGE("Location Code", Location.Code);
          END;
        END;
      END;

      IF SICAPAInvoiceLine."Coefficient Unit" <> 0 THEN BEGIN
        PurchRcptLine.SETRANGE(Quantity, SICAPAInvoiceLine."Quantite facture" * SICAPAInvoiceLine."Coefficient Unit");
      END;

      DocumentDate := SICAPAInvoiceLine."Date BL";
      IF (DocumentDate = 0D) THEN BEGIN
        DocumentDate := SICAPAInvoiceHdr."Date BL";
      END;

      IF (FORMAT(SICAPASetup."Time Find Movement") <> '') THEN BEGIN
        IF (DocumentDate = 0D) THEN BEGIN
          PurchRcptLine.SETRANGE("Posting Date", CALCDATE('-' + FORMAT(SICAPASetup."Time Find Movement"), TODAY),
                CALCDATE(SICAPASetup."Time Find Movement", TODAY));
        END ELSE BEGIN
          PurchRcptLine.SETRANGE("Posting Date", CALCDATE('-' + FORMAT(SICAPASetup."Time Find Movement"), DocumentDate),
                CALCDATE(SICAPASetup."Time Find Movement", DocumentDate));
        END;
      END;
      // Show RcptLine
      CLEAR(PagePurchRcptLine);
      PagePurchRcptLine.SETTABLEVIEW(PurchRcptLine);
      PagePurchRcptLine.SETRECORD(PurchRcptLine);
      PagePurchRcptLine.EDITABLE(FALSE);
      PagePurchRcptLine.LOOKUPMODE(TRUE);

      IF (PagePurchRcptLine.RUNMODAL = ACTION::LookupOK) THEN BEGIN
        PagePurchRcptLine.GETRECORD(PurchRcptLine);
        SICAPAInvoiceLine."Receipt No." := PurchRcptLine."Document No.";
        SICAPAInvoiceLine."Receipt Line No." := PurchRcptLine."Line No.";
        SICAPAInvoiceLine.MODIFY;
        ReturnValue := TRUE;
      END;
    END;

      END;

    PROCEDURE SetReconciliation@1000000000(VAR InvoiceHeader@1000000000 : Record 8087767;Reconciliation@1000000001 : Boolean);
    VAR
      InvoiceLines@1000000002 : Record 8087768;
    BEGIN
      InvoiceHeader.NePasRapprocher := NOT Reconciliation;
      IF (NOT Reconciliation) THEN BEGIN
        InvoiceHeader.HoroIntegration := CURRENTDATETIME;
        // Update Lines
        IF (findInvoiceLines(InvoiceHeader,InvoiceLines,InvLineType::" ")) THEN BEGIN
          REPEAT
            InvoiceLines."Receipt No." := '';
            InvoiceLines."Receipt Line No." := 0;
            InvoiceLines.User := USERID;
            InvoiceLines.MODIFY();
          UNTIL (InvoiceLines.NEXT() = 0);
        END;
      END ELSE BEGIN
        InvoiceHeader.HoroIntegration := 0DT;
      END;
      InvoiceHeader.User := USERID;
      InvoiceHeader.MODIFY();
    END;

    LOCAL PROCEDURE checkAndSetVendor@1000000003(VAR SICAPAInvoiceHdr@1000000000 : Record 8087767;VAR Vendor@1000000001 : Record 23) returnVendorFound : Boolean;
    VAR
      PageAction@1000000002 : Action;
    BEGIN
      IF (SICAPAInvoiceHdr."Vendor No." <> '') THEN BEGIN
        EXIT(Vendor.GET(SICAPAInvoiceHdr."Vendor No."));
      END;
      IF SICAPAInvoiceHdr.Emetteur <> '' THEN BEGIN
        Vendor.SETRANGE("Registration No.",SICAPAInvoiceHdr.Emetteur);
        returnVendorFound := NOT Vendor.ISEMPTY;
      END;
      IF NOT returnVendorFound AND (SICAPAInvoiceHdr."Registration No." <> '') THEN BEGIN
        Vendor.SETFILTER("Registration No.",'%1', STRSUBSTNO('%1*', SICAPAInvoiceHdr."Registration No."));
        //MESSAGE('%1', Vendor.GETFILTERS());
        returnVendorFound := NOT Vendor.ISEMPTY;
      END;
      IF NOT returnVendorFound AND (SICAPAInvoiceHdr."VAT Registration No." <> '') THEN BEGIN
        Vendor.SETRANGE("VAT Registration No.",SICAPAInvoiceHdr."VAT Registration No.");
        returnVendorFound := Vendor.FINDFIRST;
        IF NOT returnVendorFound THEN BEGIN
          IF COPYSTR(SICAPAInvoiceHdr."VAT Registration No.",1,2) = 'FR' THEN BEGIN
            Vendor.SETRANGE("Registration No.",COPYSTR(SICAPAInvoiceHdr."VAT Registration No.",3));
            returnVendorFound := NOT Vendor.ISEMPTY;
          END;
        END ELSE
          IF Vendor."Registration No." <> '' THEN
            setWarningLog(SICAPAInvoiceHdr,Vendor,Vendor.FIELDNO("Registration No."),
                          STRSUBSTNO(tVendorRegistMsg,Vendor."VAT Registration No.",Vendor."Registration No.",Vendor."No."));

      END;

      IF returnVendorFound THEN BEGIN
        Vendor.FINDFIRST;
        PageAction := ACTION::LookupOK;
        IF Vendor.COUNT > 1 THEN BEGIN
           //PageAction := PAGE.RUNMODAL(PAGE::"Vendor List",Vendor);
           setErrorLog(SICAPAInvoiceHdr,SICAPAInvoiceHdr,SICAPAInvoiceHdr.FIELDNO("Vendor No."),tSeveralVendorMatchErr);
        END;
        IF PageAction <> ACTION::LookupOK THEN
          IF CONFIRM(tVendorListCanceledQst,FALSE) THEN BEGIN
             setErrorLog(SICAPAInvoiceHdr,SICAPAInvoiceHdr,SICAPAInvoiceHdr.FIELDNO("Vendor No."),tNoVendorMatchErr);
             returnVendorFound := FALSE;
             EXIT(returnVendorFound);
          END ELSE
            ERROR(tWorkIsCanceledErr);

        SICAPAInvoiceHdr.VALIDATE("Vendor No.",Vendor."No.");
      END ELSE
        setErrorLog(SICAPAInvoiceHdr,SICAPAInvoiceHdr,SICAPAInvoiceHdr.FIELDNO("Vendor No."),tNoVendorMatchErr);
      EXIT(returnVendorFound);
    END;

    LOCAL PROCEDURE checkAndSetPaymentTerms@1000000011(VAR SICAPAInvoiceHdr@1000000001 : Record 8087767;Vendor@1000000000 : Record 23) returnCheckPassed : Boolean;
    VAR
      PaymentTerms@1000000002 : Record 3;
      TheoricalDueDate@1000000003 : Date;
    BEGIN
      returnCheckPassed := FALSE;
      IF Vendor."Payment Terms Code" = '' THEN BEGIN
        setErrorLog(SICAPAInvoiceHdr,Vendor,Vendor.FIELDNO("Payment Terms Code"),
                      STRSUBSTNO(tNoPaymentTermsVendorMsg,Vendor."No."))

      END ELSE BEGIN
        IF NOT PaymentTerms.GET(Vendor."Payment Terms Code") THEN
          setErrorLog(SICAPAInvoiceHdr,Vendor,Vendor.FIELDNO("Payment Terms Code"),
                        STRSUBSTNO(tPaymentTermsNotExistMsg,Vendor."Payment Terms Code",Vendor."No."))
        ELSE BEGIN
          IF FORMAT(PaymentTerms."Due Date Calculation") = '' THEN
            setErrorLog(SICAPAInvoiceHdr,PaymentTerms,PaymentTerms.FIELDNO("Due Date Calculation"),
                          STRSUBSTNO(tNoDueDateCalculationMsg,PaymentTerms.Code))
          ELSE BEGIN
            returnCheckPassed := TRUE;
            TheoricalDueDate := CALCDATE(PaymentTerms."Due Date Calculation",SICAPAInvoiceHdr."Date facture");

    VAR
      ItemCrossReference@1000000002 : Record 5717;

      MessageID@1000000004 : GUID;
    BEGIN
      returnCheckPassed := TRUE;
      ModifyLine := FALSE;

      {
      IF ((SICAPAInvoiceLine."Invoice Line Type" <> SICAPAInvoiceLine."Invoice Line Type"::" ") AND
         (SICAPAInvoiceLine."Invoice Line Type" <> SICAPAInvoiceLine."Invoice Line Type"::Item)) THEN BEGIN
         returnCheckPassed := TRUE;
      END ELSE BEGIN
      }
      CASE SICAPAInvoiceLine.Type OF

        SICAPAInvoiceLine.Type::Item : returnCheckPassed := TRUE;
        SICAPAInvoiceLine.Type::" " :
      //  BEGIN

                                    ItemCrossReference."Cross-Reference Type"::"Bar Code",

            returnCheckPassed := searchItemCrossRef(SICAPAInvoiceHdr,SICAPAInvoiceLine,

          IF NOT returnCheckPassed THEN
            returnCheckPassed := searchItemCrossRef(SICAPAInvoiceHdr,SICAPAInvoiceLine,ItemCrossReference."Cross-Reference Type"::Vendor
                                    ,SICAPAInvoiceHdr."Vendor No.",SICAPAInvoiceLine."Code Produit fournisseur");
          IF NOT returnCheckPassed THEN
            returnCheckPassed := searchItemCrossRef(SICAPAInvoiceHdr,SICAPAInvoiceLine,ItemCrossReference."Cross-Reference Type"::Vendor
                                    ,'',SICAPAInvoiceLine."Code Produit fournisseur");
          }
        CASE SICAPAInvoiceLine."Invoice Line Type" OF
          SICAPAInvoiceLine."Invoice Line Type"::Item : BEGIN
            returnCheckPassed := searchItem(SICAPAInvoiceHdr,SICAPAInvoiceLine);
            IF NOT returnCheckPassed THEN BEGIN
               MessageID := setErrorLog(SICAPAInvoiceHdr,SICAPAInvoiceLine,SICAPAInvoiceLine.FIELDNO("Item No. NavAgri"),
                              STRSUBSTNO(tLineNoItemMatch,SICAPAInvoiceLine.Ean13,SICAPAInvoiceLine."Code Produit fournisseur",
                                         SICAPAInvoiceLine.Type,SICAPAInvoiceLine."Market Code",
                                         SICAPAInvoiceLine."Market Description"));
              SICAPAInvoiceLine.Status := SICAPAInvoiceLine.Status::Error;
              SICAPAInvoiceLine."Log Message ID" := MessageID;
            END;
            ModifyLine := TRUE;
          END;
          ELSE BEGIN
            IF SICAPAInvoiceLine."Invoice Line Type" = SICAPAInvoiceLine."Invoice Line Type"::Comment THEN
              returnCheckPassed := SICAPAInvoiceLine."Invoice Attached Line No." <> 0
            ELSE BEGIN
              // UnitCost : seule la ligne attachee est obligatoire (Applied No. non requis)
              // Account, Charge : ligne attachee + Applied Type + Applied No. obligatoires
              returnCheckPassed := (SICAPAInvoiceLine."Invoice Attached Line No." <> 0)
                               AND (SICAPAInvoiceLine."Invoice Line Applied Type" <> SICAPAInvoiceLine."Invoice Line Applied Type"::" ")
      ;
              IF returnCheckPassed AND
                 (SICAPAInvoiceLine."Invoice Line Applied Type" IN [SICAPAInvoiceLine."Invoice Line Applied Type"::Account,
                                                                    SICAPAInvoiceLine."Invoice Line Applied Type"::Charge]) THEN
                returnCheckPassed := SICAPAInvoiceLine."Invoice Line Applied No." <> '';
            END;

            IF NOT returnCheckPassed THEN BEGIN
              returnCheckPassed := searchItem(SICAPAInvoiceHdr,SICAPAInvoiceLine);
              ModifyLine := TRUE;
            END;
            IF NOT returnCheckPassed THEN BEGIN

              ModifyLine := TRUE;
              MessageID := setErrorLog(SICAPAInvoiceHdr,SICAPAInvoiceLine,SICAPAInvoiceLine.FIELDNO(Type),
                           STRSUBSTNO(tLineTypeNotImplementedMsg,SICAPAInvoiceLine.Type,SICAPAInvoiceLine."Market Code",
                                            SICAPAInvoiceLine."Market Description"));
              SICAPAInvoiceLine.Status := SICAPAInvoiceLine.Status::Error;
              SICAPAInvoiceLine."Log Message ID" := MessageID;
            END;
          END;
        END;
      END;

      IF SICAPAInvoiceLine."Source Document No." = '' THEN BEGIN
         SICAPAInvoiceLine."Source Document No." := SICAPAInvoiceHdr."Source Document No.";
         ModifyLine := TRUE;
      END;

      checkLocationCodeFromLine(SICAPAInvoiceLine);
      //END;

      IF ModifyLine THEN
        SICAPAInvoiceLine.MODIFY(TRUE);
      EXIT(returnCheckPassed);
    END;

            (SICAPAInvoiceLineTemp."Source Amount" <> SICAPAInvoiceLine."Source Amount") OR
            (SICAPAInvoiceLineTemp."Source % VAT" <> SICAPAInvoiceLine."Source % VAT") OR
            (SICAPAInvoiceLineTemp."Automatic Reconciliation" <> SICAPAInvoiceLine."Automatic Reconciliation") OR
            (SICAPAInvoiceLineTemp."Log Message ID" <> SICAPAInvoiceLine."Log Message ID") OR
            (SICAPAInvoiceLineTemp.Status <> SICAPAInvoiceLine.Status) OR
            //(SICAPAInvoiceLineTemp."Source Document No."  <> SICAPAInvoiceLine."Source Document No.") OR
            //(SICAPAInvoiceLineTemp."Source Document Line No." <> SICAPAInvoiceLine."Source Document Line No.") OR

        END;
        EXIT(returnReconcilePassed);
      END;
    END;

    PROCEDURE searchReturnShipmentLine@1000000092(VAR SICAPAInvoiceHdr@1000000014 : Record 8087767;VAR SICAPAInvoiceLine@1000000013 : Record 8087768;VendorNo@1000000008 : Code[20];OrderNo@1000000001 : Code[20];OrderLineNo@1000000002 : Integer;ItemNo@1000000003 : Code[20];VariantCode@1000000004 : Code[10];LocationCode@1000000005 : Code[20];QtyToReceive@1000000006 : Decimal;QtyFactor@1000000011 : Decimal;InvoiceUOM@1000000012 : Code[20];VAR ReturnShipLine@1000000000 : Record 6651;VAR MultiSelect@1000000007 : Boolean) returnFound : Boolean;

      ReturnShipLine.RESET;

      // Filtre fournisseur
      IF VendorNo <> '' THEN BEGIN
        //ReturnShipLine.SETCURRENTKEY("Buy-from Vendor No.","Return Order No.","Document No.");
        ReturnShipLine.SETCURRENTKEY("Buy-from Vendor No.");
        ReturnShipLine.SETRANGE("Buy-from Vendor No.",VendorNo);

          ReturnShipLine.SETRANGE("Variant Code",VariantCode);
      END;
      IF LocationCode <> '' THEN
        ReturnShipLine.SETRANGE("Location Code",LocationCode);

      // Filtre quantit' restant Çÿ cr''diter (Qty. Shipped Not Invoiced) ¶ñ tol'rance

    LOCAL PROCEDURE fillInvoiceLineFromReturnShip@1000000093(VAR SICAPAInvoiceHdr@1000000002 : Record 8087767;VAR SICAPAInvoiceLine@1000000000 : Record 8087768;VAR ReturnShipLine@1000000001 : Record 6651;AutoReconcile@1000000003 : Boolean);
    VAR
      CMCSicapaReconcInvLine@1000000006 : Record 60101;
      ResconcilDocType@1000000004 : Integer;
    BEGIN
      // Cr'ation ou mise ÿ jour de la ligne de r'conciliation (60101)
      IF SICAPAInvoiceLine."Invoice Attached Line No." = 0 THEN
        ResconcilDocType := CMCSicapaReconcInvLine."Reconc. Doc. Type"::PostingDoc
      ELSE
        ResconcilDocType := CMCSicapaReconcInvLine."Reconc. Doc. Type"::Child;

      IF NOT CMCSicapaReconcInvLine.GET(
                SICAPAInvoiceLine."Emetteur Type",SICAPAInvoiceLine.Emetteur,
                SICAPAInvoiceLine."Destinataire Type",SICAPAInvoiceLine.Destinataire,
                SICAPAInvoiceLine."Type Facture",SICAPAInvoiceLine."Num Facture",SICAPAInvoiceLine."Num Ligne",
                ResconcilDocType,
                ReturnShipLine."Document No.",ReturnShipLine."Line No.") THEN BEGIN
        CMCSicapaReconcInvLine."Emetteur Type"      := SICAPAInvoiceLine."Emetteur Type";
        CMCSicapaReconcInvLine.Emetteur             := SICAPAInvoiceLine.Emetteur;
        CMCSicapaReconcInvLine."Destinataire Type"  := SICAPAInvoiceLine."Destinataire Type";
        CMCSicapaReconcInvLine.Destinataire         := SICAPAInvoiceLine.Destinataire;
        CMCSicapaReconcInvLine."Type Facture"       := SICAPAInvoiceLine."Type Facture";
        CMCSicapaReconcInvLine."Num Facture"        := SICAPAInvoiceLine."Num Facture";
        CMCSicapaReconcInvLine."Num Ligne"          := SICAPAInvoiceLine."Num Ligne";
        CMCSicapaReconcInvLine."Reconc. Doc. Type"  := ResconcilDocType;
        CMCSicapaReconcInvLine."Reconc. Doc. No."   := ReturnShipLine."Document No.";
        CMCSicapaReconcInvLine."Reconc. Doc. Line No." := ReturnShipLine."Line No.";
        CMCSicapaReconcInvLine.INSERT(TRUE);
      END;

      // Quantit's et montants issus du bon de retour
      CMCSicapaReconcInvLine."Reconc. Received Qty."      := ReturnShipLine.Quantity;
      CMCSicapaReconcInvLine."Reconc. Qty. Not Inv."      := ReturnShipLine."Return Qty. Shipped Not Invd.";
      CMCSicapaReconcInvLine."Reconc. Source VAT %"       := ReturnShipLine."VAT %";
      CMCSicapaReconcInvLine."Reconc. Source Unit Price"  :=
              ROUND(ReturnShipLine."Unit Cost (LCY)" * (1-(ReturnShipLine."Line Discount %"/100)));
      CMCSicapaReconcInvLine."Reconc. Source Doc. No."    := ReturnShipLine."Return Order No.";
      CMCSicapaReconcInvLine."Reconc. Source Doc. Line No." := ReturnShipLine."Return Order Line No.";
      CMCSicapaReconcInvLine.MODIFY;

      // Avertissement si le retour d'achat provient d'un ordre diff'rent de celui de la ligne SICAPA
      IF (SICAPAInvoiceLine."Reconc. Src. Doc. No." <> ReturnShipLine."Return Order No.") AND
         (SICAPAInvoiceLine."Reconc. Src. Doc. No." <> '') THEN BEGIN
        SICAPAInvoiceLine."Log Message ID" := setWarningLog(SICAPAInvoiceHdr,SICAPAInvoiceLine,
                                               SICAPAInvoiceLine.FIELDNO("Source Document No."),
                                               tOrderNoIsDifferentMsg);
        IF SICAPAInvoiceLine.Status <> SICAPAInvoiceLine.Status::Error THEN
          SICAPAInvoiceLine.Status := SICAPAInvoiceLine.Status::Warning;
      END;

      //PCDDR4 : 'cart taux TVA
      IF CMCSicapaReconcInvLine."Reconc. Source VAT %" <> SICAPAInvoiceLine."Taux Tva" THEN BEGIN
        SICAPAInvoiceLine."Log Message ID" := setWarningLog(SICAPAInvoiceHdr,SICAPAInvoiceLine,
                                               SICAPAInvoiceLine.FIELDNO("Source % VAT"),
                                               tGapVATMsg);
        IF SICAPAInvoiceLine.Status <> SICAPAInvoiceLine.Status::Error THEN
          SICAPAInvoiceLine.Status := SICAPAInvoiceLine.Status::Warning;
      END;

      SICAPAInvoiceLine."Received Quantity" += CMCSicapaReconcInvLine."Reconc. Qty. Not Inv.";
      SICAPAInvoiceLine."Source Amount"     += ROUND(CMCSicapaReconcInvLine."Reconc. Source Unit Price" *
                                               CMCSicapaReconcInvLine."Reconc. Qty. Not Inv.");

      IF SICAPAInvoiceLine."Received Quantity" <> 0 THEN

    LOCAL PROCEDURE searchAndCheckReceiptLines@1000000032(VAR SICAPAInvoiceHdr@1000000005 : Record 8087767;VAR SICAPAInvoiceLine@1000000004 : Record 8087768;VAR PurchRcptLine@1000000000 : Record 121) : Boolean;
    VAR
      QtyToCheck@1000000001 : Decimal;
      QtyToCheckMarked@1000000008 : Decimal;
      QtyMin@1000000006 : Decimal;
      QtyMax@1000000007 : Decimal;
      SourceDoc@1000000002 : Code[20];
      SourceLine@1000000003 : Integer;
    BEGIN
      getToleranceItemQty(SICAPAInvoiceHdr,SICAPAInvoiceLine,
            SICAPAInvoiceLine."Quantite facture",SICAPAInvoiceLine."Coefficient Unit",SICAPAInvoiceLine.Unite,QtyMin,QtyMax);
      IF PurchRcptLine.FINDSET THEN
         REPEAT
           IF CheckReceiptLineResconcil(SICAPAInvoiceHdr,SICAPAInvoiceLine,PurchRcptLine) THEN BEGIN
              QtyToCheck += PurchRcptLine."Qty. Rcd. Not Invoiced" * PurchRcptLine."Qty. per Unit of Measure";
              IF PurchRcptLine."Qty. Rcd. Not Invoiced" <> 0 THEN BEGIN
                IF SourceDoc = '' THEN
                   SourceDoc := PurchRcptLine."Order No."
                ELSE IF SourceDoc <> PurchRcptLine."Order No." THEN BEGIN
                   SourceDoc := 'Multiple';
                   SourceLine := 0;
                END;
                IF SourceDoc = PurchRcptLine."Order No." THEN
                   IF SourceLine = 0 THEN
                      SourceLine := PurchRcptLine."Order Line No."
                   ELSE IF SourceLine <> PurchRcptLine."Order Line No." THEN
                      SourceLine := -1;
              END;
              IF (QtyToCheck <= QtyMax) THEN BEGIN
                 PurchRcptLine.MARK(TRUE);
                 QtyToCheckMarked := QtyToCheck;
              END;
           END;
         UNTIL PurchRcptLine.NEXT = 0;

        SICAPAInvoiceLine."Reconc. Src. Doc. No." := SourceDoc;
        IF SourceLine > 0 THEN
          SICAPAInvoiceLine."Reconc. Src. Doc. Line No." := SourceLine;
      END;
      EXIT(QtyToCheckMarked >= QtyMin);
    END;

                 ReturnValue := FALSE;
              END ELSE BEGIN
                ReturnValue := TRUE;
              END;
            END;
           END ELSE BEGIN
            // the item has been shipped ?
            IF (((SICAPAInvoiceHdr."Document Type" = SICAPAInvoiceHdr."Document Type"::Invoice) AND
                (PurchaseLine."Quantity Received"  = 0)) OR
                ((SICAPAInvoiceHdr."Document Type" = SICAPAInvoiceHdr."Document Type"::"Credit Memo") AND
                (PurchaseLine."Return Qty. Shipped" = 0))) THEN BEGIN

            END ELSE BEGIN

            END;
           END;
        END;
      END;

    PROCEDURE CheckInvoiceLine@1000000026(VAR SICAPAInvoiceHdr@1000000002 : Record 8087767;VAR SICAPAInvoiceLine@1000000000 : Record 8087768;AutoReconcile@1000000003 : Boolean);
    VAR
      QtyMin@1000000004 : Decimal;
      QtyMax@1000000005 : Decimal;
      CMCSicapaReconcInvLine@1000000006 : Record 60101;
    BEGIN
      // Commiter l'etat courant en base avant les appels a setWarningLog
      // (LogManagement.AddMessage resout le RecordID/FieldNo depuis la base)
      SICAPAInvoiceLine.MODIFY;

      //PCDDR5 : prix unitaire BR complet vs prix net commande + cout ajuste
      // Source Amount = montant BR complet (article + UnitCost du fournisseur)
      // Source Unit Price = Source Amount / Received Quantity
      // Reference = "Prix net" (prix article commande) + UnitCost ramene a l'unite
      IF SICAPAInvoiceLine."Received Quantity" <> 0 THEN BEGIN
        SICAPAInvoiceLine."Source Unit Price" :=
          (SICAPAInvoiceLine."Source Amount" / SICAPAInvoiceLine."Received Quantity");
          //ROUND(SICAPAInvoiceLine."Source Amount" / SICAPAInvoiceLine."Received Quantity");
        IF ABS(SICAPAInvoiceLine."Source Unit Price" -
              (SICAPAInvoiceLine."Prix net" +
               ROUND(getUnitCostChildAmount(SICAPAInvoiceLine) / SICAPAInvoiceLine."Received Quantity"))) > 0.01 THEN BEGIN
          SICAPAInvoiceLine."Log Message ID" := setWarningLog(SICAPAInvoiceHdr,SICAPAInvoiceLine,
                                               SICAPAInvoiceLine.FIELDNO("Source Unit Price"),
                                                tGapUnitPriceMsg);
          IF SICAPAInvoiceLine.Status <> SICAPAInvoiceLine.Status::Error THEN
             SICAPAInvoiceLine.Status := SICAPAInvoiceLine.Status::Warning;
        END;
      END;

      //PCDDR6
      getToleranceItemQty(SICAPAInvoiceHdr,SICAPAInvoiceLine,SICAPAInvoiceLine."Received Quantity",
                          SICAPAInvoiceLine."Coefficient Unit", SICAPAInvoiceLine.Unite,QtyMin,QtyMax);
      IF ((SICAPAInvoiceLine."Quantite facture" < QtyMin) OR (SICAPAInvoiceLine."Quantite facture" > QtyMax)) THEN BEGIN
        SICAPAInvoiceLine."Log Message ID" := setWarningLog(SICAPAInvoiceHdr,SICAPAInvoiceLine,
                                             SICAPAInvoiceLine.FIELDNO("Received Quantity"),
                                              tGapQuantityMsg);
        IF SICAPAInvoiceLine.Status <> SICAPAInvoiceLine.Status::Error THEN
           SICAPAInvoiceLine.Status := SICAPAInvoiceLine.Status::Warning;
      END;
      //PCDDR7 : Montant_article + UnitCost doit egal Source Amount (montant BR)
      IF ((SICAPAInvoiceLine.Montant + getUnitCostChildAmount(SICAPAInvoiceLine)) <> SICAPAInvoiceLine."Source Amount") THEN BEGIN
        SICAPAInvoiceLine."Log Message ID" := setWarningLog(SICAPAInvoiceHdr,SICAPAInvoiceLine,
                                             SICAPAInvoiceLine.FIELDNO(Montant),
                                              tGapAmountMsg);
        IF SICAPAInvoiceLine.Status <> SICAPAInvoiceLine.Status::Error THEN
           SICAPAInvoiceLine.Status := SICAPAInvoiceLine.Status::Warning;
      END;
    END;

    PROCEDURE RebuildInvoiceLineReconciliati@15001020(VAR SicapaInvoiceHdr@15002000 : Record 8087767;VAR SicapaInvoiceLine@15002001 : Record 8087768);
    VAR
      SicapaReconInvLine@15002010 : Record 60101;
      ReconciliationCount@15002011 : Integer;
      VATPct@15002012 : Decimal;
      VATSet@15002013 : Boolean;
      MultiVAT@15002014 : Boolean;
      FirstReceiptNo@15002015 : Code[20];
      FirstReceiptLineNo@15002016 : Integer;
    BEGIN
      ResetInvoiceLineReconFields(SicapaInvoiceLine);

      // Uniquement les enregistrements PostingDoc (BR ou retour) contribuent a
      // Received Quantity et Source Amount. Les Child (UnitCost, Charge, Account, Comment)
      // sont pris en compte separement via getUnitCostChildAmount dans CheckInvoiceLine.
      IF findReconciliationLines(SicapaInvoiceLine,SicapaReconInvLine) THEN BEGIN
        SicapaReconInvLine.SETRANGE("Reconc. Doc. Type", SicapaReconInvLine."Reconc. Doc. Type"::PostingDoc);
        IF SicapaReconInvLine.FINDSET(FALSE,FALSE) THEN
          REPEAT
            ReconciliationCount += 1;

            IF ReconciliationCount = 1 THEN BEGIN
              FirstReceiptNo := SicapaReconInvLine."Reconc. Doc. No.";
              FirstReceiptLineNo := SicapaReconInvLine."Reconc. Doc. Line No.";
            END;

            AddReconciliationLineToInvoice(SicapaInvoiceLine,SicapaReconInvLine);

            IF NOT VATSet THEN BEGIN
              VATPct := SicapaReconInvLine."Reconc. Source VAT %";
              VATSet := TRUE;
            END ELSE
              IF VATPct <> SicapaReconInvLine."Reconc. Source VAT %" THEN
                MultiVAT := TRUE;
          UNTIL SicapaReconInvLine.NEXT = 0;
      END;

      // Source Amount = montant BR tel quel (inclut deja les UnitCost du fournisseur)
      // PCDDR7 dans CheckInvoiceLine compare Montant_article + UnitCost au Source Amount

      IF SicapaInvoiceLine."Received Quantity" <> 0 THEN
        SicapaInvoiceLine."Source Unit Price" :=
          (SicapaInvoiceLine."Source Amount" / SicapaInvoiceLine."Received Quantity");
          //ROUND(SicapaInvoiceLine."Source Amount" / SicapaInvoiceLine."Received Quantity");

      IF VATSet AND NOT MultiVAT THEN
        SicapaInvoiceLine."Source % VAT" := VATPct;

      // Compatibility only: 60101 remains the table of truth.
      IF ReconciliationCount = 1 THEN BEGIN
        SicapaInvoiceLine."Receipt No." := FirstReceiptNo;
        SicapaInvoiceLine."Receipt Line No." := FirstReceiptLineNo;
      END;

      SicapaInvoiceLine."Automatic Reconciliation" := FALSE;
    END;

      Text002@1000000003 : TextConst 'FRA=Type facture %1 non g'r'. Valeurs attendues : 380 facture, 381 avoir.';

      InvoiceLine@1000000010 : Record 8087768;
      ReconcLine@1000000011 : Record 60101;
      MessageID@1000000007 : GUID;
    BEGIN
      CanCreate := FALSE;

      IF (InvoiceHeader."Invoice No." <> '') OR (InvoiceHeader.HoroIntegration <> 0DT) THEN BEGIN

    LOCAL PROCEDURE createPurchaseLines@1000000045(VAR InvoiceHeader@1000000000 : Record 8087767;VAR PurchHeader@1000000001 : Record 38;ShowError@1000000005 : Boolean);
    VAR
      InvoiceLine@1000000002 : Record 8087768;
      ReconcLine@1000000003 : Record 60101;
      NextLineNo@1000000004 : Integer;
      MessageID@1000000006 : GUID;
    BEGIN
      NextLineNo := 0;

      IF NOT findInvoiceLines(InvoiceHeader, InvoiceLine,InvLineType::Item) THEN
        IF ShowError THEN
          ERROR(Text003, InvoiceHeader."Num Facture")
        ELSE BEGIN
          MessageID := setErrorLog(InvoiceHeader,InvoiceLine,InvoiceLine.FIELDNO("Num Facture"),
                      STRSUBSTNO(Text003, InvoiceHeader."Num Facture"));
          InvoiceLine.Status := InvoiceLine.Status::Error;
          InvoiceLine."Log Message ID" := MessageID;
          EXIT;
        END;

      // --- PASSE 1 : lignes Article (Reconc. Doc. Type = PostingDoc) ---
      REPEAT
        IF findReconciliationLines(InvoiceLine, ReconcLine) THEN BEGIN
          ReconcLine.SETRANGE("Reconc. Doc. Type", ReconcLine."Reconc. Doc. Type"::PostingDoc);
          IF ReconcLine.FINDSET THEN
            REPEAT
              NextLineNo += 10000;
              createPurchaseLineFromReconcil(InvoiceHeader,InvoiceLine,ReconcLine,PurchHeader,NextLineNo);
            UNTIL ReconcLine.NEXT = 0;
        END ELSE IF InvoiceLine.Type = InvoiceLine.Type::Item THEN BEGIN
          // Seules les lignes Article sans reconciliation sont une erreur bloquante
          IF ShowError THEN
            ERROR(Text004, InvoiceLine."Num Facture", InvoiceLine."Num Ligne")
          ELSE BEGIN
            MessageID := setErrorLog(InvoiceHeader,InvoiceLine,InvoiceLine.FIELDNO("Num Facture"),
                        STRSUBSTNO(Text004, InvoiceLine."Num Facture", InvoiceLine."Num Ligne"));
            InvoiceLine.Status := InvoiceLine.Status::Error;
            InvoiceLine."Log Message ID" := MessageID;
            EXIT;
          END;
        END;
      UNTIL InvoiceLine.NEXT = 0;
      // --- PASSE 2 : lignes Remise / Frais / Commentaire (Reconc. Doc. Type = Child) ---
      // Toutes les lignes Article ont deja ete creees et leurs ReconcLine "Linked Document..."
      // sont renseign'es : on peut donc retrouver la ligne d'achat parente.
      IF findInvoiceLines(InvoiceHeader, InvoiceLine,InvLineType::Other) THEN
      REPEAT
        IF findReconciliationLines(InvoiceLine, ReconcLine) THEN BEGIN
          ReconcLine.SETRANGE("Reconc. Doc. Type", ReconcLine."Reconc. Doc. Type"::Child);
          IF ReconcLine.FINDSET THEN
            REPEAT
              NextLineNo += 10000;
              createPurchaseLineFromChild(InvoiceHeader, InvoiceLine, ReconcLine, PurchHeader, NextLineNo);
            UNTIL ReconcLine.NEXT = 0;
        END;
      UNTIL InvoiceLine.NEXT = 0;
    END;

    LOCAL PROCEDURE createPurchaseLineFromReconcil@1000000046(VAR InvoiceHeader@1000000000 : Record 8087767;VAR InvoiceLine@1000000001 : Record 8087768;VAR ReconcLine@1000000002 : Record 60101;VAR PurchHeader@1000000003 : Record 38;VAR LineNo@1000000004 : Integer);
    VAR
      PurchLine@1000000005 : Record 39;
      PurchRcptLine@1000000006 : Record 121;
      ReturnShipLine@1000000007 : Record 6651;
      FromPostedDocDim@1000000008 : Record 359;
    BEGIN
      CLEAR(PurchLine);

      PurchLine.VALIDATE("Document Type", PurchHeader."Document Type");
      PurchLine.VALIDATE("Document No.", PurchHeader."No.");
      PurchLine.VALIDATE("Line No.", LineNo);
      // Attention gestion des types peut ^tre un CASE
      PurchLine.VALIDATE(Type, InvoiceLine.Type);
      PurchLine.VALIDATE("No.", InvoiceLine."Item No. NavAgri");

      IF InvoiceLine.Type = InvoiceLine.Type::Item THEN
        PurchLine.VALIDATE("Variant Code", InvoiceLine."Variant Code NavAgri");

      IF InvoiceHeader."Type Facture" = '381' THEN BEGIN
        getReturnShipLineFromReconcil(InvoiceLine, ReconcLine, ReturnShipLine);
        setPurchaseLineFromReturnShip(PurchLine, ReturnShipLine);
        SetFilterShippingDim(ReturnShipLine,FromPostedDocDim);
        CopyDimBufToDocDim(FromPostedDocDim,PurchLine);
      END ELSE BEGIN
        getPurchRcptLineFromReconcil(InvoiceLine, ReconcLine, PurchRcptLine);

        SetFilterReceiptDim(PurchRcptLine,FromPostedDocDim);

      END;

      setPurchaseLineQuantityAndCost(InvoiceLine, ReconcLine, PurchLine);
      setPurchaseLineSourceLink(InvoiceHeader, ReconcLine, PurchLine);
      ReconcLine.MODIFY;

      IF NOT PurchLine.INSERT(TRUE) THEN
        ERROR(Text012, PurchLine.TABLECAPTION, PurchLine."No.", PurchLine."Line No.");

      setUnitCostFromSource(PurchLine);
    END;

    LOCAL PROCEDURE createPurchaseLineFromChild@1000000065(VAR InvoiceHeader@1000000000 : Record 8087767;VAR InvoiceLine@1000000001 : Record 8087768;VAR ReconcLine@1000000008 : Record 60101;VAR PurchHeader@1000000002 : Record 38;VAR LineNo@1000000003 : Integer);

        END;

    LOCAL PROCEDURE applyParentVATPostingGroups@1000000078(VAR InvoiceHeader@1000000001 : Record 8087767;VAR InvoiceLine@1000000000 : Record 8087768;VAR ParentPurchLine@1000000002 : Record 39;VAR PurchLine@1000000003 : Record 39);

    BEGIN
      // La TVA de la ligne d'achat parente fait foi
      IF ParentPurchLine."VAT Bus. Posting Group" <> PurchLine."VAT Bus. Posting Group" THEN

        IF VATPostingSetup."VAT %" <> InvoiceLine."Taux Tva" THEN BEGIN

    LOCAL PROCEDURE applyUnitCostAdjustment@1000000067(VAR InvoiceHeader@1000000002 : Record 8087767;VAR InvoiceLine@1000000001 : Record 8087768;VAR PurchHeader@1000000000 : Record 38;Sign@1000000003 : Decimal;VAR ReconcLine@1000000007 : Record 60101);
    VAR
      ParentPurchLine@1000000004 : Record 39;
      AdditionalUnitCost@1000000005 : Decimal;
      MessageID@1000000006 : GUID;
      Text021@1000000009 : TextConst 'FRA=Ligne d''''achat parente introuvable pour l''ajustement de cout (facture %1 ligne %2).';
      Text022@1000000008 : TextConst 'FRA=Quantit' nulle sur la ligne d''''achat parente, ajustement de cout impossible (facture %1 ligne %2).';
    BEGIN
      IF NOT getLinkedPurchaseLine(PurchHeader, InvoiceLine, ParentPurchLine) THEN BEGIN
        MessageID := setErrorLog(InvoiceHeader,InvoiceLine,InvoiceLine.FIELDNO("Invoice Attached Line No."),
                    STRSUBSTNO(Text021, InvoiceLine."Num Facture", InvoiceLine."Num Ligne"));
        InvoiceLine.Status := InvoiceLine.Status::Error;
        InvoiceLine."Log Message ID" := MessageID;
        InvoiceLine.MODIFY(TRUE);
        EXIT;
      END;

      IF ParentPurchLine.Quantity = 0 THEN BEGIN
        MessageID := setErrorLog(InvoiceHeader,InvoiceLine,InvoiceLine.FIELDNO("Invoice Attached Line No."),
                    STRSUBSTNO(Text022, InvoiceLine."Num Facture", InvoiceLine."Num Ligne"));
        InvoiceLine.Status := InvoiceLine.Status::Error;
        InvoiceLine."Log Message ID" := MessageID;
        InvoiceLine.MODIFY(TRUE);
        EXIT;
      END;

      // Ajustement unitaire = montant UnitCost / quantite de la ligne article
      AdditionalUnitCost := ROUND((Sign * InvoiceLine.Montant) / ParentPurchLine.Quantity, 0.00001);
      ParentPurchLine.VALIDATE("Direct Unit Cost", ParentPurchLine."Direct Unit Cost" + AdditionalUnitCost);
      ParentPurchLine.MODIFY(TRUE);

      // Lier la ReconcLine 60101 a la ligne achat parente (dont le cout a ete majore)
      setPurchaseLineSourceLink(InvoiceHeader, ReconcLine, ParentPurchLine);
      ReconcLine.MODIFY;

      IF InvoiceLine.Status = InvoiceLine.Status::" " THEN
        InvoiceLine.Status := InvoiceLine.Status::Success;
      InvoiceLine.MODIFY(TRUE);
    END;

    LOCAL PROCEDURE setUnitCostFromSource@1000000054(VAR PurchaseLine@1000000000 : Record 39);
    VAR
      PurchRcptLine@1000000001 : Record 121;
      ReturnShipLine@1000000002 : Record 6651;
      PurchLineSource@1000000003 : Record 39;
      Price@1000000004 : Decimal;
    BEGIN
      Price := 0;

      CASE PurchaseLine."Document Type" OF
        PurchaseLine."Document Type"::Invoice:
          BEGIN
            PurchRcptLine.RESET;

            IF PurchRcptLine.FINDFIRST THEN BEGIN
              Price := ROUND(PurchRcptLine."Direct Unit Cost" * ((100 - PurchRcptLine."Line Discount %")/100));
            END;
          END;

        PurchaseLine."Document Type"::"Credit Memo":
          BEGIN
            ReturnShipLine.RESET;
            ReturnShipLine.SETRANGE("Document No.", PurchaseLine."Return Shipment No.");
            ReturnShipLine.SETRANGE("Line No.", PurchaseLine."Return Shipment Line No.");

              PurchLineSource.RESET;

              PurchLineSource.SETRANGE("Document No.", ReturnShipLine."Return Order No.");
              PurchLineSource.SETRANGE("Line No.", ReturnShipLine."Return Order Line No.");
              IF PurchLineSource.FINDFIRST THEN
                Price := ROUND(PurchLineSource."Direct Unit Cost" * ((100 - PurchLineSource."Line Discount %")/100));
            END;
          END;
      END;

      IF Price <> 0 THEN BEGIN
        PurchaseLine."Unit Price (LCY)" := Price;
        PurchaseLine.MODIFY;
      END;
    END;
