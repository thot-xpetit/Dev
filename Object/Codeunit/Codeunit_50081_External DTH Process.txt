    PROCEDURE HasCCCHeader@1000000026(VAR ContractCode@1000000000 : Code[20];ContractType@1000000001 : Code[10];CropRotationCampaignNo@1000000002 : Code[10];ContactNo@1000000003 : Code[20];VAR ErrorMessage@1000000005 : Text[250]) ReturnValue : Boolean;
    VAR
      CCCHeader@1000000004 : Record 8053208;
    BEGIN
      ReturnValue := FALSE;
      CCCHeader.SETRANGE("Contract Type", ContractType);
      CCCHeader.SETRANGE("Campaign No.", GetCampaign(CropRotationCampaignNo));
      CCCHeader.SETRANGE("Crop Rotation Campaign No.", CropRotationCampaignNo);
      CCCHeader.SETRANGE("Contact No.", ContactNo);

      IF (CCCHeader.FINDFIRST()) THEN BEGIN
        ContractCode := CCCHeader."Contract No.";
        ReturnValue := TRUE;
      END ELSE BEGIN
        IF (ContractCode <> '') THEN BEGIN
          ErrorMessage := STRSUBSTNO(tErrContractNoExists, ContractCode, ContactNo);
        END;
      END;
    END;

    LOCAL PROCEDURE GetCampaign@1000000009(CropCampaignNo@1000000000 : Code[10]) ReturnValue : Code[10];
    VAR
      recCampaign@1000000001 : Record 8052991;
      recCampaign2@1000000002 : Record 8052991;
      recCMCSetup@1000000003 : Record 8052982;
    BEGIN
      ReturnValue := '';
      IF (recCMCSetup.GET()) THEN BEGIN
        IF (recCampaign.GET(CropCampaignNo)) THEN BEGIN
          recCampaign2.RESET;
          recCampaign2.SETCURRENTKEY("Type Code","Starting Date");
          recCampaign2.SETFILTER("Starting Date", '>%1', recCampaign."Ending Date");
          recCampaign2.FINDFIRST;

          IF (recCMCSetup."Frame Contract On Next Campaig" = FALSE) THEN BEGIN
            ReturnValue := recCampaign."No.";
          END ELSE BEGIN
            ReturnValue := recCampaign2."No.";
          END;
        END;
      END;
    END;

      CCCHeaderForCommitment_L@1000000013 : Record 8053208;
    BEGIN
      ErrorMessage := '';
      IF (HasCCCHeader(ContractCode, ContractType, CropRotationCampaignNo, contactNo, ErrorMessage)) THEN BEGIN
        IF (CCCHeader.GET(ContractCode)) THEN BEGIN
            ADVCRMProcess.CheckCropRotationFromCCC(CCCHeader);

            // --> 20200930-WI00-ADV01.01.00-3497-CJY01
            CCCHeader.SetNoTriggerTrue(TRUE);
            // <-- 20200930-WI00-ADV01.01.00-3497-CJY01

              CCCHeader.VALIDATE("Commitment Code",CommitmentCode);
              IF CCCHeader."Last Modification Date" <> WORKDATE THEN BEGIN
                CCCHeader."Last Modification Date" := WORKDATE;
              END;

            END;

            // --> 20200904-Wi00-ADV01.01.00-3471-CJY01
            IF CCCHeader."Contact No." <> contactNo THEN
            // <-- 20200904-Wi00-ADV01.01.00-3471-CJY01
              CCCHeader.VALIDATE("Contact No.", contactNo);
            // --> 20200904-Wi00-ADV01.01.00-3471-CJY01
            IF CCCHeader."Campaign No." <> GetCampaign(CropRotationCampaignNo) THEN
            // <- 20200904-Wi00-ADV01.01.00-3471-CJY01
              CCCHeader.VALIDATE("Campaign No.", GetCampaign(CropRotationCampaignNo));

            // --> 20200904-Wi00-ADV01.01.00-3471-CJY01
            IF CCCHeader."Crop Rotation Campaign No." <> CropRotationCampaignNo THEN
            // <-- 20200904-Wi00-ADV01.01.00-3471-CJY01
              CCCHeader.VALIDATE("Crop Rotation Campaign No.", CropRotationCampaignNo);
            // --> 20200904-Wi00-ADV01.01.00-3471-CJY01

              CCCHeader."Last Modification Date" := WORKDATE;

              //ERROR('Message debug : error modify CU 50081');

              // --> 20200914-WI00-ADV01.01.00-3471-CJY01

              //GuidManagement_L.UpdateCCCGuid2(ContractCode,idcrm,ErrorMessage);

            //gestion du lien GUID2
