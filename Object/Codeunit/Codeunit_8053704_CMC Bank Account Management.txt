            recCustBankAcc_L.SETFILTER("Customer No.", '<>%1', '');
            IF recCustBankAcc_L.FINDSET(FALSE, FALSE) THEN BEGIN
              REPEAT
                recCustBankAccTmp_L.INIT;
                recCustBankAccTmp_L.TRANSFERFIELDS(recCustBankAcc_L, TRUE);

                  recCustomer_L.GET(recCustBankAccTmp_L."Customer No.");
                  Modify_L := FALSE;
                  Code_L := SearchNewCodeNo(Contact_L, recCustBankAccTmp_L.Code, recCustBankAccTmp_L."Bank Branch No.",

                                            recCustBankAccTmp_L."Agency Code", recCustBankAccTmp_L."RIB Key");

                    Modify_L := TRUE;
                  END;

                  IF Modify_L = FALSE THEN BEGIN
                    CLEAR(recBankAcc_L);

                    recBankAcc_L.INIT;
                    recBankAcc_L."Contact No." := Contact_L;
                    recBankAcc_L.Code := Code_L;
                    recBankAcc_L.TRANSFERFIELDS(recCustBankAccTmp_L, FALSE);
                    //Suppression du compte bancaire client
                    IF recCustBankAccDelete_L.GET(recCustBankAccTmp_L."Customer No.", recCustBankAccTmp_L.Code) THEN BEGIN
                      recCustBankAccDelete_L.DELETE;
                    END;
                    IF recCustBankAccDelete_L.GET(recCustBankAccTmp_L."Customer No.", Code_L) THEN BEGIN
                      recCustBankAccDelete_L.DELETE;
                    END;
                    //Cr'ation du compte bancaire client et contact
                    recBankAcc_L.VALIDATE("Customer RIB", TRUE);
                    IF recCustomer_L."Default Bank Account Code" = recCustBankAccTmp_L.Code THEN BEGIN
                      recBankAcc_L.VALIDATE("Default Customer RIB", TRUE);
                    END;
                    recBankAcc_L.INSERT;
                  END
                  ELSE BEGIN
                    //Suppression du compte bancaire client
                    IF recCustBankAccDelete_L.GET(recCustBankAccTmp_L."Customer No.", recCustBankAccTmp_L.Code) THEN BEGIN
                      recCustBankAccDelete_L.DELETE;
                    END;
                    IF recCustBankAccDelete_L.GET(recCustBankAccTmp_L."Customer No.", Code_L) THEN BEGIN
                      recCustBankAccDelete_L.DELETE;
                    END;
                    recBankAcc_L."Customer RIB" := FALSE;
                    recBankAcc_L.MODIFY;
                    //Cr'ation du compte bancaire client et contact
                    recBankAcc_L.VALIDATE("Customer RIB", TRUE);
                    IF recCustomer_L."Default Bank Account Code" = recBankAcc_L.Code THEN BEGIN
                      recBankAcc_L.VALIDATE("Default Customer RIB", TRUE);
                    END;
                    recBankAcc_L.MODIFY;
                  END;
                END;

                recVendorBankAccTmp_L.INSERT;
              UNTIL recVendorBankAcc_L.NEXT = 0;
            END;

            recVendorBankAccTmp_L.RESET;
            IF recVendorBankAccTmp_L.FINDSET(FALSE, FALSE) THEN BEGIN
              REPEAT
                Contact_L := SearchContactFarmerFromVendor(recVendorBankAccTmp_L."Vendor No.");
                IF Contact_L <> '' THEN BEGIN
                  recVendor_L.GET(recVendorBankAccTmp_L."Vendor No.");
                  Modify_L := FALSE;
                  Code_L := SearchNewCodeNo(Contact_L, recVendorBankAccTmp_L.Code, recVendorBankAccTmp_L."Bank Branch No.",
                                            recVendorBankAccTmp_L."Bank Account No.", recVendorBankAccTmp_L."Currency Code",
                                            recVendorBankAccTmp_L.IBAN, recVendorBankAccTmp_L."SWIFT Code",
                                            recVendorBankAccTmp_L."Agency Code", recVendorBankAccTmp_L."RIB Key");
                  IF recBankAcc_L.GET(Contact_L, Code_L) THEN BEGIN
                    Modify_L := TRUE;
                  END;

                  IF Modify_L = FALSE THEN BEGIN
                    CLEAR(recBankAcc_L);
                    //Recopie des informations du compte bancaire
                    recBankAcc_L.INIT;
                    recBankAcc_L."Contact No." := Contact_L;
                    recBankAcc_L.Code := Code_L;
                    recBankAcc_L.TRANSFERFIELDS(recVendorBankAccTmp_L, FALSE);
                    //Suppression du compte bancaire client
                    IF recVendorBankAccDelete_L.GET(recVendorBankAccTmp_L."Vendor No.", recVendorBankAccTmp_L.Code) THEN BEGIN
                      recVendorBankAccDelete_L.DELETE;
                    END;
                    IF recVendorBankAccDelete_L.GET(recVendorBankAccTmp_L."Vendor No.", Code_L) THEN BEGIN
                      recVendorBankAccDelete_L.DELETE;
                    END;
                    //Cr'ation du compte bancaire client et contact
                    recBankAcc_L.VALIDATE("Vendor RIB", TRUE);
                    IF recVendor_L."Default Bank Account Code" = recVendorBankAccTmp_L.Code THEN BEGIN
                      recBankAcc_L.VALIDATE("Default Vendor RIB", TRUE);
                    END;
                    recBankAcc_L.INSERT;
                  END
                  ELSE BEGIN
                    //Suppression du compte bancaire client
                    IF recVendorBankAccDelete_L.GET(recVendorBankAccTmp_L."Vendor No.", recVendorBankAccTmp_L.Code) THEN BEGIN
                      recVendorBankAccDelete_L.DELETE;
                    END;
                    IF recVendorBankAccDelete_L.GET(recVendorBankAccTmp_L."Vendor No.", Code_L) THEN BEGIN
                      recVendorBankAccDelete_L.DELETE;
                    END;
                    recBankAcc_L."Vendor RIB" := FALSE;
                    recBankAcc_L.MODIFY;
                    //Cr'ation du compte bancaire client et contact
                    recBankAcc_L.VALIDATE("Vendor RIB", TRUE);
                    IF recVendor_L."Default Bank Account Code" = recBankAcc_L.Code THEN BEGIN
                      recBankAcc_L.VALIDATE("Default Vendor RIB", TRUE);
                    END;
                    recBankAcc_L.MODIFY;
                  END;
                END;
              UNTIL recVendorBankAccTmp_L.NEXT = 0;
            END;
          END;

    PROCEDURE SearchContactFarmerFromCust@8053001(Customer_P@8053003 : Code[20]) Result_L : Code[20];
    VAR
      recContact_L@8053000 : Record 5050;
      recContBusRel_L@8053001 : Record 5054;
      cduMCDataMgt_L@8053002 : Codeunit 8053306;
    BEGIN
      //----------------------------------------------------------------------------------------------------------------------------------
      // Recherche le contact
      //----------------------------------------------------------------------------------------------------------------------------------
      //  Entr'es
      //    - Customer_P : No de client
      //  Sorties
      //    - No de contact
      //----------------------------------------------------------------------------------------------------------------------------------
      Result_L := '';
      recContBusRel_L.RESET;
      IF cduMCDataMgt_L.IsMCCompany(DATABASE::Customer) THEN
        recContBusRel_L.SETRANGE("MC Company Code", cduMCDataMgt_L.GetMCCompanyCode(''));
      recContBusRel_L.SETRANGE("No.", Customer_P);
      recContBusRel_L.SETRANGE("Link to Table", recContBusRel_L."Link to Table"::Customer);
      IF recContBusRel_L.FINDFIRST THEN BEGIN
        IF recContact_L.GET(recContBusRel_L."Contact No.") THEN BEGIN
          IF recContact_L."Contact Type" = recContact_L."Contact Type"::Farm THEN BEGIN
            Result_L := recContBusRel_L."Contact No.";
          END;
        END;
      END;

      EXIT(Result_L);
    END;

    PROCEDURE SearchContactFarmerFromVendor@8053004(Vendor_P@8053003 : Code[20]) Result_L : Code[20];
    VAR
      recContact_L@8053000 : Record 5050;
      recContBusRel_L@8053001 : Record 5054;
      cduMCDataMgt_L@8053002 : Codeunit 8053306;
    BEGIN
      //----------------------------------------------------------------------------------------------------------------------------------
      // Recherche le contact
      //----------------------------------------------------------------------------------------------------------------------------------
      //  Entr'es
      //    - Vendor_P : No de fournisseur
      //  Sorties
      //    - No de contact
      //----------------------------------------------------------------------------------------------------------------------------------

      recContBusRel_L.SETRANGE("Link to Table", recContBusRel_L."Link to Table"::Vendor);
      IF recContBusRel_L.FINDFIRST THEN BEGIN
        IF recContact_L.GET(recContBusRel_L."Contact No.") THEN BEGIN
          IF recContact_L."Contact Type" = recContact_L."Contact Type"::Farm THEN BEGIN
            Result_L := recContBusRel_L."Contact No.";

    PROCEDURE SearchNewCodeNo@8053000(Contact_P@8053003 : Code[20];Code_P@8053012 : Code[20];BankBranchNo_P@8053004 : Text[20];BankAccountNo_P@8053005 : Text[30];CurrencyCode_P@8053007 : Code[10];Iban_P@8053008 : Code[50];Swift_P@8053009 : Code[20];AgencyCode_P@8053010 : Text[30];RibKey_P@8053011 : Integer) Result_L : Code[20];
    VAR
      recBankAcc_L@8053006 : Record 8053474;
      Modify_L@8053002 : Boolean;
      Code_L@8053001 : Code[20];
      i_L@8053000 : Integer;
    BEGIN
      //----------------------------------------------------------------------------------------------------------------------------------
      // Recherche le code du compte bancaire
      //----------------------------------------------------------------------------------------------------------------------------------
      //  Entr'es
      //    - Contact_P : Num'ro de contact
      //    - Code_P : code du compte bancaire
      //    - BankBranchNo_P : Banque
      //    - BankAccountNo_P : Num'ro de compte bancaire
      //    - CurrencyCode_P : Code devise
      //    - Iban_P : Code IBAN
      //    - Swift_P : Code swift
      //    - AgencyCode_P ; No d'agence
      //    - RibKey_P : cl' RIB
      //  Sorties
      //    - Nouveau code pour le compte bancaire
      //----------------------------------------------------------------------------------------------------------------------------------
      Modify_L := FALSE;
      Result_L := '';

      recBankAcc_L.RESET;
      recBankAcc_L.SETRANGE("Contact No.", Contact_P);
      IF recBankAcc_L.FINDSET(FALSE,FALSE) THEN BEGIN
        REPEAT
          IF (recBankAcc_L."Bank Branch No." = BankBranchNo_P) AND
             (recBankAcc_L."Bank Account No." = BankAccountNo_P) AND
             (recBankAcc_L."Currency Code" = CurrencyCode_P) AND
             (recBankAcc_L.IBAN = Iban_P) AND
             (recBankAcc_L."SWIFT Code" = Swift_P) AND
             (recBankAcc_L."Agency Code" = AgencyCode_P) AND
             (recBankAcc_L."RIB Key" = RibKey_P) THEN BEGIN
            Modify_L := TRUE;
            Result_L := recBankAcc_L.Code;
          END;
        UNTIL (recBankAcc_L.NEXT = 0) OR (Modify_L = TRUE);
      END;

      IF Modify_L = FALSE THEN BEGIN
        i_L := 0;
        Code_L := Code_P;
        IF recBankAcc_L.GET(Contact_P, Code_L) THEN BEGIN
          REPEAT
            i_L := i_L + 1;
            Code_L := COPYSTR(Code_P, 1, MAXSTRLEN(Code_L) - 2) + '_' + FORMAT(i_L);
          UNTIL NOT(recBankAcc_L.GET(Contact_P, Code_L));
        END;
        Result_L := Code_L;
      END;

      EXIT(Result_L);
    END;

    BEGIN
