    PROCEDURE CheckItems@1000000016(VAR RadarContract@1000000001 : Record 8087776;VAR TmpItem@1000000000 : TEMPORARY Record 27);
    VAR
      ItemNo@1000000002 : Code[20];
      CourtUnealNumber@1000000003 : Integer;
      ContractNumber@1000000004 : Integer;
      ItemNumber@1000000005 : Integer;
    BEGIN
      CourtUnealNumber := 0;
      ContractNumber := 0;
      ItemNumber := 0;
      IF (TmpItem.FINDFIRST()) THEN BEGIN
        REPEAT
          ItemNumber += 1;
          // Contr"le correspondance de l'article CourtUneal
          ItemNo := MoteurInterface_G.Getcorrespondance(GlobalTemplateCode_G, 'Item', 'No.',
                                                        TmpItem."No.", FALSE, TRUE);
          IF (ItemNo = '') THEN BEGIN
            ContractNumber += 1;
          END ELSE BEGIN
            CourtUnealNumber += 1;
          END;
        UNTIL (TmpItem.NEXT() = 0);
        IF ((CourtUnealNumber > 0) AND (ContractNumber > 0)) THEN BEGIN
          RadarContract."Document Status" := RadarContract."Document Status"::Invalid;
          RadarContract."Document Status Description" := STRSUBSTNO(tErrInvalidDocument,
                                                                    RadarContract."Sales/Purchase",
                                                                    RadarContract."Blanket Order No.");

          IF (CourtUnealNumber = ItemNumber) THEN BEGIN
            RadarContract."Document Status" := RadarContract."Document Status"::CourtUneal;
            RadarContract."Document Status Description" := '';
          END;
          IF (ContractNumber = ItemNumber) THEN BEGIN
            RadarContract."Document Status" := RadarContract."Document Status"::Contract;
            RadarContract."Document Status Description" := '';
          END;

        END;
      END ELSE BEGIN
        RadarContract."Document Status" := RadarContract."Document Status"::Error;
        RadarContract."Document Status Description" := STRSUBSTNO(tErrNoItemInTheDocument,
                                                                  RadarContract."Sales/Purchase",

        END;

      END;
