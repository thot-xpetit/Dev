    PROCEDURE CalcContactBalanceLCY@8053021(CompanyCode_P@1180250002 : Code[10];LinkToTable_P@1180250003 : ' ,Customer,Vendor,Bank Account';ContactNo_P@1180250004 : Code[20]) BalanceLCY : Decimal;
    BEGIN
      //-------------------------------------------------------------------------------------------------------------------
      // Renvoie le solde DS du contact dans le groupe ou le solde DS du contact dans une soci't' du groupe
      //-------------------------------------------------------------------------------------------------------------------
      // Entr'es:
      //   - CompanyCode_P : Code soci't'
      //   - LinkToTable_P :  Client ou  Fournisseur
      //   - ContactNo_P : N› contact MS
      // Sorties:
      //   - BalanceLCY : Solde DS
      //-------------------------------------------------------------------------------------------------------------------
      CASE LinkToTable_P OF
        LinkToTable_P::Customer : // Solde DS du contact client dans une soci't' du groupe
        BEGIN
          MCDetCustLdgEntry_G.RESET;
          MCDetCustLdgEntry_G.SETCURRENTKEY("Company Code","Customer No.");
          MCDetCustLdgEntry_G.SETRANGE("Company Code",CompanyCode_P);
          MCDetCustLdgEntry_G.SETRANGE("Customer No.",ContactNo_P);
          MCDetCustLdgEntry_G.CALCSUMS("Amount (LCY)");
          EXIT(MCDetCustLdgEntry_G."Amount (LCY)");
        END;
        LinkToTable_P::Vendor : // Solde DS du contact fournisseur dans une soci't' du groupe
        BEGIN
          MCDetVendLdgEntry_G.RESET;
          MCDetVendLdgEntry_G.SETCURRENTKEY("Company Code","Vendor No.");
          MCDetVendLdgEntry_G.SETRANGE("Company Code",CompanyCode_P);
          MCDetVendLdgEntry_G.SETRANGE("Vendor No.",ContactNo_P);
          MCDetVendLdgEntry_G.CALCSUMS("Amount (LCY)");
          EXIT(MCDetVendLdgEntry_G."Amount (LCY)");
        END;
        LinkToTable_P::" " : // Solde DS du contact dans le groupe
        BEGIN
          MCDetCustLdgEntry_G.RESET;
          MCDetCustLdgEntry_G.SETCURRENTKEY("Company Code","Customer No.");
          MCDetCustLdgEntry_G.SETRANGE("Customer No.",ContactNo_P);
          MCDetCustLdgEntry_G.CALCSUMS("Amount (LCY)");

          MCDetVendLdgEntry_G.RESET;
          MCDetVendLdgEntry_G.SETCURRENTKEY("Company Code","Vendor No.");
          MCDetVendLdgEntry_G.SETRANGE("Vendor No.",ContactNo_P);
          MCDetVendLdgEntry_G.CALCSUMS("Amount (LCY)");
          EXIT(MCDetVendLdgEntry_G."Amount (LCY)" + MCDetCustLdgEntry_G."Amount (LCY)");
        END;
        ELSE
          EXIT(0);
      END;
    END;
