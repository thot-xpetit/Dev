                  MbrJnlLine_P."Date Formula"));

              CustEntry_L.SETRANGE(CustEntry_L.Open,TRUE);

              VendEntry_L.SETRANGE("Posting Date");
              VendEntry_L.SETRANGE(VendEntry_L.Open,TRUE);

                ERROR(STRSUBSTNO(Text8053004_L,MbrJnlLine_P."Account No."));

    PROCEDURE PostMbrJnlLine@8053001(MbrJnlLine_P@8053000 : Record 8053356);
    VAR
      MbrJnlLine_L@8053001 : Record 8053356;
      TypePartRec_G@8053002 : Record 8053351;
      MemberNo_L@8053003 : Code[20];
      TempMbrJnlLine_L@8053004 : Record 8053356;
      LastLineNo_L@8053005 : Integer;
      MbrJnlTemplate_L@8053006 : Record 8053354;
      MvtPostSetup_L@8053007 : Record 8053365;
    BEGIN
      MbrSetup_G.GET;
      IF MbrSetup_G."Mbr Mvt Source Code" <> '' THEN
        SrcCode_G := MbrSetup_G."Mbr Mvt Source Code"
      ELSE BEGIN
         MbrJnlTemplate_L.GET(MbrSetup_G."Member Journal Template");
         SrcCode_G := MbrJnlTemplate_L."Source Code";
       END;
      CLEAR(MemberNo_L);
      MemberNo_L := MbrJnlLine_P."Account No.";
      TypePartRec_G.RESET;
      TypePartRec_G.FINDFIRST;
      TypePartRec_G.SETFILTER(TypePartRec_G."Member Filter",MemberNo_L);
      // --> 20171003-WI00-3299-ADV01.01.00-JCD
      TypePartRec_G.SETFILTER("Date Filter",'%1..%2',0D,MbrJnlLine_P."Closing Date");
      // <-- 20171003-WI00-3299-ADV01.01.00-JCD
      IF TypePartRec_G.FINDFIRST THEN
        REPEAT
          TypePartRec_G.CALCFIELDS("Part Type Count","Part Type Amount");
          IF TypePartRec_G."Part Type Amount" <> 0 THEN BEGIN
            MbrJnlLine_L.INIT;
            MbrJnlLine_L."Capital Journal Template" := MbrSetup_G."Member Journal Template";
            MbrJnlLine_L."Capital Batch Name" := MbrSetup_G."Member Journal Batch";
            MbrJnlLine_L."Line No." := LastLineNo_G+10000;
            LastLineNo_G := MbrJnlLine_L."Line No.";
            MbrJnlLine_L."Posting Date" := MbrJnlLine_P."Posting Date";
            MbrJnlLine_L."Account Type" := MbrMvtType_G."Account Type";
            MbrJnlLine_L."Account No." := MbrMvtType_G."Account No.";
            MbrJnlLine_L.Direction := MbrMvtType_G.Direction;
            MbrJnlLine_L.Description := MbrJnlLine_P.Description;
            IF MbrJnlLine_P."G/L Posting No. Series" <> '' THEN
              MbrJnlLine_L."Document No." := NoSeriesMgt_G.GetNextNo(MbrJnlLine_P."G/L Posting No. Series",
                MbrJnlLine_L."Posting Date",TRUE)
            ELSE
              MbrJnlLine_L."Document No." := MbrJnlLine_P."Document No.";
            MbrJnlTemplate_L.GET(MbrSetup_G."Member Journal Template");
            MbrJnlLine_L."Source Code" := SrcCode_G;

            MbrJnlLine_L.Quantity := 1;
            MbrJnlLine_L.VALIDATE("Movement Type",MbrJnlLine_P."Movement Type");
            IF MvtPostSetup_L.GET(MbrJnlLine_P."Movement Type",TypePartRec_G."Part Code") THEN BEGIN
              MbrJnlLine_L."Bal. Account Type" := MbrJnlLine_L."Bal. Account Type"::G'n'ral;
              MbrJnlLine_L."Bal. Account No." := MvtPostSetup_L."Bal. Account No.";
            END;
            MbrJnlLine_L.Amount := TypePartRec_G."Part Type Amount";
            // --> 20141209-FINA09-CMC02.02.00-2567-DRN01
            MbrJnlLine_L."Posting Description" := STRSUBSTNO(Text8053000,MbrJnlLine_P."Account No.",MbrJnlLine_P."Movement Type");
            // <-- 20141209-FINA09-CMC02.02.00-2567-DRN01
            CapJnlPostLine_G.SetSourceCode(SrcCode_G);
            CapJnlPostLine_G.RunWithoutCheck(MbrJnlLine_L);
            CapJnlPostLine_G.GetGLReg(GlRec_G);
           END;
        UNTIL TypePartRec_G.NEXT=0;
    END;

    PROCEDURE ChangeMemberStatus@8053000();
    BEGIN
      Contact_G.GET(MbrJnlLine_G."Account No.");
      CASE MbrJnlLine_G."Action Message" OF
        MbrJnlLine_G."Action Message" :: "Active To Actif Without activity" :
          BEGIN
            Contact_G."Activity State" := Contact_G."Activity State":: "Active Without Activity";
          END;
        MbrJnlLine_G."Action Message" :: "Active To Not Active" :
          BEGIN
            Contact_G."Activity State" := Contact_G."Activity State":: "Not Active";
          END;
        MbrJnlLine_G."Action Message" :: "Active Without Activity To Active" :
          BEGIN
            Contact_G."Activity State" := Contact_G."Activity State":: Active;
          END;
        MbrJnlLine_G."Action Message" :: "Active Without Activity To Not Active" :
          BEGIN
            Contact_G."Activity State" := Contact_G."Activity State":: "Not Active";
          END;
        MbrJnlLine_G."Action Message" :: "Not Active To Active" :
          BEGIN
            Contact_G."Activity State" := Contact_G."Activity State":: Active;
          END;
      END;
    END;

    PROCEDURE CapitalDateNotAllowed@1(PostingDate_P@1000 : Date) : Boolean;

          AllowPostingFrom_G := MbrSetup_G."Position Starting Date";
          AllowPostingTo_G := MbrSetup_G."Posting Ending Date";

            AllowPostingTo_G := 31129999D;
       END;
      EXIT((PostingDate_P < AllowPostingFrom_G) OR (PostingDate_P > AllowPostingTo_G));
    END;
