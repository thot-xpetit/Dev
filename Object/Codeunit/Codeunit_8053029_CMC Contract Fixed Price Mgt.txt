    PROCEDURE ReturnCommercialRegion@8053004(VAR VendorNo_P@8053000 : Code[20]) Result_L : Code[20];

      ContBusRel_L@8053001 : Record 5054;
      Contact_L@8053002 : Record 5050;

    END;

    PROCEDURE CreateCollectionContract@8053005(VAR CMCMarketContract_P@8053000 : Record 8053268);
    VAR
      CMCCollectionContract_L@8053001 : Record 8052983;
      CMCMarketOffer_L@8053002 : Record 8053267;
      CMCSetup_L@8053003 : Record 8052982;
      CMCCollectionContract2_L@8053004 : Record 8052983;
      Text00001@8053005 : TextConst 'ENU="There is already a collection contract for this market sold contract ";FRA=Il existe d'j. un contrat achat collecte pour ce contrat march' vendu.';
      RecMarketContract_L@8053006 : Record 8053268;
      RecCampaign_L@8053007 : Record 8052991;
      recVendContact_L@8053008 : Record 8053160;
      cduManContact_L@8053009 : Codeunit 8053377;
      cduStatusMgt_L@8053011 : Codeunit 8052985;
      RecCollContract_L@8053010 : Record 8052983;
    BEGIN
      WITH CMCMarketContract_P DO BEGIN
        IF "Collection  Contrat No." <> '' THEN
          ERROR(Text00001);

        CLEAR(RecMarketContract_L);
        RecMarketContract_L.GET("Fixed Contract No.");

        // --> 20150507-CER29-CMC02.02.00-1172-BSI01
        RecCollContract_L.GET("Coll. Template Contract No.");
        // <-- 20150507-CER29-CMC02.02.00-1172-BSI01
        RecCampaign_L.GET(RecCollContract_L."Campaign No.");

        TESTFIELD(Status,CMCMarketContract_P.Status::Release);
        CMCSetup_L.GET;

        // --> 20150326-CER29-CMC02.02.00-1172-BSI01
        {before
        CMCMarketOffer_L.GET("Market Offer No.");
        CMCCollectionContract2_L.GET(CMCMarketOffer_L."Coll. Template Contract No.");
        after}
        IF CMCMarketOffer_L.GET("Market Offer No.") THEN;
        CMCCollectionContract2_L.GET("Coll. Template Contract No.");
        // <-- 20150326-CER29-CMC02.02.00-1172-BSI01

        CMCCollectionContract_L.INIT;
        IF CMCCollectionContract_L.INSERT(TRUE) THEN BEGIN
          "Collection  Contrat No." := CMCCollectionContract_L."No.";
          MODIFY;
        END;
        CMCCollectionContract_L.Type                    := CMCCollectionContract_L.Type::Contract;
        CMCCollectionContract_L."Campaign No."          := RecCollContract_L."Campaign No."; //3
        CMCCollectionContract_L."Contract Type"         := CMCCollectionContract2_L."Contract Type";
        CMCCollectionContract_L.VALIDATE("Vendor No."   ,"Vendor No."); //2
        CMCCollectionContract_L.VALIDATE("Template No." ,CMCMarketOffer_L."Coll. Template Contract No."); //12
        CMCCollectionContract_L.VALIDATE("Vendor No."   ,"Vendor No."); //2
        // --> 201540210-CER29-CMC02.02.00-1172-FHS01
        recVendContact_L.RESET;
        recVendContact_L.SETRANGE("Contact Code", cduManContact_L.GetContNoFromCustVentNo("Vendor No.", 1));
        recVendContact_L.SETRANGE("Prod. Activity Code", CMCSetup_L.Commercial);
        IF recVendContact_L.FINDFIRST THEN BEGIN
          CMCCollectionContract_L.VALIDATE("Salesperson Code", recVendContact_L."Salesperson Code");
          CMCCollectionContract_L.VALIDATE("Purchaser Code", recVendContact_L."Salesperson Code");
        END;
        // <-- 201540210-CER29-CMC02.02.00-1172-FHS01
        CMCCollectionContract_L."Due Date"              := "Requested Payment Date"; //7
        CMCCollectionContract_L.Quantity                := "Number Of Lots" * CMCSetup_L."MATIF Weight per Lot"; //11
        CMCCollectionContract_L.Supplement              := Supplement; //10
        CMCCollectionContract_L."Transport Cost"        := "Transport Cost";   //5
        CMCCollectionContract_L."MATIF Species Code"    := "MATIF Species Code";
        CMCCollectionContract_L."MATIF Term"            := "MATIF Term";
        CMCCollectionContract_L."Financial Expense"     := "Financial Charge"; //9
        CMCCollectionContract_L."Sold Contract No."     := "No.";  //1
        CMCCollectionContract_L."Default Location Code" := "Delivery Location Code"; //13
        CMCCollectionContract_L."Ship Starting Date"    := RecMarketContract_L."Delivery date"; //8
        CMCCollectionContract_L."Ship Ending Date"      := RecCampaign_L."Ending Date";
        CMCCollectionContract_L.Quotation               := "Sold Cotation"; //4
        CMCCollectionContract_L."Creation Date"         := "Created Date"; //6
        CMCCollectionContract_L."External Contract No." := "Reference interface Sales"; //14
        CMCCollectionContract_L."Paper Reference Contract No." := "No. 2"; //15
        CMCCollectionContract_L."Calculated Unit Price" := "Sold Cotation" - "Base Offer" - "Transport Cost" +
          Supplement + "Purchase bonus Culture" - "Financial Charge";
        IF (("Net Price Negociated" = 0)  AND ("Contract Base Negociated" = 0)) THEN
          CMCCollectionContract_L."Unit Price" := CMCCollectionContract_L."Calculated Unit Price";
        IF ("Contract Base Negociated" <> 0) THEN
          CMCCollectionContract_L."Unit Price" := CMCCollectionContract_L."Calculated Unit Price" - "Base Effect";
        IF ("Net Price Negociated" <> 0) THEN BEGIN
          "Ne Price Effect" := CMCCollectionContract_L."Calculated Unit Price" - "Net Price Negociated";
          MODIFY;
          CMCCollectionContract_L."Unit Price" := "Net Price Negociated";
        END;
        // --> 201540210-CER29-CMC02.02.00-1172-FHS01
        CMCCollectionContract_L.VALIDATE("Unit Price", CMCCollectionContract_L."Unit Price");
        // <-- 201540210-CER29-CMC02.02.00-1172-FHS01
        CMCCollectionContract_L.Margin := "Base Offer" + "Base Effect" + "Ne Price Effect"; // . v'rifier
        CMCCollectionContract_L.MODIFY;
        // --> 201540210-CER29-CMC02.02.00-1172-FHS01
        //BEFORE
        //PAGE.RUN(PAGE::"CMC Contract",CMCCollectionContract_L);
        //AFTER
        cduStatusMgt_L.LockContract(CMCCollectionContract_L);
        // <-- 201540210-CER29-CMC02.02.00-1172-FHS01
      END;
    END;
