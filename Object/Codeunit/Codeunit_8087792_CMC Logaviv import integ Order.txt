    PROCEDURE ImportOrderNav@1000000005(Type_P@1000000000 : Text[30];TableLogOrdre_P@1000000013 : Record 8053426) Result_L : Text[256];
    VAR
      "<-- 3LI LOCALS -->"@1000000005 : Integer;
      TableOrdre_L@1000000004 : Record 8053426;
      TableContratNavOrd_L@1000000003 : Record 8053268;
      TableContratNavRef_L@1000000010 : Record 8053268;
      TableContratNavMaj_L@1000000006 : Record 8053268;
      TableContratNavCon_L@1000000017 : Record 8053268;
      TableFinancial_L@1000000011 : Record 8053280;
      Erreur_L@1000000001 : Boolean;
      OrdreNavExiste_L@1000000002 : Boolean;
      ATraiter_L@1000000012 : Boolean;
      TypeTraitement_L@1000000007 : 'Fixation,Vente,Annulation';
      TypeOrdre_L@1000000009 : 'Fixation,Vente,Annulation';
      RefNavAgri_L@1000000008 : Code[20];
      IdSiloExtNum_L@1000000014 : Integer;
      IdSilo_L@1000000015 : Code[10];
      QteAvant_L@1000000016 : Decimal;
      TableContrat_L@1000000018 : Record 8053423;
    BEGIN
      Result_L := '';

      // Traiter tous les ordres Logaviv en attente.
      TableOrdre_L.SETRANGE(HoroIntegration, 0DT);
      TableOrdre_L.SETRANGE(IdOfferType, TableLogOrdre_P.IdOfferType);
      TableOrdre_L.SETRANGE(IdOrdreInterne, TableLogOrdre_P.IdOrdreInterne);

      IF TableOrdre_L.FINDSET THEN BEGIN
        REPEAT

          //--> 06/12/2017
          //R'cup'rer le contrat
          CLEAR(TableContrat_L);
          TableContrat_L.SETRANGE(IdContractExt, TableOrdre_L.IdContractExt);
          IF NOT TableContrat_L.FINDFIRST THEN BEGIN
            ERROR(Text0021, FORMAT(TableOrdre_L.IdContractExt), TableOrdre_L.TABLECAPTION);
          END ELSE BEGIN

          END;

          Erreur_L := FALSE;
          ATraiter_L := FALSE;
          // Type : Fixation, Vente, Annulation
          CASE TableOrdre_L.IdOrderStatut OF
            // Ordre fix'
            1, 5: BEGIN
              TypeTraitement_L := TypeTraitement_L::Fixation;
              TypeOrdre_L := TypeTraitement_L;
            END;
            // Ordre vendu : Si non d'j. int'gr' d'abord faire la Fixation si Type traitement FIXE
            2: BEGIN
              TypeOrdre_L := TypeTraitement_L::Vente;
      //CDAL20140604        IF ((Type_P = 'FIXE') AND (TableOrdre_L.RefNavAgriFixe = '')) THEN BEGIN
      //CDAL20140604 D'but
              IF (Type_P = 'FIXE') THEN BEGIN
      //CDAL20140604 Fin
                TypeTraitement_L := TypeTraitement_L::Fixation;
              END ELSE BEGIN
                TypeTraitement_L := TypeTraitement_L::Vente;
              END;
            END;
            // Ordre annul', expir'
            3, 4: BEGIN
              TypeTraitement_L := TypeTraitement_L::Annulation;
              TypeOrdre_L := TypeTraitement_L::Annulation;
      //CDAL20140522        IF ((TableOrdre_L.RefNavAgriFixe <> '') OR (TableOrdre_L.RefNafAgriVente <> '')) THEN BEGIN
      //CDAL20140522 D'but
              IF ((TableOrdre_L.RefNafAgriVente <> '')) THEN BEGIN
      //CDAL20140522 Fin
                Erreur_L := TRUE;
                ERROR(Text0014, TableOrdre_L.FIELDCAPTION(IdOrderStatut),
                  FORMAT(TableOrdre_L.IdOrderStatut) + '-' + TableOrdre_L.OrdreStatutLibelle,
                  TableOrdre_L.FIELDCAPTION(TableOrdre_L.IdOrdreInterne), TableOrdre_L.IdOrdreInterne);
              END;
            END;
            ELSE BEGIN
              Erreur_L := TRUE;
              ERROR(Text0014, TableOrdre_L.FIELDCAPTION(IdOrderStatut),
                FORMAT(TableOrdre_L.IdOrderStatut) + '-' + TableOrdre_L.OrdreStatutLibelle,
                TableOrdre_L.FIELDCAPTION(TableOrdre_L.IdOrdreInterne), TableOrdre_L.IdOrdreInterne);
            END;
          END;

          IF NOT Erreur_L THEN BEGIN
            IF (((Type_P = 'FIXE') AND (TypeTraitement_L = TypeTraitement_L::Fixation)) OR
              ((Type_P = 'AUTRES') AND (TypeTraitement_L <> TypeTraitement_L::Fixation))) THEN BEGIN
              ATraiter_L := TRUE;
              // Ordre fixation ou vente d'j. int'gr' ?
              IF (((TypeTraitement_L = TypeTraitement_L::Fixation) AND (TableOrdre_L.RefNavAgriFixe = '')) OR
                ((TypeTraitement_L = TypeTraitement_L::Vente) AND (TableOrdre_L.RefNafAgriVente = '')) OR
                ((TypeTraitement_L = TypeTraitement_L::Annulation) AND (TableOrdre_L.RefNafAgriVente = '') AND
                (TableOrdre_L.RefNavAgriFixe = ''))) THEN BEGIN

                OrdreNavExiste_L := FALSE;
                //-------------------------------------------------------------------------------------------------
                // En cr'ation de contrat fix' & Vente : N› Logaviv Ordre fix' ou vendu ne doit pas exister.
                //-------------------------------------------------------------------------------------------------
                TableContratNavOrd_L.SETRANGE("Reference interface Sales", FORMAT(TableOrdre_L.IdOrdreInterne));
                CASE TypeTraitement_L OF
                  TypeTraitement_L::Fixation: BEGIN
                    TableContratNavOrd_L.SETRANGE(Type, TableContratNavOrd_L.Type::Fixed);
                  END;
                  TypeTraitement_L::Vente: BEGIN
                    TableContratNavOrd_L.SETRANGE(Type, TableContratNavOrd_L.Type::Sold);
                  END;
                  TypeOrdre_L::Annulation: BEGIN
                    TableContratNavOrd_L.SETRANGE(Type, TableContratNavOrd_L.Type::Fixed);
                  END;
                END;
                IF TableContratNavOrd_L.FINDFIRST THEN BEGIN
                  Erreur_L := TRUE;
                  ERROR(Text0009, TableOrdre_L.IdOrdreInterne, TableContratNavOrd_L.TABLECAPTION,
                    TableOrdre_L.FIELDCAPTION(TableOrdre_L.IdOrdreInterne), TableOrdre_L.IdOrdreInterne);
                END;
                TableContratNavOrd_L.SETRANGE("Reference interface Sales");
                TableContratNavOrd_L.SETRANGE(Type);
                //---------------------------------------------------------
                // Si Fixation ou vente : Le contrat Nav doit exister (pour r'f'rence)
                //---------------------------------------------------------
                IF TypeTraitement_L <> TypeTraitement_L::Annulation THEN BEGIN
                  TableContratNavRef_L.SETRANGE(Type, TableContratNavRef_L.Type::Open);
                  TableContratNavRef_L.SETRANGE("Reference interface contract", FORMAT(TableOrdre_L.IdContractInt));
                  IF NOT TableContratNavRef_L.FINDFIRST THEN BEGIN
                    Erreur_L := TRUE;
                    ERROR(Text0008, TableOrdre_L.IdContractInt, TableContratNavRef_L.TABLECAPTION,
                      TableOrdre_L.FIELDCAPTION(TableOrdre_L.IdOrdreInterne), TableOrdre_L.IdOrdreInterne);
                  END;
                  TableContratNavRef_L.SETRANGE("Reference interface contract");
                  TableContratNavRef_L.SETRANGE(Type);
                  //-------------------------------------------------------------------------
                  // Si Vente : Le contrat fix' Nav doit exister (pour r'f'rence)
                  //-------------------------------------------------------------------------
                  IF TypeTraitement_L = TypeTraitement_L::Vente THEN BEGIN
                    TableContratNavRef_L.SETRANGE(Type, TableContratNavRef_L.Type::Fixed);
                    TableContratNavRef_L.SETRANGE("Reference interface contract", FORMAT(TableOrdre_L.IdContractInt));
                    TableContratNavRef_L.SETRANGE("Reference interface Sales", FORMAT(TableOrdre_L.IdOrdreInterne));
                    IF NOT TableContratNavRef_L.FINDFIRST THEN BEGIN
                      Erreur_L := TRUE;
                      ERROR(Text0008, TableOrdre_L.IdOrdreInterne, TableContratNavRef_L.TABLECAPTION,
                        TableOrdre_L.FIELDCAPTION(TableOrdre_L.IdOrdreInterne), TableOrdre_L.IdOrdreInterne);
                    END;
                    TableContratNavRef_L.SETRANGE("Reference interface contract");
                    TableContratNavRef_L.SETRANGE(Type);
                    TableContratNavRef_L.SETRANGE("Reference interface Sales");
                  END;
                  IF TableContratNavRef_L.Status <> TableContratNavRef_L.Status::Release THEN BEGIN
                    ERROR(Text0015, TableContratNavRef_L.TABLECAPTION, TableContratNavRef_L."No.", TableContratNavRef_L.Status,
                      TableOrdre_L.FIELDCAPTION(TableOrdre_L.IdOrdreInterne), TableOrdre_L.IdOrdreInterne);
                    Erreur_L := TRUE;
                  END;
                END;
              END ELSE BEGIN
                // En Modification d'ordre : On doit retrouver l'ordre avec le m^me N›.
                OrdreNavExiste_L := TRUE;
                CASE TypeTraitement_L OF
                  TypeTraitement_L::Fixation, TypeTraitement_L::Annulation: BEGIN
                    RefNavAgri_L := TableOrdre_L.RefNavAgriFixe;
                  END;
                  TypeTraitement_L::Vente: BEGIN
                    RefNavAgri_L := TableOrdre_L.RefNafAgriVente;
                  END;
       //           TypeOrdre_L::Annulation: BEGIN
      //              RefNavAgri_L := TableOrdre_L.RefNavAgriFixe???;
         //         END;
                END;
                IF NOT TableContratNavMaj_L.GET(RefNavAgri_L) THEN BEGIN
                  Erreur_L := TRUE;
                  ERROR(Text0011, RefNavAgri_L, TableContratNavMaj_L.TABLECAPTION, TableOrdre_L.IdOrdreInterne);
                END ELSE BEGIN
                  IF (FORMAT(TableOrdre_L.IdOrdreInterne) <> TableContratNavMaj_L."Reference interface Sales") THEN BEGIN
                    Erreur_L := TRUE;
                    ERROR(Text0010, TableOrdre_L.IdOrdreInterne, TableContratNavMaj_L.FIELDCAPTION("Reference interface Sales"),
                      TableContratNavMaj_L."Reference interface Sales", TableOrdre_L.FIELDCAPTION(IdOrdreInterne),
                      TableOrdre_L.IdOrdreInterne);
                  END;
                END;
              END;
              IF NOT Erreur_L THEN BEGIN
                IF TableOrdre_L.Quotit = 0 THEN BEGIN
                  ERROR(Text0012, TableOrdre_L.IdContractInt);
                  Erreur_L := TRUE;
                END;
              END;
            END;

            //================================================
            // G'n'rer la fixation ou la vente.
            //================================================
            IF ((ATraiter_L) AND (NOT Erreur_L)) THEN BEGIN
              // Annulation sur fixation et vente non int'gr' ==> M.j horoInt'gration sans traitement.
              IF ((TypeOrdre_L = TypeOrdre_L::Annulation) AND (TableOrdre_L.RefNavAgriFixe = '') AND
                (TableOrdre_L.RefNafAgriVente = '')) THEN BEGIN
                  TableOrdre_L.HoroIntegration := CURRENTDATETIME;
                  IF NOT TableOrdre_L.MODIFY(TRUE) THEN BEGIN
                    ERROR(Text0001, TableOrdre_L.IdOrdreInterne, TableOrdre_L.TABLECAPTION);
                  EXIT('FALSE');
                END;
              END ELSE BEGIN
                IF ((TypeOrdre_L = TypeOrdre_L::Annulation) AND (TableOrdre_L.RefNavAgriFixe <> '') AND
                  (TableOrdre_L.RefNafAgriVente = '')) THEN BEGIN

                  IF NOT TableContratNavMaj_L.DELETE(TRUE) THEN BEGIN
                    ERROR(Text9001, TableContratNavMaj_L."No.", TableContratNavMaj_L.TABLECAPTION, TableOrdre_L.IdOrdreInterne);
                    CLEARLASTERROR;
                    EXIT('FALSE');
                  END;
                  TableOrdre_L.HoroIntegration := CURRENTDATETIME;
                  IF NOT TableOrdre_L.MODIFY(TRUE) THEN BEGIN
                    ERROR(Text0001, TableOrdre_L.IdOrdreInterne, TableOrdre_L.TABLECAPTION);
                    EXIT('FALSE');
                  END;
                END ELSE BEGIN
                  IF NOT OrdreNavExiste_L THEN BEGIN
                    CLEAR(TableContratNavMaj_L);
                    IF NOT TableContratNavMaj_L.INSERT(TRUE) THEN BEGIN
                      ERROR(Text0001, TableContratNavMaj_L."No.", TableContratNavMaj_L.TABLECAPTION);
                      EXIT('FALSE');
                    END;
                    TableContratNavMaj_L.VALIDATE("No. 2", TableOrdre_L.IdOrdreExterne);
                    TableContratNavMaj_L.VALIDATE("Reference interface contract", TableContratNavRef_L."Reference interface contract");
                    TableContratNavMaj_L.VALIDATE("Reference interface Sales", FORMAT(TableOrdre_L.IdOrdreInterne));
                    IF TypeTraitement_L = TypeTraitement_L::Fixation THEN BEGIN
                      TableContratNavMaj_L.VALIDATE("Created Date", DT2DATE(TableOrdre_L.HoroCreated));
                    END ELSE BEGIN
                      TableContratNavMaj_L.VALIDATE("Created Date", DT2DATE(TableOrdre_L.HoroModified));
                    END;
                    TableContratNavMaj_L.VALIDATE("Offer Type", TableContratNavMaj_L."Offer Type"::Interface);
                    TableContratNavMaj_L.VALIDATE(Status, TableContratNavMaj_L.Status::Open);
                    TableContratNavMaj_L.VALIDATE("Vendor No.", TableContratNavRef_L."Vendor No.");
                    TableContratNavMaj_L.VALIDATE("Campaign No.", TableContratNavRef_L."Campaign No.");
                    TableContratNavMaj_L.VALIDATE("Deadline Fixing", TableContratNavRef_L."Deadline Fixing");
                    TableContratNavMaj_L.VALIDATE("MATIF Species Code", TableContratNavRef_L."MATIF Species Code");
                    // Erreur sur Validate : En plus pas de code dans VALIDATE ?
                    //TableContratNavMaj_L.VALIDATE("MATIF Term", TableContratNavRef_L."MATIF Term");
                    TableContratNavMaj_L."MATIF Term" := TableContratNavRef_L."MATIF Term";
                    TableContratNavMaj_L.VALIDATE("Coll. Template Contract No.", TableContratNavRef_L."Coll. Template Contract No.");
                  END;
                  IF TypeTraitement_L = TypeTraitement_L::Fixation THEN BEGIN
                    // cduContractFixedPriceMgt_G.CreateFixedContract(TableContratNav_L);
                    TableContratNavMaj_L.VALIDATE(Type, TableContratNavMaj_L.Type::Fixed);
                    IF NOT OrdreNavExiste_L THEN BEGIN
                      TableContratNavMaj_L.VALIDATE("Open Contract No.", TableContratNavRef_L."No.");
                    END;
                  END;

                  IF TypeTraitement_L = TypeTraitement_L::Vente THEN BEGIN
                    TableContratNavMaj_L.VALIDATE(Type, TableContratNavMaj_L.Type::Sold);
                    IF NOT OrdreNavExiste_L THEN BEGIN
                      TableContratNavMaj_L.VALIDATE("Open Contract No.", TableContratNavRef_L."Open Contract No.");
                      TableContratNavMaj_L.VALIDATE("Fixed Contract No.", TableContratNavRef_L."No.");
                    END;
                  END;

                  //TableContratFixNav_L.VALIDATE("Market Offer No."    ,TableContratNav_L."Market Offer No.");
                  IF STRLEN(TableOrdre_L.IdSiloExt) < 3 THEN BEGIN
                    // Avant code num'ric sur 3 ==> sans les 0 non significatif.
                    IF EVALUATE(IdSiloExtNum_L, TableOrdre_L.IdSiloExt) THEN BEGIN
                      IdSilo_L := FORMAT(IdSiloExtNum_L, 0, '<Integer,3><Filler Character,0>');
                    END ELSE BEGIN
                      IdSilo_L := TableOrdre_L.IdSiloExt;
                    END;
                  END ELSE BEGIN
                    IdSilo_L := TableOrdre_L.IdSiloExt;
                  END;
                  TableContratNavMaj_L.VALIDATE("Delivery Location Code", IdSilo_L);

                  TableContratNavMaj_L.VALIDATE("Requested Payment Date", DT2DATE(TableOrdre_L.DatePaiement));
                  TableContratNavMaj_L.VALIDATE("Purchase bonus Culture", TableOrdre_L.PrimeStorage);

                  IF TableOrdre_L.IdOrdreType = 1 THEN BEGIN
                    TableContratNavMaj_L.VALIDATE("Fixation Type", TableContratNavMaj_L."Fixation Type"::"Objective Of Cotation");
                    TableContratNavMaj_L.VALIDATE("Objective of Cotation", TableOrdre_L.ObjectifPrice);
                  END ELSE BEGIN
                    TableContratNavMaj_L.VALIDATE("Fixation Type", TableContratNavMaj_L."Fixation Type"::"At All Prices");
                    CLEAR(TableContratNavMaj_L."Objective of Cotation");
                  END;
                  IF ((NOT OrdreNavExiste_L) OR (TableContratNavMaj_L."Number Of Lots" <> TableOrdre_L.Quantity / TableOrdre_L.Quotit))
                  THEN BEGIN
                  //-->20150923-Int17-CMC02.02.00-ADVITAM-CDL005
                  // BEFORE
                  {
                    TableContratNavMaj_L.VALIDATE("Number Of Lots", TableOrdre_L.Quantity / TableOrdre_L.Quotit);
                  }
                  // AFTER
                  // - Pb proc'dure 3LI fausse pour VALIDATE sur "Number Of Lots" (ne retire pas la quantit' d'j. consomm' par lui
                  //   m^me : xRec.
                  QteAvant_L := TableContratNavMaj_L."Number Of Lots";
                  CASE TableContratNavMaj_L.Type OF
                    TableContratNavMaj_L.Type::Fixed : BEGIN
                      CLEAR(TableContratNavCon_L);
                      TableContratNavCon_L.GET(TableContratNavMaj_L."Open Contract No.");
                      TableContratNavCon_L.CALCFIELDS("Lot On FIXED For OPEN");
                      IF (((TableOrdre_L.Quantity / TableOrdre_L.Quotit) - QteAvant_L) >
                        (TableContratNavCon_L."Number Of Lots" - TableContratNavCon_L."Lot On FIXED For OPEN")) THEN BEGIN
                        ERROR(TextQtMax, (TableOrdre_L.Quantity / TableOrdre_L.Quotit) - QteAvant_L -
                          TableContratNavCon_L."Lot On FIXED For OPEN");
                      END;
                    END;
                    TableContratNavMaj_L.Type::Sold : BEGIN
                      CLEAR(TableContratNavCon_L);
                      TableContratNavCon_L.GET(TableContratNavMaj_L."Fixed Contract No.");
                      TableContratNavCon_L.CALCFIELDS("Lot On SOLD For Fixe");
                      IF (((TableOrdre_L.Quantity / TableOrdre_L.Quotit) - QteAvant_L) >
                        (TableContratNavCon_L."Number Of Lots" - TableContratNavCon_L."Lot On SOLD For Fixe")) THEN BEGIN
                        ERROR(TextQtMax, (TableOrdre_L.Quantity / TableOrdre_L.Quotit) - QteAvant_L -
                          TableContratNavCon_L."Lot On SOLD For Fixe");
                      END;
                    END;
                    ELSE BEGIN
                      ERROR(TextTypMvt, TableContratNavMaj_L.Type::Sold, TableContratNavMaj_L."No.");
                    END;
                  END;
                  TableContratNavMaj_L."Number Of Lots" := TableOrdre_L.Quantity / TableOrdre_L.Quotit;
                  //<--20150923-Int17-CMC02.02.00-ADVITAM-CDL005
                  END;
                  TableContratNavMaj_L.VALIDATE("Base Offer", TableOrdre_L.BaseAmount);
                  TableContratNavMaj_L.VALIDATE("Requested Payment Date", DT2DATE(TableOrdre_L.DatePaiement));

                  IF TableOrdre_L.DeliveryPlannedDate <> 0DT THEN BEGIN
                    TableContratNavMaj_L.VALIDATE("Delivery date", DT2DATE(TableOrdre_L.DeliveryPlannedDate));
                  END ELSE BEGIN
                    TableContratNavMaj_L.VALIDATE("Delivery date", DT2DATE(TableOrdre_L.DeliveryDate));
                  END;
                  TableContratNavMaj_L.VALIDATE("Sold Cotation", TableOrdre_L.FixedPrice);
                  TableContratNavMaj_L.VALIDATE("Transport Cost", TableOrdre_L.MontantFraisTran);
                  TableContratNavMaj_L.VALIDATE(Supplement, TableOrdre_L.IncreaseAmount);
                  // Frais financier
                  TableFinancial_L.SETRANGE("Campaign No.", TableContratNavMaj_L."Campaign No.");
                  TableFinancial_L.SETRANGE("MATIF Term", TableContratNavMaj_L."MATIF Term");
                  TableFinancial_L.SETRANGE("MATIF Species Code", TableContratNavMaj_L."MATIF Species Code");
                  TableFinancial_L.SETFILTER("Application Date", '<= %1', WORKDATE);
                  IF TableFinancial_L.FINDLAST THEN BEGIN
                    TableContratNavMaj_L."Financial Charge" := TableFinancial_L.Expense;
                  END ELSE BEGIN
                    TableFinancial_L.SETRANGE("Application Date", 0D);
                    IF TableFinancial_L.FINDFIRST THEN BEGIN
                      TableContratNavMaj_L.VALIDATE("Financial Charge", TableFinancial_L.Expense);
                    END;
                  END;

                  TableContratNavMaj_L.VALIDATE(Status, TableContratNavMaj_L.Status::Release);

                  //--> 06/12/2017 - YPR: Logaviv lot2
                  TableContratNavMaj_L."Prime Amount" := TableContrat_L.PrimeBaseMoyenne;
                  TableContratNavMaj_L.Margin := TableContrat_L.MarginAmount;
                  TableContratNavMaj_L."Contract Base Negociated" := TableContrat_L.BaseNegociee;

                  IF TableContrat_L.BaseNegociee <> 0 THEN
                  BEGIN
                    TableContratNavMaj_L."Base Offer" := TableContrat_L.PrimeBaseMoyenne + TableContrat_L.MarginAmount;

                  END ELSE BEGIN
                    TableContratNavMaj_L."Base Offer" := TableContrat_L.BaseAmount;
                  END;

                  //--< 06/12/2017 - YPR: Logaviv lot2

                  IF NOT TableContratNavMaj_L.MODIFY() THEN BEGIN
                    ERROR(Text0001, TableContratNavMaj_L."No.", TableContratNavMaj_L.TABLECAPTION, GETLASTERRORTEXT);
                    CLEARLASTERROR;
                    EXIT('FALSE');
                  END ELSE BEGIN
