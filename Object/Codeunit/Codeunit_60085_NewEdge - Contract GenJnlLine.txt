    PROCEDURE CheckFileName@100000000(FileName_P@100000000 : Text[1024]);
    VAR
      NewEdgeSAS_L@100000001 : Record 60111;
      Pos_L@100000002 : Integer;
      Found_L@100000003 : Boolean;
      FileName_L@100000004 : Text[1024];
    BEGIN
      NewEdgeSAS_L.RESET;
      NewEdgeSAS_L.SETRANGE(FileName, FileName_P);
      IF NewEdgeSAS_L.FINDFIRST THEN BEGIN
        FileName_P := DELCHR(FileName_P, '<>');
        IF (FileName_P <> '') THEN BEGIN
          Pos_L := STRLEN(FileName_P);
          REPEAT
            Found_L := (COPYSTR(FileName_P, Pos_L, 1) = '\');
            IF NOT Found_L THEN
              Pos_L := Pos_L - 1;
          UNTIL (Pos_L = 0) OR Found_L;
          IF Found_L THEN BEGIN
            FileName_L := COPYSTR(FileName_P, Pos_L+1);
          END ELSE BEGIN
            FileName_L := FileName_P;
          END;
        END;
        ERROR(STRSUBSTNO(Text60010, FileName_L));
      END;
    END;

    VAR

      LineToCut_L@1000000005 : Text[1000];

    BEGIN
      //----------------------------------------------------------------------------------------------------------------------------------
      // Lit et lance l'action de d'coupage du fichier
      //----------------------------------------------------------------------------------------------------------------------------------
      //  Entr'es
      //    - Filename_P : adresse et nom du fichier
      //    - CompanyNo_P : N› de soci't' . qui appartient le fichier
      //    - DeleteFile_P : Indique si le fichier doit ^tre supprim'
      //----------------------------------------------------------------------------------------------------------------------------------

      GetSetup();

      //Initialisation des variables locales
      Result_L := Text8053600;
      CharLF_L := 10;
      CharRC_L := 13;
      EndLine_L := FORMAT(CharLF_L) + FORMAT(CharRC_L);

      //V'rification si le fichier exist
      IF NOT(EXISTS(Filename_P)) THEN
        Result_L := Text8053603;

      //----------------------------------------------------------------------------------------------------------------------------------
      //Ouverture du fichier pour lire du texte en lecture seule
      //----------------------------------------------------------------------------------------------------------------------------------
        IF Result_L = Text8053600 THEN BEGIN
          File_L.WRITEMODE(FALSE);
          File_L.TEXTMODE(TRUE);
          IF NOT(File_L.OPEN(Filename_P)) THEN
            Result_L := Text8053601;
        END;

      //----------------------------------------------------------------------------------------------------------------------------------
      //D'coupage du fichier pour r'cup'rer chaque ligne
      //----------------------------------------------------------------------------------------------------------------------------------
        IF Result_L = Text8053600 THEN BEGIN
          LineNo_L := 1;

          File_L.CREATEINSTREAM(FileStream_L);
            REPEAT
              IF FileStream_L.READTEXT(Line_L) > 0 THEN BEGIN
                //On ne tient pas compte de la premiSre ligne comportant le nom des colonnes
                IF LineNo_L > 1 THEN BEGIN
                  //conversion de ANSI vers ASCII
                  Line_L := cduFormatFieldsMgt_L.Ansi2Ascii(Line_L);
                  Line_L := DELCHR(Line_L, '=', EndLine_L);
                  LineToCut_L := Line_L;

                  ReadLine(Filename_P, LineToCut_L);
                END;

                LineNo_L += 1;
              END;
            UNTIL FileStream_L.EOS = TRUE;

            //Fermeture du fichier aprSs traitement
            File_L.CLOSE;
        END;

      EXIT(Result_L);
    END;

    PROCEDURE ReadNextItem@8053600(VAR Var_Line_P@8053600 : Text[1000]) Result_L : Text[50];
    VAR
      Position_L@8053601 : Integer;
    BEGIN
      //----------------------------------------------------------------------------------------------------------------------------------
      // R'cupSre le prochain 'l'ment de la liste 'tant s'par' par des ; et l'enlSve de la liste
      //----------------------------------------------------------------------------------------------------------------------------------
      //  Entr'es
      //    - Var_Line_P : Ligne . traiter
      //  Sorties
      //    - Result_L : la prochain 'l'ment de la liste
      //----------------------------------------------------------------------------------------------------------------------------------

      Position_L := STRPOS(Var_Line_P, ',');
      IF Position_L > 0 THEN BEGIN
        Result_L := COPYSTR(Var_Line_P, 1, Position_L - 1);
        Var_Line_P := COPYSTR(Var_Line_P, Position_L + 1, STRLEN(Var_Line_P));
      END ELSE BEGIN
      //Dernier 'l'ment de la liste
        Result_L := Var_Line_P;
        Var_Line_P := '';
      END;
      // --> 20191122-WI00-ADV01.01.00-3444-MRU01
      //Suppression des "
      Result_L := DELCHR(Result_L, '=', '"');
      // <-- 20191122-WI00-ADV01.01.00-3444-MRU01
      EXIT(Result_L);
    END;
