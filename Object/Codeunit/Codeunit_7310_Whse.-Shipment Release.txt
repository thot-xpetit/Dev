    PROCEDURE Release@1(VAR WhseShptHeader@1000 : Record 7320);
    VAR
      Location@1004 : Record 14;
      WhsePickRqst@1003 : Record 7325;
      WhseShptLine@1001 : Record 7321;
      "<--3LI LOCALS-->"@8053000 : Integer;
      SalesRemQtyMgt_L@8053001 : Codeunit 8053008;
    BEGIN
      WITH WhseShptHeader DO BEGIN
        IF Status = Status::Released THEN
          EXIT;

        WhseShptLine.SETRANGE("No.","No.");
        WhseShptLine.SETFILTER(Quantity,'<>0');
        IF NOT WhseShptLine.FIND('-') THEN
          ERROR(Text000,TABLECAPTION,"No.");

        IF "Location Code" <> '' THEN
          Location.GET("Location Code");

        // --> 20120322-Log04-CMC02.01.00-BRA01
        CheckPalletAssignment(WhseShptHeader);
        // <-- 20120322-Log04-CMC02.01.00-BRA01

        // --> 20111209-APP01-CMC02.01.00-BRA01
        CLEAR(SalesRemQtyMgt_L);
        SalesRemQtyMgt_L.ExplodeWhseShipment(WhseShptHeader);
        // <-- 20111209-APP01-CMC02.01.00-BRA01

        REPEAT
          WhseShptLine.TESTFIELD("Item No.");
          WhseShptLine.TESTFIELD("Unit of Measure Code");
          IF Location."Directed Put-away and Pick" THEN
            WhseShptLine.TESTFIELD("Zone Code");
          IF Location."Bin Mandatory" THEN
            WhseShptLine.TESTFIELD("Bin Code");
        UNTIL WhseShptLine.NEXT = 0;

        Status := Status::Released;
        MODIFY;

        // --> 20111207-APP01-CMC02.01.00-BRA01
        WhseShptLine.RESET;
        WhseShptLine.SETRANGE("No.","No.");
        IF WhseShptLine.FINDSET THEN
          REPEAT
            WhseShptLine."Document Status" := WhseShptLine."Document Status"::Released;
            WhseShptLine.MODIFY;
          UNTIL WhseShptLine.NEXT=0;
        // <-- 20111207-APP01-CMC02.01.00-BRA01

        CreateWhsePickRqst(WhseShptHeader);

        WhsePickRqst.SETRANGE("Document Type",WhsePickRqst."Document Type"::Shipment);
        WhsePickRqst.SETRANGE("Document No.","No.");
        WhsePickRqst.SETRANGE(Status,Status::Open);
        WhsePickRqst.DELETEALL(TRUE);

        COMMIT;

      END;
    END;

    PROCEDURE Reopen@2(WhseShptHeader@1000 : Record 7320);
    VAR
      WhsePickRqst@1001 : Record 7325;
      PickWkshLine@1002 : Record 7326;
      WhseActivLine@1003 : Record 5767;
      "<--3LI LOCALS-->"@8053000 : Integer;
      WhseShptLine_L@8053001 : Record 7321;
    BEGIN
      WITH WhseShptHeader DO BEGIN
        IF Status = Status::Open THEN
          EXIT;

        PickWkshLine.SETCURRENTKEY("Whse. Document Type","Whse. Document No.");
        PickWkshLine.SETRANGE("Whse. Document Type",PickWkshLine."Whse. Document Type"::Shipment);
        PickWkshLine.SETRANGE("Whse. Document No.","No.");
        IF PickWkshLine.FIND('-') THEN
          ERROR(Text001);

        WhseActivLine.SETCURRENTKEY("Whse. Document No.","Whse. Document Type","Activity Type");
        WhseActivLine.SETRANGE("Whse. Document No.","No.");
        WhseActivLine.SETRANGE("Whse. Document Type",WhseActivLine."Whse. Document Type"::Shipment);
        WhseActivLine.SETRANGE("Activity Type",WhseActivLine."Activity Type"::Pick);
        IF WhseActivLine.FIND('-') THEN
          ERROR(Text002);

        WhsePickRqst.SETRANGE("Document Type",WhsePickRqst."Document Type"::Shipment);
        WhsePickRqst.SETRANGE("Document No.","No.");
        WhsePickRqst.SETRANGE(Status,Status::Released);
        IF WhsePickRqst.FIND('-') THEN
          REPEAT
            WhsePickRqst.Status := WhsePickRqst.Status::Open;
            WhsePickRqst.MODIFY;
          UNTIL WhsePickRqst.NEXT = 0;

        Status := Status::Open;
        MODIFY;

        // --> 20111207-APP01-CMC02.01.00-BRA01
        WhseShptLine_L.RESET;
        WhseShptLine_L.SETRANGE("No.","No.");
        IF WhseShptLine_L.FINDSET THEN
          REPEAT
            WhseShptLine_L."Document Status" := WhseShptLine_L."Document Status"::Open;
            WhseShptLine_L.MODIFY;
          UNTIL WhseShptLine_L.NEXT=0;
        // <-- 20111207-APP01-CMC02.01.00-BRA01

      END;
    END;

    LOCAL PROCEDURE CreateWhsePickRqst@3(VAR WhseShptHeader@1000 : Record 7320);
    VAR
      WhsePickRqst@1001 : Record 7325;
      Location@1002 : Record 14;
    BEGIN
      WITH WhseShptHeader DO BEGIN
        IF Location.RequirePicking("Location Code") THEN BEGIN
          WhsePickRqst."Document Type" := WhsePickRqst."Document Type"::Shipment;
          WhsePickRqst."Production Subtype" := WhsePickRqst."Production Subtype"::" ";
          WhsePickRqst."Document No." := "No.";
          WhsePickRqst.Status := Status;
          WhsePickRqst."Location Code" := "Location Code";
          WhsePickRqst."Zone Code" := "Zone Code";
          WhsePickRqst."Bin Code" := "Bin Code";
          CALCFIELDS("Completely Picked");
          WhsePickRqst."Completely Picked" := "Completely Picked";
          IF NOT WhsePickRqst.INSERT THEN
            WhsePickRqst.MODIFY;
        END;
      END;
    END;

    PROCEDURE CheckPalletAssignment@8053001(WhseShptHeader_P@8053000 : Record 7320);
    VAR
      WhseShipmentPallet_L@8053001 : Record 8053100;
      WhseShipmentPalletContent_L@8053002 : Record 8053101;
      WhseShpmtLine_L@8053003 : Record 7321;
      Text8053000_L@8053004 : TextConst 'ENU=Pallet allocated quantity is different than quantity to ship on line %1.;FRA=La quantit' affect'e sur palette est diff'rente de la quantit' . exp'dier sur ligne %1.';
      Text8053001_L@8053005 : TextConst 'ENU=The allocated weight on pallet %1 is upper than max weight.;FRA=Le poids affect' sur la palette %1 d'passe le poids maxi.';
    BEGIN
      // --> 20120322-Log04-CMC02.01.00-BRA01
      WhseShipmentPallet_L.RESET;
      WhseShipmentPallet_L.SETFILTER(WhseShipmentPallet_L."Warehouse Shipment Header No.",WhseShptHeader_P."No.");
      IF NOT WhseShipmentPallet_L.FINDFIRST THEN
        EXIT;

      WhseShpmtLine_L.SETRANGE("No.",WhseShptHeader_P."No.");
      WhseShpmtLine_L.SETFILTER(Quantity,'<>0');
      IF WhseShpmtLine_L.FINDSET THEN
        REPEAT
          WhseShipmentPalletContent_L.RESET;
          WhseShipmentPalletContent_L.SETFILTER(WhseShipmentPalletContent_L."Warehouse Shipment Header No.",WhseShpmtLine_L."No.");
          WhseShipmentPalletContent_L.SETRANGE(WhseShipmentPalletContent_L."Warehouse Shipment Line No.",WhseShpmtLine_L."Line No.");
          WhseShipmentPalletContent_L.CALCSUMS(WhseShipmentPalletContent_L."Allocated Qty");
          IF WhseShipmentPalletContent_L."Allocated Qty"<>WhseShpmtLine_L."Qty. to Ship" THEN
            ERROR(Text8053000_L,WhseShpmtLine_L."Line No.");
        UNTIL WhseShpmtLine_L.NEXT=0;

      IF WhseShipmentPallet_L.FINDSET THEN
        REPEAT
          WhseShipmentPalletContent_L.RESET;
          WhseShipmentPalletContent_L.SETFILTER(WhseShipmentPalletContent_L."Warehouse Shipment Header No.",WhseShpmtLine_L."No.");
          WhseShipmentPalletContent_L.SETRANGE(WhseShipmentPalletContent_L."Pallet No.",WhseShipmentPallet_L."Pallet No.");
          WhseShipmentPalletContent_L.CALCSUMS(WhseShipmentPalletContent_L."Content Weight");
          IF WhseShipmentPallet_L."Max. Pallet Weight"<WhseShipmentPalletContent_L."Content Weight" THEN
            ERROR(Text8053001_L,WhseShipmentPallet_L."Pallet No.");
        UNTIL WhseShipmentPallet_L.NEXT=0;
      // <-- 20120322-Log04-CMC02.01.00-BRA01
    END;
