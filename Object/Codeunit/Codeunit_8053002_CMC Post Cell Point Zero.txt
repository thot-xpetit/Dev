    PROCEDURE Code@8053002();
    BEGIN
      CheckData;
      CreatePostedIntervention;
      WITH G_recLocationCell DO BEGIN
        "Last Lot No." := "Active Lot No.";
        IF "Last Lot No." <> '' THEN
          "Active Lot No." := INCSTR("Last Lot No.")
        ELSE
          "Active Lot No." := G_recLocation."Cereal Lot Root" + '-' + "Cell Code" + '-0000001';
        "Active Species Code" := G_SpeciesCode;
        "Active Item No." := G_ItemNo;
        MODIFY(TRUE);
      END;
    END;

    PROCEDURE CreatePostedIntervention@8053000();
    VAR
      L_recPostedIntervention@8053000 : Record 8053036;
      L_recLastPostedIntervention@8053001 : Record 8053036;
      L_recItemLedgEntry@8053002 : Record 32;
    BEGIN
      WITH G_recLocationCell DO BEGIN
        L_recPostedIntervention.INIT;
        IF L_recLastPostedIntervention.FINDLAST THEN
          L_recPostedIntervention."Entry No." := L_recLastPostedIntervention."Entry No." + 1
        ELSE
          L_recPostedIntervention."Entry No." := 1;
        L_recPostedIntervention.VALIDATE("Location Code","Location Code");
        L_recPostedIntervention.VALIDATE("Cell No.","Cell Code");
        IF ValidationType_G=ValidationType_G::ZeroPoint THEN
        L_recPostedIntervention."Zero Point" := TRUE;
        L_recPostedIntervention.Comment := G_Comment;
        L_recPostedIntervention.VALIDATE("Intervention Date",G_Date);
        L_recPostedIntervention.VALIDATE("Validation Date",CURRENTDATETIME);
        L_recPostedIntervention.VALIDATE("Species Code","Active Species Code");
        L_recPostedIntervention.VALIDATE("Item No.","Active Item No.");
        IF "Active Item No." <> '' THEN
          L_recItemLedgEntry.SETRANGE("Item No.","Active Item No.");
        L_recItemLedgEntry.SETRANGE("Location Code","Location Code");
        L_recItemLedgEntry.SETRANGE("Cell Code","Cell Code");
        IF L_recItemLedgEntry.FINDLAST THEN
          L_recPostedIntervention."Last Item Ledger Entry No." := L_recItemLedgEntry."Entry No.";
        L_recPostedIntervention."Source Document No." := SourceDocNo_G;
        // --> 20150831-ADV14-CMC02.02.00-1789-SDE01
        IF (ValidationType_G=ValidationType_G::Mutation) AND (L_recPostedIntervention.Comment='') THEN
          L_recPostedIntervention.Comment:=COPYSTR(STRSUBSTNO('Mutation %1 %2',
            G_recLocationCell."Active Item No.",G_ItemNo),1,100);
        // <-- 20150831-ADV14-CMC02.02.00-1789-SDE01
        L_recPostedIntervention.INSERT(TRUE);
       END;
    END;

        RecItemJournalLine_G.VALIDATE("Campaign No.",GetCampainNo('',WORKDATE,WORKDATE));
        RecItem_L.GET(RecCellsLocation_P."Active Item No.");
        RecSpecies_L.GET(RecItem_L."Species Code");
        // --> 20150505-Wi00-CMC02.02.00-2955-FHS01
        //BEFORE
        //IF RecSpecies_L."Item Entry Typed" THEN
        //AFTER
        IF RecSpecies_L."Item Entry Typed" THEN BEGIN

            RecItemJournalLine_G."Movement Type":=RecItemJournalLine_G."Movement Type"::T10;

            RecItemJournalLine_G."Movement Type":=RecItemJournalLine_G."Movement Type"::T11;
        // <-- 20150505-Wi00-CMC02.02.00-2955-FHS01
        END;
        // --> 20150505-Wi00-CMC02.02.00-2955-FHS01
        RecItemJournalLine_G.VALIDATE("Item No.",RecCellsLocation_P."Active Item No.");
        RecItemJournalLine_G.VALIDATE("Location Code",RecCellsLocation_P."Location Code");
        RecItemJournalLine_G.VALIDATE("Cell Code",RecCellsLocation_P."Cell Code");
        RecItemJournalLine_G."Document No.":=COPYSTR(STRSUBSTNO('PZ-%1-%2',
          RecCellsLocation_P."Location Code",RecCellsLocation_P."Cell Code"), 1, 20);
        RecItemJournalLine_G.Description:=STRSUBSTNO('Point z'ro %1 %2',
          RecCellsLocation_P."Location Code",RecCellsLocation_P."Cell Code");

        // --> 20150204-WI00-CMC02.02.00-2739-DRN01
        CduDimensionsMgt_L.DimFromItemJnl(TempJnlLineDim_G,RecItemJournalLine_G,TRUE);
        // <-- 20150204-WI00-CMC02.02.00-2739-DRN01

      L_recCampaign@8053003 : Record 8052991;
    BEGIN
      Res_CampainNo := '';

      L_recCampaign.RESET;
      L_recCampaign.SETCURRENTKEY("Type Code","Starting Date","Ending Date");
      L_recCampaign.SETRANGE("Type Code", P_CampainType);
      L_recCampaign.SETFILTER("Starting Date", '<=%1', P_StartingDate);
      L_recCampaign.SETFILTER("Ending Date", '>=%1|%2', P_EndingDate, 0D);
      IF L_recCampaign.FINDFIRST THEN
        Res_CampainNo := L_recCampaign."No.";

      EXIT(Res_CampainNo);
    END;
