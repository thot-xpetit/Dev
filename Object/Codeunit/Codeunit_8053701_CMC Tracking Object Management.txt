    VAR
      Text8053000@8053000 : TextConst 'FRA=Les champs %1 et %2 doivent ^tres du m^me type.';

    PROCEDURE InitObjectTracking@8053002();
    VAR
      RecObject_L@8053000 : Record 2000000001;
      RecCMCObjectTracking_L@8053001 : Record 8053702;
      RecObjectTracking_L@8053002 : Record 2000000079;
      RecCMCObjectMgtSetup_L@8053003 : Record 8053701;
    BEGIN
      RecObject_L.SETRANGE("Company Name",'');
      RecObject_L.SETFILTER(Type,'>%1',RecObject_L.Type::TableData);
      IF RecObject_L.FINDSET THEN
        REPEAT
          RecCMCObjectTracking_L."Object Type" := RecObject_L.Type;
          RecCMCObjectTracking_L."Object ID" := RecObject_L.ID;
          RecCMCObjectTracking_L."Object Name" := RecObject_L.Name;
          RecObject_L.CALCFIELDS(Caption);
          RecCMCObjectTracking_L."Object Caption" := RecObject_L.Caption;
          RecCMCObjectTracking_L."Object Modified" := RecObject_L.Modifi';
          RecCMCObjectTracking_L."Object Date" := RecObject_L.Date;
          RecCMCObjectTracking_L."Object Time" := RecObject_L.Heure;
          RecCMCObjectTracking_L."Object BLOB Size" := RecObject_L."BLOB Size";
          RecCMCObjectTracking_L.INSERT;
        UNTIL RecObject_L.NEXT = 0;
      RecObjectTracking_L.SETCURRENTKEY("Object Timestamp");
      IF RecObjectTracking_L.FINDLAST THEN BEGIN
        RecCMCObjectMgtSetup_L.GET;
        RecCMCObjectMgtSetup_L."Last Object Timestamp" := RecObjectTracking_L."Object Timestamp";
        RecCMCObjectMgtSetup_L.MODIFY;
      END;
      COMMIT;
    END;

    PROCEDURE VerifyObjectChange@8053003();
    VAR
      RecObject_L@8053003 : Record 2000000001;
      RecCMCObjectMgtSetup_L@8053000 : Record 8053701;
      RecObjectTracking_L@8053001 : Record 2000000079;
      RecCMCObjectTracking_L@8053002 : Record 8053702;
      cduSendObjectChange_L@1000000000 : Codeunit 8053702;
    BEGIN
      //Initialisation de la table CMC Object Tracking
      IF RecCMCObjectTracking_L.ISEMPTY THEN
        InitObjectTracking;

      //V'rifier les modifications d'objet enregitr'es dans la table Object Tracking (Compilation, Design objet)
      RecCMCObjectMgtSetup_L.GET;
      RecObjectTracking_L.SETCURRENTKEY("Object Timestamp");
      RecObjectTracking_L.SETFILTER("Object Timestamp",'>%1',RecCMCObjectMgtSetup_L."Last Object Timestamp");
      RecCMCObjectTracking_L.LOCKTABLE;
      IF RecObjectTracking_L.FINDSET THEN BEGIN
        REPEAT
          //Test si l'objet existe -> Cas du renommage
          IF RecObject_L.GET(RecObjectTracking_L."Object Type",'',RecObjectTracking_L."Object ID") THEN BEGIN
            IF RecCMCObjectTracking_L.GET(RecObjectTracking_L."Object Type",RecObjectTracking_L."Object ID") THEN BEGIN
              IF (RecObject_L.Modifi' <> RecCMCObjectTracking_L."Object Modified") OR
                (RecObject_L.Date <> RecCMCObjectTracking_L."Object Date") OR
                (RecObject_L.Heure <> RecCMCObjectTracking_L."Object Time") OR
                (RecObject_L."BLOB Size" <> RecCMCObjectTracking_L."Object BLOB Size")
              THEN BEGIN
                RecCMCObjectTracking_L."Object Name" := RecObject_L.Name;
                RecObject_L.CALCFIELDS(Caption);
                RecCMCObjectTracking_L."Object Caption" := RecObject_L.Caption;
                RecCMCObjectTracking_L."Object Modified" := RecObject_L.Modifi';
                RecCMCObjectTracking_L."Object Date" := RecObject_L.Date;
                RecCMCObjectTracking_L."Object Time" := RecObject_L.Heure;
                RecCMCObjectTracking_L."Object BLOB Size" := RecObject_L."BLOB Size";
                RecCMCObjectTracking_L."Notify Object Modified" := TRUE;
                RecCMCObjectTracking_L."Type Object Modified" := RecCMCObjectTracking_L."Type Object Modified"::Modify;
                RecCMCObjectTracking_L.MODIFY;
              END ELSE
                //Cas du renommage
                IF RecCMCObjectTracking_L."Type Object Modified" = RecCMCObjectTracking_L."Type Object Modified"::Delete THEN BEGIN
                  RecCMCObjectTracking_L."Type Object Modified" := RecCMCObjectTracking_L."Type Object Modified"::New;
                  RecCMCObjectTracking_L."Notify Object Modified" := TRUE;
                  RecCMCObjectTracking_L.MODIFY;
                END;
            END ELSE BEGIN
              RecCMCObjectTracking_L."Object Type" := RecObject_L.Type;
              RecCMCObjectTracking_L."Object ID" := RecObject_L.ID;
              RecCMCObjectTracking_L."Object Name" := RecObject_L.Name;
              RecObject_L.CALCFIELDS(Caption);
              RecCMCObjectTracking_L."Object Caption" := RecObject_L.Caption;
              RecCMCObjectTracking_L."Object Modified" := RecObject_L.Modifi';
              RecCMCObjectTracking_L."Object Date" := RecObject_L.Date;
              RecCMCObjectTracking_L."Object Time" := RecObject_L.Heure;
              RecCMCObjectTracking_L."Object BLOB Size" := RecObject_L."BLOB Size";
              RecCMCObjectTracking_L."Notify Object Modified" := TRUE;
              RecCMCObjectTracking_L."Type Object Modified" := RecCMCObjectTracking_L."Type Object Modified"::New;
              RecCMCObjectTracking_L.INSERT;
            END;
          END;
        UNTIL RecObjectTracking_L.NEXT = 0;
        RecCMCObjectMgtSetup_L."Last Object Timestamp" := RecObjectTracking_L."Object Timestamp";
        RecCMCObjectMgtSetup_L.MODIFY;
      END;

      //V'rifier les suppressions d'objet qui ne sont pas enregistr'es dans la table Object Tracking
      IF RecCMCObjectTracking_L.FIND('-') THEN
        REPEAT
          IF NOT RecObject_L.GET(RecCMCObjectTracking_L."Object Type",'',RecCMCObjectTracking_L."Object ID") THEN
            IF (RecCMCObjectTracking_L."Type Object Modified" <> RecCMCObjectTracking_L."Type Object Modified"::Delete) THEN BEGIN
              RecCMCObjectTracking_L."Notify Object Modified" := TRUE;
              RecCMCObjectTracking_L."Type Object Modified" := RecCMCObjectTracking_L."Type Object Modified"::Delete;
              RecCMCObjectTracking_L.MODIFY;
            END;
        UNTIL RecCMCObjectTracking_L.NEXT = 0;

      COMMIT;

      IF NOT cduSendObjectChange_L.RUN THEN BEGIN
        cduSendObjectChange_L.ClearMailServices;
        ERROR(GETLASTERRORTEXT);
      END;
    END;

    BEGIN
    {
      // --> 3LI BEGIN
      20130416-CMCTools-DRN01 : NEWOBJ
      // <-- 3LI END
    }
    END.
  }
