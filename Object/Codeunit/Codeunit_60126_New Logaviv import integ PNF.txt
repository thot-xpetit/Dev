    PROCEDURE ImportContractPNFNav@1000000003(TableContrat_P@1000000000 : Record 8053423) Result_L : Text[256];
    VAR
      "<-- 3LI LOCALS -->"@1000000022 : Integer;
      TableContrat_L@1000000021 : Record 8053423;
      TableContratNav_L@1000000020 : Record 8053268;
      TableLogavivCrop_L@1000000019 : Record 8053422;
      TableCorrespondance_L@1000000018 : Record 8061233;
      TableOrdre_L@1000000017 : Record 8053426;
      TableUtility_L@1000000016 : Record 8053434;
      ContratNavExiste_L@1000000015 : Boolean;
      Erreur_L@1000000014 : Boolean;
      IdFarmExtNum_L@1000000013 : Integer;
      IdSiloExtNum_L@1000000012 : Integer;
      I_L@1000000011 : Integer;
      Txt_temp_L@1000000010 : Text[250];
      Temp_L@1000000009 : Text[30];
      ErreurLigne_L@1000000007 : Boolean;
      IdContractExtInteger_L@1000000001 : Integer;
      ContractNoOpen_L@1000000002 : Code[20];
      ContractNoFixed_L@1000000003 : Code[20];
      TableContratOpen_L@1000000004 : Record 8053268;
      TableContratFixed_L@1000000005 : Record 8053268;
    BEGIN
      Result_L := '';

      TableContrat_L.SETRANGE(HoroIntegration, 0DT);
      TableContrat_L.SETFILTER(IdOfferType, '8|9');

      TableContrat_L.SETRANGE(IdContractInt, TableContrat_P.IdContractInt);
      TableContrat_L.SETRANGE(TypeContract, TableContrat_P.TypeContract);

      IF TableContrat_L.FINDSET THEN BEGIN
        REPEAT
          Erreur_L := FALSE;
          IF NOT TableLogavivCrop_G.GET(TableContrat_L.IdCrop) THEN BEGIN
            Erreur_L := TRUE;
            ERROR(Text0008, TableContrat_L.IdCrop, TableLogavivCrop_G.TABLECAPTION,
              TableContrat_L.IdContractInt, TableContrat_L.TypeContract);
          END ELSE BEGIN
            // Contrat deja integre ?
            IF TableContrat_L.RefNavAgri = '' THEN BEGIN
              // En creation de contrat : N Logaviv ne doit pas exister.
              ContratNavExiste_L := FALSE;
              TableContratNav_L.SETRANGE(Type, TableContratNav_L.Type::Open);
              TableContratNav_L.SETRANGE("Reference interface contract", FORMAT(TableContrat_L.IdContractInt));
              IF (TableContratNav_L.FINDFIRST) AND (TableContrat_L.BaseAmount <> 0) THEN BEGIN
                Erreur_L := TRUE;
                ERROR(Text0009, TableContrat_L.IdContractInt, TableContratNav_L.TABLECAPTION,
                  TableContrat_L.IdContractInt, TableContrat_L.TypeContract);
              END ELSE BEGIN
                TableContratNav_L.SETRANGE(Type, TableContratNav_L.Type::Fixed);
                IF (TableContratNav_L.FINDFIRST) AND (TableContrat_L.Price <> 0) THEN BEGIN
                  Erreur_L := TRUE;
                  ERROR(Text0009, TableContrat_L.IdContractInt, TableContratNav_L.TABLECAPTION,
                    TableContrat_L.IdContractInt, TableContrat_L.TypeContract);
                END;
              END;
              TableContratNav_L.SETRANGE("Reference interface contract");
              TableContratNav_L.SETRANGE(Type);
            END ELSE BEGIN
              // En Modification de contrat : On doit retrouver le contrat avec le meme numero.
              ContratNavExiste_L := TRUE;
              IF NOT TableContratNav_L.GET(TableContrat_L.RefNavAgri) THEN BEGIN
                Erreur_L := TRUE;
                ERROR(Text0011, TableContrat_L.RefNavAgri, TableContratNav_L.TABLECAPTION,
                  TableContrat_L.IdContractInt, TableContrat_L.TypeContract);
              END ELSE BEGIN
                IF (FORMAT(TableContrat_L.IdContractInt) <> TableContratNav_L."Reference interface contract") THEN BEGIN
                  Erreur_L := TRUE;
                  ERROR(Text0010, TableContrat_L.IdContractInt,
                    TableContratNav_L.FIELDCAPTION(TableContratNav_L."Reference interface contract"),
                    TableContratNav_L."Reference interface contract", TableContrat_L.IdContractInt,
                    TableContrat_L.TypeContract);
                 END;
              END;
           END;
        END;


        IF NOT Erreur_L THEN BEGIN
          IF (TableContrat_L.IdOrderStatut <> 3) OR (ContratNavExiste_L) THEN BEGIN

            IF NOT TableLogEncod_G.GET(FORMAT(TableContrat_L.IdProduct), TableContrat_L.IdQualityExt) THEN BEGIN
              ERROR(Text0013, TableContrat_L.FIELDCAPTION(TableContrat_L.IdQualityExt),
              FORMAT(TableContrat_L.IdProduct) + '-' + TableContrat_L.IdQualityExt,
              TableContrat_L.IdContractInt, TableContrat_L.TypeContract);
            END ELSE BEGIN
              Species_G := TableLogEncod_G."Species Code";
              Modele_G := TableLogEncod_G."NavAgri Template Contract No.";
            END;
            // Recherche echeance MATIF via table de correspondance (Fields=1)
            //MESSAGE('%1',GlobalTemplateCode_G);
            TableCorrespondance_L.SETRANGE("Template code", GlobalTemplateCode_G);
            TableCorrespondance_L.SETRANGE(Table, 8053107);
            TableCorrespondance_L.SETRANGE(Fields, 1);
            TableCorrespondance_L.SETFILTER("NAV Value", '%1', Species_G + 'Ø*');
            TableCorrespondance_L.SETRANGE("Interface Value", FORMAT(TableContrat_L.IdExpiry));
            IF NOT TableCorrespondance_L.FINDFIRST() THEN BEGIN
              CLEAR(Term_G);
            END ELSE BEGIN
              Position_G := 1;
              Txt_temp_L := ToolsFile_G.TexteTrim(TableCorrespondance_L."NAV Value", 'D');
              Temp_L := ToolsFile_G.ExtractTextCsv(Txt_temp_L, Position_G, 'Ø', 30);
              Term_G := ToolsFile_G.ExtractTextCsv(Txt_temp_L, Position_G, 'Ø', 30);
            END;
            //ce cas d'erreur ne doit pas ^tre consid'r' si IsVIsPrix = vrai et type contrat = 1
            IF NOT ((TableContrat_L.IsPrixNet) AND (TableContrat_L.TypeContract = 1)) THEN
              IF (Term_G = '') THEN BEGIN
                ERROR(STRSUBSTNO(Text0013, TableContrat_L.FIELDCAPTION(TableContrat_L.IdExpiry),
                Species_G + '-' + FORMAT(TableContrat_L.IdExpiry), TableContrat_L.IdContractInt) + '-2');
              END;

            IF TableContrat_L.Quotit = 0 THEN BEGIN
              ERROR(Text0012, TableContrat_L.IdContractInt, TableContrat_L.TypeContract);
            END;

            IF TableContrat_L.Quantity = 0 THEN BEGIN
              ERROR(Text60009, TableContrat_L.IdContractInt, TableContrat_L.TypeContract);
            END;


            IF TableUtility_L.GET(TableGeneralSetup_G.IdMember, TableContrat_L.IdFarmExt) THEN BEGIN
              Vendor_G := TableUtility_L."Id NavAgri";
            END ELSE BEGIN
              ERROR(Text0008, TableContrat_L.IdFarmExt, TableUtility_L.TABLECAPTION,
               TableContrat_L.IdContractInt, TableContrat_L.TypeContract);
            END;
          END;

          IF (TableContrat_L.IdOrderStatut = 3) AND (NOT ContratNavExiste_L) THEN BEGIN
            // Annulation jamais integree : toper comme integre sans creer de contrat
            TableContrat_L.HoroIntegration := CURRENTDATETIME;
            IF NOT TableContrat_L.MODIFY(TRUE) THEN BEGIN
              ERROR(Text0001, TableContrat_L.IdContractInt, TableContrat_L.TABLECAPTION, GETLASTERRORTEXT);
              EXIT('FALSE');
            END;
          END ELSE BEGIN


            // PNF : type Open (reservation base MATIF)
            IF ((TableContrat_L.BaseAmount <> 0) AND (TableContrat_L.Price = 0) AND (TableContrat_L.TypeContract = 0)) THEN  BEGIN
              Result_L := InsertOpenContract(TableContrat_L,TableContratNav_L,ContratNavExiste_L);
              IF Result_L = 'FALSE' THEN
                EXIT;
            END;

            // Vente directe
            IF (NOT TableContrat_L.IsPrixNet) AND (TableContrat_L.TypeContract = 1) THEN  BEGIN
              //MESSAGE('vente directe ?');
              ContractNoOpen_L := GetContractNo(TableContrat_P.IdContractInt,TRUE);
              ContractNoFixed_L := GetContractNo(TableContrat_P.IdContractInt,FALSE);
              IF ContractNoOpen_L = '' THEN
                Result_L := InsertOpenContract(TableContrat_L,TableContratNav_L,ContratNavExiste_L);
              //MESSAGE('Open pass', reste fixation,vente et contrat Navagri');
              IF ContractNoFixed_L = '' THEN
                Result_L := InsertFixedSoldContract(TableContrat_L,TableContratNav_L,ContratNavExiste_L,FALSE);
              //MESSAGE('Open pass', fixation pass'e, reste vente et contrat Navagri');

              Result_L := InsertFixedSoldContract(TableContrat_L,TableContratNav_L,ContratNavExiste_L,TRUE);
              //MESSAGE('Open pass', fixation pass'e, vente pass'e, reste contrat Navagri');

              IF Result_L = 'FALSE' THEN
                EXIT;
              //MESSAGE('Open et fixation pass', reste contrat Navagri');
              Result_L := CreateDirectCollectionContract(TableContrat_L);
              IF Result_L = 'FALSE' THEN
                EXIT;
            END;

            // PRIX NET
            IF (TableContrat_L.IsPrixNet) AND (TableContrat_L.TypeContract = 1) THEN  BEGIN
              //MESSAGE('Prix net ?');
              Result_L := CreateDirectCollectionContract(TableContrat_L);
              IF Result_L = 'FALSE' THEN
                EXIT;
            END;


            // PNF : type Fixed (fixation MATIF)
            {IF (TableContrat_L.Price <> 0) AND ((TableContrat_L.BaseAmount = 0)) THEN BEGIN
              Result_L := InsertFixedSoldContract(TableContrat_L,TableContratNav_L,ContratNavExiste_L,FALSE);
              IF Result_L = 'FALSE' THEN
                EXIT;
            END;}


          END;
          END;
        UNTIL TableContrat_L.NEXT = 0;
      END;

      //a r'activer

      RunOrders();
    END;

            {IF (((TableOrdre_L.BaseAmount <> 0) AND (TableOrdre_L.FixedPrice = 0) AND (ContractNoOpen_L = '')) OR

            // PNF : type Fixed (fixation MATIF)
            IF (((TableOrdre_L.ObjectifPrice = 0) AND (TableOrdre_L.FixedPrice <> 0) AND (ContractNoFixed_L = '')) OR
              ((TableOrdre_L.ObjectifPrice <> 0) AND (TableOrdre_L.FixedPrice <> 0) AND (ContractNoFixed_L = ''))) THEN BEGIN
              Result_L := InsertFixedSoldOrder(TableOrdre_L,TableContratOpen_L,FALSE,ContractNoOpen_L,ContractNoFixed_L);
              IF Result_L = 'FALSE' THEN
                EXIT;
            END;



            // PNF : type Fixed (reservation base + fixation MATIF)
            ContractNoOpen_L := GetContractNo(TableOrdre_L.IdContractInt,TRUE);
            IF ContractNoOpen_L <> '' THEN
              TableContratOpen_L.GET(ContractNoOpen_L);
            ContractNoFixed_L := GetContractNo(TableOrdre_L.IdContractInt,FALSE);
            IF ContractNoFixed_L <> '' THEN
              TableContratFixed_L.GET(ContractNoFixed_L);
            IF (TableOrdre_L.IdOrderStatut = 2) THEN
              IF (ContractNoOpen_L <> '') AND (ContractNoFixed_L <> '') THEN BEGIN
                Result_L := InsertFixedSoldOrder(TableOrdre_L,TableContratOpen_L,TRUE,ContractNoOpen_L,ContractNoFixed_L);
                IF Result_L = 'FALSE' THEN
                  EXIT;
                CreateCollectionContract(TableOrdre_L);
              END;


         END;

      END;
