      WhseRqst.SETRANGE("Source No.", P_PurchaseOrderNo);
      WhseRqst.SETRANGE(Type,WhseRqst.Type::Inbound);
      WhseRqst.FILTERGROUP(0);
      WhseRqst.SETRANGE("Document Status",WhseRqst."Document Status"::Released);
      WhseRqst.SETRANGE("Completely Handled",FALSE);

      WhseRqst.SETCURRENTKEY("Destination No.","Source No.","Location Code");

      SourceDocSelection.LOOKUPMODE(TRUE);
      SourceDocSelection.SETTABLEVIEW(WhseRqst);
      IF SourceDocSelection.RUNMODAL <> ACTION::LookupOK THEN
        EXIT;

      SourceDocSelection.GetResult(WhseRqst);

      GetSourceDocuments.SetOneCreatedReceiptHeader(WhseReceiptHeader);

      WhseRqst.SETRANGE("Document Status");
      WhseRqst.MARKEDONLY(TRUE);

      IF NOT WhseRqst.FIND('-') THEN BEGIN
        WhseRqst.MARKEDONLY(FALSE);
        WhseRqst.SETRECFILTER;
      END;

      IF WhseRqst.FINDFIRST THEN BEGIN

        L_recPurchaseHeader.GET(L_recPurchaseHeader."Document Type"::Order, P_PurchaseOrderNo);
        L_cduReleasePurchaseDocument.Reopen(L_recPurchaseHeader);

        REPEAT

           IF WhseRqst."Location Code" <> WhseReceiptHeader."Location Code" THEN BEGIN

             L_recPurchaseLine.RESET;
             L_recPurchaseLine.SETRANGE("Document Type", L_recPurchaseHeader."Document Type");
             L_recPurchaseLine.SETRANGE("Document No.", L_recPurchaseHeader."No.");
             L_recPurchaseLine.SETRANGE("Completely Received", FALSE);
             L_recPurchaseLine.SETRANGE("Location Code", WhseRqst."Location Code");
             IF L_recPurchaseLine.FINDSET(TRUE, FALSE) THEN BEGIN
               REPEAT
                  IF L_recPurchaseLine."Quantity Received" = 0 THEN
                    L_recPurchaseLine.VALIDATE("Location Code", WhseReceiptHeader."Location Code")
                  ELSE BEGIN
                    L_recPurchaseLine2.INIT;
                    L_recPurchaseLine2.TRANSFERFIELDS(L_recPurchaseLine);

                    L_recPurchaseLine3.RESET;
                    L_recPurchaseLine3.SETRANGE("Document Type", L_recPurchaseHeader."Document Type");
                    L_recPurchaseLine3.SETRANGE("Document No.", L_recPurchaseHeader."No.");
                    L_recPurchaseLine3.FINDLAST;

                    L_recPurchaseLine2.VALIDATE("Line No.", L_recPurchaseLine3."Line No." + 10000);
                    L_recPurchaseLine2.VALIDATE("Quantity Received", 0);
                    L_recPurchaseLine2.VALIDATE("Quantity Invoiced", 0);
                    L_recPurchaseLine2.VALIDATE(Quantity, L_recPurchaseLine.Quantity - L_recPurchaseLine."Quantity Received");
                    L_recPurchaseLine2.VALIDATE("Location Code", WhseReceiptHeader."Location Code");
                    L_recPurchaseLine2.INSERT(TRUE);
                    L_recPurchaseLine.VALIDATE(Quantity, L_recPurchaseLine."Quantity Received");
                  END;
                  L_recPurchaseLine.MODIFY(TRUE);
               UNTIL L_recPurchaseLine.NEXT = 0;
             END;

           END;

        UNTIL WhseRqst.NEXT = 0;

        WhseRqst2.INIT;
        WhseRqst2.TRANSFERFIELDS(WhseRqst);
        IF WhseRqst2.INSERT(TRUE) THEN;
        WhseRqst2.VALIDATE("Location Code", WhseReceiptHeader."Location Code");
        WhseRqst2.VALIDATE("Document Status", WhseRqst2."Document Status"::Released);
        IF WhseRqst2.MODIFY(TRUE) THEN;
        WhseRqst2.COPYFILTERS(WhseRqst);

        WhseRqst.DELETEALL(TRUE);

        L_cduReleasePurchaseDocument.RUN(L_recPurchaseHeader);
        L_cduWhsePurchRelease.Release(L_recPurchaseHeader);

        WhseRqst2.SETRANGE("Location Code", WhseReceiptHeader."Location Code");

        GetSourceDocuments.USEREQUESTFORM(FALSE);
        GetSourceDocuments.SETTABLEVIEW(WhseRqst2);
        GetSourceDocuments.RUNMODAL;

      END;

    END;
