{
  OBJECT-PROPERTIES
  {
    Date=12/10/20;
    Heure=00:00:00;
    Liste versions=SP3490;
  }
  PROPERTIES
  {
    TableNo=27;
    OnRun=VAR
            CMCManagement_L@1000000000 : Codeunit 8052982;
            ItemVariant_L@1000000001 : Record 5401;
            Item_L@1000000002 : Record 27;
            ValeurRpd_L@1000000003 : Decimal;
          BEGIN
            CLEAR(CMCManagement_L);
            //on gSre d'abord l'article lui-m^me
            Item_L.RESET;
            Item_L.SETRANGE("No.","No.");
            IF Item_L.FINDFIRST THEN BEGIN
              // --> 20220203-WI00-ADV01.01.00-35609-CJY01
              {BEFORE
              Item_L."Valeur RPD" := CMCManagement_L.EcoTaxCalculationFromRecItem(Item_L,WORKDATE,'','');
              Item_L.MODIFY(TRUE);
              AFTER}
              ValeurRpd_L := CMCManagement_L.EcoTaxCalculationFromRecItem(Item_L,WORKDATE,'','');
              IF Item_L."Valeur RPD" <> ValeurRpd_L THEN BEGIN
                Item_L."Valeur RPD" := ValeurRpd_L;
                Item_L.MODIFY(TRUE);
              END;
              // <-- 20220203-WI00-ADV01.01.00-35609-CJY01
            END;
            //S'il existe des variante, on les gSre
            ItemVariant_L.RESET;
            ItemVariant_L.SETRANGE("Item No.","No.");
            IF ItemVariant_L.FINDSET THEN REPEAT
              // --> 20220203-WI00-ADV01.01.00-35609-CJY01
              {BEFORE
              ItemVariant_L."Valeur RPD" := CMCManagement_L.EcoTaxCalculationFromRecItem(Item_L,WORKDATE,'',ItemVariant_L.Code);
              ItemVariant_L.MODIFY(TRUE);
              AFTER}
              ValeurRpd_L := CMCManagement_L.EcoTaxCalculationFromRecItem(Item_L,WORKDATE,'',ItemVariant_L.Code);
              IF ItemVariant_L."Valeur RPD" <> ValeurRpd_L THEN BEGIN
                ItemVariant_L."Valeur RPD" := ValeurRpd_L;
                ItemVariant_L.MODIFY(TRUE);
              END;
              // <-- 20220203-WI00-ADV01.01.00-35609-CJY01

            UNTIL ItemVariant_L.NEXT = 0;
          END;

  }

      // <-- 3LI END

  }
}
