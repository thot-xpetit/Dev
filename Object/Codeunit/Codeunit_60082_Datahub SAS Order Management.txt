    PROCEDURE ImportAllSimpleOrder@8053747();
    VAR
      RecSASHeader_L@8053600 : Record 8053610;
      NbLineReal_L@1000000000 : Integer;
      RecSASLines_L@1000000001 : Record 8053611;
    BEGIN
      //----------------------------------------------------------------------------------------------------------------------------------
      // Import de toutes les commandes dans le SAS
      //----------------------------------------------------------------------------------------------------------------------------------
      //  Entr'es
      //  Sorties
      //----------------------------------------------------------------------------------------------------------------------------------
      RecSASHeader_L.SETCURRENTKEY("SAS Status");
      RecSASHeader_L.SETFILTER("SAS Status",'<>%1&<>%2',RecSASHeader_L."SAS Status"::Created,RecSASHeader_L."SAS Status"::Treated);
      RecSASHeader_L.SETRANGE(Deleted,FALSE);
      IF RecSASHeader_L.FINDSET(FALSE,FALSE) THEN BEGIN
        REPEAT
          // --> 20201120-WI00-ADV01.01.00-3490-CJY01
          //on ne traite que si la condition est remplie : nb ligne indiqu' dans l'ent^te = nb de lignes r'elles :
          NbLineReal_L := 0;
          RecSASLines_L.RESET;
          RecSASLines_L.SETRANGE("CRM Header Guid",RecSASHeader_L."CRM Guid");
          IF NOT RecSASLines_L.ISEMPTY THEN
            NbLineReal_L := RecSASLines_L.COUNT;

          IF RecSASHeader_L."Nb Lines Sas Order" = NbLineReal_L THEN
          // <-- 20201120-WI00-ADV01.01.00-3490-CJY01

          //tentative d'import de la commande
          ImportSimpleOrder(RecSASHeader_L);
        UNTIL RecSASHeader_L.NEXT = 0;
        CleanSAS;
      END;
    END;

    PROCEDURE ImportSimpleOrderWithoutInErro@8053746();
    VAR
      RecSASHeader_L@8053600 : Record 8053610;
      RecCMCCRMConnectorSetup_L@1000000000 : Record 8053602;
      RecSASLines_L@1000000001 : Record 8053611;
      NbLineReal_L@1000000002 : Integer;
    BEGIN
      //----------------------------------------------------------------------------------------------------------------------------------
      // Import des commandes du SAS n''tant pas en erreur
      //----------------------------------------------------------------------------------------------------------------------------------
      //  Entr'es
      //  Sorties
      //----------------------------------------------------------------------------------------------------------------------------------
      RecCMCCRMConnectorSetup_L.GET;
      RecSASHeader_L.SETCURRENTKEY("SAS Status");
      IF (RecCMCCRMConnectorSetup_L."Begin Time SAS" <> 0T) AND (RecCMCCRMConnectorSetup_L."End Time SAS" <> 0T) THEN BEGIN
        IF NOT((RecCMCCRMConnectorSetup_L."Begin Time SAS" < TIME) AND (RecCMCCRMConnectorSetup_L."End Time SAS" > TIME)) THEN
          RecSASHeader_L.SETFILTER("SAS Status",'%1|%2',RecSASHeader_L."SAS Status"::"In Progress",RecSASHeader_L."SAS Status"::" ")
        ELSE BEGIN
          RecSASHeader_L.SETCURRENTKEY("Secondary Key","SAS Status");
          RecSASHeader_L.SETFILTER("SAS Status",'%1',RecSASHeader_L."SAS Status"::Error);
        END;
      END ELSE
        RecSASHeader_L.SETFILTER("SAS Status",'<>%1&<>2',RecSASHeader_L."SAS Status"::Created,RecSASHeader_L."SAS Status"::Treated);
      RecSASHeader_L.SETRANGE(Deleted,FALSE);
      IF RecSASHeader_L.FINDSET(FALSE, FALSE) THEN BEGIN
        REPEAT
          // --> 20201120-WI00-ADV01.01.00-3490-CJY01
          //on ne traite que si la condition est remplie : nb ligne indiqu' dans l'ent^te = nb de lignes r'elles :
          NbLineReal_L := 0;
          RecSASLines_L.RESET;
          RecSASLines_L.SETRANGE("CRM Header Guid",RecSASHeader_L."CRM Guid");
          // --> 20210107-WI00-ADV01.01.00-3490-CJY01
          //on ne doit compter que les lignes qui ne sont pas en statut delete
          RecSASLines_L.SETRANGE(Deleted,FALSE);
          // <-- 20210107-WI00-ADV01.01.00-3490-CJY01
          IF NOT RecSASLines_L.ISEMPTY THEN
            NbLineReal_L := RecSASLines_L.COUNT;
          // --> 20210106-WI00-ADV01.01.00-3490-CJY01
          {REPLACE
          IF RecSASHeader_L."Nb Lines Sas Order" = NbLineReal_L THEN
          AFTER}
          IF ((RecSASHeader_L."Nb Lines Sas Order" = NbLineReal_L) OR (RecSASHeader_L."Nb Lines Sas Order" = 0)) THEN
          // <-- 20210106-WI00-ADV01.01.00-3490-CJY01
          // <-- 20201120-WI00-ADV01.01.00-3490-CJY01
          // --> 20221219-WI00-ADV01.01.00-3613-CJY01
          HeaderStartDateTime_G := CREATEDATETIME(TODAY,TIME);
          // <-- 20221219-WI00-ADV01.01.00-3613-CJY01

          //tentative d'import de la commande
          ImportSimpleOrder(RecSASHeader_L);
        UNTIL RecSASHeader_L.NEXT = 0;
        CleanSAS;
      END;
    END;

    PROCEDURE ImportSimpleOrder@8053758(RecSASHeader_P@8053600 : Record 8053610);
    VAR
      RecSASHeaderUpd_L@1000000001 : Record 8053610;
      RecSASLine_L@1000000000 : Record 8053611;
      CduDatahubSASProcessOrder_L@8053603 : Codeunit 60083;
      SalesHeader_L@1000000002 : Record 36;
      SalesLine_L@1000000003 : Record 37;
    BEGIN
      //----------------------------------------------------------------------------------------------------------------------------------
      // import d'une commande SAS
      //----------------------------------------------------------------------------------------------------------------------------------
      //  Entr'es
      //    - recSASHeader_P : ent^te de SAS de commande de vente . int'grer
      //  Sorties
      //    ERROR si erreur
      //----------------------------------------------------------------------------------------------------------------------------------
      CduDatahubSASProcessOrder_L.SetTypeSync(Type_G::Order);
      CduDatahubSASProcessOrder_L.SetParameterOrder(RecSASHeader_P);
      IF NOT CduDatahubSASProcessOrder_L.RUN THEN BEGIN
        RecSASHeaderUpd_L.GET(RecSASHeader_P."CRM Guid");
        RecSASHeaderUpd_L."SAS Status" := RecSASHeader_P."SAS Status"::Error;
        RecSASHeaderUpd_L."Synchro Error" := COPYSTR(GETLASTERRORTEXT,1,MAXSTRLEN(RecSASHeaderUpd_L."Synchro Error"));
        // --> 20221219-WI00-ADV01.01.00-3613-CJY01
        HeaderEndDateTime_G := CREATEDATETIME(TODAY,TIME);
        IF HeaderEndDateTime_G >= HeaderStartDateTime_G THEN
          RecSASHeaderUpd_L."Timing integration try" := HeaderEndDateTime_G - HeaderStartDateTime_G;
        // <-- 20221219-WI00-ADV01.01.00-3613-CJY01

        RecSASHeaderUpd_L.MODIFY;
        //gestion de la cr'ation d'une ent^te sans ligne
        //cas ou le traitement d'int'gration a rencontr' un erreur
        IF RecSASHeaderUpd_L."No." <> '' THEN
          IF SalesHeader_L.GET(SalesHeader_L."Document Type"::Order,RecSASHeaderUpd_L."No.") THEN BEGIN
            SalesLine_L.SETRANGE("Document Type",SalesHeader_L."Document Type");
            SalesLine_L.SETRANGE("Document No.",SalesHeader_L."No.");
            IF SalesLine_L.ISEMPTY THEN BEGIN
              //si pas de lignes, on supprime l'ent^te
              SalesHeader_L.DELETE(TRUE);
              //on vide le num'ro de commande Nav g'n'r'
              RecSASHeaderUpd_L."No." := '';
              RecSASHeaderUpd_L.MODIFY;
            END;
          END;
      END ELSE BEGIN
        RecSASHeaderUpd_L.GET(RecSASHeader_P."CRM Guid");
        //suppression des lignes termin'es
        RecSASLine_L.SETRANGE("CRM Header Guid",RecSASHeader_P."CRM Guid");
        RecSASLine_L.SETFILTER("SAS Status",'<>%1&<>%2',RecSASLine_L."SAS Status"::Created,RecSASLine_L."SAS Status"::Treated);
        RecSASLine_L.SETRANGE(Deleted,FALSE);
        IF RecSASLine_L.ISEMPTY THEN BEGIN
          //la commande est int'gr'e et il n'y a plus de ligne
          RecSASHeaderUpd_L."SAS Status" := RecSASHeaderUpd_L."SAS Status"::Created;
          RecSASHeaderUpd_L."Treatment Date" := CREATEDATETIME(WORKDATE,TIME);
          RecSASHeaderUpd_L."Synchro Error" := '';
          // --> 20221219-WI00-ADV01.01.00-3613-CJY01
          HeaderEndDateTime_G := CREATEDATETIME(TODAY,TIME);
          IF HeaderEndDateTime_G >= HeaderStartDateTime_G THEN
            RecSASHeaderUpd_L."Timing integration" := HeaderEndDateTime_G - HeaderStartDateTime_G;
          // <-- 20221219-WI00-ADV01.01.00-3613-CJY01

          RecSASHeaderUpd_L.MODIFY;
        END ELSE BEGIN
          //il reste des lignes en erreur
          RecSASHeaderUpd_L."SAS Status" := RecSASHeaderUpd_L."SAS Status"::Error;
          RecSASHeaderUpd_L."Synchro Error" := Text8053683;
          // --> 20221219-WI00-ADV01.01.00-3613-CJY01
          HeaderEndDateTime_G := CREATEDATETIME(TODAY,TIME);
          IF HeaderEndDateTime_G >= HeaderStartDateTime_G THEN
            RecSASHeaderUpd_L."Timing integration try" := HeaderEndDateTime_G - HeaderStartDateTime_G;
          // <-- 20221219-WI00-ADV01.01.00-3613-CJY01

          RecSASHeaderUpd_L.MODIFY;
        END;

        //gestion de la cr'ation d'une ent^te sans ligne
        //cas ou le traitement d'int'gration n'a pas rencontr' d'erreur en ent^te
        //mais ou la suite n'a pas permi de cr'e au moins ne ligne dans la commande
        IF RecSASHeaderUpd_L."No." <> '' THEN
          IF SalesHeader_L.GET(SalesHeader_L."Document Type"::Order,RecSASHeaderUpd_L."No.") THEN BEGIN
            SalesLine_L.SETRANGE("Document Type",SalesHeader_L."Document Type");
            SalesLine_L.SETRANGE("Document No.",SalesHeader_L."No.");
            IF SalesLine_L.ISEMPTY THEN BEGIN
              //si pas de lignes, on supprime l'ent^te
              SalesHeader_L.DELETE(TRUE);
              RecSASHeaderUpd_L."No." := '';
              RecSASHeaderUpd_L.MODIFY;
            END;
          END;
      END;

      COMMIT;
    END;
