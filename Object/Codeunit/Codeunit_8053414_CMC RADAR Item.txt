    PROCEDURE GetRadarParameters@1000000006() ReturnValue : Boolean;
    VAR
      RadarSetup@1000000000 : Record 8053404;
    BEGIN
      ReturnValue := FALSE;
      TempRadarParameter_G.DELETEALL();
      IF (RadarSetup.FINDSET()) THEN BEGIN
        TempRadarParameter_G.INIT();
        TempRadarParameter_G.TRANSFERFIELDS(RadarSetup);
        IF (TempRadarParameter_G.INSERT(FALSE)) THEN BEGIN
          ReturnValue := TempRadarParameter_G.FINDSET();
        END;
      END;
    END;

        RadarSetup."Last Item File Number" := TempRadarParameter_G."Last Item File Number";
        ReturnValue := RadarSetup.MODIFY(TRUE);
