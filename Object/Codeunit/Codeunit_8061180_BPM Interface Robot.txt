    PROCEDURE StartStopInterface@8061021(InterfaceID_P@8060984 : Code[20];ServiceBPM_P@8060983 : Integer;Starting_P@8060982 : Boolean);
    VAR
      InterfaceID_L@8060986 : Record 8061187;
      LogStatistic_L@8060985 : Record 8061190;
    BEGIN
      // ========================================================================
      // Mise . jour de l'interface selon action sur le service (START / STOP)
      // ========================================================================
      CASE ServiceBPM_P OF
        // --------------------------------------------------------------------
        1: BEGIN // Watcher
        // --------------------------------------------------------------------
          CASE Starting_P OF
            TRUE: BEGIN // Service is started
              InterfaceID_L.LOCKTABLE(TRUE, FALSE);
              IF InterfaceID_L.GET(InterfaceID_P) THEN BEGIN
                InterfaceID_L."Last Starting Date" := CURRENTDATETIME;
                InterfaceID_L."Last Stopping Date" := 0DT;
                InterfaceID_L."Configuration Locked" := TRUE;
                InterfaceID_L.MODIFY;
                COMMIT;
              END;
            END;
            FALSE: BEGIN // Service is stopped
              InterfaceID_L.LOCKTABLE(TRUE, FALSE);
              IF InterfaceID_L.GET(InterfaceID_P) THEN BEGIN
                InterfaceID_L."Last Stopping Date" := CURRENTDATETIME;
                InterfaceID_L.MODIFY;
                COMMIT;
              END;
            END;
          END;
        END;
        // --------------------------------------------------------------------
        2: BEGIN // Processor
        // --------------------------------------------------------------------
          CASE Starting_P OF
            TRUE: BEGIN // Service is started
              InterfaceID_L.LOCKTABLE(TRUE, FALSE);
              IF InterfaceID_L.GET(InterfaceID_P) THEN BEGIN
                InterfaceID_L."Processor Service Started" := TRUE;
                InterfaceID_L.MODIFY;
                COMMIT;
              END;
            END;
            FALSE: BEGIN // Service is stopped
              InterfaceID_L.LOCKTABLE(TRUE, FALSE);
              IF InterfaceID_L.GET(InterfaceID_P) THEN BEGIN
                InterfaceID_L."Processor Service Started" := FALSE;
                InterfaceID_L.MODIFY;
                COMMIT;
              END;
            END;
          END;
        END;
      END;
    END;

    PROCEDURE MessageProcess@8061011(MessageGUID_P@8060982 : GUID) Result_L : Boolean;
    VAR
      IncomingMessage_L@8060985 : Record 8061188;
      LogMessage_L@8060984 : Text[1024];
      LogStatistic_L@8060983 : Record 8061190;
    BEGIN
      // ========================================================================
      // TRAITEMENT d'un MESSAGE ENTRANT
      // ========================================================================
      Result_L := TRUE;

      IncomingMessage_G.GET(MessageGUID_P);

      // --------------------------------------------------------------------------------------
      LogMessage_G.Add(IncomingMessage_G."Message ID", '', '',
                       STRSUBSTNO(FORMAT(Text8060984), IncomingMessage_G."Interface ID"),
                       LogMessage_G."Log Level"::Information, 0);
      // --------------------------------------------------------------------------------------

      IF InterfaceID_G.GET(IncomingMessage_G."Interface ID") THEN BEGIN

        CASE InterfaceID_G.Product OF

          // ========================================================================
          InterfaceID_G.Product::"Dynamics NAV": BEGIN
          // ------------------------------------------------------------------------
            IF DynamicsSetup_G.GET(InterfaceID_G."Interface ID") THEN BEGIN

              Result_L := cduDynamicsRobot_G.RUN(IncomingMessage_G);

            END ELSE BEGIN

              // ------------------------------------------------------------------
              // Absence paramStre Dynamics NAV pour l'interface
              // ------------------------------------------------------------------
              LogMessage_G.Add(IncomingMessage_G."Message ID", '', '',
                               STRSUBSTNO(FORMAT(Text8060986), IncomingMessage_G."Interface ID"),
                               LogMessage_G."Log Level"::Critical, 1);

              Result_L := FALSE;

            END;
          END;
          // ------------------------------------------------------------------------

          // ========================================================================
          InterfaceID_G.Product::"Dynamics CRM": BEGIN
          // ------------------------------------------------------------------------
            IF DynamicsCRMSetup_G.GET(InterfaceID_G."Interface ID") THEN BEGIN

              Result_L := cduDynamicsCRMRobot_G.RUN(IncomingMessage_G);

            END ELSE BEGIN

              // ------------------------------------------------------------------
              // Absence paramStre Dynamics CRM pour l'interface
              // ------------------------------------------------------------------
              LogMessage_G.Add(IncomingMessage_G."Message ID", '', '',
                               STRSUBSTNO(FORMAT(Text8060987), IncomingMessage_G."Interface ID"),
                               LogMessage_G."Log Level"::Critical, 1);

              Result_L := FALSE;

            END;
          END;
          // ------------------------------------------------------------------------

          ELSE BEGIN

              // ------------------------------------------------------------------
              // Produit Interface non pris en charge
              // ------------------------------------------------------------------
              LogMessage_L := STRSUBSTNO(FORMAT(Text8060982), IncomingMessage_G."Interface ID",InterfaceID_G.Product);

              LogMessage_G.Add(IncomingMessage_G."Message ID", '', '', LogMessage_L, LogMessage_G."Log Level"::Critical, 1);

              Result_L := FALSE;

          END;

        END;

      END ELSE BEGIN

        // ------------------------------------------------------------------
        // Interface ID inconnue
        // ------------------------------------------------------------------
        LogMessage_G.Add(IncomingMessage_G."Message ID", '', '',
                         STRSUBSTNO(FORMAT(Text8060983), IncomingMessage_G."Interface ID"),
                         LogMessage_G."Log Level"::Critical, 1);

        Result_L := FALSE;

      END;



      IF IncomingMessage_G.Type = IncomingMessage_G.Type::"Scheduled Process" THEN BEGIN
        //-------------------------------------------------------------------------------------
        // Mise . jour de la planification du processus planifi'
        //-------------------------------------------------------------------------------------
        UpdateScheduledProcessus(MessageGUID_P, NOT Result_L);

      END ELSE BEGIN

        IF NOT Result_L THEN BEGIN

          IncomingMessage_G.ChangeStatus(MessageGUID_P, IncomingMessage_G.Status::Error);

          IF GETLASTERRORTEXT <> '' THEN BEGIN
            LogMessage_G.Add(MessageGUID_P, '', '', GETLASTERRORTEXT, LogMessage_G."Log Level"::Critical, 1);
            CLEARLASTERROR;
          END;


          CASE InterfaceID_G.Product OF
            // -------------------------------------------------------------
            InterfaceID_G.Product::"Dynamics NAV": BEGIN
            // -------------------------------------------------------------
              IF EXISTS(IncomingMessage_G."Parameter 1") THEN
                cduTools_G.MoveFileToFolder2(IncomingMessage_G."Parameter 1", DynamicsSetup_G."Errors Dir",TRUE, MessageGUID_P, 1);
              IF EXISTS(IncomingMessage_G."Parameter 2") THEN
                cduTools_G.MoveFileToFolder2(IncomingMessage_G."Parameter 2", DynamicsSetup_G."Errors Dir",TRUE, MessageGUID_P, 1);

            END;
            // -------------------------------------------------------------
            InterfaceID_G.Product::"Dynamics CRM": BEGIN
            // -------------------------------------------------------------
              IF EXISTS(IncomingMessage_G."Parameter 1") THEN
                cduTools_G.MoveFileToFolder2(IncomingMessage_G."Parameter 1", DynamicsCRMSetup_G."Errors Dir",TRUE, MessageGUID_P, 1);
              IF EXISTS(IncomingMessage_G."Parameter 2") THEN
                cduTools_G.MoveFileToFolder2(IncomingMessage_G."Parameter 2", DynamicsCRMSetup_G."Errors Dir",TRUE, MessageGUID_P, 1);
              IF EXISTS(IncomingMessage_G."Parameter 3") THEN
                cduTools_G.MoveFileToFolder2(IncomingMessage_G."Parameter 3", DynamicsCRMSetup_G."Errors Dir",TRUE, MessageGUID_P, 1);
              IF EXISTS(IncomingMessage_G."Parameter 4") THEN
                cduTools_G.MoveFileToFolder2(IncomingMessage_G."Parameter 4", DynamicsCRMSetup_G."Errors Dir",TRUE, MessageGUID_P, 1);
            END;
          END; // CASE

        END;

      END;

      LogMessage_G.SetRecordPosition(MessageGUID_P,
                                     cduTools_G.GetProcessRecordKey,
                                     cduTools_G.GetProcessRecordTable);


      //-------------------------------------------------------------------------------------
      // PURGE DU JOURNAL / STATISTIQUES selon PARAMETRES
      //-------------------------------------------------------------------------------------
      IF InterfaceID_G."Auto Purge" THEN BEGIN
        IncomingMessage_L.RESET;
        IncomingMessage_L.SETCURRENTKEY("Interface ID", Execution);
        IncomingMessage_L.SETRANGE("Interface ID", InterfaceID_G."Interface ID");

        IncomingMessage_L.SETFILTER(Execution, '>%1 & <%2', 0DT,
                  CREATEDATETIME(CALCDATE('<-'+FORMAT(InterfaceID_G."Auto Purge Param")+'D>', TODAY), 235959T));

                  CALCDATE('<-'+FORMAT(InterfaceID_G."Auto Purge Param")+'D>', TODAY));
        LogStatistic_L.DELETEALL(TRUE);
      END;

      EXIT(Result_L);
    END;
