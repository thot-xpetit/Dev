    PROCEDURE CheckNumberFormat@1000000001(DataToFormat@1000000000 : Text[30]) ReturnValue : Text[30];
    VAR
      index@1000000001 : Integer;
      PointChar@1000000002 : Char;
      CommaChar@1000000003 : Char;
      MinusChar@1000000004 : Char;
      ZeroChar@1000000005 : Char;
      NineChar@1000000006 : Char;
      CharToProcess@1000000007 : Char;
    BEGIN
      // <-- 20250821-CLA221680-ADVITAM-XPE
      ReturnValue := '';
      PointChar := '.';
      MinusChar := '-';
      CommaChar := ',';
      ZeroChar := '0';
      NineChar := '9';
      FOR index := 1 TO STRLEN(DataToFormat) DO BEGIN
        CharToProcess := DataToFormat[index];
        IF (
            (CharToProcess = PointChar) OR
            (CharToProcess = MinusChar) OR
            (CharToProcess = CommaChar) OR
            ((CharToProcess >= ZeroChar) AND (CharToProcess <= NineChar))
           ) THEN BEGIN
          ReturnValue := ReturnValue + FORMAT(CharToProcess);
        END;
      END;
      // --> 20250821-CLA221680-ADVITAM-XPE
    END;
