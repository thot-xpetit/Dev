OBJECT Codeunit 50033 MAJ Group Compta UNEAL
{
  OBJECT-PROPERTIES
  {
    Date=04/04/16;
    Heure=10:39:01;
    Modifi'=Oui;
    Liste versions=REP001;
  }
  PROPERTIES
  {
    OnRun=BEGIN

            //===============================================================================================================================
            //                      Mises . jour COMPTA
            //===============================================================================================================================

            Cpt  := 0;
            Cpt2 := 0;
            Fournisseur_G.RESET;
            Total := Fournisseur_G.COUNT;
            Fenetre.OPEN('Maj des groupes compta          \' +
                         'Nb lecture                      #1######\' +
                         'Nb fournisseurs                 #2######\' +
                         '@4@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@',Cpt,Cpt2);

            IF Fournisseur_G.FIND('-') THEN
            REPEAT

              IF (Fournisseur_G."Vendor Posting Group") ='ADH_APPORT' THEN BEGIN
                IF Fournisseur_G."Gen. Bus. Posting Group" ='FRANCE_D' THEN BEGIN
                Fournisseur_G."Gen. Bus. Posting Group":='ADH_FRANCE';
                END;
                IF Fournisseur_G."VAT Bus. Posting Group" ='FRANCE_D' THEN BEGIN
                Fournisseur_G."VAT Bus. Posting Group":='ADH_FRANCE';
                END;
                Fournisseur_G.MODIFY;
                Cpt2 := Cpt2 + 1;
              END;

            Cpt := Cpt + 1;
            Fenetre.UPDATE;
            Fenetre.UPDATE(1,Cpt);
            Fenetre.UPDATE(2,Cpt2);
            Fenetre.UPDATE(4,ROUND(Cpt / Total * 10000, 1));

            UNTIL Fournisseur_G.NEXT = 0;
            Fenetre.CLOSE;
          END;

  }
  CODE
  {
    VAR
      Total@1000000004 : Integer;
      Cpt@1000000005 : Integer;
      Cpt2@1000000006 : Integer;
      Cpt3@1000000022 : Integer;
      Fenetre@1000000007 : Dialog;
      Ok_lancement@1000000013 : Boolean;
      Fournisseur_G@1000000000 : Record 23;

    BEGIN
    {
      Liste modifications :

      20150404-Rpt01-ADV01.01.00_ADVITAM-KRE-001 : Mise . jour des groupes compta des fournisseurs (T23)
    }
    END.
  }
}
