    VAR

      GlobalTemplateCode_G@1000000000 : Code[20];

      Text0003@1000000016 : TextConst 'ENU=Error open file %1 : %2;FRA=Erreur ouverture fichier %1 : %2';

    PROCEDURE ImportTexte@1000000001(FileName_P@1000000000 : Text[256]) Result_L : Text[256];
    VAR
      "<-- 3LI LOCALS -->"@1000000001 : Integer;
      TableSasImport_L@1000000008 : Record 8087791;
      TableItem_L@1000000015 : Record 27;
      FileLogName_L@1000000003 : Text[256];
      FileImport_L@1000000004 : File;
      FileInStream_L@1000000005 : InStream;
      NbCarLu_L@1000000006 : Integer;
      TexteLu_L@1000000007 : Text[1024];
      Pos_L@1000000010 : Integer;
      Texte_L@1000000011 : Text[256];
      DateText_L@1000000012 : Text[10];
      EnregOK_L@1000000013 : Boolean;
      TextTemp_L@1000000002 : Text[256];
      Coef_L@1000000009 : Decimal;
      Montant_L@1000000014 : Decimal;
      Ship_L@1000000016 : Code[10];
      ShipRadar_L@1000000017 : Text[30];
    BEGIN
      //----------------------------------------------------------------------------------------------------------------------------------
      // Importation fichie DEB vers sas import
      //----------------------------------------------------------------------------------------------------------------------------------
      // Entr'es:
      //   - FileName_P : Nom du fichier PC . int'grer
      // Sorties:
      //   - Result_L   : Erreur Retourn'e, Vide si Ok
      //----------------------------------------------------------------------------------------------------------------------------------
      ToolsFile_G.LogInitFileExecute(FileName_P);
      IF NOT EXISTS(FileName_P) THEN BEGIN
        ToolsFile_G.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0002, FileName_P), 1, MAXSTRLEN(Result_L)));
      END ELSE BEGIN
        IF NOT FileImport_L.OPEN(FileName_P) THEN BEGIN
          ToolsFile_G.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0003, FileName_P, GETLASTERRORTEXT), 1, MAXSTRLEN(Result_L)));
          CLEARLASTERROR;
        END ELSE BEGIN
          FileImport_L.CREATEINSTREAM(FileInStream_L);
          IF NOT (FileInStream_L.EOS) THEN BEGIN
            // Passer la premiSre ligne
            NbCarLu_L := FileInStream_L.READTEXT(TexteLu_L);
            WHILE NOT (FileInStream_L.EOS) DO BEGIN
              EnregOK_L := TRUE;
              NbCarLu_L := FileInStream_L.READTEXT(TexteLu_L);
              CLEAR(TableSasImport_L);
              Pos_L := 1;
              // Soci't' : Non utilis'
              TextTemp_L := ToolsFile_G.ExtractTextCsv(TexteLu_L, Pos_L, ';', 30);
              // Date livraison : JJ/MM/AAAA
              Texte_L := DELCHR(ToolsFile_G.ExtractTextCsv(TexteLu_L, Pos_L, ';', 10), '=', '/');
              //-->20150910-INT05-CMC02.02.00-ADVITAM-CDL002
              // BEFORE
              {
              DateText_L := COPYSTR(Texte_L, 5, 4) + '-' + COPYSTR(Texte_L, 3, 2) + '-' + COPYSTR(Texte_L, 1, 2);
              IF DateText_L = '0000-00-00' THEN BEGIN
                TableSasImport_L."Posting Date" := 0D;
              END ELSE BEGIN
                IF NOT EVALUATE(TableSasImport_L."Posting Date", DateText_L, 9) THEN BEGIN
                  ToolsFile_G.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0006, TableSasImport_L.FIELDCAPTION("Posting Date"),
                    DateText_L, COPYSTR(TexteLu_L, 1, 12)), 1, MAXSTRLEN(Result_L)));
                END;
              END;
              }
              //<--20150910-INT05-CMC02.02.00-ADVITAM-CDL002

              // Date Facturation : JJ/MM/AAAA
              //-->20150910-INT05-CMC02.02.00-ADVITAM-CDL002
              // BEFORE
              {
              TextTemp_L := ToolsFile_G.ExtractTextCsv(TexteLu_L, Pos_L, ';', 10);
              }
              // AFTER
              Texte_L := DELCHR(ToolsFile_G.ExtractTextCsv(TexteLu_L, Pos_L, ';', 10), '=', '/');
              DateText_L := COPYSTR(Texte_L, 5, 4) + '-' + COPYSTR(Texte_L, 3, 2) + '-' + COPYSTR(Texte_L, 1, 2);
              IF DateText_L = '0000-00-00' THEN BEGIN
                TableSasImport_L."Posting Date" := 0D;
              END ELSE BEGIN
                IF NOT EVALUATE(TableSasImport_L."Posting Date", DateText_L, 9) THEN BEGIN
                  ToolsFile_G.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0006, TableSasImport_L.FIELDCAPTION("Posting Date"),
                    DateText_L, COPYSTR(TexteLu_L, 1, 12)), 1, MAXSTRLEN(Result_L)));
                END;
              END;
              //<--20150910-INT05-CMC02.02.00-ADVITAM-CDL002

              // Type document : Flux
              TextTemp_L := ToolsFile_G.ExtractTextCsv(TexteLu_L, Pos_L, ';', 30);
              CASE TextTemp_L OF
                'Introduction Facture': BEGIN
                  Coef_L := 1;
                  TableSasImport_L."Entry Type" := TableSasImport_L."Entry Type"::Purchase;
                END;
                'Introduction Avoir':BEGIN
                  Coef_L := 1;
                  TableSasImport_L."Entry Type" := TableSasImport_L."Entry Type"::Purchase;
                END;
                ELSE BEGIN
                  ToolsFile_G.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0001, TableSasImport_L.FIELDCAPTION("Entry Type"),
                    TextTemp_L, COPYSTR(TexteLu_L, 1, 12)), 1, MAXSTRLEN(Result_L)));
                END;
              END;
              // Article : NC8
              TextTemp_L := ToolsFile_G.ExtractTextCsv(TexteLu_L, Pos_L, ';', 30);
              TableItem_L.SETRANGE(TableItem_L."Tariff No.", TextTemp_L);
              IF NOT TableItem_L.FINDFIRST THEN BEGIN
                ToolsFile_G.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0008, TextTemp_L, COPYSTR(TexteLu_L, 1, 12)), 1, MAXSTRLEN(Result_L)));
              END ELSE BEGIN
                TableSasImport_L."Item No." := TableItem_L."No.";
              END;

              // Pays dest. prov.
              TextTemp_L := ToolsFile_G.ExtractTextCsv(TexteLu_L, Pos_L, ';', 30);
              // Valeur fiscale
              TextTemp_L := ToolsFile_G.ExtractTextCsv(TexteLu_L, Pos_L, ';', 30);
              IF TextTemp_L <> '' THEN BEGIN
                TextTemp_L := CONVERTSTR(TextTemp_L, '.', ',');
                IF NOT EVALUATE(Montant_L, TextTemp_L) THEN BEGIN
                  ToolsFile_G.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0006, TableSasImport_L.FIELDCAPTION("Posting Date"),
                    DateText_L, COPYSTR(TexteLu_L, 1, 12)), 1, MAXSTRLEN(Result_L)));
                END;
              END;
              CASE TableSasImport_L."Entry Type" OF
                TableSasImport_L."Entry Type"::Purchase: BEGIN
                  TableSasImport_L."Cost Amount (Actual)" := Montant_L;
                END;
                TableSasImport_L."Entry Type"::Sale: BEGIN
                  TableSasImport_L."Sales Amount (Actual)" := Montant_L;
                END;
              END;

              // Currency code.
              TextTemp_L := ToolsFile_G.ExtractTextCsv(TexteLu_L, Pos_L, ';', 30);

              // Rate.
              TextTemp_L := ToolsFile_G.ExtractTextCsv(TexteLu_L, Pos_L, ';', 30);

              // Montant
              TextTemp_L := ToolsFile_G.ExtractTextCsv(TexteLu_L, Pos_L, ';', 30);

              // R'gime
              TableSasImport_L."Transaction Specification" := ToolsFile_G.ExtractTextCsv(TexteLu_L, Pos_L, ';', 30);
              // Masse Nette
              TextTemp_L := ToolsFile_G.ExtractTextCsv(TexteLu_L, Pos_L, ';', 30);
              IF TextTemp_L <> '' THEN BEGIN
                TextTemp_L := CONVERTSTR(TextTemp_L, '.', ',');
                IF NOT EVALUATE(TableSasImport_L.Quantity, TextTemp_L) THEN BEGIN
                  ToolsFile_G.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0006, TableSasImport_L.FIELDCAPTION("Posting Date"),
                    DateText_L, COPYSTR(TexteLu_L, 1, 12)), 1, MAXSTRLEN(Result_L)));
                END;
              END;
              TableSasImport_L.Quantity := TableSasImport_L.Quantity * Coef_L;

              // PiSce
              TextTemp_L := ToolsFile_G.ExtractTextCsv(TexteLu_L, Pos_L, ';', 30);

              // Nature transaction
              TableSasImport_L."Transaction Type" := ToolsFile_G.ExtractTextCsv(TexteLu_L, Pos_L, ';', 30);
              // Condition livraison  : DEP / RDU
              ShipRadar_L := ToolsFile_G.ExtractTextCsv(TexteLu_L, Pos_L, ';', 30);
              Ship_L := MoteurInterface_G.Getcorrespondance(GlobalTemplateCode_G, 'Shipment Method', 'Code', ShipRadar_L, TRUE, TRUE);

              // Mode transport
              TableSasImport_L."Transport Method" := ToolsFile_G.ExtractTextCsv(TexteLu_L, Pos_L, ';', 30);
              // D'partement
              TableSasImport_L.Area := ToolsFile_G.ExtractTextCsv(TexteLu_L, Pos_L, ';', 30);
              CASE TableSasImport_L.Area OF
                'AIR': TableSasImport_L.Area := '62';
                'NVL': TableSasImport_L.Area := '59';
                'UAB': TableSasImport_L.Area := '62';
              END;
              // Pays d'origine
              TableSasImport_L."Country/Region Code" := ToolsFile_G.ExtractTextCsv(TexteLu_L, Pos_L, ';', 30);

              // Num'ro d'identification de l'acqu'reur C.E. : AVEC Pays : Ne pas prendre
              TextTemp_L := ToolsFile_G.ExtractTextCsv(TexteLu_L, Pos_L, ';', 30);
              IF STRLEN(TextTemp_L) > MAXSTRLEN(TableSasImport_L."VAT Registration No.") THEN BEGIN
                ToolsFile_G.LogWriteInfo(COPYSTR(STRSUBSTNO(Text0009, TableSasImport_L.FIELDCAPTION("VAT Registration No."),
                  TextTemp_L, MAXSTRLEN(TableSasImport_L."VAT Registration No."), COPYSTR(TexteLu_L, 1, 12)),
                  1, MAXSTRLEN(Result_L)));
                CLEAR(TextTemp_L);
              END;
              TableSasImport_L."VAT Registration No." := COPYSTR(TextTemp_L, 1, MAXSTRLEN(TableSasImport_L."VAT Registration No."));

              // Facture
              TableSasImport_L."Document No." := ToolsFile_G.ExtractTextCsv(TexteLu_L, Pos_L, ';', 30);

              IF Ship_L = '' THEN BEGIN
                ToolsFile_G.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0011, TableSasImport_L.FIELDCAPTION("Shipment Method Code"),
                  ShipRadar_L, TableSasImport_L."Entry No.", TableSasImport_L."Document No.") ,1, MAXSTRLEN(Result_L)));
              END;
              TableSasImport_L."Shipment Method Code" := Ship_L;

              // Ligne facture
              TextTemp_L := ToolsFile_G.ExtractTextCsv(TexteLu_L, Pos_L, ';', 30);

              // Num'ro d'identification de l'acqu'reur C.E. (Sans Pays)
              TextTemp_L := ToolsFile_G.ExtractTextCsv(TexteLu_L, Pos_L, ';', 30);

              CLEAR(TableSasImport_L.HoroIntegration);
              IF EnregOK_L THEN BEGIN
                IF NOT TableSasImport_L.INSERT(TRUE) THEN BEGIN
                  ToolsFile_G.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0007, FORMAT(TableSasImport_L."Entry No.")),
                    1, MAXSTRLEN(Result_L)));
                END;
              END;
            END;
          END;
          FileImport_L.CLOSE();
        END;

        // Pour g'rer le ROLLBACK.
        FileLogName_L := ToolsFile_G.LogName;
        ToolsFile_G.LogClose();

        IF ToolsFile_G.LogErreurType() = 'E' THEN BEGIN
          ERROR('ERREUR ' + FileLogName_L);
        END ELSE BEGIN
          IF NOT ToolsFile_G.FileRenameSav(FileName_P, 'Sav', TRUE) THEN BEGIN
            ToolsFile_G.LogWriteErreur(STRSUBSTNO(Text0004, FileName_P));
          END;
          IF ToolsFile_G.LogErreurType() = 'I' THEN BEGIN
            MoteurInterface_G.InsertLog(CodeUnit_G, FileLogName_L);
          END ELSE BEGIN
            MoteurInterface_G.InsertOk(CodeUnit_G);
          END;
        END;
      END;
    END;

    BEGIN
    {
      //--> 3LI BEGIN
      20150617-INT05-CMC02.02.00-ADVITAM-CDL001 : Import DEB externe.
      20150910-INT05-CMC02.02.00-ADVITAM-CDL002 : - Prendre la date de facture (au lieu de la date de BL).
      "Source No." ==> Vide, Client, Fournisseur
      //<-- 3LI BEGIN
    }
    END.
  }
