      SalesLine_L.SETRANGE(SalesLine_L."Header Status",SalesLine_L."Header Status"::Released);

      SalesLine_L.SETFILTER(SalesLine_L."No.",ItemNo_P);
      SalesLine_L.SETFILTER(SalesLine_L."Outstanding Quantity",'>%1',0);
      IF SalesLine_L.FINDFIRST THEN
        REPEAT
         IF (SalesHeader_L.GET(SalesLine_L."Document Type",SalesLine_L."Document No.")) AND
           (SalesHeader_L."CMC Campaign No."=CampaignNo_P)

    PROCEDURE FinishEntry@8053052(SalesForecastCode_P@8053000 : Code[20];Type_P@8053001 : 'Recipient,Salesperson';Code_P@8053002 : Code[20]);
    VAR
      SFSalesperson_L@8053003 : Record 8053205;
      SFRecipient_L@8053004 : Record 8053203;
      WarnMessage_L@8053005 : Text[250];
      Text8053000@8053006 : TextConst 'FRA=Voulez-vous terminer la saisie de la pr'vision %1 ?';
      Text8053001@8053007 : TextConst 'FRA=Voulez-vous recommencer la saisie de la pr'vision %1 ?';
    BEGIN
      CASE Type_P OF
        Type_P::Recipient:
        BEGIN
          SFRecipient_L.GET(SalesForecastCode_P,Code_P);
          IF SFRecipient_L.Finished = FALSE THEN
            WarnMessage_L := Text8053000
          ELSE
            WarnMessage_L := Text8053001;

          IF CONFIRM(WarnMessage_L,FALSE,SalesForecastCode_P) THEN BEGIN
            SFRecipient_L.Finished := NOT SFRecipient_L.Finished;
            SFRecipient_L.MODIFY;
          END;
        END;
        Type_P::Salesperson:
        BEGIN
          SFSalesperson_L.GET(SalesForecastCode_P,Code_P);
          IF SFSalesperson_L.Finished = FALSE THEN
            WarnMessage_L := Text8053000
          ELSE
            WarnMessage_L := Text8053001;

          IF CONFIRM(WarnMessage_L,FALSE,SalesForecastCode_P,Code_P) THEN BEGIN
            SFSalesperson_L.Finished := NOT SFSalesperson_L.Finished;
            SFSalesperson_L.MODIFY;
          END;
        END;
      END;
    END;

    PROCEDURE CompleteCalculOnOneRec@1000000004(SalesQuota_P@1000000001 : Record 8053260;VAR SalesLine_P@1000000000 : Record 37) : Decimal;
    VAR
      DetSalesLine_L@1000000007 : Record 8053475;
      ActualQty_L@1000000006 : Decimal;
      ItemTransEntry_L@1000000005 : Record 8053336;
      Salesperson_L@1000000004 : Record 13;
      Item_L@1000000003 : Record 27;
      CMCSalesArea_L@1000000002 : Record 8053156;
      UOMConversionRate_L@1000000008 : Decimal;
    BEGIN
      //------------------------------------------------------------------------------------------------
      // Cette fonction complete le calcul qt' quota quand on controle . la saisie et non au lancement
      // Son but est de d'falquer du calcul "standard" la quantit' possiblement existante dans les lignes
      // d'taill'es correspondant . la commande quand on la r'ouvert aprSs lancement
      // mais on doit aussi ajout' la quantit' de la ligne en train d'^tre saisie
      // Input:
      //   - SalesLine_P
      //------------------------------------------------------------------------------------------------


      // --> 20170629-WI00-ADV01.01.00-2821-CJY01
      WITH DetSalesLine_L DO BEGIN
        RESET;
        SETCURRENTKEY("Document Type",Type,"Prod. Activity Code","CMC Campaign No.","Order Date","No.");
        SETRANGE("Document Type","Document Type"::Order);
        SETRANGE("Document No.",SalesLine_P."Document No.");
        SETRANGE("Line No.",SalesLine_P."Line No.");
        SETRANGE(Type,Type::Item);
        SETRANGE("Prod. Activity Code",SalesQuota_P."Production Activity Code");
        SETRANGE("CMC Campaign No.",SalesQuota_P."Campaign No.");
        SETFILTER("Order Date",'>=%1&<=%2',SalesQuota_P."Starting Date",SalesQuota_P."Ending Date");
        CASE SalesQuota_P.Type OF
          SalesQuota_P.Type::Item :
          BEGIN
            SETRANGE("No.",SalesQuota_P."No.");
            IF SalesQuota_P."Variant Code" <> '' THEN
              SETRANGE("Variant Code",SalesQuota_P."Variant Code");
          END;
          SalesQuota_P.Type::Variety : SETRANGE("Variety Code",SalesQuota_P."No.");
          SalesQuota_P.Type::Submarket : SETRANGE("Submarket Code",SalesQuota_P."No.");
        END;
        CASE SalesQuota_P."Sales Type" OF
          SalesQuota_P."Sales Type"::Salesperson:
            SETRANGE("Salesperson Code",SalesQuota_P."Salesperson Code");
          SalesQuota_P."Sales Type"::"Sales Zone":
            SETRANGE("Sales Zone Code",SalesQuota_P."Sales Zone Code");
          SalesQuota_P."Sales Type"::"Sales Area":
            SETRANGE("Sales Area Code",SalesQuota_P."Sales Area Code");
        END;
        IF SalesQuota_P."Doc. Category Code" <> '' THEN
          SETRANGE("Doc. Category Code",SalesQuota_P."Doc. Category Code");
        CMCSalesArea_L.SETRANGE(Code,SalesQuota_P."Sales Area Code");
        CMCSalesArea_L.SETRANGE("Excluded from society quota",TRUE);
        IF CMCSalesArea_L.FINDFIRST THEN
          SETRANGE("Excluded from society quota",TRUE)
        ELSE
          SETRANGE("Excluded from society quota",FALSE);
        CALCSUMS("Outstanding Quantity");
        ActualQty_L := -"Outstanding Quantity" + SalesLine_P.Quantity;
        // --> 20190612-WI00-ADV01.01.00-3425-CJY01
        IF SalesQuota_P.Type = SalesQuota_P.Type::Variety THEN BEGIN
          //si l'on gSre un quotas ". la vari't'" on doit convertir avec l'unit' de Prevision
          UOMConversionRate_L := ConvertVarietyUnit(SalesQuota_P."No.",SalesLine_P."No.");
          ActualQty_L := ActualQty_L /UOMConversionRate_L;
        END;
        // <-- 20190612-WI00-ADV01.01.00-3425-CJY01
      END;

      EXIT(ActualQty_L);
    END;
