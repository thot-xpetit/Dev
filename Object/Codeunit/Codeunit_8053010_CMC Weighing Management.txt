    PROCEDURE SplitWeighingFromWhseShptLine@8053031(RecWhseShptLine_P@8053000 : Record 7321;Undo_P@8053006 : Boolean);
    VAR
      RecWeighingWeighbridge_L@8053001 : Record 8053094;
      RecWhseShptLine_L@8053002 : Record 7321;
      RecWhseShptLine2_L@8053004 : Record 7321;
      QtyToSplit_L@8053003 : Decimal;
      CounterSplit_L@8053005 : Integer;
      FoundSplit_L@8053007 : Boolean;
    BEGIN
      //--------------------------------------------------------------------------------------------------------
      // Eclate une pes'e sur n lignes d'exp'dition entrep"t
      // Appel dans P7336
      // Entr'es:
      //   - RecWhseShptLine_P : Record Ligne exp'dition entrep"t
      //   - Undo_P : TRUE si annulation de l''clatement
      // Sorties :
      //   -
      //--------------------------------------------------------------------------------------------------------
      WITH RecWhseShptLine_P DO BEGIN
        LOCKTABLE;
        TESTFIELD("Weighing No.");
        TESTFIELD("Posted Weighing",FALSE);
        IF Undo_P THEN BEGIN
          //En cas d'annulation, on se repositionne sur la ligne . l'origine de l''clatement
          IF "Weighing Source Line No." <> 0 THEN
            GET("No.","Weighing Source Line No.");
        END ELSE
          TESTFIELD("Weighing Source Line No.",0);

        //Recherche des lignes qui ont d'j. fait l'objet d'un 'clatement de la pes'e
        RecWhseShptLine_L.SETRANGE("No.","No.");
        RecWhseShptLine_L.SETFILTER("Line No.",'<>%1',"Line No.");
        RecWhseShptLine_L.SETRANGE("Item No.","Item No.");
        RecWhseShptLine_L.SETRANGE("Variant Code","Variant Code");
        RecWhseShptLine_L.SETRANGE("Weighing No.","Weighing No.");
        IF RecWhseShptLine_L.FINDSET THEN BEGIN
          REPEAT
            RecWhseShptLine2_L.GET(RecWhseShptLine_L."No.",RecWhseShptLine_L."Line No.");
            RecWhseShptLine2_L.VALIDATE("Qty. to Ship",0);
            RecWhseShptLine2_L."Weighing No." := 0;
            RecWhseShptLine2_L."Weighing Source Line No." := 0;
            RecWhseShptLine2_L.MODIFY;
          UNTIL RecWhseShptLine_L.NEXT = 0;
          FoundSplit_L := TRUE;
        END;

        //Eclatement de la pes'e
        //Recherche de la pes'e et du poids net . ventiler
        RecWeighingWeighbridge_L.GET("Weighing No.");
        IF NOT Undo_P THEN BEGIN
          IF RecWeighingWeighbridge_L."Net Weight" > Quantity THEN BEGIN
            VALIDATE("Qty. to Ship",Quantity);
            MODIFY;
            QtyToSplit_L := RecWeighingWeighbridge_L."Net Weight" - Quantity;
            RecWhseShptLine_L.SETRANGE("No.","No.");
            RecWhseShptLine_L.SETFILTER("Line No.",'<>%1',"Line No.");
            RecWhseShptLine_L.SETRANGE("Item No.","Item No.");
            RecWhseShptLine_L.SETRANGE("Variant Code","Variant Code");
            RecWhseShptLine_L.SETRANGE("Weighing No.",0);
            IF RecWhseShptLine_L.FINDSET THEN BEGIN
              REPEAT
                IF QtyToSplit_L > RecWhseShptLine_L.Quantity THEN BEGIN
                  RecWhseShptLine_L.VALIDATE("Qty. to Ship",RecWhseShptLine_L.Quantity);
                  QtyToSplit_L -= RecWhseShptLine_L.Quantity;
                END ELSE BEGIN
                  RecWhseShptLine_L.VALIDATE("Qty. to Ship",QtyToSplit_L);
                  QtyToSplit_L := 0;
                END;
                RecWhseShptLine_L."Weighing No." := "Weighing No.";
                RecWhseShptLine_L."Weighing Source Line No." := "Line No.";
                RecWhseShptLine_L.MODIFY;
                CounterSplit_L += 1;
              UNTIL (RecWhseShptLine_L.NEXT = 0) OR (QtyToSplit_L = 0);
              IF QtyToSplit_L > 0 THEN BEGIN
                RecWhseShptLine_L.NEXT(-1);
                RecWhseShptLine_L.VALIDATE("Qty. to Ship",RecWhseShptLine_L.Quantity + QtyToSplit_L);
                RecWhseShptLine_L.MODIFY;
              END;
              MESSAGE(Text8053013,"Weighing No.",CounterSplit_L + 1);
            END;
          END;
        END ELSE BEGIN
          IF FoundSplit_L THEN BEGIN
            VALIDATE("Qty. to Ship",RecWeighingWeighbridge_L."Net Weight");
            MODIFY;
            MESSAGE(Text8053014,"Weighing No.");
          END;
        END;
      END;
    END;
