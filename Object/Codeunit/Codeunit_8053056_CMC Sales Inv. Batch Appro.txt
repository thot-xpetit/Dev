    PROCEDURE SearchDueDate@8053005(recHeader_P@8053000 : Record 36;recLine_P@8053001 : Record 37) Result_L : Date;
    BEGIN
      //----------------------------------------------------------------------------------------------------------------------------------
      // Calcul la date d''ch'ance en fonction des paramStres de l'utilisateur
      //----------------------------------------------------------------------------------------------------------------------------------
      //  Entr'es
      //    - recHeader_P : Ent^te T36
      //    - recLine_P : Line T37
      //  Sorties
      //    - Result_L : Date d''ch'ance
      //----------------------------------------------------------------------------------------------------------------------------------
      Result_L := 0D;
      IF (TypeDueDate_G = TypeDueDate_G::Commande) AND (recHeader_P."Fixed Due Date" = FALSE) THEN BEGIN
        IF recLine_P."Due Date" = 0D THEN BEGIN
          //Pas de due date sur la ligne
          IF recHeader_P."Due Date" = 0D THEN BEGIN
            //Pas de due date sur l'ent^te
            Result_L := BeginDueDateReq_G;
          END ELSE BEGIN
            //Due date . l'ent^te
            IF recHeader_P."Due Date" <= EndingDueDateReq_G THEN BEGIN
              //Utilisation de la due date de l'ent^te
              IF recHeader_P."Due Date" >= BeginDueDateReq_G THEN BEGIN
                Result_L := recHeader_P."Due Date";
              END ELSE BEGIN
                //la date d''ch'ance est pass'e alors on utilise la date d''ch'ance min
                Result_L := BeginDueDateReq_G;
              END;
              Result_L := recHeader_P."Due Date";
            END ELSE BEGIN
              //L'ent^te de l'ent^te n'est pas dans le range
              Result_L := 0D;
            END;
          END;
        END ELSE BEGIN
          //Due date . la ligne
          IF recLine_P."Due Date" <= EndingDueDateReq_G THEN BEGIN
            //Utilisation de la due date de la ligne
            IF recLine_P."Due Date" >= BeginDueDateReq_G THEN BEGIN
              Result_L := recLine_P."Due Date";
            END ELSE BEGIN
              //la date d''ch'ance est pass'e alors on utilise la date d''ch'ance min
              Result_L := BeginDueDateReq_G;
            END;
          END ELSE BEGIN
            //L'ent^te de la ligne n'est pas dans le range
            Result_L := 0D;
          END;
        END;
      END;

      IF recHeader_P."Fixed Due Date" = TRUE THEN BEGIN
        IF recHeader_P."Due Date" <= EndingDueDateReq_G THEN
          Result_L := recHeader_P."Due Date"
        ELSE
          Result_L := 0D;
      END;

      EXIT(Result_L);
    END;

    PROCEDURE CalcAmount@8053017(recHeader_P@8053000 : Record 36;DocumentType_P@8053001 : 'Invoice,Credit Memo';DueDate_P@1000000000 : Date) Result_L : Decimal;
    VAR
      SalesHeader_L@8053002 : Record 36;
      TotalSalesLine_L@8053009 : Record 37;
      TotalSalesLineLCY_L@8053008 : Record 37;
      Cust_L@8053007 : Record 18;
      TotalAmount2_L@8053006 : Decimal;
      SalesLine_L@8053005 : Record 37;
      TempSalesLine_L@8053004 : TEMPORARY Record 37;
      SalesPost_L@8053003 : Codeunit 80;
      TempVATAmountLine1_L@8053010 : TEMPORARY Record 290;
      VATAmount_L@8053011 : Decimal;
      VATAmountText_L@8053012 : Text[30];
      ProfitLCY_L@8053017 : Decimal;
      ProfitPct_L@8053016 : Decimal;
      AdjProfitLCY_L@8053015 : Decimal;
      AdjProfitPct_L@8053014 : Decimal;
      TotalAdjCostLCY_L@8053013 : Decimal;
      Amount_L@8053018 : Decimal;
    BEGIN
      //----------------------------------------------------------------------------------------------------------------------------------
      // Calcule du Montant TOTAL TTC pour le client du document en paramStre
      //----------------------------------------------------------------------------------------------------------------------------------
      //  Entr'es
      //    - recHeader_P : Record T36 (facture ou avoir)
      //    - DocumentType_P : Option type document
      // --> 20150930-Wi00-ADV01.01.00-2265-FHS01
      //    - DueDate_P : Date d''ch'ance
      // <-- 20150930-Wi00-ADV01.01.00-2265-FHS01
      //  Sorties
      //    - la fonction retourne une valeur decimal
      //----------------------------------------------------------------------------------------------------------------------------------
      SalesHeader_L.RESET;
      SalesHeader_L.SETRANGE("Bill-to Customer No.", recHeader_P."Bill-to Customer No.");
      IF DocumentType_P = DocumentType_P::Invoice THEN BEGIN
        SalesHeader_L.SETRANGE("Document Type", SalesHeader_L."Document Type"::Invoice);
        SalesHeader_L.SETFILTER("No.",'<%1', recHeader_P."No.");
      END;
      IF DocumentType_P = DocumentType_P::"Credit Memo" THEN
        SalesHeader_L.SETRANGE("Document Type", SalesHeader_L."Document Type"::"Credit Memo");
      // --> 20150930-Wi00-ADV01.01.00-2265-FHS01
      SalesHeader_L.SETFILTER("Due Date", '<=%1', DueDate_P);
      // <-- 20150930-Wi00-ADV01.01.00-2265-FHS01
      CLEAR(Amount_L);
      IF SalesHeader_L.FINDSET(FALSE, FALSE) THEN BEGIN
        REPEAT
          CLEAR(TotalAmount2_L);
          CLEAR(SalesLine_L);
          CLEAR(TotalSalesLine_L);
          CLEAR(TotalSalesLineLCY_L);
          TempSalesLine_L.DELETEALL;
          CLEAR(TempSalesLine_L);
          CLEAR(SalesPost_L);
          SalesPost_L.GetSalesLines(SalesHeader_L, TempSalesLine_L,0);
          CLEAR(SalesPost_L);
          SalesLine_L.CalcVATAmountLines(0,SalesHeader_L, TempSalesLine_L, TempVATAmountLine1_L);
          SalesPost_L.SumSalesLinesTemp(SalesHeader_L, TempSalesLine_L, 0, TotalSalesLine_L, TotalSalesLineLCY_L,
            VATAmount_L, VATAmountText_L, ProfitLCY_L, ProfitPct_L, TotalAdjCostLCY_L);
          TotalAdjCostLCY_L := TotalSalesLineLCY_L."Unit Cost (LCY)";
          AdjProfitLCY_L := TotalSalesLineLCY_L.Amount - TotalAdjCostLCY_L;
          IF TotalSalesLineLCY_L.Amount <> 0 THEN
            AdjProfitPct_L := ROUND(AdjProfitLCY_L / TotalSalesLineLCY_L.Amount * 100, 0.1);
          IF SalesHeader_L."Prices Including VAT" THEN
            TotalAmount2_L := TotalSalesLine_L.Amount
          ELSE
            TotalAmount2_L := TotalSalesLine_L."Amount Including VAT";
          Amount_L += TotalAmount2_L;
        UNTIL SalesHeader_L.NEXT=0;
      END;

      Result_L := Amount_L;

      EXIT(Result_L);
    END;
