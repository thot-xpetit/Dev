OBJECT Codeunit 8087782 CMC OET Export Item
{
  OBJECT-PROPERTIES
  {
    Date=23/07/15;
    Heure=00:00:00;
    Liste versions=CMC02.02.00;
  }
  PROPERTIES
  {
    TableNo=8061230;
    OnRun=BEGIN
            //--> 20140317-INT19-CMC02.02.00-ADVITAM-GMX
            CodeUnit_G := Processing;
            IF "Manual Process" THEN BEGIN
              ExportItem("File in use");
            END ELSE BEGIN
              ExportItem('');
            END;
          END;

  }
  CODE
  {
    VAR
      "<-- 3LI GLOBALS -->"@1000000001 : Integer;
      CodeUnit_G@1000000000 : Integer;
      "<-- 3LI TEXT GLOBALS -->"@1000000003 : TextConst;
      Text0001@1000000002 : TextConst 'ENU=Error Table "%1" doesn''t exist \Company : %2;FRA=Erreur Table "%1" inexistante \Societe : "%2"';
      Text0003@1000000006 : TextConst 'ENU=Error Creating File %1 \%2.;FRA=Erreur cr'ation fichier %1 \%2.';
      Text0004@1000000005 : TextConst 'ENU=Error updating table %1 \Company : %2;FRA=Erreur modification table "%1" \Societe : %2';
      Text0007@1000000007 : TextConst 'ENU=Error Correspondence "%1" : "%2" not parametrized for "%3";FRA=Erreur correspondance "%1" : "%2" non param'tr'e pour "%3".';
      Text0011@1000000004 : TextConst 'ENU=Error backup copy.;FRA=Erreur copie de sauvegarde.';
      Text0012@1000000008 : TextConst 'ENU=Error backup copy.;FRA=Erreur copie "%1" vers "%2".';

    PROCEDURE "<-- 3LI FUNCTIONS -->"@1000000000();
    BEGIN
    END;

    PROCEDURE ExportItem@1000000001(FileName_P@1000000000 : Text[128]) Result_L : Text[256];
    VAR
      "<-- 3LI LOCALS -->"@1000000001 : Integer;
      TableParameter_L@1000000017 : Record 8053431;
      TableParameterMaj_L@1000000028 : Record 8053431;
      TableItem_L@1000000029 : Record 27;
      MoteurInterface_L@1000000004 : Codeunit 8061230;
      ToolsFile_L@1000000010 : Codeunit 8053405;
      FormatFieldsManage_L@1000000002 : Codeunit 8060985;
      FileNameOuvert_L@1000000036 : Boolean;
      NbCarEcrit_L@1000000003 : Integer;
      ConnectorName_L@1000000009 : Code[20];
      ImpExpFolder_L@1000000008 : Text[256];
      LogFolder_L@1000000007 : Text[256];
      ExtensionFile_L@1000000006 : Text[10];
      PrefixFile_L@1000000005 : Text[10];
      EreConnectorName_L@1000000043 : Code[20];
      EreImpExpFolder_L@1000000042 : Text[256];
      EreLogFolder_L@1000000041 : Text[256];
      EreExtensionFile_L@1000000040 : Text[10];
      ErePrefixFile_L@1000000039 : Text[10];
      ItemTemplateCode_L@1000000016 : Code[20];
      CustomerTemplateCode_L@1000000015 : Code[20];
      VendorTemplateCode_L@1000000014 : Code[20];
      ContactTemplateCode_L@1000000013 : Code[20];
      AccountTemplateCode_L@1000000012 : Code[20];
      GlobalTemplateCode_L@1000000011 : Code[20];
      FileTemp_L@1000000018 : Text[256];
      FileName_L@1000000019 : Text[256];
      EreFileName_L@1000000044 : Text[256];
      FileLogName_L@1000000020 : Text[256];
      DateTimeStart_L@1000000021 : DateTime;
      LastDateTrf_L@1000000022 : Date;
      FileNumber_L@1000000023 : Integer;
      FilePath_L@1000000025 : Text[256];
      FileN_L@1000000024 : Text[256];
      Boucle_L@1000000026 : Integer;
      FileExport_L@1000000027 : File;
      FileExportOutstream_L@1000000031 : OutStream;
      Filtre_L@1000000030 : Text[1024];
      ArticleOet_L@1000000032 : Text[30];
      ArticleTecOet_L@1000000034 : Text[30];
      ArticleSacOet_L@1000000035 : Text[30];
      CodeConditionnement_L@1000000033 : Text[2];
      FromCar_L@1000000037 : Text[1];
      ToCar_L@1000000038 : Text[1];
    BEGIN
      //----------------------------------------------------------------------------------------------------------------------------------
      // Extraction des Articles  table "Item"   pour Cr'ation du fichier OET :  AZnnnnnnnn.dat
      //----------------------------------------------------------------------------------------------------------------------------------
      // Entr'es:
      //   - FileName_P  : Nom du fichier PC . G'n'rer
      // Sorties:
      //   - Result_L    : Erreur Retourn'e, Vide si Ok
      //----------------------------------------------------------------------------------------------------------------------------------
      FileNameOuvert_L := FALSE;
      NbCarEcrit_L := 0;
      MoteurInterface_L.GetParameter(CodeUnit_G, ConnectorName_L, ImpExpFolder_L, LogFolder_L, ExtensionFile_L, PrefixFile_L);
      MoteurInterface_L.GetTemplatesCode (ConnectorName_L, ItemTemplateCode_L, CustomerTemplateCode_L, VendorTemplateCode_L,
        ContactTemplateCode_L, AccountTemplateCode_L, GlobalTemplateCode_L);
      ToolsFile_L.LogInit(LogFolder_L, 'OET');
      ToolsFile_L.LogInitCodeUnit(CodeUnit_G);

      IF FileName_P = '' THEN BEGIN
        IF (NOT TableParameter_L.FINDFIRST) THEN BEGIN
          ToolsFile_L.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0001, TableParameter_L.TABLECAPTION, COMPANYNAME), 1, MAXSTRLEN(Result_L)));
        END;
      END;

      IF ToolsFile_L.LogErreurType() <> 'E' THEN BEGIN
        IF NOT TableParameter_L.FINDFIRST THEN BEGIN
          ToolsFile_L.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0001, TableParameter_L.TABLECAPTION , COMPANYNAME), 1, MAXSTRLEN(Result_L)));
        END ELSE BEGIN
          IF TableParameter_L."Export Item" THEN BEGIN
            DateTimeStart_L := CURRENTDATETIME;
            LastDateTrf_L := DT2DATE(TableParameter_L."Last Date Export Item");
            IF FileName_P <> '' THEN BEGIN
              FileName_L := FileName_P;
            END ELSE BEGIN
              FileNumber_L := TableParameter_L."Last Item File Number" + 1;
              FileName_L := ImpExpFolder_L + '\' + DELCHR(PrefixFile_L, '<>', ' ') +
                CONVERTSTR(FORMAT(FileNumber_L, 8), ' ', '0') + '.' + DELCHR(ExtensionFile_L, '<>', ' ');
            END;
            ToolsFile_L.LogInitFileExecute(FileName_L);
            ToolsFile_L.SplitFilename(FileName_L, FilePath_L, FileN_L);
            FileTemp_L := FilePath_L + 'Temp_' + FileN_L;
            Boucle_L := 0;

            REPEAT
              TableItem_L.RESET;
              IF Boucle_L = 0 THEN BEGIN
                TableItem_L.SETRANGE("MC Last Date Modified");
                TableItem_L.SETFILTER("Last Date Modified", '>= %1', LastDateTrf_L);
              END ELSE BEGIN
                TableItem_L.SETRANGE("Last Date Modified");
                TableItem_L.SETFILTER("MC Last Date Modified", '>= %1', LastDateTrf_L);
              END;
              IF TableParameter_L."Activity Code" = '' THEN BEGIN
                TableItem_L.SETRANGE("Activity Code");
              END ELSE BEGIN
                TableItem_L.SETFILTER("Activity Code", TableParameter_L."Activity Code");
              END;
              IF TableParameter_L.ExportItemExcludeFamilies = '' THEN BEGIN
                Filtre_L := '';
              END ELSE BEGIN
                Filtre_L := '<> ' + ToolsFile_L.TexteRemplaceCar(TableParameter_L.ExportItemExcludeFamilies, '&', ' & <> ');
              END;
              TableItem_L.SETFILTER(TableItem_L."Family Code", Filtre_L);

              IF TableItem_L.FINDSET THEN BEGIN
                IF NOT FileNameOuvert_L THEN BEGIN
                  IF NOT FileExport_L.CREATE(FileTemp_L) THEN BEGIN
                    ToolsFile_L.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0003, FileTemp_L, GETLASTERRORTEXT), 1, MAXSTRLEN(Result_L)));
                    CLEARLASTERROR;
                  END ELSE BEGIN
                    FileExport_L.CREATEOUTSTREAM(FileExportOutstream_L);
                    FileNameOuvert_L := TRUE;
                  END;
                END;
                IF FileNameOuvert_L THEN BEGIN
                  REPEAT
                    IF NOT TableItem_L.Blocked THEN BEGIN
                      ArticleOet_L := MoteurInterface_L.GetReferentialCorrespondance(ItemTemplateCode_L, FALSE, TableItem_L."No.",
                        TRUE);
                      IF ArticleOet_L <> '$' THEN BEGIN
                        IF ArticleOet_L = '' THEN BEGIN
                          ToolsFile_L.LogWriteInfo(COPYSTR(STRSUBSTNO(Text0007, TableItem_L.FIELDCAPTION(TableItem_L."No."),
                            ArticleOet_L, TableItem_L."No."), 1, MAXSTRLEN(Result_L)));
                        END ELSE BEGIN
                          FileExportOutstream_L.WRITETEXT('ART', 3);
      // TODO-001 : Pr'voir code 'S' pour suppression article
                          FileExportOutstream_L.WRITETEXT('M', 1);
                          FileExportOutstream_L.WRITETEXT(PADSTR(ArticleOet_L, 8, ' '), 8);
                          // Conversion du symbole Euro
                          FromCar_L := 'ø';
                          ToCar_L := 'E';
                          FileExportOutstream_L.WRITETEXT(PADSTR(FormatFieldsManage_L.Ascii2Ansi(
                            CONVERTSTR(TableItem_L.Description, FromCar_L, ToCar_L)), 35, ' '), 35);
                          // Type d'article ('SAC' 'MP' 'PF' 'DIV')
                          IF (TableItem_L."Family Code" = 'SACO') AND (TableItem_L."Sub-Family Code" = 'SAC') THEN BEGIN
                            FileExportOutstream_L.WRITETEXT('SAC', 3);
                          END ELSE BEGIN
                            IF TableItem_L."Replenishment System" = TableItem_L."Replenishment System"::Purchase THEN BEGIN
                              FileExportOutstream_L.WRITETEXT('MP ', 3);
                            END ELSE BEGIN
                              IF TableItem_L."Replenishment System" = TableItem_L."Replenishment System"::"Prod. Order" THEN BEGIN
                                FileExportOutstream_L.WRITETEXT('PF ', 3);
                              END ELSE BEGIN
                                FileExportOutstream_L.WRITETEXT('DIV', 3);
                              END;
                            END;
                          END;
                          // Code pr'sentation ('FA' 'GR' 'MI' 'VE')
                          IF (TableItem_L."Product Presentation Code" = 'FA') OR (TableItem_L."Product Presentation Code" = 'GR') OR
                             (TableItem_L."Product Presentation Code" = 'MI') OR (TableItem_L."Product Presentation Code" = 'VE')
                             THEN BEGIN
                            FileExportOutstream_L.WRITETEXT(PADSTR(TableItem_L."Product Presentation Code", 2), 2);
                          END ELSE BEGIN
                            FileExportOutstream_L.WRITETEXT('FA', 2);
                          END;
                          // Conditionnement ('VR' 'BB' 'SA')
                          CodeConditionnement_L := MoteurInterface_L.Getcorrespondance(GlobalTemplateCode_L, 'CMC Packaging Type',
                            'Code',TableItem_L."Packaging Type", FALSE, TRUE);
                          IF CodeConditionnement_L = '' THEN BEGIN
                            ToolsFile_L.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0007,
                              TableItem_L.FIELDCAPTION(TableItem_L."Packaging Type"), TableItem_L."Packaging Type",
                              TableItem_L."No."), 1, MAXSTRLEN(Result_L)));
                          END;
                          FileExportOutstream_L.WRITETEXT(PADSTR(CodeConditionnement_L, 2), 2);
                          // Article technique
                          CLEAR(ArticleTecOet_L);
                          IF TableItem_L."Technical Product Code" <> '' THEN BEGIN
                            ArticleTecOet_L := MoteurInterface_L.GetReferentialCorrespondance(ItemTemplateCode_L, FALSE,
                              TableItem_L."Technical Product Code", TRUE);
                            IF ArticleTecOet_L = '' THEN BEGIN
                              ToolsFile_L.LogWriteInfo(COPYSTR(STRSUBSTNO(Text0007,
                                TableItem_L.FIELDCAPTION(TableItem_L."Technical Product Code"),
                                TableItem_L."Technical Product Code", TableItem_L."No."), 1, MAXSTRLEN(Result_L)));
                            END;
                          END;
                          IF ArticleTecOet_L = '' THEN BEGIN
                            ArticleTecOet_L := ArticleOet_L;
                          END;
                          FileExportOutstream_L.WRITETEXT(PADSTR(ArticleTecOet_L, 8), 8);
                          // Article sacherie
                          CLEAR(ArticleSacOet_L);
                          IF TableItem_L."Bagging Item No." <> '' THEN BEGIN
                            ArticleSacOet_L := MoteurInterface_L.GetReferentialCorrespondance(ItemTemplateCode_L, FALSE,
                              TableItem_L."Bagging Item No.", TRUE);
                            IF ArticleSacOet_L = '' THEN BEGIN
                              ToolsFile_L.LogWriteInfo(COPYSTR(STRSUBSTNO(Text0007,
                                TableItem_L.FIELDCAPTION(TableItem_L."Bagging Item No."),
                                TableItem_L."Bagging Item No.", TableItem_L."No."), 1, MAXSTRLEN(Result_L)));
                            END;
                          END;
                          FileExportOutstream_L.WRITETEXT(PADSTR(ArticleSacOet_L, 8), 8);
                          // Poids unitaire au Kg pour Sac et BigBag
                          FileExportOutstream_L.WRITETEXT(CONVERTSTR(FORMAT(TableItem_L."Gross Weight",
                            0, '<Filler Character,0><Precision,0><Sign><Integer,5><Decimals>'), ' ', '0'), 5);
                          // D'claration statistique
                          FileExportOutstream_L.WRITETEXT(PADSTR(TableItem_L."French COOP Stat. Code", 2), 2);
                          // Toxicologie
                          FileExportOutstream_L.WRITETEXT(PADSTR(TableItem_L."Security Labels", 3), 3);
      // TODO-002 D'claration statistique - Toxicologie - Environnemental - Code ONU
                          // Environnemental
                          FileExportOutstream_L.WRITETEXT('   ', 3);
                          // Code ONU
                          FileExportOutstream_L.WRITETEXT(PADSTR(TableItem_L."ICPE Category Code", 4), 4);
                          FileExportOutstream_L.WRITETEXT();
                        END;
                      END;
                    END;
                  UNTIL TableItem_L.NEXT=0;
                END;
              END;
              Boucle_L += 1;
            UNTIL Boucle_L > 1;

            IF FileNameOuvert_L THEN BEGIN
              NbCarEcrit_L := FileExport_L.LEN;
              FileExport_L.CLOSE;
              TableParameterMaj_L.LOCKTABLE(TRUE, FALSE);
              IF (TableParameterMaj_L.FINDFIRST) THEN BEGIN
                TableParameterMaj_L."Last Date Export Item" := DateTimeStart_L;
                IF (NbCarEcrit_L > 0) THEN BEGIN
                  TableParameterMaj_L."Last Item File Number" := FileNumber_L;
                END;
                IF NOT TableParameterMaj_L.MODIFY(TRUE) THEN BEGIN
                  ToolsFile_L.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0004, TableParameterMaj_L.TABLECAPTION, COMPANYNAME)
                    ,1,MAXSTRLEN(Result_L)));
                END;
              END;
            END;
          END;
        END;
      END;

      IF (NbCarEcrit_L > 0) THEN BEGIN
        IF ToolsFile_L.LogErreurType() = 'E' THEN BEGIN
          IF FileNameOuvert_L THEN BEGIN
            FILE.ERASE(FileTemp_L);
          END;
        END ELSE BEGIN
          IF RENAME(FileTemp_L, FileName_L) THEN BEGIN
          END;
          //-->20150723-INT19-CMC02.02.00-ADVITAM-CDL001
          // Copier le fichier ParamStre d'Eresis
          MoteurInterface_L.GetParameter(TableParameter_L."Export Item Eresis CodeUnit", EreConnectorName_L,
            EreImpExpFolder_L, EreLogFolder_L, EreExtensionFile_L, ErePrefixFile_L);
          EreFileName_L := EreImpExpFolder_L + '\' + DELCHR(ErePrefixFile_L, '<>', ' ') +
            CONVERTSTR(FORMAT(FileNumber_L, 8), ' ', '0') + '.' + DELCHR(EreExtensionFile_L, '<>', ' ');
          IF NOT COPY(FileName_L, EreFileName_L) THEN BEGIN
            ToolsFile_L.LogWriteInfo(COPYSTR(STRSUBSTNO(Text0012, FileName_L, EreFileName_L), 1, MAXSTRLEN(Result_L)));
          END;
          //<--20150723-INT19-CMC02.02.00-ADVITAM-CDL001
          IF NOT ToolsFile_L.FileCopySav(FileName_L, 'Sav', TRUE) THEN BEGIN
            ToolsFile_L.LogWriteErreur(STRSUBSTNO(Text0011, FileName_L));
          END;
        END;
      END ELSE BEGIN
        IF FileNameOuvert_L THEN BEGIN
          FILE.ERASE(FileTemp_L);
        END;
      END;

      FileLogName_L := ToolsFile_L.LogName;
      ToolsFile_L.LogClose();
      IF ToolsFile_L.LogErreurType() = 'E' THEN BEGIN
        ERROR('ERREUR ' + FileLogName_L);
      END ELSE BEGIN
        IF ToolsFile_L.LogErreurType() = 'I' THEN BEGIN
          MoteurInterface_L.InsertLog(CodeUnit_G, FileLogName_L);
        END ELSE BEGIN
          MoteurInterface_L.InsertOk(CodeUnit_G);
        END;
      END;
      //<-- 20140317-INT19-CMC02.02.00-ADVITAM-GMX
    END;

    BEGIN
    {
      //--> 3LI BEGIN
      20140317-INT19-CMC02.02.00-ADVITAM-GMX : Export Article OET
      20150723-INT19-CMC02.02.00-ADVITAM-CDL001 : Copier le fichier g'n'r' pour interface Eresis
      // TODO-001 : Pr'voir code 'S' pour suppression article
      // TODO-002 : Environnemental
      //<-- 3LI END
    }
    END.
  }
}
