OBJECT Codeunit 60024 Noyau Digital - Process
{
  OBJECT-PROPERTIES
  {
    Date=11/12/20;
    Heure=00:00:00;
    Liste versions=SP3512;
  }
  PROPERTIES
  {
    OnRun=BEGIN
            IF ProcessType_G = ProcessType_G::UpdatePIMCode THEN BEGIN
              UpdatePIMCode;
            END;
          END;

  }
  CODE
  {
    VAR
      ProcessType_G@100000000 : ',UpdatePIMCode';
      ItemNo_G@100000003 : Code[20];
      VariantNo_G@100000004 : Code[10];
      Text60000@100000005 : TextConst 'FRA=Code Produit NavAgri %1 inexistant en base';
      Text60001@100000006 : TextConst 'FRA=Plusieurs variantes existent dans Navagri pour le code produit %1 : mise . jour de lêid PIM impossible';
      Text60002@100000007 : TextConst 'FRA=ID PIM d'j. existant pour le code produit %1';
      PIMCode_G@100000001 : Code[20];

    PROCEDURE InitProcessType@100000002(ProcessType_P@100000000 : ',UpdatePIMCode');
    BEGIN
      ProcessType_G := ProcessType_P;
    END;

    PROCEDURE InitPIMCode@100000000(ItemNo_P@100000001 : Code[20];VariantNo_P@100000002 : Code[10];PIMCode_P@100000000 : Code[20]);
    BEGIN
      ItemNo_G := ItemNo_P;
      VariantNo_G := VariantNo_P;
      PIMCode_G := PIMCode_P;
    END;

    PROCEDURE UpdatePIMCode@100000001();
    VAR
      Item_L@100000000 : Record 27;
      ItemVariant_L@100000001 : Record 5401;
      ItemCrossReference_L@100000002 : Record 5717;
      VariantNo_L@100000003 : Code[10];
    BEGIN
      VariantNo_L := '';

      //Test existence ID PIM
      ItemCrossReference_L.RESET;
      ItemCrossReference_L.SETRANGE("Cross-Reference Type", ItemCrossReference_L."Cross-Reference Type"::"SAS Product");
      ItemCrossReference_L.SETRANGE("Cross-Reference No.", PIMCode_G);
      IF ItemCrossReference_L.FINDFIRST THEN
        IF ItemCrossReference_L."Variant Code" = '' THEN
          ERROR(STRSUBSTNO(Text60002, ItemCrossReference_L."Item No."))
        ELSE
          ERROR(STRSUBSTNO(Text60002, ItemCrossReference_L."Item No." + ' : ' + ItemCrossReference_L."Variant Code"));

      //Test existence Article / Variante
      IF NOT Item_L.GET(ItemNo_G) THEN
        ERROR(STRSUBSTNO(Text60000, ItemNo_G));
      IF VariantNo_G <> '' THEN
        IF NOT ItemVariant_L.GET(ItemNo_G, VariantNo_G) THEN
          ERROR(STRSUBSTNO(Text60000, ItemNo_G + ' : ' + VariantNo_G))
        ELSE
          VariantNo_L := ItemVariant_L.Code;

      //Test si variante non fournie et plusieurs variantes existantes
      IF VariantNo_G = '' THEN BEGIN
        ItemVariant_L.RESET;
        ItemVariant_L.SETRANGE("Item No.", ItemNo_G);
        IF ItemVariant_L.COUNT > 1 THEN
          ERROR(STRSUBSTNO(Text60001, ItemNo_G));
        IF ItemVariant_L.FINDFIRST THEN
          VariantNo_L := ItemVariant_L.Code;
      END;

      ItemCrossReference_L.RESET;
      ItemCrossReference_L."Cross-Reference Type" := ItemCrossReference_L."Cross-Reference Type"::"SAS Product";
      ItemCrossReference_L."Item No." := ItemNo_G;
      ItemCrossReference_L."Variant Code" := VariantNo_L;
      ItemCrossReference_L."Cross-Reference No." := PIMCode_G;
      IF NOT ItemCrossReference_L.INSERT THEN
        ItemCrossReference_L.MODIFY;
    END;

    BEGIN
    {
      20201211-WI00-ADV01.01.00-SP3512-MRU01 : CREATE OBJECT
    }
    END.
  }
}
