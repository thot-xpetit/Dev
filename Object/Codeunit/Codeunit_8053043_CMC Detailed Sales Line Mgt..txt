      END;

    PROCEDURE UpdateOutstandingQty@8053003(RecSalesLine_P@8053000 : Record 37);
    VAR
      RecDetailSalesLine_L@8053002 : Record 8053475;
    BEGIN
      IF RecDetailSalesLine_L.GET(RecSalesLine_P."Document Type",RecSalesLine_P."Document No.",RecSalesLine_P."Line No.") THEN BEGIN
        IF RecSalesLine_P."Document Type" = RecSalesLine_P."Document Type"::Order THEN BEGIN
          RecDetailSalesLine_L."Outstanding Quantity" := RecSalesLine_P."Outstanding Quantity";
          RecDetailSalesLine_L."Outstanding Qty. (Base)" := RecSalesLine_P."Outstanding Qty. (Base)";
        END
        ELSE IF RecSalesLine_P."Document Type" = RecSalesLine_P."Document Type"::"Return Order" THEN BEGIN
          RecDetailSalesLine_L."Outstanding Quantity" := - RecSalesLine_P."Outstanding Quantity";
          RecDetailSalesLine_L."Outstanding Qty. (Base)" := - RecSalesLine_P."Outstanding Qty. (Base)";
        END;
        RecDetailSalesLine_L.MODIFY;
      END;
    END;
