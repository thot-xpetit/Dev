  CODE
  {
    VAR
      G_recLogTourRqst@8053003 : Record 8053044;
      G_recTransferLine@8053002 : Record 5741;
      G_recLocation@8053001 : Record 14;

    PROCEDURE Release@1(P_recTransHeader@8053000 : Record 5740);
    BEGIN
      WITH P_recTransHeader DO BEGIN
        G_recLogTourRqst.RESET;
        G_recLogTourRqst.SETRANGE("Source Document Type",G_recLogTourRqst."Source Document Type"::Transfert);
        G_recLogTourRqst.SETRANGE("Source Document No.",P_recTransHeader."No.");
        G_recLogTourRqst.DELETEALL;

        G_recTransferLine.SETCURRENTKEY("Document No.", "Line No.");
        G_recTransferLine.SETRANGE("Document No.", "No.");
        G_recTransferLine.SETRANGE("Derived From Line No.", 0);
        IF G_recTransferLine.FINDSET THEN BEGIN
          REPEAT
            CreateLogTourRqst(P_recTransHeader);
          UNTIL G_recTransferLine.NEXT = 0;
        END;
      END;
    END;

    PROCEDURE Reopen@2(P_recTransHeader@8053001 : Record 5740);
    VAR
      L_recLogTourRqst@1001 : Record 8053044;
      L_LogTourType@8053000 : 'Inbound,Outbound';
    BEGIN
      WITH P_recTransHeader DO BEGIN
        G_recLogTourRqst.RESET;
        G_recLogTourRqst.SETCURRENTKEY("Source Document Type","Source Document No.","Source Document Line No.");
        G_recLogTourRqst.SETRANGE("Source Document Type", G_recLogTourRqst."Source Document Type"::Transfert);
        G_recLogTourRqst.SETRANGE("Source Document No.", "No.");
       G_recLogTourRqst.MODIFYALL("Document Status", G_recLogTourRqst."Document Status"::Open, TRUE);
      END;
    END;

    LOCAL PROCEDURE CreateLogTourRqst@3(VAR P_recTransHeader@1000 : Record 5740);
    VAR
      L_recTransferLine2@8053000 : Record 5741;
      L_recItem@8053001 : Record 27;
      L_recLocation@8053002 : Record 14;
      L_recActivityDeliveryArea@8053003 : Record 8053047;
      PostCode_L@1000000000 : Record 225;
    BEGIN
      WITH P_recTransHeader DO BEGIN
        IF L_recLocation.RequireReceive("Transfer-to Code") OR L_recLocation.RequirePutaway("Transfer-to Code") THEN BEGIN
          L_recTransferLine2.COPY(G_recTransferLine);
          L_recTransferLine2.SETRANGE("Document No.","No.");
          IF L_recItem.GET(L_recTransferLine2."Item No.") THEN BEGIN
            G_recLogTourRqst.RESET;
            G_recLogTourRqst.INIT;
            G_recLogTourRqst."Source Document Type" := G_recLogTourRqst."Source Document Type"::Transfert;
            G_recLogTourRqst."Source Document No."  := P_recTransHeader."No.";
            G_recLogTourRqst."Source Document Line No." := L_recTransferLine2."Line No.";
            G_recLogTourRqst."External Document No." := P_recTransHeader."External Document No.";
            G_recLogTourRqst."Post Code" := P_recTransHeader."Transfer-to Post Code";
            G_recLogTourRqst.City := P_recTransHeader."Transfer-to City";
            G_recLogTourRqst."Promised Delivery Date" := L_recTransferLine2."Receipt Date";
            G_recLogTourRqst."Requested Delivery Date" := L_recTransferLine2."Receipt Date";
            G_recLogTourRqst."Departure Location Code" := P_recTransHeader."Transfer-from Code";
            G_recLogTourRqst."Destination Type" := G_recLogTourRqst."Destination Type"::Location;
            G_recLogTourRqst."Destination No." := P_recTransHeader."Transfer-to Code";
            G_recLogTourRqst."Destination Name" := P_recTransHeader."Transfer-to Name";
            G_recLogTourRqst."Item No." := L_recTransferLine2."Item No.";
            G_recLogTourRqst."Variant Code" := L_recTransferLine2."Variant Code";
            G_recLogTourRqst."Item Description" := L_recTransferLine2.Description;
            G_recLogTourRqst.Adduction := FALSE;
            G_recLogTourRqst."Order Quantity" := ABS(L_recTransferLine2.Quantity);
            G_recLogTourRqst."Outstanding Quantity" := ABS(L_recTransferLine2."Outstanding Quantity");
            G_recLogTourRqst."Unit of Measure Code" := L_recTransferLine2."Unit of Measure Code";
            G_recLogTourRqst."Pallet Quantity" := 0;
            G_recLogTourRqst."Available Quantity" := 0;
            G_recLogTourRqst."Activity Code" := L_recItem."Activity Code";
            G_recLogTourRqst.Type := G_recLogTourRqst.Type::Outbound;
            G_recLogTourRqst."Document Status" := P_recTransHeader.Status::Released;
            G_recLogTourRqst."Logistic Grouping" := L_recItem."Logistic Grouping Code";

            IF L_recLocation.GET(P_recTransHeader."Transfer-from Code") THEN
              G_recLogTourRqst."Logistic Area Code" := L_recLocation."Logistic Area Code";
              IF PostCode_L.GET(P_recTransHeader."Transfer-to Post Code",P_recTransHeader."Transfer-to City") AND
                L_recActivityDeliveryArea.GET(L_recItem."Activity Code", PostCode_L."INSEE No.")
              THEN
              G_recLogTourRqst."Delivery Area Code" := L_recActivityDeliveryArea."Delivery Area Code";

            IF NOT G_recLogTourRqst.INSERT THEN
              IF G_recLogTourRqst."Outstanding Quantity" <> 0 THEN
                G_recLogTourRqst.MODIFY
              ELSE
                IF G_recLogTourRqst.FIND('=') THEN
                  G_recLogTourRqst.DELETE(TRUE);
           END;
        END;
      END;
    END;
