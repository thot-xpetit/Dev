    PROCEDURE CalculIfu@1000000000(TypeD'claration_P@1000000010 : 'IFU,DAS2';AnneeDecl_P@1000000000 : Integer;CodeUnit_P@1000000009 : Integer);
    VAR
      "<-- 3LI LOCALS -->"@1000000006 : Integer;
      TableIfuParam_L@1000000005 : Record 8087775;
      TableIfuDecl_L@1000000004 : Record 8087777;
      TableGLEntry_L@1000000003 : Record 17;
      TableGLEntryOrigine_L@1000000002 : Record 17;
      TableCVLedgerEntry_L@1000000001 : Record 8053380;
      TableContact_L@1000000012 : Record 5050;
      CVLedgerEntry_L@1000000008 : Boolean;
      AnneePaie_L@1000000007 : Integer;
      DateFiltre_L@1000000011 : Date;
      CptDeposant_L@1000000013 : Boolean;
    BEGIN
      //MessageDebug(STRSUBSTNO('17Ori:%1:%2:%3:%4:%5', TableGLEntryOrigine_L."Entry No."));
      MESSAGE('Traitement "%1" "%2".', TypeD'claration_P, AnneeDecl_P);
      TableIfuParam_L.SETRANGE("Type d'claration", TypeD'claration_P);
      TableIfuParam_L.SETRANGE("Ann'e D'claration", AnneeDecl_P);
      IF NOT TableIfuParam_L.FINDSET() THEN BEGIN
        ERROR(TextDecNonParam, AnneeDecl_P);
      END ELSE BEGIN
        // On remonte 1 ann'e en arriSre (date compta) puisque c'est la date de paiement qui est utilis'e.
        // SAUF 2016 puisque 2015 . tous d'clar' ne tenait pas compte de date paiement.
        IF (TableIfuParam_L."Ann'e D'claration" = 2016) THEN BEGIN
          DateFiltre_L := DMY2DATE(1, 1, TableIfuParam_L."Ann'e D'claration");
        END ELSE BEGIN
          DateFiltre_L := DMY2DATE(1, 1, (TableIfuParam_L."Ann'e D'claration" - 1));
        END;
        // Contr"ler Non d'j. 'dit'.
        TableIfuDecl_L.SETRANGE("Type d'claration", TableIfuParam_L."Type d'claration");
        TableIfuDecl_L.SETRANGE("Ann'e D'claration", AnneeDecl_P);
        TableIfuDecl_L.SETFILTER(HoroEdition, '<> %1', 0DT);
        IF (NOT TableIfuDecl_L.ISEMPTY) THEN BEGIN
          ERROR(TextDecDejaEdit, AnneeDecl_P);
        END ELSE BEGIN
          // Supprimer le calcul d'origine.
          CLEAR(TableIfuDecl_L);
          TableIfuDecl_L.SETRANGE("Type d'claration", TableIfuParam_L."Type d'claration");
          TableIfuDecl_L.SETRANGE("Ann'e D'claration", AnneeDecl_P);
          TableIfuDecl_L.SETRANGE("Forc' le montant", FALSE);
          TableIfuDecl_L.DELETEALL(TRUE);
          // Caculer pour chaque param'trage.
          REPEAT
            // Lire 'criture comptable correspondante au param'trage : Compte & journal.
            TableGLEntry_L.SETRANGE("G/L Account No.", TableIfuParam_L."G/L Account No.");
            TableGLEntry_L.SETFILTER("Posting Date", '>= %1', DateFiltre_L);

            IF TableIfuParam_L."Source Code" <> '' THEN BEGIN
              TableGLEntry_L.SETRANGE("Source Code", TableIfuParam_L."Source Code");
            END ELSE BEGIN
              TableGLEntry_L.SETRANGE("Source Code");
            END;
            IF TableGLEntry_L.FINDSET() THEN BEGIN
              REPEAT
                // Retouver l''criture d'origine associ' . un compte de tiers : N› transaction + Type docuement + N› document.
                TableGLEntryOrigine_L.SETFILTER("Entry No.", '<> %1', TableGLEntry_L."Entry No.");
                TableGLEntryOrigine_L.SETRANGE("Transaction No.", TableGLEntry_L."Transaction No.");
                TableGLEntryOrigine_L.SETRANGE("Document Type", TableGLEntry_L."Document Type");
                TableGLEntryOrigine_L.SETRANGE("Document No.", TableGLEntry_L."Document No.");
                TableGLEntryOrigine_L.SETFILTER("Source No.", '<> %1', '');
                IF TableGLEntryOrigine_L.FINDFIRST () THEN BEGIN
                  // Retrouver l''criture client fournisseur pour le lettrage
                  CVLedgerEntry_L := FALSE;
                  IF TableGLEntryOrigine_L."Source Type" = TableGLEntryOrigine_L."Source Type"::Customer THEN BEGIN
                    CVLedgerEntry_L := TableCVLedgerEntry_L.GET(TableCVLedgerEntry_L."Customer / Vendor"::Customer,
                        TableGLEntryOrigine_L."Entry No.");
                  END ELSE BEGIN
                    IF TableGLEntryOrigine_L."Source Type" = TableGLEntryOrigine_L."Source Type"::Vendor THEN BEGIN
                      CVLedgerEntry_L := TableCVLedgerEntry_L.GET(TableCVLedgerEntry_L."Customer / Vendor"::Vendor,
                        TableGLEntryOrigine_L."Entry No.");
                    END;
                  END;
                  IF CVLedgerEntry_L THEN BEGIN
                    CLEAR(AnneePaie_L);
                    // On utilise plus l''criture de payement pour la date r'elle de paiement mais la date th'orique dans
                    // l''criture d'origine.
                    IF (TableCVLedgerEntry_L."Due Date" <> 0D) THEN BEGIN
                      AnneePaie_L := DATE2DMY(TableCVLedgerEntry_L."Due Date", 3);
                    END ELSE BEGIN
                      AnneePaie_L := DATE2DMY(TableCVLedgerEntry_L."Posting Date", 3);
                    END;
                    IF (AnneePaie_L = 0) THEN BEGIN
                      CptDeposant_L := FALSE;
                      IF TableContact_L.GET(TableCVLedgerEntry_L."Contact No.") THEN BEGIN
                        CptDeposant_L := TableContact_L."Depositor Account";
                      END;
                      IF CptDeposant_L THEN BEGIN
                        IF (TableCVLedgerEntry_L."Due Date" <> 0D) THEN BEGIN
                          AnneePaie_L := DATE2DMY(TableCVLedgerEntry_L."Due Date", 3);
                        END ELSE BEGIN
                          AnneePaie_L := DATE2DMY(TableCVLedgerEntry_L."Posting Date", 3);
                        END;
                      END ELSE BEGIN
                        // Retrouver l''criture de paiement pour la date sur le m^me lettrage
                        IF TableCVLedgerEntry_L.GET(TableCVLedgerEntry_L."Customer / Vendor",
                          TableCVLedgerEntry_L."Original Closed by Entry No.") THEN BEGIN
                          IF (TableCVLedgerEntry_L."Due Date" <> 0D) THEN BEGIN
                            AnneePaie_L := DATE2DMY(TableCVLedgerEntry_L."Due Date", 3);
                          END;
                        END;
                      END;
                    END;
                    IF AnneePaie_L = TableIfuParam_L."Ann'e D'claration" THEN BEGIN
                      TableIfuDecl_L.SETRANGE("Type d'claration", TableIfuParam_L."Type d'claration");
                      TableIfuDecl_L.SETRANGE("Ann'e D'claration", TableIfuParam_L."Ann'e D'claration");
                      TableIfuDecl_L.SETRANGE(RubriqueIFU, TableIfuParam_L.RubriqueIFU);
                      TableIfuDecl_L.SETRANGE("Cust./Vend. No.", TableCVLedgerEntry_L."Cust./Vend. No.");
                      IF TableIfuDecl_L.FINDFIRST() THEN BEGIN
                        IF (TableIfuDecl_L."Type d'claration" = TableIfuDecl_L."Type d'claration"::DAS2) THEN BEGIN
                          // Das2 ==> TTC
                          TableIfuDecl_L.VALIDATE("Montant calcul'", TableIfuDecl_L."Montant calcul'" + TableGLEntry_L.Amount +
                            TableGLEntry_L."VAT Amount");
                        END ELSE BEGIN
                          TableIfuDecl_L.VALIDATE("Montant calcul'", TableIfuDecl_L."Montant calcul'" + TableGLEntry_L.Amount);
                        END;
                        TableIfuDecl_L.MODIFY(TRUE);
                      END ELSE BEGIN
                        CLEAR(TableIfuDecl_L);
                        TableIfuDecl_L.VALIDATE("Type d'claration", TableIfuParam_L."Type d'claration");
                        TableIfuDecl_L.VALIDATE("Ann'e D'claration", TableIfuParam_L."Ann'e D'claration");
                        TableIfuDecl_L.VALIDATE(RubriqueIFU, TableIfuParam_L.RubriqueIFU);
                        TableIfuDecl_L.VALIDATE("Cust./Vend. No.", TableCVLedgerEntry_L."Cust./Vend. No.");
                        IF (TableIfuDecl_L."Type d'claration" = TableIfuDecl_L."Type d'claration"::DAS2) THEN BEGIN
                          // Das2 ==> TTC
                          TableIfuDecl_L.VALIDATE("Montant calcul'", TableGLEntry_L.Amount + TableGLEntry_L."VAT Amount");
                        END ELSE BEGIN
                          TableIfuDecl_L.VALIDATE("Montant calcul'", TableGLEntry_L.Amount);
                        END;
                        TableIfuDecl_L.INSERT(TRUE);
                      END;
                    END;
                  END;
                END;
              UNTIL TableGLEntry_L.NEXT = 0;
            END;
          UNTIL TableIfuParam_L.NEXT = 0;
        END;
      END;
    END;
