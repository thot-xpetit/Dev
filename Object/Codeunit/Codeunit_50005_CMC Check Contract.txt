    PROCEDURE GetContracts@1000000004(CautionSetup@1000000000 : Record 50011);
    VAR
      ChangeLogEntry@1000000001 : Record 405;
      TmpCmcCollectionCtc@1000000002 : TEMPORARY Record 8052983;
      CmcCollectionContract@1000000003 : Record 8052983;
    BEGIN
      ChangeLogEntry.SETRANGE("Table No.", DATABASE::"CMC Collection Contract");
      ChangeLogEntry.SETRANGE("Type of Change", ChangeLogEntry."Type of Change"::Insertion);
      IF (CautionSetup."Last Generation Date Time" <> 0DT) THEN BEGIN
        ChangeLogEntry.SETFILTER("Date and Time", '>= %1', CautionSetup."Last Generation Date Time");
      END;
      TmpCmcCollectionCtc.DELETEALL();

      IF (ChangeLogEntry.FINDFIRST()) THEN BEGIN
        REPEAT
          CmcCollectionContract.SETRANGE("No.", ChangeLogEntry."Primary Key Field 1 Value");
          IF (CmcCollectionContract.FINDFIRST()) THEN BEGIN
            IF ((CmcCollectionContract."Campaign No." = CautionSetup."Campaign No.") AND
               (CmcCollectionContract."Species Code" =CautionSetup."Species Code") AND
               (CmcCollectionContract."Contract Type" = CautionSetup."Contract Type")) THEN BEGIN
               TmpCmcCollectionCtc.INIT();
               TmpCmcCollectionCtc.TRANSFERFIELDS(CmcCollectionContract);
               TmpCmcCollectionCtc.INSERT();
             END;
          END;
        UNTIL (ChangeLogEntry.NEXT() = 0);
        // Generate Document
        IF (TmpCmcCollectionCtc.FINDFIRST()) THEN BEGIN
          HeaderNo := 0;
          HeaderLineNo := 0;
          REPEAT
            CreateHeader(CautionSetup, TmpCmcCollectionCtc);
            CreateLine(CautionSetup, TmpCmcCollectionCtc);
          UNTIL(TmpCmcCollectionCtc.NEXT() = 0);
          CautionSetup."Last Generation Date Time" := CREATEDATETIME(TODAY, TIME);
          CautionSetup.MODIFY();
        END;
      END;
    END;
