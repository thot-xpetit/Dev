            //--> 20130927-INT05-CMC02.02.00-ADVITAM-GMX

              ExportSite("File in use");
            END ELSE BEGIN

            //<-- 20130927-INT05-CMC02.02.00-ADVITAM-GMX
          END;

  }
  CODE
  {
    VAR
      "<-- 3LI TEXT GLOBALS -->"@1000000001 : TextConst;
      Text0001@1000000000 : TextConst 'ENU=Error Table "%1" doesn''t exist \Company : %2;FRA=Erreur Table "%1" inexistante \Societe : %2';
      Text0003@1000000002 : TextConst 'ENU=Error Creating File %1 \%2.;FRA=Erreur cr'ation fichier %1 \%2.';
      Text0004@1000000003 : TextConst 'ENU=Error updating table %1;FRA=Erreur modification table "%1" \Societe : %2';
      Text0005@1000000005 : TextConst 'ENU=See the log please;FRA="Veuillez consulter le Log  "';
      Text0006@1000000006 : TextConst 'ENU=Vendor N› : %1 , Site : "%2", Pb Post Code "%3" & City "%4";FRA=Fournisseur N› : "%1" , Site : "%2'',  Pb code postal "%3" & Ville "%4"';
      Text0007@1000000007 : TextConst 'FRA=Erreur correspondance "%1" : "%2" non param'tr'e pour "%3".';
      Text0009@1000000008 : TextConst 'ENU=Pb param - Initial Vendor : % 1 / Site Vendor : %2;FRA="Pb paramStrage - Fournisseur initial : %1 / Fournisseur Site : %2 "';
      Text0011@1000000004 : TextConst 'ENU=Error backup copy.;FRA=Erreur copie de sauvegarde.';
      "<-- 3LI GLOBALS -->"@1000000010 : Integer;
      CodeUnit_G@1000000009 : Integer;

    PROCEDURE "<-- 3LI FUNCTIONS -->"@1000000000();
    BEGIN
    END;

    PROCEDURE ExportSite@1000000001(FileName_P@1000000000 : Text[256]) Result_L : Text[256];
    VAR
      "<-- 3LI LOCALS -->"@1000000002 : Integer;
      TableParameter_L@1000000018 : Record 8053404;
      TableOriginSite_L@1000000025 : Record 8053221;
      TablePostCode_L@1000000029 : Record 225;
      ToolsFile_L@1000000017 : Codeunit 8053405;
      MoteurInterface_L@1000000005 : Codeunit 8061230;
      FileNameOuvert_L@1000000001 : Boolean;
      NbCarEcrit_L@1000000003 : Integer;
      ConnectorName_L@1000000010 : Code[20];
      ImpExpFolder_L@1000000009 : Text[256];
      LogFolder_L@1000000008 : Text[256];
      ExtensionFile_L@1000000007 : Text[10];
      PrefixFile_L@1000000006 : Text[10];
      ItemTemplateCode_L@1000000016 : Code[20];
      CustomerTemplateCode_L@1000000015 : Code[20];
      VendorTemplateCode_L@1000000014 : Code[20];
      ContactTemplateCode_L@1000000013 : Code[20];
      AccountTemplateCode_L@1000000012 : Code[20];
      GlobalTemplateCode_L@1000000011 : Code[20];
      DateTimeStart_L@1000000019 : DateTime;
      FileName_L@1000000020 : Text[256];
      FileNumber_L@1000000021 : Integer;
      FileTemp_L@1000000022 : Text[256];
      FilePath_L@1000000024 : Text[256];
      FileN_L@1000000023 : Text[256];
      FileExport_L@1000000026 : File;
      FileExportOutStream_L@1000000027 : OutStream;
      FileLogName_L@1000000028 : Text[256];
      VendorRadar_L@1000000030 : Code[20];
      SiteRadar_L@1000000031 : Code[30];
      Texte_L@1000000032 : Text[30];
      ZoneNum_L@1000000033 : Decimal;
    BEGIN
      //----------------------------------------------------------------------------------------------------------------------------------
      // Extraction des sites origines table "Origin Site" pour Cr'ation du fichier Radar/Operra :  FYnnnnnnnn.dat
      //----------------------------------------------------------------------------------------------------------------------------------
      // Entr'es:
      //   - FileName_P : Nom du fichier PC . G'n'rer
      // Sorties:
      //   - Result_L : Erreur Retourn'e, Vide si Ok
      //----------------------------------------------------------------------------------------------------------------------------------
      FileNameOuvert_L := FALSE;
      NbCarEcrit_L := 0;
      MoteurInterface_L.GetParameter(CodeUnit_G, ConnectorName_L, ImpExpFolder_L, LogFolder_L, ExtensionFile_L, PrefixFile_L);
      MoteurInterface_L.GetTemplatesCode (ConnectorName_L, ItemTemplateCode_L, CustomerTemplateCode_L, VendorTemplateCode_L,
        ContactTemplateCode_L, AccountTemplateCode_L, GlobalTemplateCode_L);
      ToolsFile_L.LogInit(LogFolder_L, 'RADAR');
      ToolsFile_L.LogInitCodeUnit(CodeUnit_G);

      IF FileName_P = '' THEN BEGIN
        IF (NOT TableParameter_L.FINDFIRST) THEN BEGIN
          ToolsFile_L.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0001, TableParameter_L.TABLECAPTION, COMPANYNAME), 1, MAXSTRLEN(Result_L)));
        END;
      END;

      IF ToolsFile_L.LogErreurType() <> 'E' THEN BEGIN
        IF NOT TableParameter_L.FINDFIRST THEN BEGIN
          ToolsFile_L.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0001, TableParameter_L.TABLECAPTION ,COMPANYNAME),1,MAXSTRLEN(Result_L)));
        END ELSE BEGIN
          IF TableParameter_L."Export Origin Site RADAR" THEN BEGIN
            DateTimeStart_L := CURRENTDATETIME;
            IF FileName_P <> '' THEN BEGIN
              FileName_L := FileName_P;
            END ELSE BEGIN
              FileNumber_L := TableParameter_L."Last Site File Number" + 1;
              FileName_L := ImpExpFolder_L + '\' + DELCHR(PrefixFile_L, '<>', ' ') +
                CONVERTSTR(FORMAT(FileNumber_L,8),' ','0') + '.' + DELCHR(ExtensionFile_L, '<>', ' ');
            END;
            ToolsFile_L.LogInitFileExecute(FileName_L);
            ToolsFile_L.SplitFilename(FileName_L, FilePath_L, FileN_L);
            FileTemp_L := FilePath_L + 'Temp_' + FileN_L;

            IF TableOriginSite_L.FINDFIRST THEN BEGIN
              IF NOT FileNameOuvert_L THEN BEGIN
                IF NOT FileExport_L.CREATE(FileTemp_L) THEN BEGIN
                  ToolsFile_L.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0003, FileTemp_L, GETLASTERRORTEXT), 1, MAXSTRLEN(Result_L)));
                  CLEARLASTERROR;
                END ELSE BEGIN
                  FileExport_L.CREATEOUTSTREAM(FileExportOutStream_L);
                  FileNameOuvert_L := TRUE;
                END;
              END;
              IF FileNameOuvert_L THEN BEGIN
                REPEAT
                  VendorRadar_L := MoteurInterface_L.GetReferentialCorrespondance(VendorTemplateCode_L, FALSE,
                    TableOriginSite_L."Vendor No.", TRUE);
                  IF VendorRadar_L <> '$' THEN BEGIN
                    IF VendorRadar_L = '' THEN BEGIN
                      ToolsFile_L.LogWriteInfo(COPYSTR(STRSUBSTNO(Text0007,
                        TableOriginSite_L.FIELDCAPTION(TableOriginSite_L."Vendor No."),
                        VendorRadar_L, TableOriginSite_L."Vendor No."), 1, MAXSTRLEN(Result_L)));
                    END ELSE BEGIN
                      SiteRadar_L := MoteurInterface_L.Getcorrespondance(GlobalTemplateCode_L, 'CMC Origin Site', '',
                            TableOriginSite_L."Vendor No." + 'Ø' + TableOriginSite_L.Code, FALSE, TRUE);
                      IF COPYSTR(DELCHR(SiteRadar_L, '=', 'Ø'), 11, 3) = '' THEN BEGIN
                        ToolsFile_L.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0007, TableOriginSite_L.FIELDCAPTION(TableOriginSite_L.Code)
                          , TableOriginSite_L.Code, TableOriginSite_L."Vendor No."), 1, MAXSTRLEN(Result_L)));
                      END ELSE BEGIN
      //                  IF COPYSTR(DELCHR(SiteRadar_L, '=', 'Ø'), 1, 10) <> VendorRadar_L THEN BEGIN
      //                    ToolsFile_L.LogWriteInfo(COPYSTR(STRSUBSTNO(Text0009, VendorRadar_L,
      //                      COPYSTR(DELCHR(SiteRadar_L, '=', 'Ø'), 1, 10)), 1, MAXSTRLEN(Result_L)));
      //                  END ELSE BEGIN
                          FileExportOutStream_L.WRITETEXT('FPR',3);
                          FileExportOutStream_L.WRITETEXT('M',1);
                          FileExportOutStream_L.WRITETEXT(PADSTR(VendorRadar_L,10,' '),10);
                          FileExportOutStream_L.WRITETEXT(PADSTR(COPYSTR(DELCHR(SiteRadar_L, '=', 'Ø'), 11, 3),3,' '),3);
                          FileExportOutStream_L.WRITETEXT(PADSTR(TableOriginSite_L.City,30,' '),30);
                          // Code pays & Code Insee . partir du code postal
                          CLEAR(TablePostCode_L);
                          TablePostCode_L.SETRANGE(TablePostCode_L.Code,TableOriginSite_L."Post code");
                          TablePostCode_L.SETRANGE(TablePostCode_L.City,TableOriginSite_L.City);
                          IF NOT TablePostCode_L.FINDFIRST THEN BEGIN
                            ToolsFile_L.LogWriteInfo(COPYSTR(STRSUBSTNO(Text0006, TableOriginSite_L."Vendor No.",
                              TableOriginSite_L.Code, TableOriginSite_L."Post code", TableOriginSite_L.City), 1, MAXSTRLEN(Result_L)));
                          END;
                          // Code pays
                          Texte_L := COPYSTR(TablePostCode_L."INSEE No.", 1, 2);
                          IF NOT EVALUATE(ZoneNum_L, Texte_L) THEN BEGIN
                            FileExportOutStream_L.WRITETEXT(PADSTR(Texte_L, 3, ' '), 3);
                            FileExportOutStream_L.WRITETEXT(PADSTR(COPYSTR(TablePostCode_L."INSEE No.", 3, 5), 5, ' '), 5);
                          END ELSE BEGIN
                            FileExportOutStream_L.WRITETEXT('FR ',3);
                            FileExportOutStream_L.WRITETEXT(PADSTR(TablePostCode_L."INSEE No.",5,' '),5);
                          END;
                          // Code Insee
                          FileExportOutStream_L.WRITETEXT('.',1);
                          FileExportOutStream_L.WRITETEXT();
      //                  END;
                      END;
                    END;
                  END;
                UNTIL TableOriginSite_L.NEXT=0;

                NbCarEcrit_L := FileExport_L.LEN;
                FileExport_L.CLOSE;
                IF TableParameter_L.FINDFIRST THEN BEGIN
                  TableParameter_L."Last Date Export Site" := DateTimeStart_L;
                  IF (NbCarEcrit_L > 0) THEN BEGIN
                    TableParameter_L."Last Site File Number" := FileNumber_L;
                  END;
                  IF NOT TableParameter_L.MODIFY(TRUE) THEN BEGIN
                    ToolsFile_L.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0004,TableParameter_L.TABLECAPTION,COMPANYNAME),1
                      ,MAXSTRLEN(Result_L)));
                  END;
                END;
              END;
            END;
          END;
        END;
      END;

      IF (NbCarEcrit_L > 0) THEN BEGIN
        IF ToolsFile_L.LogErreurType() = 'E' THEN BEGIN
          IF FileNameOuvert_L THEN BEGIN
            FILE.ERASE(FileTemp_L);
          END;
        END ELSE BEGIN
          IF RENAME(FileTemp_L, FileName_L) THEN BEGIN
          END;
          IF NOT ToolsFile_L.FileCopySav(FileName_L, 'Sav', TRUE) THEN BEGIN
            ToolsFile_L.LogWriteErreur(STRSUBSTNO(Text0011, FileName_L));
          END;
        END;
      END ELSE BEGIN
        IF FileNameOuvert_L THEN BEGIN
          FILE.ERASE(FileTemp_L);
        END;
      END;

      CLEAR(Result_L);
      FileLogName_L := ToolsFile_L.LogName;
      ToolsFile_L.LogClose();
      IF ToolsFile_L.LogErreurType() = 'E' THEN BEGIN
        Result_L := 'ERREUR ' + FileLogName_L;
        ERROR(Result_L);
      END ELSE BEGIN
        IF ToolsFile_L.LogErreurType() = 'I' THEN BEGIN
          Result_L := Text0005;
          MoteurInterface_L.InsertLog(CodeUnit_G, FileLogName_L);
        END ELSE BEGIN
          MoteurInterface_L.InsertOk(CodeUnit_G);
        END;
      END;
      //<-- 20130927-INT05-CMC02.02.00-ADVITAM-GMX
    END;

    BEGIN
    {
      //--> 3LI BEGIN
      20130927-Int05-CMC02.02.00-ADVITAM-GMX : Export des Provenances/Fournisseurs interface RADAR.
      //TODO-001 : Code pays
      //<-- 3LI END
    }
    END.
  }
