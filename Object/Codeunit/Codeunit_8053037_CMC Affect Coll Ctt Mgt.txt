    PROCEDURE GetContractOutstandingQuantity@8053050(ReceiptNo_P@1000000000 : Code[20];ContractNo_P@1000000001 : Code[20];ReceiptType_P@1000000002 : 'Receipt,Transfert,Resumption';Validate_P@1000000003 : Boolean;VendorNo_P@1000000004 : Code[20];LineNo_P@1000000005 : Integer;ContractType_P@8053001 : 'Collecte,Commercialisation') : Decimal;
    VAR
      QuantityOnReceipt_L@1000000006 : Decimal;
      CollectionReceipt2_L@1000000007 : Record 8052993;
      CollectionReceiptLine_L@1000000008 : Record 8053103;
      CollectionContract_L@1000000009 : Record 8052983;
      OutstandingQuantity_L@1000000010 : Decimal;
      Campaign_L@1000000011 : Record 8052991;
      Excess_L@1000000012 : Decimal;
      StandardWeight_P@1000000013 : Decimal;
      SalesCollectionContract_L@1000000014 : Record 8053012;
      QtyReceived_L@1000000015 : Decimal;
      QtyContract_L@1000000016 : Decimal;
    BEGIN
      // On calcul la quantit' en cours de r'ception sur d'autres lignes de bon ( bon en cours ou autres lignes
      // de bon non valid's )

      OutstandingQuantity_L := 0;
      QuantityOnReceipt_L := 0;

      CollectionReceiptLine_L.RESET;
      CollectionReceiptLine_L.SETCURRENTKEY("Contract No.");
      CollectionReceiptLine_L.SETRANGE("Contract No.",ContractNo_P);
      IF CollectionReceiptLine_L.FINDSET THEN
        REPEAT
         IF NOT Validate_P THEN BEGIN
            IF (ReceiptType_P<>CollectionReceiptLine_L."Receipt Type") OR (ReceiptNo_P<>CollectionReceiptLine_L."No.") OR
              (LineNo_P<> CollectionReceiptLine_L."Line No.")
           THEN
             QuantityOnReceipt_L := QuantityOnReceipt_L + CollectionReceiptLine_L.Quantity;
         END ELSE
          QuantityOnReceipt_L := QuantityOnReceipt_L + CollectionReceiptLine_L.Quantity;
        UNTIL CollectionReceiptLine_L.NEXT = 0;

      IF ContractType_P= ContractType_P::Collecte THEN BEGIN // contrat de collecte
        CollectionContract_L.GET(ContractNo_P);
        Campaign_L.GET(CollectionContract_L."Campaign No.");
        CollectionContract_L.CALCFIELDS("Qty On Post. Recept Line");
        QtyReceived_L:=CollectionContract_L."Qty On Post. Recept Line";
        QtyContract_L:= CollectionContract_L.Quantity +
        // --> 20140926-CER06-TER01.01.00-RMZ01
                    ROUND(CollectionContract_L.Quantity * (CollectionContract_L."Threshold Percentage" / 100),0.00001);
        // <-- 20140919-CER06-TER01.01.00-RMZ01
      END ELSE BEGIN    // contrat de commercialisation
        SalesCollectionContract_L.GET(ContractNo_P);
        Campaign_L.GET(SalesCollectionContract_L."Campaign No.");
        SalesCollectionContract_L.CALCFIELDS("Quantity Received");
        QtyReceived_L:=SalesCollectionContract_L."Quantity Received";
        QtyContract_L:= SalesCollectionContract_L.Quantity +
        // --> 20140926-CER06-TER01.01.00-RMZ01
              ROUND(CollectionContract_L.Quantity * (CollectionContract_L."Threshold Percentage" / 100),0.00001);
        // <-- 20140919-CER06-TER01.01.00-RMZ01
      END;
      OutstandingQuantity_L := QtyContract_L - QtyReceived_L - QuantityOnReceipt_L;
