OBJECT Codeunit 81 Sales-Post (Yes/No)
{
  OBJECT-PROPERTIES
  {
    Date=30/10/14;
    Heure=00:00:00;
    Liste versions=NAVW15.00,CMB6.00.10.06,CMC02.01.00,CMB6.00.10.09;
  }
  PROPERTIES
  {
    TableNo=36;
    OnRun=BEGIN
            SalesHeader.COPY(Rec);
            // --> 20110826-8060990/085-CMB6.00.10.04-FEC01
            {BEFORE
            Code;
            AFTER}
            SalesSetup_G.GET;
            CASE SalesSetup_G."Sales Doc. Posting Method" OF
              SalesSetup_G."Sales Doc. Posting Method"::Standard : Code;
              SalesSetup_G."Sales Doc. Posting Method"::Confirmation : BEGIN
                CLEAR(SalesPost_G);
                SalesPost_G.InitReport(SalesHeader,FALSE,SalesSetup_G."Sales Doc. Posting Option",TRUE);
                SalesPost_G.RUNMODAL;
              END;
              SalesSetup_G."Sales Doc. Posting Method"::"Confirmation (Not Editable)": BEGIN
                CLEAR(SalesPost_G);
                SalesPost_G.InitReport(SalesHeader,FALSE,SalesSetup_G."Sales Doc. Posting Option",FALSE);
                SalesPost_G.RUNMODAL;
              END;
            END;
            // <-- 20110826-8060990/085-CMB6.00.10.04-FEC01
            Rec := SalesHeader;
          END;

  }
  CODE
  {
    VAR
      Text000@1000 : TextConst 'ENU=&Ship,&Invoice,Ship &and Invoice;FRA=&Livrer,&Facturer,Livrer &et facturer';
      Text001@1001 : TextConst 'ENU=Do you want to post the %1?;FRA=Souhaitez-vous valider le document %1 ?';
      Text002@1002 : TextConst 'ENU=&Receive,&Invoice,Receive &and Invoice;FRA=&R'ceptionner,&Facturer,R'ceptionner &et facturer';
      SalesHeader@1003 : Record 36;
      SalesPost@1004 : Codeunit 80;
      Selection@1005 : Integer;
      "<<3LI GLOBALS>>"@8060982 : Integer;
      SalesSetup_G@8060983 : Record 311;
      SalesPost_G@8060984 : Report 8060995;

    LOCAL PROCEDURE Code@1();
    VAR
      "<<3LI LOCALS>>"@8053000 : Integer;
      SplitInvoice_L@8053501 : Boolean;
      recLine_L@8053502 : Record 37;
    BEGIN
      WITH SalesHeader DO BEGIN
        //<-- 20131115-Log03-CMC02.01.00-922-FHU01
        SplitInvoice_L := FALSE;
        //--> 20131115-Log03-CMC02.01.00-922-FHU01

        // --> 20140314-8060990/104-CMB6.00.10.09-1815-FEC01
        "Post Drop Shipment Only" := FALSE;
        // --< 20140314-8060990/104-CMB6.00.10.09-1815-FEC01

        CASE "Document Type" OF
          "Document Type"::Order:
            BEGIN
              // --> 3LI-20120808-8060990/085-CMB6.00.10.06-JDL01 avant
              //Selection := STRMENU(Text000,3);
              //apres
              Selection := STRMENU(Text000,SalesSetup_G."Sales Doc. Posting Option"+1);
              // <-- 20120808-8060990/085-CMB6.00.10.06-JDL01
              IF Selection = 0 THEN
                EXIT;
              //--> 20131115-Log03-CMC02.01.00-922-FHU01
              recLine_L.RESET;
              recLine_L.SETRANGE("Document Type", "Document Type");
              recLine_L.SETRANGE("Document No.", "No.");
              recLine_L.SETFILTER(recLine_L."Prepayment %", '<>0');
              IF NOT(recLine_L.ISEMPTY) THEN BEGIN
                IF Selection = 3 THEN BEGIN
                  SplitInvoice_L := TRUE;
                END;
              END;
              IF SplitInvoice_L = FALSE THEN BEGIN
              //<-- 20131115-Log03-CMC02.01.00-922-FHU01
              Ship := Selection IN [1,3];
              Invoice := Selection IN [2,3];
              //--> 20131115-Log03-CMC02.01.00-922-FHU01
              END
              ELSE BEGIN
                Ship := TRUE;
                Invoice := FALSE;
              END;
              //<-- 20131115-Log03-CMC02.01.00-922-FHU01
            END;
          "Document Type"::"Return Order":
            BEGIN
              // --> 3LI-20120808-8060990/085-CMB6.00.10.06-JDL01 avant
              //Selection := STRMENU(Text002,3);
              //apres
              Selection := STRMENU(Text002,SalesSetup_G."Sales Doc. Posting Option"+1);
              // <-- 3LI-20120808-8060990/085-CMB6.00.10.06-JDL01
              IF Selection = 0 THEN
                EXIT;
              Receive := Selection IN [1,3];
              Invoice := Selection IN [2,3];
            END ELSE
              IF NOT
                 CONFIRM(
                   Text001,FALSE,
                   "Document Type")
              THEN
                EXIT;
        END;
        SalesPost.RUN(SalesHeader);
        //--> 20131115-Log03-CMC02.01.00-922-FHU01
        IF SplitInvoice_L = TRUE THEN BEGIN
          SalesHeader.GET("Document Type", "No.");
          Ship := FALSE;
          Invoice := TRUE;
          SalesPost.RUN(SalesHeader);
        END;
        //<-- 20131115-Log03-CMC02.01.00-922-FHU01
        COMMIT;
      END;
    END;

    BEGIN
    {
      // --> 3LI BEGIN
      20110826-8060990/085-CMB6.00.10.04-FEC01 : ADDCODE (Gestion de la validation des documents de ventes).
      20110826-8060990/085-CMB6.00.10.04-FEC02 : ADDGLOBAL (SalesSetup_G).
      20110826-8060990/085-CMB6.00.10.04-FEC03 : ADDGLOBAL (SalesPost_G).
      20110826-8060990/085-CMB6.00.10.04-FEC04 : ADDGLOBAL (DirectDocPostingMgt_G).
      20120808-8060990/085-CMB6.00.10.06-JDL01 : ADDCODE (gestion de l'option par d'faut sur le code standard)
      20131115-Log03-CMC02.01.00-922-FHU01 : ADDCODE (r'alise en 2 temps la livraison et la facturation dans le cas o- il y a un acompte)
      20140314-8060990/104-CMB6.00.10.09-1815-FEC01 : ADDCODE.

      // <-- 3LI END
    }
    END.
  }
}
