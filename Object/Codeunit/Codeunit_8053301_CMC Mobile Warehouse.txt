    PROCEDURE LinkBarcode@8053001(P_sourceItem@8053000 : Code[20];P_targetItem@8053001 : Code[20];P_BarcodeNo@8053002 : Code[20]);
    VAR
      L_recItem@8053005 : Record 27;
      L_recItemCrossRef@8053003 : Record 5717;
    BEGIN
      IF P_sourceItem <> '' THEN BEGIN
        L_recItemCrossRef.RESET;
        L_recItemCrossRef.SETCURRENTKEY("Cross-Reference Type","Cross-Reference No.","Item No.");
        L_recItemCrossRef.SETRANGE("Cross-Reference Type", L_recItemCrossRef."Cross-Reference Type"::"Bar Code");
        L_recItemCrossRef.SETRANGE("Cross-Reference No.", P_BarcodeNo);
        L_recItemCrossRef.SETRANGE("Item No.", P_sourceItem);
        IF L_recItemCrossRef.FINDFIRST THEN BEGIN
          // ----------------------------------------------------
          // Suppression du code bar associ' . l'article
          // ----------------------------------------------------
          L_recItemCrossRef.DELETE(TRUE);
        END;
      END;

      // ----------------------------------------------------
      // Association du code bar . l'article s'lectionn'
      // ----------------------------------------------------
      L_recItem.GET(P_targetItem);
      CLEAR(L_recItemCrossRef);
      L_recItemCrossRef.RESET;
      L_recItemCrossRef.INIT;
      L_recItemCrossRef.VALIDATE("Item No.", P_targetItem);
      L_recItemCrossRef.VALIDATE("Cross-Reference Type", L_recItemCrossRef."Cross-Reference Type"::"Bar Code");
      L_recItemCrossRef.VALIDATE("Cross-Reference No.", P_BarcodeNo);
      L_recItemCrossRef.INSERT(TRUE);
    END;

    PROCEDURE AddPhysInvLine@8053003(P_JnlTemplate@8053000 : Code[10];P_JnlBatch@8053001 : Code[10];P_ItemNo@8053002 : Code[20];P_Quantity@8053003 : Decimal;P_UserID@8053004 : Text[30];P_Location@8053007 : Code[10];P_Bin@8053008 : Code[20];P_Update@1180250002 : Boolean);
    VAR
      L_recPhysInvJnl@8053005 : Record 8053072;
      L_NewLineNo@8053006 : Integer;
    BEGIN
      L_recPhysInvJnl.RESET;
      L_recPhysInvJnl.SETCURRENTKEY("Journal Template Name", "Journal Batch Name", "Line No.");
      L_recPhysInvJnl.SETRANGE("Journal Template Name", P_JnlTemplate);
      L_recPhysInvJnl.SETRANGE("Journal Batch Name", P_JnlBatch);
      L_recPhysInvJnl.SETRANGE("Item No.", P_ItemNo);
      L_recPhysInvJnl.SETRANGE("Location Code", P_Location);
      IF P_Bin <> '' THEN
        L_recPhysInvJnl.SETRANGE("Bin Code", P_Bin);

      IF NOT L_recPhysInvJnl.FINDFIRST THEN BEGIN
        // ---------------------------------------------------------------
        // Creation d'une nouvelle ligne
        // ---------------------------------------------------------------
        L_recPhysInvJnl.SETRANGE("Item No.");
        L_recPhysInvJnl.SETRANGE("Location Code");
        L_recPhysInvJnl.SETRANGE("Bin Code");
        IF L_recPhysInvJnl.FINDLAST THEN
          L_NewLineNo := L_recPhysInvJnl."Line No." + 10
        ELSE
          L_NewLineNo := 10;

        L_recPhysInvJnl.RESET;
        L_recPhysInvJnl.INIT;
        L_recPhysInvJnl.VALIDATE("Journal Template Name", P_JnlTemplate);
        L_recPhysInvJnl.VALIDATE("Journal Batch Name", P_JnlBatch);
        L_recPhysInvJnl.VALIDATE("Line No.", L_NewLineNo);
        L_recPhysInvJnl.VALIDATE("Item No.", P_ItemNo);
        L_recPhysInvJnl.VALIDATE(Quantity, P_Quantity);

        IF P_Location <> '' THEN
          L_recPhysInvJnl.VALIDATE("Location Code", P_Location);

        IF P_Bin <> '' THEN
          L_recPhysInvJnl.VALIDATE("Bin Code", P_Bin);

        L_recPhysInvJnl.VALIDATE("Inventaire Date", WORKDATE);
        L_recPhysInvJnl.VALIDATE("Inventaire Time", TIME);
        L_recPhysInvJnl.VALIDATE("User ID", P_UserID);
        L_recPhysInvJnl.INSERT(TRUE);

      END ELSE BEGIN
        // ---------------------------------------------------------------
        // CUMUL sur ligne existante si Mode UPDATE
        // ---------------------------------------------------------------
        IF (P_Update) THEN
          L_recPhysInvJnl.VALIDATE(Quantity, P_Quantity)
        ELSE
          L_recPhysInvJnl.VALIDATE(Quantity, L_recPhysInvJnl.Quantity + P_Quantity);
        L_recPhysInvJnl.VALIDATE("Inventaire Date", WORKDATE);
        L_recPhysInvJnl.VALIDATE("Inventaire Time", TIME);
        L_recPhysInvJnl.VALIDATE("User ID", P_UserID);
        L_recPhysInvJnl.MODIFY(TRUE);

      END;
    END;
