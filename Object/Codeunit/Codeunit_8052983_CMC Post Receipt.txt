            RecOldItemLdgEntry_L."Reserved Quantity"),RecOldItemLdgEntry_L,RecItemJnlLine_L);

    PROCEDURE SuggestReceiptAffecttoResump@8053008(VAR PostedReceipts_P@8053000 : Record 8052995;VAR QtyToAffect_P@8053001 : Decimal);

      // - QtyToAffect_P : quantit' restante . affecter

    END;
