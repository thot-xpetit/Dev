    PROCEDURE CreateReceipt@1000000001(ExelienceHeader_P@1000000000 : Record 60078;VAR CMCCollectionReceipt_P@1000000001 : Record 8052993;ContractNo_P@1000000002 : Code[10]) : Boolean;
    VAR
      CMCCollectionReceiptLine_L@1000000010 : Record 8053103;
      ItemSNO_L@1000000009 : Record 60100;
      Item_L@1000000008 : Record 27;
      ReceiptDateTime_L@1000000006 : DateTime;
      CMBAdvSetupMgt_L@1000000005 : Codeunit 8061290;
      CMCSpeciesSetup_L@1000000004 : Record 8053323;
      CMBInterfaceCorresp_L@1000000011 : Record 8061233;
      CMCSetup_L@60000 : Record 8052982;
      SeedsSetup_L@60001 : Record 60106;
      NetWeight_L@60002 : Decimal;
      StandardWeight_L@60003 : Decimal;
    BEGIN
      WITH ExelienceHeader_P DO BEGIN
        CMCSetup_L.GET;
        // --> 20190110-Wi00-ADV01.01.00-3358-MRU01
        //R'cup'ration des paramStres semences
        SeedsSetup_L.GET;
        // <-- 20190110-Wi00-ADV01.01.00-3358-MRU01

        CMCCollectionReceipt_P."Receipt Type" := CMCCollectionReceipt_P."Receipt Type"::Receipt;
        CMCCollectionReceipt_P.INSERT(TRUE);

        CMCCollectionReceipt_P.VALIDATE("Campaign No.", "Campaign Year");

        // --> 20190110-Wi00-ADV01.01.00-3358-MRU01
        //R'cup'ration du code magasin via le modSle de correspondance
        CMBInterfaceCorresp_L.RESET;
        CMBInterfaceCorresp_L.SETRANGE("Template code", TextTemplateLocation);
        CMBInterfaceCorresp_L.SETRANGE("Interface Value", "Management Site");
        {BEFORE
        CMBInterfaceCorresp_L.FINDFIRST;
        CMCCollectionReceipt_P.VALIDATE("Location Code", CMBInterfaceCorresp_L."NAV Value");
        AFTER}
        //R'cup'ration du code magasin via le modSle de correspondance, s'il n'existe pas alors magasin par d'faut (Param Semences)
        IF CMBInterfaceCorresp_L.FINDFIRST THEN
          CMCCollectionReceipt_P.VALIDATE("Location Code", CMBInterfaceCorresp_L."NAV Value")
        ELSE
          CMCCollectionReceipt_P.VALIDATE("Location Code", SeedsSetup_L."Seed Location Code");
        // <-- 20190110-Wi00-ADV01.01.00-3358-MRU01

        CMCCollectionReceipt_P.VALIDATE("Vendor No.", COPYSTR("Third Code", 2));
        ItemSNO_L.GET("Item No.");
        Item_L.GET(ItemSNO_L."Correspondance num'ro article");
        CMCCollectionReceipt_P.VALIDATE("Species Code", Item_L."Species Code");
        CMCCollectionReceipt_P.VALIDATE("Item No.", Item_L."No.");
        EVALUATE(ReceiptDateTime_L, "Receipt DateTime");
        CMCCollectionReceipt_P."Creation Date" := ReceiptDateTime_L;

        //Traitement particulier pour la date comptabilisation : ni elle n'est pas dans la p'riode FranceAgrimer, alors TODAY
        IF (DT2DATE(ReceiptDateTime_L) >= CMCSetup_L."ONIC Allow Posting From")
            AND (DT2DATE(ReceiptDateTime_L) <= CMCSetup_L."ONIC Allow Posting To") THEN
          CMCCollectionReceipt_P."Posting Date" := DT2DATE(ReceiptDateTime_L)
        ELSE
          CMCCollectionReceipt_P."Posting Date" := TODAY;

        CMCCollectionReceipt_P."Shipment Date" := DT2DATE(ReceiptDateTime_L);
        EVALUATE(CMCCollectionReceipt_P."Net Weight", "Net Weight");
        CMCCollectionReceipt_P."Net Weight" := CMCCollectionReceipt_P."Net Weight"/1000;
        EVALUATE(CMCCollectionReceipt_P."Standard Weight", "Standard Weight");
        // --> 20190513-WI00-ADV01.01.00-3358-MRU01
        {BEFORE
        CMCCollectionReceipt_P."Weight Standards Already" := TRUE;
        AFTER}
        CMCCollectionReceipt_P."Weight Standards Already" := FALSE;
        // <-- 20190513-WI00-ADV01.01.00-3358-MRU01

        CMCCollectionReceipt_P.VALIDATE("Collection Type", CMCCollectionReceipt_P."Collection Type"::Purchase);

        // --> 20190130-Wi00-ADV01.01.00-3358-MRU01
        IF COMPANYNAME = 'TERNOVEO' THEN
        // <-- 20190130-Wi00-ADV01.01.00-3358-MRU01
          // --> 20190429-Wi00-ADV01.01.00-3358-MRU01
          //CMCCollectionReceipt_P."Manual Weighing No." := '0';
          CMCCollectionReceipt_P.VALIDATE("Manual Weighing No.", '0');
          // <-- 20190429-Wi00-ADV01.01.00-3358-MRU01

        CMCCollectionReceipt_P."Notebook Weighing No." := '0';

        //Si Vari't' n'cessaire, r'cup'ration de l'information
        IF CMBAdvSetupMgt_L.GetAdvancedParameterValueBool('CER-RCPT-VARIETOBLIG') THEN BEGIN
          CMCCollectionReceipt_P."Variety Code" := Item_L."Variety Code";
        END;

        //Si Bon de livraison agriculteur n'cessaire, on remplit le champ
        IF CMBAdvSetupMgt_L.GetAdvancedParameterValueBool('CER-RCPT-BLAGRIOBLIG') THEN
          CMCCollectionReceipt_P."Vendor Delivery Note" := '0';

        //Si N› Scell' n'cessaire, on remplit le champ
        CMCSpeciesSetup_L.GET(Item_L."Species Code");
        IF CMCSpeciesSetup_L."Mandatory Seal" THEN
          CMCCollectionReceipt_P."Seal No." := '0';

        CMCCollectionReceipt_P."Contract No." := ContractNo_P;
        CMCCollectionReceipt_P."EXELIENCE Receipt Reference" := "Receipt Reference";

        //Si Desinsectisation n'cessaire, on remplit le champ
        IF NOT CMCSetup_L."Disinsectization default to No" THEN
          CMCCollectionReceipt_P.Disinsectization := CMCCollectionReceipt_P.Disinsectization::No;

        //Si N› Bon Manuel n'cessaire, on remplit le champ
        IF CMBAdvSetupMgt_L.GetAdvancedParameterValueBool('CER-RCPT-TICKMAOBLIG') THEN
          CMCCollectionReceipt_P."Manual Voucher No." := CMCCollectionReceipt_P."No.";

        CMCCollectionReceipt_P.MODIFY(TRUE);

        InsertReceiptAnalysis(ExelienceHeader_P, CMCCollectionReceipt_P);

        // --> 20190513-WI00-ADV01.01.00-3358-MRU01
        CMCCollectionReceipt_P.VALIDATE("Net Weight");
        CMCCollectionReceipt_P.MODIFY;
        // <-- 20190513-WI00-ADV01.01.00-3358-MRU01

        CLEAR(CMCCollectionReceiptLine_L);
        CMCCollectionReceiptLine_L.LOCKTABLE;
        CMCCollectionReceiptLine_L.INIT;
        CMCCollectionReceiptLine_L."Receipt Type" := CMCCollectionReceipt_P."Receipt Type";
        CMCCollectionReceiptLine_L."No." := CMCCollectionReceipt_P."No.";

        CMCCollectionReceiptLine_L.VALIDATE("Contract No.", ContractNo_P);
        IF NOT CMCCollectionReceiptLine_L.INSERT THEN
          CMCCollectionReceiptLine_L.MODIFY;

        // --> 20190513-WI00-ADV01.01.00-3358-MRU01
        RecRef_G.GETTABLE(CMCCollectionReceipt_P);
        EVALUATE(NetWeight_L, "Net Weight");
        NetWeight_L := NetWeight_L / 1000;
        EVALUATE(StandardWeight_L, "Standard Weight");
        InsertNotes(RecRef_G, NetWeight_L, StandardWeight_L);
        // <-- 20190513-WI00-ADV01.01.00-3358-MRU01

      END;
    END;

    PROCEDURE InsertReceiptAnalysis@1000000015(ExelienceHeader_P@1000000006 : Record 60078;VAR CMCCollectionReceipt_P@1000000002 : Record 8052993);
    VAR
      ExelienceLine_L@1000000000 : Record 60103;
      CMCCollReceiptAnalysis_L@1000000001 : Record 8052994;
      CMBInterfaceCorresp_L@1000000003 : Record 8061233;
      SeedsSetup_L@60000 : Record 60106;
    BEGIN
      ExelienceLine_L.RESET;
      ExelienceLine_L.SETRANGE("Sequence Flow No.", ExelienceHeader_P."Sequence Flow No.");
      ExelienceLine_L.SETRANGE("Sequence Header No.", ExelienceHeader_P."Sequence No.");
      IF ExelienceLine_L.FINDFIRST THEN REPEAT
        WITH ExelienceLine_L DO BEGIN
          // --> 20190110-Wi00-ADV01.01.00-3358-MRU01
          {BEFORE
          // --> 20181105-WI00-ADV01.01.00-3358-MRU01
          {BEFORE
          //Si le Code Caract'ristique est diff'rent de TextFeatureImmat, cr'ation d'une ligne d'analyse
          //Sinon, renseignement du num'ro d'immatriculation v'hicule
          IF "Feature Code" <> TextFeatureImmat THEN BEGIN
          AFTER}
          //Si le Code Caract'ristique est diff'rent de TextFeatureImmat et TextFeatureModLiv, cr'ation d'une ligne d'analyse
          IF ("Feature Code" <> TextFeatureImmat) AND ("Feature Code" <> TextFeatureModLiv) THEN BEGIN
          // <-- 20181105-WI00-ADV01.01.00-3358-MRU01
          AFTER}
          //Cr'ation d'une ligne d'analyse si le Code caract'ristique n'a pas de lien avec l'immatriculation ou la livraison
          IF ("Feature Code" <> TextFeatureImmat) AND ("Feature Code" <> TextFeatureModLiv)
                                                  AND ("Feature Code" <> TextFeatureBlLiv) THEN BEGIN
          // <-- 20190110-Wi00-ADV01.01.00-3358-MRU01
            CMBInterfaceCorresp_L.RESET;
            CMBInterfaceCorresp_L.SETRANGE("Template code", TextTemplateAnalysis);
            CMBInterfaceCorresp_L.SETRANGE("Interface Value", "Feature Code");
            // --> 20190130-Wi00-ADV01.01.00-3358-MRU01
            //Suppression du contr"le durant la phase d'analyse des donn'es : ne retourne pas forc'ment un enregistrement
            {BEFORE
            CMBInterfaceCorresp_L.FINDFIRST;
            AFTER}
            IF CMBInterfaceCorresp_L.FINDFIRST THEN BEGIN
            // <-- 20190130-Wi00-ADV01.01.00-3358-MRU01

              CLEAR(CMCCollReceiptAnalysis_L);
              CMCCollReceiptAnalysis_L.LOCKTABLE;
              CMCCollReceiptAnalysis_L.INIT;
              CMCCollReceiptAnalysis_L."Receipt Type" := CMCCollectionReceipt_P."Receipt Type";
              CMCCollReceiptAnalysis_L."Receipt No." := CMCCollectionReceipt_P."No.";
              CMCCollReceiptAnalysis_L."Analysis Code" := CMBInterfaceCorresp_L."NAV Value";
              EVALUATE(CMCCollReceiptAnalysis_L.Value, "Feature Value");
              CMCCollReceiptAnalysis_L.VALIDATE(Value);
              IF NOT CMCCollReceiptAnalysis_L.INSERT THEN
                CMCCollReceiptAnalysis_L.MODIFY;
            // --> 20190130-Wi00-ADV01.01.00-3358-MRU01
            END;
            // <-- 20190130-Wi00-ADV01.01.00-3358-MRU01
          END ELSE BEGIN
          // --> 20181105-WI00-ADV01.01.00-3358-MRU01
            {BEFORE
            CMCCollectionReceipt_P."Trucks Registration" := "Feature Value";
            CMCCollectionReceipt_P.MODIFY;
            AFTER}
            IF "Feature Code" = TextFeatureImmat THEN BEGIN
              CMCCollectionReceipt_P."Trucks Registration" := "Feature Value";
              CMCCollectionReceipt_P.MODIFY;
            END;

            IF "Feature Code" = TextFeatureModLiv THEN BEGIN
              IF UPPERCASE("Feature Value") = 'MULTIPLICATEUR' THEN
                CMCCollectionReceipt_P."Transport Type" := CMCCollectionReceipt_P."Transport Type"::Vendor;
              IF UPPERCASE("Feature Value") = 'TRANSPORTEUR' THEN BEGIN
                CMCCollectionReceipt_P."Transport Type" := CMCCollectionReceipt_P."Transport Type"::Company;
                // --> 20190110-Wi00-ADV01.01.00-3358-MRU01
                {BEFORE
                CMCCollectionReceipt_P.VALIDATE("Shipping Agent Code", 'EXELIENCE');
                AFTER}
                SeedsSetup_L.GET;
                CMCCollectionReceipt_P.VALIDATE("Shipping Agent Code", SeedsSetup_L."Seed Shipping Agent Code");
                // <-- 20190110-Wi00-ADV01.01.00-3358-MRU01
              END;
              CMCCollectionReceipt_P.MODIFY;
            END;

            // --> 20190110-Wi00-ADV01.01.00-3358-MRU01
            IF "Feature Code" = TextFeatureBlLiv THEN BEGIN
              CMCCollectionReceipt_P."Vendor Delivery Note" := "Feature Value";
              // --> 20190130-Wi00-ADV01.01.00-3358-MRU01
              IF COMPANYNAME = 'UNEAL' THEN
                // --> 20190429-Wi00-ADV01.01.00-3358-MRU01
                //CMCCollectionReceipt_P."Manual Weighing No." := "Feature Value";
                CMCCollectionReceipt_P.VALIDATE("Manual Weighing No.", "Feature Value");
                // <-- 20190429-Wi00-ADV01.01.00-3358-MRU01
              // <-- 20190130-Wi00-ADV01.01.00-3358-MRU01
              CMCCollectionReceipt_P.MODIFY;
            END;
            // <-- 20190110-Wi00-ADV01.01.00-3358-MRU01
          // <-- 20181105-WI00-ADV01.01.00-3358-MRU01
          END;

        END;
      UNTIL ExelienceLine_L.NEXT = 0;
    END;

    PROCEDURE CreateDelivery@60000(ExelienceHeader_P@60001 : Record 60078;VAR CMCDelivery_P@60000 : Record 8053013;VAR CMCCollectionReceipt_P@60003 : Record 8052993);
    VAR
      SeedsSetup_L@60002 : Record 60106;
      CMCSalesContract_L@60004 : Record 8053012;
      DeliveryDateTime_L@60005 : DateTime;
      CMCSetup_L@60006 : Record 8052982;
      VendorName_L@1000000000 : Text[50];
      CommentLength_L@1000000001 : Integer;
      CMBAdvSetupMgt_L@100000000 : Codeunit 8061290;
    BEGIN
      // --> 20190110-Wi00-ADV01.01.00-3358-MRU01
      WITH ExelienceHeader_P DO BEGIN
        CMCSetup_L.GET;
        SeedsSetup_L.GET;

        CMCDelivery_P."Delivery Type" := CMCDelivery_P."Delivery Type"::Delivery;
        CMCDelivery_P.INSERT(TRUE);

        CMCDelivery_P.VALIDATE("Campaign No.", "Campaign Year");
        CMCDelivery_P.VALIDATE("Location Code", SeedsSetup_L."Seed Location Code");
        CMCDelivery_P.VALIDATE("Customer No.", SeedsSetup_L."Seed Customer No.");
        CMCDelivery_P.VALIDATE("Species Code", CMCCollectionReceipt_P."Species Code");

        //R'cup'ration du contrat de vente associ'

        CMCSalesContract_L.SETRANGE("Customer No.", CMCDelivery_P."Customer No.");
        CMCSalesContract_L.SETRANGE("Species Code", CMCDelivery_P."Species Code");
        CMCSalesContract_L.SETRANGE(Status, CMCSalesContract_L.Status::Locked);
        CMCSalesContract_L.FINDFIRST;

        //Traitement particulier pour la date comptabilisation : ni elle n'est pas dans la p'riode FranceAgrimer, alors TODAY
        IF (DT2DATE(DeliveryDateTime_L) >= CMCSetup_L."ONIC Allow Posting From")
            AND (DT2DATE(DeliveryDateTime_L) <= CMCSetup_L."ONIC Allow Posting To") THEN
          CMCDelivery_P."Posting Date" := DT2DATE(DeliveryDateTime_L)
        ELSE
          CMCDelivery_P."Posting Date" := TODAY;

        EVALUATE(DeliveryDateTime_L, "Receipt DateTime");
        CMCDelivery_P."Sales Date" := DT2DATE(DeliveryDateTime_L);
        CMCDelivery_P."Requested Delivery Date" := DT2DATE(DeliveryDateTime_L);

        CMCDelivery_P.VALIDATE("Contract No.", CMCSalesContract_L."No.");

        IF CMCCollectionReceipt_P."Transport Type" = CMCCollectionReceipt_P."Transport Type"::Vendor THEN
          // --> 20190123-Wi00-ADV01.01.00-3358-MRU01
          //CMCDelivery_P."Shipping Type" := CMCDelivery_P."Shipping Type"::Company;
          CMCDelivery_P."Shipping Type" := CMCDelivery_P."Shipping Type"::Customer;
          CMCDelivery_P.VALIDATE("Shipping Agent Code", SeedsSetup_L."Seed Shipping Agent Code");
          CMCDelivery_P."Trucker Name" := ' ';
          // <-- 20190123-Wi00-ADV01.01.00-3358-MRU01
        IF CMCCollectionReceipt_P."Transport Type" = CMCCollectionReceipt_P."Transport Type"::Company THEN BEGIN
          // --> 20190123-Wi00-ADV01.01.00-3358-MRU01
          //CMCDelivery_P."Shipping Type" := CMCDelivery_P."Shipping Type"::Customer;
          //CMCDelivery_P.VALIDATE("Shipping Agent Code", SeedsSetup_L."Seed Shipping Agent Code");
          //CMCDelivery_P."Trucker Name" := ' ';
          CMCDelivery_P."Shipping Type" := CMCDelivery_P."Shipping Type"::Company;
          // <-- 20190123-Wi00-ADV01.01.00-3358-MRU01
        END;

        CMCDelivery_P."Transport Method" := '3'; //Route par d'faut
        CMCDelivery_P.Disinsectization := CMCDelivery_P.Disinsectization::No;
        CMCDelivery_P."Trailers Registration No." := ' ';

        CMCDelivery_P."Creation Date" := DeliveryDateTime_L;

        CMCDelivery_P."User Id" := 'EXELIENCE';

        //Observations : N› Client;Nom Client (tronqu' si total d'passe la taille du champ Comment);R'f'rence Apport
        VendorName_L := CMCCollectionReceipt_P."Vendor Name";
        CommentLength_L := (STRLEN(CMCCollectionReceipt_P."Vendor No.") + STRLEN(VendorName_L) + STRLEN("Receipt Reference") + 2);
        IF CommentLength_L > MAXSTRLEN(CMCDelivery_P.Comment) THEN BEGIN
          VendorName_L := COPYSTR(VendorName_L, 1, STRLEN(VendorName_L) - (CommentLength_L - MAXSTRLEN(CMCDelivery_P.Comment)));
        END;

        CMCDelivery_P.Comment := STRSUBSTNO('%1;%2;%3', CMCCollectionReceipt_P."Vendor No.", VendorName_L
                                           , "Receipt Reference");

        // --> 20190513-WI00-ADV01.01.00-3358-MRU01
        {BEFORE
        CMCDelivery_P."Net Weight" := CMCCollectionReceipt_P."Standard Weight";
        CMCDelivery_P."Standard Weight" := CMCCollectionReceipt_P."Standard Weight";
        CMCDelivery_P."Weight Standards Already" := TRUE;
        AFTER}
        CMCDelivery_P."Net Weight" := CMCCollectionReceipt_P."Net Weight";
        CMCDelivery_P."Standard Weight" := CMCCollectionReceipt_P."Standard Weight";
        CMCDelivery_P."Weight Standards Already" := FALSE;
        // <-- 20190513-WI00-ADV01.01.00-3358-MRU01

        // --> 20190130-Wi00-ADV01.01.00-3358-MRU01
        //Reprise de l'information pr'sente dans le bon de r'ception cr''
        {BEFORE
        CMCDelivery_P."Manual Weighing No." := '0';
        AFTER}
        // --> 20190429-Wi00-ADV01.01.00-3358-MRU01
        //CMCDelivery_P."Manual Weighing No." := CMCCollectionReceipt_P."Manual Weighing No.";
        CMCDelivery_P.VALIDATE("Manual Weighing No.", CMCCollectionReceipt_P."Manual Weighing No.");
        // <-- 20190429-Wi00-ADV01.01.00-3358-MRU01
        // <-- 20190130-Wi00-ADV01.01.00-3358-MRU01
        CMCDelivery_P."Notebook Weighing No." := '0';
        CMCDelivery_P."Clean Containing" := CMCDelivery_P."Clean Containing"::Yes;

        //Si N› Bon Manuel n'cessaire, on remplit le champ
        IF CMBAdvSetupMgt_L.GetAdvancedParameterValueBool('CER-LIVR-TICKMAOBLIG') THEN
          CMCDelivery_P."Manual Voucher No." := CMCDelivery_P."No.";

        CMCDelivery_P.MODIFY(TRUE);

        InsertDeliveryAnalysis(CMCDelivery_P, CMCCollectionReceipt_P);

        // --> 20190513-WI00-ADV01.01.00-3358-MRU01
        CMCDelivery_P.VALIDATE("Net Weight");
        CMCDelivery_P.MODIFY;
        // <-- 20190513-WI00-ADV01.01.00-3358-MRU01

        InsertLastLoading(CMCDelivery_P);

        CMCDelivery_P.VALIDATE("Conformity Containing", CMCDelivery_P."Conformity Containing"::Yes);
        CMCDelivery_P.Status := CMCDelivery_P.Status::Released;
