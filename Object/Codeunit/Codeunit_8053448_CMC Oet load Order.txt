    VAR
      "<-- 3LI LOCALS -->"@1000000005 : Integer;
      int_temp_L@1000000016 : Integer;
      intPos_L@1000000014 : Integer;
      intLg_L@1000000015 : Integer;
      intNbLu_L@1000000009 : Integer;
      Txt_temp_L@1000000003 : Text[1024];
      TxtLu_L@1000000010 : Text[1024];
      ImpExpFolder_L@1000000006 : Text[250];
      LogFolder_L@1000000017 : Text[250];
      ExtensionFile_L@1000000031 : Text[10];
      PrefixFile_L@1000000030 : Text[10];
      FileLogName_L@1000000019 : Text[250];

      OrderNo_L@1000000001 : Code[20];
      ConnectorName_L@1000000013 : Code[20];
      FileExport_L@1000000007 : File;
      FileInStream_L@1000000008 : InStream;
      TableSetup_L@1000000012 : Record 8053431;
      TableOetOrder_L@1000000045 : Record 8053432;
      TableSalesHeader_L@1000000000 : Record 36;
      TableOetExped_L@1000000020 : Record 8053443;
      Trame_L@1000000018 : Text[3];
    BEGIN
      //----------------------------------------------------------------------------------------------------------------------------------
      // Importation fichier texte
      // - Figer la commande.
      //----------------------------------------------------------------------------------------------------------------------------------
      // Entr'es:
      //   - FileName_P             : Nom du fichier PC . int'grer
      // Sorties:
      //   - Result_L   : Erreur Retourn'e, Vide si Ok
      //----------------------------------------------------------------------------------------------------------------------------------
      IF ((TableSetup_L.FINDFIRST) AND (((Partenaire_P = 'OET') AND (TableSetup_L.ExpedOetActive)) OR
        ((Partenaire_P = 'ERESIS') AND (TableSetup_L.ExpedEresisActive)))) THEN BEGIN

        MoteurInterface_G.GetParameter(CodeUnit_P, ConnectorName_L, ImpExpFolder_L, LogFolder_L, ExtensionFile_L, PrefixFile_L);
        ToolsFile_G.LogInit(LogFolder_L, Partenaire_P);
        ToolsFile_G.LogInitCodeUnit(CodeUnit_P);
        ToolsFile_G.LogInitFileExecute(FileName_P);

        IF NOT EXISTS(FileName_P) THEN BEGIN
          ToolsFile_G.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0009, FileName_P), 1, MAXSTRLEN(Result_L)));
        END ELSE BEGIN
          IF NOT FileExport_L.OPEN(FileName_P) THEN BEGIN

            CLEARLASTERROR;
          END ELSE BEGIN
            FileExport_L.CREATEINSTREAM(FileInStream_L);
            WHILE NOT (FileInStream_L.EOS) DO BEGIN
              intNbLu_L := FileInStream_L.READTEXT(TxtLu_L);

              intPos_L := 1;

              Trame_L := ToolsFile_G.ExtractText(TxtLu_L, intPos_L, 3);
              OrderNo_L := ToolsFile_G.ExtractText(TxtLu_L, intPos_L, 8);

                'ERR': BEGIN
                  ToolsFile_G.LogWriteInfo(COPYSTR(STRSUBSTNO(Text0006, OrderNo_L), 1, MAXSTRLEN(Result_L)));
                END;
                'ENT': BEGIN
                  Txt_temp_L := ToolsFile_G.ExtractText(TxtLu_L, intPos_L, 3);
                  IF NOT EVALUATE(OrderLineNo_L, Txt_temp_L) THEN BEGIN
                    ToolsFile_G.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0008,
                      TableOetOrder_L.FIELDCAPTION(TableOetOrder_L."Line No. Extern"), Txt_temp_L, TxtLu_L),
                      1, MAXSTRLEN(Result_L)));
                  END ELSE BEGIN
                    TableOetOrder_L.SETRANGE(TableOetOrder_L.OrderExtern, OrderNo_L);
                    TableOetOrder_L.SETRANGE(TableOetOrder_L."Line No. Extern", OrderLineNo_L);
                    IF NOT TableOetOrder_L.FINDFIRST() THEN BEGIN
                      ToolsFile_G.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0010, OrderNo_L, OrderLineNo_L, TableOetOrder_L.TABLECAPTION),
                        1, MAXSTRLEN(Result_L)));
                    END ELSE BEGIN
                      // <<-- CLARILOG211811
                      IF (TableOetOrder_L."Sale Transfert" = TableOetOrder_L."Sale Transfert"::Sale) THEN BEGIN
                      // -->> CLARILOG211811
                        IF NOT TableSalesHeader_L.GET(TableOetOrder_L."Document Type", TableOetOrder_L."Document No.") THEN BEGIN
                          ToolsFile_G.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0010, TableOetOrder_L."Document Type",
                            TableOetOrder_L."Document No.", TableSalesHeader_L.TABLECAPTION),1, MAXSTRLEN(Result_L)));
                        END ELSE BEGIN
                          IF NOT TableSalesHeader_L."Freezed Document" THEN BEGIN
                            //-->20150702-Int19-CMC02.02.00-ADVITAM-CDL001
                            // V'rifier que ligne non dans un Bon d'exp'dition
                            TableOetExped_L.SETRANGE(OrderNo, TableOetOrder_L.OrderExtern);
                            TableOetExped_L.SETRANGE(OrderLineNo, TableOetOrder_L."Line No. Extern");
                            IF TableOetExped_L.ISEMPTY THEN BEGIN
                              // Pas dans une exp'dition on peut d'figer.
                              //<--20150702-Int19-CMC02.02.00-ADVITAM-CDL001
                              TableSalesHeader_L."Freezed Document" := TRUE;
                              IF NOT TableSalesHeader_L.MODIFY(TRUE) THEN BEGIN
                                ToolsFile_G.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0102, TableOetOrder_L."Document Type",
                                  TableOetOrder_L."Document No.", TableSalesHeader_L.TABLECAPTION),1, MAXSTRLEN(Result_L)));
                              END;
                            //-->20150702-Int19-CMC02.02.00-ADVITAM-CDL001
                            END;
                            //<--20150702-Int19-CMC02.02.00-ADVITAM-CDL001
                          END;
                        END;
                      // <<-- CLARILOG211811
                      END ELSE BEGIN
                      END;
                      // -->> CLARILOG211811
                    END;
                  END;
                END;
              END;
            END;
          END;
        END;
        FileExport_L.CLOSE();
        IF ToolsFile_G.LogErreurType() <> 'E' THEN BEGIN
          IF NOT ToolsFile_G.FileRenameSav(FileName_P, 'Sav', TRUE) THEN BEGIN
            ToolsFile_G.LogWriteErreur(STRSUBSTNO(Text0014, FileName_P));
          END;
        END;
        CLEAR(Result_L);
        FileLogName_L := ToolsFile_G.LogName;
        ToolsFile_G.LogClose();
        IF ToolsFile_G.LogErreurType() = 'E' THEN BEGIN
          Result_L := 'ERREUR ' + FileLogName_L;
          ERROR(Result_L);
        END ELSE BEGIN
          //TableSetup_L."Last Date Import" := CURRENTDATETIME;
          //IF TableSetup_L.MODIFY(TRUE) THEN BEGIN
          //END;
          IF ToolsFile_G.LogErreurType() = 'I' THEN BEGIN
            Result_L := Text0101;
            MoteurInterface_G.InsertLog(CodeUnit_P, FileLogName_L);
          END ELSE BEGIN
            MoteurInterface_G.InsertOk(CodeUnit_P);
          END;
        END;
      END;
