    LOCAL PROCEDURE Code@1();
    BEGIN
      WITH ItemJnlLine DO BEGIN
        ItemJnlTemplate.GET("Journal Template Name");
        ItemJnlTemplate.TESTFIELD("Force Posting Report",FALSE);
        IF ItemJnlTemplate.Recurring AND (GETFILTER("Posting Date") <> '') THEN
          FIELDERROR("Posting Date",Text000);

        // --> 20160105-Wi00-ADV01.01.00-2561-OAN01
        // BEFORE
        {
        // --> 20140516-WI00-CMC02.02.00-2120-FEC01
        IF (NOT AddPermissionMgt_G.AllowItemJnlPost) AND (ItemJnlLine."Movement Type"=0) THEN
          ERROR(Text8053001);
        // --< 20140516-WI00-CMC02.02.00-2120-FEC01
        }
        // AFTER
        IF NOT (ItemJnlTemplate.Type = ItemJnlTemplate.Type::"Phys. Inventory") AND
        (NOT AddPermissionMgt_G.AllowItemJnlPost) AND (ItemJnlLine."Movement Type"=0) THEN
          ERROR(Text8053001);

        IF (ItemJnlTemplate.Type = ItemJnlTemplate.Type::"Phys. Inventory") AND (NOT AddPermissionMgt_G.AllowItemInventoryJnlPost) THEN
          ERROR(Text8053001);
        // <-- 20160105-Wi00-ADV01.01.00-2561-OAN01

        IF NOT CONFIRM(Text001,FALSE) THEN
          EXIT;

        TempJnlBatchName := "Journal Batch Name";
       // --> 20131114-Cer25/001-CMC02.02.00-KEM01
       ItemJnlPostBatch.SetCell(Cell_G);
       // <-- 20131114-Cer25/001-CMC02.02.00-KEM01
        ItemJnlPostBatch.RUN(ItemJnlLine);

        // --> 20140319-Gen15/001-CMC02.02.00-FBE01
        //IF "Line No." = 0 THEN
        IF ("Line No." = 0) AND NOT "Allocate on IC Partners" THEN
        // --> 20140319-Gen15/001-CMC02.02.00-FBE01
          MESSAGE(Text002)
        ELSE
          IF TempJnlBatchName = "Journal Batch Name" THEN
            MESSAGE(Text003)
          ELSE
            MESSAGE(
              Text004 +
              Text005,
              "Journal Batch Name");

        IF NOT FIND('=><') OR (TempJnlBatchName <> "Journal Batch Name") THEN BEGIN
          RESET;
          FILTERGROUP(2);
          SETRANGE("Journal Template Name","Journal Template Name");
          SETRANGE("Journal Batch Name","Journal Batch Name");
          FILTERGROUP(0);
          "Line No." := 1;
        END;
      END;
    END;
