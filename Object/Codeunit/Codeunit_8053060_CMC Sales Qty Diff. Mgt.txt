    VAR
      ExtraQty_L@8053518 : Decimal;
      NewLineQty_L@8053517 : Decimal;
      recProcPurch_L@8053516 : Record 5721;
      PercentEcart_L@8053515 : Decimal;
      recSetup_L@8053514 : Record 8053261;
      SetupFinded_L@8053513 : Boolean;
      recSalesLine_L@8053512 : Record 37;
      recContrat_L@8053511 : Record 37;
      recSalesHeader_L@8053509 : Record 36;
      recCmcSetup_L@8053502 : Record 8052982;
      LinkedOrderLine_L@8053501 : Record 37;
      recItem_L@8053000 : Record 27;
      recVariant_L@8053001 : Record 5401;
      CalcIndv_L@8053002 : Decimal;
      OldPrepaymentAmount_L@8053003 : Decimal;
      cduDefalq_L@8053100 : Codeunit 8053008;
    BEGIN
      //----------------------------------------------------------------------------------------------------------------------------------
      // Gestion de la livraison des surplus
      //----------------------------------------------------------------------------------------------------------------------------------
      //  Entr'es
      //    - QtyToShipVarr_P : quantit' . livrer
      //    - Var recSalesLineVar_P : Ligne de vente . mettre . jour
      //  Sorties
      //    - Result_L : Nouvelle ligne suite . acompte
      //----------------------------------------------------------------------------------------------------------------------------------
      //pas de traitement si la ligne est en LD ou commande sp'ciale
      Result_L := FALSE;
      IF recSalesLineVar_P.Type <> recSalesLineVar_P.Type::Item THEN
        EXIT(Result_L);

      // --> CMC-8052300/125|903487-NED
      IF recItem_L.GET(recSalesLineVar_P."No.") AND (recItem_L."Item Tracking Code"<>'')  AND
          (QtyToShipVar_P > recSalesLineVar_P.Quantity) THEN
        ERROR(Text8053503,recItem_L."No.");
      // <--CMC-8052300/125|903487-NED
      IF recSalesLineVar_P."Purchasing Code" <> '' THEN BEGIN
        IF recProcPurch_L.GET(recSalesLineVar_P."Purchasing Code") THEN BEGIN
          IF (recProcPurch_L."Drop Shipment" = TRUE) OR (recProcPurch_L."Special Order" = TRUE) THEN BEGIN
            IF recSalesLineVar_P.Quantity < recSalesLineVar_P."Quantity Shipped" + QtyToShipVar_P THEN BEGIN
              OldPrepaymentAmount_L := recSalesLineVar_P."Prepmt. Line Amount";
              recSalesLineVar_P.VALIDATE(Quantity, recSalesLineVar_P."Quantity Shipped" + QtyToShipVar_P);
              recSalesLineVar_P.MODIFY(TRUE);
            END;
            EXIT(Result_L);
          END;
        END;
      END;

      //Pas de nouvelle ligne en cas d'acompte
      recCmcSetup_L.GET;
      IF (recSalesLineVar_P."Prepmt. Line Amount" = 0) OR (recCmcSetup_L."Cancel Sales Prepayment Checks") THEN BEGIN
        recCmcSetup_L.GET;
        IF recCmcSetup_L."New Line On Exceed Qty To Ship" = TRUE THEN BEGIN
          IF (recSalesLineVar_P."Blanket Order No." <> '') AND (recSalesLineVar_P."Blanket Order Line No." <> 0) THEN BEGIN
            //la ligne est li'e . un contrat
            recContrat_L.RESET;
            recContrat_L.SETRANGE("Document Type", recContrat_L."Document Type"::"Blanket Order");
            recContrat_L.SETRANGE("Document No.", recSalesLineVar_P."Blanket Order No.");
            recContrat_L.SETRANGE("Line No.", recSalesLineVar_P."Blanket Order Line No.");
            IF recContrat_L.FINDFIRST THEN BEGIN
              ExtraQty_L := QtyToShipVar_P + recContrat_L."Quantity Shipped" - recContrat_L.Quantity;
              IF ExtraQty_L > 0 THEN BEGIN
                IF (recContrat_L.Quantity - recContrat_L."Quantity Shipped") = 0 THEN BEGIN
                  ERROR(Text8053501);
                END;
                PercentEcart_L := (ExtraQty_L / (recContrat_L.Quantity - recContrat_L."Quantity Shipped")) * 100;
                IF NOT recItem_L.GET(recContrat_L."No.") THEN
                  recItem_L.INIT;

                // --> 20160322-Gen14-ADV01.01.00-2767-BDN01
                // BEFORE
                {SetupFinded_L := SearchSetup(recSetup_L,FALSE,PercentEcart_L,recContrat_L."Unit of Measure Code",recContrat_L."No.",
                                              recContrat_L."Species Code",recContrat_L."Sub-Family Code",recContrat_L."Family Code",
                                              recContrat_L."Activity Code",recItem_L);}
                // AFTER
                SetupFinded_L := SearchSetup(recSetup_L, FALSE, PercentEcart_L, recContrat_L."Unit of Measure Code", recContrat_L."No.",
                                             recContrat_L."Species Code",recContrat_L."Sub-Family Code",recContrat_L."Family Code",
                                             recContrat_L."Activity Code",recItem_L,recSalesLineVar_P."Location Code",
                                             GetLocationType(recSalesLineVar_P."Location Code"));
                // <-- 20160322-Gen14-ADV01.01.00-2767-BDN01

                IF SetupFinded_L = TRUE THEN BEGIN
                  //un param'trage a 't' trouv'
                  IF recSetup_L.Processing = recSetup_L.Processing::Blocking THEN
                    //Il faut afficher un message d'erreur
                    ERROR(Text8053500, recSalesLineVar_P."No.", ((recContrat_L.Quantity - recContrat_L."Quantity Shipped") *
                      (1 + (recSetup_L."Max. % Difference" / 100))), recContrat_L."Unit of Measure Code");

                  IF recSetup_L.Processing = recSetup_L.Processing::"New Line" THEN BEGIN
                    //Il faut cr'er une nouvelle ligne
                    //La nouvelle ligne cr''e va prendre la totalit' du surplus
                    NewLineQty_L := ExtraQty_L;
                    IF recSetup_L."New Line Qty" = recSetup_L."New Line Qty"::"Simple Difference" THEN BEGIN
                      //La nouvelle ligne cr''e va prendre la quantit' au del. du max
                      NewLineQty_L := NewLineQty_L - ((recContrat_L.Quantity - recContrat_L."Quantity Shipped") *
                                                     (recSetup_L."Max. % Difference" / 100));
                    END;
                    //on met . jour la quantit' . livr'e sur la ligne initiale
                    QtyToShipVar_P := QtyToShipVar_P - NewLineQty_L;

                    IF recSalesLineVar_P."Variant Code" <> '' THEN BEGIN
                      IF recVariant_L.GET(recSalesLineVar_P."No.", recSalesLineVar_P."Variant Code") THEN BEGIN
                        IF recVariant_L."Indiv. Base Qty" > 0 THEN BEGIN
                          CalcIndv_L := ROUND(QtyToShipVar_P / recVariant_L."Indiv. Base Qty", 1, '=');
                          NewLineQty_L := NewLineQty_L + QtyToShipVar_P - (CalcIndv_L * recVariant_L."Indiv. Base Qty");
                          QtyToShipVar_P := CalcIndv_L * recVariant_L."Indiv. Base Qty";
                        END;
                      END;
                    END
                    ELSE BEGIN
                      IF recItem_L.GET(recSalesLineVar_P."No.") THEN BEGIN
                        IF recItem_L."Indiv. Base Qty" > 0 THEN BEGIN
                          CalcIndv_L := ROUND(QtyToShipVar_P / recItem_L."Indiv. Base Qty", 1, '=');
                          NewLineQty_L := NewLineQty_L + QtyToShipVar_P - (CalcIndv_L * recItem_L."Indiv. Base Qty");
                          QtyToShipVar_P := CalcIndv_L * recItem_L."Indiv. Base Qty";
                        END;
                      END;
                    END;
                    CreateNewLine(recSalesLineVar_P, recSalesLine_L, NewLineQty_L);
                    UpdateWhse(recSalesLineVar_P, NewLineQty_L, recSalesLine_L);
                  END;
                END;
              END;
            END;
          END
          ELSE BEGIN
            //la ligne est sans contrat
            ExtraQty_L := QtyToShipVar_P + recSalesLineVar_P."Quantity Shipped" - recSalesLineVar_P.Quantity;
            IF ExtraQty_L > 0 THEN BEGIN
              IF (recSalesLineVar_P.Quantity - recSalesLineVar_P."Quantity Shipped") = 0 THEN BEGIN
                ERROR(Text8053501);
              END;
              PercentEcart_L := (ExtraQty_L / (recSalesLineVar_P.Quantity - recSalesLineVar_P."Quantity Shipped")) * 100;
              IF NOT recItem_L.GET(recSalesLineVar_P."No.") THEN
                recItem_L.INIT;

              // --> 20160322-Gen14-ADV01.01.00-2767-BDN01
              //
              // BEFORE
              {SetupFinded_L := SearchSetup(recSetup_L, TRUE, PercentEcart_L, recSalesLineVar_P."Unit of Measure Code",
                                              recSalesLineVar_P."No.",recSalesLineVar_P."Species Code",recSalesLineVar_P."Sub-Family Code"
      ,
                                            recSalesLineVar_P."Family Code", recSalesLineVar_P."Activity Code", recItem_L);}
              // AFTER
              SetupFinded_L := SearchSetup(recSetup_L, TRUE, PercentEcart_L, recSalesLineVar_P."Unit of Measure Code",
                recSalesLineVar_P."No.", recSalesLineVar_P."Species Code", recSalesLineVar_P."Sub-Family Code",
                                           recSalesLineVar_P."Family Code", recSalesLineVar_P."Activity Code", recItem_L,
                                           recSalesLineVar_P."Location Code",GetLocationType(recSalesLineVar_P."Location Code"));
              // <-- 20160322-Gen14-ADV01.01.00-2767-BDN01

              IF SetupFinded_L = TRUE THEN BEGIN
                //un param'trage a 't' trouv'
                IF recSetup_L.Processing = recSetup_L.Processing::Blocking THEN
                  //Il faut afficher un message d'erreur
                  ERROR(Text8053500, recSalesLineVar_P."No.", ((recSalesLineVar_P.Quantity - recSalesLineVar_P."Quantity Shipped") *
                    (1 + (recSetup_L."Max. % Difference" / 100))), recSalesLineVar_P."Unit of Measure Code");
                IF recSetup_L.Processing = recSetup_L.Processing::"New Line" THEN BEGIN
                  //Il faut cr'er une nouvelle ligne
                  //La nouvelle ligne cr''e va prendre la totalit' du surplus
                  NewLineQty_L := ExtraQty_L;
                  IF recSetup_L."New Line Qty" = recSetup_L."New Line Qty"::"Simple Difference" THEN BEGIN
                    //La nouvelle ligne cr''e va prendre la quantit' au del. du max
                    NewLineQty_L := NewLineQty_L - ((recSalesLineVar_P.Quantity - recSalesLineVar_P."Quantity Shipped") *
                                                   (recSetup_L."Max. % Difference" / 100));
                  END;
                  //on met . jour la quantit' . livr'e sur la ligne initiale
                  QtyToShipVar_P := QtyToShipVar_P - NewLineQty_L;

                  IF recSalesLineVar_P."Variant Code" <> '' THEN BEGIN
                    IF recVariant_L.GET(recSalesLineVar_P."No.", recSalesLineVar_P."Variant Code") THEN BEGIN
                      IF recVariant_L."Indiv. Base Qty" > 0 THEN BEGIN
                        CalcIndv_L := ROUND(QtyToShipVar_P / recVariant_L."Indiv. Base Qty", 1, '=');
                        NewLineQty_L := NewLineQty_L + QtyToShipVar_P - (CalcIndv_L * recVariant_L."Indiv. Base Qty");
                        QtyToShipVar_P := CalcIndv_L * recVariant_L."Indiv. Base Qty";
                      END;
                    END;
                  END
                  ELSE BEGIN
                    IF recItem_L.GET(recSalesLineVar_P."No.") THEN BEGIN
                      IF recItem_L."Indiv. Base Qty" > 0 THEN BEGIN
                        CalcIndv_L := ROUND(QtyToShipVar_P / recItem_L."Indiv. Base Qty", 1, '=');
                        NewLineQty_L := NewLineQty_L + QtyToShipVar_P - (CalcIndv_L * recItem_L."Indiv. Base Qty");
                        QtyToShipVar_P := CalcIndv_L * recItem_L."Indiv. Base Qty";
                      END;
                    END;
                  END;
                  CreateNewLine(recSalesLineVar_P, recSalesLine_L, NewLineQty_L);
                  UpdateWhse(recSalesLineVar_P, NewLineQty_L, recSalesLine_L);
                END;
              END;
            END;
          END;
        END;
      END
      ELSE BEGIN
        ExtraQty_L := 0;
        IF recSalesLineVar_P."Linked Order No." <> '' THEN BEGIN
          IF LinkedOrderLine_L.GET(recSalesLineVar_P."Document Type",
              recSalesLineVar_P."Linked Order No.", recSalesLineVar_P."Linked Order Line No.") THEN BEGIN
            //la ligne a un acompte
            ExtraQty_L := QtyToShipVar_P + LinkedOrderLine_L.Quantity - LinkedOrderLine_L."Outstanding Quantity";
            IF ExtraQty_L > recSalesLineVar_P.Quantity THEN BEGIN
              ExtraQty_L := QtyToShipVar_P - LinkedOrderLine_L."Outstanding Quantity";
            END;
          END;
        END
        ELSE BEGIN
          //la ligne est un acompte
          ExtraQty_L := QtyToShipVar_P - recSalesLineVar_P."Outstanding Quantity";
        END;
        IF ExtraQty_L > 0 THEN BEGIN
          //Il y a un surplus, il faut splitter la ligne avec la totalit' du surplus
          Result_L := TRUE;
          NewLineQty_L := ExtraQty_L;
          //on met . jour la quantit' . livr'e sur la ligne initiale
          QtyToShipVar_P := QtyToShipVar_P - NewLineQty_L;

          CreateNewLine(recSalesLineVar_P, recSalesLine_L, NewLineQty_L);
          UpdateWhse(recSalesLineVar_P, NewLineQty_L, recSalesLine_L);
        END;
      END;

      EXIT(Result_L);
    END;

    PROCEDURE UpdateWhse@8053004(recSalesLine_P@8053109 : Record 37;NewLineQty_P@8053110 : Decimal;recNewSalesLine_P@8053000 : Record 37);
    VAR
      recWhsShipLine_L@8053111 : Record 7321;
      cduReleaseWhseShptDoc_L@8053112 : Codeunit 7310;
      recWhsShipHeader_L@8053113 : Record 7320;
      recWhsShipLine2_L@8053114 : Record 7321;
      NewWhsLineNo_L@8053115 : Integer;
      OldLineQtyToShip_L@8053116 : Decimal;
    BEGIN
      //----------------------------------------------------------------------------------------------------------------------------------
      // Insertion nouvelle ligne de livraison et mise . jour de celle existante
      //----------------------------------------------------------------------------------------------------------------------------------
      //  Entr'es
      //    - recSalesLine_P : Ligne de commande . livrer
      //    - NewLineQty_P : Nouvelle quantit' de livraison calcul'e pr'c'demment
      //    - recNewSalesLine_P : Ligne d'j. cr''e
      //  Sorties
      //----------------------------------------------------------------------------------------------------------------------------------
      recWhsShipLine_L.RESET;
      recWhsShipLine_L.SETRANGE("Source Type", DATABASE::"Sales Line");
      recWhsShipLine_L.SETRANGE("Source No.", recSalesLine_P."Document No.");
      recWhsShipLine_L.SETRANGE("Source Line No.", recSalesLine_P."Line No.");
      IF recWhsShipLine_L.FINDFIRST THEN BEGIN
        //recherche du nouveau num'ro de ligne
        NewWhsLineNo_L := recWhsShipLine_L."Line No." + 10000;
        recWhsShipLine2_L.RESET;
        recWhsShipLine2_L.SETRANGE("No.", recWhsShipLine_L."No.");
        recWhsShipLine2_L.SETFILTER("Line No.", '>%1', recWhsShipLine_L."Line No.");
        IF recWhsShipLine2_L.FINDLAST THEN
          NewWhsLineNo_L := recWhsShipLine2_L."Line No." + 10000;

        recWhsShipHeader_L.GET(recWhsShipLine_L."No.");
        cduReleaseWhseShptDoc_L.Reopen(recWhsShipHeader_L);
        OldLineQtyToShip_L := recWhsShipLine_L."Qty. to Ship" - NewLineQty_P;
        recWhsShipLine_L.VALIDATE("Qty. to Ship", OldLineQtyToShip_L);
        recWhsShipLine_L.VALIDATE("Qty. to Ship (Base)", OldLineQtyToShip_L);
        recWhsShipLine_L.MODIFY(TRUE);
        recWhsShipLine2_L.INIT;
        recWhsShipLine2_L.TRANSFERFIELDS(recWhsShipLine_L);
        recWhsShipLine2_L."No." := recWhsShipLine_L."No.";
        recWhsShipLine2_L."Line No." := NewWhsLineNo_L;
        recWhsShipLine2_L."Source Type" := recWhsShipLine2_L."Source Type";
        recWhsShipLine2_L."Source Subtype" := recWhsShipLine2_L."Source Subtype";
        recWhsShipLine2_L."Source No." := recWhsShipLine2_L."Source No.";
        recWhsShipLine2_L."Source Line No." := recNewSalesLine_P."Line No.";
        recWhsShipLine2_L.Quantity := NewLineQty_P;
        recWhsShipLine2_L.VALIDATE(Quantity, NewLineQty_P);
        recWhsShipLine2_L.VALIDATE("Qty. (Base)", NewLineQty_P);
        recWhsShipLine2_L.VALIDATE("Qty. to Ship", NewLineQty_P);
        recWhsShipLine2_L.VALIDATE("Qty. to Ship (Base)", NewLineQty_P);
        recWhsShipLine2_L.INSERT(TRUE);
        cduReleaseWhseShptDoc_L.Release(recWhsShipHeader_L);
      END;
    END;
