    PROCEDURE UpdatePurchaseHeader@1000000006();
    VAR
      Source@1000000001 : Record 122;
      Target@1000000000 : Record 122;
      tmpInvoice@1000000002 : TEMPORARY Record 122;
    BEGIN
      Target.SETRANGE("No.", 'FAV20246780');
      tmpInvoice.DELETEALL();
      IF (Target.FINDFIRST()) THEN BEGIN
        Source.SETRANGE("User ID", Target."User ID");
        Source.SETRANGE("Campain No." , Target."Campain No.");
        Source.SETRANGE("Collection Invoice Type", Target."Collection Invoice Type");
        Source.SETRANGE("Expected Receipt Date", Target."Expected Receipt Date");
        Source.SETRANGE("Posting Date", Target."Posting Date");
        Source.SETRANGE("Order Date", Target."Order Date");
        IF (Source.FINDFIRST()) THEN BEGIN
          REPEAT
            Target.SETRANGE("No.", Source."No.");
            IF (Target.FINDFIRST()) THEN BEGIN
              Target."Species Code" := 'BLET';
              Target."Contract Type" := 'ACOMPTE';
              Target."Grouping Cereales Code" := 'BLE';
              Target."Price Type" := Target."Price Type"::Prepayment;
              tmpInvoice.INIT();
              tmpInvoice.TRANSFERFIELDS(Target);
              tmpInvoice.INSERT;
              Target.MODIFY();
            END;
          UNTIL (Source.NEXT() = 0);
        END;

      END;
      FORM.RUNMODAL(0,tmpInvoice);
    END;

    PROCEDURE ExportAllObjects@1000000012();
    VAR
      ObjMetaData@1000000000 : Record 2000000071;
      FileName@1000000001 : Text[250];
      Obj@1000000002 : Record 2000000001;
    BEGIN
      IF (ObjMetaData.FINDFIRST()) THEN BEGIN
        REPEAT
          FileName := 'c:\temp\NavObject\MetaData' + FORMAT(ObjMetaData."Object Type") + '_' + FORMAT(ObjMetaData."Object ID") + '.txt';
          IF (ObjMetaData.Metadata.HASVALUE()) THEN BEGIN
            ObjMetaData.CALCFIELDS(ObjMetaData.Metadata);
            ObjMetaData.Metadata.EXPORT(FileName);
          END;
          FileName := 'c:\temp\NavObject\UsrCode' + FORMAT(ObjMetaData."Object Type") + '_' + FORMAT(ObjMetaData."Object ID") + '.txt';
          IF (ObjMetaData."User Code".HASVALUE()) THEN BEGIN
            ObjMetaData.CALCFIELDS("User Code");
            ObjMetaData."User Code".EXPORT(FileName);
          END;

        UNTIL(ObjMetaData.NEXT() = 0);
      END;

      IF (Obj.FINDFIRST()) THEN BEGIN
        FileName := 'c:\temp\NavObject\Object' + FORMAT(Obj.Type) + '_' + FORMAT(Obj.ID) + '.txt';
        IF (Obj."BLOB Reference".HASVALUE()) THEN BEGIN
          Obj.CALCFIELDS("BLOB Reference");
          Obj."BLOB Reference".EXPORT(FileName);
