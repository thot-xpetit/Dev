    OnRun=VAR
            Setup_L@1000000000 : Record 8052982;
            RecRef@1000000002 : RecordRef;
            xRecRef@1000000001 : RecordRef;
          BEGIN
            Setup_L := Rec;
            Setup_L.LOCKTABLE;
            Setup_L.FIND;

          END;

    PROCEDURE CheckModification@3(VAR RecRef@1000 : RecordRef;VAR xRecRef@1001 : RecordRef) Boo_R : Boolean;
    VAR
      FldRef@1003 : FieldRef;
      xFldRef@1004 : FieldRef;
      i@1002 : Integer;
      Field_L@1000000000 : Record 2000000041;
    BEGIN
      Boo_R := TRUE;
      FOR i := 1 TO RecRef.FIELDCOUNT DO BEGIN
        FldRef := RecRef.FIELDINDEX(i);
        xFldRef := xRecRef.FIELDINDEX(i);
        Field_L.GET(RecRef.NUMBER,FldRef.NUMBER);
        IF Field_L.Class =Field_L.Class::Normal THEN
          IF FORMAT(FldRef.VALUE) <> FORMAT(xFldRef.VALUE) THEN BEGIN
            IF NOT CONFIRM(Text60000,FALSE,Field_L."Field Caption",xFldRef.VALUE,FldRef.VALUE) THEN BEGIN
              Boo_R := FALSE;
              EXIT;
            END;
           END;
      END;
    END;
