      Msg@1000000004 : Text[1024];
      tErrLineNo@1000000005 : TextConst 'ENU=At line : %1 in the file : %2;FRA=A la ligne : %1 dans le fichier : %2';
    BEGIN
      Msg := ErrMsg;
      {
      IF ((LineNo <> 0) AND (FileName <> '')) THEN BEGIN
        Msg := Msg + ' ' + STRSUBSTNO(tErrLineNo, LineNo, FileName);
      END;
      }
      CASE (LogType) OF
        LogType::Info : BEGIN
          ToolsFile_G.LogWriteInfo(Msg);
        END;
        LogType::Error : BEGIN
          ToolsFile_G.LogWriteErreur(Msg);
        END;
      END;
    END;
