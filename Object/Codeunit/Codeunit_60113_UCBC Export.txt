    PROCEDURE ConvText@1000000003(Text_P@1000000000 : Text[1024]) Text_L : Text[1024];
    VAR
      Char_L@1000000001 : Char;
      Len_L@1000000002 : Integer;
      Pos_L@1000000003 : Integer;
    BEGIN
      Len_L := STRLEN(Text_P);
      WHILE Pos_L < Len_L DO BEGIN
        EVALUATE(Char_L,COPYSTR(Text_P,Pos_L+1,1));
        IF ((Char_L >= 48) AND (Char_L <= 57)) OR
           ((Char_L >= 65) AND (Char_L <= 90)) OR
           ((Char_L >= 97) AND (Char_L <= 122)) THEN
          Text_L := Text_L + FORMAT(Char_L);

        Pos_L +=1;
      END;
    END;

    PROCEDURE ConvDec2Text@1000000006(Value_P@1000000000 : Decimal) Text_L : Text[1024];
    VAR
      Value_L@1000000001 : Decimal;
      Text2_L@1000000002 : Text[30];
    BEGIN
      Value_L := ABS(ROUND(Value_P * 1000 ,1) MOD 1000);
      Text2_L := FORMAT(Value_L,0,'<Integer>') ;
      IF Value_L < 10 THEN
        Text2_L := '00' + Text2_L
      ELSE IF Value_L < 100 THEN
        Text2_L := '0' + Text2_L;

      Text_L := FORMAT(Value_P,0,'<Integer>,') + Text2_L;
      Text_L := Text_L + FORMAT(Value_P,0,'<Sign>') ;
    END;
