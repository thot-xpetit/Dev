            TempPostedReceiptsLine_L.RESET;

            RecPostedReceiptsLine_L.SETRANGE("Prepay CreditMemo No.",'');
            RecPostedReceiptsLine_L.SETRANGE(Quantity,Quantity);
            IF RecPostedReceiptsLine_L.FINDFIRST THEN BEGIN
               // Affectation du nouveau contrat (stock' dans Pre Contract No.)
               RecPostedReceiptsLine_L.VALIDATE("Contract No.","Pre Contract No.");
               RecPostedReceiptsLine_L.MODIFY;
            END;

            // Verrouillage du bon de r'ception
            RecPostedReceipts_L.GET("Receipt Type","No.");
            CduStatusMgt_L.ReleasePostedReceipt(RecPostedReceipts_L);

            // Copie du r'sultat dans Rec pour retourner les donn'es modifi'es
            Rec := RecPostedReceiptsLine_L;
