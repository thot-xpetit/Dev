    PROCEDURE Export@1000000000(FileName_P@1000000019 : Text[256]) Result_L : Text[256];
    VAR
      "<-- 3LI LOCALS -->"@1000000002 : Integer;
      ToolsFile_L@1000000001 : Codeunit 8053405;
      TableMajCdeFermatel_L@1000000000 : Record 8053410;
      TableSalesHead_L@1000000008 : Record 36;
      TableSalesLine_L@1000000013 : Record 37;
      Connector_L@1000000012 : Code[20];
      FileName_L@1000000020 : Text[256];
      FileTemp_L@1000000027 : Text[256];
      FilePath_L@1000000028 : Text[256];
      FileN_L@1000000029 : Text[256];
      ExtensionFile_L@1000000011 : Text[10];
      PrefixFile_L@1000000010 : Text[10];
      ItemTemplateCode_L@1000000009 : Code[20];
      CustomerTemplateCode_L@1000000007 : Code[20];
      VendorTemplateCode_L@1000000006 : Code[20];
      ContactTemplateCode_L@1000000005 : Code[20];
      AccountTemplateCode_L@1000000004 : Code[20];
      GlobalTemplateCode_L@1000000003 : Code[20];
      ConnectorName_L@1000000018 : Code[20];
      ImpExpFolder_L@1000000017 : Text[250];
      LogFolder_L@1000000016 : Text[250];
      FileLogName_L@1000000015 : Text[250];
      FileExport_L@1000000022 : File;
      FileOutstream_L@1000000021 : OutStream;
      FileExportOpen_L@1000000023 : Boolean;
      w_Horo_L@1000000024 : Text[50];
      Tableparameter_L@1000000025 : Record 8053409;
      Status_L@1000000026 : Text[1];
      Wqte_L@1000000030 : Decimal;
      wdliv_L@1000000031 : Date;
      numnavcde_old_L@1000000032 : Code[20];
      TxtMessage_L@1000000033 : Text[256];
    BEGIN
      //----------------------------------------------------------------------------------------------------------------------------------
      // Maj Zone  Accuse Reception Cde  Fermatel  .
      //----------------------------------------------------------------------------------------------------------------------------------
      MoteurInterface_G.GetParameter(CodeUnit_G, ConnectorName_L, ImpExpFolder_L, LogFolder_L, ExtensionFile_L, PrefixFile_L);
      MoteurInterface_G.GetTemplatesCode(ConnectorName_L,ItemTemplateCode_L,CustomerTemplateCode_L,VendorTemplateCode_L,
        ContactTemplateCode_L, AccountTemplateCode_L, GlobalTemplateCode_L);
      ToolsFile_L.LogInit(LogFolder_L, 'FERMATEL');
      ToolsFile_L.LogInitCodeUnit(CodeUnit_G);

      w_Horo_L := FORMAT(CURRENTDATETIME, 0, '<Year4>-<Month,2>-<Day,2><Hours24,2><Minutes,2><Seconds,2><Second dec>');
      //-->20150630-Int09-CMC02.02.00-ADVITAM-CDL001
      w_Horo_L := ToolsFile_L.TexteRemplaceCar(w_Horo_L, ',', '_');
      //<--20150630-Int09-CMC02.02.00-ADVITAM-CDL001

      // Nom du fichier pass' en paramStre ou Nom constitu' par r'p. du Robot & Nom fic. r'cup'r' du robot
      IF FileName_P <> '' THEN BEGIN
        FileName_L := FileName_P;
      END ELSE BEGIN
        FileName_L := ImpExpFolder_L + '\' + DELCHR(PrefixFile_L, '<>', ' ') + w_Horo_L + '.' + ExtensionFile_L;
      END;
      FileExportOpen_L:=FALSE;

      ToolsFile_L.LogInitFileExecute(FileName_L);

      IF NOT Tableparameter_L.FINDFIRST THEN BEGIN
        ToolsFile_L.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0001, Tableparameter_L.TABLECAPTION , COMPANYNAME), 1, MAXSTRLEN(Result_L)));
      END ELSE BEGIN
        IF Tableparameter_L."Accused Receipt" THEN BEGIN
          // on boucle sur tous les enreg sans accuse de reception  esr
          TableMajCdeFermatel_L.SETCURRENTKEY(TableMajCdeFermatel_L."Document Type",TableMajCdeFermatel_L.NumNavCde);
          TableMajCdeFermatel_L.SETRANGE(TableMajCdeFermatel_L."Document Type",TableMajCdeFermatel_L."Document Type"::Order);
          TableMajCdeFermatel_L.SETRANGE(TableMajCdeFermatel_L.Accus'Recep,' ');
          IF TableMajCdeFermatel_L.FINDSET() THEN BEGIN
            // Rupture commande
            REPEAT
              IF numnavcde_old_L <> TableMajCdeFermatel_L.NumNavCde THEN BEGIN
                CLEAR(Status_L);
                CLEAR(Wqte_L);
                CLEAR(wdliv_L);
                IF NOT TableSalesHead_L.GET(TableMajCdeFermatel_L."Document Type",TableMajCdeFermatel_L.NumNavCde) THEN BEGIN

                  Status_L :='5';

                END ELSE BEGIN
                  IF TableSalesHead_L.Status = TableSalesHead_L.Status::Released THEN BEGIN
                    Status_L := '2';

                    TableSalesLine_L.SETRANGE(TableSalesLine_L."Document Type", TableMajCdeFermatel_L."Document Type");
                    TableSalesLine_L.SETRANGE(TableSalesLine_L."Document No.", TableMajCdeFermatel_L.NumNavCde);
                    TableSalesLine_L.SETRANGE(TableSalesLine_L.Type, TableSalesLine_L.Type::Item);
                    IF TableSalesLine_L.FINDSET() THEN BEGIN
                      REPEAT
                        IF TableSalesLine_L."Quantity before delivery" <> 0 THEN BEGIN
                          Wqte_L +=TableSalesLine_L."Quantity before delivery";
                        END ELSE BEGIN
                          Wqte_L +=TableSalesLine_L.Quantity;
                        END;
                          wdliv_L := TableSalesLine_L."Requested Delivery Date";
                      UNTIL  TableSalesLine_L.NEXT = 0;
                      IF Wqte_L = 0 THEN BEGIN
                         Status_L :='2';
                      END;
                    END;
                  END;
                END;

                // Ecrire fichier Texte
                //  ***
                IF Status_L <> '' THEN BEGIN
                // creation avec status a 5 Si Inexistant ligne cde
                  IF FileExportOpen_L= FALSE THEN BEGIN
                    ToolsFile_L.SplitFilename(FileName_L, FilePath_L, FileN_L);
                    FileTemp_L := FilePath_L + 'Temp_' + FileN_L;
                    TxtMessage_L := ToolsFile_L.FileCreateNotExist(FileName_L, 'Temp_', FileExport_L);
                    IF TxtMessage_L <> '' THEN BEGIN
                      ToolsFile_L.LogWriteErreur(COPYSTR(TxtMessage_L, 1, 256));
                    END ELSE BEGIN
                      FileExport_L.CREATEOUTSTREAM(FileOutstream_L);
                      FileExportOpen_L := TRUE;
                    END;
                  END;
                  IF FileExportOpen_L THEN BEGIN
                  // creation Enreg fichier txt
                    FileOutstream_L.WRITETEXT(PADSTR(TableMajCdeFermatel_L.Zlot, 10, '0'), 10);
                    FileOutstream_L.WRITETEXT(PADSTR(TableMajCdeFermatel_L.Zelev, 6, ' '), 6);
                    FileOutstream_L.WRITETEXT(PADSTR('0', 5, '0'), 5);
                    FileOutstream_L.WRITETEXT(PADSTR(FORMAT(TableMajCdeFermatel_L.Zdatsai,
                     0,'<Year4><Month,2><Day,2><Hours24,2><Minutes,2><Seconds,2>'),14 , '0'), 14);
                    FileOutstream_L.WRITETEXT(PADSTR(TableMajCdeFermatel_L.Zalima, 5, ' '), 5);
                    FileOutstream_L.WRITETEXT(PADSTR(TableMajCdeFermatel_L.Zalim, 5, ' '), 5);
                    FileOutstream_L.WRITETEXT(FORMAT(Wqte_L, 0,
                      '<Precision,3:3><Sign><Integer,3><Filler Character,0><Decimals><Comma,.>'), 7);
                       FileOutstream_L.WRITETEXT(PADSTR(FORMAT(wdliv_L, 0,'<Year4><Month,2><Day,2>'),14 , '0'), 14);
                    FileOutstream_L.WRITETEXT(PADSTR(Status_L,1,' ' ),1);
                    FileOutstream_L.WRITETEXT(PADSTR(TableMajCdeFermatel_L.Zcdnum,10, ' '), 10);
                    FileOutstream_L.WRITETEXT();
                  END;
                   // ***
                END;

                numnavcde_old_L := TableMajCdeFermatel_L.NumNavCde;

              END;
              // Maj Accuse Fichier import Cde
              IF Status_L <> '' THEN BEGIN
                TableMajCdeFermatel_L.Accus'Recep := Status_L;
                TableMajCdeFermatel_L.DateAccusedOrder := TODAY;
                IF NOT TableMajCdeFermatel_L.MODIFY(TRUE) THEN BEGIN
                  ToolsFile_L.LogWriteErreur(STRSUBSTNO(Text0010, TableMajCdeFermatel_L.TABLECAPTION,TableMajCdeFermatel_L.Zlot));
                END;
              END;
            UNTIL TableMajCdeFermatel_L.NEXT = 0;

            IF FileExportOpen_L THEN BEGIN
              FileExport_L.CLOSE;
              IF ToolsFile_L.LogErreurType() <> 'E' THEN BEGIN
                IF RENAME(FileTemp_L, FileName_L) THEN BEGIN
                END;
                IF NOT ToolsFile_L.FileCopySav(FileName_L, 'Sav', TRUE) THEN BEGIN
                  Result_L := STRSUBSTNO(Text0011, FileName_L);
                  ToolsFile_L.LogWriteErreur(Result_L);
                END;
              END;
            END;

            FileLogName_L := ToolsFile_L.LogName;
            ToolsFile_L.LogClose();
            IF ToolsFile_L.LogErreurType() = 'E' THEN BEGIN
              Result_L := 'ERREUR ' + FileLogName_L;
              IF NOT ERASE(FileTemp_L) THEN BEGIN
              END;
            END ELSE BEGIN
              IF ToolsFile_L.LogErreurType() = 'I' THEN BEGIN
                Result_L := Text0004;
                MoteurInterface_G.InsertLog(CodeUnit_G, FileLogName_L);
              END ELSE BEGIN
                MoteurInterface_G.InsertOk(CodeUnit_G);
              END;
            END;
          END;
          Tableparameter_L."Last Date Accuse Reception" := CURRENTDATETIME;
          IF NOT Tableparameter_L.MODIFY(TRUE) THEN BEGIN
            ToolsFile_L.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0003,Tableparameter_L.TABLECAPTION,COMPANYNAME),1,MAXSTRLEN(Result_L)));
          END;
        END;
      END;
      //<--20130924-Int09-CMC02.02.00-ADVITAM-ESR
    END;

      //<-- 3LI END
