    LOCAL PROCEDURE CreateMailMergeHeader@3(StyleSheetCode@1016 : Code[20];VAR wrdMergefile@1011 : Automation "{3C2554FE-2B41-4543-B3E8-E64296773DA5} 2.1:{40D88563-F0DA-4B3D-AD1B-34EFA8C3978A}:Unknown Automation Server.MergeHandler";MergeFieldsOnly@1000 : Boolean;MergeFileName@1001 : Text[260]);
    VAR
      I@1007 : Integer;
      StyleSheetHeader@1003 : Record 680;
      StyleSheetTables@1013 : Record 681;
      StyleSheetFields@1002 : Record 682;
      RecRef@1014 : RecordRef;
      FldRef@1015 : FieldRef;
      AddFieldText@1006 : Text[100];
      FieldCount@1004 : Integer;
    BEGIN
      IF NOT wrdMergefile.CreateFile(MergeFileName) THEN
        ERROR(Text005 + Text006);

      // Create HTML Header source
      WITH wrdMergefile DO BEGIN
        StyleSheetTables.SETRANGE("Style Sheet Code",StyleSheetCode);
        IF StyleSheetTables.FINDFIRST THEN BEGIN
          REPEAT
            // Add Merge Field for Multi-Line Begin
            IF StyleSheetTables."Multiple Lines" THEN BEGIN
              AddField(Text020 + ' ' + StyleSheetFunctions.StdName(StyleSheetTables."Table Name",TRUE));
              FieldCount += 1;
            END;

            StyleSheetFields.SETRANGE("Style Sheet Code",StyleSheetTables."Style Sheet Code");
            StyleSheetFields.SETRANGE("Table No.",StyleSheetTables."Table No.");

            // Add CAPTION Merge Fields for Selected Fields
            StyleSheetFields.SETRANGE("Include Caption",TRUE);
            IF StyleSheetFields.FINDFIRST THEN BEGIN
              RecRef.OPEN(StyleSheetFields."Table No.");
              REPEAT
                IF StyleSheetFields."Include Caption" THEN BEGIN
                  FldRef := RecRef.FIELD(StyleSheetFields."Field No.");
                  AddFieldText := RecRef.NAME + ' ' + FldRef.NAME;
                  AddField(Text022 + ' ' + AddFieldText);
                  FieldCount += 1;
                END;
              UNTIL (StyleSheetFields.NEXT = 0);
              RecRef.CLOSE;
            END;

            // Add Merge Fields
            StyleSheetFields.SETRANGE("Include Caption");
            IF StyleSheetFields.FINDFIRST THEN BEGIN
              RecRef.OPEN(StyleSheetFields."Table No.");
              REPEAT
                FldRef := RecRef.FIELD(StyleSheetFields."Field No.");
                AddFieldText := RecRef.NAME + ' ' + FldRef.NAME;
                StyleSheetTables.CALCFIELDS(Relationships);
                IF StyleSheetTables.Relationships AND NOT StyleSheetTables."Multiple Lines" THEN
                  AddField(Text023 + ' ' + AddFieldText)
                ELSE
                  AddField(AddFieldText);
                FieldCount += 1;
              UNTIL (StyleSheetFields.NEXT = 0);
              RecRef.CLOSE;
            END;

            // Add Merge Fields For Multi-Line End
            IF StyleSheetTables."Multiple Lines" THEN BEGIN
              AddField(Text021 + ' ' + StyleSheetFunctions.StdName(StyleSheetTables."Table Name",TRUE));
              FieldCount += 1;
            END;

          UNTIL (StyleSheetTables.NEXT = 0);
        END;

        // Add General Information
        StyleSheetHeader.GET(StyleSheetCode);
        IF StyleSheetHeader."Include Data Link" THEN BEGIN
          AddField('DataLink');
          FieldCount += 1;
        END;

        IF StyleSheetHeader."Include Work Date" THEN BEGIN
          AddField('WorkDate');
          FieldCount += 1;
        END;

        WriteLine;

        // Mergesource must be at least two lines
        IF MergeFieldsOnly THEN BEGIN
          FOR I := 1 TO FieldCount DO
            AddField('');
          WriteLine;
          CloseFile;
        END;
      END;
    END;

      XMLDoc.save(SaveStyleSheetStream);
      DOMError(XMLDoc);
      StylesheetHeader.MODIFY;
      COMMIT;

      DOMError(XMLDoc);

      CLEAR(XMLDoc);
      CLEAR(MailMergeXMLDoc);

      Status.CLOSE;

      IF ShowProcessingMessages THEN BEGIN
        IF CONFIRM(Text011,FALSE) THEN BEGIN
          StylesheetHeader."Version No." += 1;
          StylesheetHeader.MODIFY;
          StyleSheetFunctions.SendStyleSheet(StylesheetHeader);
        END;
      END ELSE BEGIN
        StylesheetHeader."Version No." += 1;
        StylesheetHeader.MODIFY;
        StyleSheetFunctions.SendStyleSheet(StylesheetHeader);
      END;
    END;
