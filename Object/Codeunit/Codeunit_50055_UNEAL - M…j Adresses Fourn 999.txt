    OnRun=BEGIN

            //===============================================================================================================================
            //                    M.j Table 8053221  Adresses D'part des Fournisseurs
            //===============================================================================================================================

            // 17/09/2015 cr'ation des adresses 999 si inexistantes, pour Fournisseurs non Exploitation Agricole.
            Cpt  := 0;
            Cpt2 := 0;
            Fournisseur.RESET;
            Total := Fournisseur.COUNT;
            Fenetre.OPEN('Maj information Fournisseur ...        \' +
                         'Nb lecture fourn                #1######\' +
                         'Nb cr'ation . 999               #2######\' +
                         '@4@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@',Cpt,Cpt2);

            IF Fournisseur.FIND('-') THEN
            REPEAT

              EVALUATE(Code_Num,Fournisseur."No.");
              IF Code_Num < 10000000 THEN BEGIN
                IF Contact.GET(Fournisseur."No.") THEN BEGIN
                  IF Contact."Contact Type" <> Contact."Contact Type"::Farm THEN BEGIN

                    Code10_G := '999' ;
                    IF NOT t8053221.GET(Fournisseur."No.",Code10_G) THEN BEGIN
                      t8053221.INIT;
                      t8053221."Vendor No." := Fournisseur."No.";
                      t8053221.Code := Code10_G;
                      t8053221.Name := Fournisseur.Name;
                      t8053221.Address := 'ARRAS.' ;
                      t8053221."Post code" := '62000';
                      t8053221.City := 'ARRAS';

                      t8053221.INSERT;
                      Cpt2 := Cpt2 +1;
                    END;
                  END;
                END;
              END;

              Cpt := Cpt + 1;

              Fenetre.UPDATE;
              Fenetre.UPDATE(1,Cpt);
              Fenetre.UPDATE(2,Cpt2);

              Fenetre.UPDATE(4,ROUND(Cpt / Total * 10000, 1));

            UNTIL Fournisseur.NEXT = 0;
            Fenetre.CLOSE;
          END;

  }
