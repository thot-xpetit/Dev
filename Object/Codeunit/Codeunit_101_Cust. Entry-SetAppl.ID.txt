    PROCEDURE SetApplId@1(VAR CustLedgEntry@1000 : Record 21;ApplyingCustLedgEntry@1003 : Record 21;AppliedAmount@1004 : Decimal;PmtDiscAmount@1002 : Decimal;AppliesToID@1001 : Code[20]);
    BEGIN
      CustLedgEntry.LOCKTABLE;
      IF CustLedgEntry.FIND('-') THEN BEGIN
        // Make Applies-to ID
        IF CustLedgEntry."Applies-to ID" <> '' THEN
          CustEntryApplID := ''
        ELSE BEGIN
          CustEntryApplID := AppliesToID;
          IF CustEntryApplID = '' THEN BEGIN
            CustEntryApplID := USERID;
            IF CustEntryApplID = '' THEN
              CustEntryApplID := '***';
          END;
        END;

        // Set Applies-to ID
        REPEAT
          CustLedgEntry.TESTFIELD(Open,TRUE);
          //-->20140619-FINA13-CMC02.02.00-2155-BDN01
          {BEFORE
          //-->20140514-FINA13-CMC02.02.00-2044-BDN01
          IF CustLedgEntry."Applies-to ID" <> '' THEN BEGIN
            IF NOT CONFIRM(Text8053000,FALSE,CustLedgEntry."Document No.") THEN
              ERROR('');
          END;
          //<--20140514-FINA13-CMC02.02.00-2044-BDN01

          //-->20140619-FINA13-CMC02.02.00-2155-BDN01
          IF CustLedgEntry."Applies-to ID" = '' THEN BEGIN
            CompensProp_G.RESET;
            CompensProp_G.SETRANGE("SettOff Type",CompensProp_G."SettOff Type"::Proposed);
            CompensProp_G.SETRANGE("Entry Type",CompensProp_G."Entry Type"::Customer);
            CompensProp_G.SETRANGE("Entry No.",CustLedgEntry."Entry No.");
            IF CompensProp_G.FINDFIRST THEN
              IF NOT CONFIRM(Text8053001,FALSE,CompensProp_G."SettOff Applies-to ID") THEN
                ERROR('');
          END;
          //<--20140619-FINA13-CMC02.02.00-2155-BDN01
          }
          //<--20140619-FINA13-CMC02.02.00-2155-BDN01
          CustLedgEntry."Applies-to ID" := CustEntryApplID;
          IF CustLedgEntry."Applies-to ID" = '' THEN BEGIN
            CustLedgEntry."Accepted Pmt. Disc. Tolerance" := FALSE;
            CustLedgEntry."Accepted Payment Tolerance" := 0;
          END;
          // Set Amount to Apply
          IF ((CustLedgEntry."Amount to Apply" <> 0) AND (CustEntryApplID = '')) OR
            (CustEntryApplID = '')
          THEN
            CustLedgEntry."Amount to Apply" := 0
          ELSE
            IF CustLedgEntry."Amount to Apply" = 0 THEN BEGIN
              CustLedgEntry.CALCFIELDS("Remaining Amount");
              CustLedgEntry."Amount to Apply" := CustLedgEntry."Remaining Amount"
            END;

          IF CustLedgEntry."Entry No." = ApplyingCustLedgEntry."Entry No." THEN
            CustLedgEntry."Applying Entry" := ApplyingCustLedgEntry."Applying Entry";
          CustLedgEntry.MODIFY;
        UNTIL CustLedgEntry.NEXT = 0;
      END;
    END;
