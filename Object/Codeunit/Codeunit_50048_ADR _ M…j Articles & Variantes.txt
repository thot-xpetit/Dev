  PROPERTIES
  {
    OnRun=BEGIN

            // M.j "Expl Agric" Tiers MS si non FARM et FARM dans Contact UNEAL - pass' en PROD le 19/01/2015 18H15
            {
            Cpt  := 0;
            Cpt2 := 0;
            RelationMS.RESET;
            Total := RelationMS.COUNT;
            Fenetre.OPEN('Maj information TiersMS ...           \' +
                         'Nb lecture                    #1######\' +
                         'Nb mise . jour Expl.Agric.    #2######\' +
                         '@4@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@',Cpt,Cpt2);

            IF RelationMS.FIND('-') THEN
            REPEAT

              IF (RelationMS."MC Company Code"='UNEAL') AND (RelationMS.Type= 5 ) AND (RelationMS."Contact type"= 0 ) THEN BEGIN
                IF TiersMS.GET(RelationMS."No.") THEN BEGIN
                  IF Contact.GET(RelationMS."Contact code") THEN BEGIN
                    IF (TiersMS."Contact Type" = 0) AND (Contact."Contact Type" = 1) THEN BEGIN
                      TiersMS."Contact Type":=TiersMS."Contact Type"::Farm; //exploitant agricole
                      TiersMS.MODIFY;
                      Cpt2 := Cpt2 +1;
                    END;
                  END;
                END;
              END;

              Cpt := Cpt + 1;

              Fenetre.UPDATE;
              Fenetre.UPDATE(1,Cpt);
              Fenetre.UPDATE(2,Cpt2);

              Fenetre.UPDATE(4,ROUND(Cpt / Total * 10000, 1));

            UNTIL RelationMS.NEXT = 0;
            Fenetre.CLOSE;
            }

            //  M.j "Expl Agric" Contact St' si non FARM et FARM dans Tiers MS - pass' en PROD le 20/01/2015 8H35 (47 cas).
            {
            Cpt  := 0;
            Cpt2 := 0;
            RelationMS.RESET;
            Total := RelationMS.COUNT;
            Fenetre.OPEN('Maj information Contact ...           \' +
                         'Nb lecture                    #1######\' +
                         'Nb mise . jour Expl.AGric.    #2######\' +
                         '@4@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@',Cpt,Cpt2);

            IF RelationMS.FIND('-') THEN
            REPEAT

              IF (RelationMS."MC Company Code"='TERNOVEO') AND (RelationMS.Type= 5 ) AND (RelationMS."Contact type"= 0 ) THEN BEGIN
                IF TiersMS.GET(RelationMS."No.") THEN BEGIN
                  IF Contact.GET(RelationMS."Contact code") THEN BEGIN
                    IF (TiersMS."Contact Type" = 1) AND (Contact."Contact Type" <> 1) THEN BEGIN
                      Contact."Contact Type":=Contact."Contact Type"::Farm; //exploitant agricole
                      Contact.MODIFY;
                      Cpt2 := Cpt2 +1;
                    END;
                  END;
                END;
              END;
              Cpt := Cpt + 1;

              Fenetre.UPDATE;
              Fenetre.UPDATE(1,Cpt);
              Fenetre.UPDATE(2,Cpt2);

              Fenetre.UPDATE(4,ROUND(Cpt / Total * 10000, 1));

            UNTIL RelationMS.NEXT = 0;
            Fenetre.CLOSE;
            }
          END;

  }
