OBJECT Codeunit 8052999 CMC Interv. Jnl.-Post Batch
{
  OBJECT-PROPERTIES
  {
    Date=01/08/11;
    Heure=00:00:00;
    Liste versions=CMC01.04;
  }
  PROPERTIES
  {
    TableNo=8053034;
    Permissions=TableData 5634=imd;
    OnRun=BEGIN
            G_recInterventionJnlLine.COPY(Rec);
            Code;
            Rec := G_recInterventionJnlLine;
          END;

  }
  CODE
  {
    VAR
      Text000@1000 : TextConst 'ENU=cannot exceed %1 characters;FRA=ne peut pas d'passer %1 caractSres';
      Text001@1001 : TextConst 'ENU=Journal Batch Name    #1##########\\;FRA=Nom feuille             #1##########\\';
      Text002@1002 : TextConst 'ENU=Checking lines        #2######\;FRA=V'rification des lignes #2######\';
      Text003@1003 : TextConst 'ENU=Posting lines         #3###### @4@@@@@@@@@@@@@;FRA=Validation des lignes   #3###### @4@@@@@@@@@@@@@';
      Text004@1004 : TextConst 'ENU=A maximum of %1 posting number series can be used in each journal.;FRA=Chaque feuille peut utiliser au maximum %1 souche(s) de num'ros.';
      G_recInterventionJnlLine@1005 : Record 8053034;
      G_recInterventionJnlTempl@1006 : Record 8053039;
      G_recInterventionJnlBatch@1007 : Record 8053038;
      G_LedgEntryDim@1009 : Record 355;
      G_recInterventionJnlLine2@1011 : Record 8053034;
      G_recInterventionJnlLine3@1012 : Record 8053034;
      G_NoSeries@1013 : TEMPORARY Record 308;
      G_InterventionJnlPostLine@1015 : Codeunit 8052998;
      G_InterventionJnlCheckLine@1016 : Codeunit 8052997;
      G_NoSeriesMgt@1017 : Codeunit 396;
      G_NoSeriesMgt2@1018 : ARRAY [10] OF Codeunit 396;
      G_DimMgt@1019 : Codeunit 408;
      G_Window@1020 : Dialog;
      G_LineCount@1021 : Integer;
      G_StartLineNo@1022 : Integer;
      G_NoOfRecords@1023 : Integer;
      G_NoOfPostingNoSeries@1028 : Integer;
      G_PostingNoSeriesNo@1029 : Integer;

    PROCEDURE Code@1();
    VAR
      L_recJnlLineDim@1000 : Record 356;
      L_recTempJnlLineDim@1001 : TEMPORARY Record 356;
      L_UpdateAnalysisView@1002 : Codeunit 410;
    BEGIN
      WITH G_recInterventionJnlLine DO BEGIN
        SETRANGE("Journal Template Name","Journal Template Name");
        SETRANGE("Journal Batch Name","Journal Batch Name");
        IF RECORDLEVELLOCKING THEN
          LOCKTABLE;

        G_recInterventionJnlTempl.GET("Journal Template Name");
        G_recInterventionJnlBatch.GET("Journal Template Name","Journal Batch Name");
        IF STRLEN(INCSTR(G_recInterventionJnlBatch.Name)) > MAXSTRLEN(G_recInterventionJnlBatch.Name) THEN
          G_recInterventionJnlBatch.FIELDERROR(Name,STRSUBSTNO(Text000,MAXSTRLEN(G_recInterventionJnlBatch.Name)));

        IF NOT FIND('=><') THEN BEGIN
          COMMIT;
          "Line No." := 0;
          EXIT;
        END;

        G_Window.OPEN(Text001 + Text002 + Text003);
        G_Window.UPDATE(1,"Journal Batch Name");

        // Check lines
        G_LineCount := 0;
        G_StartLineNo := "Line No.";
        REPEAT
          G_LineCount := G_LineCount + 1;
          G_Window.UPDATE(2,G_LineCount);
          L_recJnlLineDim.SETRANGE("Table ID",DATABASE::"CMC Intervention Journal");
          L_recJnlLineDim.SETRANGE("Journal Template Name","Journal Template Name");
          L_recJnlLineDim.SETRANGE("Journal Batch Name","Journal Batch Name");
          L_recJnlLineDim.SETRANGE("Journal Line No.","Line No.");
          L_recJnlLineDim.SETRANGE("Allocation Line No.",0);
          L_recTempJnlLineDim.DELETEALL;
          G_DimMgt.CopyJnlLineDimToJnlLineDim(L_recJnlLineDim,L_recTempJnlLineDim);
          G_InterventionJnlCheckLine.RunCheck(G_recInterventionJnlLine,L_recTempJnlLineDim);
          IF NEXT = 0 THEN
            FIND('-');
        UNTIL "Line No." = G_StartLineNo;
        G_NoOfRecords := G_LineCount;

        G_LedgEntryDim.LOCKTABLE;

        // Post lines
        G_LineCount := 0;
        FIND('-');
        REPEAT
          G_LineCount := G_LineCount + 1;
          G_Window.UPDATE(3,G_LineCount);
          G_Window.UPDATE(4,ROUND(G_LineCount / G_NoOfRecords * 10000,1));
          L_recJnlLineDim.SETRANGE("Table ID",DATABASE::"CMC Intervention Journal");
          L_recJnlLineDim.SETRANGE("Journal Template Name","Journal Template Name");
          L_recJnlLineDim.SETRANGE("Journal Batch Name","Journal Batch Name");
          L_recJnlLineDim.SETRANGE("Journal Line No.","Line No.");
          L_recJnlLineDim.SETRANGE("Allocation Line No.",0);
          L_recTempJnlLineDim.DELETEALL;
          G_DimMgt.CopyJnlLineDimToJnlLineDim(L_recJnlLineDim,L_recTempJnlLineDim);
          G_InterventionJnlPostLine.RunWithOutCheck(G_recInterventionJnlLine,L_recTempJnlLineDim);
        UNTIL NEXT = 0;

        INIT;

        // Update/delete lines
        IF NOT RECORDLEVELLOCKING THEN BEGIN
          L_recJnlLineDim.LOCKTABLE(TRUE,TRUE);
          LOCKTABLE(TRUE,TRUE);
        END;
        G_recInterventionJnlLine2.COPYFILTERS(G_recInterventionJnlLine);
        G_recInterventionJnlLine2.SETFILTER("Intervention Code",'<>%1','');
        IF G_recInterventionJnlLine2.FINDLAST THEN; // Remember the last line
          L_recJnlLineDim.SETRANGE("Table ID",DATABASE::"CMC Intervention Journal");
        L_recJnlLineDim.COPYFILTER("Journal Template Name","Journal Template Name");
        L_recJnlLineDim.COPYFILTER("Journal Batch Name","Journal Batch Name");
        L_recJnlLineDim.SETRANGE("Allocation Line No.",0);
        G_recInterventionJnlLine3.COPY(G_recInterventionJnlLine);
        IF G_recInterventionJnlLine3.FIND('-') THEN
          REPEAT
            L_recJnlLineDim.SETRANGE("Journal Line No.",G_recInterventionJnlLine3."Line No.");
            L_recJnlLineDim.DELETEALL;
            G_recInterventionJnlLine3.DELETE;
          UNTIL G_recInterventionJnlLine3.NEXT = 0;
        G_recInterventionJnlLine3.RESET;
        G_recInterventionJnlLine3.SETRANGE("Journal Template Name","Journal Template Name");
        G_recInterventionJnlLine3.SETRANGE("Journal Batch Name","Journal Batch Name");
        IF NOT G_recInterventionJnlLine3.FINDLAST THEN
          IF INCSTR("Journal Batch Name") <> '' THEN BEGIN
            G_recInterventionJnlBatch.GET("Journal Template Name","Journal Batch Name");
            G_recInterventionJnlBatch.DELETE;
            G_recInterventionJnlBatch.Name := INCSTR("Journal Batch Name");
            IF G_recInterventionJnlBatch.INSERT THEN;
            "Journal Batch Name" := G_recInterventionJnlBatch.Name;
          END;

        G_recInterventionJnlLine3.SETRANGE("Journal Batch Name","Journal Batch Name");
        IF (G_recInterventionJnlBatch."No. Series" = '') AND NOT G_recInterventionJnlLine3.FINDLAST THEN BEGIN
          G_recInterventionJnlLine3.INIT;
          G_recInterventionJnlLine3."Journal Template Name" := "Journal Template Name";
          G_recInterventionJnlLine3."Journal Batch Name" := "Journal Batch Name";
          G_recInterventionJnlLine3."Line No." := 10000;
          G_recInterventionJnlLine3.INSERT;
        END;
        IF G_recInterventionJnlBatch."No. Series" <> '' THEN
          G_NoSeriesMgt.SaveNoSeries;
        IF G_NoSeries.FIND('-') THEN
          REPEAT
            EVALUATE(G_PostingNoSeriesNo,G_NoSeries.Description);
            G_NoSeriesMgt2[G_PostingNoSeriesNo].SaveNoSeries;
          UNTIL G_NoSeries.NEXT = 0;

        COMMIT;
        CLEAR(G_InterventionJnlCheckLine);
        CLEAR(G_InterventionJnlPostLine);
      END;
      L_UpdateAnalysisView.UpdateAll(0,TRUE);
      COMMIT;
    END;

    BEGIN
    {
      // <-- 3LI BEGIN

      20110601-Wi00-CMC01.04 : Version RTM 01.04 - NavAgri

      // --> 3LI END
    }
    END.
  }
