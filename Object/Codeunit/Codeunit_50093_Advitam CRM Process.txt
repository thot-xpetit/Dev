      // Get CCC contract linked to Crop Rotation
      ContractType.RESET();
      ContractType.SETRANGE(Type, ContractType.Type::CCC);
      IF (ContractType.FINDFIRST()) THEN BEGIN
        REPEAT
          CCCContract.SETRANGE("Contract Type", ContractType.Code);
          CCCContract.SETRANGE("Crop Rotation Campaign No.", CropRotation."Campaign No.");

            REPEAT
              IF (CCCContract."Synchro Status" <> CCCContract."Synchro Status"::"To validate") THEN BEGIN
                IF (CCCContractModified.GET(CCCContract."Contract No.")) THEN BEGIN
                  CCCContractModified.SystemCreated(TRUE);
                  CCCContractModified.VALIDATE("Synchro Status", CCCContractModified."Synchro Status"::"To validate");

                END;
              END;
            UNTIL (CCCContract.NEXT() = 0);
          END;

      END;
    END;
