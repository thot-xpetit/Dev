    PROCEDURE CalcTotalDiscount@1180250002(VAR SalesLine_P@1180250002 : Record 37;VAR Currency_P@1180250003 : Record 4) : Decimal;
    VAR
      GrossPrice_L@1000000005 : Decimal;
      NetPrice_L@1180250004 : ARRAY [5] OF Decimal;
    BEGIN
      WITH SalesLine_P DO BEGIN
        GrossPrice_L := ROUND(Quantity * "Unit Price",Currency_P."Amount Rounding Precision");
        NetPrice_L[1] := GrossPrice_L - ((GrossPrice_L * "Level 1 Discount %" ) / 100);
        NetPrice_L[2] := NetPrice_L[1] - ((NetPrice_L[1] * "Level 2 Discount %" ) / 100);
        NetPrice_L[3] := NetPrice_L[2] - ((NetPrice_L[2] * "Level 3 Discount %" ) / 100);
        NetPrice_L[4] := NetPrice_L[3] - ((NetPrice_L[3] * "Level 4 Discount %" ) / 100);
        NetPrice_L[5] := NetPrice_L[4] - ((NetPrice_L[4] * "Level 5 Discount %" ) / 100);
        IF GrossPrice_L <> 0 THEN
          EXIT(ABS(100 * (1 - (NetPrice_L[5] / GrossPrice_L))))
        ELSE
          EXIT(0);
      END;
    END;

    PROCEDURE SalesLineLineDiscExists@1000000003(VAR SalesLineMultiDisc_P@1180250002 : Record 8060984;SalesHeader_P@1001 : Record 36;VAR SalesLine_P@1000 : Record 37;ShowAll_P@1002 : Boolean) : Boolean;
    BEGIN
      WITH SalesLine_P DO BEGIN
        SetUoM(ABS(Quantity),"Qty. per Unit of Measure");
        IF (Type = Type::Item) AND Item_G.GET("No.") THEN BEGIN
          FindSalesLineDisc(
            SalesLineMultiDisc_P,"Bill-to Customer No.",SalesHeader_P."Bill-to Contact No.",
            "Customer Disc. Group",'',"No.",Item_G."Item Disc. Group","Variant Code","Unit of Measure Code",
            SalesHeader_P."Currency Code",SalesHeaderStartDate(SalesHeader_P,DateCaption_G),ShowAll_P);
          EXIT(TempSalesLineMultiDisc_G.FIND('-'));
        END;
      END;
      EXIT(FALSE);
    END;

    PROCEDURE FindSalesLineDisc@1000000002(VAR ToSalesLineDisc_P@1006 : Record 8060984;CustNo_P@1003 : Code[20];ContNo_P@1001 : Code[20];CustDiscGrCode_P@1002 : Code[10];CampaignNo_P@1010 : Code[20];ItemNo_P@1004 : Code[20];ItemDiscGrCode_P@1005 : Code[10];VariantCode_P@1014 : Code[10];UOM_P@1013 : Code[10];CurrencyCode_P@1012 : Code[10];StartingDate_P@1011 : Date;ShowAll_P@1000 : Boolean);
    VAR
      FromSalesLineDisc_L@1007 : Record 8060984;
      TempCampaignTargetGr_L@1009 : TEMPORARY Record 7030;
      InclCampaigns_L@1008 : Boolean;
    BEGIN
      WITH FromSalesLineDisc_L DO BEGIN
        SETFILTER("Ending Date",'%1|>=%2',0D,StartingDate_P);
        SETFILTER("Variant Code",'%1|%2',VariantCode_P,'');
        IF NOT ShowAll_P THEN BEGIN
          SETRANGE("Starting Date",0D,StartingDate_P);
          SETFILTER("Currency Code",'%1|%2',CurrencyCode_P,'');
          IF UOM_P <> '' THEN
            SETFILTER("Unit of Measure Code",'%1|%2',UOM_P,'');
        END;

        ToSalesLineDisc_P.RESET;
        ToSalesLineDisc_P.DELETEALL;
        FOR "Sales Type" := "Sales Type"::Customer TO "Sales Type"::Campaign DO
          IF ("Sales Type" = "Sales Type"::"All Customers") OR
             (("Sales Type" = "Sales Type"::Customer) AND (CustNo_P <> '')) OR
             (("Sales Type" = "Sales Type"::"Customer Disc. Group") AND (CustDiscGrCode_P <> '')) OR
             (("Sales Type" = "Sales Type"::Campaign) AND
                 NOT ((CustNo_P = '') AND (ContNo_P = '') AND (CampaignNo_P = '')))
          THEN BEGIN
            InclCampaigns_L := FALSE;

            SETRANGE("Sales Type","Sales Type");
            CASE "Sales Type" OF
              "Sales Type"::"All Customers":
                SETRANGE("Sales Code");
              "Sales Type"::Customer:
                SETRANGE("Sales Code",
                CustNo_P);
              "Sales Type"::"Customer Disc. Group":
                SETRANGE("Sales Code",CustDiscGrCode_P);
              "Sales Type"::Campaign:
                BEGIN
                  InclCampaigns_L := ActivatedCampaignExists(TempCampaignTargetGr_L,CustNo_P,ContNo_P,CampaignNo_P);
                  SETRANGE("Sales Code",TempCampaignTargetGr_L."Campaign No.");
                END;
            END;

            REPEAT
              SETRANGE(Type,Type::Item);
              SETRANGE(Code,ItemNo_P);
              CopySalesMulDiscToSalesMulDisc(FromSalesLineDisc_L,ToSalesLineDisc_P);

              IF ItemDiscGrCode_P <> '' THEN BEGIN
                SETRANGE(Type,Type::"Item Disc. Group");
                SETRANGE(Code,ItemDiscGrCode_P);
                CopySalesMulDiscToSalesMulDisc(FromSalesLineDisc_L,ToSalesLineDisc_P);
              END;

              IF InclCampaigns_L THEN BEGIN
                InclCampaigns_L := TempCampaignTargetGr_L.NEXT <> 0;
                SETRANGE("Sales Code",TempCampaignTargetGr_L."Campaign No.");
              END;
            UNTIL NOT InclCampaigns_L;
        END;
      END;
    END;

    LOCAL PROCEDURE SalesHeaderStartDate@31(SalesHeader_P@1000 : Record 36;VAR DateCaption_P@1001 : Text[30]) : Date;
    BEGIN
      WITH SalesHeader_P DO
        IF "Document Type" IN ["Document Type"::Invoice,"Document Type"::"Credit Memo"] THEN BEGIN
          DateCaption_P := FIELDCAPTION("Posting Date");
          EXIT("Posting Date")
        END ELSE BEGIN
          DateCaption_P := FIELDCAPTION("Order Date");
          EXIT("Order Date");
        END;
    END;

    LOCAL PROCEDURE ActivatedCampaignExists@21(VAR ToCampaignTargetGr_P@1001 : Record 7030;CustNo_P@1000 : Code[20];ContNo_P@1002 : Code[20];CampaignNo_P@1004 : Code[20]) : Boolean;
    VAR
      FromCampaignTargetGr@1003 : Record 7030;
      Cont@1005 : Record 5050;
    BEGIN
      WITH FromCampaignTargetGr DO BEGIN
        ToCampaignTargetGr_P.RESET;
        ToCampaignTargetGr_P.DELETEALL;

        IF CampaignNo_P <> '' THEN BEGIN
          ToCampaignTargetGr_P."Campaign No." := CampaignNo_P;
          ToCampaignTargetGr_P.INSERT;
        END ELSE BEGIN
          SETRANGE(Type,Type::Customer);
          SETRANGE("No.",CustNo_P);
          IF FINDSET THEN
            REPEAT
              ToCampaignTargetGr_P := FromCampaignTargetGr;
              ToCampaignTargetGr_P.INSERT;
            UNTIL NEXT = 0
          ELSE BEGIN
            IF Cont.GET(ContNo_P) THEN BEGIN
              SETRANGE(Type,Type::Contact);
              SETRANGE("No.",Cont."Company No.");
              IF FINDSET THEN
                REPEAT
                  ToCampaignTargetGr_P := FromCampaignTargetGr;
                  ToCampaignTargetGr_P.INSERT;
                UNTIL NEXT = 0;
            END;
          END;
        END;
        EXIT(ToCampaignTargetGr_P.FINDFIRST);
      END;
    END;

    LOCAL PROCEDURE CopySalesMulDiscToSalesMulDisc@1000000004(VAR FromSalesLineDisc_P@1000 : Record 8060984;VAR ToSalesLineDisc_P@1001 : Record 8060984);
    BEGIN
      WITH ToSalesLineDisc_P DO BEGIN
        IF FromSalesLineDisc_P.FIND('-') THEN
          REPEAT
            IF FromSalesLineDisc_P."Line Discount %" <> 0 THEN BEGIN
              ToSalesLineDisc_P := FromSalesLineDisc_P;
              INSERT;
            END;
          UNTIL FromSalesLineDisc_P.NEXT = 0;
      END;
    END;

    PROCEDURE CalcBestLineDisc@1180250003(VAR SalesLineDisc_P@1000 : Record 8060984;DiscountLevel_P@1000000000 : 'R1,R2,R3,R4');
    VAR
      BestSalesLineDisc_L@1002 : Record 8060984;
    BEGIN
      WITH SalesLineDisc_P DO BEGIN
        SETRANGE("Discount Level",DiscountLevel_P);
        IF FIND('-') THEN
          REPEAT
            IF IsInMinQty("Unit of Measure Code","Minimum Quantity") THEN
              CASE TRUE OF
                ((BestSalesLineDisc_L."Currency Code" = '') AND ("Currency Code" <> '')) OR
                ((BestSalesLineDisc_L."Variant Code" = '') AND ("Variant Code" <> '')):
                  BestSalesLineDisc_L := SalesLineDisc_P;
                ((BestSalesLineDisc_L."Currency Code" = '') OR ("Currency Code" <> '')) AND
                ((BestSalesLineDisc_L."Variant Code" = '') OR ("Variant Code" <> '')):
                  IF BestSalesLineDisc_L."Line Discount %" < "Line Discount %" THEN
                    BestSalesLineDisc_L := SalesLineDisc_P;
              END;
          UNTIL NEXT = 0;
      END;

      SalesLineDisc_P := BestSalesLineDisc_L;
    END;

    LOCAL PROCEDURE SetUoM@23(Qty_P@1000 : Decimal;QtyPerUoM_P@1001 : Decimal);

    LOCAL PROCEDURE IsInMinQty@7(UnitofMeasureCode@1003 : Code[10];MinQty@1000 : Decimal) : Boolean;
    BEGIN
      IF UnitofMeasureCode = '' THEN
        EXIT(MinQty <= QtyPerUOM_G * Qty_G);
      EXIT(MinQty <= Qty_G);
    END;
