    PROCEDURE LaunchCheck@1000000002();
    VAR
      CRMSetup@1000000000 : Record 60156;
      SASCrmError@1000000001 : Report 50104;
      Filename@1000000002 : Text[1024];
      Result@1000000003 : Boolean;
    BEGIN
      Result := FALSE;
      IF (CRMSetup.GET()) THEN BEGIN
        IF ((CRMSetup."Error Integration File Path" <> '') AND
           (CRMSetup."Error Integration Filename" <> '')) THEN BEGIN
          Filename := CRMSetup."Error Integration File Path";
          IF (Filename[STRLEN(Filename)] <> '\') THEN BEGIN
            Filename := Filename + '\'
          END;
          Filename := Filename + CRMSetup."Error Integration Filename";
          CASE (CRMSetup."Type File Error") OF
            CRMSetup."Type File Error"::Csv : BEGIN
              Result := ExportData(Filename);
            END;
            CRMSetup."Type File Error"::Excel : BEGIN
              Result := SASCrmError.SAVEASEXCEL(Filename);
            END;
            CRMSetup."Type File Error"::Text : BEGIN
              Result := ExportData(Filename);
            END;
          END;
          IF (Result) THEN BEGIN
            IF (CRMSetup."Email Notification" <> '') THEN BEGIN
              IF (SendEmail(CRMSetup."Email Notification",
                            Filename,
                            Text001,
                            Text002)) THEN BEGIN
                ERASE(Filename);
              END ELSE BEGIN
              END;
            END;
          END ELSE BEGIN
            ToolsFile_G.LogWriteErreur(ErrExportData);
          END;
        END ELSE BEGIN
          ToolsFile_G.LogWriteErreur(ErrNoFileSetup);
        END;
      END ELSE BEGIN
        ToolsFile_G.LogWriteErreur(ErrNoSetup);
      END;
    END;
