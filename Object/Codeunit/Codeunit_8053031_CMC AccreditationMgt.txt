    PROCEDURE VerifyContactAccr@8053001(ContactNo_P@8053000 : Code[20];ReferenceDate_P@8053001 : Date;CMCAccreditationCheck_P@8053002 : Record 8053313) : Boolean;
    VAR
      CMCContactAccreditation_L@8053003 : Record 8053314;
      CMCAccreditation_L@1000000001 : Record 8053310;
    BEGIN
      CMCContactAccreditation_L.SETRANGE("Company Contact No.", ContactNo_P);
      CMCContactAccreditation_L.SETFILTER("Contact No.", '<>%1','');
      CMCContactAccreditation_L.SETRANGE("Accreditation Code", CMCAccreditationCheck_P."Accreditation Code");
      // --> 20220426-WI00-ADV01.01.00-SP3583-CJY01
      //dans le cas des pr'curseurs, on ne doit consid'rer comme valide que les lignes dont le N› agr'ment est non vide
      CMCAccreditation_L.GET(CMCAccreditationCheck_P."Accreditation Code");
      IF CMCAccreditation_L."Explosive material" THEN
        CMCContactAccreditation_L.SETFILTER("Accreditation No.",'<>%1','');
      // <-- 20220426-WI00-ADV01.01.00-SP3583-CJY01

      // --> 20220406-WI00-ADV01.01.00-SP3583-CJY01
      //le contr"le sur date d'but et date de fin devienne param'trable
      IF CMCAccreditation_L."Control Date" = CMCAccreditation_L."Control Date"::yes THEN BEGIN
      // <-- 20220406-WI00-ADV01.01.00-SP3583-CJY01
      CMCContactAccreditation_L.SETFILTER("Starting Date", '<=%1|%2', ReferenceDate_P,0D);
      CMCContactAccreditation_L.SETFILTER("Ending Date", '>=%1|%2', ReferenceDate_P,0D);
      // --> 20220406-WI00-ADV01.01.00-SP3583-CJY01
      END;
      // <-- 20220406-WI00-ADV01.01.00-SP3583-CJY01
      CMCContactAccreditation_L.SETFILTER("Accreditation Auth. Code", '%1', CMCAccreditationCheck_P."Accred. Authorization Filter");
      // --> 20220406-WI00-ADV01.01.00-SP3583-CJY01
      {BEFORE
      CMCContactAccreditation_L.SETFILTER("Accreditation Type Code", '%1', CMCAccreditationCheck_P."Accred. Type Filter");
      AFTER}
      IF CMCAccreditationCheck_P."Accred. Type Filter" <> '' THEN
        CMCContactAccreditation_L.SETFILTER("Accreditation Type Code",CMCAccreditationCheck_P."Accred. Type Filter")
      ELSE
        CMCContactAccreditation_L.SETRANGE("Accreditation Type Code",'');

      // <-- 20220406-WI00-ADV01.01.00-SP3583-CJY01
      IF NOT CMCContactAccreditation_L.FINDLAST THEN
        IF CMCAccreditationCheck_P.Blocking THEN
          ERROR(Text8053001, ContactNo_P,  CMCAccreditationCheck_P."Accreditation Code")
        ELSE
          MESSAGE(Text8053001, ContactNo_P,  CMCAccreditationCheck_P."Accreditation Code");

      EXIT(TRUE);
    END;
