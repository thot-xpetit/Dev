    PROCEDURE InOutCall@1000000001(_UserID@1000000004 : Text[30];_CallID@1000000003 : Integer;_CallingPhoneNo@1000000002 : Text[30];_CalledPhoneNo@1000000001 : Text[30];VAR _TextInfo@1000000000 : Text[1024];_Direction@1000000005 : Integer) Return : Text[1024];
    BEGIN
      _CallingPhoneNo := TelephonyUtils.CleanPhoneNo(_CallingPhoneNo, '');
      _CalledPhoneNo  := TelephonyUtils.CleanPhoneNo(_CalledPhoneNo, '');

      CalledCompany := GetCalledCompany(_CalledPhoneNo);

      WITH TelephonyPhonebook DO BEGIN
        RESET();
        FilterSourceCompanies();

        SETRANGE("Phone No.", _CallingPhoneNo);

        IF '' <> CalledCompany THEN SETFILTER("Source Company", '%1', CalledCompany);

        IF      ISEMPTY     THEN Return := _NoEntryFound(_CallID, _CallingPhoneNo, _CalledPhoneNo, _TextInfo, _Direction)
        ELSE IF 1 = COUNT() THEN Return := _OneEntryFound(_CallID, _CallingPhoneNo, _CalledPhoneNo, _TextInfo, _Direction)
                            ELSE Return := _ManyEntriesFound(_CallID, _CallingPhoneNo, _CalledPhoneNo, _TextInfo, _Direction);
      END;
    END;

    LOCAL PROCEDURE _NoEntryFound@8062007(_CallID@1000000000 : Integer;_CallingPhoneNo@8062002 : Text[30];_CalledPhoneNo@8053600 : Text[30];VAR _TextInfo@8062000 : Text[1024];_Direction@1000000001 : Integer) : Text[1024];

    LOCAL PROCEDURE _OneEntryFound@8062006(_CallID@1000000000 : Integer;_CallingPhoneNo@8062002 : Text[30];_CalledPhoneNo@8053600 : Text[30];VAR _TextInfo@8062000 : Text[1024];_Direction@1000000001 : Integer) : Text[1024];
    BEGIN
      GetSetup();

      WITH TelephonyPhonebook DO BEGIN
        FINDFIRST();

        CreateLogEntry(_CallID, "Source Company", "Source Type", "Source No.", "Phone No.", _Direction, CalledCompany);
        CompleteLogEntry(TODAY, _CallID, TelephonyPhonebook);

        _TextInfo := CreateTextInfo(_CalledPhoneNo, _CallingPhoneNo);
        IF (TelephonySetup."Use Only Phonebook Card") OR ("Source Type"::External = "Source Type") THEN
          EXIT(CreatePhonebookCardURL("Source Company", "Source Company", "Source Type", "Source No.", "Phone No."))

        ELSE IF (TelephonySetup."Prevent Change Company") AND (COMPANYNAME <> "Source Company") THEN
          EXIT(CreatePhonebookCardURL(COMPANYNAME, "Source Company", "Source Type", "Source No.", "Phone No."))

        ELSE
          EXIT(CreateURL("Source Company", 'ONE', GetSetupPageNo("Source Type", CalledCompany), "Source Type", "Source No.", '', 0));
      END;
    END;

    LOCAL PROCEDURE _ManyEntriesFound@8062008(_CallID@1000000000 : Integer;_CallingPhoneNo@8062003 : Text[30];_CalledPhoneNo@8053600 : Text[30];VAR _TextInfo@8062000 : Text[1024];_Direction@1000000001 : Integer) : Text[1024];
    VAR

    END;

    PROCEDURE GetSetupPageNo@8062010(_SourceType@8062001 : Integer;_CompanyCalled@8062000 : Text[30]) : Integer;
    BEGIN
      GetSetup();

      WITH TelephonySetup DO BEGIN
        IF TelephonySetupPages.GET(Type, Code, _SourceType, _CompanyCalled) THEN EXIT(TelephonySetupPages."Page No.");
        IF TelephonySetupPages.GET(Type, Code, _SourceType, '')             THEN EXIT(TelephonySetupPages."Page No.");

        IF Type::User = Type THEN IF TelephonySetupPages.GET(Type::Profile, GetUserProfile(), _SourceType, _CompanyCalled) THEN EXIT(
      TelephonySetupPages."Page No.");
        IF Type::User = Type THEN IF TelephonySetupPages.GET(Type::Profile, GetUserProfile(), _SourceType, '')             THEN EXIT(
      TelephonySetupPages."Page No.");

        IF Type::Global = Type THEN EXIT(0);

        IF TelephonySetupPages.GET(Type::Global, '', _SourceType, _CompanyCalled) THEN EXIT(TelephonySetupPages."Page No.");
        IF TelephonySetupPages.GET(Type::Global, '', _SourceType, '')             THEN EXIT(TelephonySetupPages."Page No.");

        EXIT(0);
      END;
    END;

    PROCEDURE GetCalledCompany@8062020(_PhoneNo@8062000 : Text[30]) : Text[1024];
    VAR
      _CompaniesFilter@8062001 : Text[1024];
    BEGIN
      WITH TelephonySetupCompanies DO BEGIN
        SETRANGE("Phone No.", _PhoneNo);
        IF ISEMPTY THEN EXIT('');
        IF 1 = COUNT THEN BEGIN
          FINDFIRST();
          EXIT("Company Name");
        END ELSE BEGIN
          FINDSET();
          REPEAT
            IF '' <> _CompaniesFilter THEN _CompaniesFilter += '|';
            _CompaniesFilter += "Company Name";
          UNTIL 0 = NEXT();
          EXIT(_CompaniesFilter);
        END;
      END;
    END;

    PROCEDURE CreateURL@8062012(_NAVCompany@8062007 : Text[50];_Action@8062010 : Code[20];_Page@8062000 : Integer;_Type@8062005 : Integer;_Code@8062001 : Text[250];_PhoneNo@8062006 : Text[30];_CallID@8053602 : Integer) return : Text[1024];
    VAR
      URLBase@8062002 : TextConst 'ENU=DynamicsNAV://%1/%2/%3/';
      URLReport@8062003 : TextConst 'ENU="runreport?report=%1"';
      URLReportFilter@8062004 : TextConst 'ENU="&filter=%22%1%22.%22%2%22:%3"';
      _NAVServer@8053600 : Text[50];
      _NAVInstance@8053601 : Text[50];
    BEGIN
      WITH TelephonySetup DO BEGIN
        GET(Type::Global, '');
        _NAVServer   := "NAV Server";
        _NAVInstance := "NAV Instance";
      END;

      IF ('' <> _NAVServer) AND ('' <> _NAVInstance) AND ('' <> _NAVCompany) THEN
        return := STRSUBSTNO(URLBase, _NAVServer, _NAVInstance, _NAVCompany)
      ELSE return := STRSUBSTNO(URLBase, '', '', '');

      return += STRSUBSTNO(URLReport, 8062000);
      return += STRSUBSTNO(URLReportFilter, 'GET', 'Action', _Action);

      IF 0  <> _Page    THEN return += STRSUBSTNO(URLReportFilter, 'GET', 'Page', _Page);
      IF 0  <> _Type    THEN return += STRSUBSTNO(URLReportFilter, 'GET', 'Type', _Type);
      IF '' <> _Code    THEN return += STRSUBSTNO(URLReportFilter, 'GET', 'Code', _Code);
      IF '' <> _PhoneNo THEN return += STRSUBSTNO(URLReportFilter, 'GET', 'Phone No.', _PhoneNo);
      IF 0  <> _CallID  THEN return += STRSUBSTNO(URLReportFilter, 'GET', 'CallID', _CallID);
    END;

    PROCEDURE CreateTextInfo@8062026(_InternalPhoneNo@8053604 : Text[30];_ExternalPhoneNo@8053605 : Text[30]) : Text[1024];
    VAR
      _Count@8062000 : Integer;
      TextInfoBase@8053600 : TextConst 'FRA=%1@%2@%3';
      TextInfoMany@8053601 : TextConst 'ENU=Many Entries;FRA=Plusieurs entr'es';
      TextInfoNone@8053602 : TextConst 'ENU=Unknown;FRA=Inconnu';
      _CalledCompany@8053603 : Text[30];
    BEGIN
      WITH TelephonyPhonebook DO BEGIN

        IF '' = CalledCompany THEN _CalledCompany := _InternalPhoneNo ELSE _CalledCompany := CalledCompany;
        IF ISEMPTY THEN EXIT("Phone No." + '@' + _CalledCompany);

        CASE COUNT OF
          1: IF Name = Company THEN EXIT (Name + '@' + _CalledCompany)
                               ELSE IF '' = Company THEN EXIT(Name + '@' + _CalledCompany)
                                                    ELSE EXIT('(' + Company + ') ' + Name + '@' + _CalledCompany)
          ELSE BEGIN
            _Count := COUNT;
            FINDFIRST();
            SETRANGE(Company, Company);
            IF _Count = COUNT THEN IF '' = Company THEN EXIT(_ExternalPhoneNo + '@' + _CalledCompany)
                                                   ELSE EXIT('(' + Company + ') ' + _ExternalPhoneNo + '@' + _CalledCompany)
            ELSE BEGIN
              SETRANGE(Company);
              EXIT(_ExternalPhoneNo + '@' + _CalledCompany);
            END;
          END;
        END
      END;
    END;

            3: BEGIN
              _RecSalesperson.SETPOSITION(_RecordRef.GETPOSITION());
              PAGE.RUN(_PageNo, _RecSalesperson);

            5: BEGIN

              PAGE.RUN(_PageNo, _RecLocation);

        'MANY': BEGIN

          _PhonebookList.SetCallID(_CallID);
