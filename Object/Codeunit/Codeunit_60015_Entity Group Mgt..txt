    PROCEDURE FarmerListManagement@60006();
    VAR
      recFarmer_L@60000 : Record 5050;
      recFarmerTmp_L@60001 : TEMPORARY Record 5050;
      recCustomizedLists_L@60005 : Record 60038;
      recCustomizableList_L@60007 : Record 60039;
      recChooseCustomListTmp_L@60008 : TEMPORARY Record 21;
      dlgWindow@60002 : Dialog;
      Text60000_L@60003 : TextConst 'ENU=Farmer #1########;FRA=Agriculteur #1########';
      codeFarmerList_L@60006 : Code[20];
      intCountPosition_L@60004 : Integer;
    BEGIN
      recCustomizedLists_L.GET;

      recFarmerTmp_L.RESET;
      recFarmerTmp_L.DELETEALL;

      recChooseCustomListTmp_L.RESET;
      recChooseCustomListTmp_L.DELETEALL;

      FOR intCountPosition_L := 1 TO 6 DO BEGIN
        CLEAR(codeFarmerList_L);

        CASE intCountPosition_L OF
          1: codeFarmerList_L := recCustomizedLists_L."Farmer List 1";
          2: codeFarmerList_L := recCustomizedLists_L."Farmer List 2";
          3: codeFarmerList_L := recCustomizedLists_L."Farmer List 3";
          4: codeFarmerList_L := recCustomizedLists_L."Farmer List 4";
          5: codeFarmerList_L := recCustomizedLists_L."Farmer List 5";
          6: codeFarmerList_L := recCustomizedLists_L."Farmer List 6";
        END;
        IF codeFarmerList_L <> '' THEN BEGIN
          CLEAR(recChooseCustomListTmp_L);
          recChooseCustomListTmp_L."Entry No." := intCountPosition_L;
          recChooseCustomListTmp_L.Open := TRUE;
          recChooseCustomListTmp_L."Customer No." := codeFarmerList_L;

          recCustomizableList_L.RESET;
          recCustomizableList_L.SETRANGE(Type,recCustomizableList_L.Type::Farmer);
          recCustomizableList_L.SETRANGE(Field,codeFarmerList_L);
          IF recCustomizableList_L.FINDFIRST THEN
            recChooseCustomListTmp_L.Description := recCustomizableList_L."Field Name";
          recChooseCustomListTmp_L.INSERT;
        END;
      END;

      recChooseCustomListTmp_L.RESET;
      IF recChooseCustomListTmp_L.COUNT > 1 THEN BEGIN
        recChooseCustomListTmp_L.FINDFIRST;
        IF NOT (PAGE.RUNMODAL(PAGE::"Define Customizable List",recChooseCustomListTmp_L) = ACTION::LookupOK) THEN
          EXIT;
      END;

      dlgWindow.OPEN(Text60000_L);

      recChooseCustomListTmp_L.RESET;
      recChooseCustomListTmp_L.SETRANGE(Open,TRUE);
      IF recChooseCustomListTmp_L.FINDFIRST THEN
        REPEAT
          dlgWindow.UPDATE(1,'');

          intCountPosition_L := recChooseCustomListTmp_L."Entry No.";
          codeFarmerList_L := recChooseCustomListTmp_L."Customer No.";

          IF codeFarmerList_L <> '' THEN BEGIN
            recCustomizableList_L.RESET;
            recCustomizableList_L.SETRANGE(Type,recCustomizableList_L.Type::Farmer);
            recCustomizableList_L.SETRANGE("Field Aux",intCountPosition_L);
            IF recCustomizableList_L.FINDFIRST THEN
              REPEAT
                recFarmer_L.RESET;
                recFarmer_L.SETRANGE("Contact Type",recFarmer_L."Contact Type"::Farm);
                CASE intCountPosition_L OF
                  1: recFarmer_L.SETRANGE("Farmer List 1",recCustomizableList_L.Code);
                  2: recFarmer_L.SETRANGE("Farmer List 2",recCustomizableList_L.Code);
                  3: recFarmer_L.SETRANGE("Farmer List 3",recCustomizableList_L.Code);
                  4: recFarmer_L.SETRANGE("Farmer List 4",recCustomizableList_L.Code);
                  5: recFarmer_L.SETRANGE("Farmer List 5",recCustomizableList_L.Code);
                  6: recFarmer_L.SETRANGE("Farmer List 6",recCustomizableList_L.Code);
                END;
                IF recFarmer_L.FINDFIRST THEN;
                  REPEAT
                    dlgWindow.UPDATE(1,recFarmer_L."No.");
                    CLEAR(recFarmerTmp_L);
                    IF NOT recFarmerTmp_L.GET(recFarmer_L."No.") THEN BEGIN
                      recFarmerTmp_L.TRANSFERFIELDS(recFarmer_L);
                      recFarmerTmp_L.INSERT;
                    END;
                  UNTIL recFarmer_L.NEXT = 0;
              UNTIL recCustomizableList_L.NEXT = 0;
          END;
        UNTIL recChooseCustomListTmp_L.NEXT = 0;

      dlgWindow.CLOSE;

      IF recFarmerTmp_L.FINDFIRST THEN;

      PAGE.RUNMODAL(PAGE::"CMC Farmer List",recFarmerTmp_L);
    END;

    PROCEDURE CaptionClass@60011(CaptionExpr@1001 : Text[80]) : Text[80];
    VAR
      recCustomizedLists_L@60000 : Record 60038;
      CaptionType@1003 : Text[80];
      CaptionRef@1004 : Text[80];
      OptionalParam1@1005 : Text[80];
      OptionalParam2@1006 : Text[80];
      CommaPosition@1007 : Integer;
    BEGIN
      IF NOT recCustomizedLists_L.GET THEN
        EXIT('');

      CommaPosition := STRPOS(CaptionExpr,',');
      IF (CommaPosition > 0) THEN BEGIN
        CaptionType := COPYSTR(CaptionExpr,1,CommaPosition - 1);
        CaptionRef := COPYSTR(CaptionExpr,CommaPosition + 1);
        CommaPosition := STRPOS(CaptionRef,',');
        IF (CommaPosition > 0) THEN BEGIN
          OptionalParam1 := COPYSTR(CaptionRef,CommaPosition + 1);
          CaptionRef := COPYSTR(CaptionRef,1,CommaPosition - 1);
          CommaPosition := STRPOS(OptionalParam1,',');
          IF (CommaPosition > 0) THEN BEGIN
            OptionalParam2 := COPYSTR(OptionalParam1,CommaPosition + 1);
            OptionalParam1 := COPYSTR(OptionalParam1,1,CommaPosition - 1);
          END ELSE BEGIN
            OptionalParam2 := '';
          END;
        END ELSE BEGIN
          OptionalParam1 := '';
          OptionalParam2 := '';
        END;
        CASE CaptionType OF
          '1':  // FARMER
            BEGIN
              CASE CaptionRef OF
                '1': EXIT(recCustomizedLists_L."Farmer List 1");
                '2': EXIT(recCustomizedLists_L."Farmer List 2");
                '3': EXIT(recCustomizedLists_L."Farmer List 3");
                '4': EXIT(recCustomizedLists_L."Farmer List 4");
                '5': EXIT(recCustomizedLists_L."Farmer List 5");
                '6': EXIT(recCustomizedLists_L."Farmer List 6");
              END;
            END;
          '2':  // CUSTOMER
            BEGIN
              CASE CaptionRef OF
                '1': EXIT(recCustomizedLists_L."Customer List 1");
                '2': EXIT(recCustomizedLists_L."Customer List 2");
                '3': EXIT(recCustomizedLists_L."Customer List 3");
                '4': EXIT(recCustomizedLists_L."Customer List 4");
                '5': EXIT(recCustomizedLists_L."Customer List 5");
                '6': EXIT(recCustomizedLists_L."Customer List 6");
              END;
            END;
          '3':  // VENDOR
            BEGIN
              CASE CaptionRef OF
                '1': EXIT(recCustomizedLists_L."Vendor List 1");
                '2': EXIT(recCustomizedLists_L."Vendor List 2");
                '3': EXIT(recCustomizedLists_L."Vendor List 3");
                '4': EXIT(recCustomizedLists_L."Vendor List 4");
                '5': EXIT(recCustomizedLists_L."Vendor List 5");
                '6': EXIT(recCustomizedLists_L."Vendor List 6");
