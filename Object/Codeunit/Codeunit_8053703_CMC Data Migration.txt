OBJECT Codeunit 8053703 CMC Data Migration
{
  OBJECT-PROPERTIES
  {
    Date=30/10/14;
    Heure=00:00:00;
    Liste versions=CMCTools;
  }
  PROPERTIES
  {
    OnRun=BEGIN
            //Step 1
            //Import tables 8053323, 8053703, 8053704
            //Pour VAESKEN -> Ajouter un enregistrement de migration de donn'es sur la table 8053700 . vider au pr'alalable
            //et importer le fichier XML CMC2009R2-20130725-RIM informations pes'es
            //Ex'cution du codeunit Passe 1

            //Step 2
            //Import du build
            //Ex'cution du codeunit Passe 2 et 3

            // ************************************ Step 1 **********************************************************************************
            {
            //Transfert des param'trages de la table CMC Species vers la table CMC Species Setup
            IF RecSpecies_G.FINDSET AND RecSpeciesSetup_G.ISEMPTY THEN
              REPEAT
                IF RecCompany_G.FINDSET THEN
                  REPEAT
                    RecSpeciesSetup_G.CHANGECOMPANY(RecCompany_G.Name);
                    RecSpeciesSetup_G.TRANSFERFIELDS(RecSpecies_G);
                    RecSpeciesSetup_G.INSERT;
                  UNTIL RecCompany_G.NEXT = 0;
                RecSpecies_G."Standard Item No." := '';
                RecSpecies_G."Due Date" := 0D;
                RecSpecies_G."Biologic Species" := FALSE;
                RecSpecies_G."Product Type" := 0;
                RecSpecies_G."Default Increase Price" := 0;
                RecSpecies_G."Increase Starting Month" := 0;
                RecSpecies_G."Out Of Harvest" := FALSE;
                RecSpecies_G."Rich Product" := '';
                RecSpecies_G."Poor Product" := '';
                RecSpecies_G."Declassed Product" := '';
                RecSpecies_G."Mandatory Seal" := FALSE;
                RecSpecies_G.MODIFY;
              UNTIL RecSpecies_G.NEXT = 0;

            //Modification de la structure de la table CMC Campaign DataPerCompany -> Yes
            //Sauvegarde de la table CMC Campaign -> CMC Campaign Temp
            IF RecCampaign_G.FINDSET AND RecCampaignTemp_G.ISEMPTY THEN BEGIN
              REPEAT
                RecCampaignTemp_G.TRANSFERFIELDS(RecCampaign_G);
                RecCampaignTemp_G."Max Frame Contract Date" := 0D;
                RecCampaignTemp_G.INSERT;
              UNTIL RecCampaign_G.NEXT = 0;
              RecCampaign_G.DELETEALL;
            END;

            //Suppression de donn'es suite modification structure table CMC Setup
            IF RecCompany_G.FINDSET THEN
              REPEAT
                RecCMCSetup_G.CHANGECOMPANY(RecCompany_G.Name);
                IF RecCMCSetup_G.GET THEN BEGIN
                  RecCMCSetup_G."Frame Contract Nos." := '';
                  RecCMCSetup_G.MODIFY;
                END;
              UNTIL RecCompany_G.NEXT = 0;

            //Suppression de donn'es suite modification structure table CMC MC Item
            IF RecMCItem_G.FINDSET THEN
              REPEAT
                RecMCItem_G."Grouping 1" := '';
                RecMCItem_G."Grouping 2" := '';
                RecMCItem_G."Utilisation Code" := '';
                RecMCItem_G."Hectare Dose" := 0;
                RecMCItem_G."Product Strategy Code" := '';
                RecMCItem_G."Purch. Contract Cereals" := FALSE;
                RecMCItem_G."Sales Contract Cereals" := FALSE;
                RecMCItem_G.MODIFY;
              UNTIL RecMCItem_G.NEXT = 0;

            //Suppression de donn'es suite modification structure table CMC MC Item Variant
            IF RecMCItemVariant_G.FINDSET THEN
              REPEAT
                RecMCItemVariant_G."Grouping 1" := '';
                RecMCItemVariant_G."Grouping 2" := '';
                RecMCItemVariant_G.MODIFY;
              UNTIL RecMCItemVariant_G.NEXT = 0;

            //Suppression de donn'es li'es . la pes'e dans plusieurs tables
            cduObjectManagement_G.DeleteFieldData;

            //MAJ d'sinsectisation table CMC Posted Receipts
            //Uniquement pour VAESKEN
            //Sauvegarde de la table CMC Posted Receipts -> CMC Posted Receipts Temp
            IF RecCompany_G.FINDSET THEN
              REPEAT
                RecPostedReceipts_G.CHANGECOMPANY(RecCompany_G.Name);
                RecPostedReceiptsTmp_G.CHANGECOMPANY(RecCompany_G.Name);
                IF RecPostedReceipts_G.FINDSET THEN
                  REPEAT
                    RecPostedReceiptsTmp_G.TRANSFERFIELDS(RecPostedReceipts_G);
                    RecPostedReceiptsTmp_G.INSERT;
                  UNTIL RecPostedReceipts_G.NEXT = 0;
              UNTIL RecCompany_G.NEXT = 0;
            }
            // ************************************ Step 1 **********************************************************************************

            // ************************************ Step 2 **********************************************************************************
            {
            //Restauration des donn'es de la table CMC Campaign dans chaque soci't'
            IF RecCampaignTemp_G.FINDSET AND NOT RecInfoTable_G.GET('',DATABASE::"CMC Campaign") THEN BEGIN
              REPEAT
                IF RecCompany_G.FINDSET THEN
                  REPEAT
                    RecCampaign_G.CHANGECOMPANY(RecCompany_G.Name);
                    RecCampaign_G.TRANSFERFIELDS(RecCampaignTemp_G);
                    RecCampaign_G.INSERT;
                  UNTIL RecCompany_G.NEXT = 0;
              UNTIL RecCampaignTemp_G.NEXT = 0;
              RecCampaignTemp_G.DELETEALL;
            END;

            //MAJ Table CMC Post. Weighing Weighbridge
            //Uniquement pour VAESKEN
            IF RecCompany_G.FINDSET THEN
              REPEAT
                RecPostedReceipts_G.CHANGECOMPANY(RecCompany_G.Name);
                RecPostWeighingWeighbridge_G.CHANGECOMPANY(RecCompany_G.Name);
                IF RecPostedReceipts_G.FINDSET THEN
                  REPEAT
                    RecPostWeighingWeighbridge_G.GET(RecPostedReceipts_G."Weighing No.");
                    RecPostWeighingWeighbridge_G."Manual Weighing No." := RecPostedReceipts_G."Manual Weighing No.";
                    RecPostWeighingWeighbridge_G.MODIFY;
                  UNTIL RecPostedReceipts_G.NEXT = 0;
              UNTIL RecCompany_G.NEXT = 0;

            //Restauration d'sinsectisation table CMC Posted Receipts . partir de la sauvegarde
            //Uniquement pour VAESKEN
            IF RecCompany_G.FINDSET THEN
              REPEAT
                RecPostedReceipts_G.CHANGECOMPANY(RecCompany_G.Name);
                RecPostedReceiptsTmp_G.CHANGECOMPANY(RecCompany_G.Name);
                IF RecPostedReceipts_G.FINDSET THEN
                  REPEAT
                    RecPostedReceiptsTmp_G.GET(RecPostedReceipts_G."Receipt Type",RecPostedReceipts_G."No.");
                    IF RecPostedReceiptsTmp_G."Disinsectization Input" THEN BEGIN
                      RecPostedReceipts_G."Disinsectization Input" := RecPostedReceipts_G."Disinsectization Input"::Curative;
                      RecPostedReceipts_G.MODIFY;
                    END;
                  UNTIL RecPostedReceipts_G.NEXT = 0;
                RecPostedReceiptsTmp_G.DELETEALL;
              UNTIL RecCompany_G.NEXT = 0;
            }
            // ************************************ Step 2 **********************************************************************************

            // ************************************ Step 3 **********************************************************************************
            {
            //MAJ paramStres avanc's
            cduInitAdvSetup_G.InitParameterSetup;
            }
            // ************************************ Step 3 **********************************************************************************
          END;

  }

    BEGIN
