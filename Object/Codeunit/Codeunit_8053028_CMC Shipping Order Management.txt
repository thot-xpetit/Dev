    PROCEDURE CreateShipOrderFrmPurchOrder@8053000(PurchOrderLine_P@8053000 : Record 39);
    VAR
      ShippingOrder_L@8053001 : Record 8053265;
      SalesOrder_L@8053002 : Record 36;
    BEGIN
      ShippingOrder_L.RESET;
      ShippingOrder_L.SETRANGE("Origin Document Type",ShippingOrder_L."Origin Document Type"::"Purchase Order");
      ShippingOrder_L.SETRANGE("Origin Document No.",PurchOrderLine_P."Document No.");
      ShippingOrder_L.SETRANGE("Origin Document Line No.",PurchOrderLine_P."Line No.");
      IF ShippingOrder_L.FINDFIRST THEN
        ERROR(STRSUBSTNO(MsgError8053000_G,PurchOrderLine_P.TABLECAPTION,PurchOrderLine_P."Line No.",
          ShippingOrder_L.TABLECAPTION,ShippingOrder_L."No."));

      ShippingOrder_L.INIT;
      ShippingOrder_L."Origin Document Type" := ShippingOrder_L."Origin Document Type"::"Purchase Order";
      ShippingOrder_L."Origin Document No."  := PurchOrderLine_P."Document No.";
      ShippingOrder_L."Origin Document Line No." := PurchOrderLine_P."Line No.";
      ShippingOrder_L."From Type" := ShippingOrder_L."From Type"::Vendor;
      ShippingOrder_L.VALIDATE("From Code",PurchOrderLine_P."Buy-from Vendor No.");
      IF PurchOrderLine_P."Drop Shipment" THEN BEGIN
        IF SalesOrder_L.GET(SalesOrder_L."Document Type"::Order,PurchOrderLine_P."Sales Order No.") THEN BEGIN
          ShippingOrder_L."Destination Type" := ShippingOrder_L."Destination Type"::Customer;
          ShippingOrder_L.VALIDATE("Destination Code",SalesOrder_L."Sell-to Customer No.");
          ShippingOrder_L.VALIDATE("Destination Adress Code",SalesOrder_L."Ship-to Code");
         END;
      END ELSE BEGIN
        ShippingOrder_L."Destination Type" := ShippingOrder_L."Destination Type"::Location;
        ShippingOrder_L.VALIDATE("Destination Code",PurchOrderLine_P."Location Code");
        ShippingOrder_L.VALIDATE("Destination Adress Code",PurchOrderLine_P."Ship-to Code");
      END;
      ShippingOrder_L.SetSystemCreated(TRUE);
      ShippingOrder_L.VALIDATE("Item No.",PurchOrderLine_P."No.");
      ShippingOrder_L.VALIDATE("Variant Code",PurchOrderLine_P."Variant Code");
      ShippingOrder_L.Quantity := PurchOrderLine_P.Quantity;
      ShippingOrder_L.INSERT(TRUE);
      IF NOT HideDialog_G THEN
        MESSAGE(STRSUBSTNO(Text8053000_G,ShippingOrder_L.TABLECAPTION,ShippingOrder_L."No."));
    END;

    PROCEDURE CreateTransportPurchaseOrder@8053008(VAR ShippingOrder_P@8053000 : Record 8053265);
    VAR
      PurchOrder_L@8053001 : Record 38;
      PurchLine_L@8053002 : Record 39;
      CMCSetup_L@8053003 : Record 8052982;
    BEGIN
      ShippingOrder_P.TESTFIELD("Ship. Purchase Order No.",'');
      ShippingOrder_P.TESTFIELD("Shipping Vendor Code");
      IF NOT CONFIRM(Text8053001_G,FALSE) THEN
        EXIT;

      // cr'ation d'ent^te achat
      PurchOrder_L.INIT;
      PurchOrder_L."Document Type" := PurchOrder_L."Document Type"::Order;
      PurchOrder_L.INSERT(TRUE);
      PurchOrder_L.VALIDATE("Buy-from Vendor No.",ShippingOrder_P."Shipping Vendor Code");
      PurchOrder_L.VALIDATE("Posting Date",WORKDATE);
      CASE TRUE  OF
        ShippingOrder_P."Origin Document Type" IN [ShippingOrder_P."Origin Document Type"::"Sales Order",
        ShippingOrder_P."Origin Document Type"::"Transfer Order"]:
        BEGIN
          PurchOrder_L.VALIDATE("Location Code",ShippingOrder_P."From Code");
        END;
        ShippingOrder_P."Origin Document Type" = ShippingOrder_P."Origin Document Type"::"Purchase Order":
        BEGIN
          PurchOrder_L.VALIDATE("Location Code",ShippingOrder_P."Destination Code");
        END;
      END;
      PurchOrder_L.MODIFY;

      CMCSetup_L.GET;
      CMCSetup_L.TESTFIELD("Transport Item Code");
      // cr'ation d'une ligne
      PurchLine_L.INIT;
      PurchLine_L."Document Type" := PurchLine_L."Document Type"::Order;
      PurchLine_L."Document No." := PurchOrder_L."No.";
      PurchLine_L."Line No." := 10000;
      PurchLine_L."Buy-from Vendor No." := PurchOrder_L."Buy-from Vendor No.";
      PurchLine_L.VALIDATE(Type,PurchLine_L.Type::Item);
      PurchLine_L.VALIDATE("No.",CMCSetup_L."Transport Item Code");
      PurchLine_L.VALIDATE("Location Code",PurchOrder_L."Location Code");
      CASE ShippingOrder_P."Purchase Invoice Type" OF
        ShippingOrder_P."Purchase Invoice Type"::"Per Shipping (Qty=1)" :
        BEGIN
          ShippingOrder_P.TESTFIELD("Number Of Shipping Type");
          PurchLine_L.VALIDATE(Quantity,ShippingOrder_P."Number Of Shipping Type");
        END;
        ShippingOrder_P."Purchase Invoice Type"::"Per Quantity" :
        BEGIN
          ShippingOrder_P.TESTFIELD(Quantity);
          PurchLine_L.VALIDATE(Quantity,ShippingOrder_P.Quantity);
        END;
      END;
      PurchLine_L.INSERT;

      ShippingOrder_P."Ship. Purchase Order No." := PurchOrder_L."No.";
      ShippingOrder_P."Ship. Purchase Order Line No." := PurchLine_L."Line No.";
      ShippingOrder_P.MODIFY;
      IF NOT HideDialog_G THEN
        MESSAGE(Text8053002_G,PurchOrder_L."No.");
    END;
