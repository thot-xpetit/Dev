    PROCEDURE ReturnNbPeriod@8053003(_P_StartingDate@8053000 : Date;_P_EndingDate@8053001 : Date;_P_TypePeriode@8053002 : 'Quinzaine,Mois') RES_NbPeriode : Integer;
    VAR
      L_Date1@8053003 : Date;
      L_Date2@8053004 : Date;
      Date_L@1000000000 : Record 2000000007;
      NoOfMonths_L@1000000001 : Integer;
      SameMonth_L@1000000002 : Boolean;
    BEGIN
      RES_NbPeriode := 1;

      IF _P_StartingDate <= _P_EndingDate THEN BEGIN
        IF _P_StartingDate <> 0D THEN BEGIN
          CASE _P_TypePeriode OF
            _P_TypePeriode::Quinzaine: BEGIN
              L_Date1 := _P_StartingDate;
              L_Date2 := _P_EndingDate;

              IF (DATE2DMY(L_Date1,2) = DATE2DMY(L_Date2,2)) AND (DATE2DMY(L_Date1,3) = DATE2DMY(L_Date2,3)) THEN
                SameMonth_L := TRUE;

               IF SameMonth_L THEN BEGIN
                  IF DATE2DMY(L_Date1, 1) < 16 THEN BEGIN
                    IF DATE2DMY(L_Date2, 1) < 16 THEN
                   RES_NbPeriode := 1
                 ELSE
                   RES_NbPeriode := 2;
                  END ELSE
                    RES_NbPeriode := 1
               END ELSE BEGIN
                 IF DATE2DMY(L_Date1,1) < 16 THEN
                   RES_NbPeriode := 2
                 ELSE
                   RES_NbPeriode := 1;

                 IF DATE2DMY(L_Date2,1) < 16 THEN
                   RES_NbPeriode += 1
                 ELSE
                   RES_NbPeriode += 2;

                  L_Date1 := DMY2DATE(1, DATE2DMY(L_Date1, 2), DATE2DMY(L_Date1, 3));
                  L_Date2 := DMY2DATE(1, DATE2DMY(L_Date2, 2), DATE2DMY(L_Date2, 3));

                   Date_L.SETRANGE("Type de p'riode",Date_L."Type de p'riode"::Mois);
                   Date_L.SETRANGE("D'but p'riode",L_Date1,L_Date2);
                  RES_NbPeriode += (Date_L.COUNT - 2) * 2;
               END;
             END;

             _P_TypePeriode::Mois : BEGIN
              L_Date1 := DMY2DATE(1, DATE2DMY(_P_StartingDate, 2), DATE2DMY(_P_StartingDate, 3));
              L_Date2 := DMY2DATE(1, DATE2DMY(_P_EndingDate, 2), DATE2DMY(_P_EndingDate, 3));

              Date_L.SETRANGE("Type de p'riode",Date_L."Type de p'riode"::Mois);
              Date_L.SETRANGE("D'but p'riode", L_Date1, L_Date2);
              RES_NbPeriode := Date_L.COUNT;
             END;
          END;
        END;
      END ELSE
        RES_NbPeriode := 0;
      EXIT(RES_NbPeriode);
    END;

    PROCEDURE SumQt'Repr@8053006(VAR P_PostedReceipts@8053004 : Record 8052995;L_Qt'ToInv@8053001 : Decimal) Res_Qt' : Decimal;

      L_PostedReceipts.COPYFILTERS(P_PostedReceipts);

      L_PostedReceipts.SETRANGE("Posted Receipt Selected",TRUE);

      IF L_PostedReceipts.FIND('-') THEN BEGIN
        REPEAT
          L_ResteALettrer:=(L_PostedReceipts."Quantity To Invoice"-L_PostedReceipts."Quantity To Lettred");
          IF L_Qt'ToInv+L_ResteALettrer<0 THEN BEGIN
            L_PostedReceipts."Quantity To Lettred":=L_PostedReceipts."Quantity To Lettred"-L_Qt'ToInv;
            P_PostedReceipts."Quantity To Lettred":=P_PostedReceipts."Quantity To Invoice";
            P_PostedReceipts.MODIFY;

          END ELSE BEGIN
            L_PostedReceipts."Quantity To Lettred":=L_PostedReceipts."Quantity To Invoice";
            P_PostedReceipts."Quantity To Lettred":=P_PostedReceipts."Quantity To Lettred"- L_ResteALettrer ;
            L_Qt'ToInv:=P_PostedReceipts."Quantity To Invoice"-P_PostedReceipts."Quantity To Lettred";
            P_PostedReceipts.MODIFY;
          END;
          L_PostedReceipts.MODIFY;
        UNTIL (L_PostedReceipts.NEXT=0) OR (L_Qt'ToInv<=0) OR (L_Out=TRUE);
        Res_Qt':=P_PostedReceipts."Quantity To Lettred";
      END;

      EXIT(Res_Qt');
    END;

              // mise . jour de la ligne brute
              PostedReceiptsLine_L.VALIDATE(Quantity,PostedReceiptsLine_L.Quantity-PostedReceiptsLine2_L.Quantity);
            END;

            PostedReceiptsLine_L."Product Type" := PostedReceiptsLine_L."Product Type"::"Rich Product";
            PostedReceiptsLine_L.MODIFY;

            IF NOT PostedReceipts_P.Calibration THEN BEGIN
              PostedReceipts_P.Calibration := TRUE ;
              PostedReceipts_P.MODIFY
            END ;

          UNTIL PostedReceiptsLine_L.NEXT = 0;
        END;
      UNTIL PostedReceiptAnalysis_L.NEXT = 0;

    END;
