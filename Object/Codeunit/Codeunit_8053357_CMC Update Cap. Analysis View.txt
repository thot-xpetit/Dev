    LOCAL PROCEDURE CalculatePeriodStart@4(PostingDate_P@1000 : Date;DateCompression_P@1001 : Integer) : Date;
    VAR
      AccountingPeriod_L@1002 : Record 50;
    BEGIN
      IF PostingDate_P = CLOSINGDATE(PostingDate_P) THEN
        EXIT(PostingDate_P);

      CASE DateCompression_P OF
        AnalysisReport_G."Compression Date"::Week :
          PostingDate_P := CALCDATE('<CW+1D-1W>',PostingDate_P);
        AnalysisReport_G."Compression Date"::Month :
          PostingDate_P := CALCDATE('<CM+1D-1M>',PostingDate_P);
        AnalysisReport_G."Compression Date"::Quarter :
          PostingDate_P := CALCDATE('<CQ+1D-1Q>',PostingDate_P);
        AnalysisReport_G."Compression Date"::Year :
          PostingDate_P := CALCDATE('<CY+1D-1Y>',PostingDate_P);
        AnalysisReport_G."Compression Date"::Period :
          BEGIN
            IF PostingDate_P <> PrevPostingDate_G THEN BEGIN
              PrevPostingDate_G := PostingDate_P;
              AccountingPeriod_L.SETRANGE("Starting Date",0D,PostingDate_P);
              IF AccountingPeriod_L.FINDLAST THEN BEGIN
                PrevCalculatedPostingDate_G := AccountingPeriod_L."Starting Date"
              END ELSE
                PrevCalculatedPostingDate_G := PostingDate_P;
            END;
            PostingDate_P := PrevCalculatedPostingDate_G;
          END;
      END;
      EXIT(PostingDate_P);
    END;

    LOCAL PROCEDURE UpdateWindowCounter@7(EntryNo_L@1000 : Integer);
    BEGIN
      WinUpdateCounter_G := WinUpdateCounter_G + 1;
      WinTime2_G := TIME;
      IF (WinTime2_G > WinTime1_G + 1000) OR (EntryNo_L = WinLastEntryNo_G) THEN BEGIN
        IF WinLastEntryNo_G <> 0 THEN
          Window_G.UPDATE(3,ROUND(EntryNo_L / WinLastEntryNo_G*10000,1));
        WinPrevEntryNo_G := EntryNo_L;
        WinTotalCounter_G := WinTotalCounter_G + WinUpdateCounter_G;
        IF WinTime2_G <> WinTime1_G THEN
          Window_G.UPDATE(4,ROUND(WinUpdateCounter_G * (1000 / (WinTime2_G - WinTime1_G)),1));
        IF WinTime2_G <> WinTime0_G THEN
          Window_G.UPDATE(5,ROUND(WinTotalCounter_G * (1000 / (WinTime2_G - WinTime0_G)),1));
        WinTime1_G := WinTime2_G;
        WinUpdateCounter_G := 0;
      END;
