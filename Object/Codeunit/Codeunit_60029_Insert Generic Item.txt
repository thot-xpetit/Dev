OBJECT Codeunit 60029 Insert Generic Item
{
  OBJECT-PROPERTIES
  {
    Date=03/01/18;
    Heure=00:00:00;
    Liste versions=LogA07-SCT00.00.00;
  }
  PROPERTIES
  {
    TableNo=60075;
    OnRun=BEGIN
            InsertGenrericItem(Rec);
          END;

  }
  CODE
  {

    PROCEDURE InsertGenrericItem@60012(VAR RecPhytoSalesImport_P@60000 : Record 60075);
    VAR
      RecPhytoInterfaceSetup_L@60014 : Record 60076;
      RecSalesHeader_L@60005 : Record 36;
      RecSalesLine_L@60006 : Record 37;
      RecSalesLine1_L@60007 : Record 37;
      RecSalesLine2_L@60010 : Record 37;
      RecItem_L@60008 : Record 27;
      RecWhseShptHeader_L@60002 : Record 7320;
      RecWhseShptLine_L@60001 : Record 7321;
      RecPhytoSalesAdd_L@60011 : Record 60075;
      RecPhytoSalesImport_L@60012 : Record 60075;
      ReleaseSalesDoc@60013 : Codeunit 414;
      ImportWhseShptPhyto_L@60015 : Codeunit 60027;
      CmdNo_L@60003 : Code[20];
      FreezedDoc_L@60004 : Boolean;
    BEGIN
      CmdNo_L := '';
      RecPhytoInterfaceSetup_L.GET;
      RecWhseShptHeader_L.GET(RecPhytoSalesImport_P."No.");
      CLEAR(RecWhseShptLine_L);
      RecWhseShptLine_L.SETRANGE("No.",RecPhytoSalesImport_P."No.");
      RecWhseShptLine_L.SETRANGE("Source Type",37);
      RecWhseShptLine_L.SETRANGE("Source Subtype",1);
      IF NOT RecPhytoSalesImport_P."Generic Item" AND NOT RecPhytoSalesImport_P."Added Line" THEN
        RecWhseShptLine_L.SETRANGE("Item No.",RecPhytoSalesImport_P."Item No.");
      IF RecWhseShptLine_L.FINDFIRST THEN BEGIN
        RecSalesHeader_L.GET(RecWhseShptLine_L."Source Subtype",RecWhseShptLine_L."Source No.");
        FreezedDoc_L := RecSalesHeader_L."Freezed Document";
        IF FreezedDoc_L THEN BEGIN
          RecSalesHeader_L."Freezed Document" := FALSE;
          RecSalesHeader_L.MODIFY;
        END;
        IF RecSalesHeader_L.Status = RecSalesHeader_L.Status :: Open THEN
          ReleaseSalesDoc.PerformManualRelease(RecSalesHeader_L);

        CLEAR(RecSalesLine_L);
        RecSalesLine_L."Document Type" := RecWhseShptLine_L."Source Subtype";
        RecSalesLine_L."Document No." := RecWhseShptLine_L."Source No.";

        CLEAR(RecSalesLine1_L);
        RecSalesLine1_L.SETRANGE("Document Type",RecWhseShptLine_L."Source Subtype");
        RecSalesLine1_L.SETRANGE("Document No.",RecWhseShptLine_L."Source No.");
        IF RecSalesLine1_L.FINDLAST THEN
          RecSalesLine_L."Line No." := RecSalesLine1_L."Line No." + 10000
        ELSE
          RecSalesLine_L."Line No." := 10000;
        IF NOT RecSalesLine_L.INSERT(TRUE) THEN BEGIN
          RecItem_L.GET(RecPhytoInterfaceSetup_L."Generic Item No.");
          CmdNo_L := ImportWhseShptPhyto_L.InsertSalesHeader(RecPhytoSalesImport_P,RecSalesHeader_L,
                        RecPhytoInterfaceSetup_L."Generic Item No.",RecPhytoSalesImport_P."Quantity Handled",
                            RecWhseShptLine_L."Location Code",'',RecItem_L."Sales Unit of Measure");

          RecSalesLine2_L.GET(RecSalesLine2_L."Document Type" :: Order,CmdNo_L,10000);
        END ELSE BEGIN
          RecSalesLine_L.SuspendStatusCheck(TRUE);
          RecSalesLine_L.VALIDATE(Type,RecSalesLine_L.Type :: Item);
          RecSalesLine_L.VALIDATE("No.",RecPhytoInterfaceSetup_L."Generic Item No.");
          RecSalesLine_L.UpdateQtyFromWhse(TRUE);
          RecSalesLine_L.VALIDATE("Location Code",RecWhseShptLine_L."Location Code");

          RecSalesLine_L.VALIDATE(Quantity,RecPhytoSalesImport_P."Quantity Handled");
          RecSalesLine_L."Order Date" := WORKDATE;
          IF RecSalesLine_L."Promised Delivery Date" = 0D THEN
            RecSalesLine_L.VALIDATE("Requested Delivery Date",WORKDATE);
          RecSalesLine_L.VALIDATE("Promised Delivery Date",WORKDATE);
          RecSalesLine_L.MODIFY(TRUE);
          CLEAR(RecSalesLine2_L);
          RecSalesLine2_L := RecSalesLine_L;
        END;

        ImportWhseShptPhyto_L.GetSourceDoc(RecWhseShptHeader_L,RecSalesLine2_L,RecPhytoSalesImport_P);

        CLEAR(RecPhytoSalesAdd_L);
        RecPhytoSalesAdd_L.TRANSFERFIELDS(RecPhytoSalesImport_P);
        // --> 20171129-LogA07-SCT00.00.00-3313-NED
        {
        CLEAR(RecPhytoSalesImport_L);
        IF RecPhytoSalesImport_L.FINDLAST THEN
          RecPhytoSalesAdd_L."Entry No." := RecPhytoSalesImport_L."Entry No." + 1
        ELSE
          RecPhytoSalesAdd_L."Entry No." := 1;
        }
        // <-- 20171129-LogA07-SCT00.00.00-3313-NED
        RecPhytoInterfaceSetup_L.TESTFIELD("Generic Item No.");
        RecPhytoSalesAdd_L."Item No." := RecPhytoInterfaceSetup_L."Generic Item No.";
        RecPhytoSalesAdd_L."Qty. to Handle" := RecSalesLine2_L.Quantity;
        RecPhytoSalesAdd_L."Quantity Handled" := RecSalesLine2_L.Quantity;
        RecPhytoSalesAdd_L."Added Line" := FALSE;
        RecPhytoSalesAdd_L."Import Status" := RecPhytoSalesAdd_L."Import Status" :: Treated;
        RecPhytoSalesAdd_L."Error Message" := '';
        RecPhytoSalesAdd_L."Import Error"  := FALSE;
        RecPhytoSalesAdd_L.Treated := TRUE;
        RecPhytoSalesAdd_L."Source Entry No." := RecPhytoSalesImport_L."Entry No.";
        RecPhytoSalesAdd_L."Generic Item" := TRUE;
        RecPhytoSalesAdd_L."Order No." := RecSalesLine2_L."Document No.";
        RecPhytoSalesAdd_L."Order Line No." := RecSalesLine2_L."Line No.";
        // --> 20171129-LogA07-SCT00.00.00-3313-NED
        //RecPhytoSalesAdd_L.INSERT;
        RecPhytoSalesAdd_L.INSERT(TRUE);
        // <-- 20171129-LogA07-SCT00.00.00-3313-NED
      END;
      COMMIT;
    END;

    BEGIN
    {
      // --> 3LI BEGIN
      20170502-LogA07-SCT00.00.00-3208-NED : ADDOBJ
      20171129-LogA07-SCT00.00.00-3313-NED : MODCODE (Adaptation)
      // <-- 3LI END
    }
    END.
  }
}
