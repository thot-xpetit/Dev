      END;

    PROCEDURE DateNotAllowed@1(PostingDate@1000 : Date) : Boolean;
    BEGIN
      // --> 20130531-Gen08-CMC02.02.00-BRA01
      // --> 20150602-GEN08-CMC02.02.00-1961-DRN01
      IF NOT ItemJnlPosting_G THEN BEGIN
      // <-- 20150602-GEN08-CMC02.02.00-1961-DRN01
        GLSetup.GET;
        IF USERID <> '' THEN
          IF UserSetup.GET(USERID) THEN BEGIN
            // --> 20150602-GEN08-CMC02.02.00-1961-DRN01
            //BEFORE
            // --> 20151123-GEN08-ADV01.01.00-2539-CJY01
            //R'activation code :
            InventorySetup_G.GET;
            // <-- 20151123-GEN08-ADV01.01.00-2539-CJY01
            // <-- 20150602-GEN08-CMC02.02.00-1961-DRN01
            IF UserSetup."User Groupe Code" <> '' THEN
              IF NOT UserGroup_G.GET(UserSetup."User Groupe Code") THEN
                UserGroup_G.INIT;
            IF (UserGroup_G."Post. Allow.on Server Workdate" OR UserSetup."Post. Allow.on Server Workdate") THEN BEGIN
              // --> 20150602-GEN08-CMC02.02.00-1961-DRN01
              //BEFORE
              // --> 20151123-GEN08-ADV01.01.00-2539-CJY01
              //R'activation code :
              IF ((GLSetup."Server Workdate" <> PostingDate) AND (InventorySetup_G."Inventory Diff. Posting Date" <> PostingDate)) THEN
              // <-- 20151123-GEN08-ADV01.01.00-2539-CJY01
              //AFTER
              IF (GLSetup."Server Workdate" <> PostingDate) THEN
              // <-- 20150602-GEN08-CMC02.02.00-1961-DRN01
                EXIT(TRUE);
            END;
          END;
      // --> 20150602-GEN08-CMC02.02.00-1961-DRN01
      END;
      // <-- 20150602-GEN08-CMC02.02.00-1961-DRN01
      // <-- 20130531-Gen08-CMC02.02.00-BRA01

      IF (AllowPostingFrom = 0D) AND (AllowPostingTo = 0D) THEN BEGIN
       IF USERID <> '' THEN
         IF UserSetup.GET(USERID) THEN BEGIN
           AllowPostingFrom := UserSetup."Allow Posting From";
           AllowPostingTo := UserSetup."Allow Posting To";
         END;
         IF (AllowPostingFrom = 0D) AND (AllowPostingTo = 0D) THEN BEGIN
          GLSetup.GET;
          AllowPostingFrom := GLSetup."Allow Posting From";
          AllowPostingTo := GLSetup."Allow Posting To";
         END;
         IF AllowPostingTo = 0D THEN
           AllowPostingTo := 31129999D;
         END;


         EXIT((PostingDate < AllowPostingFrom) OR (PostingDate > AllowPostingTo));
    END;
