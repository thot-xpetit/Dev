        SalesLine2_L.VALIDATE("No.",SalesHeader_P."CMC Prep. Deduct. G/L Acc. No.");
        // --> 20260409-WI00-ADV01.01.00-43404-IJI
        //BEFORE
        //SalesLine2_L.VALIDATE(Quantity,1);
        //SalesLine2_L.VALIDATE("Unit Price",-AmntToDeduct_L);
        //AFTER
        SalesLine2_L.VALIDATE(Quantity,-1);
        SalesLine2_L.VALIDATE("Unit Price",ABS(AmntToDeduct_L));

        SLPrepBuffer_L.INIT;

        SLPrepBuffer_L."Document No." := SalesLine2_L."Document No.";
        SLPrepBuffer_L."Document Line No." := SalesLine2_L."Line No.";
        SLPrepBuffer_L."Customer No." := SalesHeader_P."Bill-to Customer No.";
        SLPrepBuffer_L."G/L Account No." := SalesHeader_P."CMC Prep. Deduct. G/L Acc. No.";

        SLPrepBuffer_L.Type := SalesLine2_L.Type;
        SLPrepBuffer_L."No." := SalesLine2_L."No.";
        SLPrepBuffer_L.Description := SalesLine2_L.Description;

    END;
