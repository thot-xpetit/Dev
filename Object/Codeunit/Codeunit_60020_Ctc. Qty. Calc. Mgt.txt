    PROCEDURE GetContract@1000000000(VAR tmpContract@1000000000 : TEMPORARY Record 8052983;SpeciesCode@1000000002 : Code[10];CompaignNo@1000000003 : Code[10];DateFilter@1000000005 : Text[250];ReceiptDateFilter@1000000006 : Text[250]);
    VAR
      CalculationSetup@1000000001 : Record 60093;
      ErrorMsg@1000000004 : Text[255];
    BEGIN
      // check if all data are valid
      IF (CheckData(SpeciesCode, CompaignNo, DateFilter, ReceiptDateFilter, ErrorMsg)) THEN BEGIN
        // Initialize the internal data
        Initialize(tmpContract);
        // Search the calculation setup
        CalculationSetup.SETRANGE("Campaign No.", CompaignNo);
        CalculationSetup.SETRANGE("Species Code", SpeciesCode);
        IF (CalculationSetup.FINDFIRST()) THEN BEGIN
          InitDialog() ;
          NoOfRecsSetup := CalculationSetup.COUNT;
          REPEAT
            // Search the contracts
            SetContracts(CalculationSetup, tmpContract, DateFilter, ReceiptDateFilter, TRUE);
            UpdateDialog(1, CurrRecSetup, NoOfRecsSetup);
          UNTIL(CalculationSetup.NEXT() = 0);
          // Add total per Vendor
          SetTotalVendor(tmpContract, CompaignNo, SpeciesCode);
          // calculate the quantity to be applied
          CalculateApplyQty(tmpContract, CompaignNo, SpeciesCode);
          CloseDialog();
        END;
        IF (tmpContract.FINDFIRST()) THEN;
        tmpContract.SETCURRENTKEY("Paper Reference Contract No.");
      END ELSE BEGIN
        ERROR(ErrorMsg);
      END;
    END;

    PROCEDURE SetContracts@1000000005(CalculationSetup@1000000000 : Record 60093;VAR tmpContract@1000000001 : TEMPORARY Record 8052983;DateFilter@1000000003 : Text[250];DateReceiptFilter@1000000004 : Text[250];CheckReceipt@1000000005 : Boolean);
    VAR
      CollectionContract@1000000002 : Record 8052983;
      InsertContract@1000000006 : Boolean;
    BEGIN
      CollectionContract.SETCURRENTKEY(Type,"Vendor No.","Campaign No.","Collection Type");
      CollectionContract.SETRANGE(Type, CollectionContract.Type::Contract);
      CollectionContract.SETRANGE("Species Code", CalculationSetup."Species Code");
      CollectionContract.SETRANGE("Item No.", CalculationSetup."Item No.");
      CollectionContract.SETRANGE("Campaign No.", CalculationSetup."Campaign No.");
      CollectionContract.SETFILTER("Contract Type", '%1|%2', CalculationSetup."Reference Contract Type",
                                                             CalculationSetup."PMG Contract Type");

      IF (CustomerNoFilter <> '') THEN BEGIN
        //CollectionContract.SETFILTER("Vendor No.", '%1', CustomerNoFilter);
        CollectionContract.SETFILTER("Vendor No.", CustomerNoFilter);
      END;
      //IF ((DateMin <> 0D) OR (DateMax <> 0D)) THEN BEGIN
      //  CollectionContract.SETRANGE("Creation Date", DateMin, DateMax);
      //END;

        NoOfRecsCtc := CollectionContract.COUNT;
        CurrRecCtc := 0;

          // get the receipts line only if checkreceipt

            InsertContract := GetReceiptLines(CollectionContract, DateFilter, DateReceiptFilter);
          END ELSE BEGIN
            InsertContract := TRUE;
          END;
          IF (InsertContract) THEN BEGIN
            // Create the contract line to adjust quantity
            tmpContract.INIT();
            tmpContract.TRANSFERFIELDS(CollectionContract);
            tmpContract."Calculated Unit Price" := 0;
            // Create the ordre key
            tmpContract."Paper Reference Contract No." := tmpContract."Vendor No." + '-' + PADSTR(FORMAT(CalculationSetup.Order), 4, '0');
            tmpContract.Priority := 1;

              IF (NOT tmpContract.INSERT()) THEN;
            END;
          END;
          UpdateDialogText(2, tmpContract."No.");
        UNTIL(CollectionContract.NEXT() = 0);
      END;
    END;

    PROCEDURE UpdateDialog@1000000006(index@1000000000 : Integer;VAR curRec@1000000001 : Integer;Total@1000000003 : Integer);
    VAR
      Progress@1000000002 : Integer;
    BEGIN
      curRec += 1;
      IF Total <= 100 THEN BEGIN
          Progress := ((curRec / Total * 10000) DIV 1);
      END  ELSE BEGIN

    PROCEDURE CalculateQuantity@1000000012(VAR TmpCtc@1000000000 : TEMPORARY Record 8052983;Setup@1000000001 : Record 60093);
    VAR
      RefContract@1000000002 : Record 8052983;
      QtyBase@1000000003 : Decimal;
      ReceiptQty@1000000004 : Decimal;
      PctCtc@1000000005 : Decimal;
      CalcQty@1000000006 : Boolean;
      PctRef@1000000007 : Decimal;
      PctPMG@1000000008 : Decimal;
      IsSpecificCtc@1000000009 : Boolean;
    BEGIN
      CalcQty := FALSE;
      PctRef := Setup."% Reference Contract";
      PctPMG := Setup."% PMG Contract";

      IsSpecificCtc := FALSE;

      IF (NOT HasInitialQuantity(TmpCtc."No.", QtyBase)) THEN BEGIN
        // IF Contract Type has not contrepartie then PCT = 100% and Not the % of reference Contract
        // Get Initial Qty From RefContract
        //IF ((Setup."Reference Contract Type" = Setup."PMG Contract Type") AND
        //   (Setup."% Reference Contract"=100) AND
        //   (Setup."% PMG Contract" = 100)) THEN BEGIN
        IF (IsSpecificContract(TmpCtc, Setup, PctRef, PctPMG)) THEN BEGIN
           RefContract := TmpCtc;
           CalcQty := TRUE;
           IsSpecificCtc := TRUE;
        END ELSE BEGIN
          RefContract.SETRANGE(Type, TmpCtc.Type);
          RefContract.SETRANGE("Vendor No.", TmpCtc."Vendor No.");
          RefContract.SETRANGE("Campaign No.", TmpCtc."Campaign No.");
          RefContract.SETRANGE("Species Code", TmpCtc."Species Code");
          RefContract.SETRANGE("Item No.", TmpCtc."Item No.");
          RefContract.SETRANGE("Contract Type", Setup."Reference Contract Type");
          IF (RefContract.FINDFIRST()) THEN BEGIN
            CalcQty := TRUE;
          END;
        END;
        IF (CalcQty) THEN BEGIN
          //IF (RefContract."Threshold Percentage" = 0) THEN BEGIN
            IF (TmpCtc."Contract Type" = Setup."Reference Contract Type") THEN BEGIN
              //PctCtc := Setup."% Reference Contract";
              PctCtc := PctRef;
            END ELSE BEGIN
              IF (TmpCtc."Contract Type" = Setup."PMG Contract Type") THEN BEGIN
                //PctCtc := Setup."% PMG Contract";
                PctCtc := PctPMG;
              END;
            END;
            //QtyBase := ((RefContract.Quantity * 100) / Setup."% Reference Contract");
            QtyBase := ((RefContract.Quantity * 100) / PctRef);
            //QtyBase := QtyBase - RefContract.Quantity;
            QtyBase := (QtyBase * PctCtc) / 100;
            AddInitialQuantity(TmpCtc."No.", QtyBase);
          //END;
        END;
      END ELSE BEGIN
      END;
      //InitValue
      // OutStanding Quantity
      TmpCtc."Outstanding Quantity" := 0;
      // Quantity Base
      TmpCtc."MATIF Basis Complement" := 0;
      // Receive Quantity
      TmpCtc.Margin := 0;
      // Set the Base quantity
      TmpCtc."MATIF Basis Complement" := QtyBase;
      ReceiptQty := GetReceiptQty(TmpCtc, TRUE);
      // Applied Quantity
      TmpCtc.Supplement := ReceiptQty;
      // Calculate the outstanding qty
      TmpCtc."Outstanding Quantity" := TmpCtc."MATIF Basis Complement" - ReceiptQty;
      // Calculate the quantity for all recepipt
      TmpCtc."Financial Expense" := GetQuantityAllReceipt(TmpCtc,  GetReceiptQty(TmpCtc, FALSE), Setup);
      // contract is 100% depository (True) or 100% Acompte (FALSE)
      TmpCtc."Farm Storage" := IsDepositoryContract(TmpCtc);
      // PrimerFerme Contract
      //TmpCtc.Cancelled :=((Setup."Reference Contract Type" = Setup."PMG Contract Type") AND
      //                    (Setup."% Reference Contract"=100) AND
      //                    (Setup."% PMG Contract" = 100));
      TmpCtc.Cancelled := IsSpecificCtc;
      {
      //IF (TmpCtc."Contract Type" = Setup."PMG Contract Type") THEN BEGIN
        //IF (TmpCtc."Threshold Percentage" = 0) THEN BEGIN
          // Get Setup."PMG Contract Type"
          RefContract.SETRANGE(Type, TmpCtc.Type);
          RefContract.SETRANGE("Vendor No.", TmpCtc."Vendor No.");
          RefContract.SETRANGE("Campaign No.", TmpCtc."Campaign No.");
          RefContract.SETRANGE("Species Code", TmpCtc."Species Code");
          RefContract.SETRANGE("Item No.", TmpCtc."Item No.");
          RefContract.SETRANGE("Contract Type", Setup."Reference Contract Type");
          IF (RefContract.FINDFIRST()) THEN BEGIN
            IF (RefContract."Threshold Percentage" = 0) THEN BEGIN
              //InitValue
              TmpCtc."Outstanding Quantity" := 0;
              TmpCtc."MATIF Basis Complement" := 0;
              TmpCtc.Margin := 0;
              QtyBase := ((RefContract.Quantity * 100) / Setup."% Reference Contract");
              TmpCtc."MATIF Basis Complement" := QtyBase - RefContract.Quantity;
              ReceiptQty := GetReceiptQty(TmpCtc);
              TmpCtc."Outstanding Quantity" := TmpCtc."MATIF Basis Complement" - ReceiptQty;
            END;
          END;
        //END;
      //END;
      }
    END;

    PROCEDURE SetApplyQuantity@1000000011(VAR TmpCtc@1000000000 : TEMPORARY Record 8052983) ReturnValue : Boolean;
    VAR
      QtyToAffect@1000000001 : Decimal;
    BEGIN
      ReturnValue := FALSE;
      // Get the Temporary Posted Receptions Line
      TmpCMCPostedReceiptLine.RESET();
      TmpCMCPostedReceiptLine.SETRANGE("Distributed Vendor No.", TmpCtc."Vendor No.");
      IF (TmpCMCPostedReceiptLine.FINDFIRST()) THEN BEGIN
        TmpCtc.Margin := TmpCMCPostedReceiptLine.Quantity;
        {
        IF ((TmpCMCPostedReceiptLine."Quantity To Affect" > 0) AND (TmpCtc."Outstanding Quantity" > 0)) THEN BEGIN
            QtyToAffect := Min(TmpCMCPostedReceiptLine."Quantity To Affect",
                               TmpCtc."Outstanding Quantity");
            TmpCtc."Calculated Unit Price" := QtyToAffect;
            TmpCMCPostedReceiptLine."Quantity To Affect" := TmpCMCPostedReceiptLine."Quantity To Affect" - QtyToAffect;
            TmpCMCPostedReceiptLine.MODIFY();
        END;
        }
        ReturnValue := TRUE;
      END;
      TmpCMCPostedReceiptLine.RESET();
    END;

    PROCEDURE CalculateApplyQty@1000000013(VAR tmpContract@1000000000 : TEMPORARY Record 8052983;CompaignNo@1000000005 : Code[10];SpeciesCode@1000000006 : Code[10]);
    VAR
      QtyToAffect@1000000001 : Decimal;
      CtcQtySetup@1000000002 : Record 60093;
      QtyPMG@1000000003 : Decimal;
      QtyRef@1000000004 : Decimal;
      OutstandingQty@1000000007 : Decimal;
      ReceiptQty@1000000008 : Decimal;
      AffectedQty@1000000009 : Decimal;
      Repository@1000000010 : Boolean;
      PctRef@1000000011 : Decimal;
      PctPMG@1000000012 : Decimal;
      UseTotalQty@1000000013 : Boolean;
      AlreadyAffectedQty@1000000014 : Decimal;
    BEGIN
      // Calculate the qty to apply at the contract
      TmpCMCPostedReceiptLine.RESET();

      IF (TmpCMCPostedReceiptLine.FINDFIRST()) THEN BEGIN
        REPEAT
          IF (GetTotalLine(TmpTotalConrtact, TmpCMCPostedReceiptLine."Distributed Vendor No.")) THEN BEGIN
            QtyToAffect := TmpTotalConrtact.Margin;
            UseTotalQty := TRUE;
          END ELSE BEGIN
            QtyToAffect := TmpCMCPostedReceiptLine."Quantity To Affect";
            UseTotalQty := FALSE;
          END;
          CtcQtySetup.RESET();
          ReceiptQty := 0;
          AffectedQty := 0;
          // Loop arount the qtysetup
          CtcQtySetup.SETRANGE("Campaign No.", CompaignNo);
          CtcQtySetup.SETRANGE("Species Code", SpeciesCode);
          IF (CtcQtySetup.FINDFIRST()) THEN BEGIN
            REPEAT
              Repository := TRUE;
              // The the Temporary Contract
              tmpContract.RESET();
              tmpContract.SETCURRENTKEY("Paper Reference Contract No.");
              tmpContract.SETRANGE("Vendor No.", TmpCMCPostedReceiptLine."Distributed Vendor No.");
              tmpContract.SETRANGE("Species Code", CtcQtySetup."Species Code");
              tmpContract.SETRANGE("Item No.", CtcQtySetup."Item No.");
              tmpContract.SETFILTER("Contract Type", '%1|%2', CtcQtySetup."Reference Contract Type", CtcQtySetup."PMG Contract Type");
              tmpContract.SETFILTER(Type, '<>%1', 3);

              // The contract is identified 100% depository
              tmpContract.CALCSUMS("Threshold Percentage");
              Repository := Repository AND (tmpContract."Threshold Percentage" = 0);

              IF (tmpContract.FINDFIRST()) THEN BEGIN
                // Calculate the Proportionnal Qty
                IsSpecificContract(tmpContract, CtcQtySetup, PctRef, PctPMG);
                //QtyPMG := (QtyToAffect * CtcQtySetup."% PMG Contract") / 100;
                //QtyRef := (QtyToAffect * CtcQtySetup."% Reference Contract") / 100;
                QtyPMG := (QtyToAffect * PctPMG) / 100;
                QtyRef := (QtyToAffect * PctRef) / 100;

                REPEAT
                  OutstandingQty := tmpContract."Outstanding Quantity";
                  IF (UseTotalQty) THEN BEGIN
                    AlreadyAffectedQty := tmpContract.Supplement;
                  END ELSE BEGIN
                    AlreadyAffectedQty := 0;
                  END;

                  // if contract adjust quantity and Not 100% depository, the outstanding quantity is QtyToAffect
                  //IF (tmpContract."Farm Storage") THEN BEGIN
                  IF ((tmpContract."Farm Storage") AND (NOT Repository)) THEN BEGIN
                    OutstandingQty := QtyToAffect;
                  END;
                  //IF (tmpContract."Threshold Percentage" = 0) THEN BEGIN
                  IF (Repository) THEN BEGIN
                    // 100% Depository
                    IF (tmpContract."Contract Type" = CtcQtySetup."Reference Contract Type") THEN BEGIN
                      tmpContract."Calculated Unit Price" := Min(GetQuantity(QtyRef, AlreadyAffectedQty) , OutstandingQty);
                    END;
                    IF (tmpContract."Contract Type" = CtcQtySetup."PMG Contract Type") THEN BEGIN
                      tmpContract."Calculated Unit Price" := Min(GetQuantity(QtyPMG, AlreadyAffectedQty), OutstandingQty);
                    END;
                  END ELSE BEGIN
                    // 100% Deposit
                    // << CLA69425-201910-21-XPE1
                    {
                    IF (tmpContract."Contract Type" = CtcQtySetup."Reference Contract Type") THEN BEGIN
                      tmpContract."Calculated Unit Price" := GetQuantity(QtyRef, AlreadyAffectedQty);
                    END;
                    IF (tmpContract."Contract Type" = CtcQtySetup."PMG Contract Type") THEN BEGIN
                      tmpContract."Calculated Unit Price" := GetQuantity(QtyPMG, AlreadyAffectedQty);
                    END ELSE BEGIN
                      tmpContract."Calculated Unit Price" := GetQuantity(QtyToAffect, AlreadyAffectedQty);
                    END;
                    }
                    CASE tmpContract."Contract Type" OF
                      CtcQtySetup."Reference Contract Type" : BEGIN
                        tmpContract."Calculated Unit Price" := GetQuantity(QtyRef, AlreadyAffectedQty);
                      END;
                      CtcQtySetup."PMG Contract Type" : BEGIN

                    //QtyRef := (QtyToAffect * CtcQtySetup."% Reference Contract") / 100;

                    QtyRef := (QtyToAffect * PctRef) / 100;

                UNTIL(tmpContract.NEXT() = 0);

                tmpContract.MODIFYALL("Blocking Characteristics", Repository);
