    PROCEDURE CreateCommentInvLine@60001();
    BEGIN
      // creation of comment invoice line with contract no.
      PurchLineRec_G.INIT;
      PurchLineRec_G.VALIDATE("Document Type",PurchHeaderRec_G."Document Type");
      PurchLineRec_G.VALIDATE("Document No.",PurchHeaderRec_G."No.");
      PurchLineRec_G."Line No." := GetNextInvLineNo();
      PurchLineRec_G.Description := STRSUBSTNO(Text50000,ContractNoRec_G."No.",BrokerageInvSchLinesRec_G."Starting Date",
        BrokerageInvSchLinesRec_G."Ending Date");
      PurchLineRec_G.INSERT(TRUE);
    END;

    PROCEDURE CreateBrokerLine@60000(ItemCode_P@60004 : Code[20];LocationCode_P@60005 : Code[10]);
    VAR
      CMCSetupRec_L@60000 : Record 8052982;
      TranspMethSetupRec_L@60001 : Record 60023;
      DocDimensionRec_L@60003 : Record 357;
      GLSetupRec_L@60007 : Record 98;
      DefaultDimRec_L@60008 : Record 352;
      LocationRec_L@60006 : Record 14;
      DimensionValueRec_L@60009 : Record 349;
    BEGIN
      CMCSetupRec_L.GET;
      GLSetupRec_L.GET;
      IF CMCSetupRec_L."GL account for broker invoice" = '' THEN
        ERROR(STRSUBSTNO(Text50001),CMCSetupRec_L.FIELDCAPTION("GL account for broker invoice"));

      PurchLineRec_G.INIT;
      PurchLineRec_G.VALIDATE("Document Type",PurchHeaderRec_G."Document Type");
      PurchLineRec_G.VALIDATE("Document No.",PurchHeaderRec_G."No.");
      PurchLineRec_G."Line No." := GetNextInvLineNo();
      PurchLineRec_G.VALIDATE(Type,PurchLineRec_G.Type :: "G/L Account");
      PurchLineRec_G.VALIDATE("No.",CMCSetupRec_L."GL account for broker invoice");
      PurchLineRec_G.VALIDATE(Quantity,BrokerageInvSchLinesRec_G."Qty to Invoice");
      PurchLineRec_G.VALIDATE("Direct Unit Cost",ContractNoRec_G."Broker Com. Unit Price");
      PurchLineRec_G.INSERT(TRUE);

      //-- insert Item Dimension from item contract
      IF DefaultDimRec_L.GET(DATABASE::Item,ItemCode_P,DefaultDimRec_L.Type::"Regular Dimension",
        GLSetupRec_L."Shortcut Dimension 2 Code") THEN BEGIN
        DocDimensionRec_L.INIT;
        DocDimensionRec_L."Table ID" := DATABASE::"Purchase Line";
        DocDimensionRec_L."Document Type" := DocDimensionRec_L."Document Type"::Invoice;
        DocDimensionRec_L."Document No." :=  PurchHeaderRec_G."No.";
        DocDimensionRec_L."Line No." :=  PurchLineRec_G."Line No.";
        DocDimensionRec_L.Type := DocDimensionRec_L.Type::"Regular Dimension";
        DocDimensionRec_L."Dimension Code" := DefaultDimRec_L."Dimension Code";
        DocDimensionRec_L."Dimension Value Code" := DefaultDimRec_L."Dimension Value Code";
        DocDimensionRec_L.INSERT(TRUE);
      END;

      //-- insert Loc Dimension from default Location contract
      IF LocationRec_L.GET(LocationCode_P) THEN BEGIN
        IF DimensionValueRec_L.GET(GLSetupRec_L."Shortcut Dimension 1 Code",LocationRec_L."Analytic Section") THEN BEGIN
          DocDimensionRec_L.INIT;
          DocDimensionRec_L."Table ID" := DATABASE::"Purchase Line";
          DocDimensionRec_L."Document Type" := DocDimensionRec_L."Document Type"::Invoice;
          DocDimensionRec_L."Document No." :=  PurchHeaderRec_G."No.";
          DocDimensionRec_L."Line No." :=  PurchLineRec_G."Line No.";
          DocDimensionRec_L.Type := DocDimensionRec_L.Type::"Regular Dimension";
          DocDimensionRec_L."Dimension Code" := DimensionValueRec_L."Dimension Code";
          DocDimensionRec_L."Dimension Value Code" := DimensionValueRec_L.Code;
          DocDimensionRec_L.INSERT(TRUE);
        END;
      END;
    END;
