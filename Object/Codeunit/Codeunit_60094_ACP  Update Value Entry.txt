    PROCEDURE CreateACP@1000000001();
    BEGIN
      // Cr'ation COST PLUS mode UPDATE
      //

      // Table Tempo pour cr'ation Value Entry Cost Plus
      CLEAR(TempACPPurchLine_G);
      IF NOT TempACPPurchLine_G.ISEMPTY THEN
        TempACPPurchLine_G.DELETEALL;
      CLEAR(TempPurchLine_G);
      IF NOT TempPurchLine_G.ISEMPTY THEN
        TempPurchLine_G.DELETEALL;

      // Initialisation TmpPurchLine
      CLEAR(TempPurchLine_G);
      TempPurchLine_G."CMC Campaign No."     := TbItemLedgerEntry_G."Campaign No.";
      TempPurchLine_G."Unit of Measure Code" := TbItemLedgerEntry_G."Unit of Measure Code";
      TempPurchLine_G."Buy-from Vendor No."  := TbItemLedgerEntry_G."Buy-from Vendor No.";
      TempPurchLine_G."Pay-to Vendor No."    := TbItemLedgerEntry_G."Pay-to Vendor No.";
      TempPurchLine_G."No."                  := TbItemLedgerEntry_G."Item No.";
      TempPurchLine_G."Variant Code"         := TbItemLedgerEntry_G."Variant Code";
      TempPurchLine_G."Species Code"         := TbItemLedgerEntry_G."Species No.";
      TempPurchLine_G."Sub-Family Code"      := TbValueEntry_G."Sub-Family Code";
      TempPurchLine_G."Family Code"          := TbValueEntry_G."Family Code";
      TempPurchLine_G."Activity Code"        := TbValueEntry_G."Activity Code";
      TempPurchLine_G."Document Type"        := TbItemLedgerEntry_G."Document Type";
      TempPurchLine_G."Document No."         := TbItemLedgerEntry_G."Document No.";
      TempPurchLine_G."Line No."             := TbItemLedgerEntry_G."Document Line No.";
      TempPurchLine_G.Quantity               := TbValueEntry_G."Valued Quantity";
      TempPurchLine_G."Line Discount Amount" := 0;
      TempPurchLine_G."Direct Unit Cost"     := TbValueEntry_G."Cost per Unit";
      TempPurchLine_G."Order Date"           := TbItemLedgerEntry_G."Order Date";

      // Table Tempo du r'sultat
      CLEAR(TempACPPurchLine_G);

      // Cr'ation dans table tempo si correspondance avec param'trage . appliquer
      CuACPManagement_G.FindAcpWithSetup(TempPurchLine_G,TempACPPurchLine_G,TempPurchLine_G."Order Date",ACPCode_G);

      // Cr'ation Value Entry ACP
      IF TempACPPurchLine_G.FINDSET THEN
        REPEAT
          CLEAR(TbValueEntryACP_G);
          TbValueEntryACP_G.INIT;
          TbValueEntryACP_G."Value Entry No."  := TbValueEntry_G."Entry No.";
          TbValueEntryACP_G."Acp Code"         := ACPCode_G;
          TbValueEntryACP_G.Rank               := TempACPPurchLine_G.Rank;
          TbValueEntryACP_G."Base Value"       := TempACPPurchLine_G."Base Value";
          TbValueEntryACP_G."Acp %"            := TempACPPurchLine_G."ACP %";
          TbValueEntryACP_G."Acp Amount"       := TempACPPurchLine_G."ACP Amount";
          TbValueEntryACP_G."ACP Vendor No."   := TempACPPurchLine_G."ACP vendor No.";
          //TEST
          {BEFORE
          TbValueEntryACP_G."Final Price"      := TbValueEntryACP_G."Base Value" - TbValueEntryACP_G."Acp Amount";
          AFTER}
          TbValueEntryACP_G."Final Price"      := CuACPManagement_G.GetPreviousFinalPrice(TbValueEntryACP_G) -
                                                  TbValueEntryACP_G."Acp Amount";
          //TEST
          TbValueEntryACP_G.INSERT(TRUE);
        UNTIL TempACPPurchLine_G.NEXT = 0;

      // Cr'ation ACP Ledger Entry
      CuACPManagement_G.CreateAcpLedgerEntry(TbValueEntry_G,FALSE,ACPCode_G);
      //fonction de refresh . ajouter sur les Acp Value Entry Suivantes
      CuACPManagement_G.ManageNextAcpEntry(TbValueEntryACP_G);
    END;

    PROCEDURE ACPToUpdate@1000000005();
    VAR
      AmountInv_L@1000000000 : Decimal;
    BEGIN
      //
      // Mise . jour des 'critures ACP . partir d'un code ACP
      //

      CLEAR(TbPurchACPLedgerEntry_G);
      TbPurchACPLedgerEntry_G.SETRANGE("Item Ledger Entry Type",TbValueEntry_G."Item Ledger Entry Type");
      TbPurchACPLedgerEntry_G.SETRANGE("Item Ledger Entry No.",TbValueEntry_G."Item Ledger Entry No.");
      TbPurchACPLedgerEntry_G.SETRANGE("Value Entry No.",TbValueEntry_G."Entry No.");
      TbPurchACPLedgerEntry_G.SETRANGE("ACP Code" ,ACPCode_G);
      // --> 20200313-WI00-ADV01.01.00-3446-CJY01
      TbPurchACPLedgerEntry_G.SETRANGE("ACP  No. final doc. purch.",0);
      // <-- 20200313-WI00-ADV01.01.00-3446-CJY01

      //on ne doit pas traiter une 'criture sold'e volontairement
      // --> 20200416-WI00-ADV01.01.00-3446-CJY01
      TbPurchACPLedgerEntry_G.SETRANGE(Canceled,FALSE);
      // <-- 20200416-WI00-ADV01.01.00-3446-CJY01

      IF TbPurchACPLedgerEntry_G.FINDFIRST THEN BEGIN
          // Mise . jour Value Entry ACP
          UpdateACPValueEntry;
          // V'rification que la nouvelle valeur ACP n'est pas > au montant d'j. factur'
          AmountInv_L := ABS(TbPurchACPLedgerEntry_G."ACP Unit Amount (Invoice)") - ABS(TbPurchACPLedgerEntry_G.CalcACPUnitPriceOut);

          IF TbValueEntryACP_G."Acp Amount" < AmountInv_L THEN
                   ERROR(STRSUBSTNO(TxtErr001_G,TbPurchACPLedgerEntry_G."Document No.",TbPurchACPLedgerEntry_G."Document Line No.",
                              TbPurchACPLedgerEntry_G."Entry No.",AmountInv_L,TbValueEntryACP_G."Acp Amount"));

          // Mise . jour ACP Value Ledger Entry
          UpdateACPValueLedgerEntry;
          //gestion de l''criture d'taill'es de type "mise . jour "
          CuACPManagement_G.ManageAcpDetailEntry(TbPurchACPLedgerEntry_G,1);
          //enfin, mise . jour 'ventuelle du prix final des "Value Entry ACP" suivantes
          CuACPManagement_G.ManageNextAcpEntry(TbValueEntryACP_G);
      END;
    END;

    PROCEDURE UpdateACPValueEntry@1000000010();
    BEGIN

      // Mise . jour Value Entry Acp
      //

      TbValueEntryACP_G.GET(TbValueEntry_G."Entry No.",ACPCode_G);

      IF ACPType_G = ACPType_G::Pourcentage THEN BEGIN
         // Mise . jour du montant unitaire . partir du % saisi
         TbValueEntryACP_G.VALIDATE("Acp %",ACPPct_G);
         TbValueEntryACP_G."Acp Amount" := 0;
         // --> 20200609-WI00-ADV01.01.00-3446-SCR01
         IF ACPPct_G > 0 THEN
         // --< 20200609-WI00-ADV01.01.00-3446-SCR01
         // --> 20200625-WI00-ADV01.01.00-3446-SCR01
         { BEFORE
           TbValueEntryACP_G.VALIDATE("Acp Amount",
                                    ROUND(TbValueEntryACP_G."Base Value" * ACPPct_G /100,
                                                                              UnitAmtRndPrecision_G,'='));
           AFTER }
           TbValueEntryACP_G.VALIDATE("Acp Amount",
                                    ROUND(TbValueEntryACP_G."Base Value" * ACPPct_G /100,0.01,'='));
          // <-- 20200625-WI00-ADV01.01.00-3446-SCR01
      END;

      IF ACPType_G = ACPType_G::"Montant unitaire" THEN BEGIN
         // Mise . jour du montant unitaire . partir de la valeur saisie
         // --> 20200609-WI00-ADV01.01.00-3446-SCR01
          //TbValueEntryACP_G.VALIDATE("Acp Amount",ACPValue_G);
         // <-- 20200609-WI00-ADV01.01.00-3446-SCR01
          TbValueEntryACP_G."Acp %" := 0;
          TbValueEntryACP_G.VALIDATE("Acp Amount",ACPValue_G);

      END;

      TbValueEntryACP_G."Final Price" := CuACPManagement_G.GetPreviousFinalPrice(TbValueEntryACP_G) -
                                                  TbValueEntryACP_G."Acp Amount";

      TbValueEntryACP_G.MODIFY(TRUE);
    END;

    PROCEDURE UpdateACPValueLedgerEntry@1000000006();
    BEGIN


      TbPurchACPLedgerEntry_G."ACP value" := TbValueEntryACP_G."Acp Amount";
      TbPurchACPLedgerEntry_G."ACP %" := TbValueEntryACP_G."Acp %";

      // --> 20200609-WI00-ADV01.01.00-3446-SCR01
      { BEFORE
      IF TbValueEntryACP_G."Acp %" <> 0 THEN
        TbPurchACPLedgerEntry_G."Valuation Type" := TbPurchACPLedgerEntry_G."Valuation Type"::Percentage
      ELSE
        TbPurchACPLedgerEntry_G."Valuation Type" := TbPurchACPLedgerEntry_G."Valuation Type"::"Unit Amount";
      AFTER }
      IF ACPType_G = ACPType_G::Pourcentage THEN
        TbPurchACPLedgerEntry_G."Valuation Type" := TbPurchACPLedgerEntry_G."Valuation Type"::Percentage
      ELSE
        TbPurchACPLedgerEntry_G."Valuation Type" := TbPurchACPLedgerEntry_G."Valuation Type"::"Unit Amount";
      // <-- 20200609-WI00-ADV01.01.00-3446-SCR01

      CASE TbPurchACPLedgerEntry_G."Document Type" OF
        // R'ceptiion, Retour
        TbPurchACPLedgerEntry_G."Document Type"::"Purchase Receipt",
        TbPurchACPLedgerEntry_G."Document Type"::"Purchase Return Shipment" : BEGIN

          TbPurchACPLedgerEntry_G."ACP Unit Amount (Receipt)" :=
                          ROUND(TbValueEntryACP_G."Acp Amount",
                          UnitAmtRndPrecision_G);

          TbPurchACPLedgerEntry_G."ACP Amount (Receipt)" :=
                          ROUND(TbPurchACPLedgerEntry_G."ACP Unit Amount (Receipt)" * TbValueEntry_G."Valued Quantity",
                          AmtRndPrecision_G);
        END;
        // Facture, Avoir
        TbPurchACPLedgerEntry_G."Document Type"::"Purchase Credit Memo",
        TbPurchACPLedgerEntry_G."Document Type"::"Purchase Invoice" : BEGIN

          TbPurchACPLedgerEntry_G."ACP Unit Amount (Invoice)" :=
                          ROUND(TbValueEntryACP_G."Acp Amount",
                          UnitAmtRndPrecision_G);

          TbPurchACPLedgerEntry_G."ACP Amount (Invoice)" :=
                          ROUND(TbPurchACPLedgerEntry_G."ACP Unit Amount (Invoice)" * TbValueEntry_G."Valued Quantity"
                          ,AmtRndPrecision_G);

          CuACPManagement_G.CalcRevPurchACPExpAmtToBalance(TbPurchACPLedgerEntry_G, TbValueEntry_G."Invoiced Quantity");

          TbPurchACPLedgerEntry_G.CALCFIELDS("Invoiced Acp Unit Price");
          // --> 20200609-WI00-ADV01.01.00-3446-SCR01
          TbPurchACPLedgerEntry_G."ACP Unit Price Outstanding" := 0;
          IF TbValueEntryACP_G."Acp Amount" > 0 THEN
          // <-- 20200609-WI00-ADV01.01.00-3446-SCR01
            TbPurchACPLedgerEntry_G."ACP Unit Price Outstanding" :=
                        ROUND(TbValueEntryACP_G."Acp Amount"-TbPurchACPLedgerEntry_G."Invoiced Acp Unit Price",UnitAmtRndPrecision_G);

        END;

      END;


      TbPurchACPLedgerEntry_G.MODIFY(TRUE);
    END;
