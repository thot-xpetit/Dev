    PROCEDURE RunCheck@1(VAR P_recInterventionJnlLine@1000 : Record 8053034;VAR P_recJnlLineDim@1001 : Record 356);
    VAR
      L_TableID@1002 : ARRAY [10] OF Integer;
      L_No@1003 : ARRAY [10] OF Code[20];
      "<-- 3LI LOCALS -->"@8051300 : Integer;
      IntervationType_L@8051301 : Record 8053033;
    BEGIN
      WITH P_recInterventionJnlLine DO BEGIN
        IF "Intervention Code" = '' THEN
          EXIT;
        TESTFIELD("Intervention Code");
        TESTFIELD("Location Code");
        TESTFIELD("Cell No.");
        IF (IntervationType_L.GET("Intervention Code")) AND IntervationType_L.Disinsectization THEN BEGIN
         TESTFIELD("Disinsectization Code");
         TESTFIELD("Disinsectization Dosage");
         TESTFIELD("Disinsectization Period");
        END;
        IF NOT G_DimMgt.CheckJnlLineDimComb(P_recJnlLineDim) THEN
          ERROR(Text000,TABLECAPTION,"Journal Template Name","Journal Batch Name","Line No.",G_DimMgt.GetDimCombErr);

        L_TableID[1] := DATABASE::"CMC Intervention Type";
        L_No[1] := "Intervention Code";
        IF NOT G_DimMgt.CheckJnlLineDimValuePosting(P_recJnlLineDim,L_TableID,L_No) THEN
          IF "Line No." <> 0 THEN
            ERROR(Text001,TABLECAPTION,"Journal Template Name","Journal Batch Name","Line No.",G_DimMgt.GetDimValuePostingErr)
          ELSE
            ERROR(G_DimMgt.GetDimValuePostingErr);
      END;
    END;

    BEGIN
    {

                                                       Disinsectization Period);

      // --> 3LI END
    }
