    PROCEDURE DeleteNonPointedLines@60002(VAR BankAccRec_P@60001 : Record 273);
    VAR
      BankAccReconLine_L@60000 : Record 274;
      Text60000_L@60002 : TextConst 'ENU=Do you want to delete the unapplied lines ?;FRA=Voulez-vous supprimer les lignes non point'es du rapprochement %1 ?';
    BEGIN
      WITH BankAccRec_P DO BEGIN

        IF NOT CONFIRM(Text60000_L,FALSE,"Statement No.") THEN
          EXIT;

        BankAccReconLine_L.RESET;
        BankAccReconLine_L.SETRANGE("Bank Account No.","Bank Account No.");
        BankAccReconLine_L.SETRANGE("Statement No.","Statement No.");
        BankAccReconLine_L.SETRANGE(Pointed,FALSE);
        IF BankAccReconLine_L.FINDFIRST THEN
          REPEAT
            BankAccReconLine_L.DELETE(TRUE);
          UNTIL BankAccReconLine_L.NEXT = 0;

      END;
    END;

    PROCEDURE UpdatePointedAmount@60003(VAR BankAccRecLine_P@60000 : Record 274);
    BEGIN
      WITH BankAccRecLine_P DO
        IF Pointed THEN
          "Pointed Amount" := "Statement Amount"
        ELSE
          "Pointed Amount" := 0;
    END;

    PROCEDURE UpdateBRDisplay@60004(VAR PtedTotalBalance_P@60000 : Decimal;BankAccRecon_P@60001 : Record 273;PointedAmnt_P@60002 : Decimal;TempBankAccReconLine_P@60003 : Record 274);
    BEGIN
      PtedTotalBalance_P := BankAccRecon_P."Balance Last Statement" - PointedAmnt_P;
      IF TempBankAccReconLine_P.CALCSUMS("Pointed Amount") THEN
        PtedTotalBalance_P := PtedTotalBalance_P + TempBankAccReconLine_P."Pointed Amount";
    END;

    PROCEDURE CopyBalanceAcc@60005(BalAccType_P@60002 : Integer;BalAccNo_P@60000 : Code[20];VAR GenJnlLine_P@60001 : Record 81);
    BEGIN
      IF BalAccNo_P <> '' THEN BEGIN
        GenJnlLine_P."Bal. Account Type" := BalAccType_P;
        GenJnlLine_P.VALIDATE("Bal. Account No.",BalAccNo_P);
      END;
    END;

    BEGIN
    {

      20101213-8060990/067-CMB6.00.01-FEC02 : ADDFUNCT (CopyBalanceAcc).
      // <-- 3LI END
    }
