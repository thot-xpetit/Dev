    PROCEDURE CheckVariationLines@1000000000(FileName_P@1000000000 : Text[250]);
    VAR
      OldPACCT_L@1000000001 : Text[50];
      NewEdgeSAS_L@1000000002 : Record 60111;
      PACCTWithoutVariation_L@1000000003 : Text[250];
      NewEdgeSASVariation_L@1000000004 : Record 60111;
    BEGIN
      // --> 20200429-WI00-ADV01.01.00-3480-MRU01
      // Pour chaque compte du fichier, v'rification si pr'sence ou non d'une ligne de variation
      OldPACCT_L := '';

      NewEdgeSAS_L.RESET;
      NewEdgeSAS_L.SETCURRENTKEY(PACCT,PFC,PCTYM);
      NewEdgeSAS_L.SETRANGE(FileName, FileName_P);
      IF NewEdgeSAS_L.FINDSET THEN REPEAT
        IF OldPACCT_L <> NewEdgeSAS_L.PACCT THEN BEGIN
          NewEdgeSASVariation_L.RESET;
          NewEdgeSASVariation_L.SETRANGE(FileName, FileName_P);
          NewEdgeSASVariation_L.SETRANGE(PACCT, NewEdgeSAS_L.PACCT);
          NewEdgeSASVariation_L.SETRANGE(PRECID, 'C');
          IF NOT NewEdgeSASVariation_L.FINDFIRST THEN BEGIN
            IF PACCTWithoutVariation_L = '' THEN
              PACCTWithoutVariation_L := COPYSTR(NewEdgeSAS_L.PACCT, STRLEN(NewEdgeSAS_L.PACCT) - 2)
            ELSE
              PACCTWithoutVariation_L += ' - ' + COPYSTR(NewEdgeSAS_L.PACCT, STRLEN(NewEdgeSAS_L.PACCT) - 2);
          END;

          OldPACCT_L := NewEdgeSAS_L.PACCT;
        END;
      UNTIL NewEdgeSAS_L.NEXT = 0;

      IF PACCTWithoutVariation_L <> '' THEN
        ToolsFile_G.LogWriteInfo(STRSUBSTNO(Text60001, PACCTWithoutVariation_L));
      // <-- 20200429-WI00-ADV01.01.00-3480-MRU01
    END;

      // <-- 3LI END
