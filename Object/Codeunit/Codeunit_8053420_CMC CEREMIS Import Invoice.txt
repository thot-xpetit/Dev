    PROCEDURE ImportInvoiceRecord@1000000006() Result_L : Text[256];
    VAR
      "<-- 3LI LOCALS -->"@1000000002 : Integer;
      TableSasImport_L@1000000001 : Record 8053407;
      TableSasImport2_L@1000000006 : Record 8053407;
      TraitOk_L@1000000000 : Boolean;
      FileLogName_L@1000000003 : Text[256];
      NoPrec_L@1000000004 : Code[10];
    BEGIN
      //----------------------------------------------------------------------------------------------------------------------------------
      // Importation des Factures dans Nav : copie du SAS vers Table ????
      //----------------------------------------------------------------------------------------------------------------------------------
      // Entr'es:
      //
      // Sorties:
      //   - Result_L    : Erreur Retourn'e, Vide si Ok
      //----------------------------------------------------------------------------------------------------------------------------------
      IF (NOT GetCeremisParameters()) THEN BEGIN
      //IF NOT TableParameter_L.FINDFIRST THEN BEGIN
        ToolsFile_G.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0001, TempCeremisParameter.TABLECAPTION ,COMPANYNAME),1,MAXSTRLEN(Result_L)));
      END ELSE BEGIN

        TableSasImport_L.SETRANGE(TableSasImport_L.Type, TableSasImport_L.Type::Invoice);
        TableSasImport_L.SETRANGE(TableSasImport_L."Integration Date", 0D);

        CLEAR(NoPrec_L);

        IF TableSasImport_L.FINDSET THEN BEGIN
          REPEAT
            // Test d'but rupture : N› Commande
            IF ((NoPrec_L <> TableSasImport_L.No)) THEN BEGIN
              NoPrec_L := TableSasImport_L.No;

              // Pour traiter avec COMMIT et ROLLBACK par Exp'dition
              IF NOT CODEUNIT.RUN(TempCeremisParameter."Invoice CodeUnit 02", TableSasImport_L) THEN BEGIN
                ToolsFile_G.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0021, NoPrec_L + ':' +
                  FORMAT(TableSasImport_L.Type)  + '-' + TableSasImport_L.No + '-' + TableSasImport_L."Sub-No"),
                  1, MAXSTRLEN(Result_L)));
                ToolsFile_G.LogWriteErreur(COPYSTR(GETLASTERRORTEXT, 1, MAXSTRLEN(Result_L)));
              END ELSE BEGIN
                COMMIT;
                IF TableSasImport2_L.GET(TableSasImport_L.Type, TableSasImport_L.No, TableSasImport_L."Sub-No") THEN BEGIN
                END;
                ToolsFile_G.LogWriteInfo(COPYSTR(STRSUBSTNO(Text0022, NoPrec_L, TableSasImport2_L."NAV No"), 1, MAXSTRLEN(Result_L)));
              END;
            END;
          UNTIL TableSasImport_L.NEXT = 0;
        END;
      END;

      // Pour g'rer le ROLLBACK.
      CLEAR(Result_L);
      FileLogName_L := ToolsFile_G.LogName;
      ToolsFile_G.LogClose();

      IF ToolsFile_G.LogErreurType() = 'E' THEN BEGIN
        Result_L := 'ERREUR ' + FileLogName_L;
        ERROR(Result_L);
      END ELSE BEGIN
        IF ToolsFile_G.LogErreurType() = 'I' THEN BEGIN
          Result_L := Text0008;
          MoteurInterface_G.InsertLog(CodeUnit_G, FileLogName_L);
        END ELSE BEGIN
          MoteurInterface_G.InsertOk(CodeUnit_G);
        END;
      END;
      //<--20140401-Int07-CMC02.02.00-ADVITAM-CDL
    END;

    PROCEDURE GetCeremisParameters@1000000002() ReturnValue : Boolean;

      CeremisSetup@1000000000 : Record 8053406;
    BEGIN

      TempCeremisParameter.DELETEALL();

      END;
    END;
