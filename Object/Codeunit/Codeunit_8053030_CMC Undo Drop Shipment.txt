    LOCAL PROCEDURE Code@2();

      ItemLedgerEntryPurch_L@1000000003 : Record 32;
      ItemApplicationEntry@1000000004 : Record 339;
    BEGIN
      WITH SalesShptLine_G DO BEGIN
        CLEAR(ItemJnlPostLine_G);
        SETCURRENTKEY("Item Shpt. Entry No.");
        SETRANGE(Correction,FALSE);

        REPEAT
          IF NOT HideDialog_G THEN
            Window_L.OPEN(Text003);
          CheckSalesShptLine(SalesShptLine_G);
        UNTIL NEXT = 0;

        ServItem_L.SETCURRENTKEY("Sales/Serv. Shpt. Document No.");
        ServItem_L.SETRANGE("Sales/Serv. Shpt. Document No.","Document No.");
        IF ServItem_L.FIND('-') THEN
          IF NOT HideDialog_G THEN
            DeleteServItems_L := CONFIRM(Text004,TRUE)
          ELSE
            DeleteServItems_L := TRUE;

        FIND('-');
        REPEAT
          TempGlobalItemLedgEntry_G.RESET;
          IF NOT TempGlobalItemLedgEntry_G.ISEMPTY THEN
            TempGlobalItemLedgEntry_G.DELETEALL;

          IF NOT HideDialog_G THEN
            Window_L.OPEN(Text001);
          // --> 20160330-WI00-ADV01.01.00-2887-CJY01
          //D'l'trage
          ItemLedgerEntrySales_L.GET(SalesShptLine_G."Item Shpt. Entry No.");
          //on va chercher l''criture achat lettr' avec notre 'criture vente
          ItemApplicationEntry.RESET;
          ItemApplicationEntry.SETRANGE("Outbound Item Entry No.",ItemLedgerEntrySales_L."Entry No.");
          IF ItemApplicationEntry.FINDSET THEN
            ItemLedgerEntryPurch_L.GET(ItemApplicationEntry."Inbound Item Entry No.");

          IF ItemLedgerEntrySales_L.Positive THEN
            RemoveApplications(ItemLedgerEntrySales_L."Entry No.",ItemLedgerEntryPurch_L."Entry No.")
          ELSE
            RemoveApplications(ItemLedgerEntryPurch_L."Entry No.",ItemLedgerEntrySales_L."Entry No.");
          // <-- 20160330-WI00-ADV01.01.00-2887-CJY01
          ItemShptEntryNo_L := PostSaleItemJnlLine(SalesShptLine_G,DocLineNo_L);
          InsertNewShipmentLine(SalesShptLine_G,ItemShptEntryNo_L,DocLineNo_L);
          UpdateSalesOrderLine(SalesShptLine_G);

          SalesShipLine_L.RESET;
          SalesShipLine_L.SETCURRENTKEY("Document No.");
          SalesShipLine_L.SETRANGE("Document No.",SalesShptLine_G."Document No.");
          SalesShipLine_L.SETRANGE("CVO Item Attached Line",SalesShptLine_G."Line No.");
          IF SalesShipLine_L.FINDFIRST THEN
            InsertNewShipmentLineForGLAcc(SalesShipLine_L);

          SalesShipLine_L.RESET;
          SalesShipLine_L.SETCURRENTKEY("Document No.");
          SalesShipLine_L.SETRANGE("Document No.",SalesShptLine_G."Document No.");
          SalesShipLine_L.SETRANGE("CVO Corn Item Attached Line",SalesShptLine_G."Line No.");
          IF SalesShipLine_L.FINDFIRST THEN
            InsertNewShipmentLineForGLAcc(SalesShipLine_L);

          SalesShipLine_L.RESET;
          SalesShipLine_L.SETCURRENTKEY("Document No.");
          SalesShipLine_L.SETRANGE("Document No.",SalesShptLine_G."Document No.");
          SalesShipLine_L.SETRANGE("RPD Item Attached Line",SalesShptLine_G."Line No.");
          IF SalesShipLine_L.FINDFIRST THEN
            InsertNewShipmentLineForGLAcc(SalesShipLine_L);

          IF ("Blanket Order No." <> '') AND ("Blanket Order Line No." <> 0) THEN
            UpdateSalesBlanketOrder(SalesShptLine_G);

          IF DeleteServItems_L THEN
            DeleteSalesShptLineServItems(SalesShptLine_G);

          "Quantity Invoiced" := Quantity;
          "Qty. Invoiced (Base)" := "Quantity (Base)";
          "Qty. Shipped Not Invoiced" := 0;
          Correction := TRUE;
          MODIFY;

          // Traitement de la ligne d'achat
          // --> 20160330-WI00-ADV01.01.00-2887-CJY01
          {BEFORE
          GetPurchReceiptLine(SalesShptLine_G,PurchRcptLine_L);
          AFTER}
          //on r'cupSre la ligne d'achat
          PurchRcptLine_L.GET(ItemLedgerEntryPurch_L."Document No.",
                              ItemLedgerEntryPurch_L."Document Line No.");
          // <-- 20160330-WI00-ADV01.01.00-2887-CJY01

          CheckPurchRcptLine(PurchRcptLine_L);
          // --> 20160330-WI00-ADV01.01.00-2887-CJY01
          {BEFORE
          ItemRcptEntryNo_L := PostPurchItemJnlLine(PurchRcptLine_L,DocLineNo_L,ItemShptEntryNo_L);
          AFTER}
          ItemRcptEntryNo_L := PostPurchItemJnlLine(PurchRcptLine_L,DocLineNo_L,PurchRcptLine_L."Item Rcpt. Entry No.");
          // <-- 20160330-WI00-ADV01.01.00-2887-CJY01
          InsertNewReceiptLine(PurchRcptLine_L,ItemRcptEntryNo_L,DocLineNo_L);
          UpdatePurchOrderLine(PurchRcptLine_L);

          IF (PurchRcptLine_L."Blanket Order No." <> '') AND (PurchRcptLine_L."Blanket Order Line No." <> 0) THEN
            UpdatePurchBlanketOrder(PurchRcptLine_L);

          PurchRcptLine_L."Quantity Invoiced" := Quantity;
          PurchRcptLine_L."Qty. Invoiced (Base)" := "Quantity (Base)";
          PurchRcptLine_L."Qty. Rcd. Not Invoiced" := 0;
          PurchRcptLine_L.Correction := TRUE;
          PurchRcptLine_L.MODIFY;

          // --> 20200604-WI00-ADV01.01.00-ID50146-MRU01
          //Annulation de la ligne de CRIV si elle existe (Import EXELIENCE - SP3375)
          PurchRcptLine_L.RESET;
          PurchRcptLine_L.SETCURRENTKEY("Document No.");
          PurchRcptLine_L.SETRANGE("Document No.",ItemLedgerEntryPurch_L."Document No.");
          PurchRcptLine_L.SETRANGE("CVO Item Attached Line",ItemLedgerEntryPurch_L."Document Line No.");
          IF PurchRcptLine_L.FINDFIRST THEN
            InsertNewRcptLineForGLAcc(PurchRcptLine_L);
          // <-- 20200604-WI00-ADV01.01.00-ID50146-MRU01
        UNTIL NEXT = 0;

        InvtSetup_G.GET;
        IF InvtSetup_G."Automatic Cost Adjustment" <> InvtSetup_G."Automatic Cost Adjustment"::Never THEN BEGIN
          InvtAdjmt_G.SetProperties(TRUE,InvtSetup_G."Automatic Cost Posting");
          InvtAdjmt_G.MakeMultiLevelAdjmt;
        END;
      END;
    END;
