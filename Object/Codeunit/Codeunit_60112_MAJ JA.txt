      ToolsFile_L.LogInit(LogFolder_L, 'MAJ_JA');
      ToolsFile_L.LogInitCodeUnit(CodeUnit_G);
      IF (NOT TableParam_L.FINDFIRST) THEN BEGIN
        ToolsFile_L.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0001, TableParam_L.TABLECAPTION, COMPANYNAME), 1, MAXSTRLEN(Result_L)));
      END;
      //le cas d'erreur "manque filtre campagne" n'est . faire que si on est pas en param'trage RAZ
      IF (TableParam_L."Filtre campagne" = '') AND (TableParam_L."RAZ JA existant" = FALSE) THEN

        Contact_L.RESET;
        Contact_L.SETRANGE("Farmer List 2",'JA');
        IF NOT Contact_L.ISEMPTY THEN
          Contact_L.MODIFYALL("Farmer List 2",'');
      END
      ELSE BEGIN
        IF ToolsFile_L.LogErreurType() <> 'E' THEN BEGIN
          //on balaie les contrat JA avec le filtre campagne
          ContratCCA_L.RESET;
          ContratCCA_L.SETRANGE("Frame Contract Type",'JA');
          ContratCCA_L.SETFILTER("Campaign No.",TableParam_L."Filtre campagne");
          //MESSAGE('nombre %1',ContratCCA_L.COUNT);
          IF ContratCCA_L.FINDSET THEN REPEAT
            IF Contact_L.GET(ContratCCA_L."Contact No.") THEN BEGIN
              Contact_L."Farmer List 2" := 'JA';
              Contact_L.MODIFY;
            END;
          UNTIL ContratCCA_L.NEXT = 0;
        END;
      END;
      CLEAR(Result_L);
      FileLogName_L := ToolsFile_L.LogName;
      //MESSAGE('%1',FileLogName_L);
      ToolsFile_L.LogClose();
      IF ToolsFile_L.LogErreurType() = 'E' THEN BEGIN
        Result_L := 'ERREUR ' + FileLogName_L;
        ERROR(Result_L);
      END ELSE BEGIN
        IF ToolsFile_L.LogErreurType() = 'I' THEN BEGIN
          Result_L := Text0005;
          //-->20131209-Int13-CMC02.02.00-ADVITAM-CDL
          //BEFORE
          {
          MoteurInterface_L.InsertLog(CodeUnit_L, FileLogName_L);
          }
          //AFTER
          MoteurInterface_L.InsertLog(CodeUnit_G, FileLogName_L);
          //<--20131209-Int13-CMC02.02.00-ADVITAM-CDL
        END ELSE BEGIN
          //-->20131209-Int13-CMC02.02.00-ADVITAM-CDL
          //BEFORE
          {
          MoteurInterface_L.InsertOk(CodeUnit_L);
          }
          //AFTER
          MoteurInterface_L.InsertOk(CodeUnit_G);
          //<--20131209-Int13-CMC02.02.00-ADVITAM-CDL
