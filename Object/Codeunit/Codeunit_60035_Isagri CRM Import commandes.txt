    PROCEDURE ReadNextItem@8053600(VAR Var_Line_P@8053600 : Text[1000]) Result_L : Text[50];
    VAR
      Position_L@8053601 : Integer;
    BEGIN
      //----------------------------------------------------------------------------------------------------------------------------------
      // R'cupSre le prochain 'l'ment de la liste 'tant s'par' par des ; et l'enlSve de la liste
      //----------------------------------------------------------------------------------------------------------------------------------
      //  Entr'es
      //    - Var_Line_P : Ligne . traiter
      //  Sorties
      //    - Result_L : la prochain 'l'ment de la liste
      //----------------------------------------------------------------------------------------------------------------------------------

      Position_L := STRPOS(Var_Line_P, ';');
      IF Position_L > 0 THEN BEGIN
        Result_L := COPYSTR(Var_Line_P, 1, Position_L - 1);
        Var_Line_P := COPYSTR(Var_Line_P, Position_L + 1, STRLEN(Var_Line_P));
      END ELSE BEGIN
      //Dernier 'l'ment de la liste
        Result_L := Var_Line_P;
        Var_Line_P := '';
      END;
      //MESSAGE('-%1-',Result_L);
      EXIT(Result_L);
    END;
