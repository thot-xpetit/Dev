    PROCEDURE CheckStatCap@8053011(FarmerNo_P@8053001 : Code[20];CampaignNo_P@8053000 : Code[20]) : Boolean;
    VAR
      CurrCampaign_L@8053014 : Record 8052991;
      TempCampaign_L@8053013 : TEMPORARY Record 8052991;
      NbCampaign_L@8053011 : Integer;
      CapitalRates_L@8053010 : Record 8053363;
      I_L@8053009 : Integer;
      PartRec_L@8053008 : Record 8053351;
      AnalysisEntry_L@8053007 : Record 8053362;
      Contact_L@8053006 : Record 5050;
      AnalysisReportLine_L@8053005 : Record 8053361;
      Campaign_L@8053004 : Record 8052991;
      Average_L@8053003 : Decimal;
      CapitalEntry_L@8053002 : Record 8053357;
    BEGIN
      // *** Calcul capital statutaire***//
      CLEAR(StatCap_G);
      CurrCampaign_L.GET(CampaignNo_P);
      CurrCampaign_L.TESTFIELD("Statutory Capital Table Code");
      CurrCampaign_L.TESTFIELD("Campaigns Number");
      TempCampaign_L.DELETEALL;
      I_L := CurrCampaign_L."Campaigns Number";
      REPEAT
        TempCampaign_L.INIT;
        TempCampaign_L.TRANSFERFIELDS(CurrCampaign_L);
        TempCampaign_L.INSERT;
        I_L -= 1;
      UNTIL (CurrCampaign_L.NEXT(-1) = 0) OR (I_L = 0) ;

      Contact_L.GET(FarmerNo_P);
      Campaign_L.GET(CampaignNo_P);
      AnalysisReportLine_L.RESET;
      AnalysisReportLine_L.SETFILTER(AnalysisReportLine_L."Contact Filter", FarmerNo_P);
      AnalysisReportLine_L.SETRANGE(Code, Campaign_L."Statutory Capital Table Code");
      AnalysisReportLine_L.SETRANGE("Line Type", AnalysisReportLine_L."Line Type"::"Funnel Filter");
      IF AnalysisReportLine_L.FINDSET THEN
        REPEAT
          CLEAR(Average_L);
          IF TempCampaign_L.FINDFIRST THEN
            REPEAT
              AnalysisEntry_L.RESET;
              AnalysisEntry_L.SETCURRENTKEY(Code, Line, Date, "Contact No.", "Entry No.");
              AnalysisEntry_L.SETRANGE(Code, Campaign_L."Statutory Capital Table Code");
              AnalysisEntry_L.SETRANGE(Line, AnalysisReportLine_L.Line);
              AnalysisEntry_L.SETFILTER(AnalysisEntry_L."Contact No.", Contact_L."No.");
              AnalysisEntry_L.SETRANGE(Date, TempCampaign_L."Starting Date", TempCampaign_L."Ending Date");
              AnalysisEntry_L.CALCSUMS(Amount);
              CapitalRates_G.GET(CampaignNo_P, Campaign_L."Statutory Capital Table Code", AnalysisReportLine_L.Line) ;
              CapitalRates_G.TESTFIELD(Rate);
              CapitalRates_G.Rate := CapitalRates_G.Rate / 100;
              Average_L += AnalysisEntry_L.Amount;
            UNTIL TempCampaign_L.NEXT=0;
            Average_L := Average_L / Campaign_L."Campaigns Number";
            StatCap_G += Average_L * CapitalRates_G.Rate;
        UNTIL AnalysisReportLine_L.NEXT = 0;
        StatCap_G := ROUND(StatCap_G, GLSetup_G."Amount Rounding Precision");
      EXIT(StatCap_G = DistrJnlLine_G."Statutory Capital");
    END;

    PROCEDURE CheckPartToRestraint@8053012(FarmerNo_P@8053002 : Code[20]) : Boolean;
    VAR
      DtlPart_L@8053000 : Record 8053352;
      PartRec_L@8053001 : Record 8053351;
      Dividend_L@8053003 : Decimal;
      Interest_L@8053004 : Decimal;
      PltSocial_L@8053005 : Decimal;
      Contact_L@8053006 : Record 5050;
      Restraint_L@8053007 : Decimal;
      Ret_L@8053008 : Decimal;
      CapitalSetup_L@8053009 : Record 8053350;
      DtlPartRec_L@8053010 : Record 8053352;
    BEGIN
      Contact_L.GET(FarmerNo_P);
      PartRec_L.RESET;
      PartRec_L.SETFILTER(PartRec_L."Member Filter", FarmerNo_P);
      PartRec_L.SETFILTER("Date Filter", '..%1', DistrJnlLine_G."Posting Date");
      PartRec_L.CALCFIELDS(PartRec_L."Part Type Count");
      PartRec_L.SETFILTER(PartRec_L."Part Type Count", '<>%1', 0);
      CLEAR(Dividend_L);
      CLEAR(Interest_L);
      CLEAR(PltSocial_L);
      IF PartRec_L.FINDSET THEN
        REPEAT
          DtlPart_L.RESET;
          DtlPart_L.ASCENDING(FALSE);
          DtlPart_L.SETFILTER("Effective Date", '..%1', DistrJnlLine_G."Closing Date");
          DtlPart_L.SETRANGE("Part Code", PartRec_L."Part Code");
          IF DtlPart_L.FINDFIRST THEN BEGIN
            PartRec_L.CALCFIELDS("Part Type Count", PartRec_L."Part Type Amount");
            // *** Calcul dividende ***//
            DtlPart_L.TESTFIELD("Dividend Amount");
            Dividend_L += PartRec_L."Part Type Count" * DtlPart_L."Dividend Amount";
            // *** Calcul int'r^t ***//
            IF Contact_L."Partner Type" = Contact_L."Partner Type"::Cooperator THEN
              Interest_L += PartRec_L."Part Type Amount" * (DtlPart_L."Interest Rate for Coop. Member" / 100)
            ELSE
              IF Contact_L."Partner Type" = Contact_L."Partner Type"::"Not Cooperator" THEN
                Interest_L += PartRec_L."Part Type Amount" * (DtlPart_L."Interest Rate for NotCoop. Mbr" / 100);
            // *** Calcul pr'lSvement social *** //
            IF (Contact_L."Moral Or Natural Person" = Contact_L."Moral Or Natural Person"::Natural) THEN BEGIN
              IF DtlPart_L."Apply Dividend Social Prlt." THEN
                PltSocial_L += (PartRec_L."Part Type Count" * DtlPart_L."Dividend Amount") * (DtlPart_L."Social Retrievement" / 100);
              IF (Contact_L."Partner Type" = Contact_L."Partner Type"::"Not Cooperator") AND
                (DtlPart_L."Apply N Coop Interset S. Prlt")
              THEN
                PltSocial_L += (PartRec_L."Part Type Amount" * (DtlPart_L."Interest Rate for NotCoop. Mbr" / 100)) *
                  (DtlPart_L."Social Retrievement" / 100);
              IF (Contact_L."Partner Type" = Contact_L."Partner Type"::Cooperator) AND
                 (DtlPart_L."Apply Coop Interset S. Prlt")
              THEN
                PltSocial_L += (PartRec_L."Part Type Amount" * (DtlPart_L."Interest Rate for Coop. Member" / 100)) *
                  (DtlPart_L."Social Retrievement" / 100);
             END;
          END;
        UNTIL PartRec_L.NEXT=0;

      Dividend_L := ROUND(Dividend_L, GLSetup_G."Amount Rounding Precision");
      Interest_L := ROUND(Interest_L, GLSetup_G."Amount Rounding Precision");
      PltSocial_L := ROUND(PltSocial_L, GLSetup_G."Amount Rounding Precision");

      // ** Calcul capital r'el . ce jour **//
      RealCap_G := CalcRealCap(Contact_L);

      // *** Calcul Retenue souscription ***//
      IF StatCap_G>RealCap_G THEN BEGIN
        IF (NOT Contact_L.IsYoungFarmerCtt(DistrJnlLine_G."Campaign No.")) THEN BEGIN
          Restraint_L := Dividend_L + Interest_L - PltSocial_L + DistrJnlLine_G.Discount;
          IF Restraint_L > (StatCap_G-RealCap_G) THEN
            Restraint_L := StatCap_G-RealCap_G;
        END ELSE BEGIN
          IF Contact_L."Moral Or Natural Person" = Contact_L."Moral Or Natural Person"::Moral THEN
            Restraint_L := ROUND((StatCap_G-RealCap_G) * ((100 - Contact_L.GetYoungFarmerCttPart(DistrJnlLine_G."Campaign No.")) / 100),
              GLSetup_G."Amount Rounding Precision");
          Ret_L := Dividend_L + Interest_L - PltSocial_L + DistrJnlLine_G.Discount;
          IF Restraint_L > Ret_L THEN
            Restraint_L := Ret_L;
        END;
      END;
      CapitalSetup_L.GET;
      CapitalSetup_L.TESTFIELD("Restraint Part Type");
      DtlPartRec_L.RESET;
      DtlPartRec_L.ASCENDING(FALSE);
      DtlPartRec_L.SETFILTER("Effective Date", '..%1', DistrJnlLine_G."Posting Date");
      DtlPartRec_L.SETRANGE("Part Code", CapitalSetup_L."Restraint Part Type");
      IF DtlPartRec_L.FINDFIRST THEN
        Restraint_L := Restraint_L-(Restraint_L MOD  DtlPartRec_L."Unit Value");
      EXIT(Restraint_L = DistrJnlLine_G."Sousc. Ret");
    END;

    PROCEDURE CheckDocumentNo@11(VAR CapJnlLine2_G@8053000 : Record 8053356);

      WITH CapJnlLine2_G DO BEGIN

              "Document No." := LastPostedDocNo_G
            ELSE BEGIN
              IF NOT NoSeries_G.GET("Posting No. Series") THEN BEGIN
                NoOfPostingNoSeries_G := NoOfPostingNoSeries_G + 1;
                IF NoOfPostingNoSeries_G > ARRAYLEN(NoSeriesMgt2) THEN
                  ERROR(Text8053025, ARRAYLEN(NoSeriesMgt2));
                NoSeries_G.Code := "Posting No. Series";
                NoSeries_G.Description := FORMAT(NoOfPostingNoSeries_G);
                NoSeries_G.INSERT;
              END;
              LastDocNo_G := "Document No.";
              EVALUATE(PostingNoSeriesNo_G, NoSeries_G.Description);
              "Document No." := NoSeriesMgt2[PostingNoSeriesNo_G].GetNextNo("Posting No. Series", "Posting Date",TRUE);
              LastPostedDocNo_G := "Document No.";
            END;
      END;
    END;
