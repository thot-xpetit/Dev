OBJECT Codeunit 60108 Interface MAJ Engagement
{
  OBJECT-PROPERTIES
  {
    Date=17/02/22;
    Heure=00:00:00;
    Liste versions=;
  }
  PROPERTIES
  {
    TableNo=8061230;
    OnRun=VAR
            MajListVendor_L@1000000009 : Codeunit 60109;
            CodeUnit_L@1000000008 : Integer;
            ToolsFile_G@1000000007 : Codeunit 8053405;
            MoteurInterface_L@1000000006 : Codeunit 8061230;
            ConnectorName_L@1000000005 : Code[20];
            ImpExpFolder_L@1000000004 : Text[250];
            LogFolder_L@1000000003 : Text[250];
            FileLogName_L@1000000002 : Text[250];
            ExtensionFile_L@1000000001 : Text[10];
            PrefixFile_L@1000000000 : Text[10];
            CmCCampain_L@1000000011 : Record 8052991;
          BEGIN
            CodeUnit_L := Processing;
            MoteurInterface_L.GetParameter(CodeUnit_L, ConnectorName_L, ImpExpFolder_L, LogFolder_L, ExtensionFile_L, PrefixFile_L);
            ToolsFile_G.LogInit(LogFolder_L, 'NAVAGRI');
            ToolsFile_G.LogInitCodeUnit(CodeUnit_L);

            Compteur_G := 0;

            CmCCampain_L.RESET;
            CmCCampain_L.SETCURRENTKEY("Type Code","Starting Date","Ending Date");
            CmCCampain_L.SETFILTER("Starting Date",'<=%1',WORKDATE);
            CmCCampain_L.SETFILTER("Ending Date",'>=%1',WORKDATE);
            IF CmCCampain_L.FINDLAST THEN BEGIN
              Campain_G := CmCCampain_L."No.";
              MajListVendor_L.SetCampagne(Campain_G);
            END
            ELSE
              ERROR(Text60000,WORKDATE);

            //on balaie tous les fournisseurs
            Vendor_G.RESET;

            NbFournisseur_G := Vendor_G.COUNT;
            IF Vendor_G.FINDSET THEN
              BEGIN
                IF GUIALLOWED THEN
                  Diag_G.OPEN(Text60001 + Text60002, Compteur_G, NbFournisseur_G);

                REPEAT
                  Compteur_G := Compteur_G + 1;
                  IF NOT MajListVendor_L.RUN(Vendor_G) THEN BEGIN
                    IF GUIALLOWED THEN
                      Diag_G.CLOSE;
                    ToolsFile_G.LogWriteInfo(COPYSTR(STRSUBSTNO(GETLASTERRORTEXT),1,1024));
                    COMMIT;
                  END
                  ELSE BEGIN
                    IF GUIALLOWED THEN
                      Diag_G.UPDATE();
                    COMMIT;

                  END;

                UNTIL Vendor_G.NEXT = 0;
                IF GUIALLOWED THEN
                  Diag_G.CLOSE;
              END;

            FileLogName_L := ToolsFile_G.LogName;
            ToolsFile_G.LogClose();

            // Pour g'rer le ROLLBACK.
            IF ToolsFile_G.LogErreurType() = 'E' THEN BEGIN
              ERROR('ERREUR ' + FileLogName_L);
            END ELSE BEGIN
              IF ToolsFile_G.LogErreurType() = 'I' THEN BEGIN
                MoteurInterface_L.InsertLog(CodeUnit_L, FileLogName_L);
            MESSAGE('I');
              END ELSE BEGIN
            MESSAGE('OK');
                MoteurInterface_L.InsertOk(CodeUnit_L);
              END;
            END;
          END;

  }
  CODE
  {
    VAR
      Diag_G@1000000003 : Dialog;
      Compteur_G@1000000002 : Integer;
      NbFournisseur_G@1000000001 : Integer;
      Campain_G@1000000000 : Code[10];
      Text60000@1000000004 : TextConst 'FRA=Impossible de d'terminer la campagne . prendre en compte . partir de la date de travail %1';
      Vendor_G@1000000005 : Record 23;
      Text60001@1000000006 : TextConst 'FRA=Fournisseur  #1######';
      Text60002@1000000007 : TextConst 'FRA=Sur un total de #2######';

    BEGIN
    {
      // --> 3LI BEGIN
      20220216-WI00-ADV01.01.00-3584-CJY01 : ADDOBJECT
      // <-- 3LI END
    }
    END.
  }
}
