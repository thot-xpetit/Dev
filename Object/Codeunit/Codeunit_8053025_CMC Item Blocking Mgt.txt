    PROCEDURE BlockAllFlows@8053000(ItemBlockingSetup_P@8053000 : Record 8053230;Unblock_P@8053026 : Boolean);
    BEGIN
      //------------------------------------------------------------------------------------------------------------------------------
      // The function blocks all possible flows for the current line.
      // Input:
      //   - ItemBlockingSetup - holds the item blocking parametrization
      //-------------------------------------------------------------------------------------------------------------------------------

      WITH ItemBlockingSetup_P DO BEGIN
        CASE "Blocking Type" OF

          "Blocking Type"::Location:
            BlockAllFlowsForLocation(ItemBlockingSetup_P,Unblock_P);

          "Blocking Type"::"Location Type":
            BlockAllFlowsForLocation(ItemBlockingSetup_P,Unblock_P);

          "Blocking Type"::Salesperson:
            BlockAllFlowsForSalesPerson(ItemBlockingSetup_P,Unblock_P);

          "Blocking Type"::"Responsibility Center":
            BlockAllFlowsForRespCenter(ItemBlockingSetup_P,Unblock_P);

          "Blocking Type"::Customer:
            BlockAllFlowsForCustomer(ItemBlockingSetup_P,Unblock_P);

          "Blocking Type"::Vendor:
            BlockAllFlowsForVendor(ItemBlockingSetup_P,Unblock_P);
        END;
      END;
    END;
