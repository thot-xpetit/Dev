OBJECT Codeunit 60059 Sales-Post DS + Print (Yes/No)
{
  OBJECT-PROPERTIES
  {
    Date=07/08/17;
    Heure=00:00:00;
    Liste versions=ADV01.01.00;
  }
  PROPERTIES
  {
    TableNo=36;
    OnRun=BEGIN
            SalesHeader.COPY(Rec);
            Code;
            Rec := SalesHeader;
          END;

  }
  CODE
  {
    VAR
      Text001@1001 : TextConst 'ENU=Do you want to post and print the %1?;FRA=Souhaitez-vous valider et imprimer le document %1 ?';
      SalesHeader@1003 : Record 36;
      SalesPost@1004 : Codeunit 80;
      SalesShptHeader@1000000001 : Record 110;
      ReportSelection@1000000000 : Record 77;

    LOCAL PROCEDURE Code@1();
    BEGIN
      WITH SalesHeader DO BEGIN

        TESTFIELD("Document Type","Document Type"::Order);

        IF NOT CONFIRM(Text001,FALSE,"Document Type") THEN
          EXIT;

        Ship := TRUE;
        Invoice := FALSE;
        "Post Drop Shipment Only" := TRUE;

        SalesPost.RUN(SalesHeader);

        GetReport(SalesHeader);

        COMMIT;

      END;
    END;

    PROCEDURE GetReport@3(VAR SalesHeader@1000 : Record 36);
    BEGIN
      WITH SalesHeader DO BEGIN
          TESTFIELD("Document Type","Document Type"::Order);
          TESTFIELD(Ship,TRUE);
          TESTFIELD(Invoice,FALSE);
          SalesShptHeader."No." := "Last Shipping No.";
          SalesShptHeader.SETRECFILTER;
          PrintReport(ReportSelection.Usage::"S.Shipment");
      END;
    END;

    LOCAL PROCEDURE PrintReport@1000000000(ReportUsage@1000 : Integer);
    BEGIN
      ReportSelection.RESET;
      ReportSelection.SETRANGE(Usage,ReportUsage);
      ReportSelection.FINDSET;
      REPEAT
        ReportSelection.TESTFIELD("Report ID");
        CASE ReportUsage OF
          ReportSelection.Usage::"S.Shipment": BEGIN
            REPORT.RUN(ReportSelection."Report ID",FALSE,FALSE,SalesShptHeader);
          END;
        END;
      UNTIL ReportSelection.NEXT = 0;
    END;

    BEGIN
    {
      // --> 3LI BEGIN
      20170807-Wi00-ADV01.01.00-2409-CDX01 : NEWOBJ
      // <-- 3LI END
    }
    END.
  }
}
