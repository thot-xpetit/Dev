    PROCEDURE SearchNormalPrice@1000000003() : Decimal;
    VAR
      SalesHeaderEphemeral_L@1000000000 : Record 36;
      SalesLineEphemeral_L@1000000001 : Record 37;
      ItemNo_L@1000000005 : Code[20];
      VariantNo_L@1000000004 : Code[10];

      SalesLineEphemeralImpacted_L@1000000006 : Record 37;
      SalesHeaderEphemeralToDel_L@1000000007 : Record 36;
      Sessions_L@1000000008 : Record 2000000009;
    BEGIN
      //fonction charg'e de simuler la saisie manuelle d'une ligne afin de r'cup'rer le tarif qui descend naturellement
      //en cours d''laboration

      //pr'paration couple article/variante
      FormatItemVariant(ItemVariantNo_G,ItemNo_L,VariantNo_L);
      SalesHeaderEphemeral_L.SetHideValidationDialog(TRUE);
      SalesHeaderEphemeral_L.SetHideCollectifChoice(TRUE);
      SalesHeaderEphemeral_L.VALIDATE("Document Type", SalesHeaderEphemeral_L."Document Type"::Order);
      //si on veut pouvoir utiliser plusieurs instance en simultan'e du WS, il ne faut pas un num'ro de document identique
      Sessions_L.SETRANGE("Ma session",TRUE);
      Sessions_L.FINDFIRST;
      SalesHeaderEphemeral_L."No." := 'WS Price' + ' ' + FORMAT(Sessions_L."ID connexion");
      SalesHeaderEphemeral_L.INSERT(TRUE);
      SalesHeaderEphemeral_L.VALIDATE("Document Date",WORKDATE);
      //SalesHeaderEphemeral_L.VALIDATE("Order Date", WORKDATE);
      SalesHeaderEphemeral_L.VALIDATE("Posting Date", SalesHeaderEphemeral_L."Order Date");




      //SalesHeaderEphemeral_L.VALIDATE("Posting Date", RecSASHeader_L."Order Date");
      SalesHeaderEphemeral_L.VALIDATE("Sell-to Customer No.", SellToCustNo_G);
      //SalesHeaderEphemeral_L.VALIDATE("Order Date", RecSASHeader_L."Date Commande");
      //les 'l'ment suivant sont apparu n'cessaire au lancement
      //date livraison demand'

      //on doit basculer de record en record pour 'viter les soucis d'enregistrement mis . jour et ne pas utiliser de COMMIT
      SalesHeaderEphemeral_L.VALIDATE("Requested Delivery Date",WORKDATE);

      //SalesHeaderEphemeral_L.VALIDATE("Shipment Method Code",'LIV');
      SalesHeaderEphemeral_L.VALIDATE("Shipment Method Code",ShipMethodCode_G);
      // --> 20210610-WI00-ADV01.01.00-3545-CJY01
      SalesHeaderEphemeral_L.VALIDATE("CMC Campaign No.",CampainNo_G);
      // <-- 20210610-WI00-ADV01.01.00-3545-CJY01

      SalesHeaderEphemeral_L.MODIFY(TRUE);
      SalesLineEphemeral_L.SetHideValidationDialog(TRUE);
      SalesLineEphemeral_L.SuspendQuotasMessage(TRUE);
      SalesLineEphemeral_L.VALIDATE("Document Type", SalesLineEphemeral_L."Document Type"::Order);
      SalesLineEphemeral_L.VALIDATE("Document No.", SalesHeaderEphemeral_L."No.");
      SalesLineEphemeral_L.VALIDATE("Line No.", 10000);
      SalesLineEphemeral_L.INSERT;

      SalesLineEphemeral_L.VALIDATE(Type, SalesLineEphemeral_L.Type::Item);

      SalesLineEphemeral_L.VALIDATE("No.", ItemNo_L);
      IF VariantNo_L <> '' THEN
        SalesLineEphemeral_L.VALIDATE("Variant Code", VariantNo_L);
      // --> 20210610-WI00-ADV01.01.00-3545-CJY01
      SalesLineEphemeral_L.VALIDATE("CMC Campaign No.",CampainNo_G);
      // <-- 20210610-WI00-ADV01.01.00-3545-CJY01
      SalesLineEphemeral_L.VALIDATE(Quantity, Qty_G);

      SalesLineEphemeral_L.MODIFY;

      //gestion du lancement pour le prix net
      RecSalesHeader_L.GET(SalesLineEphemeral_L."Document Type", SalesLineEphemeral_L."Document No.");
      IF RecSalesHeader_L.Status <> RecSalesHeader_L.Status::Released THEN BEGIN
        RecSalesHeader_L.GET(RecSalesHeader_L."Document Type", RecSalesHeader_L."No.");
        RecSalesHeader_L."Is Manual Release" := TRUE;
        RecSalesHeader_L.MODIFY;
        cduReleaseSalesDoc_L.DisableEmail(TRUE);
        cduReleaseSalesDoc_L.SetFromSync(TRUE);

        cduReleaseSalesDoc_L.RUN(RecSalesHeader_L); //lancement
      END;


      //on doit rechercher un record mis . jour
      SalesLineEphemeralImpacted_L.GET(SalesLineEphemeral_L."Document Type",SalesLineEphemeral_L."Document No.",
        SalesLineEphemeral_L."Line No.");
      UnitPrice_G := SalesLineEphemeralImpacted_L."Unit Price";
      NetUnitPrice_G := SalesLineEphemeralImpacted_L."Net Unit Price";

      //on rouvre avant le delete

      RecSalesHeader_L.GET(SalesHeaderEphemeral_L."Document Type"::Order, SalesHeaderEphemeral_L."No.");
      IF RecSalesHeader_L.Status <> RecSalesHeader_L.Status::Open THEN BEGIN
        cduReleaseSalesDoc_L.SetFromSync(TRUE);
        cduReleaseSalesDoc_L.Reopen(RecSalesHeader_L); //rouvrir
      END;

      //on ne doit laisser aucune trace de notre commande eph'mSre
      //on la recharge :
      SalesHeaderEphemeralToDel_L.GET(SalesHeaderEphemeral_L."Document Type"::Order, SalesHeaderEphemeral_L."No.");
      SalesHeaderEphemeralToDel_L.DELETE(TRUE);
      EndTime_G := TIME;
    END;

    PROCEDURE FormatItemVariant@1000000014(OriginalFormat_P@1000000000 : Code[31];VAR ItemNo_P@1000000001 : Code[20];VAR VariantNo_P@1000000002 : Code[10]);
    VAR
      Position_L@1000000003 : Integer;
    BEGIN
      //recherche de la pr'sence du caractSre ":"
      Position_L := STRPOS(OriginalFormat_P,':');
      IF Position_L <> 0 THEN BEGIN
        //une variante est pr'sente
        ItemNo_P := COPYSTR(OriginalFormat_P,1,Position_L - 1);
        VariantNo_P := COPYSTR(OriginalFormat_P,Position_L + 1,31);;

      END
      ELSE BEGIN
        //Il n'y a pas de variante
        ItemNo_P := OriginalFormat_P;
        VariantNo_P := '';
      END;
    END;
