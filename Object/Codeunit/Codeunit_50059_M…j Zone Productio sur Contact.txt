  {
    Date=05/10/15;
    Heure=16:41:24;
    Modifi'=Oui;
    Liste versions=REP001;
  }
  PROPERTIES
  {
    OnRun=BEGIN

            //{
            VendeurContact.RESET;
            Total := VendeurContact.COUNT;
            Fenetre.OPEN('Maj information VendeurContact ...      \' +
                         'Nb lecture                      #1######\' +
                         'Nb mises . jour                 #2######\' +
                         '@3@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@',Cpt,Cpt2);

            IF VendeurContact.FIND('-') THEN
            REPEAT
            // traitement sur les PA BOVIN
                IF (VendeurContact."Prod. Activity Code" = 'PA') AND (VendeurContact."Sales Network Code" = 'BOVIN') THEN BEGIN
                // rattach' le TC '2B2' sur la r'gion 'BZ1'
                  IF VendeurContact."Salesperson Code" = '2B2' THEN BEGIN
                    VendeurContact.DELETE;
                    VendeurContact."Sales Area Code" := 'BZ1';
                    VendeurContact.INSERT;
                    Cpt2 := Cpt2 +1;
                  END;
                // rattach' le TC '3B1' sur la r'gion 'BZ1'
                  IF VendeurContact."Salesperson Code" = '3B1' THEN BEGIN
                    VendeurContact.DELETE;
                    VendeurContact."Sales Area Code" := 'BZ1';
                    VendeurContact.INSERT;
                    Cpt2 := Cpt2 +1;
                  END;
                // rattach' les TC '4B1' et '4B2' sur la r'gion 'BZ4'
                  IF (VendeurContact."Salesperson Code" = '4B1') OR (VendeurContact."Salesperson Code" = '4B2') THEN BEGIN
                    VendeurContact.DELETE;
                    VendeurContact."Sales Area Code" := 'BZ4';
                    VendeurContact.INSERT;
                    Cpt2 := Cpt2 +1;
                  END;

                END;

              Cpt := Cpt + 1;
              Fenetre.UPDATE;
              Fenetre.UPDATE(1,Cpt);
              Fenetre.UPDATE(2,Cpt2);
              Fenetre.UPDATE(3,ROUND(Cpt / Total * 10000, 1));

            UNTIL VendeurContact.NEXT = 0;
            Fenetre.CLOSE;
            //}
          END;

  }
  CODE

      Total@1000000004 : Integer;
      Cpt@1000000005 : Integer;
      Cpt2@1000000006 : Integer;
      Cpt3@1000000022 : Integer;
      Fenetre@1000000007 : Dialog;
      ToolsFile_L@1000000029 : Codeunit 8053405;

      Texte2_G@1000000031 : Text[50];

    BEGIN
    END.
