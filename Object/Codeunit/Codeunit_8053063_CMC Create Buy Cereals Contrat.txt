    LOCAL PROCEDURE CreateCollecContFromFrameCont@8053001(recContract_P@8053000 : Record 8053239);
    VAR
      recContratColl_L@8053001 : Record 8052983;
      recContractLine_L@8053002 : Record 8053240;
      recContractType_L@8053003 : Record 8053198;
      recLigneModele_L@8053004 : Record 8053235;
      recModele_L@8053008 : Record 8052983;
      Vendor_L@8053006 : Code[10];
      recVendorBus_L@8053005 : Record 5054;
      recCompagny_L@8053007 : Record 8053075;
      cduStatusMgt_L@8053009 : Codeunit 8052985;
      recContacSalespers_L@8053010 : Record 8053160;
      recSetup_L@8053011 : Record 8053602;
      Modify_L@8053012 : Boolean;
    BEGIN
      //----------------------------------------------------------------------------------------------------------------------------------
      // Cr'ation du contrat d'achat collection depuis le contrat de vente APPRO
      //----------------------------------------------------------------------------------------------------------------------------------
      //  Entr'es
      //    - recContract_P : Contrat de vente APPRO
      //  Sorties
      //----------------------------------------------------------------------------------------------------------------------------------
      recContratColl_L.RESET;
      recContratColl_L.SETCURRENTKEY("Paper Reference Contract No.");
      recContratColl_L.SETRANGE("Paper Reference Contract No.", recContract_P."No.");
      IF recContratColl_L.FINDFIRST THEN
        Modify_L := TRUE
      ELSE
        Modify_L := FALSE;

      recContractLine_L.RESET;
      recContractLine_L.SETRANGE("Frame Contract No.", recContract_P."No.");
      IF (recContractLine_L.COUNT = 1) AND recContractLine_L.FINDFIRST THEN BEGIN
        IF recContractType_L.GET(recContract_P."Frame Contract Type") THEN BEGIN
          IF (recContractType_L.Type = recContractType_L.Type::Collect) OR
            (recContractType_L.Type = recContractType_L.Type::TAF)
          THEN BEGIN
            //Recherche de la ligne de modSle
            recLigneModele_L.RESET;
            recLigneModele_L.SETRANGE("Frame Contract Type", recContract_P."Frame Contract Type");
            recLigneModele_L.SETRANGE("Item Funnel Code", recContractLine_L."Item Funnel Code");
            IF recLigneModele_L.FINDFIRST THEN BEGIN
              recModele_L.GET(recLigneModele_L."Collect Contract Template");
              //Recherche du fournisseur
              recVendorBus_L.RESET;
              recVendorBus_L.SETCURRENTKEY("Link to Table", "Contact No.", "MC Company Code");
              recVendorBus_L.SETRANGE("Link to Table", recVendorBus_L."Link to Table"::Vendor);
              recVendorBus_L.SETRANGE("Contact No.", recContract_P."Contact No.");
              recCompagny_L.RESET;
              recCompagny_L.SETRANGE("Company Name", COMPANYNAME);
              IF recCompagny_L.FINDFIRST THEN
                recVendorBus_L.SETRANGE("MC Company Code", recCompagny_L.Code)
              ELSE
                recVendorBus_L.SETRANGE("MC Company Code", '');
              IF recVendorBus_L.FINDFIRST THEN BEGIN
                IF SalespersonCode_G = '' THEN BEGIN
                  recSetup_L.GET();
                  recContacSalespers_L.RESET;
                  recContacSalespers_L.SETRANGE("Contact Code", recVendorBus_L."No.");
                  recContacSalespers_L.SETRANGE("Prod. Activity Code", recSetup_L."Activity PV");
                  recContacSalespers_L.SETRANGE("Sales Network Code", recSetup_L."Sales Network PV");
                  IF recContacSalespers_L.FINDFIRST THEN BEGIN
                    SalespersonCode_G := recContacSalespers_L."Salesperson Code";
                  END;
                END;

                IF Modify_L = FALSE THEN BEGIN
                  recContratColl_L.INIT;
                  recContratColl_L.INSERT(TRUE);
                END ELSE BEGIN
                  cduStatusMgt_L.OpenContract(recContratColl_L);
                END;
                recContratColl_L.Type := recContratColl_L.Type::Contract;
                recContratColl_L."Campaign No." := recContract_P."Campaign No.";
                recContratColl_L."Contract Type" := recModele_L."Contract Type";
                // --> 20161027-WI00-ADV01.01.00-22043-BSI01
                IF recContratColl_L."Template No." <> recLigneModele_L."Collect Contract Template" THEN
                // <-- 20161027-WI00-ADV01.01.00-22043-BSI01
                recContratColl_L.VALIDATE("Template No.", recLigneModele_L."Collect Contract Template");
                IF (recContractType_L.Type = recContractType_L.Type::Collect) THEN
                  recContratColl_L."Collection Type" := recContratColl_L."Collection Type"::Purchase
                ELSE
                  recContratColl_L."Collection Type" := recContratColl_L."Collection Type"::"Storage for Job";
                // --> 20161027-WI00-ADV01.01.00-22043-BSI01
                IF recContratColl_L."Vendor No." <> recVendorBus_L."No." THEN
                // <-- 20161027-WI00-ADV01.01.00-22043-BSI01
                recContratColl_L.VALIDATE("Vendor No.", recVendorBus_L."No.");
                // --> 20161027-WI00-ADV01.01.00-22043-BSI01
                IF recContratColl_L."Salesperson Code" <> SalespersonCode_G THEN
                // <-- 20161027-WI00-ADV01.01.00-22043-BSI01
                recContratColl_L.VALIDATE("Salesperson Code", SalespersonCode_G);
                recContratColl_L.Description := recContract_P.Description;
                recContratColl_L.Quantity := recContractLine_L."Area (Ha)";
                recContratColl_L."Paper Reference Contract No." := recContract_P."No.";
                recContratColl_L.MODIFY;
                //Verouillage du contrat
                cduStatusMgt_L.LockContract(recContratColl_L);
              END
              ELSE
                ERROR(Text8053001);
            END
            ELSE
              ERROR(Text8053002, recContractLine_L.FIELDCAPTION("Item Funnel Code"), recContractLine_L."Item Funnel Code");
          END
          ELSE
            ERROR(Text8053003, recContractType_L.Code);
        END
        ELSE
          ERROR(Text8053004, recContract_P."Frame Contract Type");
      END
      ELSE
        ERROR(Text8053005);
    END;

    PROCEDURE SetSalesperson@8053000(SalespersonCode_P@8053000 : Code[10]);
    BEGIN
      //----------------------------------------------------------------------------------------------------------------------------------
      // R'cup'ration du code vendeur
      //----------------------------------------------------------------------------------------------------------------------------------
      //  Entr'es
      //    - SalespersonnCode_P : Code vendeur
      //  Sorties
      //----------------------------------------------------------------------------------------------------------------------------------
      SalespersonCode_G := SalespersonCode_P;
    END;
