                          ,TableSasImport_L.No,TableSasImport_L."Sub-No") ,1, MAXSTRLEN(Result_L)));
                        TraitOK_L := FALSE;
                      END ELSE BEGIN
                        IF NOT TableQualityContract_L.GET(TableSalesContract_L."No.", Analysis_L) THEN BEGIN
                          QualityCreate_L := TRUE;
                          TableQualityContract_L.VALIDATE(TableQualityContract_L."Contract No.", TableSalesContract_L."No.");
                          TableQualityContract_L.VALIDATE(TableQualityContract_L."Analysis Code", Analysis_L);
                        END ELSE BEGIN
                          QualityCreate_L := FALSE;
                        END;
                        IF (TableQualityContract_L.Value <> TableSasQuality_L.Value) OR
                         (TableQualityContract_L."Min Range Value" <> TableSasQuality_L."Min Range Value") OR
                         (TableQualityContract_L."Max Range Value" <> TableSasQuality_L."Max Range Value") OR
                         (TableQualityContract_L.Comments <> TableSasQuality_L.Comment) THEN BEGIN
                          TableQualityContract_L.VALIDATE(TableQualityContract_L.Value, TableSasQuality_L.Value);
                            TableQualityContract_L.VALIDATE(TableQualityContract_L."Min Range Value", TableSasQuality_L."Min Range Value")
      ;
                            TableQualityContract_L.VALIDATE(TableQualityContract_L."Max Range Value", TableSasQuality_L."Max Range Value")
      ;
                          TableQualityContract_L.VALIDATE(TableQualityContract_L.Comments, TableSasQuality_L.Comment);
                          ModifQual_L := TRUE;
                        END;

                        IF QualityCreate_L THEN BEGIN
                          IF NOT TableQualityContract_L.INSERT(TRUE) THEN BEGIN
                            ToolsFile_G.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0010, TableQualityContract_L.TABLECAPTION,
                              TableSalesContract_L."No."), 1, MAXSTRLEN(Result_L)));
                            TraitOK_L := FALSE;
                          END;
                        END ELSE BEGIN
                          IF ModifQual_L THEN BEGIN
                            IF NOT TableQualityContract_L.MODIFY(TRUE) THEN BEGIN
                              ToolsFile_G.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0014, TableQualityContract_L.TABLECAPTION,
                                TableSalesContract_L."No."), 1, MAXSTRLEN(Result_L)));
                              TraitOK_L := FALSE;
                            END;
                          END;
                        END;
                      END;
                    END;

                  UNTIL TableSasQuality_L.NEXT = 0;
                END;
              END;

              //----------------------------------------------
              // Mise . jour des incidences du contrat
              //----------------------------------------------
              IF TraitOK_L THEN BEGIN
                CLEAR(TableSasImpact_L);
                TableSasImpact_L.SETRANGE(TableSasImpact_L."Contract No", TableSasImport_L.No);
                TableSasImpact_L.SETRANGE(TableSasImpact_L."Sub-Contract No", TableSasImport_L."Sub-No");
                IF TableSasImpact_L.FINDSET THEN BEGIN
                  REPEAT
                    ModifIncid_L := FALSE;
                    IF NOT TableImpactContract_L.GET(TableSalesContract_L."No.", TableSasImpact_L."Impact Code") THEN BEGIN
                      ImpactCreate_L := TRUE;
                      TableImpactContract_L.VALIDATE(TableImpactContract_L."Contract No.", TableSalesContract_L."No.");
                      TableImpactContract_L.VALIDATE(TableImpactContract_L.Code, TableSasImpact_L."Impact Code");
                    END ELSE BEGIN
                      ImpactCreate_L := FALSE;
                    END;
                    IF (TableImpactContract_L.Description <> TableSasImpact_L.Description) OR
                     (TableImpactContract_L."Value (ø)" <> TableSasImpact_L.Value) OR
                     (TableImpactContract_L."Application Base" <> TableSasImpact_L."Application Base") THEN BEGIN
                      TableImpactContract_L.VALIDATE(TableImpactContract_L.Description, TableSasImpact_L.Description);
                      TableImpactContract_L.VALIDATE(TableImpactContract_L."Value (ø)", TableSasImpact_L.Value);
                      TableImpactContract_L.VALIDATE(TableImpactContract_L."Application Base", TableSasImpact_L."Application Base");
                      ModifIncid_L := TRUE;
                    END;
                    IF ImpactCreate_L THEN BEGIN
                      IF NOT TableImpactContract_L.INSERT(TRUE) THEN BEGIN
                        ToolsFile_G.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0010, TableImpactContract_L.TABLECAPTION,
                          TableSalesContract_L."No."), 1, MAXSTRLEN(Result_L)));
                        TraitOK_L := FALSE;
                      END;
                    END ELSE BEGIN
                      IF ModifIncid_L THEN BEGIN
                        IF NOT TableImpactContract_L.MODIFY(TRUE) THEN BEGIN
                          ToolsFile_G.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0014, TableImpactContract_L.TABLECAPTION,
                            TableSalesContract_L."No."), 1, MAXSTRLEN(Result_L)));
                          TraitOK_L := FALSE;
                        END;
                      END;
                    END;

                  UNTIL TableSasImpact_L.NEXT = 0;
                END;
              END;
            END;

            // Positionnement du statut
            IF TraitOK_L THEN BEGIN
              IF TableSasImport_L."Closing Date" <> 0D THEN BEGIN
                TableSalesContract_L.VALIDATE(TableSalesContract_L.Status, TableSalesContract_L.Status::Closed);
              END ELSE BEGIN
                TableSalesContract_L.Status := TableSalesContract_L.Status::Locked;
              END;
      MESSAGE ('3Je force %1, TableSalesContract_L."Standard Weight Deliveries");
              IF NOT TableSalesContract_L.MODIFY(FALSE) THEN BEGIN
                ToolsFile_G.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0014, TableSalesContract_L.TABLECAPTION, TableSalesContract_L."No.")
                  , 1, MAXSTRLEN(Result_L)));
                TraitOK_L := FALSE;
              END;
            END;

            IF TraitOK_L THEN BEGIN
              TableSasImport_L."Integration Date" := TODAY;
              TableSasImport_L."NAV No" := TableSalesContract_L."No.";
              IF NOT TableSasImport_L.MODIFY(TRUE) THEN BEGIN
                ToolsFile_G.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0005, TableSasImport_L.No, TableSasImport_L."Sub-No"), 1
                  , MAXSTRLEN(Result_L)));
              END;
            END;
          UNTIL TableSasImport_L.NEXT = 0;
        END;
      END;

      // Pour g'rer le ROLLBACK.
      CLEAR(Result_L);
      FileLogName_L := ToolsFile_G.LogName;
      ToolsFile_G.LogClose();

      IF ToolsFile_G.LogErreurType() = 'E' THEN BEGIN
        Result_L := 'ERREUR ' + FileLogName_L;
        ERROR(Result_L);
      END ELSE BEGIN
        IF ToolsFile_G.LogErreurType() = 'I' THEN BEGIN
          Result_L := Text0008;
          MoteurInterface_G.InsertLog(CodeUnit_G, FileLogName_L);
        END ELSE BEGIN
          MoteurInterface_G.InsertOk(CodeUnit_G);
        END;
      END;
      }
      //<--CDL-20140709
      //<--20140107-Int07-CMC02.02.00-ADVITAM-GMX
    END;
