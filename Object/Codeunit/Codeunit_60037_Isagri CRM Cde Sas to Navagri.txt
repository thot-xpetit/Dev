    PROCEDURE VerifLignesBrut@1000000019(VAR IsagriSASOrderHeader_P@1000000005 : Record 60094);
    VAR
      Customer_L@1000000004 : Record 18;
      ShipmentMethod_L@1000000003 : Record 10;
      Location_L@1000000002 : Record 14;
      ShipToAddress_L@1000000001 : Record 222;
      IsagriSASOrderLines_L@1000000000 : Record 60095;
    BEGIN
      //lignes
      IsagriSASOrderLines_L.RESET;
      // --> 20190708-WI00-ADV01.01.00-3422-CJY01
      IsagriSASOrderLines_L.SETRANGE(Origine,IsagriSASOrderHeader_P.Origine);
      // <-- 20190708-WI00-ADV01.01.00-3422-CJY01
      IsagriSASOrderLines_L.SETRANGE("Isagri CRM Header Guid",IsagriSASOrderHeader_P."Isagri CRM Guid");
      IF IsagriSASOrderLines_L.FINDSET THEN REPEAT
        //code client
        IF NOT Customer_L.GET(IsagriSASOrderLines_L."Sell-to Customer No.") THEN
          ERROR(Text60004,IsagriSASOrderLines_L."Sell-to Customer No.",IsagriSASOrderHeader_P."Isagri CRM Guid",
            IsagriSASOrderLines_L."Isagri CRM Line Guid");
        //gestion code article/variante envoy' par Isagri
        TranslateItemVariant(IsagriSASOrderLines_L);


        //gestion tarifs : PMA/PMP
        CLEAR(DateDebutAutomne_G);
        CLEAR(DateFinAutomne_G);
        CLEAR(DateDebutPrintemps_G);
        CLEAR(DateFinPrintemps_G);
        CLEAR(GroupePrixClientPrixMoyen_G);
        CLEAR(SaisonProduit_G);
        ConstitutionDatesSaison(IsagriSASOrderLines_L);
        IF DateDebutAutomne_G <> 0D THEN
          IsagriSASOrderLines_L.VALIDATE("Date debut liv Automne", DateDebutAutomne_G);

        IF DateFinAutomne_G <> 0D THEN
          IsagriSASOrderLines_L.VALIDATE("Date fin liv Automne", DateFinAutomne_G);

        IF DateDebutPrintemps_G <> 0D THEN
          IsagriSASOrderLines_L.VALIDATE("Date debut liv Printemps", DateDebutPrintemps_G);

        IF DateFinPrintemps_G <> 0D THEN
          IsagriSASOrderLines_L.VALIDATE("Date fin liv Printemps", DateFinPrintemps_G);

        {IF NOT IsagriSASOrderLines_L."Prix negocie OK" THEN
          IF ((IsagriSASOrderLines_L."Application Prix Tarif" = FALSE) AND (IsagriSASOrderLines_L."Application Prix Moyen" = FALSE) AND
            (IsagriSASOrderLines_L."Unit Price" = 0)) THEN
        }
        IF GroupePrixClientPrixMoyen_G <> '' THEN BEGIN
          IsagriSASOrderLines_L.VALIDATE("Gestion Prix", IsagriSASOrderLines_L."Gestion Prix"::"PMP/PMA");
          IsagriSASOrderLines_L.VALIDATE("Saison Produit",SaisonProduit_G);
          IsagriSASOrderLines_L.VALIDATE("Groupe Prix Client",GroupePrixClientPrixMoyen_G);
        END;

        IsagriSASOrderLines_L.VALIDATE("Rec Status",IsagriSASOrderHeader_P."Rec Status"::"Ok to send to Navagri");
        //IsagriSASOrderHeader_P.VALIDATE("Format Brut Ok",TRUE);
        IsagriSASOrderLines_L.MODIFY(TRUE);
      UNTIL IsagriSASOrderLines_L.NEXT = 0;
    END;

    PROCEDURE TranslateItemVariant@1000000002(VAR IsagriSASOrderLines_L@1000000002 : Record 60095);
    VAR
      Position_L@1000000001 : Integer;
    BEGIN
      //fonction utilis'e pour determiner article et eventuellement variante dans Isagri
      //sachant que Isagri nous envoi ou "code article" ou "code article/code variante"
      //si "/" d'tect'er on doit d'couper
      //Position := STRPOS(String, SubString)
      //
      Position_L := STRPOS(IsagriSASOrderLines_L."Item/Variant Isagri", '/');
      IF Position_L = 0 THEN BEGIN
        IsagriSASOrderLines_L.VALIDATE("Item No.",IsagriSASOrderLines_L."Item/Variant Isagri");
        IsagriSASOrderLines_L.MODIFY;
      END ELSE
      BEGIN
          IF Position_L >= 2 THEN BEGIN
            //article
            IsagriSASOrderLines_L.VALIDATE("Item No.",COPYSTR(IsagriSASOrderLines_L."Item/Variant Isagri",1,Position_L - 1));
            //variante
            IsagriSASOrderLines_L.VALIDATE(IsagriSASOrderLines_L."Variant Code",COPYSTR(IsagriSASOrderLines_L."Item/Variant Isagri",
              Position_L + 1,STRLEN(IsagriSASOrderLines_L."Item/Variant Isagri")));
            IsagriSASOrderLines_L.MODIFY;
          END ELSE
          BEGIN
            ERROR(Text60005,IsagriSASOrderLines_L."Item/Variant Isagri");
          END;
        END;
    END;
