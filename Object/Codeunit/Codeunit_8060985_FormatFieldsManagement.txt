    PROCEDURE "NumToDec(xxyzn)"@60022(Numeric_P@60000 : Variant;Separator_P@60001 : Text[1];NbDecimal_P@60002 : Integer) Result : Text[30];
    VAR
      Precision_L@60003 : Decimal;
      Numeric_L@60004 : Decimal;
    BEGIN
      {
      Numeric_P     Valeur num'ric . convertir
      Separator_P   Type de s'parateur
      NbDecimal_P   Nombre de d'cimale
      }

      Numeric_L   := Numeric_P;
      Precision_L := 1 / POWER(10, NbDecimal_P);
      Numeric_L   := ROUND(Numeric_L, Precision_L);
      Result      := DELCHR(CONVERTSTR(DELCHR(FORMAT(ROUND(Numeric_L*POWER(10, NbDecimal_P),1),0,'<Integer><Decimal>'),'=',','),'.',',')
      );

      IF STRLEN(Result) < NbDecimal_P THEN BEGIN
        Result := STRSUBSTNO('%1%2%3%4', 0, Separator_P, 0, Result);
        EXIT;
      END;

      IF STRLEN(Result) = NbDecimal_P THEN BEGIN
        Result := STRSUBSTNO('0%1%2', Separator_P, Result);
        EXIT;
      END;

      IF STRLEN(Result) > NbDecimal_P THEN BEGIN
        Result := STRSUBSTNO('%1%2%3', COPYSTR(Result, 1, STRLEN(Result)-NbDecimal_P)
                                     , Separator_P
                                     , COPYSTR(Result,STRLEN(Result)-NbDecimal_P+1,NbDecimal_P)
                                     );
        EXIT;
      END;
    END;

    PROCEDURE FormatLine@1120003(Data_P@1120004 : Variant;FieldLen_P@1120003 : Integer;PadChar_P@1120002 : Char;"Left/Right_P"@1120001 : 'Right,Left';EndOfText_P@1120000 : Char) : Text[250];

      {ParamStres:
       Data_P       : Data

       PadChar_P    : CaractSre ajout'
       Left/Right_P : Cadrage  Gauche/Droit
       EndOfText_P  : CaractSre fin de champ}

      // Init
      //----------------------------------------------------------------------------------------------------------------------------------
      NUL_L  := 0;
      Data_L := FORMAT(Data_P);

      CASE EndOfText_P OF

      // No character end of text
      //----------------------------------------------------------------------------------------------------------------------------------
        NUL_L: BEGIN

      // same lenght
      //----------------------------------------------------------------------------------------------------------------------------------
          IF STRLEN(Data_L) = FieldLen_P THEN EXIT (Data_L);

      // Length higher
      //----------------------------------------------------------------------------------------------------------------------------------
          IF STRLEN(Data_L) > FieldLen_P THEN BEGIN
            CASE "Left/Right_P" OF
              "Left/Right_P"::Right:  EXIT(TruncStrRight(Data_L,FieldLen_P));

            END;
          END;

      // Length lower
      //----------------------------------------------------------------------------------------------------------------------------------

            CASE "Left/Right_P" OF
              "Left/Right_P"::Right: EXIT(PadStrRight(Data_L, FieldLen_P, PadChar_P));
              "Left/Right_P"::Left:  EXIT(PadStrLeft(Data_L, FieldLen_P, PadChar_P));
            END;
          END;
        END;

      // No character end of text
      //----------------------------------------------------------------------------------------------------------------------------------
        ELSE BEGIN

      // same lenght
      //----------------------------------------------------------------------------------------------------------------------------------
          IF STRLEN(Data_L) = FieldLen_P THEN EXIT (Data_L+FORMAT(EndOfText_P));

      // Length higher
      //----------------------------------------------------------------------------------------------------------------------------------
          IF STRLEN(Data_L) > FieldLen_P THEN BEGIN
            CASE "Left/Right_P" OF
              "Left/Right_P"::Right: EXIT (TruncStrRight(Data_L, FieldLen_P)+FORMAT(EndOfText_P));

      // Length lower

          IF STRLEN(Data_L) < FieldLen_P THEN BEGIN
            CASE "Left/Right_P" OF
              "Left/Right_P"::Right: EXIT (PadStrRight(Data_L, FieldLen_P, PadChar_P)+FORMAT(EndOfText_P));
              "Left/Right_P"::Left:  EXIT (PadStrLeft(Data_L, FieldLen_P, PadChar_P)+FORMAT(EndOfText_P));
            END;
          END;
        END;
      END;
    END;
