    PROCEDURE CashExportMerge@1000000000(FileName_P@1000000002 : Text[128];Directory_P@1000000015 : Text[128]) Result_L : Text[256];
    VAR
      "<-- 3LI LOCALS -->"@1000000003 : Integer;
      DernNoTrait_L@1000000024 : Integer;
      FileExport_L@1000000005 : File;
      FileExportOutStream_L@1000000009 : OutStream;
      TableCashSetup_L@1000000012 : Record 8053400;
      TableBankAccount_L@1000000001 : Record 271;
      TableBankAccountPostingGroup_L@1000000011 : Record 277;
      ToolsFile_L@1000000000 : Codeunit 8053405;
      MoteurInterface_L@1000000007 : Codeunit 8061230;
      ExtComDlgMgt_L@1000000006 : Codeunit 8060994;
      CodeUnit_L@1000000004 : Integer;
      ConnectorName_L@1000000020 : Code[20];
      ImpExpFolder_L@1000000019 : Text[256];
      LogFolder_L@1000000018 : Text[256];
      ExtensionFile_L@1000000023 : Text[10];
      PrefixFile_L@1000000022 : Text[10];
      FileName_L@1000000017 : Text[256];
      FilePath_L@1000000016 : Text[256];
      FileN_L@1000000013 : Text[256];
      FileTemp_L@1000000010 : Text[256];
      FileLogName_L@1000000008 : Text[256];
      NbCarEcrit_L@1000000014 : Integer;
      FileNameOuvert_L@1000000021 : Boolean;
    BEGIN
      //----------------------------------------------------------------------------------------------------------------------------------
      // Extraction des 'critures > au dernier ID trait' vers fichier texte longueur fixe
      //----------------------------------------------------------------------------------------------------------------------------------
      // Entr'es:
      //   - FileName_P : Nom du fichier PC . G'n'rer ou vide si utilisation du nom classique sss + MM + JJ + '.TXT'
      //   - Directory_P : Nom du r'pertoire ou placer le fichier si FileName_P non pr'ciser ==> Nom classique
      // Sorties:
      //   - Result_L : Erreur Retourn'e, Vide si Ok
      //----------------------------------------------------------------------------------------------------------------------------------
      //-->20130521-Int02-CMC02.02.00-ADVITAM-CDL
      FileNameOuvert_L := FALSE;
      CodeUnit_L := 8053400;
      MoteurInterface_L.GetParameter(CodeUnit_L, ConnectorName_L, ImpExpFolder_L, LogFolder_L, ExtensionFile_L, PrefixFile_L);
      ToolsFile_L.LogInit(LogFolder_L, 'TRESO');
      //-->20131112-Int02-CMC02.02.00-ADVITAM-CDL
      ToolsFile_L.LogInitCodeUnit(CodeUnit_L);
      //<--20131112-Int02-CMC02.02.00-ADVITAM-CDL

      //-->20130628-Int02-CMC02.02.00-ADVITAM-CDL
      CLEAR(DernNoTrait_L);
      //<--20130628-Int02-CMC02.02.00-ADVITAM-CDL

      //<--20130521-Int02-CMC02.02.00-ADVITAM-CDL
      IF (NOT TableCashSetup_L.FINDFIRST) OR
        ((TableCashSetup_L."Export RAPPRO") AND (TableCashSetup_L."Company Name" = '')) THEN BEGIN

        Result_L :=  STRSUBSTNO(Text0003);
        ToolsFile_L.LogWriteErreur(Result_L);
      END ELSE BEGIN
        IF (TableCashSetup_L."Export RAPPRO") THEN BEGIN
          IF FileName_P <> '' THEN BEGIN
            FileName_L := FileName_P;
          END ELSE BEGIN
            IF Directory_P <> '' THEN BEGIN
              FileName_L := Directory_P + '\' + TableCashSetup_L."Company Name" +
                FORMAT(CURRENTDATETIME, 0,'<Year4><Month,2><Day,2><Hours24,2><Minutes,2><Seconds,2><Second dec>') + '.txt';
            END ELSE BEGIN
              FileName_L := ImpExpFolder_L + '\' + PrefixFile_L +
                FORMAT(CURRENTDATETIME, 0,'<Year4><Month,2><Day,2><Hours24,2><Minutes,2><Seconds,2><Second dec>') + '.' + ExtensionFile_L;
            END;
          END;
          ToolsFile_L.LogInitFileExecute(FileName_L);
          TableBankAccount_L.SETCURRENTKEY(TableBankAccount_L."Entry No.");
          // Filtre N› > au dernier trait'
          TableBankAccount_L.SETFILTER(TableBankAccount_L."Entry No.", '> %1', TableCashSetup_L."Last Entry No RAPPRO");
          IF TableBankAccount_L.FINDSET THEN BEGIN
            //-->20130521-Int02-CMC02.02.00-ADVITAM-CDL
            ToolsFile_L.SplitFilename(FileName_L, FilePath_L, FileN_L);
            FileTemp_L := FilePath_L + 'Temp_' + FileN_L;
            // BEFORE
            {
            Result_L := ToolsFile_L.FileCreateNotExist(FileName_L, '.temp', FileExport_L);
            }
            // AFTER
            Result_L := ToolsFile_L.FileCreateNotExist(FileName_L, 'Temp_', FileExport_L);
            //<--20130521-Int02-CMC02.02.00-ADVITAM-CDL
            IF Result_L <> '' THEN BEGIN
              ToolsFile_L.LogWriteErreur(Result_L);
            END ELSE BEGIN
              FileNameOuvert_L := TRUE;
              FileExport_L.CREATEOUTSTREAM(FileExportOutStream_L);

              REPEAT
                FileExportOutStream_L.WRITETEXT(PADSTR(TableCashSetup_L."Company Name", 3, ' '), 3);
                // Compte Zone "Bank Acc. Posting Group" Prendre la relation de la table 277 "Bank Account Posting Group"
                IF TableBankAccountPostingGroup_L.GET(TableBankAccount_L."Bank Acc. Posting Group") THEN
                  FileExportOutStream_L.WRITETEXT(PADSTR(TableBankAccountPostingGroup_L."G/L Bank Account No.",
                    9, ' '), 9)
                ELSE
                  FileExportOutStream_L.WRITETEXT(PADSTR(TableBankAccount_L."Bank Acc. Posting Group",
                    9, ' '), 9);
                // N› ligne
                FileExportOutStream_L.WRITETEXT('00000', 5);
                // Origine
                FileExportOutStream_L.WRITETEXT('A', 1);
                // Date op'ration
                FileExportOutStream_L.WRITETEXT(PADSTR(FORMAT(TableBankAccount_L."Posting Date", 0,
                  '<Year4><Month,2><Day,2>'), 8, '0'), 8);
                // Libell'
                FileExportOutStream_L.WRITETEXT(PADSTR(TableBankAccount_L.Description, 30, ' '), 30);
                // ChSque
                //-->20130220-Int02-CMC02.02.00-ADVITAM-CDL
                FileExportOutStream_L.WRITETEXT(PADSTR(TableBankAccount_L."External Document No.", 7, ' '), 7);
                //<--20130220-Int02-CMC02.02.00-ADVITAM-CDL
                // Montant en centime Signe Devant
                IF (TableBankAccount_L."Amount (LCY)" < 0) THEN BEGIN
                  FileExportOutStream_L.WRITETEXT('-', 1);
                END ELSE BEGIN
                  FileExportOutStream_L.WRITETEXT('+', 1);
                END;
                FileExportOutStream_L.WRITETEXT(CONVERTSTR(FORMAT(ABS(TableBankAccount_L."Amount (LCY)") * 100,
                    0, '<Filler Character,0><Precision,0><Sign><Integer,12><Decimals>'),
                    ' ', '0'), 12);
                //-->20130402-Int02-CMC02.02.00-ADVITAM-CDL
                // Mode rSglement
                FileExportOutStream_L.WRITETEXT(PADSTR(TableBankAccount_L."Payment Method Code", 10, ' '), 10);
                //<--20130402-Int02-CMC02.02.00-ADVITAM-CDL
                FileExportOutStream_L.WRITETEXT();
                //-->20130628-Int02-CMC02.02.00-ADVITAM-CDL
                DernNoTrait_L := TableBankAccount_L."Entry No.";
                //<--20130628-Int02-CMC02.02.00-ADVITAM-CDL
              UNTIL TableBankAccount_L.NEXT=0;
              //-->20130521-Int02-CMC02.02.00-ADVITAM-CDL
              NbCarEcrit_L := FileExport_L.LEN;
              //-->20130220-Int02-CMC02.02.00-ADVITAM-CDL
              // BEFORE
              {
              IF (FileExport_L.LEN > 0) THEN BEGIN
                FileExport_L.CLOSE;
                IF NOT ToolsFile_L.FileCopySav(FileName_L, 'Sav', TRUE) THEN BEGIN
                  Result_L := STRSUBSTNO(Text0011, FileName_L);
                  ToolsFile_L.LogWriteErreur(Result_L);
                END;
                //<--20130220-Int02-CMC02.02.00-ADVITAM-CDL
              END ELSE BEGIN
                FileExport_L.CLOSE;
                FILE.ERASE(FileName_L + '.temp');
              END;
              // Enregistrement dernier Id trait' et date export.
              TableCashSetup_L."Last Entry No RAPPRO" := TableBankAccount_L."Entry No.";
              TableCashSetup_L."Last Date Export RAPPRO" := CURRENTDATETIME;
              IF NOT TableCashSetup_L.MODIFY(TRUE) THEN BEGIN
                Result_L :=  STRSUBSTNO(Text0003, FileName_L);
                ToolsFile_L.LogWriteErreur(Result_L);
              END;
              }
              // AFTER
              FileExport_L.CLOSE;
              //<--20130521-Int02-CMC02.02.00-ADVITAM-CDL
            END;
          END;
        END;
      END;

      //-->20130521-Int02-CMC02.02.00-ADVITAM-CDL
      // BEFORE
      {
      ToolsFile_L.LogClose();
      }
      // AFTER
      CLEAR(Result_L);
      FileLogName_L := ToolsFile_L.LogName;
      ToolsFile_L.LogClose();
      IF ToolsFile_L.LogErreurType() = 'E' THEN BEGIN
        IF (NbCarEcrit_L > 0) THEN BEGIN
          ERASE(FileTemp_L);
        END;
        Result_L := 'ERREUR ' + FileLogName_L;
        ERROR(Result_L);
      END ELSE BEGIN
        IF (NbCarEcrit_L > 0) THEN BEGIN
          IF RENAME(FileTemp_L, FileName_L) THEN BEGIN
          END;
          IF NOT ToolsFile_L.FileCopySav(FileName_L, 'Sav', TRUE) THEN BEGIN
            Result_L := STRSUBSTNO(Text0011, FileName_L);
            ToolsFile_L.LogWriteErreur(Result_L);
          END;
        END ELSE BEGIN
          IF FileNameOuvert_L THEN BEGIN
            FILE.ERASE(FileTemp_L);
          END;
        END;
        // Enregistrement dernier Id trait' et date export.
        //-->20130628-Int02-CMC02.02.00-ADVITAM-CDL
        // BEFORE
        {
        TableCashSetup_L."Last Entry No RAPPRO" := TableBankAccount_L."Entry No.";
        }
        // AFTER
        IF DernNoTrait_L <> 0 THEN BEGIN
          TableCashSetup_L."Last Entry No RAPPRO" := DernNoTrait_L;
        //<--20130628-Int02-CMC02.02.00-ADVITAM-CDL
          TableCashSetup_L."Last Date Export RAPPRO" := CURRENTDATETIME;
          IF NOT TableCashSetup_L.MODIFY(TRUE) THEN BEGIN
            Result_L :=  STRSUBSTNO(Text0003, FileName_L);
            ToolsFile_L.LogWriteErreur(Result_L);
          END;
        //-->20130628-Int02-CMC02.02.00-ADVITAM-CDL
        END;
        //<--20130628-Int02-CMC02.02.00-ADVITAM-CDL

        IF ToolsFile_L.LogErreurType() = 'I' THEN BEGIN
          Result_L := Text0012;
          MoteurInterface_L.InsertLog(CodeUnit_L, FileLogName_L);
        END ELSE BEGIN
          MoteurInterface_L.InsertOk(CodeUnit_L);
        END;
      END;
      //<--20130521-Int02-CMC02.02.00-ADVITAM-CDL
      //<--20130118-Int02-CMC02.02.00-ADVITAM-CDL
    END;

    BEGIN
    {
      //--> 3LI BEGIN
      20130118-Int02-CMC02.02.00-ADVITAM-CDL : - Fonction CashExportMerge : Export rapprochement tr'sorerie
      20130220-Int02-CMC02.02.00-ADVITAM-CDL : - Ajout zone N› ChSque
      20130402-Int02-CMC02.02.00-ADVITAM-CDL : - Ajout zone mode de rSglement
      20130521-Int02-CMC02.02.00-ADVITAM-CDL : - Ajout Robot & correspondance. Remplacement tout LogWrite Par LogWriteErreur Sans
        utilisation de variable (2 lignes en 1 ligne).
      20130628-Int02-CMC02.02.00-ADVITAM-CDL : - ProblSme si pas d''criture dernier N› trait' remis . 0.
      20131112-Int02-CMC02.02.00-ADVITAM-CDL : - Ajout appel fonction ToolsFile_L.LogInitCodeUnit.
      //<-- 3LI END
    }
    END.
