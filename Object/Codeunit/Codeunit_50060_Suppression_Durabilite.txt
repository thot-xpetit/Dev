OBJECT Codeunit 50060 Suppression_Durabilite
{
  OBJECT-PROPERTIES
  {
    Date=23/01/17;
    Heure=14:09:41;
    Modifi'=Oui;
    Liste versions=REP001;
  }
  PROPERTIES
  {
    OnRun=VAR
            ZoneDec@1000000000 : Decimal;
          BEGIN

            //===============================================================================================================================
            //                      Suppression donn'es DURABILITE
            //===============================================================================================================================
            Cpt  := 0;
            Cpt2 := 0;
            Durability_G.RESET;
            Total := Durability_G.COUNT;
            Fenetre.OPEN('Maj information durabilit'              \' +
                         'Nb lecture                      #1######\' +
                         'Nb MAJ                          #2######\' +
                         '@4@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@',Cpt,Cpt2);

            Durability_G.SETRANGE(Durability_G."Campaign No.",'2017','2019');

            IF Durability_G.FIND('-') THEN
            REPEAT

              IF (COMPANYNAME = 'UNEAL') OR  (COMPANYNAME = 'TERNOVEO')  THEN BEGIN
                IF (Durability_G."Campaign No." = '2017')
                  OR (Durability_G."Campaign No." = '2018')
                  OR (Durability_G."Campaign No." = '2019') THEN BEGIN
                  Durability_G.DELETE;
                  Cpt2 := Cpt2 + 1;
                END;
              END;

            Cpt := Cpt + 1;
            Fenetre.UPDATE;
            Fenetre.UPDATE(1,Cpt);
            Fenetre.UPDATE(2,Cpt2);
            Fenetre.UPDATE(4,ROUND(Cpt / Total * 10000, 1));

            UNTIL Durability_G.NEXT = 0;
            Fenetre.CLOSE;
            //}
          END;

  }
  CODE
  {
    VAR
      Total@1000000004 : Integer;
      Cpt@1000000005 : Integer;
      Cpt2@1000000006 : Integer;
      Cpt3@1000000022 : Integer;
      Fenetre@1000000007 : Dialog;
      Ok_lancement@1000000013 : Boolean;
      Durability_G@1000000000 : Record 8053125;

    BEGIN
    END.
  }
}
