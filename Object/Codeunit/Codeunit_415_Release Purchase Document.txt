    PROCEDURE Reopen@1(VAR PurchHeader@1000 : Record 38);

      MailMgt_L@8060985 : Codeunit 8060999;

      WITH PurchHeader DO BEGIN
        IF Status = Status::Open THEN
          EXIT;

          WhsePurchRelease.Reopen(PurchHeader);
          // <-- 20110601-Wi00-CMC01.04
          G_cduLogisticsTourMngt.Reopen(PurchHeader);
          // --> 20110601-Wi00-CMC01.04
        END;

        PurchLine.SETRANGE("Document No.","No.");
        PurchLine.SETFILTER(Type,'>0');
        PurchLine.SETFILTER(Quantity,'<>0');
        IF PurchLine.FIND('-') THEN
          REPEAT
            PurchLine.Amount := 0;
            PurchLine."Amount Including VAT" := 0;
            PurchLine."VAT Base Amount" := 0;
            PurchLine.InitOutstandingAmount;
            PurchLine.MODIFY;
          UNTIL PurchLine.NEXT = 0;

      END;

                  CMCVariantEntrySetup."Purch. Entry Check"::Mandatory:
