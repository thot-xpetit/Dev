    PROCEDURE UpdateMCBankAccount@8053023(MCContactBankAccount_P@1180250002 : Record 8053135);
    VAR
      RecContBusRel_L@1180250003 : Record 5054;
      RecCustBankAccount_L@1180250004 : Record 287;
      RecVendBankAccount_L@1180250005 : Record 288;
    BEGIN
      //----------------------------------------------------------------------------------------------------------------------------------
      // MAJ des comptes bancaires client/fournisseur dans toutes les soci't's ou le compte bancaire est pr'sent.Cr'ation si inexistant
      //----------------------------------------------------------------------------------------------------------------------------------
      // Entr'es:
      //   - MCContactBankAccount_P : Enregistrement compte bancaire contact
      // Sorties:
      //   -
      //----------------------------------------------------------------------------------------------------------------------------------
      RecContBusRel_L.RESET;
      RecContBusRel_L.SETRANGE("Contact No.",MCContactBankAccount_P."Contact No.");
      IF RecContBusRel_L.FINDSET THEN BEGIN
        REPEAT
          CASE RecContBusRel_L."Link to Table" OF
            RecContBusRel_L."Link to Table"::Customer :
            BEGIN
              IF NOT (cduMCDataMgt_G.GetMCCompanyCode(COMPANYNAME) = RecContBusRel_L."MC Company Code") THEN
                RecCustBankAccount_L.CHANGECOMPANY(GetMCCompanyName(RecContBusRel_L."MC Company Code"));
              RecCustBankAccount_L.SETRANGE("Customer No.",RecContBusRel_L."No.");
              RecCustBankAccount_L.SETRANGE(Code,MCContactBankAccount_P.Code);
              IF RecCustBankAccount_L.FINDFIRST THEN BEGIN
                RecCustBankAccount_L.TRANSFERFIELDS(MCContactBankAccount_P);
                RecCustBankAccount_L.MODIFY;
              END ELSE BEGIN
                RecCustBankAccount_L.TRANSFERFIELDS(MCContactBankAccount_P);
                RecCustBankAccount_L.INSERT;
              END;
           END;
            RecContBusRel_L."Link to Table"::Vendor :
            BEGIN
              IF NOT (cduMCDataMgt_G.GetMCCompanyCode(COMPANYNAME) = RecContBusRel_L."MC Company Code") THEN
                RecVendBankAccount_L.CHANGECOMPANY(GetMCCompanyName(RecContBusRel_L."MC Company Code"));
              RecVendBankAccount_L.SETRANGE("Vendor No.",RecContBusRel_L."No.");
              RecVendBankAccount_L.SETRANGE(Code,MCContactBankAccount_P.Code);
              IF RecVendBankAccount_L.FINDFIRST THEN BEGIN
                RecVendBankAccount_L.TRANSFERFIELDS(MCContactBankAccount_P);
                RecVendBankAccount_L.MODIFY;
              END ELSE BEGIN
                RecVendBankAccount_L.TRANSFERFIELDS(MCContactBankAccount_P);
                RecVendBankAccount_L.INSERT;
              END;
            END;
          END;
        UNTIL RecContBusRel_L.NEXT=0;
      END;
    END;

                RecDefaultDim2_L.INSERT;
              UNTIL RecDefaultDim_L.NEXT = 0;
          END;

          UpdateQuotes(RecCust_L);
          Message_L += STRSUBSTNO(Text8053000,RecCust_L.TABLECAPTION,RecCust_L."No.",GetMCCompanyName(RecContBusRel_P."MC Company Code")
      );
        END;
      END;

      EXIT(Message_L);
    END;
