                    END;

                    IF NOT ErreurEntete_L THEN BEGIN

                        ToolsFile_L.LogWriteErreur(STRSUBSTNO(Text0010, TableFactLig_L.TABLECAPTION,

              END;

              END;
            END;
          END;
          ModifAnalytique;
        END;
      END;
