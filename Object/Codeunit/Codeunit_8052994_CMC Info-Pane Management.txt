    PROCEDURE CalcWeighAndCubForTour@8053012(P_WSHeader@8053000 : Record 7320;P_WaC@8053003 : 'W,C,Q') Res_Quantity : Decimal;
    VAR
      L_recWSLine@8053001 : Record 7321;
      L_CuWMSMGt@8053002 : Codeunit 7302;
      L_Weight@8053004 : Decimal;
      L_Weight2@8053006 : Decimal;
      L_Cubage@8053005 : Decimal;
      L_Cubage2@8053007 : Decimal;
      L_Qty@8053008 : Decimal;
    BEGIN
      Res_Quantity := 0;
      L_Weight:=0;
      L_Cubage:=0;
      L_Qty:=0;


      L_recWSLine.RESET;
      L_recWSLine.SETRANGE("No.",P_WSHeader."No.");
      IF L_recWSLine.FINDSET(FALSE,FALSE) THEN
        REPEAT
          L_CuWMSMGt.CalcCubageAndWeight(L_recWSLine."Item No.",L_recWSLine."Unit of Measure Code",L_recWSLine.Quantity,
            L_Cubage2,L_Weight2);
          L_Weight:=L_Weight+L_Weight2;
          L_Cubage:=L_Cubage+L_Cubage2;
          L_Qty:=L_Qty+L_recWSLine.Quantity;
        UNTIL L_recWSLine.NEXT=0;

      IF P_WaC=0 THEN
        Res_Quantity:=L_Weight;
      IF P_WaC=1 THEN
        Res_Quantity:=L_Cubage;
      IF P_WaC=2 THEN
        Res_Quantity:=L_Qty;
      EXIT(Res_Quantity)
    END;
