      // Calcul du TOTAL Quantit' RECUES
      // -------------------------------------------------------------------------------
      RES_Value := 0;

      L_recCMCCollectionContract.RESET;
      L_recCMCCollectionContract.SETCURRENTKEY(Type,"Collection Type","Species Code","Campaign No.",Status);
      L_recCMCCollectionContract.SETRANGE(Type, L_recCMCCollectionContract.Type::Contract);

        REPEAT
          L_recCMCCollectionContract.CALCFIELDS("Quantity Received");
          L_ContractQty := L_recCMCCollectionContract."Quantity Received";
          IF L_recCMCCollectionContract.Status <> L_recCMCCollectionContract.Status::Finished THEN BEGIN
            L_OutstandingQty := L_recCMCCollectionContract.Quantity - L_recCMCCollectionContract."Quantity Received";
            IF L_OutstandingQty > 0 THEN
              L_ContractQty := L_ContractQty + L_OutstandingQty;
          END;

          IF P_Result = P_Result::Value THEN
            RES_Value := RES_Value + (L_ContractQty * L_recCMCCollectionContract."Unit Price")
          ELSE
            RES_Value := RES_Value + L_ContractQty;
        UNTIL L_recCMCCollectionContract.NEXT = 0;
      END;

    END;

    BEGIN

      // Calcul du TOTAL Quantit' ACHAT MATIF OPTION

      RES_Value := 0;

      L_recCMCMATIFContract.RESET;
      L_recCMCMATIFContract.SETCURRENTKEY(Type, Way, "Species Code", "Campaign No.", Strategy, "Due Date", Status);

        L_recCMCMATIFContract.SETRANGE("Due Date", CALCDATE('<-1M+CM>', P_DueDateFilter)+1, CALCDATE('<CM>', P_DueDateFilter));

      IF L_recCMCMATIFContract.FIND('-') THEN BEGIN
        IF G_recCMCSetup.GET THEN BEGIN
          REPEAT
            IF (P_CurrentPrice > L_recCMCMATIFContract.Strike) THEN BEGIN
              IF ((L_recCMCMATIFContract.Way = L_recCMCMATIFContract.Way::Purchase)   AND
                  (L_recCMCMATIFContract.Option = L_recCMCMATIFContract.Option::Call))
              THEN BEGIN

                  L_ContractQty := L_recCMCMATIFContract."Lot Number" * G_recCMCSetup."MATIF Weight per Lot";
                  L_Price := P_CurrentPrice - L_recCMCMATIFContract.Strike - L_recCMCMATIFContract.Price;
                END ELSE BEGIN
                  IF P_Result = P_Result::Value THEN
                    L_ContractQty := L_recCMCMATIFContract."Lot Number" * G_recCMCSetup."MATIF Weight per Lot"
                  ELSE
                    L_ContractQty := 0;
                  L_Price := -L_recCMCMATIFContract.Price;
                END;
              END;
            END ELSE BEGIN
              IF ((L_recCMCMATIFContract.Way = L_recCMCMATIFContract.Way::Sales)     AND
                  (L_recCMCMATIFContract.Option = L_recCMCMATIFContract.Option::Put))
              THEN BEGIN
                IF (NOT L_recCMCMATIFContract.Complete) THEN BEGIN
                  L_ContractQty := L_recCMCMATIFContract."Lot Number" * G_recCMCSetup."MATIF Weight per Lot";
                  L_Price := L_recCMCMATIFContract.Strike - P_CurrentPrice + L_recCMCMATIFContract.Price;
                END ELSE BEGIN
                  IF P_Result = P_Result::Value THEN
                    L_ContractQty := L_recCMCMATIFContract."Lot Number" * G_recCMCSetup."MATIF Weight per Lot"
                  ELSE
                    L_ContractQty := 0;
                  L_Price := L_recCMCMATIFContract.Price;
                END;
              END;
            END;

            IF P_Result = P_Result::Value THEN
              RES_Value := RES_Value + (L_ContractQty * L_Price)
            ELSE
              RES_Value := RES_Value + L_ContractQty;
          UNTIL L_recCMCMATIFContract.NEXT = 0;
        END;
      END;

      EXIT(RES_Value);
