      // *** Calcul capital r'el***//
      CLEAR(RealCap_P);

      Contact_L.GET(FarmerNo_P);
      PartRec_L.RESET;
      PartRec_L.SETFILTER("Member Filter",Contact_L."No.");
      PartRec_L.SETFILTER("Date Filter",'..%1',PostingDate_P);
      IF PartRec_L.FINDFIRST THEN
      PartRec_L.CALCFIELDS("Total Part Amount");
      RealCap_P := PartRec_L."Total Part Amount";
      RealCap_P := ROUND(RealCap_P,GLSetup_G."Amount Rounding Precision");

      // *** Calcul capital statutaire***//
      CLEAR(StatCap_P);
      CurrCampaign_L.GET(CampaignNo_P);
      CurrCampaign_L.TESTFIELD("Statutory Capital Table Code");
      CurrCampaign_L.TESTFIELD("Campaigns Number");
      TempCampaign_L.DELETEALL;
      I_L := CurrCampaign_L."Campaigns Number";
      REPEAT
         TempCampaign_L.INIT;
         TempCampaign_L.TRANSFERFIELDS(CurrCampaign_L);
         TempCampaign_L.INSERT;
       I_L-=1;
      UNTIL (CurrCampaign_L.NEXT(-1)=0) OR (I_L=0) ;


      Campaign_L.GET(CampaignNo_P);
      AnalysisReportLine_L.RESET;
      AnalysisReportLine_L.SETFILTER(AnalysisReportLine_L."Contact Filter",FarmerNo_P);
      AnalysisReportLine_L.SETRANGE(Code,Campaign_L."Statutory Capital Table Code");
      AnalysisReportLine_L.SETRANGE("Line Type",AnalysisReportLine_L."Line Type"::"Funnel Filter");
      IF AnalysisReportLine_L.FINDSET THEN
        REPEAT
          CLEAR(Average_L);
          IF TempCampaign_L.FINDFIRST THEN
            REPEAT
              AnalysisEntry_L.RESET;
              AnalysisEntry_L.SETCURRENTKEY(Code,Line,Date,"Contact No.","Entry No.");
              AnalysisEntry_L.SETRANGE(Code,Campaign_L."Statutory Capital Table Code");
              AnalysisEntry_L.SETRANGE(Line,AnalysisReportLine_L.Line);

            StatCap_P += Average_L* CapitalRates_L.Rate;

      StatCap_P := ROUND(StatCap_P,GLSetup_G."Amount Rounding Precision");
    END;

    PROCEDURE BooRistourne@1000000001(AnalysisReportLine_P@1000000000 : Record 8053361;ContactNo_P@1000000001 : Code[20];CampainNo_P@1000000002 : Code[10];CustomerNo_P@1000000007 : Code[20]) Boo_R : Boolean;
    VAR
      SupFrameContractHead_L@1000000003 : Record 8053239;
      FrameContract_L@1000000004 : Record 8053208;
      FrameContractType_L@1000000005 : Record 8053198;
      SalesCrMemoHeader_L@1000000006 : Record 114;
    BEGIN
      // --> 20161012-WI00-ADV01.01.00-3140-JCD01
      Boo_R := FALSE;

      CASE AnalysisReportLine_P."Disc. Contrat Type" OF
      '' : Boo_R := TRUE;
      'CCC' : BEGIN
                FrameContract_L.SETCURRENTKEY("Contact No.","Contract Type","Date Signed");
                FrameContract_L.SETRANGE("Contact No.",ContactNo_P);
                FrameContract_L.SETRANGE("Contract Type",AnalysisReportLine_P."Disc. Contrat Type");
                FrameContract_L.SETRANGE("Campaign No.",CampainNo_P);
                IF AnalysisReportLine_P."Disc. Commitment Code Filter" <> '' THEN
                  FrameContract_L.SETFILTER("Commitment Code",AnalysisReportLine_P."Disc. Commitment Code Filter");
                IF NOT FrameContract_L.ISEMPTY THEN
                  Boo_R := TRUE;
              END;
      ELSE    BEGIN
                SupFrameContractHead_L.SETCURRENTKEY("Contact No.","Frame Contract Type","Creation Date");
                SupFrameContractHead_L.SETRANGE("Contact No.",ContactNo_P);
                SupFrameContractHead_L.SETRANGE("Frame Contract Type",AnalysisReportLine_P."Disc. Contrat Type");
                SupFrameContractHead_L.SETRANGE("Campaign No.",CampainNo_P);
                IF AnalysisReportLine_P."Disc. Commitment Code Filter" <> '' THEN
                  SupFrameContractHead_L.SETFILTER("Commitment Code",AnalysisReportLine_P."Disc. Commitment Code Filter");
                IF NOT SupFrameContractHead_L.ISEMPTY THEN
                  Boo_R := TRUE;
              END;
      END;
      // <-- 20161012-WI00-ADV01.01.00-3140-JCD01

      // --> 20161102-WI00-ADV01.01.00-3140-JCD01
      //Le contrat doit ^tre honor' pour avoir une prime (pour les CCA ou JA).
      //S'il y a une prime et qu'il est honor' FrameContractType_L."Doc. Category Code" donne la cat'gorie de la facture
      // S'il existe un avoir avec cette cat'gorie, l'agriculteur a bien honor' son contrat...
      IF Boo_R AND (CustomerNo_P <> '') THEN
        IF FrameContractType_L.GET(AnalysisReportLine_P."Disc. Contrat Type") THEN
          IF FrameContractType_L."Doc. Category Code" <> '' THEN BEGIN
            SalesCrMemoHeader_L.RESET;
            SalesCrMemoHeader_L.SETCURRENTKEY("Sell-to Customer No.","External Document No.");
            SalesCrMemoHeader_L.SETRANGE("Sell-to Customer No.",CustomerNo_P);
            SalesCrMemoHeader_L.SETRANGE("CMC Campaign No.",CampainNo_P);
            SalesCrMemoHeader_L.SETRANGE("Doc. Category Code",FrameContractType_L."Doc. Category Code");
            IF SalesCrMemoHeader_L.ISEMPTY THEN
              Boo_R := FALSE;
          END;
      // <-- 20161102-WI00-ADV01.01.00-3140-JCD01

      // --> 20191002-WI00-ADV01.01.00-3442-MRU01
      //Dans le cadre d'un contrat CCA : on ne tient plus compte de la pr'sence d'un avoir mais du champ Conformit' pour paiement
      IF AnalysisReportLine_P."Disc. Contrat Type" = 'CCA' THEN BEGIN
        SupFrameContractHead_L.SETRANGE("Conformity For Payment", SupFrameContractHead_L."Conformity For Payment"::Yes);
        IF NOT SupFrameContractHead_L.ISEMPTY THEN
          Boo_R := TRUE
        ELSE
          Boo_R := FALSE;
      END;
      // <-- 20191002-WI00-ADV01.01.00-3442-MRU01
    END;

    BEGIN

                changement place chargement "campagne")
      20160926-Wi00-ADV01.01.00-3140-JCD01 : MODCODE (Ajout seuil de montant ristourne, dividende et distribution totale)
      20161012-WI00-ADV01.01.00-3140-JCD01 : ADDCODE (Gestion de filtre type contrat et code engagement dans les ristournes)
                                             MODFUNCT (BooRistourne)
