OBJECT Codeunit 8053000 CMC Intervention Jnl.-Post
{
  OBJECT-PROPERTIES
  {
    Date=01/08/11;
    Heure=00:00:00;
    Liste versions=CMC01.04;
  }
  PROPERTIES
  {
    TableNo=8053034;
    OnRun=BEGIN
            G_recInterventionJnlLine.COPY(Rec);
            Code;
            Rec.COPY(G_recInterventionJnlLine);
          END;

  }
  CODE
  {
    VAR
      Text000@1000 : TextConst 'ENU=Do you want to post the journal lines?;FRA=Souhaitez-vous valider les lignes de la feuille ?';
      Text001@1001 : TextConst 'ENU=There is nothing to post.;FRA=Il n''y a rien . valider.';
      Text002@1002 : TextConst 'ENU=The journal lines were successfully posted.;FRA=Les lignes de la feuille ont 't' valid'es avec succSs.';
      Text003@1003 : TextConst 'ENU=The journal lines were successfully posted. You are now in the %1 journal.;FRA=Les lignes de la feuille ont 't' valid'es avec succSs. Vous vous trouvez maintenant dans la feuille %1.';
      G_recInterventionJnlTempl@1004 : Record 8053039;
      G_recInterventionJnlLine@1005 : Record 8053034;
      G_InterventionJnlPostBatch@1006 : Codeunit 8052999;
      G_TempJnlBatchName@1007 : Code[10];

    LOCAL PROCEDURE Code@1();
    BEGIN
      WITH G_recInterventionJnlLine DO BEGIN
        G_recInterventionJnlTempl.GET("Journal Template Name");

        IF NOT CONFIRM(Text000,FALSE) THEN
          EXIT;

        G_TempJnlBatchName := "Journal Batch Name";

        G_InterventionJnlPostBatch.RUN(G_recInterventionJnlLine);

        IF "Line No." = 0 THEN
          MESSAGE(Text001)
        ELSE
          IF G_TempJnlBatchName = "Journal Batch Name" THEN
            MESSAGE(Text002)
          ELSE
            MESSAGE(
              Text003,
              "Journal Batch Name");

        IF NOT FIND('=><') OR (G_TempJnlBatchName <> "Journal Batch Name") THEN BEGIN
          RESET;
          FILTERGROUP := 2;
          SETRANGE("Journal Template Name","Journal Template Name");
          SETRANGE("Journal Batch Name","Journal Batch Name");
          FILTERGROUP := 0;
          "Line No." := 1;
        END;
      END;
    END;

    BEGIN
    {
      // <-- 3LI BEGIN

      20110601-Wi00-CMC01.04 : Version RTM 01.04 - NavAgri

      // --> 3LI END
    }
    END.
  }
}
