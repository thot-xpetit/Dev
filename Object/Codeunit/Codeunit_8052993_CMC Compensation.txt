    PROCEDURE CMCApplyVendEntryformEntry@10(VAR P_Rec_ApplyingVendLedgEntry@1000 : Record 25);
    BEGIN
      P_Rec_ApplyingVendLedgEntry.TESTFIELD(P_Rec_ApplyingVendLedgEntry."Document Type",
      P_Rec_ApplyingVendLedgEntry."Document Type"::Invoice);
      IF NOT P_Rec_ApplyingVendLedgEntry.Open THEN
        ERROR(Text8053018);

      IF Vendor.GET(P_Rec_ApplyingVendLedgEntry."Vendor No.") THEN BEGIN
        IF Vendor."Setting-off Allowed"=FALSE THEN IF NOT CONFIRM(Text8053010,FALSE) THEN ERROR(Text8053020);
      END;

      IF P_Rec_ApplyingVendLedgEntry."SettOff Applies-to ID"='' THEN BEGIN
        P_Rec_ApplyingVendLedgEntry."SettOff Applies-to ID":=P_Rec_ApplyingVendLedgEntry."Document No.";
        P_Rec_ApplyingVendLedgEntry.MODIFY;
      END;

      CreateTmpCMCSetOff(P_Rec_ApplyingVendLedgEntry);

      PAGE.RUN(PAGE::"CMC Apply compensation",G_REC_CMCSetOffLedgerEntryTmp);
    END;
