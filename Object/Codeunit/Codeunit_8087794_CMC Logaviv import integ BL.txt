    PROCEDURE ImportBLNav@1000000006(TableBL_P@1000000013 : Record 8053427) Result_L : Text[256];
    VAR
      "<-- 3LI LOCALS -->"@1000000001 : Integer;
      TableBL_L@1000000000 : Record 8053427;
      TableContrat_L@1000000002 : Record 8053423;
      TableReceipt_L@1000000012 : Record 8052995;
      TableReceiptLine_L@1000000004 : Record 8053104;

      Result_L := '';

      // Traiter tous les ordres Logaviv en attente.
      TableBL_L.SETRANGE(HoroIntegration, 0DT);
      TableBL_L.SETRANGE(IdContractInt, TableBL_P.IdContractInt);
      TableBL_L.SETRANGE(IdBlInterne, TableBL_P.IdBlInterne);

      IF TableBL_L.FINDSET THEN BEGIN
        REPEAT
          Erreur_L := FALSE;
          ATraiter_L := FALSE;
          IF NOT TableContrat_L.GET(TableBL_L.IdContractInt) THEN BEGIN
            Erreur_L := TRUE;
            ERROR(COPYSTR(STRSUBSTNO(Text0008, TableBL_L.IdContractInt, TableContrat_L.TABLECAPTION, TableBL_L.IdBlInterne,
              TableBL_L.IdContractExt), 1, MAXSTRLEN(Result_L)));
          END ELSE BEGIN
            Position_L := 1;
            wNo_L := ToolsFile_L.ExtractTextCsv(TableBL_L.IdBlExterne, Position_L, '-', 30);
            IF NOT EVALUATE(w_NoLigne_L, ToolsFile_L.ExtractTextCsv(TableBL_L.IdBlExterne, Position_L, '-', 30)) THEN BEGIN
              ERROR(COPYSTR(STRSUBSTNO(Text0018, TableBL_L.IdContractInt, TableBL_L.IdBlInterne,
                TableBL_L.IdContractExt), 1, MAXSTRLEN(Result_L)));
            END ELSE BEGIN
              IF NOT TableReceiptLine_L.GET(TableReceiptLine_L."Receipt Type"::Receipt, wNo_L, w_NoLigne_L) THEN BEGIN
                ERROR(COPYSTR(STRSUBSTNO(Text0008, TableBL_L.IdContractInt, TableReceiptLine_L.TABLECAPTION,
                  TableBL_L.IdBlInterne, TableBL_L.IdContractExt), 1, MAXSTRLEN(Result_L)));
              END ELSE BEGIN
                IF NOT TableReceipt_L.GET(TableReceiptLine_L."Receipt Type"::Receipt, wNo_L) THEN BEGIN
                  ERROR(COPYSTR(STRSUBSTNO(Text0008, TableBL_L.IdContractInt, TableReceipt_L.TABLECAPTION,
                    TableBL_L.IdBlInterne, TableBL_L.IdContractExt), 1, MAXSTRLEN(Result_L)));
                END ELSE BEGIN
                  IF NOT TableContratNav_L.GET(TableContrat_L.RefNavAgri) THEN BEGIN
                    ERROR(COPYSTR(STRSUBSTNO(Text0008, TableContrat_L.RefNavAgri, TableContratNav_L.TABLECAPTION,
                      TableBL_L.IdBlInterne, TableBL_L.IdContractExt), 1, MAXSTRLEN(Result_L)));
                  END ELSE BEGIN
                    // Si silo vide : le remplir
                    IF TableContratNav_L."Default Location Code" = '' THEN BEGIN
                      TableContratNav_L.Status := TableContratNav_L.Status::Open;
                      IF NOT TableContratNav_L.MODIFY() THEN BEGIN
                        ERROR(COPYSTR(STRSUBSTNO(Text0001, TableContrat_L.RefNavAgri, TableContratNav_L.TABLECAPTION,
                          TableBL_L.IdBlInterne, TableBL_L.IdContractExt), 1, MAXSTRLEN(Result_L)));
                      END;

                      IF STRLEN(TableBL_L.IdSiloExt) < 3 THEN BEGIN
                        // Avant code num'ric sur 3 ==> sans les 0 non significatif.
                        IF EVALUATE(IdSiloExtNum_L, TableBL_L.IdSiloExt) THEN BEGIN
                          TableContratNav_L.VALIDATE("Default Location Code",
                            FORMAT(IdSiloExtNum_L, 0, '<Integer,3><Filler Character,0>'));
                        END ELSE BEGIN
                          TableContratNav_L.VALIDATE("Default Location Code", TableBL_L.IdSiloExt);
                        END;
                      END ELSE BEGIN
                        TableContratNav_L.VALIDATE("Default Location Code", TableBL_L.IdSiloExt);
                      END;
                      IF NOT TableContratNav_L.MODIFY(TRUE) THEN BEGIN
                        ERROR(COPYSTR(STRSUBSTNO(Text0001, TableContrat_L.RefNavAgri, TableContratNav_L.TABLECAPTION,
                          TableBL_L.IdBlInterne, TableBL_L.IdContractExt), 1, MAXSTRLEN(Result_L)));
                      END;
                      TableContratNav_L.Status := TableContratNav_L.Status::Locked;
                      IF NOT TableContratNav_L.MODIFY() THEN BEGIN
                        ERROR(COPYSTR(STRSUBSTNO(Text0001, TableContrat_L.RefNavAgri, TableContratNav_L.TABLECAPTION,
                          TableBL_L.IdBlInterne, TableBL_L.IdContractExt), 1, MAXSTRLEN(Result_L)));
                      END;
                    END;

                    CMCStatusMgt_L.OpenPostedReceipt(TableReceipt_L);

                    TableReceiptLine_L.VALIDATE("Contract No.", TableContrat_L.RefNavAgri);
                    IF NOT TableReceiptLine_L.MODIFY(TRUE) THEN BEGIN
                      ERROR(COPYSTR(STRSUBSTNO(Text0001, TableBL_L.IdBlInterne, TableBL_L.TABLECAPTION, TableBL_L.IdBlInterne),
                        1, MAXSTRLEN(Result_L)));
                    END;
                    // Pour recalculer quantit' affect'e
                    CMCStatusMgt_L.ReleasePostedReceipt(TableReceipt_L);
                    // Toper comme int'gr'
                    TableBL_L.HoroIntegration := CURRENTDATETIME;
                    IF NOT TableBL_L.MODIFY(TRUE) THEN BEGIN
                      ERROR(COPYSTR(STRSUBSTNO(Text0001, TableBL_L.IdBlInterne, TableBL_L.TABLECAPTION, TableBL_L.IdBlInterne),
                        1, MAXSTRLEN(Result_L)));
                    END;
                  END;
                END;
              END;
            END;
          END;
        UNTIL TableBL_L.NEXT = 0;
      END;
