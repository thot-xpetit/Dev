    PROCEDURE RollUpPrice@1480011(VAR MixSalesLine_P@1480000 : Record 60032;RequestType_P@1480006 : 'Manual,Modify,Delete,Insert');
    VAR
      MixSalesLine2_L@1480003 : Record 60032;
      KitLine@1480001 : Record 37;
      Item@1480008 : Record 27;
      GLSetup@1480004 : Record 98;
      CustCheckCreditLimit@1480002 : Codeunit 312;
      UnitPrice@1480005 : Decimal;
    BEGIN
      WITH MixSalesLine_P DO BEGIN
        KitLine.GET("Document Type","Document No.","Document Line No.");
        Item.GET(KitLine."No.");
        IF (RequestType_P IN [RequestType_P::Modify,RequestType_P::Delete,RequestType_P::Insert]) AND
           (Item."Roll-up Kit Pricing" <> Item."Roll-up Kit Pricing"::Always)
        THEN
          EXIT;
        Item.TESTFIELD("Roll-up Kit Pricing");

        IF ("Line No." = 0) OR (RequestType_P = RequestType_P::Insert) THEN
          UnitPrice := "Quantity per" * "Unit Price"
        ELSE
          UnitPrice := 0;
        MixSalesLine2_L.SETRANGE("Document Type","Document Type");
        MixSalesLine2_L.SETRANGE("Document No.","Document No.");
        MixSalesLine2_L.SETRANGE("Document Line No.","Document Line No.");
        IF MixSalesLine2_L.FINDSET THEN
          REPEAT
            IF "Line No." = MixSalesLine2_L."Line No." THEN  // not necessary written to database yet.
              CASE RequestType_P OF
                RequestType_P::Manual:
                  UnitPrice := UnitPrice + MixSalesLine2_L."Quantity per" * MixSalesLine2_L."Unit Price";
                RequestType_P::Modify:
                  UnitPrice := UnitPrice + "Quantity per" * "Unit Price";
              END
            ELSE
              UnitPrice := UnitPrice + MixSalesLine2_L."Quantity per" * MixSalesLine2_L."Unit Price";
          UNTIL MixSalesLine2_L.NEXT = 0;
        GLSetup.GET;
        UnitPrice := ROUND(UnitPrice,GLSetup."Unit-Amount Rounding Precision");
        IF KitLine."Unit Price" <> UnitPrice * KitLine."Qty. per Unit of Measure" THEN BEGIN
          KitLine.VALIDATE("Unit Price",UnitPrice * KitLine."Qty. per Unit of Measure"); // to test status and to set outst. amt.
          IF (RequestType_P <> RequestType_P::Insert) AND
             ("Document Type" <= "Document Type"::Invoice) AND
             ((KitLine."Outstanding Amount" + KitLine."Shipped Not Invoiced") > 0)
          THEN
            CustCheckCreditLimit.SalesLineCheck(KitLine);

          IF "Line No." <> 0 THEN BEGIN
            CASE RequestType_P OF
              RequestType_P::Modify:
                BEGIN
                  MODIFY;
                  KitLine.MODIFY;
                END;
              RequestType_P::Manual,RequestType_P::Insert:
                KitLine.MODIFY;
              RequestType_P::Delete:
                BEGIN
                  LOCKTABLE;
                  KitLine.MODIFY;
                END;
            END;
          END;
        END;
      END;
    END;
