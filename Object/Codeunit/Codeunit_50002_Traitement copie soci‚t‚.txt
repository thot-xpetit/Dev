                  CMC_MC_Masterdata_Relation.SETRANGE(CMC_MC_Masterdata_Relation."MC Company Code",CMC_MC_Company.Code);
                  CMC_MC_Masterdata_Relation.SETRANGE(CMC_MC_Masterdata_Relation.Type,CMC_MC_Masterdata_Relation.Type::"G/L Account");

                    CMC_MC_Masterdata_Relation.INIT;
                    CMC_MC_Masterdata_Relation.VALIDATE(CMC_MC_Masterdata_Relation."MC Company Code",CMC_MC_Company.Code);
                    CMC_MC_Masterdata_Relation.Type              := CMC_MC_Masterdata_Relation.Type::"G/L Account";
                    CMC_MC_Masterdata_Relation.VALIDATE(CMC_MC_Masterdata_Relation."No.",GL_Account."No.");
                    CMC_MC_Masterdata_Relation.INSERT;
                  END;
                UNTIL GL_Account.NEXT = 0;
              END;
