    PROCEDURE GetBOMVersion@1(BOMHeaderNo_P@1000 : Code[20];Date_P@1001 : Date;OnlyCertified_P@1002 : Boolean) : Code[10];
    VAR
      MixBOMVersion_L@1003 : Record 60027;
    BEGIN

      MixBOMVersion_L.SETRANGE("Mix BOM No.",BOMHeaderNo_P);

    END;

    PROCEDURE CopyBOM@60000(MixHeaderNo_P@1000 : Code[20];FromVersionCode_P@1001 : Code[10];CurrentBOMHeader_P@1002 : Record 60025;ToVersionCode_P@1003 : Code[10]);
    VAR
      FromProdBOMComponent_L@1004 : Record 60026;
      ToProdBOMComponent_L@1005 : Record 60026;
      ProdBomVersion_L@1008 : Record 60027;
      ProdBOMHeader_L@1462000 : Record 60025;
    BEGIN
      IF (CurrentBOMHeader_P."No." = MixHeaderNo_P) AND
        (FromVersionCode_P = ToVersionCode_P)
      THEN
        ERROR(Text60000,CurrentBOMHeader_P.TABLECAPTION);

      ProdBOMHeader_L.GET(MixHeaderNo_P);
      IF ProdBOMHeader_L.Type <> CurrentBOMHeader_P.Type THEN
        ERROR(
          Text60003,
          ProdBOMHeader_L.TABLECAPTION,
          ProdBOMHeader_L.FIELDCAPTION(Type));

      IF ToVersionCode_P = '' THEN BEGIN
        IF CurrentBOMHeader_P.Status = CurrentBOMHeader_P.Status::Certified THEN
          ERROR(
            Text60001,
            CurrentBOMHeader_P.FIELDCAPTION(Status),
            CurrentBOMHeader_P.TABLECAPTION,
            CurrentBOMHeader_P."No.",
            CurrentBOMHeader_P.Status);
      END ELSE BEGIN
        ProdBomVersion_L.GET(
          CurrentBOMHeader_P."No.",ToVersionCode_P);
        IF ProdBomVersion_L.Status = ProdBomVersion_L.Status::Certified THEN
          ERROR(
            Text60002,
            ProdBomVersion_L.FIELDCAPTION(Status),
            ProdBomVersion_L.TABLECAPTION,
            ProdBomVersion_L."Mix BOM No.",
            ProdBomVersion_L."Version Code",
            ProdBomVersion_L.Status);
      END;

      ToProdBOMComponent_L.SETRANGE("Mix BOM No.",CurrentBOMHeader_P."No.");
      ToProdBOMComponent_L.SETRANGE("Version Code",ToVersionCode_P);
      ToProdBOMComponent_L.DELETEALL;

      {
      ToProdBOMCompComment.SETRANGE("Production BOM No.",CurrentBOMHeader."No.");
      ToProdBOMCompComment.SETRANGE("Version Code",ToVersionCode);
      ToProdBOMCompComment.DELETEALL;
      }

      FromProdBOMComponent_L.SETRANGE("Mix BOM No.",MixHeaderNo_P);
      FromProdBOMComponent_L.SETRANGE("Version Code",FromVersionCode_P);

      IF FromProdBOMComponent_L.FIND('-') THEN
        REPEAT
          ToProdBOMComponent_L := FromProdBOMComponent_L;
          ToProdBOMComponent_L."Mix BOM No." := CurrentBOMHeader_P."No.";
          ToProdBOMComponent_L."Version Code" := ToVersionCode_P;
          ToProdBOMComponent_L.INSERT;
        UNTIL FromProdBOMComponent_L.NEXT = 0;

      {
      FromProdBOMCompComment_l.SETRANGE("mix BOM No.",BOMHeaderNo);
      FromProdBOMCompComment_l.SETRANGE("Version Code",FromVersionCode);

      IF FromProdBOMCompComment.FIND('-') THEN
        REPEAT
          ToProdBOMCompComment := FromProdBOMCompComment;
          ToProdBOMCompComment."Production BOM No." := CurrentBOMHeader."No.";
          ToProdBOMCompComment."Version Code" := ToVersionCode;
          ToProdBOMCompComment.INSERT;
        UNTIL FromProdBOMCompComment.NEXT = 0;
      }
    END;

    LOCAL PROCEDURE BuildWhereUsedList@60004(Type@1000 : ' ,Item,Production BOM';No@1001 : Code[20];"Calc.Date"@1002 : Date;Level@1003 : Integer;Quantity@1004 : Decimal);
    VAR
      ItemAssembly_L@1005 : Record 27;
      MixBOMComponent@1006 : Record 60026;
    BEGIN
      IF Level > 30 THEN
        EXIT;

      IF Type = Type::"Production BOM" THEN BEGIN
        ItemAssembly_L.SETCURRENTKEY("Mix BOM No.");
        ItemAssembly_L.SETRANGE("Mix BOM No.",No);
        IF ItemAssembly_L.FIND('-') THEN
          REPEAT
            WhereUsedList_G."Entry No." := NextWhereUsedEntryNo_G;
            WhereUsedList_G."Item No." := ItemAssembly_L."No.";
            WhereUsedList_G.Description := ItemAssembly_L.Description;
            WhereUsedList_G."Level Code" := Level;
            WhereUsedList_G."Quantity Needed" :=
              Quantity *
              (1 + ItemAssembly_L."Scrap %" / 100) *
              UOMMgt_G.GetQtyPerUnitOfMeasure(
                ItemAssembly_L,
                ItemAssembly_L."Base Unit of Measure") /
              UOMMgt_G.GetQtyPerUnitOfMeasure(
                ItemAssembly_L,
                GetBOMUnitOfMeasure(
                  ItemAssembly_L."Mix BOM No.",
                  GetBOMVersion(
                    ItemAssembly_L."Mix BOM No.",
                    "Calc.Date",FALSE)));
            // 20140319-wi00-CMC02.02.00-1906-JLT01 : START -->
            // BEFORE
            // WhereUsedList."Version Code" := VersionMgt.GetBOMVersion(No,"Calc.Date",FALSE);
            // AFTER
            WhereUsedList_G."Version Code" := GetBOMVersion(No,"Calc.Date",TRUE);
            // 20140319-wi00-CMC02.02.00-1906-JLT01 : STOP <--
            WhereUsedList_G.INSERT;
            NextWhereUsedEntryNo_G := NextWhereUsedEntryNo_G + 1;
            IF MultiLevel_G THEN
              BuildWhereUsedList(
                1,
                ItemAssembly_L."No.",
                "Calc.Date",
                Level + 1,
                WhereUsedList_G."Quantity Needed");
          UNTIL ItemAssembly_L.NEXT = 0;
      END;


      MixBOMComponent.SETCURRENTKEY(Type,"No.");
      MixBOMComponent.SETRANGE(Type,Type);
      MixBOMComponent.SETRANGE("No.",No);
      IF "Calc.Date" <> 0D THEN BEGIN
        MixBOMComponent.SETFILTER("Starting Date",'%1|..%2',0D,"Calc.Date");
        MixBOMComponent.SETFILTER("Ending Date",'%1|%2..',0D,"Calc.Date");
      END;

      IF MixBOMComponent.FIND('-') THEN
        REPEAT
          IF GetBOMVersion(
               MixBOMComponent."Mix BOM No.","Calc.Date",TRUE) =
               MixBOMComponent."Version Code"
          THEN BEGIN
            MixBOMHeader_G.GET(MixBOMComponent."Mix BOM No.");
            BuildWhereUsedList(
              2,
              MixBOMComponent."Mix BOM No.",
              "Calc.Date",
              Level,
              Quantity);
          END;
        UNTIL MixBOMComponent.NEXT = 0;
    END;
