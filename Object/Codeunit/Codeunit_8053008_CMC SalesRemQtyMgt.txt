    PROCEDURE FindBestVariant@8053008(VAR ItemVariant_P@8053000 : Record 5401;WheShipmentLine_P@8053002 : Record 7321;VAR VariantFoundCode_P@8053006 : Code[10];VAR VariantInventory_P@8053005 : Decimal) : Boolean;

          IF ((Item_L.Inventory - Item_L."Qty on Warehouse Shipment" - VariantDetail_L.Quantity) > 0) AND
             ((ItemVarIndQty_L > VariantIndivQty_L) OR
             //--> 20140723-APP30-TER01.01.00-PDC01
             //BEFORE
             //((Item_L.Inventory-Item_L."Qty on Warehouse Shipment"-VariantDetail_L.Quantity)<VariantInventory_P)) THEN
             //AFTER
             ((Item_L.Inventory-Item_L."Qty on Warehouse Shipment"-VariantDetail_L.Quantity)>VariantInventory_P))
             //<-- 20140723-APP30-TER01.01.00-PDC01
          AFTER  }
          IF RecCMCSetup_L.GET THEN;
          CASE RecCMCSetup_L."Variant Stock alloc. Priority" OF

             ((ItemVarIndQty_L > VariantIndivQty_L) OR QuantityOk_L )
          // --> 20140910-APP30-TER01.01.00-FBE02

            VariantIndivQty_L := ItemVarIndQty_L;
          END;
          IF (VariantInventory_P = 0) AND ((Item_L.Inventory - Item_L."Qty on Warehouse Shipment" - VariantDetail_L.Quantity) > 0 )
          THEN BEGIN
            VariantInventory_P :=Item_L.Inventory-Item_L."Qty on Warehouse Shipment"-VariantDetail_L.Quantity;

          END;
        UNTIL ItemVariant_P.NEXT=0;

    PROCEDURE ReturnNumber@1000000001(Quantity_P@1000000000 : Decimal;DivQty_P@1000000001 : Decimal;Direction_P@1000000002 : 'More,Less') Result_L : Decimal;
    VAR
      MaxIteraction_L@1000000003 : Integer;
    BEGIN
      //20140710-APP14-TER01.01.00-RMZ01
      // --------------------------------------------------------------------------------------------------------------------------
      // To return the next divisible number
      // --------------------------------------------------------------------------------------------------------------------------
      // Inputs
      //    - Quantity_P - quantity of order line
      //    - DivQty_P - quantity divisable
      // Outputs
      //    - Result_L - quantity of the next divisible number
      IF (Quantity_P = 0) OR (DivQty_P = 0) THEN
        Result_L := Quantity_P
      ELSE BEGIN
        IF Quantity_P MOD DivQty_P = 0 THEN
          Result_L := Quantity_P
        ELSE BEGIN
          MaxIteraction_L := 0;
          Result_L := Quantity_P;
          WHILE (Result_L MOD DivQty_P <> 0) OR (MaxIteraction_L >= 100) DO BEGIN
            IF Direction_P = Direction_P::More THEN
              Result_L := Result_L + 1
            ELSE
              Result_L := Result_L - 1;
            MaxIteraction_L := MaxIteraction_L + 1;
          END;
        END;
      END;

      EXIT(Result_L);
    END;
