      END;

      PeriodStart_L := PeriodStart_G;
      PeriodEnd_L   := PeriodEnd_G;

      IF FileMgt_G.LogErreurType() <> 'E' THEN BEGIN
        CLEAR(Cust_L);
        IF CMCAccoutingCenter_G <> '' THEN
          Cust_L.SETRANGE("Customer Code Center",CMCAccoutingCenter_G);
        IF NOT Cust_L.ISEMPTY THEN BEGIN
          Cust_L.FINDSET;

            //-->20150925-INT40/001-CMC02.02.00-VMR01

            }
            //AFTER
            //<--20150925-INT40/001-CMC02.02.00-VMR01
            CustLedgerEntry_L.SETRANGE("Posting Date", PeriodStart_L, PeriodEnd_L);
            //CustLedgerEntry_L.SETRANGE("Export CER",FALSE);
            IF NOT CustLedgerEntry_L.ISEMPTY THEN BEGIN
              CustLedgerEntry_L.FINDSET;
              REPEAT
                IF NOT DataNotIsEmpty(CustLedgerEntry_L) THEN BEGIN
                  IF NOT IsFileCreate_L THEN

                  GetDataHeader(CustLedgerEntry_L, FileOutstream_L, Line_L);
                END;
              UNTIL CustLedgerEntry_L.NEXT = 0;
            END;
            //FOURNISSEUR
            //-->20151019-INT40/001-CMC02.02.00-VMR0
            //BEFORE
            {
            IF Contact_L.GET(Cust_L."Primary Contact No.") THEN BEGIN
              IF  Contact_L."Vendor No."<>'' THEN  BEGIN
            }
            //AFTER
            Vendor_L.SETRANGE("Customer No.",Cust_L."No.");
            IF Vendor_L.FINDFIRST THEN BEGIN

                CLEAR(VendLedgerEntry_L);
                VendLedgerEntry_L.SETCURRENTKEY("Document Type","Vendor No.","Posting Date");
                //-->20151016-INT40/001-CMC02.02.00-VMR01
                //BEFORE
                //VendLedgerEntry_L.SETRANGE("Vendor No.",Contact_L."Vendor No.");
                //AFTER
                VendLedgerEntry_L.SETRANGE("Vendor No.",Vendor_L."No.");
                //<--20151019-INT40/001-CMC02.02.00-VMR01

                //-->20150925-INT40/001-CMC02.02.00-VMR01
                //BEFORE
                {
                VendLedgerEntry_L.SETFILTER("Document Type",'%1|%2|%3', VendLedgerEntry_L."Document Type"::Invoice,
                  VendLedgerEntry_L."Document Type"::"Credit Memo", VendLedgerEntry_L."Document Type"::"Finance Charge Memo");
                }
                //AFTER
                //<--20159025-INT40/001-CMC02.02.00-VMR01
                VendLedgerEntry_L.SETRANGE("Posting Date", PeriodStart_L, PeriodEnd_L);
                //VendLedgerEntry_L.SETRANGE("Export CER",FALSE);
                IF NOT VendLedgerEntry_L.ISEMPTY THEN BEGIN
                  VendLedgerEntry_L.FINDSET;
                  REPEAT
                    IF NOT DataNotIsEmptyVendor(VendLedgerEntry_L) THEN BEGIN
                      IF NOT IsFileCreate_L THEN

                      IF NOT IsUNB_L THEN

                      IF NOT IsUNH_L THEN

                    END;
                  UNTIL VendLedgerEntry_L.NEXT = 0;
                END;
              //END;
            END;
            //FOURNISSEURS
          UNTIL Cust_L.NEXT = 0;

          // UNT M 1 Fin de message
          // Fonction : Terminer et v'rifier l'achSvement d'un message.
          //----------------------------------------------------------------------------------------------------------------------
          Line_L := FormatLine('UNT',0,' ',Pos_G::Left,'+');

    PROCEDURE FormatLine@1120003(Data_P@1120004 : Variant;FieldLen_P@1120003 : Integer;PadChar_P@1120002 : Char;"Left/Right_P"@1120001 : 'Right,Left';EndOfText_P@1120000 : Char) : Text[250];
    VAR
      Data_L@1120008 : Text[250];
      Line_L@1120007 : Text[250];
      I_L@1120006 : Integer;
      NUL_L@1120005 : Char;
    BEGIN
      {ParamStres:
       Data_P       : Data
       FieldLen_P   : Longueur du champ  (Si=0, alors on prend la longueur de Data_P)
       PadChar_P    : CaractSre ajout'
       Left/Right_P : Cadrage  Gauche/Droit
       EndOfText_P  : CaractSre fin de champ}

      // Init
      //----------------------------------------------------------------------------------------------------------------------------------
      NUL_L  := 0;
      Data_L := FORMAT(Data_P);

      IF FieldLen_P = 0 THEN
        FieldLen_P := STRLEN(Data_L);

      CASE EndOfText_P OF
        // No character end of text
        //------------------------------------------------------------------------------------------------------------------------------
        NUL_L:
        BEGIN

        // same lenght
        //------------------------------------------------------------------------------------------------------------------------------
          IF STRLEN(Data_L) = FieldLen_P THEN EXIT (Data_L);

          // Length higher
      //------------------------------------------------------------------------------------------------------------------------------
          IF STRLEN(Data_L) > FieldLen_P THEN BEGIN
            CASE "Left/Right_P" OF
              "Left/Right_P"::Right:
                EXIT(TruncStrRight(Data_L,FieldLen_P));
              "Left/Right_P"::Left:
                EXIT(TruncStrLeft(Data_L, FieldLen_P));
            END;
          END;

          // Length lower
      //------------------------------------------------------------------------------------------------------------------------------
          IF STRLEN(Data_L) < FieldLen_P THEN BEGIN
            CASE "Left/Right_P" OF
              "Left/Right_P"::Right:
                EXIT(PadStrRight(Data_L, FieldLen_P, PadChar_P));
              "Left/Right_P"::Left:
                EXIT(PadStrLeft(Data_L, FieldLen_P, PadChar_P));
            END;
          END;
        END;
        // No character end of text
        //------------------------------------------------------------------------------------------------------------------------------
        ELSE BEGIN
          // same lenght
      //------------------------------------------------------------------------------------------------------------------------------
          IF STRLEN(Data_L) = FieldLen_P THEN
            EXIT (Data_L+FORMAT(EndOfText_P));

          // Length higher
      //------------------------------------------------------------------------------------------------------------------------------
          IF STRLEN(Data_L) > FieldLen_P THEN BEGIN
            CASE "Left/Right_P" OF
              "Left/Right_P"::Right:
                EXIT (TruncStrRight(Data_L, FieldLen_P)+FORMAT(EndOfText_P));
              "Left/Right_P"::Left:
                EXIT (TruncStrRight(Data_L, FieldLen_P)+FORMAT(EndOfText_P));
            END;
          END;

          // Length lower
      //------------------------------------------------------------------------------------------------------------------------------
          IF STRLEN(Data_L) < FieldLen_P THEN BEGIN
            CASE "Left/Right_P" OF
              "Left/Right_P"::Right:
                EXIT (PadStrRight(Data_L, FieldLen_P, PadChar_P)+FORMAT(EndOfText_P));
              "Left/Right_P"::Left:
                EXIT (PadStrLeft(Data_L, FieldLen_P, PadChar_P)+FORMAT(EndOfText_P));
            END;
          END;
        END;
      END;
    END;
