    Heure=18:33:05;
    Modifi'=Oui;

  }
  PROPERTIES
  {
    OnRun=VAR
            DateFin_L@1000000000 : Date;
            DateFinNouvelle_L@1000000001 : Date;
          BEGIN

            //===============================================================================================================================
            //                      Mises . jour ALITMENT
            //===============================================================================================================================

            Cpt  := 0;
            Cpt2 := 0;
            Article_G.RESET;
            Total := Article_G.COUNT;
            Fenetre.OPEN('Maj des groupes compta          \' +
                         'Nb lecture                      #1######\' +
                         'Nb flux modifi'                 #2######\' +
                         '@4@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@',Cpt,Cpt2);

            //FILTE TEMPORAIRE SUR 1 ARTICLE
            //Article_G.SETFILTER(Article_G."No.",'26846');
            Article_G.SETFILTER(Article_G."Activity Code", 'ALI');

            IF Article_G.FIND('-') THEN
            REPEAT

            PrixVente_G.RESET;
            PrixVente_G.SETFILTER(PrixVente_G."Sales Type",'Tous les clients');
            PrixVente_G.SETFILTER(PrixVente_G."Item No.",Article_G."No.");

            IF PrixVente_G.FIND('+') THEN BEGIN
               EVALUATE(DateFin_L,'31/01/2017');

              IF PrixVente_G."Ending Date" = DateFin_L  THEN BEGIN
                  EVALUATE(DateFinNouvelle_L,'30/05/2017');
                  PrixVente_G.VALIDATE("Ending Date",DateFinNouvelle_L);
                  PrixVente_G.MODIFY;

                Cpt2 := Cpt2 + 1;
              END;
            END;

            Cpt := Cpt + 1;
            Fenetre.UPDATE;
            Fenetre.UPDATE(1,Cpt);
            Fenetre.UPDATE(2,Cpt2);
            Fenetre.UPDATE(4,ROUND(Cpt / Total * 10000, 1));

            UNTIL Article_G.NEXT = 0;
            Fenetre.CLOSE;
          END;

  }
