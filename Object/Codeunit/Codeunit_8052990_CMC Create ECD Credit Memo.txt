    PROCEDURE CalcTotalQuantity@8053000(VAR P_ValueEntry@8053002 : Record 5802) L_TotalQuantityToAssign : Decimal;
    VAR
      "<< 3LI LOCALS >>"@8053001 : Integer;
      L_ValueEntryTemp@8053000 : TEMPORARY Record 5802;
      L_OldVendoNo@8053003 : Code[20];
    BEGIN
      // ---------------------------------------------------------------------------------------------------------------------------------
      // Calcul de la quantit' totale des 'critures client.
      // ---------------------------------------------------------------------------------------------------------------------------------
      // Entree :
      //  P_ValueEntry : Ecritures valeur . sommer
      // Sortie :
      //  L_TotalQuantityToAssign : Quantit' totale factur'e
      // ---------------------------------------------------------------------------------------------------------------------------------
      SetRecordsToTemp(P_ValueEntry , L_ValueEntryTemp);
      L_TotalQuantityToAssign := 0;
      L_ValueEntryTemp.SETCURRENTKEY("Source Type","Source No.","Item No.","Posting Date","Entry Type",Adjustment);
      IF L_ValueEntryTemp.FINDSET THEN
        REPEAT
          G_AmountToAss:=G_AmountToAss+(L_ValueEntryTemp."Invoiced Quantity" *L_ValueEntryTemp."Unit ECD Porposed");
          L_OldVendoNo:=L_ValueEntryTemp."Source No.";
        UNTIL L_ValueEntryTemp.NEXT=0;
      L_ValueEntryTemp.CALCSUMS("Invoiced Quantity");
      L_TotalQuantityToAssign := L_ValueEntryTemp."Invoiced Quantity";
      EXIT(L_TotalQuantityToAssign);
    END;

    PROCEDURE CheckSelectionValidity@8053005(VAR ValueEntry_P@8053000 : Record 5802);
    VAR
      "<< 3LI LOCALS >>"@8053003 : Integer;
      L_ValueEntryTemp@8053002 : TEMPORARY Record 5802;
      L_TotalQuantity@8053001 : Decimal;
      L_ValueEntry@8053004 : Record 5802;
      L_ValueEntryTemp2@8053005 : TEMPORARY Record 5802;
      L_UnitAmount@8053006 : Decimal;
      L_CMCSetup@8053007 : Record 8052982;
    BEGIN
      // ---------------------------------------------------------------------------------------------------------------------------------
      // teste les enregistrements pour connaitre la validit' de ce qui a 't' s'lectionn'.
      // retourne une erreur dans le cas d'une incoh'rence.
      // ---------------------------------------------------------------------------------------------------------------------------------
      // Entrees :
      //  - ValueEntry_P : Ecritures valeur . tester
      // Sorties :
      //  - aucune
      // ---------------------------------------------------------------------------------------------------------------------------------
      CLEAR(L_ValueEntryTemp);
      CLEAR(L_ValueEntryTemp2);

      L_CMCSetup.GET;
      L_CMCSetup.TESTFIELD("ECD Cr.Memo Item Charge");

      SetRecordsToTemp(ValueEntry_P,L_ValueEntryTemp);

      IF NOT L_ValueEntryTemp.FINDFIRST THEN
        ERROR(Text8053004);

      L_ValueEntryTemp.RESET;
      L_ValueEntryTemp.SETFILTER("Source No.",'<>%1',ValueEntry_P."Source No.");
      IF L_ValueEntryTemp.FINDFIRST THEN
        ERROR(Text8053001);

      L_ValueEntryTemp.RESET;
      L_ValueEntryTemp.SETFILTER("Item No.",'<>%1',ValueEntry_P."Item No.");
      IF L_ValueEntryTemp.FINDFIRST THEN
        ERROR(Text8053002);

      IF G_CreditMemoAmount = 0 THEN
        ERROR(Text8053003);

      SetRecordsToTemp(ValueEntry_P,L_ValueEntryTemp2);

      L_ValueEntryTemp.RESET;
      L_ValueEntryTemp.SETCURRENTKEY("Item Ledger Entry No.");
      L_ValueEntryTemp.FIND('-');
      REPEAT
        L_ValueEntry.RESET;
        L_ValueEntry.SETCURRENTKEY("Item Ledger Entry No.");
        L_ValueEntry.SETRANGE("Item Ledger Entry No.",L_ValueEntryTemp."Item Ledger Entry No.");
        L_ValueEntry.SETRANGE("Item Ledger Entry Type",L_ValueEntry."Item Ledger Entry Type"::Purchase);
        L_ValueEntry.SETRANGE("Entry Type",L_ValueEntry."Entry Type"::"Direct Cost");
        L_ValueEntry.SETRANGE("Expected Cost",FALSE);
        L_ValueEntry.SETRANGE("Document Type",L_ValueEntry."Document Type"::"Purchase Invoice");
        L_ValueEntry.SETFILTER("Entry No.",'<>%1',L_ValueEntryTemp."Entry No.");
        IF L_ValueEntry.FINDSET THEN
          REPEAT
            IF NOT L_ValueEntryTemp2.GET(L_ValueEntry."Entry No.") THEN
              ERROR(STRSUBSTNO(Text8053006,L_ValueEntry."Entry No.",L_ValueEntry."Document Type",L_ValueEntry."Document No."));
          UNTIL L_ValueEntry.NEXT = 0;
      UNTIL L_ValueEntryTemp.NEXT=0;
      CLEAR(L_ValueEntryTemp2);

      L_TotalQuantity := CalcTotalQuantity(ValueEntry_P);
      L_UnitAmount := G_CreditMemoAmount / L_TotalQuantity;

      L_ValueEntryTemp.RESET;
      L_ValueEntryTemp.SETCURRENTKEY("Item Ledger Entry No.");
      L_ValueEntryTemp.FIND('-');
      REPEAT
        L_ValueEntryTemp.VALIDATE("Outstanding ECD", L_ValueEntryTemp."Outstanding ECD" - L_UnitAmount);
        L_ValueEntryTemp.VALIDATE("Invoiced Cost ECD", L_ValueEntryTemp."Invoiced Cost ECD" - L_UnitAmount);
      UNTIL L_ValueEntryTemp.NEXT=0;
    END;

    PROCEDURE CreateCreditMemo@8053006(VAR ValueEntry_P@8053000 : Record 5802);
    VAR
      "<< 3LI LOCALS >>"@8053004 : Integer;
      L_ValueEntryTemp@8053003 : TEMPORARY Record 5802;
      L_ValueEntryTemp2@8053013 : TEMPORARY Record 5802;
      L_TotalQuantity@8053002 : Decimal;
      L_UnitAmount@8053001 : Decimal;
      L_PurchaseHeader@8053005 : Record 38;
      L_PurchaseLine@8053006 : Record 39;
      L_PurchItemChargeAssignment@8053007 : Record 5805;
      L_CMCSetup@8053008 : Record 8052982;
      L_ItemLedgerEntry@8053009 : Record 32;
      L_LineNo@8053010 : Integer;
      L_LinePurchNo@8053011 : Integer;
      L_Item@8053012 : Record 27;
    BEGIN
      // ---------------------------------------------------------------------------------------------------------------------------------
      // g'nSre l'avoir de RFC et le valide.
      // ---------------------------------------------------------------------------------------------------------------------------------
      // Entrees :
      //  - ValueEntry_P : Ecritures valeur . partir desquelles la RFC sera calcul'e
      // Sorties :
      //  - aucune
      // ---------------------------------------------------------------------------------------------------------------------------------

      SetRecordsToTemp(ValueEntry_P,L_ValueEntryTemp);

      SetRecordsToTemp2(ValueEntry_P,L_ValueEntryTemp2);

      L_PurchaseHeader.INIT;
      L_PurchaseHeader."Document Type" := L_PurchaseHeader."Document Type"::"Credit Memo";
      L_PurchaseHeader."No." := '';
      L_PurchaseHeader.INSERT(TRUE);
      L_PurchaseHeader.VALIDATE("Buy-from Vendor No.",G_VendorNo);
      L_PurchaseHeader.VALIDATE(ECD , TRUE);
      L_PurchaseHeader.VALIDATE("Posting Date",G_PostingDate);
      L_PurchaseHeader.VALIDATE("Document Date" ,G_DocDate);
      L_PurchaseHeader.VALIDATE("Posting Description", Text8053010);
      // --> 20260224-WI00-ADV01.01.00-42635-IJI
      L_PurchaseHeader."Send to Approved Platform" := L_PurchaseHeader."Send to Approved Platform"::Yes;
      // <-- 20260224-WI00-ADV01.01.00-42635-IJI
      L_PurchaseHeader.MODIFY(TRUE);

      L_LinePurchNo:=10000;
      L_CMCSetup.GET;

      L_ValueEntryTemp2.RESET;
      IF L_ValueEntryTemp2.FINDSET THEN
        REPEAT
          L_Item.GET(L_ValueEntryTemp2."Item No.");
          L_PurchaseLine.INIT;
          L_PurchaseLine."Document Type" := L_PurchaseLine."Document Type"::"Credit Memo";
          L_PurchaseLine.VALIDATE("Document No.", L_PurchaseHeader."No.");
          L_PurchaseLine.VALIDATE("Line No.",L_LinePurchNo);
          L_PurchaseLine.INSERT(TRUE);
          L_PurchaseLine.VALIDATE(Type,L_PurchaseLine.Type::"Charge (Item)");
          L_PurchaseLine.VALIDATE("No.",L_CMCSetup."ECD Cr.Memo Item Charge");
          L_PurchaseLine.VALIDATE(Description,
                           COPYSTR(STRSUBSTNO(Text8053007, 'RCF',L_Item.Description),
                                                    1,MAXSTRLEN(L_PurchaseLine.Description)));
          L_PurchaseLine.VALIDATE(Quantity,L_ValueEntryTemp2."Invoiced Quantity");
          L_PurchaseLine.VALIDATE("VAT Prod. Posting Group",L_Item."VAT Prod. Posting Group");
          L_PurchaseLine.VALIDATE("Direct Unit Cost",L_ValueEntryTemp2."Unit ECD Porposed"/L_PurchaseLine.Quantity);
          L_PurchaseLine.MODIFY(TRUE);
          L_LinePurchNo:=L_LinePurchNo+10000;


          L_ValueEntryTemp.SETCURRENTKEY("Item Ledger Entry No.");
          L_ValueEntryTemp.SETRANGE("Item No." ,L_ValueEntryTemp2."Item No.");
          L_ValueEntryTemp.FINDSET;
          REPEAT

            L_ItemLedgerEntry.GET(L_ValueEntryTemp."Item Ledger Entry No.");
            L_PurchItemChargeAssignment.RESET;
            L_PurchItemChargeAssignment.SETRANGE("Document Type",L_PurchaseLine."Document Type");
            L_PurchItemChargeAssignment.SETRANGE("Document No.",L_PurchaseLine."Document No.");
            L_PurchItemChargeAssignment.SETRANGE("Document Line No.",L_PurchaseLine."Line No.");
            L_LineNo := 10000;
            IF L_PurchItemChargeAssignment.FINDLAST THEN
              L_LineNo := L_PurchItemChargeAssignment."Line No." + 10000;

            L_PurchItemChargeAssignment.INIT;
            L_PurchItemChargeAssignment."Document Type" := L_PurchaseLine."Document Type";
            L_PurchItemChargeAssignment."Document No." := L_PurchaseLine."Document No.";
            L_PurchItemChargeAssignment."Document Line No." := L_PurchaseLine."Line No.";
            L_PurchItemChargeAssignment."Line No." := L_LineNo;
            L_PurchItemChargeAssignment."Applies-to Doc. Type" := GetDocType(L_ItemLedgerEntry."Document Type");
            L_PurchItemChargeAssignment."Applies-to Doc. No." := L_ItemLedgerEntry."Document No.";
            L_PurchItemChargeAssignment."Applies-to Doc. Line No." := L_ItemLedgerEntry."Document Line No.";
            L_PurchItemChargeAssignment.VALIDATE("Item Charge No.", L_CMCSetup."ECD Cr.Memo Item Charge");
            L_PurchItemChargeAssignment.VALIDATE("Item No.", L_ItemLedgerEntry."Item No.");
            L_PurchItemChargeAssignment.VALIDATE("Unit Cost",L_PurchaseLine."Unit Cost (LCY)");
            L_PurchItemChargeAssignment."ECD Invoice Entry":=L_ValueEntryTemp."Entry No.";
            L_PurchItemChargeAssignment.INSERT(TRUE);

            L_PurchItemChargeAssignment.VALIDATE("Qty. to Assign", L_PurchItemChargeAssignment."Qty. to Assign" +
               L_ValueEntryTemp."Valued Quantity");
            L_PurchItemChargeAssignment."Amount to Assign" := L_PurchItemChargeAssignment."Qty. to Assign" *
              L_ValueEntryTemp."Unit ECD Porposed";
            L_PurchItemChargeAssignment.MODIFY(TRUE);
          UNTIL L_ValueEntryTemp.NEXT =0;
        UNTIL L_ValueEntryTemp2.NEXT=0;
    END;

    PROCEDURE SetRecordsToTemp2@8053012(VAR P_ValueEntry@8053000 : Record 5802;VAR ValueEntryTemp_P@8053001 : TEMPORARY Record 5802);
    BEGIN
      // ---------------------------------------------------------------------------------------------------------------------------------
      // Copie les enregistrements en table temporaire pour traitement pour regroupement article
      // ---------------------------------------------------------------------------------------------------------------------------------
      // Entrees :
      //  - P_ValueEntry : Ecritures valeur . copier
      //  - ValueEntryTemp_P : Table temporaire . remplir
      // Sorties :
      //  - aucune
      // ---------------------------------------------------------------------------------------------------------------------------------
      ValueEntryTemp_P.DELETEALL;

      IF P_ValueEntry.FINDSET(FALSE,FALSE) THEN
        REPEAT
          ValueEntryTemp_P.SETRANGE("Item No.",P_ValueEntry."Item No.");
          IF NOT ValueEntryTemp_P.FINDFIRST THEN BEGIN
            ValueEntryTemp_P.RESET;
            ValueEntryTemp_P := P_ValueEntry;
            ValueEntryTemp_P."Unit ECD Porposed":=ValueEntryTemp_P."Unit ECD Porposed"*P_ValueEntry."Invoiced Quantity";
            ValueEntryTemp_P.INSERT;
          END ELSE BEGIN
            ValueEntryTemp_P."Invoiced Quantity":=ValueEntryTemp_P."Invoiced Quantity"+P_ValueEntry."Invoiced Quantity";
            ValueEntryTemp_P."Unit ECD Porposed":=ValueEntryTemp_P."Unit ECD Porposed"+
              (P_ValueEntry."Unit ECD Porposed"*P_ValueEntry."Invoiced Quantity");
            ValueEntryTemp_P.MODIFY;
          END;
        UNTIL P_ValueEntry.NEXT = 0;
    END;
