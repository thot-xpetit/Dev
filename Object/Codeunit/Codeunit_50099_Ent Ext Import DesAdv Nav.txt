      //dans le cas d'un "AVIS EXPEDITION AGRICULTEUR" et dans le cas d'une commande vente

      IF TableExped_P."Flag Pack" THEN

      BooUseLotFictif_L := FALSE;
      QteTotaleLot_L := 0;

      TableTraca_L.RESET;
      TableTraca_L.SETRANGE("No Exped",TableExped_P.No);
      TableTraca_L.SETRANGE(HoroIntegration,0DT);
      IF TableTraca_L.FINDSET THEN REPEAT
        QteTotaleLot_L := QteTotaleLot_L + TableTraca_L."Expedition quantity";
      UNTIL TableTraca_L.NEXT = 0;

        // Si param'trage : "TraŒabilit' uniquement si param'trage" : V'rifier que le code traŒabilit' de l'article et sur lot ou date de

        IF RecPhytoInterfaceSetup_L."Use Default Lot SICAPA" THEN
