    PROCEDURE UpdateProgBalance@1000000002(VAR CVLedgerEntry_P@1000000000 : Record 8053380);
    VAR
      ProgBalance_L@1000000001 : Decimal;
      ProgAmount_L@1000000002 : Decimal;
      Contact_L@1000000003 : Record 5050;
      CMCProgBal_L@1000000004 : Record 8053385;
    BEGIN
      WITH CVLedgerEntry_P DO BEGIN
        IF NOT Contact_L.GET(GETFILTER("Contact No.")) THEN
          EXIT;
        ProgBalance_L := 0;
        ProgAmount_L := 0;
        IF FIND('-') THEN
          REPEAT
            CVLedgerEntry_P.CALCFIELDS("Remaining Amt. (LCY)");
            CVLedgerEntry_P.CALCFIELDS("Amount (LCY)");
            ProgBalance_L := ProgBalance_L + CVLedgerEntry_P."Remaining Amt. (LCY)";
            ProgAmount_L := ProgAmount_L + CVLedgerEntry_P."Amount (LCY)";
            IF NOT CMCProgBal_L.GET(CMCProgBal_L."Entry Type"::"CV Entry",CVLedgerEntry_P."Entry No.") THEN BEGIN
              CMCProgBal_L.INIT;
              CMCProgBal_L."Entry Type" := CMCProgBal_L."Entry Type"::"CV Entry";
              CMCProgBal_L."Entry No." := CVLedgerEntry_P."Entry No.";
              CMCProgBal_L.INSERT;
            END;
            CMCProgBal_L.Amount := ProgAmount_L;
            CMCProgBal_L."Remaining Amount" := ProgBalance_L;
            CMCProgBal_L.MODIFY;
          UNTIL NEXT = 0;
      END;
    END;

    PROCEDURE RefreshCVLetter2@8053012(ContactNo_P@8053001 : Code[20]);
    VAR
      CVLedgerEntry_L@8053000 : Record 8053380;
      recLedgerEntryTmp_L@1000000005 : ARRAY [6] OF TEMPORARY Record 8053380;
      Letter_L@8053004 : Text[30];
      recCustLedgerEntry_L@1000000002 : Record 21;
      recVendorLedgerEntry_L@1000000001 : Record 25;
    BEGIN
      //Neutralisation
      EXIT;

      CVLedgerEntry_L.RESET;
      CVLedgerEntry_L.SETCURRENTKEY("Contact No.","Posting Date");
      CVLedgerEntry_L.SETRANGE("Contact No.",ContactNo_P);
      CVLedgerEntry_L.MODIFYALL("CV Entry Letter", '');

      IF CVLedgerEntry_L.FINDSET(FALSE,FALSE) THEN BEGIN
        REPEAT
          recLedgerEntryTmp_L[1] := CVLedgerEntry_L;
          CASE CVLedgerEntry_L."Customer / Vendor" OF
            CVLedgerEntry_L."Customer / Vendor"::Customer:
            BEGIN
              recCustLedgerEntry_L.GET(CVLedgerEntry_L."Entry No.");
              recLedgerEntryTmp_L[1].Letter:=recCustLedgerEntry_L.Letter;
            END;
            CVLedgerEntry_L."Customer / Vendor"::Vendor:
            BEGIN
              recVendorLedgerEntry_L.GET(CVLedgerEntry_L."Entry No.");
              recLedgerEntryTmp_L[1].Letter:=recVendorLedgerEntry_L.Letter;
            END;
          END;
          recLedgerEntryTmp_L[1].INSERT;
        UNTIL CVLedgerEntry_L.NEXT = 0;
      END;

      Letter_L:='AAA';

      recLedgerEntryTmp_L[1].RESET;
      recLedgerEntryTmp_L[1].SETCURRENTKEY("Customer / Vendor","Cust./Vend. No.",Letter);
      recLedgerEntryTmp_L[1].SETRANGE("Customer / Vendor",recLedgerEntryTmp_L[1]."Customer / Vendor"::Customer);
      recLedgerEntryTmp_L[1].SETRANGE("Cust./Vend. No.",ContactNo_P);
      recLedgerEntryTmp_L[1].SETFILTER(Letter,'<>%1','');
      IF recLedgerEntryTmp_L[1].FINDSET(FALSE,FALSE) THEN
        REPEAT
          IF recLedgerEntryTmp_L[1]."CV Entry Letter"='' THEN BEGIN
            recLedgerEntryTmp_L[1].SETRANGE(Letter,recLedgerEntryTmp_L[1].Letter);
            recLedgerEntryTmp_L[1].SETRANGE("Setting Off",TRUE);
            IF recLedgerEntryTmp_L[1].ISEMPTY THEN BEGIN // pas de compensation
              recLedgerEntryTmp_L[1].SETRANGE("Setting Off");
              recLedgerEntryTmp_L[1].MODIFYALL("CV Entry Letter",Letter_L);
              recLedgerEntryTmp_L[1].FINDLAST;
              recLedgerEntryTmp_L[1].SETFILTER(Letter,'<>%1','');
            END ELSE BEGIN // compensation exsitante
              recLedgerEntryTmp_L[1].FINDSET(FALSE,FALSE);
              REPEAT  // pour chaque ecr compens client
                recLedgerEntryTmp_L[2].RESET; // on cherche la compensation four
                recLedgerEntryTmp_L[2].SETRANGE("Customer / Vendor",recLedgerEntryTmp_L[2]."Customer / Vendor"::Vendor);
                recLedgerEntryTmp_L[2].SETRANGE("Cust./Vend. No.",ContactNo_P);
                recLedgerEntryTmp_L[2].SETRANGE("Transaction No.",recLedgerEntryTmp_L[1]."Transaction No.");
                recLedgerEntryTmp_L[2].SETRANGE("Setting Off" ,TRUE);
                IF recLedgerEntryTmp_L[2].FINDSET(FALSE,FALSE) THEN // il existe des ecr compensation fournisseur
                  REPEAT   // maj des 'critures lettr'es avec la compensation four
                    recLedgerEntryTmp_L[3].SETCURRENTKEY("Customer / Vendor","Cust./Vend. No.",Letter);
                    recLedgerEntryTmp_L[3].SETRANGE("Customer / Vendor",recLedgerEntryTmp_L[3]."Customer / Vendor"::Vendor);
