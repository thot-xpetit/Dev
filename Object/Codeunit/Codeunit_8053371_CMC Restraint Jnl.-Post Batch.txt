    LOCAL PROCEDURE Code@7();
    VAR
      CapJnlBatch2_G@8053000 : Record 8053355;
      TempCapJnlLine_G@8053001 : TEMPORARY Record 8053356;
      TempCapJnlLine2_G@1011 : Record 8053356;
    BEGIN
      CLEAR(CapJnlCheckLine_G);
      CLEAR(CapJnlPostLine_G);
      WITH CapJnlLine_G DO
       BEGIN
        SETRANGE("Capital Journal Template","Capital Journal Template");
        SETRANGE("Capital Batch Name","Capital Batch Name");
        IF RECORDLEVELLOCKING THEN
          LOCKTABLE;
        CapJnlTemplate_G.GET(CapJnlLine_G."Capital Journal Template");
        CapJnlBatch_G.GET("Capital Journal Template","Capital Batch Name");
        IF STRLEN(INCSTR(CapJnlBatch_G.Name)) > MAXSTRLEN(CapJnlBatch_G.Name) THEN
          CapJnlBatch_G.FIELDERROR(Name,STRSUBSTNO(Text8053000,MAXSTRLEN(CapJnlBatch_G.Name)));
        IF NOT FIND('=><') THEN BEGIN
          "Line No." := 0;
          COMMIT;
          EXIT;
         END;
        IF NOT G_WinDialog THEN BEGIN
          IF ShowDialog() THEN BEGIN
            Window_G.OPEN(Text8053001 +Text8053002 +Text8053007);
            Window_G.UPDATE(1,"Capital Batch Name");
           END;
         END;

        // Check lines
        LineCount_G := 0;
        StartLineNo_G := "Line No.";
        REPEAT
          LineCount_G := LineCount_G + 1;
          IF NOT G_WinDialog THEN
            IF ShowDialog() THEN
              Window_G.UPDATE(2,LineCount_G);

          CapJnlCheckLine_G.RunCheck(CapJnlLine_G);
          IF NEXT = 0 THEN
            FINDFIRST;
        UNTIL "Line No." = StartLineNo_G;
        NoOfRecords_G := LineCount_G;

      //Check Doc No
        FINDSET(TRUE,FALSE);
        REPEAT
          LineCount_G := LineCount_G + 1;
          IF NOT EmptyLine THEN
            IF (CapJnlBatch_G."No. Series" <> '') AND ("Document No." <> LastDocNo_G) THEN
              TESTFIELD("Document No.",NoSeriesMgt_G.GetNextNo(CapJnlBatch_G."No. Series","Posting Date",FALSE));
        UNTIL NEXT = 0;

        // Post lines
        LineCount_G := 0;
        LastDocNo_G := '';
        LastPostedDocNo_G := '';
        NoOfReversingRecords_G := 0;
        FINDSET(TRUE,FALSE);
        REPEAT
          CapJnlLine3_G := CapJnlLine_G;
          WITH CapJnlLine3_G DO
           BEGIN
            LineCount_G := LineCount_G + 1;

            IF NOT G_WinDialog THEN BEGIN
              IF ShowDialog() THEN BEGIN
                Window_G.UPDATE(5,LineCount_G);
                Window_G.UPDATE(6,ROUND(LineCount_G / NoOfRecords_G * 10000,1));
               END;
             END;
            CheckDocumentNo(CapJnlLine3_G);
            CapJnlPostLine_G.RunWithoutCheck(CapJnlLine3_G);
           END;
        UNTIL NEXT = 0;

        // Update/delete lines
         BEGIN
          IF NOT RECORDLEVELLOCKING THEN
            LOCKTABLE(TRUE,TRUE);
            CapJnlLine2_G.COPY(CapJnlLine_G);
            CapJnlLine2_G.SETFILTER("Account No.",'<>%1','');
            IF CapJnlLine2_G.FINDLAST THEN; // Remember the last line
            CapJnlLine3_G.COPY(CapJnlLine_G);
            IF CapJnlLine3_G.FINDSET(TRUE,FALSE) THEN
              REPEAT
                CapJnlLine3_G.DELETE;
              UNTIL CapJnlLine3_G.NEXT = 0;
            CapJnlLine3_G.RESET;
            CapJnlLine3_G.SETRANGE("Capital Journal Template","Capital Journal Template");
            CapJnlLine3_G.SETRANGE("Capital Batch Name","Capital Batch Name");
            IF NOT CapJnlLine3_G.FINDLAST THEN
              IF INCSTR("Capital Batch Name") <> '' THEN BEGIN
                CapJnlBatch_G.DELETE;
                CapJnlBatch_G.Name := INCSTR("Capital Batch Name");
                IF CapJnlBatch_G.INSERT THEN;
                "Capital Batch Name" := CapJnlBatch_G.Name;
               END;

            CapJnlLine3_G.SETRANGE("Capital Batch Name","Capital Batch Name");
            IF (CapJnlBatch_G."No. Series" = '') AND NOT CapJnlLine3_G.FINDLAST THEN BEGIN
              CapJnlLine3_G.INIT;
              CapJnlLine3_G."Capital Journal Template" := "Capital Journal Template";
              CapJnlLine3_G."Capital Batch Name" := "Capital Batch Name";
              CapJnlLine3_G."Line No." := 10000;
              CapJnlLine3_G.INSERT;
              TempCapJnlLine2_G := CapJnlLine2_G;
              CapJnlLine3_G.SetUpNewLine(TempCapJnlLine2_G,TRUE);
              CapJnlLine3_G.MODIFY;
             END;
         END;
        IF CapJnlBatch_G."No. Series" <> '' THEN
          NoSeriesMgt_G.SaveNoSeries;
        IF NoSeries_G.FINDSET THEN
          REPEAT
            EVALUATE(PostingNoSeriesNo_G,NoSeries_G.Description);
            NoSeriesMgt2[PostingNoSeriesNo_G].SaveNoSeries;
          UNTIL NoSeries_G.NEXT = 0;
        COMMIT;
       END;
    END;

    PROCEDURE CheckDocumentNo@11(VAR CapJnlLine2_G@8053000 : Record 8053356);
    BEGIN
      WITH CapJnlLine2_G DO
        BEGIN
         IF "Posting No. Series" = '' THEN
            "Posting No. Series" := CapJnlBatch_G."No. Series"
         ELSE
          IF NOT EmptyLine THEN
            IF "Document No." = LastDocNo_G THEN
              "Document No." := LastPostedDocNo_G
            ELSE BEGIN
              IF NOT NoSeries_G.GET("Posting No. Series") THEN BEGIN
                NoOfPostingNoSeries_G := NoOfPostingNoSeries_G + 1;
                IF NoOfPostingNoSeries_G > ARRAYLEN(NoSeriesMgt2) THEN
                  ERROR(Text8053025,ARRAYLEN(NoSeriesMgt2));
                NoSeries_G.Code := "Posting No. Series";
                NoSeries_G.Description := FORMAT(NoOfPostingNoSeries_G);
                NoSeries_G.INSERT;
               END;
              LastDocNo_G := "Document No.";
              EVALUATE(PostingNoSeriesNo_G,NoSeries_G.Description);
              "Document No." := NoSeriesMgt2[PostingNoSeriesNo_G].GetNextNo("Posting No. Series","Posting Date",TRUE);
              LastPostedDocNo_G := "Document No.";
             END;
      END;
    END;
