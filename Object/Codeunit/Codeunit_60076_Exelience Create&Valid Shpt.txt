          IF NOT Vendor_L.GET("Buy-from Vendor No.") THEN

                CollectionTax_L.RESET;
                CollectionTax_L.SETRANGE(Fees, CollectionTax_L.Fees::CVO);
                IF CollectionTax_L.FINDFIRST THEN
                  IF recTax_L.GET(CollectionTax_L.Code, PurchaseLine_P."VAT Bus. Posting Group") THEN BEGIN

                  PurchaseLine_L.SETRANGE("Document No.", "Document No.");
                  PurchaseLine_L.SETRANGE("CVO Item Attached Line", "Line No.");
                  IF TaxUnitPrice_L <> 0 THEN BEGIN
                    IF NOT PurchaseLine_L.FINDFIRST THEN BEGIN
                      InitTaxLine(PurchaseLine_P,PurchaseLine_L,TaxUnitPrice_L,CollectionTax_L."G/L Account","Line No.",'');

                    END;
                    PurchaseLine_L.VALIDATE("Prepayment VAT Identifier", PurchaseLine_L."Prepayment VAT Identifier");
                    ModifyTaxLine(PurchaseLine_P,PurchaseLine_L,TaxUnitPrice_L);
                  END ELSE
                    IF PurchaseLine_L.FINDFIRST THEN BEGIN
                      IF (PurchaseLine_L."Qty. to Receive" <> PurchaseLine_P."Qty. to Receive") OR
                        (PurchaseLine_L."Qty. to Invoice" <> PurchaseLine_P."Qty. to Invoice")
                      THEN BEGIN
                        PurchaseLine_L.DELETE(TRUE);
                      END;
                    END;
                END;
                  END;
              END;
            END;
          END;
        END;
      END;
      // --> 20190520-WI00-ADV01.01.00-3375-MRU01
    END;

    PROCEDURE CalcCVO@8053035(PurchaseLine_P@60000 : Record 39;campaign_P@8053008 : Code[20];"CVO Credit Tax Code_P"@8053000 : Code[20];Rec_item_p@8053009 : Record 27) UnitPriceCVO_L : Decimal;

      CollectionTax_L@8053007 : Record 8053007;
      CollectedTax_L@8053006 : Record 8053006;
      UnitsMeasureCMC_L@8053005 : Record 8053025;
      UnitsMeasureCMC2_L@8053004 : Record 8053025;

      ItemOfUnit2_L@8053001 : Record 5404;
      coeff_l@8053003 : Decimal;
    BEGIN
      // --> 20190520-WI00-ADV01.01.00-3375-MRU01

      // Recherche du taux de CVO
      CollectedTax_L.SETRANGE("Campain No.",campaign_P);
      CollectedTax_L.SETRANGE("Species code", '');
      CollectedTax_L.SETFILTER("Starting date", '<=%1', WORKDATE);
      CollectedTax_L.SETRANGE("Tax Code","CVO Credit Tax Code_P");
      IF CollectedTax_L.FINDLAST THEN BEGIN
        UnitsMeasureCMC2_L.RESET;
        UnitsMeasureCMC2_L.SETRANGE(Code, CollectedTax_L."Unit Code");

        UnitsMeasureCMC_L.RESET;
        UnitsMeasureCMC_L.SETRANGE(Code, PurchaseLine_P."Unit of Measure Code");
