    PROCEDURE ExportGlobalDimension@1000000000(FileName_P@1000000000 : Text[256]) Result_L : Text[256];
    VAR
      "<-- 3LI LOCALS -->"@1000000006 : Integer;
      FileExport_L@1000000002 : File;
      FileExportOutStream_L@1000000001 : OutStream;
      tableGlobalDimension_L@1000000003 : Record 349;
      TableCompany_L@1000000008 : Record 2000000006;
      TableParameter_L@1000000007 : Record 8053402;
      TableLedgerSetup_L@1000000012 : Record 98;
      Text0001@1000000010 : TextConst 'FRA=.ŸÎ'S^O"-';
      Text0002@1000000009 : TextConst 'FRA=aaceeeiou';
      ToolsFile_L@1000000011 : Codeunit 8053405;
      MoteurInterface_L@1000000005 : Codeunit 8061230;
      ConnectorName_L@1000000018 : Code[20];
      ImpExpFolder_L@1000000017 : Text[256];
      LogFolder_L@1000000016 : Text[256];
      ExtensionFile_L@1000000022 : Text[10];
      PrefixFile_L@1000000021 : Text[10];
      FileName_L@1000000015 : Text[256];
      FileLogName_L@1000000014 : Text[256];
      FilePath_L@1000000020 : Text[256];
      FileN_L@1000000019 : Text[256];
      FileTemp_L@1000000004 : Text[256];
      CodeUnit_L@1000000013 : Integer;
      FileNameOuvert_L@1000000024 : Boolean;
      NbCarEcrit_L@1000000023 : Integer;
    BEGIN
      //----------------------------------------------------------------------------------------------------------------------------------
      // Extraction des Produits table "GlobalDimension"  dans toutes les societes pour Cr'ation du fichier Dimo : Dimprodf.txt
      // Sorties des erreurs dand Log_AAAAMMJJHHMNSSmmmn.txt
      //----------------------------------------------------------------------------------------------------------------------------------
      // Entr'es:
      //   - FileName_P : Nom du fichier PC . G'n'rer
      // Sorties:
      //   - Result_L : Erreur Retourn'e, Vide si Ok
      //----------------------------------------------------------------------------------------------------------------------------------
      FileNameOuvert_L := FALSE;
      CodeUnit_L := 8053409;
      MoteurInterface_L.GetParameter(CodeUnit_L, ConnectorName_L, ImpExpFolder_L, LogFolder_L, ExtensionFile_L, PrefixFile_L);
      ToolsFile_L.LogInit(LogFolder_L, 'DIMO');
      //-->20131112-Int03-CMC02.02.00-ADVITAM-CDL
      ToolsFile_L.LogInitCodeUnit(CodeUnit_L);
      //<--20131112-Int03-CMC02.02.00-ADVITAM-CDL
      // Nom du fichier pass' en paramStre ou Nom constitu' par r'p. du Robot & Nom fic. r'cup'r' du robot
      IF FileName_P = '' THEN BEGIN
        IF (ExtensionFile_L = '') OR (PrefixFile_L = '') THEN BEGIN
           ToolsFile_L.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0002, 'ROBOT', COMPANYNAME), 1, MAXSTRLEN(Result_L)));
        END ELSE BEGIN
          FileName_L := ImpExpFolder_L + '\' + PrefixFile_L + '.' + ExtensionFile_L;
        END;
      END ELSE BEGIN
        FileName_L := FileName_P;
      END;

      IF ToolsFile_L.LogErreurType() <> 'E' THEN BEGIN
        IF TableCompany_L.FINDSET THEN BEGIN
          ToolsFile_L.LogInitFileExecute(FileName_L);
          ToolsFile_L.SplitFilename(FileName_L, FilePath_L, FileN_L);
          FileTemp_L := FilePath_L + 'Temp_' + FileN_L;
          IF NOT FileExport_L.CREATE(FileTemp_L) THEN BEGIN
            ToolsFile_L.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0003, FileTemp_L, GETLASTERRORTEXT), 1, MAXSTRLEN(Result_L)));
            CLEARLASTERROR;
          END ELSE BEGIN
            FileExport_L.CREATEOUTSTREAM(FileExportOutStream_L);
            FileNameOuvert_L := TRUE;
             REPEAT
              tableGlobalDimension_L.CHANGECOMPANY(TableCompany_L.Name);
              TableParameter_L.CHANGECOMPANY(TableCompany_L.Name);
              TableLedgerSetup_L.CHANGECOMPANY(TableCompany_L.Name);
              IF (NOT TableParameter_L.FINDSET) THEN BEGIN
                  ToolsFile_L.LogWriteInfo(COPYSTR(STRSUBSTNO(Text0001,TableParameter_L.TABLECAPTION,TableCompany_L.Name), 1
                    , MAXSTRLEN(Result_L)));
              END ELSE BEGIN
                IF TableParameter_L."Export Files DIMO" THEN BEGIN
                  IF NOT TableLedgerSetup_L.FINDFIRST THEN BEGIN
                    ToolsFile_L.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0001, TableLedgerSetup_L.TABLECAPTION,TableCompany_L.Name), 1
                      , MAXSTRLEN(Result_L)));
                  END ELSE BEGIN
                    //CLEAR(tableGlobalDimension_L); Ne pas faire de CLEAR quand CHANGECOMPANY
                    tableGlobalDimension_L.SETRANGE(tableGlobalDimension_L."Dimension Code",TableLedgerSetup_L."Global Dimension 2 Code");
                    IF tableGlobalDimension_L.FINDSET THEN BEGIN
                      REPEAT
                        IF tableGlobalDimension_L.Blocked THEN BEGIN
                        END ELSE BEGIN
                          FileExportOutStream_L.WRITETEXT(PADSTR(tableGlobalDimension_L.Code,4,' '),4);
                          FileExportOutStream_L.WRITETEXT('|', 1);
                          FileExportOutStream_L.WRITETEXT(PADSTR(UPPERCASE(CONVERTSTR(tableGlobalDimension_L.Name, Text0006, Text0007)),
                            30, ' '), 30);
                          FileExportOutStream_L.WRITETEXT('|', 1);
                          FileExportOutStream_L.WRITETEXT('99999999', 8);
                          FileExportOutStream_L.WRITETEXT('|', 1);
                          FileExportOutStream_L.WRITETEXT(PADSTR(TableParameter_L."Company Name",3,' '),3);
                          FileExportOutStream_L.WRITETEXT();
                        END;
                      UNTIL tableGlobalDimension_L.NEXT=0;
                    END;
                  END;
                  TableParameter_L."Last Date Export DIMPROD" := CURRENTDATETIME;
                  IF NOT TableParameter_L.MODIFY(TRUE) THEN BEGIN
                    ToolsFile_L.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0004, TableParameter_L.TABLECAPTION), 1, MAXSTRLEN(Result_L)));
                  END;
                END;
              END;
            UNTIL TableCompany_L.NEXT=0;
            NbCarEcrit_L := FileExport_L.LEN;
            FileExport_L.CLOSE;
          END;
        END;
      END;

      IF (NbCarEcrit_L > 0) THEN BEGIN
        IF ToolsFile_L.LogErreurType() = 'E' THEN BEGIN
          IF FileNameOuvert_L THEN BEGIN
            FILE.ERASE(FileTemp_L);
          END;
        END ELSE BEGIN
          IF FILE.ERASE(FileName_L) THEN BEGIN
          END;
          IF FILE.RENAME(FileTemp_L, FileName_L) THEN BEGIN
          END;
          IF NOT ToolsFile_L.FileCopySav(FileName_L, 'Sav', TRUE) THEN BEGIN
            ToolsFile_L.LogWriteErreur(STRSUBSTNO(Text0011, FileName_L));
          END;
        END;
      END ELSE BEGIN
        IF FileNameOuvert_L THEN BEGIN
          FILE.ERASE(FileTemp_L);
        END;
      END;

      CLEAR(Result_L);
      FileLogName_L := ToolsFile_L.LogName;
      ToolsFile_L.LogClose();
      IF ToolsFile_L.LogErreurType() = 'E' THEN BEGIN
        Result_L := 'ERREUR ' + FileLogName_L;
        ERROR(Result_L);
      END ELSE BEGIN
        IF ToolsFile_L.LogErreurType() = 'I' THEN BEGIN
          Result_L := Text0005;
          MoteurInterface_L.InsertLog(CodeUnit_L, FileLogName_L);
        END ELSE BEGIN
          MoteurInterface_L.InsertOk(CodeUnit_L);
        END;
      END;
      //<--20130205-Int03-CMC02.02.00-ADVITAM-ESR
    END;

      //<-- 3LI END
    }
