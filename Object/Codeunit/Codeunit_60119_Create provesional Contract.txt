    VAR
      RecProvCollectionContract_L@1000000003 : Record 8052983;
      RecContractTemplate_L@1000000004 : Record 8052983;
      Contact_L@1000000009 : Record 5050;
      cduStatusMgt_L@1000000008 : Codeunit 8052985;
      CMCSetup_L@1000000007 : Record 8052982;
      NoSeriesMgt_L@1000000006 : Codeunit 396;
      PurchContractType_L@1000000010 : Record 8053096;
      CMCContactSalesPerson_L@1000000012 : Record 8053160;
      CMCFrameContractType_L@1000000013 : Record 8053198;
      Vendor_L@1000000011 : Record 23;
      Customer_L@1000000014 : Record 18;
      L_recTemplateContractQuality@1000000015 : Record 8052984;
      L_recContractQuality@1000000016 : Record 8052984;
      RecContractPrioQuality_L@1000000017 : Record 60068;
      RecContractPrioQualityIns_L@1000000018 : Record 60068;
      RecContractCondQuality_L@1000000020 : Record 60069;
      RecContractCondQualityIns_L@1000000019 : Record 60069;
      RecConSP_L@1000000022 : Record 8053160;
      RecCMCSetup_L@1000000021 : Record 8052982;
    BEGIN
      RecContractTemplate_L.SETCURRENTKEY(Type,"Campaign No.","Species Code","Item No.",Status,"Contract Type");
      RecContractTemplate_L.SETRANGE(Type,RecContractTemplate_L.Type::Template);
      RecContractTemplate_L.SETRANGE(Status,RecContractTemplate_L.Status::Locked);
      RecContractTemplate_L.SETRANGE("Campaign No.",Campaign_P);
      RecContractTemplate_L.SETRANGE("Contract Type",ContractType_P);
      RecContractTemplate_L.SETRANGE("Species Code",Species_P);
      RecContractTemplate_L.SETRANGE("Item No.",Item_P);
      IF RecContractTemplate_L.FIND('-') THEN BEGIN

        Contact_L.GET(Vendor_P);
        Vendor_L.GET(Contact_L."Vendor No.");
        Customer_L.GET(Vendor_L."Customer No.");
        CMCSetup_L.GET;
        CMCSetup_L.TESTFIELD(CMCSetup_L."Collection Contract Nos.");
        PurchContractType_L.GET(RecContractTemplate_L."Contract Type");

        //CMCFrameContractType_L.GET(ContractType_P);
        //CMCContactSalesPerson_L.RESET;
        //CMCContactSalesPerson_L.SETRANGE("Contact Code",Contact_L."No.");
        //CMCContactSalesPerson_L.SETRANGE("Prod. Activity Code" ,CMCFrameContractType_L."Prod. Activity Code");
        //CMCContactSalesPerson_L.SETRANGE("Sales Network Code", CMCFrameContractType_L."Sales Network Code");
        //CMCContactSalesPerson_L.FINDFIRST;

        RecProvCollectionContract_L.INIT;
        RecProvCollectionContract_L."No." := NoSeriesMgt_L.GetNextNo(CMCSetup_L."Collection Contract Nos.",WORKDATE,TRUE);
        RecProvCollectionContract_L.INSERT(TRUE);
        RecProvCollectionContract_L.Type := RecProvCollectionContract_L.Type::Contract;
        RecProvCollectionContract_L."Collection Type" := RecProvCollectionContract_L."Collection Type"::Purchase;
        RecProvCollectionContract_L."Contract Type" := RecContractTemplate_L."Contract Type";
        //RecProvCollectionContract_L.VALIDATE("Template No.",RecContractTemplate_L."No.");
        RecProvCollectionContract_L."Template No.":=RecContractTemplate_L."No.";

        RecProvCollectionContract_L.Description := PurchContractType_L.Description;
        RecProvCollectionContract_L.VALIDATE("Vendor No.",Contact_L."Vendor No.");
        RecProvCollectionContract_L.VALIDATE("Campaign No.",RecContractTemplate_L."Campaign No.");

        RecProvCollectionContract_L."Salesperson Code" := RecContractTemplate_L."Salesperson Code";
        RecProvCollectionContract_L."Purchaser Code" := RecContractTemplate_L."Purchaser Code";

        IF (RecProvCollectionContract_L."Salesperson Code" = '')OR(RecProvCollectionContract_L."Purchaser Code" = '') THEN BEGIN
          IF Customer_L."Salesperson Code"<>'' THEN BEGIN
            RecProvCollectionContract_L."Salesperson Code" := Customer_L."Salesperson Code";
            RecProvCollectionContract_L."Purchaser Code" := Customer_L."Salesperson Code";
          END;

          IF Vendor_L."Purchaser Code"<>'' THEN BEGIN
            RecProvCollectionContract_L."Salesperson Code" := Vendor_L."Purchaser Code";
            RecProvCollectionContract_L."Purchaser Code" := Vendor_L."Purchaser Code";
          END;
        END;


        RecCMCSetup_L.GET;
        CLEAR(RecConSP_L);
        RecConSP_L.SETRANGE("Contact Code", RecProvCollectionContract_L."Vendor No.");
        RecConSP_L.SETRANGE("Prod. Activity Code",RecCMCSetup_L."Market Prod Activity");
        RecConSP_L.SETRANGE("Sales Network Code",RecCMCSetup_L.Commercial);
        IF RecConSP_L.FIND('-') THEN
           RecProvCollectionContract_L.VALIDATE("Salesperson Code",RecConSP_L."Salesperson Code");


        IF RecProvCollectionContract_L."Salesperson Code"= '' THEN BEGIN
          //ERROR(Txt0004,RecProvCollectionContract_L."No.",Contact_L."Vendor No.");
          //Insertion du log
          RecBatchLog_G.CreateLog(RecBatchLog_G."Source Type"::Vendor,Vendor_P,
            STRSUBSTNO(Txt0004,RecProvCollectionContract_L."No.",Contact_L."Vendor No."),
            RecBatchLog_G."Process Type"::"Create Provesional Contract");
          CountError_G += 1;
        END;

        RecProvCollectionContract_L.VALIDATE("Species Code",RecContractTemplate_L."Species Code");
        RecProvCollectionContract_L.VALIDATE("Item No.",RecContractTemplate_L."Item No.");
        RecProvCollectionContract_L.VALIDATE(Quantity,RecContractTemplate_L.Quantity);
        RecProvCollectionContract_L.VALIDATE("Unit Price",RecContractTemplate_L."Unit Price");
        RecProvCollectionContract_L.VALIDATE("Due Date",RecContractTemplate_L."Due Date");
        RecProvCollectionContract_L.VALIDATE(Parity,RecContractTemplate_L.Parity);
        IF RecContractTemplate_L."Bonus Due Code"<>'' THEN
          RecProvCollectionContract_L.VALIDATE("Bonus Due Code",RecContractTemplate_L."Bonus Due Code");
        //RecProvCollectionContract_L.VALIDATE("Template No.",RecContractTemplate_L."No.");

        // ------------------------------------------------------------------------
        // Copie des Qualit's Contrat d'finies sur le modSle
        // ------------------------------------------------------------------------
        L_recTemplateContractQuality.RESET;
        L_recTemplateContractQuality.SETCURRENTKEY("Contract No.", "Analysis Code");
        L_recTemplateContractQuality.SETRANGE("Contract No.", RecContractTemplate_L."No.");
        IF L_recTemplateContractQuality.FINDSET(FALSE, FALSE) THEN
          REPEAT
             L_recContractQuality.INIT;
             L_recContractQuality.VALIDATE("Contract No.", RecProvCollectionContract_L."No.");
             L_recContractQuality.VALIDATE("Analysis Code", L_recTemplateContractQuality."Analysis Code");
             L_recContractQuality.TRANSFERFIELDS(L_recTemplateContractQuality, FALSE);
             L_recContractQuality.INSERT(TRUE);
          UNTIL L_recTemplateContractQuality.NEXT = 0;

        // ------------------------------------------------------------------------
        // Copie des priorit's et conditions Qualit's Contrat d'finies sur le modSle
        // ------------------------------------------------------------------------
        CLEAR(RecContractPrioQuality_L);
        RecContractPrioQuality_L.SETRANGE("Contract No.",RecContractTemplate_L."No.");
        IF RecContractPrioQuality_L.FINDSET THEN
        REPEAT
          RecContractPrioQualityIns_L.INIT;
          RecContractPrioQualityIns_L.TRANSFERFIELDS(RecContractPrioQuality_L);
          RecContractPrioQualityIns_L."Contract No." := RecProvCollectionContract_L."No.";
          RecContractPrioQualityIns_L.INSERT;

          CLEAR(RecContractCondQuality_L);
          RecContractCondQuality_L.SETRANGE("Contract No.",RecContractTemplate_L."No.");
          RecContractCondQuality_L.SETRANGE("Priority Code",RecContractPrioQuality_L."Priority Code");
          IF RecContractCondQuality_L.FINDSET THEN
          REPEAT
            RecContractCondQualityIns_L.INIT;
            RecContractCondQualityIns_L.TRANSFERFIELDS(RecContractCondQuality_L);
            RecContractCondQualityIns_L."Contract No." := RecProvCollectionContract_L."No.";
            RecContractCondQualityIns_L.INSERT;
          UNTIL RecContractCondQuality_L.NEXT = 0;
        UNTIL RecContractPrioQuality_L.NEXT = 0;

        cduStatusMgt_L.LockContract(RecProvCollectionContract_L);

        ContractCount_G +=1;
        EXIT(TRUE);
      END ELSE BEGIN
        //ERROR(Txt0003,ContractType_P);
        //Insertion du log
        RecBatchLog_G.CreateLog(RecBatchLog_G."Source Type"::Vendor,Vendor_P,
          STRSUBSTNO(Txt0003,ContractType_P,Campaign_P,Item_P,Species_P),
          RecBatchLog_G."Process Type"::"Create Provesional Contract");
        CountError_G += 1;
        EXIT(FALSE);
      END;
    END;
