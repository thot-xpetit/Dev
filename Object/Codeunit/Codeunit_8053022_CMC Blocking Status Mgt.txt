    PROCEDURE CheckBlockingOnRelease@8053002(DocumentType_P@8053001 : 'Quote,Order,Invoice,Credit Memo,Blanket Order,Return Order';DocumentNo_P@8053000 : Code[20]);
    VAR
      Customer_L@8053010 : Record 18;
      BlockingStatusDetail_L@8053009 : Record 8053227;
      BlockingStatusDetailTMP_L@8053012 : TEMPORARY Record 8053227;
      Activity_L@8053008 : Record 8052985;
      Family_L@8053007 : Record 8052986;
      SubFamily_L@8053006 : Record 8052987;
      Species_L@8053005 : Record 8052988;
      Item_L@8053004 : Record 27;
      TXT001_L@8053011 : TextConst 'FRA="%1 = %2 pour la ligne %3 = %4 ,%5 = %6 , %7 = %8 de la page %9"';
      SalesHeader_L@8053002 : Record 36;
      SalesLine_L@8053003 : Record 37;
      SalesPerson_L@8053018 : Record 13;
      CMBEmailLogEntry_L@8053016 : Record 8061374;
      BlockingStatusCode_L@8053013 : Code[10];
      ToSend@8053015 : Boolean;
      EntryNo_L@8053017 : Integer;
      ContNo_L@8053014 : Code[20];
      Contact_L@8053019 : Record 5050;
      ContSalesPerson_L@8053020 : Record 8053160;
      MailSent_L@8053021 : Boolean;
      CustPostingGroup_L@1000000001 : Record 92;
    BEGIN
      IF DocumentType_P <> DocumentType_P::Order THEN
       EXIT;

      SalesHeader_L.GET(DocumentType_P,DocumentNo_P) ;

      ContNo_L := CVLedgerMgt_G.GetContNoFromCustVentNo(SalesHeader_L."Bill-to Customer No.",0);
      IF ContNo_L <> '' THEN BEGIN
        Contact_L.GET(ContNo_L);
        BlockingStatusCode_L := Contact_L."Blocking Status Code";
      END ELSE BEGIN
        Customer_L.GET(SalesHeader_L."Bill-to Customer No.");
        BlockingStatusCode_L := Customer_L."Blocking Status Code";
      END;

      WITH BlockingStatusDetail_L DO BEGIN
        SalesLine_L.RESET;
        SalesLine_L.SETRANGE(SalesLine_L."Document Type",DocumentType_P);
        SalesLine_L.SETRANGE(SalesLine_L."Document No.",DocumentNo_P);
        SalesLine_L.SETRANGE(SalesLine_L.Type,SalesLine_L.Type::Item);
        IF SalesLine_L.FINDSET THEN
          REPEAT
           RESET;
           SETRANGE(BlockingStatusDetail_L."Blocking Status Code",BlockingStatusCode_L);
           IF FINDSET THEN
              REPEAT
                Item_L.GET(SalesLine_L."No.");
                ToSend := FALSE;
                CASE Type OF
                  Type::Item :
                    IF Item_L."No." = "No." THEN
                      ToSend := TRUE;

                  Type::Species :
                    IF Item_L."Species Code" = "No." THEN
                      ToSend := TRUE;

                  Type::Subfamily :
                    IF Item_L."Sub-Family Code" = "No." THEN
                      ToSend := TRUE;

                  Type::Family :
                    IF Item_L."Family Code" = "No." THEN
                      ToSend := TRUE;

                  Type::Activity :
                    IF Item_L."Activity Code" = "No." THEN
                      ToSend := TRUE;
                END;

                IF ToSend THEN
                  IF (BlockingStatusDetail_L."Group Account Filter" <> '') THEN
                    IF CustPostingGroup_L.GET(SalesHeader_L."Customer Posting Group") THEN
                      IF CustPostingGroup_L."Receivables Account" <> '' THEN
                        IF STRPOS(BlockingStatusDetail_L."Group Account Filter",CustPostingGroup_L."Receivables Account") = 0 THEN
                          ToSend := FALSE;

                IF ToSend THEN BEGIN
                  IF NOT BlockingStatusDetailTMP_L.GET("Blocking Status Code",Type,"No.") THEN BEGIN
                    Customer_L.GET(SalesHeader_L."Bill-to Customer No.");
                    IF ("Salesperson Notif. Limit LCY" <> 0) AND (CalcDueBalance(Customer_L."No.",
                      BlockingStatusDetail_L) > "Salesperson Notif. Limit LCY")
                    THEN BEGIN
                      IF SalesLine_L."Salesperson Code" = '' THEN
                        ERROR(Text8053001,SalesLine_L."Line No.");
                      BlockingStatusDetailTMP_L.INIT;
                      BlockingStatusDetailTMP_L.TRANSFERFIELDS(BlockingStatusDetail_L);
                      SalesPerson_L.GET(SalesLine_L."Salesperson Code");
                      BlockingStatusDetailTMP_L."E-Mail" := SalesPerson_L."E-Mail";
                      BlockingStatusDetailTMP_L.INSERT;
                    END;
                  END;
                END;
              UNTIL NEXT = 0;
          UNTIL SalesLine_L.NEXT =0 ;
       END;

      CMBEmailLogEntry_L.LOCKTABLE;
      CMBEmailLogEntry_L.RESET;
      EntryNo_L :=0;
      IF CMBEmailLogEntry_L.FINDLAST THEN
        EntryNo_L := CMBEmailLogEntry_L."Entry No.";

      BlockingStatusDetailTMP_L.RESET;
      IF BlockingStatusDetailTMP_L.FINDSET THEN
        REPEAT
          MailSent_L := FALSE;
          IF (ContNo_L <> '') THEN BEGIN
            ContSalesPerson_L.RESET;
            ContSalesPerson_L.SETRANGE("Contact Code",ContNo_L);
            ContSalesPerson_L.SETRANGE("Debit Responsible",TRUE);
            IF ContSalesPerson_L.FINDSET THEN
              REPEAT
                IF (ContSalesPerson_L."Salesperson Code" <> '') AND (SalesPerson_L.GET(ContSalesPerson_L."Salesperson Code")) AND
                  (SalesPerson_L."E-Mail" <> '')
                THEN BEGIN
                  CMBEmailLogEntry_L.INIT;
                  EntryNo_L += 1;
                  CMBEmailLogEntry_L."Entry No." := EntryNo_L;
                  CMBEmailLogEntry_L.Status := CMBEmailLogEntry_L.Status ::Waiting;
                  CMBEmailLogEntry_L.Module := CMBEmailLogEntry_L.Module::Sales;
                  CMBEmailLogEntry_L."Document Type" := CMBEmailLogEntry_L."Document Type"::Order;
                  CMBEmailLogEntry_L."Document No." := DocumentNo_P ;
                  CMBEmailLogEntry_L."Creation Date Time" := CURRENTDATETIME;
                  CMBEmailLogEntry_L."E-Mail Template Code" := BlockingStatusDetailTMP_L."Salesperson Notif. e-mail Tmpl";
                  CMBEmailLogEntry_L.Recipients := SalesPerson_L."E-Mail";
                  CMBEmailLogEntry_L.INSERT;

                  MailSent_L := TRUE;
                END;
              UNTIL ContSalesPerson_L.NEXT = 0;
          END;

          IF NOT MailSent_L THEN BEGIN
            CMBEmailLogEntry_L.INIT;
            EntryNo_L += 1;
            CMBEmailLogEntry_L."Entry No." := EntryNo_L;
            CMBEmailLogEntry_L.Status := CMBEmailLogEntry_L.Status ::Waiting;
            CMBEmailLogEntry_L.Module := CMBEmailLogEntry_L.Module::Sales;
            CMBEmailLogEntry_L."Document Type" := CMBEmailLogEntry_L."Document Type"::Order;
            CMBEmailLogEntry_L."Document No." := DocumentNo_P ;
            CMBEmailLogEntry_L."Creation Date Time" := CURRENTDATETIME;
            CMBEmailLogEntry_L."E-Mail Template Code" := BlockingStatusDetailTMP_L."Salesperson Notif. e-mail Tmpl";
            CMBEmailLogEntry_L.Recipients := BlockingStatusDetailTMP_L."E-Mail";
            CMBEmailLogEntry_L.INSERT ;
          END;
      UNTIL BlockingStatusDetailTMP_L.NEXT = 0;
    END;

    PROCEDURE CalcDueBalance@8053003(CustNo_P@8053002 : Code[20];BlockStatusDetail_P@8053003 : Record 8053227) : Decimal;
    VAR
      CVLedgerMgt_L@8053000 : Codeunit 8053377;
      ContNo_L@8053001 : Code[20];
      CVCollBalance_L@8053004 : Record 8053382;
      BalanceDue_L@8053005 : Decimal;
    BEGIN
      BalanceDue_L := 0;

      ContNo_L := CVLedgerMgt_L.GetContNoFromCustVentNo(CustNo_P,0);
      IF ContNo_L = '' THEN
        ERROR(Text8053000,CustNo_P);

      CVCollBalance_L.RESET;
      CVCollBalance_L.SETRANGE("Contact No.",ContNo_L);
      IF BlockStatusDetail_P."Group Account Filter" <> '' THEN
        CVCollBalance_L.SETFILTER("G/L Account No.",BlockStatusDetail_P."Group Account Filter");
      CVCollBalance_L.SETFILTER("Date Filter",'..%1',WORKDATE);
      IF CVCollBalance_L.FINDSET THEN
        REPEAT
          CVCollBalance_L.CALCFIELDS("Balance Due (LCY)");
          BalanceDue_L := BalanceDue_L + CVCollBalance_L."Balance Due (LCY)";
        UNTIL CVCollBalance_L.NEXT = 0;
      EXIT(BalanceDue_L);
    END;
