                    CostPlusLedgerEntry_L."Valuation Type" := CostPlusLedgerEntry_L."Valuation Type"::Percentage
                  ELSE
                    CostPlusLedgerEntry_L."Valuation Type" := CostPlusLedgerEntry_L."Valuation Type"::"Unit Amount";

                  IF CostPlusList_L."Valuation Type" = CostPlusList_L."Valuation Type"::Percentage THEN
                    CostPlusLedgerEntry_L."Valuation Type" := CostPlusLedgerEntry_L."Valuation Type"::Percentage

                  CostPlusLedgerEntry_L.INSERT(TRUE);
                  ManageCostPlusDetailEntry(CostPlusLedgerEntry_L,0);
                  //--------------------------------------------------------------
                  CostPlusLedgerEntry_L.INIT;
                  CostPlusLedgerEntry_L.TRANSFERFIELDS(ValueEntry_P_, FALSE);
                  CostPlusValueEntryNo_L += 1;
                  CostPlusLedgerEntry_L."Entry No." := CostPlusValueEntryNo_L + 1;
                  CostPlusLedgerEntry_L."Value Entry No." := ValueEntry_P_."Entry No.";
                  //---------------------------------------------------------------
                UNTIL ValueEntryCostPlus_L.NEXT = 0;
              END;
            END;
          END;
          // ----------------------------------------------------------------
          ValueEntry_P_."Document Type"::"Purchase Credit Memo",
          ValueEntry_P_."Document Type"::"Purchase Invoice" : BEGIN

              IF NOT IsPurchaseCostPlusDocument(ValueEntry_P_) THEN BEGIN

                      CostPlusLedgerEntry_L."Cost Plus Code" := ValueEntryCostPlus_L."Cost Plus Code";

      ;

                        ROUND(CostPlusLedgerEntry_L."C. Plus Unit Amount (Invoice)",UnitAmtRndPrecision_L);

            //END;

    PROCEDURE ManageCostPlusDetailEntry@1000000003(CostPlusEntry_P@1000000003 : Record 60127;EntryType_P@1000000002 : 'Initial,Update,Cancel,User Update');
    VAR
      DetCostPlusEntry_L@1000000000 : Record 60128;
      DetCostPlusEntry2_L@1000000004 : Record 60128;
      NextEntryNo_L@1000000005 : Integer;
    BEGIN
      DetCostPlusEntry_L.RESET;
      DetCostPlusEntry_L.SETCURRENTKEY("Sales / Purchase","Cost Plus Entry No.");
      DetCostPlusEntry_L.SETRANGE("Sales / Purchase",DetCostPlusEntry_L."Sales / Purchase"::Purchase);
      DetCostPlusEntry_L.SETRANGE("Cost Plus Entry No.",CostPlusEntry_P."Entry No.");
      DetCostPlusEntry_L.CALCSUMS("Amount (Shpmt/Rcpt)");
      DetCostPlusEntry_L.CALCSUMS("Amount (Invoice)");

      // --> 20200626-WI00-ADV01.01.00-3446-CJY01
      //modification pour permettre la g'n'ration d'une 'criture origine . 0 si le param'trage
      //du Cost Plus utilis' le veut
      {BEFORE
      IF (CostPlusEntry_P."Cost Plus Amount (Receipt)" <> DetCostPlusEntry_L."Amount (Shpmt/Rcpt)") OR
         (CostPlusEntry_P."Cost Plus Amount (Invoice)" <> DetCostPlusEntry_L."Amount (Invoice)") THEN BEGIN
      AFTER}
      IF (EntryType_P = EntryType_P::Initial) OR
         (CostPlusEntry_P."Cost Plus Amount (Receipt)" <> DetCostPlusEntry_L."Amount (Shpmt/Rcpt)") OR
         (CostPlusEntry_P."Cost Plus Amount (Invoice)" <> DetCostPlusEntry_L."Amount (Invoice)") THEN BEGIN
      // <-- 20200626-WI00-ADV01.01.00-3446-CJY01
        DetCostPlusEntry2_L.RESET;
        IF DetCostPlusEntry2_L.FINDLAST THEN
          NextEntryNo_L := DetCostPlusEntry2_L."Entry No." + 1
        ELSE
          NextEntryNo_L := 1;

        DetCostPlusEntry2_L.INIT;
        DetCostPlusEntry2_L."Entry No." := NextEntryNo_L;
        DetCostPlusEntry2_L."Sales / Purchase" := DetCostPlusEntry_L."Sales / Purchase"::Purchase;
        DetCostPlusEntry2_L."Cost Plus Entry No." := CostPlusEntry_P."Entry No.";
        DetCostPlusEntry2_L."Amount (Shpmt/Rcpt)" :=
          CostPlusEntry_P."Cost Plus Amount (Receipt)" - DetCostPlusEntry_L."Amount (Shpmt/Rcpt)";
        DetCostPlusEntry2_L."Amount (Invoice)" := CostPlusEntry_P."Cost Plus Amount (Invoice)" - DetCostPlusEntry_L."Amount (Invoice)";
        DetCostPlusEntry2_L."Entry Type" := EntryType_P;
        DetCostPlusEntry2_L."Posting Date" := CostPlusEntry_P."Posting Date";
        DetCostPlusEntry2_L."User ID" := USERID;
        DetCostPlusEntry2_L."Creation Date" := TODAY;
        DetCostPlusEntry2_L."Creation Time" := TIME;
        //gestion de la date 'criture compta article
        DetCostPlusEntry2_L."Item Ledger Entry Posting Date" := CostPlusEntry_P."Item Ledger Entry Posting Date";

        DetCostPlusEntry2_L.INSERT;

      END;
    END;

    PROCEDURE CalcRevPurchPlusExpToBalance@8053011(VAR CostPlusLedgerEntry_P@8053001 : Record 60127;InvoicedQty_P@8053000 : Decimal);
    VAR
      ItemLedgerEntry_L@8053005 : Record 32;
      OldExpectedQty_L@8053003 : Decimal;
      OldExpectedAmount_L@8053002 : ARRAY [3] OF Decimal;
      InvoicedQty_L@60000 : Decimal;
      GLSetup_L@1000000001 : Record 98;
      AmtRndPrecision_L@1000000002 : Decimal;
      UnitAmtRndPrecision_L@1000000003 : Decimal;
      CostPlusLedgerEntry_L@1000000004 : Record 60127;
    BEGIN

      CLEAR(OldExpectedAmount_L);
      GLSetup_L.GET;
      AmtRndPrecision_L := GLSetup_L."Amount Rounding Precision";
      UnitAmtRndPrecision_L := GLSetup_L."Unit-Amount Rounding Precision";

      WITH CostPlusLedgerEntry_L DO BEGIN
        SETCURRENTKEY("Item Ledger Entry No.","Entry Type");
        SETRANGE("Item Ledger Entry No.",CostPlusLedgerEntry_P."Item Ledger Entry No.");
        SETRANGE("Entry Type","Entry Type"::"Direct Cost");
        SETFILTER("Document Type", '%1|%2', "Document Type"::"Purchase Receipt", "Document Type"::"Purchase Return Shipment");
        //SETFILTER("Entry No.",'<%1',ECDLedgerEntry_P."Entry No.");
        SETRANGE("Item Charge No.",'');
        SETRANGE("Cost Plus Code",CostPlusLedgerEntry_P."Cost Plus Code");
        IF FINDSET THEN
          REPEAT
            OldExpectedQty_L := OldExpectedQty_L + "Valued Quantity";
            //OldExpectedAmount_L[1] := OldExpectedAmount_L[1] + ACPLedgerEntry_L."Net Amount (Receipt)";
            OldExpectedAmount_L[2] := OldExpectedAmount_L[2] + CostPlusLedgerEntry_L."Cost Plus Amount (Receipt)";
            OldExpectedAmount_L[3] := OldExpectedAmount_L[3] + CostPlusLedgerEntry_L."C. Plus Unit Amount (Receipt)";
          UNTIL NEXT = 0;
        IF OldExpectedQty_L <> 0 THEN BEGIN
          //ECDLedgerEntry_P."Net Amount (Receipt)" :=
            //-ROUND((InvoicedQty_P * OldExpectedAmount_L[1]) / OldExpectedQty_L, GLSetup_L."Amount Rounding Precision");

          CostPlusLedgerEntry_P."Cost Plus Amount (Receipt)" :=
            -ROUND((InvoicedQty_P * OldExpectedAmount_L[2]) / OldExpectedQty_L, AmtRndPrecision_L);

          CostPlusLedgerEntry_P."C. Plus Unit Amount (Receipt)" :=
            -ROUND((InvoicedQty_P * OldExpectedAmount_L[3]) / OldExpectedQty_L, UnitAmtRndPrecision_L);
        END;
      END;
    END;

    PROCEDURE ManageCostPlusPurchLineOld@1000000002(PurchLine_P@1000000000 : Record 39);
    VAR
      CostPlusPurchLines_L@1000000001 : Record 60121;
      PurchaseHeader_L@1000000002 : Record 38;
      CostPlusList_L@1000000003 : Record 60118;
      CostPlusDetails_L@1000000004 : Record 60119;
      TempCostPlusPurchLines_L@1000000005 : TEMPORARY Record 60121;
      i_L@1000000006 : Integer;
      j_L@1000000007 : Integer;
      TotalDiscount_L@1000000008 : Decimal;
      PreviousFinalPrice_L@1000000009 : Decimal;
      CostPlusLedgerEntry_L@1000000010 : Record 60127;
    BEGIN
      //principe g'n'ral :
      //dans le cas d'une ligne retour ou avoir : on ne passe pas par le param'trage Cost Plus mais :
      //si il y a eu extraction on r'cupSre les lignes Cost Plus existantes sur la ligne extraite
      //si la ligne extraite n'a pas de ligne Cost Plus, on ne cr'e pas non plus de ligne Cost Plus sur la ligne retour/avoir

      WITH PurchLine_P DO BEGIN
        PurchaseHeader_L.GET("Document Type", "Document No.");
        //on devra g'rer le cas des retours qui utiliseront les ACP li' . la r'ception originale afin de garantir
        //la sym'trie entre r'ception et retour
        IF ((PurchaseHeader_L."Document Type" <> PurchaseHeader_L."Document Type"::Order) AND
          (PurchaseHeader_L."Document Type" <> PurchaseHeader_L."Document Type"::"Return Order") AND
          (PurchaseHeader_L."Document Type" <> PurchaseHeader_L."Document Type"::Invoice) AND
          (PurchaseHeader_L."Document Type" <> PurchaseHeader_L."Document Type"::"Credit Memo")) THEN
            EXIT;

        TempCostPlusPurchLines_L.RESET;
        TempCostPlusPurchLines_L.DELETEALL;
        //suppression des 'ventuelles lignes Cost Plus avant re-cr'ation
        CostPlusPurchLines_L.SETRANGE("Document Type", "Document Type");
        CostPlusPurchLines_L.SETRANGE("Document No.", "Document No.");
        CostPlusPurchLines_L.SETRANGE("Document Line No.", "Line No.");

        IF NOT CostPlusPurchLines_L.ISEMPTY THEN
          CostPlusPurchLines_L.DELETEALL(TRUE);
        CostPlusPurchLines_L.RESET;


        IF ((PurchaseHeader_L."Document Type" = PurchaseHeader_L."Document Type"::Order) OR
          (PurchaseHeader_L."Document Type" = PurchaseHeader_L."Document Type"::Invoice)) THEN BEGIN
          IF PurchaseHeader_L."Document Type" = PurchaseHeader_L."Document Type"::Order THEN
            FindCostPlusWithSetup(PurchLine_P,TempCostPlusPurchLines_L,PurchaseHeader_L."Order Date",'')
          ELSE
            FindCostPlusWithSetup(PurchLine_P,TempCostPlusPurchLines_L,PurchLine_P."Order Date",'');

        END
        ELSE BEGIN

          //cas des retours ou avoir non bas's sur le param'trage mais sur la ligne r'ception origininale
          // --> 20200217-WI00-ADV01.01.00-3446-CJY01
          {BEFORE
          IF (("Doc Type Extraction" = "Doc Type Extraction"::Receipt) AND
          AFTER}
          IF (("Doc Type Extraction" <> "Doc Type Extraction"::" ") AND
          // <-- 20200217-WI00-ADV01.01.00-3446-CJY01

             ("No. Doc Extraction" <> '') AND ("No. Line Doc Extraction" <> 0)) THEN BEGIN
              //recherches des lignes ACP g'n'r'es sur la ligne extraite
              CostPlusLedgerEntry_L.RESET;
              CostPlusLedgerEntry_L.SETRANGE("Item No.","No.");
              CostPlusLedgerEntry_L.SETRANGE("Item Ledger Entry Type",CostPlusLedgerEntry_L."Item Ledger Entry Type"::Purchase);
              IF "Doc Type Extraction" = "Doc Type Extraction"::Receipt THEN
                CostPlusLedgerEntry_L.SETRANGE("Document Type",CostPlusLedgerEntry_L."Document Type"::"Purchase Receipt")
              ELSE
                // --> 20200217-WI00-ADV01.01.00-3446-CJY01
                {BEFORE
                CostPlusLedgerEntry_L.SETRANGE("Document Type",CostPlusLedgerEntry_L."Document Type"::"Purchase Credit Memo");
                AFTER}
                CostPlusLedgerEntry_L.SETRANGE("Document Type",CostPlusLedgerEntry_L."Document Type"::"Purchase Invoice");
                // <-- 20200217-WI00-ADV01.01.00-3446-CJY01

              CostPlusLedgerEntry_L.SETRANGE("Document No.","No. Doc Extraction");
              CostPlusLedgerEntry_L.SETRANGE("Document Line No.","No. Line Doc Extraction");
              // --> 20200313-WI00-ADV01.01.00-3446-CJY01
              //on remplace le critSre "Cost Plus Code" <> '' par un autre critSre car les lignes doc final conserve finalement
              //le code "code plus" original
              {BEFORE
              CostPlusLedgerEntry_L.SETFILTER("Cost Plus Code",'<>%1','');
              AFTER
              }
              CostPlusLedgerEntry_L.SETRANGE("Cost Pl. No. final doc. purch.",0);
              // <-- 20200313-WI00-ADV01.01.00-3446-CJY01
              IF CostPlusLedgerEntry_L.FINDSET THEN REPEAT
                TempCostPlusPurchLines_L.INIT;
                TempCostPlusPurchLines_L.VALIDATE("Document Type", "Document Type");
                TempCostPlusPurchLines_L.VALIDATE("Document No.", "Document No.");
                TempCostPlusPurchLines_L.VALIDATE("Document Line No.", "Line No.");
                TempCostPlusPurchLines_L.VALIDATE("Cost Plus Code", CostPlusLedgerEntry_L."Cost Plus Code");
                IF Quantity <> 0 THEN BEGIN
                  TotalDiscount_L := ROUND("Line Discount Amount" / ABS(Quantity), 0.00001);
                  TempCostPlusPurchLines_L.VALIDATE("Base Value", "Direct Unit Cost" - TotalDiscount_L);
                END
                ELSE BEGIN
                  TempCostPlusPurchLines_L.VALIDATE("Base Value", "Direct Unit Cost");
                END;
                TempCostPlusPurchLines_L.VALIDATE("Cost Plus %", CostPlusLedgerEntry_L."Cost Plus %");
                TempCostPlusPurchLines_L.VALIDATE("Cost Plus Amount", ROUND(CostPlusLedgerEntry_L."Cost Plus value", 0.01, '='));
                TempCostPlusPurchLines_L.VALIDATE("Cost Plus vendor No.",CostPlusLedgerEntry_L."Cost Plus Vendor No.");
                TempCostPlusPurchLines_L.VALIDATE(Rank, CostPlusLedgerEntry_L.Rank);
                IF TempCostPlusPurchLines_L.INSERT(TRUE) THEN;

              UNTIL CostPlusLedgerEntry_L.NEXT = 0;

          END;

        END;
        TempCostPlusPurchLines_L.RESET;
        TempCostPlusPurchLines_L.SETCURRENTKEY(Rank);
        IF NOT TempCostPlusPurchLines_L.FINDSET THEN
          EXIT;

        CostPlusPurchLines_L.INIT;
        CostPlusPurchLines_L := TempCostPlusPurchLines_L;
        CostPlusPurchLines_L."Final Price" := CostPlusPurchLines_L."Base Value" + CostPlusPurchLines_L."Cost Plus Amount";
        CostPlusPurchLines_L.INSERT(TRUE);

        PreviousFinalPrice_L := CostPlusPurchLines_L."Final Price";
        TotalDiscount_L += CostPlusPurchLines_L."Cost Plus Amount";


        IF TempCostPlusPurchLines_L.NEXT <> 0 THEN REPEAT
          CostPlusPurchLines_L.INIT;
          CostPlusPurchLines_L := TempCostPlusPurchLines_L;
          //ligne ci-dessous incoh'rente, . supprimer une fois s-r
          {IF CostPlusList_L."Valuation Type" = CostPlusList_L."Valuation Type"::"Unit Amount" THEN
            CostPlusPurchLines_L.VALIDATE("Cost Plus Amount",

             ("No. Doc Extraction" <> '') AND ("No. Line Doc Extraction" <> 0)) THEN BEGIN

                TempCostPlusPurchLines_L.VALIDATE("Document Line No.", "Line No.");

        PreviousFinalPrice_L := CostPlusPurchLines_L."Final Price";
        TotalDiscount_L += CostPlusPurchLines_L."Cost Plus Amount";

          //ligne ci-dessous incoh'rente, . supprimer une fois s-r
          {IF CostPlusList_L."Valuation Type" = CostPlusList_L."Valuation Type"::"Unit Amount" THEN

                                       ROUND(CostPlusPurchLines_L."Base Value" * CostPlusPurchLines_L."Cost Plus %" /100, 0.01, '='));

          CostPlusPurchLines_L.INSERT(TRUE);

          PreviousFinalPrice_L := CostPlusPurchLines_L."Final Price";
          TotalDiscount_L += CostPlusPurchLines_L."Cost Plus Amount";




        UNTIL TempCostPlusPurchLines_L.NEXT = 0;

      END;
