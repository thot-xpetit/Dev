    PROCEDURE CleanPhoneNo@8062003(_PhoneNo@8062000 : Text[30];_CountryCode@8062001 : Code[10]) _CleanedPhoneNo : Text[30];
    BEGIN
      _CleanedPhoneNo := '+';

      _PhoneNo := DELCHR(_PhoneNo, '<', ' ');

      IF      ('00' = COPYSTR(_PhoneNo, 1, 2)) THEN _PhoneNo := COPYSTR(_PhoneNo, 3)
      ELSE IF ('0'  = COPYSTR(_PhoneNo, 1, 1)) THEN _CleanedPhoneNo += FORMAT(GetCountryCallingCode(_CountryCode));

      _CleanedPhoneNo += DELCHR(DELCHR(_PhoneNo, '=', DELCHR(_PhoneNo, '=', '0123456789')), '<', '0');

      IF '+' = _CleanedPhoneNo THEN EXIT('');
    END;

    END;

    VAR
      _TodayText@8062001 : TextConst 'ENU=Today;FRA=Aujourd''hui';
      _YesterdayText@8062002 : TextConst 'ENU=Yesterday;FRA=Hier';
      _DaysAgoText@8062003 : TextConst 'ENU=%1 days ago;FRA=Il y a %1 jours';
      _WeekAgoText@8062005 : TextConst 'ENU=Last week;FRA=La semaine derniSre';
      _WeeksAgoText@8062004 : TextConst 'ENU=%1 weeks ago;FRA=Il y a %1 semaines';
      _MonthAgoText@8062006 : TextConst 'ENU=Last month;FRA=Le mois dernier';
      _MonthsAgoText@8062007 : TextConst 'ENU=%1 months ago;FRA=Il y a %1 mois';
      _YearAgoText@8062008 : TextConst 'ENU=Last year;FRA=L''ann'e derniSre';
      _YearsAgoText@8062009 : TextConst 'ENU=%1 years ago;FRA=Il y a %1 ans';
      _LongTimeAgoText@8062011 : TextConst 'ENU=Long time ago;FRA=Il y a longtemps';
    BEGIN
      IF TODAY >= _Date THEN BEGIN

        IF TODAY = _Date               THEN EXIT(_TodayText);
        IF TODAY - 1 = _Date           THEN EXIT(_YesterdayText);
        IF CALCDATE('<CW-1W>') < _Date THEN EXIT(STRSUBSTNO(_DaysAgoText, TODAY - _Date));
        IF CALCDATE('<CW-2W>') < _Date THEN EXIT(_WeekAgoText);
        IF CALCDATE('<CM-1M>') < _Date THEN EXIT(STRSUBSTNO(_WeeksAgoText, DATE2DWY(TODAY, 2) - DATE2DWY(_Date, 2)));
        IF CALCDATE('<CM-2M>') < _Date THEN EXIT(_MonthAgoText);
        IF CALCDATE('<CY-1Y>') < _Date THEN EXIT(STRSUBSTNO(_MonthsAgoText, DATE2DMY(TODAY, 2) - DATE2DMY(_Date, 2)));
        IF CALCDATE('<CY-2Y>') < _Date THEN EXIT(_YearAgoText);
        IF CALCDATE('<CY-6Y>') < _Date THEN EXIT(STRSUBSTNO(_YearsAgoText, DATE2DMY(TODAY, 3) - DATE2DMY(_Date, 3)));
                                            EXIT(_LongTimeAgoText);

      END ELSE                              EXIT(FORMAT(_Date));
    END;

    PROCEDURE TimeUserFriendly@8062001(_Time@8062000 : Time) : Text[1024];

      _NowText@8062001 : TextConst 'ENU=Just now;FRA=Il y a un instant';

      _HoursAgoText@8062005 : TextConst 'ENU=%1 hours ago;FRA=Il y a %1 heures';
    BEGIN
      IF TIME >= _Time THEN BEGIN

        IF TIME - 300000 < _Time  THEN EXIT(_NowText);
        IF TIME - 3300000 < _Time THEN EXIT(STRSUBSTNO(_MinutesAgoText, ROUND((TIME - _Time) / 60000, 1, '<')));
        IF TIME - 6300000 < _Time THEN EXIT(_HourAgoText);
                                       EXIT(STRSUBSTNO(_HoursAgoText, ROUND((TIME - _Time + 900000) / 3600000, 1, '<')));

      END ELSE                         EXIT(FORMAT(_Time));
    END;
