    VAR
      MailServices_G@1000000000 : Automation "{837703CD-59DD-45D3-B3B4-A7EB1FEC8260} 1.0:{76727E59-9512-4D85-9B93-36E3D39C7707}:'BusinessSolutions3LI.MailServices'.MailService";

    PROCEDURE NotifyObjectChange@8053003();
    VAR
      RecCMCObjectTracking_L@8053001 : Record 8053702;
      RecCMCObjectTracking2_L@8053002 : Record 8053702;
      RecCMCObjectMgtSetup_L@8053003 : Record 8053701;
      ErrorMailServices_L@8053004 : Text[1024];
      SendSmtp_L@8053005 : Boolean;
    BEGIN
      //Envoi un mail de notification des objets modifi's

      RecCMCObjectTracking_L.SETCURRENTKEY("Notify Object Modified");
      RecCMCObjectTracking_L.SETRANGE("Notify Object Modified",TRUE);
      RecCMCObjectTracking_L.LOCKTABLE;
      IF RecCMCObjectTracking_L.FINDSET THEN
        IF ISCLEAR(MailServices_G) THEN BEGIN
          RecCMCObjectMgtSetup_L.GET;
          CREATE(MailServices_G);
          // Adresse du serveur SMTP
          MailServices_G.HostAddress := RecCMCObjectMgtSetup_L."SMTP Server";
          // Authentification au serveur SMTP
          MailServices_G.UseCredentials(RecCMCObjectMgtSetup_L."SMTP User",RecCMCObjectMgtSetup_L."SMTP Password",'');
          // Adresse mail emetteur
          MailServices_G.SetFrom(RecCMCObjectMgtSetup_L."SMTP Email From",'');
          // Adresses mail destinataires
          MailServices_G.AddRecipient(RecCMCObjectMgtSetup_L."SMTP Email To",'');
          // Sujet du mail
          MailServices_G.Subject := COMPANYNAME + ' : Notification modification d''objet';
          // Corps du mail
          MailServices_G.AppendHtmlBody(
            '<div style="font-family: Arial; font-size: x-small";><H1 style="font-size: small;">Liste des objets modifi's</H1>',TRUE);
          MailServices_G.AppendHtmlBody(
            '<table border="1" style="font-family: Arial; font-size: x-small";><tr><th>Type objet</th><th>ID objet</th>' +
            '<th>Nom objet</th><th>Modifi'</th><th>Date</th><th>Heure</th><th>Type modification</th></tr>',
            TRUE);
          REPEAT
            MailServices_G.AppendHtmlBody(
              '<tr><td>' + FORMAT(RecCMCObjectTracking_L."Object Type") + '</td>' +
              '<td>' + FORMAT(RecCMCObjectTracking_L."Object ID") + '</td>' +
              '<td>' + FORMAT(RecCMCObjectTracking_L."Object Caption") + '</td>' +
              '<td>' + FORMAT(RecCMCObjectTracking_L."Object Modified") + '</td>' +
              '<td>' + FORMAT(RecCMCObjectTracking_L."Object Date") + '</td>' +
              '<td>' + FORMAT(RecCMCObjectTracking_L."Object Time") + '</td>' +
              '<td>' + FORMAT(RecCMCObjectTracking_L."Type Object Modified") + '</td></tr>',
              TRUE);
            RecCMCObjectTracking2_L.GET(RecCMCObjectTracking_L."Object Type",RecCMCObjectTracking_L."Object ID");
            RecCMCObjectTracking2_L."Notify Object Modified" := FALSE;
            IF RecCMCObjectTracking2_L."Type Object Modified" < RecCMCObjectTracking2_L."Type Object Modified"::Delete THEN BEGIN
              RecCMCObjectTracking2_L."Type Object Modified" := 0;
              RecCMCObjectTracking2_L.MODIFY;
            END ELSE
              RecCMCObjectTracking2_L.DELETE;
          UNTIL RecCMCObjectTracking_L.NEXT = 0;

          // Envoi du mail
          IF NOT MailServices_G.SendSmtp THEN
            ERROR(MailServices_G.LastErrorDescription);
        END;
    END;

    PROCEDURE ClearMailServices@1000000001();
    BEGIN
      // Lib'ration de l'instance COM
      CLEAR(MailServices_G);
    END;

    }
    END.
