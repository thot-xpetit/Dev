    PROCEDURE ProcessContact@1000000001(VAR contact@1000000000 : Record 5050);
    VAR
      SalesRevenueCalcSetup@1000000001 : Record 8053661;
      LastAmountSalesRevenue@1000000002 : Decimal;
      SavedSalesPersonCode@1000000003 : Code[10];
      ContactModified@1000000004 : Boolean;
    BEGIN
      // <-- CLA220216
      ContactModified := FALSE;
      SavedSalesPersonCode := contact."Salesperson Code";
      // --> CLA220216

      IF (NOT DebugMode) THEN BEGIN
        contact."Salesperson Code" := '';
      END;
      WriteFile('#' + contact."No." + '#');

      IF (SalesRevenueCalcSetup.FINDFIRST()) THEN BEGIN
        LastAmountSalesRevenue := 0;
        REPEAT
          CalcSalesRevenueContact(contact, SalesRevenueCalcSetup, LastAmountSalesRevenue, ContactModified);
        UNTIL (SalesRevenueCalcSetup.NEXT() = 0);
        IF (NOT DebugMode) THEN BEGIN
          // <-- CLA220216
          IF (NOT ContactModified) THEN BEGIN
            contact."Salesperson Code" := SavedSalesPersonCode;
          END;
          // --> CLA220216
          contact.MODIFY(TRUE);
          UpdateCustomer(contact);
        END ELSE BEGIN
          ToolsFile.LogWriteInfo(STRSUBSTNO(tNoChange, contact."No.", SavedSalesPersonCode));
        END;
      END ELSE BEGIN
        ToolsFile.LogWriteInfo(tSetupMissing);
      END;
    END;

    PROCEDURE CalcSalesRevenueContact@1000000002(VAR contact@1000000000 : Record 5050;VAR SaleRevenueCalcSetup@1000000001 : Record 8053661;VAR LastAmountSalesRevenue@1000000004 : Decimal;VAR ModifiedContact@1000000005 : Boolean);
    VAR
      CtcSalesPerson@1000000002 : Record 8053160;
      SalesRevenue@1000000003 : Decimal;
    BEGIN
      // Get The Appropriate TC from Setup Calculation and contact
      CtcSalesPerson.SETRANGE("Contact Code", contact."No.");
      CtcSalesPerson.SETRANGE("Prod. Activity Code", SaleRevenueCalcSetup."Prod. Activity Code");
      CtcSalesPerson.SETRANGE("Sales Network Code", SaleRevenueCalcSetup."Sales Network Code");
      IF (CtcSalesPerson.FINDFIRST()) THEN BEGIN
        // Has a TC
        SalesRevenue := GetSalesRevenue(contact, SaleRevenueCalcSetup);
        IF ((SalesRevenue > LastAmountSalesRevenue) AND
           (SalesRevenue > SaleRevenueCalcSetup."Minimum Amount Sales Revenue")) THEN BEGIN
           contact."Salesperson Code" := CtcSalesPerson."Salesperson Code";
           LastAmountSalesRevenue := SalesRevenue;
           // <-- CLA220216
           ModifiedContact := TRUE;
           // --> CLA220216
           WriteFile(contact."No." + '#' +
                     contact.Name + '#' +
                     contact."Salesperson Code" + '#' +
                     CtcSalesPerson."Prod. Activity Code" + '#' +
                     CtcSalesPerson."Sales Network Code" + '#' +
                     CtcSalesPerson."Salesperson Code" + '#' +
                     FORMAT(SalesRevenue) + '#' +
                     'TRUE');
        END ELSE BEGIN

        END;
      END ELSE BEGIN
        ToolsFile.LogWriteInfo(STRSUBSTNO(tNoTCFound, contact."No.",
                                                      SaleRevenueCalcSetup.FIELDCAPTION("Prod. Activity Code"),
                                                      SaleRevenueCalcSetup."Prod. Activity Code",
                                                      SaleRevenueCalcSetup.FIELDCAPTION("Sales Network Code"),
                                                      SaleRevenueCalcSetup."Sales Network Code"));
      END;
    END;

      SalesInvLine.SETRANGE("Posting Date", BeginPostingDate, EndPostingDate);
      SalesInvLine.SETRANGE(Type, SalesInvLine.Type::Item);

        SalesInvLine.SETFILTER("Species Code", SaleRevenueCalcSetup."Species Code Filter");
      END;
      IF (SalesInvLine.FINDFIRST()) THEN BEGIN
        REPEAT
          IF (DetailledDebugMode) THEN BEGIN
            WriteFile('Facture ' +
                      SalesInvLine."Document No." + ' ' +
                      FORMAT(SalesInvLine."Line No.") + ' ' +
                      FORMAT(SalesInvLine.Type) + ' ' +
                      SalesInvLine."No." + ' ' +
                      FORMAT(SalesInvLine.Quantity) + ' ' +
                      FORMAT(SalesInvLine."Amount Including VAT"));
          END;
          ReturnValue += SalesInvLine."Amount Including VAT";
        UNTIL (SalesInvLine.NEXT() = 0);
      END;
      // Get Sales Credit Memo
      SalesCrMemoLine.SETRANGE("Sell-to Customer No.", contact."Customer No.");
      SalesCrMemoLine.SETRANGE("Posting Date", BeginPostingDate, EndPostingDate);
      SalesCrMemoLine.SETRANGE(Type, SalesInvLine.Type::Item);

        SalesCrMemoLine.SETFILTER("Activity Code", SaleRevenueCalcSetup."Activity Code Filter");
      END;

        SalesCrMemoLine.SETFILTER("Family Code", SaleRevenueCalcSetup."Family Code Filter");
      END;
      IF (SaleRevenueCalcSetup."Sub-Family Code Filter" <> '') THEN BEGIN
        SalesCrMemoLine.SETFILTER("Sub-Family Code", SaleRevenueCalcSetup."Sub-Family Code Filter");
      END;
      IF (SaleRevenueCalcSetup."Species Code Filter" <> '') THEN BEGIN
        SalesCrMemoLine.SETFILTER("Species Code", SaleRevenueCalcSetup."Species Code Filter");
      END;
      IF (SalesCrMemoLine.FINDFIRST()) THEN BEGIN
        REPEAT
          IF (DetailledDebugMode) THEN BEGIN
            WriteFile('Avoir ' +
                      SalesCrMemoLine."Document No." + ' ' +
                      FORMAT(SalesCrMemoLine."Line No.") + ' ' +
                      FORMAT(SalesCrMemoLine.Type) + ' ' +
                      SalesCrMemoLine."No." + ' ' +
                      FORMAT(SalesCrMemoLine.Quantity) + ' ' +
                      FORMAT(SalesCrMemoLine."Amount Including VAT"));
          END;
          ReturnValue -= SalesCrMemoLine."Amount Including VAT";
        UNTIL (SalesCrMemoLine.NEXT() = 0);
      END;
    END;
