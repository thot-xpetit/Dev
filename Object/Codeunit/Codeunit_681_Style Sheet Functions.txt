    PROCEDURE RemoveStyleSheetLinks@14(StyleSheetHeader@1000 : Record 680;StyleSheetType@1001 : 'Form,Page';ShowPrompt@1003 : Boolean;RemoveStyleSheet@1005 : Boolean);
    VAR
      StyleSheetLink@1002 : Record 685;
      StyleSheet@1004 : Record 2000000066;
    BEGIN
      IF ShowPrompt THEN
        IF NOT CONFIRM(Text015,FALSE) THEN
          EXIT;

      StyleSheetLink.SETCURRENTKEY("Template Code","Object Type",StyleSheetLink."Version Date",StyleSheetLink."Version No.");
      StyleSheetLink.SETRANGE("Template Code",StyleSheetHeader.Code);

      IF StyleSheetType = StyleSheetType::Form THEN
        StyleSheetLink.SETRANGE("Object Type",StyleSheetLink."Object Type"::Form);
      IF StyleSheetType = StyleSheetType::Page THEN
        StyleSheetLink.SETRANGE("Object Type",StyleSheetLink."Object Type"::Page);

      StyleSheetLink.SETFILTER("Version Date",'..%1',TODAY);

      IF RemoveStyleSheet THEN
        StyleSheetLink.SETFILTER("Version No.",'<=%1',StyleSheetHeader."Version No.")
      ELSE
        StyleSheetLink.SETFILTER("Version No.",'<%1',StyleSheetHeader."Version No.");

      IF StyleSheetLink.FINDSET THEN BEGIN
        REPEAT
          IF StyleSheet.GET(StyleSheetLink."Style Sheet ID") THEN
            StyleSheet.DELETE;
        UNTIL (StyleSheetLink.NEXT = 0);
        StyleSheetLink.DELETEALL;
      END;
    END;

    PROCEDURE ExportDefinition@8(StyleSheetCode@1000 : Code[20]);
    VAR
      DefinitionFilename@1001 : Text[260];
      Action@1002 : Integer;
      FilterString@1003 : Text[250];
      DefaultFileType@1004 : Integer;
      DefaultFileName@1005 : Text[260];
      WindowTitle@1006 : Text[30];
      FilePtr@1007 : File;
      OutStream@1008 : OutStream;
      StyleSheetHeader@1009 : Record 680;
      NamePos@1010 : Integer;
    BEGIN
      WindowTitle := 'Style Sheet Definition Files';
      DefaultFileName := '';
      DefaultFileType := 4;
      FilterString := Text013;
      Action := 1;

      DefinitionFilename := OpenFile(WindowTitle,DefaultFileName,DefaultFileType,FilterString,Action);

      IF DefinitionFilename <> '' THEN BEGIN
        IF STRPOS(DefinitionFilename,'.') = 0 THEN
          DefinitionFilename += '.XML';

        FilePtr.CREATE(DefinitionFilename);
        FilePtr.CREATEOUTSTREAM(OutStream);

        StyleSheetHeader.SETRANGE(Code,StyleSheetCode);
        XMLPORT.EXPORT(XMLPORT::"Style Sheet Definition",OutStream,StyleSheetHeader);
        FilePtr.CLOSE;

        StyleSheetHeader.GET(StyleSheetCode);

        IF STRPOS(DefinitionFilename,'.') <> 0 THEN BEGIN
          NamePos := STRLEN(DefinitionFilename);
          WHILE DefinitionFilename[NamePos] <> '.' DO
            NamePos -= 1;
        END;

        IF CONFIRM(Text010,TRUE) THEN
          ExportMailMerge(StyleSheetHeader,COPYSTR(DefinitionFilename,1,NamePos) + 'DOC');
      END;
    END;

    PROCEDURE ImportDefinition@7(VAR StyleSheetHeader@1000 : Record 680);
    VAR
      ImportStyleSheetDefinition@1001 : XMLport 680;
      StyleSheetHeaderCode@1002 : Code[20];
      DefinitionFilename@1003 : Text[260];
      Action@1004 : Integer;
      FilterString@1005 : Text[250];
      DefaultFileType@1006 : Integer;
      DefaultFileName@1007 : Text[260];
      WindowTitle@1008 : Text[30];
      FilePtr@1009 : File;
      InStream@1010 : InStream;
      StyleSheetMgt@1011 : Codeunit 680;
    BEGIN
      WindowTitle := 'Style Sheet Definition Files';
      DefaultFileName := '';
      DefaultFileType := 4;
      FilterString := Text013;
      Action := 0;

      DefinitionFilename := OpenFile(WindowTitle,DefaultFileName,DefaultFileType,FilterString,Action);

      IF DefinitionFilename <> '' THEN BEGIN
        FilePtr.OPEN(DefinitionFilename);
        FilePtr.CREATEINSTREAM(InStream);

        ImportStyleSheetDefinition.SETSOURCE(InStream);
        ImportStyleSheetDefinition.IMPORT;

        FilePtr.CLOSE;

        StyleSheetHeaderCode := ImportStyleSheetDefinition.GetStylesheetHeaderCode;
        StyleSheetHeader.GET(StyleSheetHeaderCode);

        IF CONFIRM(Text011,TRUE) THEN
          ImportMailMerge(StyleSheetHeader);

        StyleSheetHeader.CALCFIELDS("Mail Merge Document");

        IF StyleSheetHeader."Mail Merge Document".HASVALUE THEN
          IF CONFIRM(Text012,TRUE) THEN
            StyleSheetMgt.ConvertMailMergeToStyleSheet(StyleSheetHeader);
      END;
    END;

    PROCEDURE StdName@24(TempName@1000 : Text[103];MailMerge@1001 : Boolean) : Text[50];
    VAR
      ICount@1003 : Integer;
      TempNamePct1@1005 : Text[50];
      TempNamePct2@1006 : Text[50];
      Pos@1007 : Integer;
      InvalidCharList@1008 : Text[30];
    BEGIN
      InvalidCharList := ' .!@#$^&*()_+-={}[]|\/';          // % will be replaced by Pct

      IF MailMerge THEN
        InvalidCharList := COPYSTR(InvalidCharList,2);

      Pos := STRPOS(TempName,'%');
      IF Pos > 0 THEN BEGIN
        TempNamePct1 := COPYSTR(TempName,1,Pos - 1);
        TempNamePct2 := COPYSTR(TempName,Pos + 1);
        TempName := TempNamePct1 + 'Pct' + TempNamePct2;
      END;

      FOR ICount := 1 TO STRLEN(InvalidCharList) DO
        TempName := DELCHR(TempName,'=',FORMAT(InvalidCharList[ICount]));

      IF MailMerge THEN
        TempName := CONVERTSTR(TempName,' ','_');

      EXIT(TempName);
    END;

    PROCEDURE OpenFile@6(WindowTitle@1000 : Text[50];DefaultFileName@1001 : Text[260];DefaultFileType@1002 : ' ,Text,Excel,Word,Custom';FilterString@1003 : Text[250];Action@1004 : 'Open,Save') : Text[260];
    VAR
      CommonDialogControl@1005 : OCX "{F9043C85-F6F2-101A-A3C9-08002B2F49FB}:'Microsoft Common Dialog Control, version 6.0'";
      FilenameFound@1006 : Boolean;
      Pos@1007 : Integer;
      ParseFilename@1008 : Text[260];
    BEGIN
      // Common Dialog Value Definitions:
      // DefaultFileType values => 0 = '', 1 = Text, 2 = Excel, 3 = Word, 4 = Custom
      // Action values => 0 = Open, 1 = Save

      Pos := STRPOS(DefaultFileName,'\');
      ParseFilename := DefaultFileName;
      IF Pos <> 0 THEN BEGIN
        WHILE NOT FilenameFound DO BEGIN
          ParseFilename := COPYSTR(ParseFilename,Pos + 1);
          Pos := STRPOS(ParseFilename,'\');
          FilenameFound := (Pos = 0);
        END;
        CommonDialogControl.FileName := ParseFilename;
        DefaultFileName := COPYSTR(DefaultFileName,1,STRLEN(DefaultFileName) - STRLEN(ParseFilename));
      END ELSE
        CommonDialogControl.FileName := DefaultFileName;
      CommonDialogControl.DialogTitle := WindowTitle;

      IF DefaultFileType = DefaultFileType::Custom THEN BEGIN
        IF STRPOS(UPPERCASE(FilterString),'.DOC') > 0 THEN
          DefaultFileType := DefaultFileType::Word;
        IF STRPOS(UPPERCASE(FilterString),'.XL') > 0 THEN
          DefaultFileType := DefaultFileType::Excel;
      END;

      CASE DefaultFileType OF
        DefaultFileType::Text:
          CommonDialogControl.Filter := Text003;
        DefaultFileType::Excel:
          CommonDialogControl.Filter := Text004;
        DefaultFileType::Word:
          CommonDialogControl.Filter := Text005;
        DefaultFileType::Custom:
          CommonDialogControl.Filter := FilterString;
      END;

      IF STRPOS(DefaultFileName,'\') <> 0 THEN
        CommonDialogControl.InitDir := DefaultFileName;

      IF Action = Action::Open THEN
        CommonDialogControl.ShowOpen
      ELSE
        CommonDialogControl.ShowSave;

      EXIT(CommonDialogControl.FileName);
    END;
