OBJECT Codeunit 95002 Update
{
  OBJECT-PROPERTIES
  {
    Date=26/03/15;
    Heure=00:00:00;
    Liste versions=Update;
  }
  PROPERTIES
  {
    OnRun=VAR
            Selection@60000 : Integer;
          BEGIN
            CASE DIALOG.STRMENU('Etape 1,Etape 2',0) OF
              1 : Etape1;
              2 : Etape2;
            END;
          END;

  }
  CODE
  {

    PROCEDURE Etape1@60000();
    VAR
      RecTranInvMgt_L@60000 : Record 60017;
      RecT1_L@60001 : Record 90005;
    BEGIN
      CLEAR(RecTranInvMgt_L);
      IF RecTranInvMgt_L.FIND('-') THEN
      REPEAT
        RecT1_L.INIT;
        RecT1_L."Entry No." := IncrementEntryNo;
        RecT1_L.TRANSFERFIELDS(RecTranInvMgt_L);
        RecT1_L.INSERT;
        RecTranInvMgt_L.DELETE;
      UNTIL RecTranInvMgt_L.NEXT = 0;
      MESSAGE('Etape 1 termin'e');
    END;

    PROCEDURE Etape2@60001();
    VAR
      RecTranInvMgt_L@60001 : Record 60017;
      RecT1_L@60000 : Record 90005;
    BEGIN
      CLEAR(RecT1_L);
      IF RecT1_L.FIND('-') THEN
      REPEAT
        RecTranInvMgt_L.INIT;
        RecTranInvMgt_L.TRANSFERFIELDS(RecT1_L);
        RecTranInvMgt_L.INSERT;
      UNTIL RecT1_L.NEXT = 0;
      MESSAGE('Etape 2 termin'e');
    END;

    PROCEDURE IncrementEntryNo@60002() Res_EntryNo : Integer;
    VAR
      RecT1_L@60000 : Record 90005;
    BEGIN
      CLEAR(RecT1_L);
      IF RecT1_L.FIND('+') THEN
        Res_EntryNo := RecT1_L."Entry No." + 1
      ELSE Res_EntryNo := 1;
    END;

    BEGIN
    END.
  }
}
