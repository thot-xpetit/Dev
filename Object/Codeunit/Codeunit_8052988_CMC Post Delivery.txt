    PROCEDURE CreateItemJnlLinesForTransfer@8053009(RecPostedDelivery_P@8053000 : Record 8053014;PostingDate_P@8053001 : Date) Res_EntryNo : Integer;
    VAR
      AppliedQty_L@8053003 : Decimal;
      RecItemLdgEntry_L@8053004 : Record 32;
      RecOldItemLdgEntry_L@8053005 : Record 32;
      RecItem_L@8053006 : Record 27;
      EntryFindMethod_L@8053007 : Text[1];
      FirstApplication_L@8053008 : Boolean;
      TotalRemQty_L@8053010 : Decimal;
    BEGIN
      //Splitter lignes de transfert en fonction du stock disponible
      AppliedQty_L := RecPostedDelivery_P."Standard Weight";
      FirstApplication_L := TRUE;
      RecItemLdgEntry_L.SETCURRENTKEY("Item No.",Open,"Variant Code",Positive,"Location Code","Posting Date");
      RecItemLdgEntry_L.SETRANGE("Item No.",RecPostedDelivery_P."Item No.");
      RecItemLdgEntry_L.SETRANGE(Open,TRUE);
      RecItemLdgEntry_L.SETRANGE(Positive,TRUE);
      RecItemLdgEntry_L.SETRANGE("Location Code",RecPostedDelivery_P."Location Code");
      IF RecItemLdgEntry_L.FINDSET THEN
        REPEAT
          RecItemLdgEntry_L.CALCFIELDS("Reserved Quantity");
          TotalRemQty_L += RecItemLdgEntry_L."Remaining Quantity" - RecItemLdgEntry_L."Reserved Quantity"
        UNTIL RecItemLdgEntry_L.NEXT = 0;
      IF AppliedQty_L > TotalRemQty_L THEN
        ERROR(Text8053007,RecPostedDelivery_P."Item No.");
      RecItem_L.GET(RecPostedDelivery_P."Item No.");
      REPEAT
        IF FirstApplication_L THEN BEGIN
          FirstApplication_L := FALSE;
          IF RecItem_L."Costing Method" = RecItem_L."Costing Method"::LIFO THEN
            EntryFindMethod_L := '+'
          ELSE
            EntryFindMethod_L := '-';
          RecItemLdgEntry_L.FIND(EntryFindMethod_L);
        END ELSE
          CASE EntryFindMethod_L OF
            '-':
              IF RecItemLdgEntry_L.NEXT = 0 THEN
                EXIT;
            '+':
              IF RecItemLdgEntry_L.NEXT(-1) = 0 THEN
                EXIT;
          END;
        RecOldItemLdgEntry_L.COPY(RecItemLdgEntry_L);
        RecOldItemLdgEntry_L.CALCFIELDS("Reserved Quantity");
        //L''criture article . lettrer couvre la quantit' . transf'rer
        IF AppliedQty_L <= RecOldItemLdgEntry_L."Remaining Quantity" - RecOldItemLdgEntry_L."Reserved Quantity" THEN BEGIN
          //Validation du transfert sur la quantit' totale de l''criture article . lettrer
          Res_EntryNo := PostItemJnlLinesForTransfer(RecPostedDelivery_P,AppliedQty_L,RecOldItemLdgEntry_L);
          AppliedQty_L := 0;
        END ELSE BEGIN
          //Validation du transfert sur la quantit' restante de l''criture . lettrer
          Res_EntryNo := PostItemJnlLinesForTransfer(RecPostedDelivery_P,
            RecOldItemLdgEntry_L."Remaining Quantity" - RecOldItemLdgEntry_L."Reserved Quantity",RecOldItemLdgEntry_L);
          //D'duction de la quantit' . lettrer
          AppliedQty_L := AppliedQty_L - (RecOldItemLdgEntry_L."Remaining Quantity" - RecOldItemLdgEntry_L."Reserved Quantity");
        END;
      UNTIL FALSE OR (AppliedQty_L <= 0);
    END;
