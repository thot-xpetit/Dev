        RecPurHeader_G."No." := Text8053018;

        RecPurHeader_G."Species Code" := CMCPostedReceipts_G."Species Code";

        RecPurHeader_G.VALIDATE("Document Date", DocDateReq_G);

          IF CollContract_G.GET(recPostedRecLine_P."Contract No.") THEN BEGIN
            // date nulle dans contrat ou date < date demand'e
            IF (CollContract_G."Due Date" = 0D) OR (CollContract_G."Due Date" <= ReceiptDate_G) THEN BEGIN
              CerealsInvSch_L.GET(CampaingCode_G, ReceiptDate_G); // on prends 'ch'ance planning
              RecPurHeader_G.VALIDATE("Due Date", CerealsInvSch_L."Due Date");

              CLEAR(DueDate_L);
              CerealsInvSch_L.RESET;
              CerealsInvSch_L.SETRANGE("Campaign No.", CampaingCode_G);
              CerealsInvSch_L.SETRANGE("Collect Ending Date", 0D, CollContract_G."Due Date"); // sur date 'ch'ance contrat
              IF CerealsInvSch_L.FINDLAST THEN
                RecPurHeader_G.VALIDATE("Due Date", CerealsInvSch_L."Due Date")

                CerealsInvSch_L.GET(CampaingCode_G, ReceiptDate_G);
                RecPurHeader_G.VALIDATE("Due Date", CerealsInvSch_L."Due Date");
              END;
            END;

              RecPurHeader_G.VALIDATE("Due Date", CollContract_G."Due Date");
          END;
          IF DueDate_G <> 0D THEN
            RecPurHeader_G.VALIDATE("Due Date", DueDate_G);
        END;
        CMCCreateCollectInvoiceOpt_G.InfoToHeader(RecPurHeader_G,recPostedRecLine_P); // 20151203-Wi00-ADV-2465-JLT01
        RecPurHeader_G.MODIFY(TRUE);

        NextNo_G := 10000;
        PreviousVendor_G := CMCPostedReceipts_G."Vendor No.";
        PreviousItem_G := recPostedRecLine_P."Item Contract No.";
        PreviousContrat_G := recPostedRecLine_P."Contract No.";

      recPostedReceiptsLine_L."Purchase Document No." := RecPurHeader_G."No.";

      // --------------------------------------------------------------------

      // Cr'ation Ligne Facture pour Article collect' SI NON FACTUREE
      // --------------------------------------------------------------------
      IF (PreviousVariety_G = '') OR ((PreviousVariety_G = VNR_G) AND  (recPostedRecLine_P."Variety Code"<>'')) THEN BEGIN
        CLEAR(RecPurLine_G);
        RecPurLine_G.INIT;
        RecPurLine_G."Document Type" := RecPurHeader_G."Document Type";
        RecPurLine_G."Document No." := RecPurHeader_G."No.";
        RecPurLine_G."Line No." := NextNo_G;
        RecPurLine_G.Description := '';
        NextNo_G += 10000;
        RecPurLine_G.INSERT(TRUE);
        RecPurLine_G.INIT;
        RecPurLine_G."Document Type" := RecPurHeader_G."Document Type";
        RecPurLine_G."Document No." := RecPurHeader_G."No.";
        RecPurLine_G."Line No." := NextNo_G;

        NextNo_G += 10000;
        RecPurLine_G.Type := RecPurLine_G.Type::" ";
        IF Variety_L.GET(recPostedRecLine_P."Variety Code") THEN BEGIN
          RecPurLine_G.Description := COPYSTR(STRSUBSTNO(Text8053017, Variety_L."Variety Code", Variety_L.Description), 1, 50);
          PreviousVariety_G := recPostedRecLine_P."Variety Code";
        END ELSE BEGIN
          RecPurLine_G.Description := Text8053016;
          PreviousVariety_G := VNR_G;
        END;
        // --> 20150312-CER15-CMC02.02.00-2820-FHS01
        //IF recPostedRecLine_P."VAT Bus. Posting Group" <> '' THEN
        //  RecPurLine_G.VALIDATE("VAT Bus. Posting Group", recPostedRecLine_P."VAT Bus. Posting Group");
        // <-- 20150312-CER15-CMC02.02.00-2820-FHS01

      RecPurLine_G.INSERT(TRUE);
      END;

      CLEAR(RecPurLine_G);
      RecPurLine_G.INIT;
      RecPurLine_G."Document Type" := RecPurHeader_G."Document Type";
      RecPurLine_G."Document No." := RecPurHeader_G."No.";
      RecPurLine_G."Line No." := NextNo_G;
      NextNo_G += 10000;
      RecPurLine_G.Type := RecPurLine_G.Type::" ";

      RecPurLine_G."Document No." := RecPurHeader_G."No.";

      RecPurLine_G.SETRANGE("Document No.", RecPurHeader_G."No.");

      recPostedRecLine_P.SETRANGE("Line No.");
      recPostedRecLine_P.SETRANGE("No.");
      RecPurLine_G.SETRANGE("Document Type", RecPurHeader_G."Document Type");
      RecPurLine_G.SETRANGE("Document No.", RecPurHeader_G."No.");
      IF RecPurLine_G.FINDLAST THEN
        NextNo_G := RecPurLine_G."Line No." + 10000
      ELSE
        NextNo_G := 10000;
    END;

    PROCEDURE CreateInvoice@8053005();
    VAR
      recPurchLineVariatPrice_L@8053000 : Record 8053005;
      recPostedReceiptsLine_L@8053001 : Record 8053104;
      DocDim_L@8053002 : Record 357;
    BEGIN
      RecPurHeaderInsert_G.INIT;
      RecPurHeaderInsert_G.COPY(RecPurHeader_G);
      RecPurHeaderInsert_G."Document Type" := RecPurHeaderInsert_G."Document Type"::Invoice;
      RecPurHeaderInsert_G."No." := '';
      RecPurHeaderInsert_G.SetHideCollectifChoice(TRUE);

      RecPurHeaderInsert_G.INSERT(TRUE);

      RecPurHeaderInsert_G.VALIDATE("Posting Date", PostingDateReq_G);
      RecPurHeaderInsert_G.VALIDATE("Document Date", DocDateReq_G);
      RecPurHeaderInsert_G.VALIDATE("Due Date", RecPurHeader_G."Due Date");
      RecPurHeaderInsert_G."Vendor Invoice No." := RecPurHeaderInsert_G."No.";

      RecPurHeaderInsert_G."Grouping Code" := GroupingCode_G;

      RecPurHeaderInsert_G.MODIFY(TRUE);

      CLEAR(RecPurLine_G);
      RecPurLine_G.RESET;
      RecPurLine_G.SETRANGE("Document Type", RecPurHeader_G."Document Type");
      RecPurLine_G.SETRANGE("Document No.", RecPurHeader_G."No.");
      IF RecPurLine_G.FINDSET(FALSE, FALSE) THEN
        REPEAT
          RecPurLineInsert_G.INIT;
          RecPurLineInsert_G.TRANSFERFIELDS(RecPurLine_G);
          RecPurLineInsert_G."Document No." := RecPurHeaderInsert_G."No.";
          RecPurLineInsert_G.INSERT(TRUE);
        UNTIL RecPurLine_G.NEXT=0;

      recPurchLineVariatPrice_G.RESET;
      recPurchLineVariatPrice_G.SETRANGE("Document Type", recPurchLineVariatPrice_G."Document Type"::Invoice);
      recPurchLineVariatPrice_G.SETFILTER("Document No.", RecPurHeader_G."No.");
      IF recPurchLineVariatPrice_G.FINDSET(FALSE, FALSE) THEN
        REPEAT
          recPurchLineVariatPrice_L.INIT;
          recPurchLineVariatPrice_L.TRANSFERFIELDS(recPurchLineVariatPrice_G);
          recPurchLineVariatPrice_L."Document Type" := recPurchLineVariatPrice_G."Document Type"::Invoice;
          recPurchLineVariatPrice_L."Document No." := RecPurHeaderInsert_G."No.";
          recPurchLineVariatPrice_L.INSERT(TRUE);
        UNTIL recPurchLineVariatPrice_G.NEXT = 0;

      TempDocDim_G.RESET;
      TempDocDim_G.SETRANGE("Table ID", 39);
      TempDocDim_G.SETRANGE("Document Type", TempDocDim_G."Document Type"::Invoice);
      TempDocDim_G.SETRANGE("Document No.", RecPurHeader_G."No.");
      IF TempDocDim_G.FINDSET THEN
        REPEAT
          DocDim_L.INIT;
          DocDim_L.TRANSFERFIELDS(TempDocDim_G);
          DocDim_L.VALIDATE("Table ID", 39);
          DocDim_L.VALIDATE("Document Type", DocDim_L."Document Type"::Invoice);
          DocDim_L.VALIDATE("Document No.", RecPurHeaderInsert_G."No.");
          DocDim_L.VALIDATE("Line No." , TempDocDim_G."Line No.");
          DocDim_L.VALIDATE("Dimension Code", TempDocDim_G."Dimension Code");
          DocDim_L.VALIDATE("Dimension Value Code", TempDocDim_G."Dimension Value Code");
          DocDim_L.INSERT(TRUE);
        UNTIL TempDocDim_G.NEXT = 0;
        TempDocDim_G.DELETEALL(TRUE);

      // --------------------------------------------------------------------
      // Mise . jour de la r'ception Valid'e
      // --------------------------------------------------------------------
      recPostedReceiptsLine_L.SETCURRENTKEY("Purchase Document No.");
      recPostedReceiptsLine_L.SETRANGE("Purchase Document No.", RecPurHeader_G."No.");
      recPostedReceiptsLine_L.MODIFYALL("Purchase Document Type", RecPurHeaderInsert_G."Document Type", FALSE);
      recPostedReceiptsLine_L.MODIFYALL("Purchase Document No.", RecPurHeaderInsert_G."No.", FALSE);
      // --------------------------------------------------------------------

      recPurchLineVariatPrice_G.DELETEALL(TRUE);
      RecPurHeader_G.DELETE(TRUE);
      CLEAR(RecPurHeader_G);

      InvoiceCount_G := InvoiceCount_G + 1;
    END;
