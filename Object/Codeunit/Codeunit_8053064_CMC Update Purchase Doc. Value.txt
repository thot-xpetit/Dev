            RecPurchBlanketLineDetails_G.RESET;

            RecPurchBlanketLineDetails_G.SETRANGE("Blanket Order Line No.",RecPurchBlanketLine_G."Line No.");

            RecPurchBlanketLineDetails_G.SETFILTER("Ending Date", '>=%1',RecPurchaseHeader_G."Posting Date");
            IF RecPurchBlanketLineDetails_G.FINDFIRST THEN
              VALIDATE("Direct Unit Cost",RecPurchBlanketLineDetails_G."Price/Cost")
            ELSE
              VALIDATE("Direct Unit Cost",RecPurchBlanketLine_G."Direct Unit Cost");
            MODIFY;
          END;
        END;
        // --> 20200113-WI00-ADV01.01.00-3446-SCR01
        CLEAR(CuACPManagement_G);
        CuACPManagement_G.ManageAcpPurchLine(RecPurchaseLine_G);
        CLEAR(CuCostPlusManagement_G);
        CuCostPlusManagement_G.ManageCostPlusPurchLine(RecPurchaseLine_G);
        // <-- 20200113-WI00-ADV01.01.00-3446-SCR01
      END;
    END;
