    LOCAL PROCEDURE ParseSkipBlanks@1101257006(VAR APos@1101257001 : Integer;AMacro@1101257000 : Text[1024]);

    LOCAL PROCEDURE ParseDecimalLevel1@1101257000(VAR AResult@1101257005 : Decimal;VAR AParsePos@8053001 : Integer;AMacro@8053000 : Text[1024]);

          '+': AResult := AResult + fltResult;

        END;
      END;
    END;

      WHILE COPYSTR(AMacro, AParsePos, 1) IN ['*', '/','C'] DO BEGIN

      END;

    END;
