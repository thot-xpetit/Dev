      L_recInterventionJnlTempl.SETRANGE("Form ID",P_FormID);

      IF UserSetup_L.GET(USERID) THEN BEGIN
        IF UserSetup_L."Stock Resp. Ctr. Filter" <> '' THEN
          L_recInterventionJnlTempl.SETRANGE("Responsibility Center",UserSetup_L."Stock Resp. Ctr. Filter");
      END;

      CASE L_recInterventionJnlTempl.COUNT OF
        0:
          IF L_recInterventionJnlTempl.GETFILTER("Responsibility Center") <> '' THEN
            ERROR(Text004,L_recInterventionJnlTempl.GETFILTER("Responsibility Center"))
          ELSE BEGIN
            L_recInterventionJnlTempl.INIT;
            L_recInterventionJnlTempl.Name := Text000;
            L_recInterventionJnlTempl.Description := Text001;
            L_recInterventionJnlTempl.VALIDATE("Form ID");
            L_recInterventionJnlTempl.INSERT;
            COMMIT;
          END;
        1:
          L_recInterventionJnlTempl.FIND('-');
        ELSE
          P_JnlSelected := FORM.RUNMODAL(0,L_recInterventionJnlTempl) = ACTION::LookupOK;
      END;
      IF P_JnlSelected THEN BEGIN
        P_recInterventionJnlLine.FILTERGROUP := 2;
        P_recInterventionJnlLine.SETRANGE("Journal Template Name",L_recInterventionJnlTempl.Name);
        P_recInterventionJnlLine.FILTERGROUP := 0;
        IF G_OpenFromBatch THEN BEGIN
          P_recInterventionJnlLine."Journal Template Name" := '';
          FORM.RUN(L_recInterventionJnlTempl."Form ID",P_recInterventionJnlLine);
        END;
      END;
    END;

    PROCEDURE OpenJnlBatch@12(VAR p_recInterventionJnlBatch@1000 : Record 8053038);

      L_recInterventionJnlTemplate@1002 : Record 8053039;
      L_recInterventionJnlLine@1004 : Record 8053034;
      L_JnlSelected@1003 : Boolean;
    BEGIN
      IF p_recInterventionJnlBatch.GETFILTER("Journal Template Name") <> '' THEN
        EXIT;
      p_recInterventionJnlBatch.FILTERGROUP(2);
      IF p_recInterventionJnlBatch.GETFILTER("Journal Template Name") <> '' THEN BEGIN
        p_recInterventionJnlBatch.FILTERGROUP(0);
        EXIT;
      END;
      p_recInterventionJnlBatch.FILTERGROUP(0);

      IF NOT p_recInterventionJnlBatch.FIND('-') THEN BEGIN
        IF NOT L_recInterventionJnlTemplate.FIND('-') THEN
          TemplateSelection(0,L_recInterventionJnlLine,L_JnlSelected);
        IF L_recInterventionJnlTemplate.FIND('-') THEN
          CheckTemplateName(L_recInterventionJnlTemplate.Name,p_recInterventionJnlBatch.Name);
      END;
      p_recInterventionJnlBatch.FIND('-');
      L_JnlSelected := TRUE;
      IF p_recInterventionJnlBatch.GETFILTER("Journal Template Name") <> '' THEN
         L_recInterventionJnlTemplate.SETRANGE(Name,p_recInterventionJnlBatch.GETFILTER("Journal Template Name"));
      CASE L_recInterventionJnlTemplate.COUNT OF
        1:
          L_recInterventionJnlTemplate.FIND('-');
        ELSE
          L_JnlSelected := FORM.RUNMODAL(0,L_recInterventionJnlTemplate) = ACTION::LookupOK;
      END;
      IF NOT L_JnlSelected THEN
        ERROR('');

      p_recInterventionJnlBatch.FILTERGROUP(2);
      p_recInterventionJnlBatch.SETRANGE("Journal Template Name",L_recInterventionJnlTemplate.Name);
      p_recInterventionJnlBatch.FILTERGROUP(0);
    END;
