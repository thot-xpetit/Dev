OBJECT Codeunit 8053354 CMC Capital Jnl.-B.Post
{
  OBJECT-PROPERTIES
  {
    Date=30/10/14;
    Heure=00:00:00;
    Liste versions=CMC02.02.00;
  }
  PROPERTIES
  {
    TableNo=8053355;
    OnRun=BEGIN
            CapJnlBatch_G.COPY(Rec);
            Code;
            Rec.COPY(CapJnlBatch_G);
          END;

  }
  CODE
  {
    VAR
      Text8053000@1000 : TextConst 'ENU=Do you want to post the journals?;FRA=Souhaitez-vous valider les feuilles ?';
      Text8053001@1001 : TextConst 'ENU=The journals were successfully posted.;FRA=Les feuilles ont 't' valid'es avec succSs.';
      Text8053002@1002 : TextConst 'ENU="It was not possible to post all of the journals. ";FRA="Il n''a pas 't' possible de valider l''ensemble des feuilles. "';
      Text8053003@1003 : TextConst 'ENU=The journals that were not successfully posted are now marked.;FRA=Les feuilles qui n''ont pas 't' valid'es avec succSs sont maintenant marqu'es.';
      CapJnlTemplate_G@8053000 : Record 8053354;
      CapJnlBatch_G@8053001 : Record 8053355;
      CapJnlLine_G@8053002 : Record 8053356;
      CapJnlPostBatch_G@8053003 : Codeunit 8053353;
      JnlWithErrors_G@8053004 : Boolean;

    LOCAL PROCEDURE Code@1();
    BEGIN
      WITH CapJnlBatch_G DO
       BEGIN
        CapJnlTemplate_G.GET("Capital Journal Template");
        IF NOT CONFIRM(Text8053000,FALSE) THEN
          EXIT;
        FIND('-');
        REPEAT
          CapJnlLine_G."Capital Journal Template" := "Capital Journal Template";
          CapJnlLine_G."Capital Batch Name" := Name;
          CapJnlLine_G."Line No." := 1;
          CLEAR(CapJnlPostBatch_G);
          IF CapJnlPostBatch_G.RUN(CapJnlLine_G) THEN
            MARK(FALSE)
          ELSE BEGIN
            MARK(TRUE);
            JnlWithErrors_G := TRUE;
           END;
        UNTIL NEXT = 0;
        IF NOT JnlWithErrors_G THEN
          MESSAGE(Text8053001)
        ELSE
          MESSAGE(Text8053002 + Text8053003);
        IF NOT FIND('=><') THEN BEGIN
          RESET;
          FILTERGROUP(2);
          SETRANGE("Capital Journal Template","Capital Journal Template");
          FILTERGROUP(0);
          Name := '';
         END;
       END;
    END;

    BEGIN
    {
      // --> 3LI BEGIN
      20130222-FINA05-CMC02.02.00-WSA01 : OBJETCREATE
      // <-- 3LI END
    }
    END.
  }
}
