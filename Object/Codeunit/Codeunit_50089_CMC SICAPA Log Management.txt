    PROCEDURE AddMessage@1000000000(MessageID@1000000001 : GUID;ContextID@1000000002 : RecordRef;FromRecRef@1000000003 : RecordRef;FieldNo@1000000004 : Integer;Type@1000000005 : Integer;MessageText@1000000006 : Text[500]);
    VAR
      CMCSICAPAImportInvoiceLog@1000000000 : Record 60059;
      CMCSICAPAImportInvoiceLog2@1000000007 : Record 60059;
      LineNo@1000000008 : Integer;
    BEGIN
      CMCSICAPAImportInvoiceLog2.SETRANGE("Message ID",MessageID);
      IF CMCSICAPAImportInvoiceLog2.FINDLAST THEN
        LineNo := CMCSICAPAImportInvoiceLog2."Entry No.";
      LineNo += 1;

      CMCSICAPAImportInvoiceLog.INIT;
      CMCSICAPAImportInvoiceLog."Message ID" := MessageID;
      CMCSICAPAImportInvoiceLog."Entry No." := LineNo;
      CMCSICAPAImportInvoiceLog."Context ID" := ContextID.RECORDID;
      CMCSICAPAImportInvoiceLog."From RecordID" := FromRecRef.RECORDID;
      CMCSICAPAImportInvoiceLog.Type := Type;
      CMCSICAPAImportInvoiceLog."Field No." := FieldNo;
      CMCSICAPAImportInvoiceLog."Message Text" := COPYSTR(MessageText,1, MAXSTRLEN(CMCSICAPAImportInvoiceLog."Message Text"));
      IF STRLEN(MessageText) > MAXSTRLEN(CMCSICAPAImportInvoiceLog."Message Text") THEN
         CMCSICAPAImportInvoiceLog."Message Text" := COPYSTR(MessageText,501, MAXSTRLEN(CMCSICAPAImportInvoiceLog."Message Text"));
      CMCSICAPAImportInvoiceLog."User ID" := USERID;
      CMCSICAPAImportInvoiceLog."Log Datetime" := CURRENTDATETIME;
      CMCSICAPAImportInvoiceLog.INSERT();
    END;

    PROCEDURE DeleteAllMessage@1000000001(ContextVar@1000000000 : Variant);
    VAR
      CMCSICAPAImportInvoiceLog@1000000001 : Record 60059;
      FromRecordRef@1000000002 : RecordRef;
    BEGIN
      FromRecordRef.GETTABLE(ContextVar);
      CMCSICAPAImportInvoiceLog.SETCURRENTKEY("Context ID");
      CMCSICAPAImportInvoiceLog.SETRANGE("Context ID",FromRecordRef.RECORDID);
      IF NOT CMCSICAPAImportInvoiceLog.ISEMPTY THEN
         CMCSICAPAImportInvoiceLog.DELETEALL;
    END;

    BEGIN
    END.
