    PROCEDURE GetDiscountBaseValue@8053012(VAR ValueEntryDiscount_P@8053000 : Record 8053252;PrevFinalPrice_P@8053004 : Decimal) : Decimal;
    VAR
      ValueEntryDisc_L@8053001 : Record 8053252;
      PurchDisc_L@8053002 : Record 8053247;
      ValueEntry_L@8053003 : Record 5802;
    BEGIN
      //----------------------------------------------------------------------------------------------------------
      // renvoi le montant servant de base de calcul pour le calcul RFC
      //----------------------------------------------------------------------------------------------------------
      // Entr'es:
      //   - ValueEntryDiscount_P : La Value Entry Discount sur laquelle on travail
      //   - PrevFinalPrice_P : Eventuel Prix de la RFC pr'c'dente
      // Sorties:
      //   - Montant Base
      //----------------------------------------------------------------------------------------------------------


      PurchDisc_L.GET(ValueEntryDiscount_P."Discount Code");

      CASE PurchDisc_L."Disc. Calculation Type" OF
        PurchDisc_L."Disc. Calculation Type"::"Chain Discount" : BEGIN
          IF PrevFinalPrice_P <> 0 THEN
            EXIT(PrevFinalPrice_P)
          ELSE BEGIN
            ValueEntryDisc_L.RESET;
            ValueEntryDisc_L.SETCURRENTKEY("Value Entry No.",Rank);
            ValueEntryDisc_L.SETRANGE("Value Entry No.",ValueEntryDiscount_P."Value Entry No.");
            ValueEntryDisc_L.SETFILTER(Rank,'<=%1',ValueEntryDiscount_P.Rank);
            IF ValueEntryDisc_L.FINDLAST THEN
              EXIT(ValueEntryDisc_L."Final Price")
            ELSE BEGIN
              ValueEntry_L.GET(ValueEntryDiscount_P."Value Entry No.");
              EXIT(ValueEntry_L."Cost per Unit");
            END;
          END;
        END;
        PurchDisc_L."Disc. Calculation Type"::"Sum Discount" : BEGIN
          ValueEntry_L.GET(ValueEntryDiscount_P."Value Entry No.");
          EXIT(ValueEntry_L."Cost per Unit");
        END;
      END;

      EXIT(0);
    END;

    PROCEDURE UpdateDiscount@8053004(VAR ValueEntryDisc_P@8053000 : Record 8053252;ValuationType_P@8053001 : 'New ECD Amount,New ECD Percentage,Change ECD amount,Apply Purch. Discount';ModificationValue_P@8053002 : Decimal;PrevFinalPrice_P@8053003 : Decimal) : Decimal;
    VAR
      CMCPurchDiscValue_L@8053004 : Record 8053248;
    BEGIN
      CASE ValuationType_P OF
        ValuationType_P::"New ECD Amount" : BEGIN
          IF ValueEntryDisc_P."Discount Code" <> CMCSetup_G."Except. Purch. ECD Disc. Code" THEN BEGIN
            ValueEntryDisc_P."Discount Amount" := ModificationValue_P;
            ValueEntryDisc_P."Discount %" := 0;
            SetFinalPrice(ValueEntryDisc_P,PrevFinalPrice_P);
          END ELSE BEGIN
            IF UpdateType_G <> UpdateType_G::"Unapply Budgeted Unit Cost" THEN BEGIN
              IF ValueEntryDisc_P."Discount Amount" <> 0 THEN BEGIN
                IF ValueEntryDisc_P."Base Value" > ValueEntryDisc_P."Final Price" THEN
                  ValueEntryDisc_P."Discount Amount" := ValueEntryDisc_P."Base Value" - ValueEntryDisc_P."Final Price"
                ELSE
                  IF ValueEntryDisc_P."Base Value" < ValueEntryDisc_P."Final Price" THEN
                    ERROR(Text8053018,ValueEntryReference_G."Entry No.")
                  ELSE BEGIN
                    ValueEntryDisc_P."Discount %" := 0;
                    ValueEntryDisc_P."Discount Amount" := 0;
                  END;
              END ELSE BEGIN
                ValueEntryDisc_P."Discount Amount" := ModificationValue_P;
                SetFinalPrice(ValueEntryDisc_P,PrevFinalPrice_P);
              END;
            END ELSE BEGIN
              ValueEntryDisc_P."Discount %" := 0;
              ValueEntryDisc_P."Discount Amount" := 0;
              ValueEntryDisc_P."Final Price" := ValueEntryDisc_P."Base Value";
            END;
          END;
        END;
        ValuationType_P::"New ECD Percentage" : BEGIN
          ValueEntryDisc_P."Discount Amount" :=
                                   ROUND(ValueEntryDisc_P."Base Value" * ModificationValue_P / 100, 0.01, '=');
          ValueEntryDisc_P."Discount %" := ModificationValue_P;
          SetFinalPrice(ValueEntryDisc_P,PrevFinalPrice_P);
        END;
        ValuationType_P::"Change ECD amount" : BEGIN
          ValueEntryDisc_P."Discount Amount" := ValueEntryDisc_P."Discount Amount" + ModificationValue_P;
          SetFinalPrice(ValueEntryDisc_P,PrevFinalPrice_P);
        END;
        ValuationType_P::"Apply Purch. Discount": BEGIN
          GetPurchDiscount(CMCPurchDiscValue_L);
          CASE CMCPurchDiscValue_L."Valuation Type" OF
            CMCPurchDiscValue_L."Valuation Type" ::Percentage : BEGIN
              ValueEntryDisc_P."Discount %" := CMCPurchDiscValue_L."Discount Value";
              ValueEntryDisc_P."Discount Amount" :=
                                 ROUND(ValueEntryDisc_P."Base Value" * CMCPurchDiscValue_L."Discount Value"/100, 0.01, '=');
            END;
            CMCPurchDiscValue_L."Valuation Type" ::"Unit Amount" : BEGIN
              ValueEntryDisc_P."Discount Amount" := CMCPurchDiscValue_L."Discount Value";
            END;
          END;
          SetFinalPrice(ValueEntryDisc_P,PrevFinalPrice_P);
        END;
      END;

      ValueEntryDisc_P.MODIFY(TRUE);

      EXIT(ValueEntryDisc_P."Final Price");
    END;
