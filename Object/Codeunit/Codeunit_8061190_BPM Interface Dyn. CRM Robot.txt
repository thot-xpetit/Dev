    PROCEDURE DynamicsCRMInterface@8061007(MessageGUID_P@8060982 : GUID) Result_L : Boolean;
    VAR
      ConfirmFileName@8060984 : Text[1024];
      ConfirmFile@8060983 : File;
    BEGIN
      // ============================================================================
      // !!! --------------------- INTERFACE DYNAMICS CRM ----------------------- !!!
      // ============================================================================
      Result_L := TRUE;

      IncomingMessage_G.GET(MessageGUID_P);                  // Message
      InterfaceID_G.GET(IncomingMessage_G."Interface ID");   // Interface
      DynamicsCRMSetup_G.GET(InterfaceID_G."Interface ID");     // ParamStres

      WorkFileParam1_G := '';
      WorkFileParam2_G := '';
      WorkFileParam3_G := '';
      WorkFileParam4_G := '';

      IF IncomingMessage_G."Parameter 1" <> '' THEN
        IF EXISTS(IncomingMessage_G."Parameter 1") THEN
          WorkFileParam1_G := DynamicsCRMSetup_G."Work Dir" + '\' + cduTools_G.GetFileName(IncomingMessage_G."Parameter 1");

      IF IncomingMessage_G."Parameter 2" <> '' THEN
        IF EXISTS(IncomingMessage_G."Parameter 2") THEN
          WorkFileParam2_G := DynamicsCRMSetup_G."Work Dir" + '\' + cduTools_G.GetFileName(IncomingMessage_G."Parameter 2");

      IF IncomingMessage_G."Parameter 3" <> '' THEN
        IF EXISTS(IncomingMessage_G."Parameter 3") THEN
          WorkFileParam3_G := DynamicsCRMSetup_G."Work Dir" + '\' + cduTools_G.GetFileName(IncomingMessage_G."Parameter 3");

      IF IncomingMessage_G."Parameter 4" <> '' THEN
        IF EXISTS(IncomingMessage_G."Parameter 4") THEN
          WorkFileParam4_G := DynamicsCRMSetup_G."Work Dir" + '\' + cduTools_G.GetFileName(IncomingMessage_G."Parameter 4");

      IF IncomingMessage_G.Type = IncomingMessage_G.Type::"Event" THEN
        LogStatistic_G.AddLogStatistic(IncomingMessage_G."Interface ID", LogStatistic_G.Type::"Event",
                                       IncomingMessage_G."Event Code")
      ELSE
        LogStatistic_G.AddLogStatistic(IncomingMessage_G."Interface ID", LogStatistic_G.Type::Processus,
                                       IncomingMessage_G."Processus Code");

      //------------------------------------------------------------------------
      // V'rification des paramStres d'interface
      //------------------------------------------------------------------------
      IF NOT CheckDynamicsCRMParameters THEN BEGIN
        IF GUIALLOWED THEN
          MESSAGE(Text8061009);
        EXIT(FALSE);
      END;
      //------------------------------------------------------------------------


      // ============================================================================
      // TRAITEMENT EVENEMENT / PROCESSUS Dynamics CRM
      // ============================================================================
      CASE IncomingMessage_G.Type OF

        // ----------------------------------------------------------------
        IncomingMessage_G.Type::"Event": BEGIN     // EVENEMENT
        // ----------------------------------------------------------------
          Result_L := DynamicsCRMEventProcess;
        END;

        // ----------------------------------------------------------------
        IncomingMessage_G.Type::"Scheduled Process": BEGIN // PLANIFIE
        // ----------------------------------------------------------------
          Result_L := ScheduledProcess;
        END;

      END;

      // ====================================================================================
      // FIN EXECUTION MESSAGE
      // ====================================================================================

      // ------------------------------------------------------------------------
      // Cas o- demande de mise en pause en cours d'attente et fin de traitement.
      // --> D'blocage de l'utilisateur.
      // ------------------------------------------------------------------------
      IF InterfaceID_G."Interface Stop Dir" <> '' THEN BEGIN
        ConfirmFileName := InterfaceID_G."Interface Stop Dir" + '\BPM-' + InterfaceID_G."Interface ID" + '-RequestStopOK.BPM';
        IF NOT EXISTS(ConfirmFileName) THEN BEGIN
          IF ConfirmFile.CREATE(ConfirmFileName) THEN
            ConfirmFile.CLOSE;
        END;
      END;
      // ------------------------------------------------------------------------

      EXIT(Result_L);
    END;

    LOCAL PROCEDURE DynamicsCRMEventProcess@8061006() Result_L : Boolean;
    VAR
      Processus_L@8060986 : Record 8061182;
      EventCode_L@8060985 : Code[50];
      ExecutionStartDateTime_L@8060984 : DateTime;
      Duration_L@8060983 : Decimal;
      LogSeq_L@8060982 : Integer;
    BEGIN
      // ============================================================================
      // TRAITEMENT EVENEMENT DYNAMICS CRM
      // ============================================================================
      Result_L := TRUE;

      // -------------------------------------------------------------------------------
      // Traitement d'un fichier d''v'nement ...
      // -------------------------------------------------------------------------------
      EventCode_L := IncomingMessage_G."Event Code";

      IF Event_G.GET(EventCode_L) THEN BEGIN

        IF Event_G.Enabled THEN BEGIN

          // --------------------------------------------------------------------------
          // JOURNAL : Prise en compte d'un 'v'nement
          // --------------------------------------------------------------------------
          LogMessage_G.Add(IncomingMessage_G."Message ID", '', '',
                           STRSUBSTNO(FORMAT(Text8060982), Event_G.Code),
                           LogMessage_G."Log Level"::Information, 1);

          // ==========================================================================
          // Ex'cution du processus rattach' . l''v'nement reŒu ...
          // ==========================================================================
          Processus_L.RESET;
          Processus_L.SETCURRENTKEY("Event Code", "Interface ID");
          Processus_L.SETRANGE("Event Code", Event_G.Code);
          Processus_L.SETRANGE("Interface ID", IncomingMessage_G."Interface ID");
          IF Processus_L.FINDFIRST THEN BEGIN

            IF Processus_L.Enabled THEN BEGIN

              // ----------------------------------------------------------------------
              // JOURNAL : Ex'cution du Processus 'v'nementiel
              // ----------------------------------------------------------------------
              LogSeq_L := LogMessage_G.Add(IncomingMessage_G."Message ID", Processus_L.Code, '',
                                           STRSUBSTNO(FORMAT(Text8060983), Processus_L.Code),
                                           LogMessage_G."Log Level"::Information, 1);

              ExecutionStartDateTime_L := CURRENTDATETIME;

              Result_L := ExecuteProcessus(IncomingMessage_G."Interface ID", Processus_L.Code, TRUE);

              Duration_L := ROUND((CURRENTDATETIME - ExecutionStartDateTime_L) / 1000, 0.01);

              LogMessage_G.UpdateDuration(IncomingMessage_G."Message ID", LogSeq_L, Duration_L);

            END ELSE BEGIN

              // ----------------------------------------------------------------------
              // JOURNAL : Processus 'v'nementiel d'sactiv'
              // ----------------------------------------------------------------------
              LogMessage_G.Add(IncomingMessage_G."Message ID", Processus_L.Code, '',
                               STRSUBSTNO(FORMAT(Text8060984), Processus_L.Code),
                               LogMessage_G."Log Level"::Warning, 1);

            END;

            // ----------------------------------------------------------------------
            // ARCHIVAGE / SUPPRESSION du fichier EvSnement
            // ----------------------------------------------------------------------
            IF Result_L THEN BEGIN

              IncomingMessage_G.GET(IncomingMessage_G."Message ID");
              IF IncomingMessage_G.Status = IncomingMessage_G.Status::Pending THEN BEGIN
                IncomingMessage_G.ChangeStatus(IncomingMessage_G."Message ID", IncomingMessage_G.Status::Completed);

                IF InterfaceID_G."Auto Archive" THEN BEGIN

                  IF EXISTS(WorkFileParam1_G) THEN
                    cduTools_G.MoveFileToFolder2(WorkFileParam1_G, DynamicsCRMSetup_G."Archives Dir",TRUE,IncomingMessage_G."Message ID"
      ,1);
                  IF EXISTS(WorkFileParam2_G) THEN
                    cduTools_G.MoveFileToFolder2(WorkFileParam2_G, DynamicsCRMSetup_G."Archives Dir",TRUE,IncomingMessage_G."Message ID"
      ,1);
                  IF EXISTS(WorkFileParam3_G) THEN
                    cduTools_G.MoveFileToFolder2(WorkFileParam3_G, DynamicsCRMSetup_G."Archives Dir",TRUE,IncomingMessage_G."Message ID"
      ,1);
                  IF EXISTS(WorkFileParam4_G) THEN
                    cduTools_G.MoveFileToFolder2(WorkFileParam4_G, DynamicsCRMSetup_G."Archives Dir",TRUE,IncomingMessage_G."Message ID"
      ,1);

                END ELSE BEGIN

                  IF EXISTS(WorkFileParam1_G) THEN cduTools_G.DeleteFile(WorkFileParam1_G);
                  IF EXISTS(WorkFileParam2_G) THEN cduTools_G.DeleteFile(WorkFileParam2_G);
                  IF EXISTS(WorkFileParam3_G) THEN cduTools_G.DeleteFile(WorkFileParam3_G);
                  IF EXISTS(WorkFileParam4_G) THEN cduTools_G.DeleteFile(WorkFileParam4_G);

                END;
              END;

            END;
            // ----------------------------------------------------------------------

          END ELSE BEGIN
            ActionOnNoProcessEvent; //--------------- JOURNAL - Pas de processus li'.
          END;

        END ELSE BEGIN
          ActionOnDisabledEvent; //------------------ JOURNAL - Ev'nement d'sactiv'.
        END;

      END ELSE BEGIN
        ActionOnUnknownEvent;   //------------------- JOURNAL - Ev'nement inconnu.
      END;

      EXIT(Result_L);
    END;

    LOCAL PROCEDURE ScheduledProcess@8061005() Result_L : Boolean;
    VAR
      Processus_L@8060985 : Record 8061182;
      ExecutionStartDateTime_L@8060984 : DateTime;
      Duration_L@8060983 : Decimal;
      LogSeq_L@8060982 : Integer;
    BEGIN
      // ============================================================================
      // TRAITEMENT PROCESSUS PLANIFIE DYNAMICS CRM
      // ============================================================================
      Result_L := TRUE;

      IF Processus_L.GET(IncomingMessage_G."Processus Code", IncomingMessage_G."Interface ID") THEN BEGIN

        // ----------------------------------------------------------------------
        // JOURNAL : Prise en compte du Processus Planifi'
        // ----------------------------------------------------------------------
        LogMessage_G.Add(IncomingMessage_G."Message ID", Processus_L.Code, '',
                         STRSUBSTNO(FORMAT(Text8061008), Processus_L.Code),
                         LogMessage_G."Log Level"::Information, 1);

        // ----------------------------------------------------------------------
        // JOURNAL : Ex'cution du Processus planifi'
        // ----------------------------------------------------------------------
        LogSeq_L := LogMessage_G.Add(IncomingMessage_G."Message ID", Processus_L.Code, '',
                         STRSUBSTNO(FORMAT(Text8060983), Processus_L.Code),
                         LogMessage_G."Log Level"::Information, 1);

        ExecutionStartDateTime_L := CURRENTDATETIME;

        Result_L := ExecuteProcessus(IncomingMessage_G."Interface ID", Processus_L.Code, FALSE);

        Duration_L := ROUND((CURRENTDATETIME - ExecutionStartDateTime_L) / 1000, 0.01);

        LogMessage_G.UpdateDuration(IncomingMessage_G."Message ID", LogSeq_L, Duration_L);

      END;

      IF Result_L THEN
        IncomingMessage_G.ChangeStatus(IncomingMessage_G."Message ID", IncomingMessage_G.Status::Completed);

      EXIT(Result_L);
    END;

    LOCAL PROCEDURE ExecuteProcessus@8061004(InterfaceID_P@8060984 : Code[20];ProcessusCode_P@8060983 : Code[30];EventProcess@8060982 : Boolean) Result_L : Boolean;
    VAR
      SequenceTask_L@8060992 : Record 8061183;
      ContinueToNextSequence_L@8060991 : Boolean;
      ExecutionStartDateTime_L@8060990 : DateTime;
      Duration_L@8060989 : Decimal;
      LogSeq_L@8060988 : Integer;
      TotalTask_L@8060987 : Integer;
      CurrentTask_L@8060986 : Integer;
      Progression_L@8060985 : Integer;
    BEGIN
      // ============================================================================
      // BOUCLE DE LECTURE DES SEQUENCES ACTIVEES D'UN PROCESSUS
      // ============================================================================
      Result_L := TRUE;

      LogStatistic_G.AddLogStatistic(IncomingMessage_G."Interface ID", LogStatistic_G.Type::Processus, ProcessusCode_P);

      IncomingMessage_G.SetProcessusCode(IncomingMessage_G."Message ID", ProcessusCode_P);
      IncomingMessage_G.GET(IncomingMessage_G."Message ID");

      SequenceTask_L.RESET;
      SequenceTask_L.SETCURRENTKEY(Code, "Interface ID", Sequence);
      SequenceTask_L.SETRANGE(Code, ProcessusCode_P);
      SequenceTask_L.SETRANGE("Interface ID", InterfaceID_P);
      SequenceTask_L.SETRANGE(Enabled, TRUE);

      IF SequenceTask_L.FINDSET(FALSE, FALSE) THEN BEGIN

        TotalTask_L := SequenceTask_L.COUNT;
        CurrentTask_L := 0;

        REPEAT

          CurrentRetry_G := 0;

          // --------------------------------------------------------------------------
          // JOURNAL : Ex'cution de la tüche ...
          // --------------------------------------------------------------------------
          LogSeq_L := LogMessage_G.Add(IncomingMessage_G."Message ID", SequenceTask_L.Code, SequenceTask_L.Task,
                                       STRSUBSTNO(FORMAT(Text8060986), SequenceTask_L.Sequence, SequenceTask_L.Task),
                                       LogMessage_G."Log Level"::Information, 2);

          CurrentTask_L := CurrentTask_L + 1;

          Progression_L := ROUND((CurrentTask_L / TotalTask_L) * 10000, 1);

          IncomingMessage_G.SetTaskCode(IncomingMessage_G."Message ID", SequenceTask_L.Task, Progression_L);

          IncomingMessage_G.GET(IncomingMessage_G."Message ID");

          ExecutionStartDateTime_L := CURRENTDATETIME;

          ContinueToNextSequence_L := ExecuteTask(ProcessusCode_P, InterfaceID_P, SequenceTask_L.Sequence, SequenceTask_L.Task, TRUE);

          Duration_L := ROUND((CURRENTDATETIME - ExecutionStartDateTime_L) / 1000, 0.01);

          LogMessage_G.UpdateDuration(IncomingMessage_G."Message ID", LogSeq_L, Duration_L);

          SLEEP(100); // Temporisation

        UNTIL ((SequenceTask_L.NEXT = 0) OR (NOT ContinueToNextSequence_L));

        Result_L := ContinueToNextSequence_L;

      END ELSE
        // --------------------------------------------------------------------------
        // JOURNAL : Aucune tüche d'finie dans le processus ...
        // --------------------------------------------------------------------------
        LogMessage_G.Add(IncomingMessage_G."Message ID", ProcessusCode_P, '',
                         STRSUBSTNO(FORMAT(Text8060985), ProcessusCode_P, IncomingMessage_G."Event Code"),
                         LogMessage_G."Log Level"::Warning, 1);

      EXIT(Result_L);
    END;

          // ---------------------------------------------------------

      //----------------------------------------------------------------------------------

      CLEAR(cduDynamicsCRMLib2);

    LOCAL PROCEDURE ActionOnErrorExecution@8061002(ProcessusCode_P@8060986 : Code[30];InterfaceID_P@8060985 : Code[20];ProcessusSeq_P@8060984 : Integer;WriteJournal_P@8060983 : Boolean;ForceRetry_P@8060982 : Boolean) Result_L : Boolean;
    VAR
      ProcessusTask_L@8060989 : Record 8061183;
      EndProcessus_L@8060988 : Boolean;
      LogMessage_L@8060987 : Text[1024];
    BEGIN
      Result_L := TRUE;

      ProcessusTask_L.GET(ProcessusCode_P,InterfaceID_P,ProcessusSeq_P);

      IF ForceRetry_P AND (CurrentRetry_G < 10) THEN BEGIN

        CurrentRetry_G := CurrentRetry_G + 1;

        // --------------------------------------------------------------------------
        // JOURNAL : Deadlock ou TimeStamp error - Option RETRY
        // --------------------------------------------------------------------------
        IF WriteJournal_P THEN

          LogMessage_G.Add(IncomingMessage_G."Message ID", ProcessusTask_L.Code, ProcessusTask_L.Task,
                           STRSUBSTNO(FORMAT(Text8061010), ProcessusTask_L.Task, CurrentRetry_G),
                           LogMessage_G."Log Level"::Warning, 2);

          EndProcessus_L := ExecuteTask(ProcessusCode_P, InterfaceID_P, ProcessusSeq_P, ProcessusTask_L.Task, TRUE);

        IF NOT EndProcessus_L THEN BEGIN
          // --------------------------------------------------------------------------
          // JOURNAL : Erreur d'ex'cution - FIN DU PROCESSUS
          // --------------------------------------------------------------------------
          IF WriteJournal_P AND (CurrentRetry_G = 10) THEN

              CurrentRetry_G := 0;

        END;

      END ELSE BEGIN

        CASE ProcessusTask_L."Action on Error" OF

          ProcessusTask_L."Action on Error"::Continue: BEGIN

            // --------------------------------------------------------------------------
            // JOURNAL : Erreur d'ex'cution - Option CONTINUER
            // --------------------------------------------------------------------------
            IF WriteJournal_P THEN

              LogMessage_G.Add(IncomingMessage_G."Message ID", ProcessusTask_L.Code, ProcessusTask_L.Task,
                               STRSUBSTNO(FORMAT(Text8060994), ProcessusTask_L.Task),
                               LogMessage_G."Log Level"::Warning, 2);

            EndProcessus_L := TRUE;

          END;

          ProcessusTask_L."Action on Error"::Stop: BEGIN

            // --------------------------------------------------------------------------
            // JOURNAL : Erreur d'ex'cution - Option STOP
            // --------------------------------------------------------------------------
            IF WriteJournal_P THEN

              LogMessage_G.Add(IncomingMessage_G."Message ID", ProcessusTask_L.Code, ProcessusTask_L.Task,
                               STRSUBSTNO(FORMAT(Text8060995), ProcessusTask_L.Task),
                               LogMessage_G."Log Level"::Warning, 2);

            EndProcessus_L := FALSE;

          END;

          ProcessusTask_L."Action on Error"::Retry: BEGIN

            // --------------------------------------------------------------------------
            // JOURNAL : Erreur d'ex'cution - Option RETRY
            // --------------------------------------------------------------------------
            IF WriteJournal_P THEN

              LogMessage_G.Add(IncomingMessage_G."Message ID", ProcessusTask_L.Code, ProcessusTask_L.Task,
                               STRSUBSTNO(FORMAT(Text8060996), ProcessusTask_L.Task),
                               LogMessage_G."Log Level"::Warning, 2);

            EndProcessus_L := ExecuteTask(ProcessusCode_P, InterfaceID_P, ProcessusSeq_P, ProcessusTask_L.Task, FALSE);

            IF NOT EndProcessus_L THEN
              // --------------------------------------------------------------------------
              // JOURNAL : Erreur d'ex'cution - FIN DU PROCESSUS
              // --------------------------------------------------------------------------
              IF WriteJournal_P THEN

                LogMessage_G.Add(IncomingMessage_G."Message ID", ProcessusTask_L.Code, ProcessusTask_L.Task,
                                 STRSUBSTNO(FORMAT(Text8060997), ProcessusTask_L.Task),
                                 LogMessage_G."Log Level"::Warning, 2);
          END;

          ProcessusTask_L."Action on Error"::Task: BEGIN

              // --------------------------------------------------------------------------
              // JOURNAL : Erreur d'ex'cution - Option TACHE
              // --------------------------------------------------------------------------
              IF WriteJournal_P THEN BEGIN

                LogMessage_L := STRSUBSTNO(FORMAT(Text8060998), ProcessusTask_L.Task, ProcessusTask_L."Task on Error");

                LogMessage_G.Add(IncomingMessage_G."Message ID", ProcessusTask_L.Code, ProcessusTask_L.Task,
                                 LogMessage_L, LogMessage_G."Log Level"::Warning, 2);
              END;

              EndProcessus_L := ExecuteTask(ProcessusCode_P, InterfaceID_P, ProcessusSeq_P, ProcessusTask_L."Task on Error", FALSE);

              IF NOT EndProcessus_L THEN
                // --------------------------------------------------------------------------
                // JOURNAL : Erreur d'ex'cution - FIN DU PROCESSUS
                // --------------------------------------------------------------------------
                IF WriteJournal_P THEN

                  LogMessage_G.Add(IncomingMessage_G."Message ID", ProcessusTask_L.Code, ProcessusTask_L.Task,
                                   STRSUBSTNO(FORMAT(Text8060999), ProcessusTask_L."Task on Error"),
                                   LogMessage_G."Log Level"::Warning, 2);

          END;

        END; // CASE

      END; // ELSE IF ForceRetry

      Result_L := EndProcessus_L;

      EXIT(Result_L);
    END;

    PROCEDURE SetBPMStop@8060995(InterfaceID_P@8060985 : Code[20];RequestInterfaceLock_P@8060984 : Boolean;VAR RequestLockID_P@8060983 : Integer;ShowWaitMessage_P@8060982 : Boolean);
    VAR
      CompanyInformation@8060996 : Record 79;
      IncomingMessage_L@8060995 : Record 8061188;
      FileRec@8060994 : Record 2000000022;
      Task@8060993 : Record 8061184;
      dialogWaitMessage_L@8060992 : Dialog;
      RequestFile@8060991 : File;
      ConfirmFile@8060990 : File;
      RequestFileName@8060989 : Text[250];
      RequestFileNameRoot@8060988 : Text[250];
      ConfirmFileName@8060987 : Text[250];
      i@8060986 : Integer;
    BEGIN
      IF NOT InterfaceID_G.GET(InterfaceID_P) THEN EXIT;

      IF InterfaceID_G."Interface Stop Dir" = '' THEN EXIT;

      ConfirmFileName := InterfaceID_G."Interface Stop Dir" + '\BPM-' + InterfaceID_G."Interface ID" + '-RequestStopOK.BPM';
      RequestFileNameRoot := InterfaceID_G."Interface Stop Dir" + '\BPM-' + InterfaceID_G."Interface ID";

      IF RequestInterfaceLock_P THEN BEGIN
        //-----------------------------------------------------------------------------
        // Creation du FICHIER de mise en pause
        //-----------------------------------------------------------------------------
        RequestLockID_P := 0;
        RequestFileName := RequestFileNameRoot + '-RequestStop.' + USERID + '-' + FORMAT(RequestLockID_P);
        WHILE EXISTS(RequestFileName) DO BEGIN
          RequestLockID_P := RequestLockID_P + 1;
          RequestFileName := RequestFileNameRoot + '-RequestStop.' + USERID + '-' + FORMAT(RequestLockID_P);
        END;

        RequestFile.CREATE(RequestFileName);
        RequestFile.CLOSE;

        //-----------------------------------------------------------------------------
        // Attente du FICHIER de Confirmation de pause
        // Uniquement si Ordre Navision en cours.
        //-----------------------------------------------------------------------------
        IF InterfaceID_G."Processor Service Started" THEN BEGIN

          IncomingMessage_L.SETCURRENTKEY("Interface ID", Status);
          IncomingMessage_L.SETRANGE("Interface ID", InterfaceID_G."Interface ID");
          IncomingMessage_L.SETRANGE(Status, IncomingMessage_L.Status::Pending);
          IF NOT IncomingMessage_L.ISEMPTY THEN BEGIN
            IF ShowWaitMessage_P THEN BEGIN
              dialogWaitMessage_L.OPEN(Text8061013);
              dialogWaitMessage_L.UPDATE(1, InterfaceID_G."Interface ID");
            END;

            WHILE NOT EXISTS(ConfirmFileName) DO
              SLEEP(1000);

            IF ShowWaitMessage_P THEN
              dialogWaitMessage_L.CLOSE;
          END;

        END;

      END ELSE BEGIN
        //-----------------------------------------------------------------------------
        // Suppression du FICHIER de mise en pause
        //-----------------------------------------------------------------------------
        RequestFileName := RequestFileNameRoot + '-RequestStop.' + USERID + '-' + FORMAT(RequestLockID_P);
        IF EXISTS(RequestFileName) THEN
          cduTools_G.DeleteFile(RequestFileName);

        FileRec.RESET;
        FileRec.SETCURRENTKEY(Path);
        FileRec.SETFILTER(Path, 'C:\');           // R'initialisation
        IF FileRec.FINDFIRST THEN;                // R'initialisation
        FileRec.SETFILTER(Path, InterfaceID_G."Interface Stop Dir" + '\');
        FileRec.SETRANGE("Is a file", TRUE);
        FileRec.SETFILTER(Name, '%1', 'BPM-' + InterfaceID_G."Interface ID" + '-RequestStop.*');
        IF FileRec.COUNT = 0 THEN
          IF EXISTS(ConfirmFileName) THEN
            cduTools_G.DeleteFile(ConfirmFileName);

      END;
    END;

    BEGIN
    {
      // --> 3LI BEGIN
      20120101-8060990/091-CMB6.00.10.05-PRD01 : ADDOBJ (Nouvel objet)
      // <-- 3LI END
    }
