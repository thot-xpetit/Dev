  CODE
  {
    VAR
      "<-- 3LI GLOBALS -->"@1000000001 : Integer;
      CodeUnit_G@1000000000 : Integer;
      "<-- 3LI TEXT GLOBALS -->"@1000000003 : TextConst;
      Text0001@1000000002 : TextConst 'ENU=Error Table : %1 doesn''t exist for company : "%2";FRA=Erreur Table : "%1"  Inexistante.';
      Text0002@1000000005 : TextConst 'ENU=Error open file %1 : %2;FRA=Erreur ouverture fichier %1 : %2';
      Text0009@1000000004 : TextConst 'ENU=Error file %1 was not exist.;FRA=Erreur fichier %1 inexistant.';

      TextEcrOuv@1000000007 : TextConst 'FRA=Info : Ecriture "%1" non ouverte pour Id. Externe "%2" : Ligne non mise . jour.';
      TextCopySav@1000000008 : TextConst 'ENU=Error backup copy.;FRA=Erreur copie de sauvegarde.';
      TextVoirLog@1000000009 : TextConst 'FRA=Veuillez consulter le log.';
      TextNonAtt@1000000010 : TextConst 'FRA=Info : Ecriture "%1" non en attente pour Id. Externe "%2" : Ligne non mise . jour.';
      TextErrMaj@1000000011 : TextConst 'FRA=Erreur : Ecriture "%1" Appel pour Id. Externe "%2" : Ligne non mise . jour.';
      TextInfoMaj@1000000012 : TextConst 'FRA=Info : Nb ligne BAP "%1" : Nombre ligne mise . jour "%2" : Nombre erreur "%3".';
      TextIdExtInex@1000000013 : TextConst 'FRA=Erreur : Id. Externe "%1" non r'f'renc' : Ligne non mise . jour.';

    PROCEDURE "<-- 3LI FUNCTIONS -->"@1000000001();

    PROCEDURE ImportBap@1000000000(FileName_P@1000000002 : Text[128]) Result_L : Text[256];
    VAR
      "<-- 3LI LOCALS -->"@1000000005 : Integer;
      TableParameter_L@1000000006 : Record 8053402;
      TableVendorLedgEnt_L@1000000017 : Record 25;
      ToolsFile_L@1000000000 : Codeunit 8053405;
      MoteurInterface_L@1000000012 : Codeunit 8061230;
      FormatFieldsMan_L@1000000015 : Codeunit 8060985;
      ConnectorName_L@1000000010 : Code[20];
      ImpExpFolder_L@1000000009 : Text[256];
      LogFolder_L@1000000008 : Text[256];
      ExtensionFile_L@1000000007 : Text[10];
      PrefixFile_L@1000000004 : Text[10];
      FileImport_L@1000000003 : File;
      FileInStream_L@1000000001 : InStream;
      NumExt_L@1000000011 : Text[64];
      BAP_L@1000000013 : Text[1];
      intPos_L@1000000018 : Integer;
      intNbLu_L@1000000016 : Integer;
      TxtLu_L@1000000014 : Text[1024];
      FileLogName_L@1000000019 : Text[256];
      NbLigneBAP_L@1000000020 : Integer;
      NbLigneMaj_L@1000000021 : Integer;
      NbLigneErr_L@1000000022 : Integer;
    BEGIN
      //-->20170627-Int03-CMC02.02.00-ADVITAM-CDL
      //----------------------------------------------------------------------------------------------------------------------------------
      // Importation des bons . payer de BaseWare
      // - Lire "Ecriture comptable fournisseur" (T25) de l'identifiant BaseWare ("Id externe").
      //   + Si BAP = 'O' ==> M.j "Attente" = " " si 'criture "Ouverte".
      //----------------------------------------------------------------------------------------------------------------------------------
      // Entr'es:
      //   - FileName_P : Nom du fichier PC . int'grer
      // Sorties:
      //   - Result_L : Erreur Retourn'e, Vide si Ok
      //----------------------------------------------------------------------------------------------------------------------------------
      MoteurInterface_L.GetParameter(CodeUnit_G, ConnectorName_L, ImpExpFolder_L, LogFolder_L, ExtensionFile_L, PrefixFile_L);

      ToolsFile_L.LogInit(LogFolder_L, 'COMPTA_BAP');
      ToolsFile_L.LogInitCodeUnit(CodeUnit_G);

      ToolsFile_L.LogInitFileExecute(FileName_P);
      IF (NOT TableParameter_L.FINDFIRST) THEN BEGIN
        ToolsFile_L.LogWriteInfo(COPYSTR(STRSUBSTNO(Text0001, TableParameter_L.TABLECAPTION), 1
          , MAXSTRLEN(Result_L)));
      END ELSE BEGIN
        IF (TableParameter_L."Import Files BAP DIMO") THEN BEGIN
          IF NOT EXISTS(FileName_P) THEN BEGIN
            Result_L := COPYSTR(STRSUBSTNO(Text0009, FileName_P), 1, MAXSTRLEN(Result_L));
            ToolsFile_L.LogWriteErreur(Result_L);
          END ELSE BEGIN
            IF NOT FileImport_L.OPEN(FileName_P) THEN BEGIN
              Result_L := COPYSTR(STRSUBSTNO(Text0002, FileName_P, GETLASTERRORTEXT), 1, MAXSTRLEN(Result_L));
              CLEARLASTERROR;
              ToolsFile_L.LogWriteErreur(Result_L);
            END ELSE BEGIN
              FileImport_L.CREATEINSTREAM(FileInStream_L);
              WHILE NOT (FileInStream_L.EOS) DO BEGIN
                intNbLu_L := FileInStream_L.READTEXT(TxtLu_L);
                TxtLu_L := FormatFieldsMan_L.Ansi2Ascii(TxtLu_L);
                NbLigneBAP_L += 1;

                intPos_L := 1;
                // N› Base Ware
                NumExt_L := ToolsFile_L.ExtractTextTrim(TxtLu_L, intPos_L, 64);
                IF (NumExt_L = '') THEN BEGIN
                    Result_L := COPYSTR(STRSUBSTNO(TextIdExtObli), 1, MAXSTRLEN(Result_L));
                    ToolsFile_L.LogWriteInfo(Result_L);
                END ELSE BEGIN
                  // N› Base Ware
                  BAP_L := ToolsFile_L.ExtractText(TxtLu_L, intPos_L, 1);
                  IF BAP_L = 'O' THEN BEGIN
                    // Recherche des lignes . modifier . partir du N› externe
                    TableVendorLedgEnt_L.SETRANGE(TableVendorLedgEnt_L."External ID", NumExt_L);
                    IF NOT TableVendorLedgEnt_L.FINDSET THEN BEGIN
                      Result_L := COPYSTR(STRSUBSTNO(TextIdExtInex, NumExt_L), 1, MAXSTRLEN(Result_L));
                      ToolsFile_L.LogWriteErreur(Result_L);
                    END ELSE BEGIN
                      REPEAT
                        IF NOT TableVendorLedgEnt_L.Open THEN BEGIN
                          Result_L := COPYSTR(STRSUBSTNO(TextEcrOuv, TableVendorLedgEnt_L."Entry No."
                            ,TableVendorLedgEnt_L."External ID"), 1, MAXSTRLEN(Result_L));
                          ToolsFile_L.LogWriteInfo(Result_L);
                          NbLigneErr_L += 1;
                        END ELSE BEGIN
                          IF TableVendorLedgEnt_L."On Hold" = '' THEN BEGIN
                            Result_L := COPYSTR(STRSUBSTNO(TextNonAtt, TableVendorLedgEnt_L."Entry No."
                              ,TableVendorLedgEnt_L."External ID"), 1, MAXSTRLEN(Result_L));
                            ToolsFile_L.LogWriteInfo(Result_L);
                            NbLigneErr_L += 1;
                          END ELSE BEGIN
                            TableVendorLedgEnt_L."On Hold" := '';
                            // Mise . jour directe non autoris'e.
                            IF NOT CODEUNIT.RUN(CODEUNIT::"Vend. Entry-Edit", TableVendorLedgEnt_L) THEN BEGIN
      //                      IF NOT TableVendorLedgEnt_L.MODIFY(TRUE) THEN BEGIN
                              Result_L := COPYSTR(STRSUBSTNO(TextErrMaj, TableVendorLedgEnt_L."Entry No."
                                ,TableVendorLedgEnt_L."External ID"), 1, MAXSTRLEN(Result_L));
                              //-->20180319-Int03-CMC02.02.00-ADVITAM-CDL
                              // Before
                              {
                              ToolsFile_L.LogWriteInfo(Result_L);
                              }
                              // After
                              ToolsFile_L.LogWriteErreur(Result_L);
                              //--<20180319-Int03-CMC02.02.00-ADVITAM-CDL
                              NbLigneErr_L += 1;
                            END ELSE BEGIN
                              NbLigneMaj_L += 1;
                            END;
                          END;
                        END;
                      UNTIL TableVendorLedgEnt_L.NEXT = 0;
                    END;
                  END;
                END;
              END;
              FileImport_L.CLOSE();
              IF ToolsFile_L.LogErreurType() <> 'E' THEN BEGIN
                IF NOT ToolsFile_L.FileRenameSav(FileName_P, 'Sav', TRUE) THEN BEGIN
                  ToolsFile_L.LogWriteErreur(STRSUBSTNO(TextCopySav, FileName_P));
                END;
              END;
            END;
          END;
        END;
      END;
      //Result_L := COPYSTR(STRSUBSTNO(TextInfoMaj, FORMAT(NbLigneBAP_L), FORMAT(NbLigneMaj_L)
      //  ,FORMAT(NbLigneErr_L)), 1, MAXSTRLEN(Result_L));
      //ToolsFile_L.LogWriteInfo(Result_L);

      CLEAR(Result_L);
      FileLogName_L := ToolsFile_L.LogName;
      ToolsFile_L.LogClose();

      IF ToolsFile_L.LogErreurType() = 'E' THEN BEGIN
        Result_L := 'ERREUR ' + FileLogName_L;
        ERROR(Result_L);
      END ELSE BEGIN
        IF ToolsFile_L.LogErreurType() = 'I' THEN BEGIN
          Result_L := TextVoirLog;
          MoteurInterface_L.InsertLog(CodeUnit_G, FileLogName_L);
        END ELSE BEGIN
          MoteurInterface_L.InsertOk(CodeUnit_G);
        END;
      END;
      //<--20170627-Int03-CMC02.02.00-ADVITAM-CDL
    END;

    BEGIN
    {
      Documentation()
      //--> 3LI BEGIN
      20170627-Int03-CMC02.02.00-ADVITAM-CDL : - Importation des bons . payer BaseWare :
                                                 + Mettre . jour 'criture fournisseur : Zone en Attente pour "Id Externet" BaseWare.
      20180319-Int03-CMC02.02.00-ADVITAM-CDL : - Message bloquant pour retour en erreur appel fonction Nav de mise . jour
                                                 + "TextErrMaj" chang'
                                                   Avant : Info : Ecriture "%1" mise . jour pour Id. Externe "%2" : Ligne non mise . jour.
                                                   AprSs : Erreur : Ecriture "%1" Appel pour Id. Externe "%2" : Ligne non mise . jour.
      //<-- 3LI END
    }
    END.
  }
