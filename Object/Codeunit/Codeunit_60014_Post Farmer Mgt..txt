          recMATIFContractSales_L.CALCFIELDS("Nb Lot Invoiced");

            FirstSignOffDate_L := recMATIFContractSales_L."Sign-Off Date";

              REPEAT
                // --> 20161118-WI00-ADV01.01.00-2814-OLE01
                IF recMATIFContractPurc_L."Source Option Contract" <> '' THEN
                  IncludeCommission_L := FALSE;
                // <-- 20161118-WI00-ADV01.01.00-2814-OLE01

                Lotsales_L := 0;
                LotPurch_L := 0;
                recMATIFContractPurc_L.CALCFIELDS("Nb Lot Invoiced");
                recMATIFContractSales_L.CALCFIELDS("Nb Lot Invoiced");
                IF recMATIFContractPurc_L."Nb Lot Invoiced" < recMATIFContractPurc_L."Lot Number" THEN BEGIN
                  FirstSignOffDate_L := recMATIFContractSales_L."Sign-Off Date";
                  // --> 20151104-CER126-TER01.01.00-2427-FBE01
                  Way_L := Way_L::Sales;
                  // <-- 20151104-CER126-TER01.01.00-2427-FBE01
                  IF (FirstSignOffDate_L = 0D) OR (recMATIFContractPurc_L."Sign-Off Date" < FirstSignOffDate_L) THEN BEGIN
                    FirstSignOffDate_L := recMATIFContractPurc_L."Sign-Off Date";
                    // --> 20151104-CER126-TER01.01.00-2427-FBE01
                    Way_L := Way_L::Purchase;
                    // <-- 20151104-CER126-TER01.01.00-2427-FBE01
                  END;
                  IF recMATIFContractPurc_L."Nb Lot Invoiced" +
                      (recMATIFContractSales_L."Lot Number" - recMATIFContractSales_L."Nb Lot Invoiced")
                       > recMATIFContractPurc_L."Lot Number" THEN BEGIN
                    LotPurch_L := recMATIFContractPurc_L."Nb Lot Invoiced" +
                          (recMATIFContractPurc_L."Lot Number" - recMATIFContractPurc_L."Nb Lot Invoiced");
                    Lotsales_L := recMATIFContractSales_L."Nb Lot Invoiced" +
                          (recMATIFContractPurc_L."Lot Number" - recMATIFContractPurc_L."Nb Lot Invoiced");
                  END
                  ELSE BEGIN
                    LotPurch_L := recMATIFContractPurc_L."Nb Lot Invoiced"
                          + (recMATIFContractSales_L."Lot Number" - recMATIFContractSales_L."Nb Lot Invoiced");
                    Lotsales_L := recMATIFContractSales_L."Nb Lot Invoiced"
                          + (recMATIFContractSales_L."Lot Number" - recMATIFContractSales_L."Nb Lot Invoiced");

                  // <-- 20161118-WI00-ADV01.01.00-2814-OLE01
                  Amount_P := (recMATIFContractSales_L."Farmer Price" - recMATIFContractPurc_L."Farmer Price" -
                        recMATIFContractArbFarmer_G.Commission);

                  IF (Lotsales_L <> 0) AND (LotPurch_L <> 0) THEN BEGIN
                    IF (Amount_P <= 0) AND (recSalesHeaderVar_P."Document Type" = recSalesHeaderVar_P."Document Type"::Invoice) THEN

                             recMATIFContractPurc_L,recMATIFContractSales_L,Way_L);
                      END;
                    IF (Amount_P > 0) AND (recSalesHeaderVar_P."Document Type" = recSalesHeaderVar_P."Document Type"::"Credit Memo")
                      THEN BEGIN
                        CreateDocFirstLine(recSalesHeaderVar_P,(Lotsales_L - recMATIFContractSales_L."Nb Lot Invoiced")
                            * recCMCSetup_G."MATIF Weight per Lot",Amount_P,FirstSignOffDate_L,
                            recMATIFContractPurc_L."Farmer Price",recMATIFContractSales_L."Farmer Price",
                             recMATIFContractPurc_L,recMATIFContractSales_L,Way_L);
                      END;
                  END;
                END;
              UNTIL recMATIFContractPurc_L.NEXT = 0;
          END;
        UNTIL recMATIFContractSales_L.NEXT = 0;
    END;
