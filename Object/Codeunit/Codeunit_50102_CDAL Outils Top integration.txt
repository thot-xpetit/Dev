      TableOetExped2_L@1000000006 : Record 8053443;

      TableSalesLine_L@1000000003 : Record 37;
      OetExpedTout_L@1000000004 : Boolean;
      Fin_L@1000000005 : Boolean;
    BEGIN
      IF TableOetExped_L.FINDSET THEN BEGIN
        REPEAT
          // Commande
          TableOetCmde_L.SETRANGE(TableOetCmde_L.OrderExtern, TableOetExped_L.OrderNo);
          TableOetCmde_L.SETRANGE("Line No. Extern", TableOetExped_L.OrderLineNo);
          IF TableOetCmde_L.FINDFIRST THEN BEGIN
            IF TableSalesHeader_L.GET(TableOetCmde_L."Document Type", TableOetCmde_L."Document No.") THEN BEGIN
              IF TableSalesHeader_L."Freezed Document" THEN BEGIN
                // Toutes lignes commandes dans exp'dition
                OetExpedTout_L := TRUE;
                TableSalesLine_L.SETRANGE("Document Type", TableSalesHeader_L."Document Type");
                TableSalesLine_L.SETRANGE("Document No.", TableSalesHeader_L."No.");
                IF TableSalesLine_L.FINDSET THEN BEGIN
                  Fin_L := FALSE;
                  REPEAT
                    // Exp'dition OET ?
                    // V'rifier que ligne non dans un Bon d'exp'dition
                    TableOetExped2_L.SETRANGE(OrderNo, TableOetCmde_L.OrderExtern);
                    TableOetExped2_L.SETRANGE(OrderLineNo, TableOetCmde_L."Line No. Extern");
                    IF TableOetExped2_L.ISEMPTY THEN BEGIN
                      OetExpedTout_L := FALSE;
                    END;
                      // Pas dans une exp'dition on peut d'figer.
                    IF OetExpedTout_L THEN BEGIN
                      IF TableSalesLine_L.NEXT = 0 THEN BEGIN
                        Fin_L := TRUE;
                      END;
                    END;
                  UNTIL Fin_L OR NOT OetExpedTout_L;
                END;
                IF OetExpedTout_L THEN BEGIN
                  // Pas dans une exp'dition on peut d'figer.
                  //<--20150702-Int19-CMC02.02.00-ADVITAM-CDL001
                  TableSalesHeader_L."Freezed Document" := FALSE;
                  TableSalesHeader_L.MODIFY(TRUE);
                END;
              END;
            END;
          END;
        UNTIL TableOetExped_L.NEXT = 0;
      END;
    END;

                  END;
                END;
                I_L += 1
              UNTIL I_L > 2;
            END;
          UNTIL TableProDat_L.NEXT = 0;
        END;
        MESSAGE('Modification %1, Identique %2, Erreur %3', NbModif_L, NbIdem_L, NbErreur_L);
      END;
    END;

    END.
