    PROCEDURE SetVendor@1000000007(VAR ImportInvHeader@1000000000 : Record 8087767);
    VAR
      VendorNo@1000000001 : Code[20];
      Vendor@1000000002 : Record 23;
      SicapaSetup@1000000003 : Record 8053408;
    BEGIN
      VendorNo := '';
      IF (ImportInvHeader."Source Type" = ImportInvHeader."Source Type"::EDI) THEN BEGIN
        IF (SicapaSetup.GET()) THEN BEGIN
          IF (SicapaSetup.Vendor <> '') THEN BEGIN
            VendorNo := SicapaSetup.Vendor;
          END ELSE BEGIN
            VendorNo := MoteurInterface_G.GetReferentialCorrespondance(VendorTemplateCode_G, TRUE,
                          ImportInvHeader.Emetteur, TRUE);
          END;
        END ELSE BEGIN
          VendorNo := MoteurInterface_G.GetReferentialCorrespondance(VendorTemplateCode_G, TRUE,
                        ImportInvHeader.Emetteur, TRUE);
        END;
      END ELSE BEGIN
        IF ((ImportInvHeader.Emetteur = '') AND (ImportInvHeader."Registration No."  = '')) THEN BEGIN
          VendorNo := '';
          ImportInvHeader.Status := ImportInvHeader.Status::Warning;
          ImportInvHeader."Status Message" := TErrNoVendor;
        END ELSE BEGIN
          Vendor.RESET();
          IF ((ImportInvHeader.Emetteur <> '') AND (ImportInvHeader."Registration No."  <> '')) THEN BEGIN
            Vendor.SETRANGE(Vendor."Registration No.", ImportInvHeader.Emetteur);
            IF (Vendor.COUNT > 1) THEN BEGIN
              ImportInvHeader.Status := ImportInvHeader.Status::Warning;
              ImportInvHeader."Status Message" := STRSUBSTNO(TErrMultiVendor, ImportInvHeader.Emetteur);
            END ELSE BEGIN
              IF (Vendor.FINDFIRST()) THEN BEGIN
                VendorNo := Vendor."No.";
              END ELSE BEGIN
                ImportInvHeader.Status := ImportInvHeader.Status::Warning;
                ImportInvHeader."Status Message" := TErrNoVendor;
              END;
            END;
          END ELSE BEGIN
            IF ((ImportInvHeader.Emetteur <> '') AND (ImportInvHeader."Registration No."  = '')) THEN BEGIN
              Vendor.SETRANGE(Vendor."Registration No.", ImportInvHeader.Emetteur);
              IF (Vendor.COUNT > 1) THEN BEGIN
                ImportInvHeader.Status := ImportInvHeader.Status::Warning;
                ImportInvHeader."Status Message" := STRSUBSTNO(TErrMultiVendor, ImportInvHeader.Emetteur);
              END ELSE BEGIN
                IF (Vendor.FINDFIRST()) THEN BEGIN
                  VendorNo := Vendor."No.";
                END ELSE BEGIN
                  ImportInvHeader.Status := ImportInvHeader.Status::Warning;
                  ImportInvHeader."Status Message" := TErrNoVendor;
                END;
              END;
            END ELSE BEGIN
              IF ((ImportInvHeader.Emetteur = '') AND (ImportInvHeader."Registration No."  <> '')) THEN BEGIN
                Vendor.SETFILTER(Vendor."Registration No.", '%1*',ImportInvHeader."Registration No.");
                IF (Vendor.COUNT > 1) THEN BEGIN
                  ImportInvHeader.Status := ImportInvHeader.Status::Warning;
                  ImportInvHeader."Status Message" := STRSUBSTNO(TErrMultiVendor, ImportInvHeader.Emetteur);
                END ELSE BEGIN
                  IF (Vendor.FINDFIRST()) THEN BEGIN
                    VendorNo := Vendor."No.";
                  END ELSE BEGIN
                    ImportInvHeader.Status := ImportInvHeader.Status::Warning;
                    ImportInvHeader."Status Message" := TErrNoVendor;
                  END;
                END;
              END ELSE BEGIN
                VendorNo := '';
                ImportInvHeader.Status := ImportInvHeader.Status::Warning;
                ImportInvHeader."Status Message" := TErrNoVendor;
              END;
            END;
          END
        END
      END;
      Vendor.RESET();
      IF (ImportInvHeader."Vendor No." = '') THEN BEGIN
        IF (VendorNo <> '') THEN BEGIN
          IF (Vendor.GET(VendorNo)) THEN BEGIN
            ImportInvHeader.VALIDATE("Vendor No.", Vendor."No.");
            ImportInvHeader.VALIDATE("VAT Registration No.", Vendor."VAT Registration No.");
          END;
        END;
      END;
      //ImportInvHeader.MODIFY();
    END;
