OBJECT Codeunit 50063 MAJ Campagne T_Prix TERNOVEO
{
  OBJECT-PROPERTIES
  {
    Date=01/07/16;
    Heure=10:39:50;
    Modifi'=Oui;
    Liste versions=REP001;
  }
  PROPERTIES
  {
    OnRun=VAR
            PrixVenteCampVide_G@1000000000 : Record 7002;
            CptErr_L@1000000001 : Integer;
          BEGIN

            //===============================================================================================================================
            //                      Mises . jour APPROS
            //===============================================================================================================================

            Cpt  := 0;
            Cpt2 := 0;
            CptErr_L := 0;
            PrixVente_G.RESET;
            Total := PrixVente_G.COUNT;
            Fenetre.OPEN('Maj donnees prix vente                  \' +
                         'Nb lecture                      #1######\' +
                         'Nb mises a jour                 #2######\' +
                         'Nb Erreur                       #3######\' +
                                      '@4@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@',Cpt,Cpt2);

            PrixVente_G.SETFILTER(PrixVente_G."Campaign No.", '<> %1', '');
            IF PrixVente_G.FIND('-') THEN
            REPEAT
              IF (PrixVente_G."Campaign No." <> '') THEN BEGIN
                // V'rifier si existe pas d'j. sur campagne vide.
                IF PrixVenteCampVide_G.GET(PrixVente_G."Item No.",PrixVente_G."Sales Type",PrixVente_G."Sales Code",
                  PrixVente_G."Starting Date",PrixVente_G."Currency Code",PrixVente_G."Variant Code",
                  PrixVente_G."Unit of Measure Code",PrixVente_G."Minimum Quantity",'') THEN BEGIN
                  CptErr_L := CptErr_L + 1;
                END ELSE BEGIN
                  PrixVente_G.RENAME(PrixVente_G."Item No.",PrixVente_G."Sales Type",PrixVente_G."Sales Code",
                    PrixVente_G."Starting Date",PrixVente_G."Currency Code",PrixVente_G."Variant Code",
                    PrixVente_G."Unit of Measure Code",PrixVente_G."Minimum Quantity",'');
                  Cpt2 := Cpt2 + 1;
                END;
              END;

            Cpt := Cpt + 1;
            Fenetre.UPDATE;
            Fenetre.UPDATE(1,Cpt);
            Fenetre.UPDATE(2,Cpt2);
            Fenetre.UPDATE(3,CptErr_L);
            Fenetre.UPDATE(4,ROUND(Cpt / Total * 10000, 1));

            UNTIL PrixVente_G.NEXT = 0;
            MESSAGE ('Cpt = %1; Cpt2 = %2; CptErr = %3', Cpt, Cpt2, CptErr_L);

            Fenetre.CLOSE;
          END;

  }
  CODE
  {
    VAR
      Total@1000000004 : Integer;
      Cpt@1000000005 : Integer;
      Cpt2@1000000006 : Integer;
      Cpt3@1000000022 : Integer;
      Fenetre@1000000007 : Dialog;
      Ok_lancement@1000000013 : Boolean;
      ToolsFile_L@1000000029 : Codeunit 8053405;
      Texte1_G@1000000030 : Text[50];
      Texte2_G@1000000031 : Text[50];
      Code10_G@1000000034 : Code[10];
      PrixVente_G@1000000000 : Record 7002;

    BEGIN
    {
      Liste modifications :

      20160630-Rpt01-ADV01.01.00_ADVITAM-KRE-001 : Suppression de la campagne sur les lignes de prix de vente
    }
    END.
  }
}
