    PROCEDURE ExportContactVet@1000000000(FileName_P@1000000000 : Text[128]) Result_L : Text[256];
    VAR
      "<-- 3LI LOCALS -->"@1000000001 : Integer;
      TableParameter_L@1000000002 : Record 8053413;
      ConnectorName_L@1000000009 : Code[20];
      ImpExpFolder_L@1000000008 : Text[256];
      LogFolder_L@1000000007 : Text[256];
      ExtensionFile_L@1000000006 : Text[10];
      PrefixFile_L@1000000005 : Text[10];
      FileName_L@1000000021 : Text[256];
      FilePath_L@1000000023 : Text[256];
      FileN_L@1000000022 : Text[256];
      NbCarEcrit_L@1000000027 : Integer;
      FileLogName_L@1000000029 : Text[256];
      TxtTemp_L@1000000046 : Text[255];
    BEGIN
      //----------------------------------------------------------------------------------------------------------------------------------
      // Extraction des contacts v't'rinatire : table "Contact"   pour Cr'ation du fichier WMS :  WMSCVSSAAMMJJhhMNSSsss.dat
      // Sorties des erreurs dans WMS_AAAAMMJJHHMNSSmmmn.txt
      //----------------------------------------------------------------------------------------------------------------------------------
      // Entr'es:
      //   - FileName_P  : Nom du fichier PC . G'n'rer
      // Sorties:
      //   - Result_L    : Erreur Retourn'e, Vide si Ok
      //----------------------------------------------------------------------------------------------------------------------------------
      FileNameOuvert_G := FALSE;
      NbCarEcrit_L := 0;
      MoteurInterface_G.GetParameter(CodeUnit_G, ConnectorName_L, ImpExpFolder_L, LogFolder_L, ExtensionFile_L, PrefixFile_L);
      MoteurInterface_G.GetTemplatesCode (ConnectorName_L, ItemTemplateCode_G, CustomerTemplateCode_G, VendorTemplateCode_G,
        ContactTemplateCode_G, AccountTemplateCode_G, GlobalTemplateCode_G);
      ToolsFile_G.LogInit(LogFolder_L, 'WMS');
      ToolsFile_G.LogInitCodeUnit(CodeUnit_G);

      // Nom du fichier pass' en paramStre ou Nom constitu' par r'p. du Robot & Nom fic. r'cup'r' du robot
      IF FileName_P = '' THEN BEGIN
        IF (NOT TableParameter_L.FINDFIRST) THEN BEGIN
          ToolsFile_G.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0001, TableParameter_L.TABLECAPTION, COMPANYNAME), 1, MAXSTRLEN(Result_L)));
        END;
      END;

      IF ToolsFile_G.LogErreurType() <> 'E' THEN BEGIN
        IF NOT TableParameter_L.FINDFIRST THEN BEGIN
          ToolsFile_G.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0001, TableParameter_L.TABLECAPTION , COMPANYNAME), 1, MAXSTRLEN(Result_L)));
        END ELSE BEGIN
          IF FileName_P <> '' THEN BEGIN
            FileName_L := FileName_P;
          END ELSE BEGIN
            FileName_L := ImpExpFolder_L + '\' + DELCHR(PrefixFile_L, '<>', ' ') +
              FORMAT(CURRENTDATETIME, 0,'<Year4><Month,2><Day,2><Hours24,2><Minutes,2><Seconds,2><Second dec>') + '.'+ExtensionFile_L;
          END;
          ToolsFile_G.LogInitFileExecute(FileName_L);
          ToolsFile_G.SplitFilename(FileName_L, FilePath_L, FileN_L);
          FileTemp_G := FilePath_L + 'Temp_' + FileN_L;

          TableContact_G.RESET;
          TableContact_G.SETRANGE(Vetenarian, TRUE);
          IF TableContact_G.FINDSET THEN BEGIN
            REPEAT
              ExportContactTexte();
            UNTIL TableContact_G.NEXT=0;
          END;

          IF FileNameOuvert_G THEN BEGIN
            NbCarEcrit_L := FileExport_G.LEN;
            FileExport_G.CLOSE;
          END;

        END;
      END;

      IF (NbCarEcrit_L > 0) THEN BEGIN
        IF ToolsFile_G.LogErreurType() = 'E' THEN BEGIN
          IF FileNameOuvert_G THEN BEGIN
            FILE.ERASE(FileTemp_G);
          END;
        END ELSE BEGIN
          IF RENAME(FileTemp_G, FileName_L) THEN BEGIN
          END;
        END;
      END ELSE BEGIN
        IF FileNameOuvert_G THEN BEGIN
          FILE.ERASE(FileTemp_G);
        END;
      END;

      CLEAR(Result_L);
      FileLogName_L := ToolsFile_G.LogName;
      ToolsFile_G.LogClose();
      IF ToolsFile_G.LogErreurType() = 'E' THEN BEGIN
        Result_L := 'ERREUR ' + FileLogName_L;
        ERROR(Result_L);
      END ELSE BEGIN
        IF ToolsFile_G.LogErreurType() = 'I' THEN BEGIN
          Result_L := Text0005;
          MoteurInterface_G.InsertLog(CodeUnit_G, FileLogName_L);
        END ELSE BEGIN
          MoteurInterface_G.InsertOk(CodeUnit_G);
        END;
      END;
    END;

    PROCEDURE ExportContactTexte@1000000001() Result_L : Text[256];
    VAR
      "<-- 3LI LOCALS -->"@1000000001 : Integer;
      TableCustomer_L@1000000034 : Record 18;
      TableCodPost_L@1000000033 : Record 225;
      FormatFieldsManage_L@1000000000 : Codeunit 8060985;
      Tab_L@1000000030 : Char;
      TxtEcrit_L@1000000031 : Text[1024];
      ZoneNum_L@1000000044 : Integer;
      FormeJuridique_L@1000000045 : Text[2];
      TxtTemp_L@1000000046 : Text[255];
    BEGIN
      //----------------------------------------------------------------------------------------------------------------------------------
      // Ecrit contact dans fichier texte
      //----------------------------------------------------------------------------------------------------------------------------------
      // Entr'es:
      //   -
      // Sorties:
      //   - Result_L    : Erreur Retourn'e, Vide si Ok
      //----------------------------------------------------------------------------------------------------------------------------------
      IF NOT FileNameOuvert_G THEN BEGIN
        IF NOT FileExport_G.CREATE(FileTemp_G) THEN BEGIN
          ToolsFile_G.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0003, FileTemp_G, GETLASTERRORTEXT), 1, MAXSTRLEN(Result_L)));
          CLEARLASTERROR;
        END ELSE BEGIN
          FileExport_G.CREATEOUTSTREAM(FileExportOutstream_G);
          FileNameOuvert_G := TRUE;
        END;
      END;
      IF FileNameOuvert_G THEN BEGIN
        Tab_L :=9;
        TxtEcrit_L := '';
        TxtEcrit_L += FormatFieldsManage_L.Ascii2Ansi(COMPANYNAME) + FORMAT(Tab_L);
        TxtEcrit_L += FormatFieldsManage_L.Ascii2Ansi(TableContact_G."No.") + FORMAT(Tab_L);
        TxtEcrit_L += FormatFieldsManage_L.Ascii2Ansi(TableContact_G.Name) + FORMAT(Tab_L);
        TxtEcrit_L += FormatFieldsManage_L.Ascii2Ansi(TableContact_G.Address) + FORMAT(Tab_L);
        TxtEcrit_L += FormatFieldsManage_L.Ascii2Ansi(TableContact_G."Address 2") + FORMAT(Tab_L);
        TxtEcrit_L += FormatFieldsManage_L.Ascii2Ansi(TableContact_G."Post Code") + FORMAT(Tab_L);
        TxtEcrit_L += FormatFieldsManage_L.Ascii2Ansi(TableContact_G.City) + FORMAT(Tab_L);
        TxtEcrit_L += FormatFieldsManage_L.Ascii2Ansi(TableContact_G."Search Name") + FORMAT(Tab_L);
        TxtEcrit_L += FormatFieldsManage_L.Ascii2Ansi(TableContact_G."Registration No.") + FORMAT(Tab_L);
        TxtEcrit_L += FormatFieldsManage_L.Ascii2Ansi(TableContact_G."Country/Region Code") + FORMAT(Tab_L);
        // Indicatif T'l'phone
        CASE TRUE OF
          TableContact_G."Country/Region Code" = 'FR':
            TxtEcrit_L += '33' + FORMAT(Tab_L);
          TableContact_G."Country/Region Code" = 'BE':
            TxtEcrit_L += '32' + FORMAT(Tab_L);
          TableContact_G."Country/Region Code" = 'NL':
            TxtEcrit_L += '31' + FORMAT(Tab_L);
          TableContact_G."Country/Region Code" = 'GB':
            TxtEcrit_L += '44' + FORMAT(Tab_L);
          ELSE BEGIN
            TxtEcrit_L += '00' + FORMAT(Tab_L);
          END;
        END;
        TxtEcrit_L += FormatFieldsManage_L.Ascii2Ansi(DELCHR(DELCHR(TableContact_G."Phone No.", '=', ' '), '=', '.'))
          + FORMAT(Tab_L);
        TxtEcrit_L += FormatFieldsManage_L.Ascii2Ansi(DELCHR(DELCHR(TableContact_G."Fax No.", '=', ' '), '=', '.'))
          + FORMAT(Tab_L);
        TxtEcrit_L += FormatFieldsManage_L.Ascii2Ansi(DELCHR(DELCHR(TableContact_G."Mobile Phone No.",'=',' '),'=','.'))
          + FORMAT(Tab_L);
        TxtEcrit_L += FormatFieldsManage_L.Ascii2Ansi(TableContact_G."E-Mail") + FORMAT(Tab_L);
        // D'partement + Commune (Code Insee)
        CLEAR(TableCodPost_L);
        TableCodPost_L.SETRANGE(TableCodPost_L.Code, TableContact_G."Post Code");
        TableCodPost_L.SETRANGE(TableCodPost_L.City, TableContact_G.City);
        IF TableCodPost_L.FINDFIRST THEN BEGIN
          IF NOT EVALUATE(ZoneNum_L, COPYSTR(TableCodPost_L."INSEE No.", 1, 2)) THEN BEGIN
            TxtEcrit_L += '99' + FORMAT(Tab_L);
            TxtEcrit_L += '999' + FORMAT(Tab_L);
          END ELSE BEGIN
            TxtEcrit_L += COPYSTR(TableCodPost_L."INSEE No.", 1, 2) + FORMAT(Tab_L);
            TxtEcrit_L += COPYSTR(TableCodPost_L."INSEE No.", 3, 3) + FORMAT(Tab_L);
          END;
        END ELSE BEGIN
          IF TableContact_G."Country/Region Code" <> 'FR' THEN BEGIN
            TxtEcrit_L += '99' + FORMAT(Tab_L);
            TxtEcrit_L += '999' + FORMAT(Tab_L);
          END ELSE BEGIN
            TxtEcrit_L += '' + FORMAT(Tab_L);
             TxtEcrit_L += '' + FORMAT(Tab_L);
            ToolsFile_G.LogWriteInfo(COPYSTR(STRSUBSTNO(Text0007, TableContact_G."No.", TableContact_G."Post Code",
              TableContact_G.City), 1, MAXSTRLEN(Result_L)));
          END;
        END;

        // Date cr'ation (Absent dans NavAgri --> Date Modification)
        TxtEcrit_L += FORMAT(TableContact_G."Last Date Modified", 0,'<Year4><Month,2><Day,2>') +FORMAT(Tab_L);
        // Date modification
        TxtEcrit_L += FORMAT(TableContact_G."Last Date Modified", 0,'<Year4><Month,2><Day,2>') +FORMAT(Tab_L);

        CLEAR(FormeJuridique_L);
        IF TableContact_G."Salutation Code" <> '' THEN BEGIN

            'Salutation', 'Code', TableContact_G."Salutation Code", FALSE, TRUE));
          IF FormeJuridique_L = '' THEN BEGIN
            ToolsFile_G.LogWriteInfo(COPYSTR(STRSUBSTNO(Text0010, TableContact_G.FIELDNAME("Salutation Code"),
              TableContact_G."Salutation Code", TableContact_G."No."), 1, MAXSTRLEN(Result_L)));
          END;
        END;
        TxtEcrit_L += TableContact_G."Salutation Code" + FORMAT(Tab_L);
        TxtEcrit_L += FormeJuridique_L + FORMAT(Tab_L);

        TxtEcrit_L += FORMAT(TableContact_G."Date Company Creation", 0,'<Year4><Month,2><Day,2>') + FORMAT(Tab_L);

        FileExportOutstream_G.WRITETEXT(TxtEcrit_L);
        FileExportOutstream_G.WRITETEXT();
      END;
