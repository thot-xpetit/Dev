    PROCEDURE DimFromPurch@8053000(VAR P_TemDocDim@8053001 : TEMPORARY Record 357;VAR P_PurchaseLine@8053000 : Record 39);
    VAR
      ItemCode_L@8053005 : Code[20];
      LocationCode_L@8053004 : Code[10];
      RecPurchLine_L@8053003 : Record 39;
      RecDefaultDimension_L@8053002 : Record 352;
      PurchHd_L@1000000000 : Record 38;
      PostedCollReceipt_L@1000000001 : Record 8052995;
      PostedDelivery_L@1000000002 : Record 8053014;
      PurchReceiptHd_L@1000000003 : Record 120;
      DecDim_L@1000000004 : Record 357;
    BEGIN
      // appeler  . partir du CU 90
      G_CMCSetUp.GET;
      G_GLSetup.GET;

      ItemCode_L:='';
      LocationCode_L:='';

      IF (P_PurchaseLine.Type=P_PurchaseLine.Type::Item) AND (P_PurchaseLine."No."<>'') THEN BEGIN
        // --> 20150731-APP27-TER01.01.00-2017-PDC01
        //BEFORE
        //ItemCode_L:=P_PurchaseLine."No.";

        //AFTER
        // --> 20150806-APP27-TER01.01.00-2017-PDC01
        IF (P_PurchaseLine."Document Type"=P_PurchaseLine."Document Type"::Invoice) AND
           PurchReceiptHd_L.GET(P_PurchaseLine."Receipt No.") THEN BEGIN
          IF NOT PurchHd_L.GET(PurchHd_L."Document Type"::Order,PurchReceiptHd_L."Order No.") THEN
            CLEAR(PurchHd_L);
        END ELSE
        // <-- 20150731-APP27-TER01.01.00-2017-PDC01
        PurchHd_L.GET(P_PurchaseLine."Document Type",P_PurchaseLine."Document No.");

        CASE TRUE OF
          (PurchHd_L."Source Document Type"=PurchHd_L."Source Document Type"::"Collection Receipt") AND
          PostedCollReceipt_L.GET(PostedCollReceipt_L."Receipt Type"::Receipt,PurchHd_L."Source Document No.") :
            ItemCode_L:=PostedCollReceipt_L."Item No.";

          (PurchHd_L."Source Document Type"=PurchHd_L."Source Document Type"::Delivery) AND
          PostedDelivery_L.GET(PostedDelivery_L."Delivery Type"::Delivery,PurchHd_L."Source Document No.") :
            ItemCode_L:=PostedDelivery_L."Item No.";

          (PurchHd_L."Source Document Type"=PurchHd_L."Source Document Type"::"Ind. Transfer") AND
          PostedDelivery_L.GET(PostedDelivery_L."Delivery Type"::Transfert,PurchHd_L."Source Document No.") :
            ItemCode_L:=PostedDelivery_L."Item No.";

          (PurchHd_L."Source Document Type"=PurchHd_L."Source Document Type"::Resumption) AND
          PostedDelivery_L.GET(PostedDelivery_L."Delivery Type"::Resumption,PurchHd_L."Source Document No.") :
            ItemCode_L:=PostedDelivery_L."Item No.";

          ELSE
            ItemCode_L:=P_PurchaseLine."No.";
        END;
        // <-- 20150731-APP27-TER01.01.00-2017-PDC014

        LocationCode_L:=P_PurchaseLine."Location Code";
      END ELSE BEGIN
        // ligne frais annexes associ'
        IF P_PurchaseLine."Item Charge Root Line No." <> 0 THEN BEGIN
          RecPurchLine_L.GET(P_PurchaseLine."Document Type",P_PurchaseLine."Document No.",P_PurchaseLine."Item Charge Root Line No.");
          ItemCode_L:=RecPurchLine_L."No.";
          LocationCode_L:=RecPurchLine_L."Location Code";
        END;
      END;

      IF NOT G_recItem.GET(ItemCode_L) THEN EXIT;

      // --> 20160602-WI00-ADV01.01.00-2056-JCD01 : Pas de chargement d'axe analytique Location si CG et Axe d'j. pr'sent
      IF (P_PurchaseLine.Type = P_PurchaseLine.Type::"G/L Account") THEN
        IF (LocationCode_L = '') THEN // La ligne n'est pas li'e . une ligne article
          IF G_CMCSetUp."Location Dimension" <> '' THEN BEGIN
            DecDim_L.RESET;
            DecDim_L.SETRANGE("Table ID",39);
            DecDim_L.SETRANGE("Document Type",P_PurchaseLine."Document Type");
            DecDim_L.SETRANGE("Document No.",P_PurchaseLine."Document No.");
            DecDim_L.SETRANGE("Line No.",P_PurchaseLine."Line No.");
            DecDim_L.SETRANGE("Dimension Code",G_CMCSetUp."Location Dimension");
            IF NOT DecDim_L.ISEMPTY THEN //Analytique lieu saisi prioritaire
              EXIT;
          END;
      // <-- 20160602-WI00-ADV01.01.00-2056-JCD01

      // --> 20151106-WI00-TER01.01.00-2338-BSI01
      {before
      G_recLocation.INIT;
      IF G_recLocation.GET(P_PurchaseLine."Location Code") THEN ;
      after}
      IF NOT G_recLocation.GET(P_PurchaseLine."Location Code") THEN
        CLEAR(G_recLocation);
      // <-- 20151106-WI00-TER01.01.00-2338-BSI01

      FOR i := 0 TO 4 DO BEGIN
        IF GetCMCDim(i) <> '' THEN BEGIN
          IF NOT G_recDimValue.GET(GetCMCDim(i),GetDimValue(G_recItem,G_recLocation,i)) THEN
            CreatDimVal(GetCMCDim(i),GetDimValue(G_recItem,G_recLocation,i),GetCMCDimDescrp(i,GetDimValue(G_recItem,G_recLocation,i)));
            CreatTempDocDim(P_TemDocDim,DATABASE::"Purchase Line",P_PurchaseLine."Document No.",P_PurchaseLine."Line No.",
              P_PurchaseLine."Document Type",GetCMCDim(i),GetDimValue(G_recItem,G_recLocation,i),P_PurchaseLine."Analytical Quantity")
        END;
        IF GetCMCDim(i) = G_GLSetup."Global Dimension 1 Code" THEN BEGIN
          P_PurchaseLine."Shortcut Dimension 1 Code":=GetDimValue(G_recItem,G_recLocation,i);
          P_PurchaseLine.MODIFY;
        END;
        IF GetCMCDim(i) = G_GLSetup."Global Dimension 2 Code" THEN BEGIN
          P_PurchaseLine."Shortcut Dimension 2 Code":=GetDimValue(G_recItem,G_recLocation,i);
          P_PurchaseLine.MODIFY;
        END
      END;

      RecDefaultDimension_L.RESET;
      RecDefaultDimension_L.SETRANGE("Table ID",27);
      RecDefaultDimension_L.SETRANGE("No.",ItemCode_L);
      IF RecDefaultDimension_L.FINDSET THEN
        REPEAT
          CreatTempDocDim(P_TemDocDim,DATABASE::"Purchase Line",P_PurchaseLine."Document No.",P_PurchaseLine."Line No.",
            P_PurchaseLine."Document Type",RecDefaultDimension_L."Dimension Code",RecDefaultDimension_L."Dimension Value Code",
            P_PurchaseLine."Analytical Quantity");
          IF RecDefaultDimension_L."Dimension Code" = G_GLSetup."Global Dimension 1 Code" THEN BEGIN
            P_PurchaseLine."Shortcut Dimension 1 Code":=RecDefaultDimension_L."Dimension Value Code";
            P_PurchaseLine.MODIFY;
          END;
          IF RecDefaultDimension_L."Dimension Code" = G_GLSetup."Global Dimension 2 Code" THEN BEGIN
            P_PurchaseLine."Shortcut Dimension 2 Code":=RecDefaultDimension_L."Dimension Value Code";
            P_PurchaseLine.MODIFY;
          END
        UNTIL RecDefaultDimension_L.NEXT=0;
    END;

          LocationCode_L:=RecSalesLine_L."Location Code";
        END;

        // produit gratuit
        IF P_SalesLine."Free Item Root Line"<>0 THEN BEGIN
          RecSalesLine_L.GET(P_SalesLine."Document Type",P_SalesLine."Document No.",P_SalesLine."Free Item Root Line");

        END;

          IF RecSalesLine_L.Type <> RecSalesLine_L.Type::Item THEN
            OriginalItemLine(RecSalesLine_L);
          // <-- 20160301-WI00-ADV01.01.00-2358-JCD01

    PROCEDURE DimFromItemJnl@8053019(VAR P_TemJnlDim@8053001 : TEMPORARY Record 356;VAR P_ItemJnl@8053000 : Record 83;P_DimFromCMCObj@8053003 : Boolean);
    VAR
      L_@8053002 : Record 39;
      RecDefaultDimension_L@8053004 : Record 352;
      RecDefaultDimension2_L@8053007 : Record 352;
      RecNewItem_L@8053005 : Record 27;
      Mutation_L@8053006 : Boolean;
    BEGIN
      // appeler  . partir du CU 22
      G_CMCSetUp.GET;
      G_GLSetup.GET;
      IF P_ItemJnl."Item No." = '' THEN
        EXIT;
      IF (P_ItemJnl."Journal Template Name"='') AND (P_DimFromCMCObj=FALSE) AND (P_ItemJnl."Item Charge No."='') AND
        (P_ItemJnl."BO Doc. No."='')
      THEN
        EXIT;
      IF NOT G_recItem.GET(P_ItemJnl."Item No.") THEN
        EXIT;

      // --> 20150217-WI00-CMC02.02.00-2763-DRN01
      //Cas de la mutation sur article diff'rent
      IF (P_ItemJnl."Entry Type" = P_ItemJnl."Entry Type"::Transfer) THEN
        IF (P_ItemJnl."Item No." <> P_ItemJnl."New Item No.") AND (P_ItemJnl."New Item No." <> '') THEN BEGIN
          RecNewItem_L.GET(P_ItemJnl."New Item No.");
          Mutation_L := TRUE;
        END;
      // <-- 20150217-WI00-CMC02.02.00-2763-DRN01

      // --> 20151106-WI00-TER01.01.00-2338-BSI01
      {before
      G_recLocation.INIT;
      IF G_recLocation.GET(P_ItemJnl."Location Code") THEN;
      after}
      IF NOT G_recLocation.GET(P_ItemJnl."Location Code") THEN
        CLEAR(G_recLocation);
      // <-- 20151106-WI00-TER01.01.00-2338-BSI01


      FOR i := 0 TO 4 DO BEGIN
        IF GetCMCDim(i) <> '' THEN BEGIN
          IF NOT G_recDimValue.GET(GetCMCDim(i),GetDimValue(G_recItem,G_recLocation,i)) THEN
            CreatDimVal(GetCMCDim(i),GetDimValue(G_recItem,G_recLocation,i),GetCMCDimDescrp(i,GetDimValue(G_recItem,G_recLocation,i)));
          // --> 20150217-WI00-CMC02.02.00-2763-DRN01
          IF Mutation_L THEN
            IF NOT G_recDimValue.GET(GetCMCDim(i),GetDimValue(RecNewItem_L,G_recLocation,i)) THEN
              CreatDimVal(GetCMCDim(i),GetDimValue(RecNewItem_L,G_recLocation,i),
                GetCMCDimDescrp(i,GetDimValue(RecNewItem_L,G_recLocation,i)));
          // <-- 20150217-WI00-CMC02.02.00-2763-DRN01
          IF P_ItemJnl."Entry Type" = P_ItemJnl."Entry Type"::Transfer THEN BEGIN
            // --> 20150217-WI00-CMC02.02.00-2763-DRN01
            IF Mutation_L THEN
              CreatTempJnlDim(P_TemJnlDim,DATABASE::"Item Journal Line",P_ItemJnl."Journal Template Name",P_ItemJnl."Journal Batch Name"
      ,
                P_ItemJnl."Line No.",GetCMCDim(i),GetDimValue(G_recItem,G_recLocation,i),GetDimValue(RecNewItem_L,G_recLocation,i))
            ELSE
              CreatTempJnlDim(P_TemJnlDim,DATABASE::"Item Journal Line",P_ItemJnl."Journal Template Name",P_ItemJnl."Journal Batch Name"
      ,
                P_ItemJnl."Line No.",GetCMCDim(i),GetDimValue(G_recItem,G_recLocation,i),GetDimValue(G_recItem,G_recLocation,i))
            // <-- 20150217-WI00-CMC02.02.00-2763-DRN01
          END ELSE
            CreatTempJnlDim(P_TemJnlDim,DATABASE::"Item Journal Line",P_ItemJnl."Journal Template Name",P_ItemJnl."Journal Batch Name",
              P_ItemJnl."Line No.",GetCMCDim(i),GetDimValue(G_recItem,G_recLocation,i),'')
        END;
        IF GetCMCDim(i) = G_GLSetup."Global Dimension 1 Code" THEN BEGIN
          P_ItemJnl."Shortcut Dimension 1 Code" := GetDimValue(G_recItem,G_recLocation,i);
          IF P_ItemJnl."Entry Type" = P_ItemJnl."Entry Type"::Transfer THEN BEGIN
            // --> 20150217-WI00-CMC02.02.00-2763-DRN01
            IF Mutation_L THEN
              P_ItemJnl."New Shortcut Dimension 1 Code" := GetDimValue(RecNewItem_L,G_recLocation,i)
            ELSE
            // <-- 20150217-WI00-CMC02.02.00-2763-DRN01
              P_ItemJnl."New Shortcut Dimension 1 Code" := GetDimValue(G_recItem,G_recLocation,i);
          END;
        END;
        IF GetCMCDim(i) = G_GLSetup."Global Dimension 2 Code" THEN BEGIN
          P_ItemJnl."Shortcut Dimension 2 Code" := GetDimValue(G_recItem,G_recLocation,i);
          IF P_ItemJnl."Entry Type" = P_ItemJnl."Entry Type"::Transfer THEN BEGIN
            // --> 20150217-WI00-CMC02.02.00-2763-DRN01
            IF Mutation_L THEN
              P_ItemJnl."New Shortcut Dimension 2 Code" := GetDimValue(RecNewItem_L,G_recLocation,i)
            ELSE
            // <-- 20150217-WI00-CMC02.02.00-2763-DRN01
              P_ItemJnl."New Shortcut Dimension 2 Code" := GetDimValue(G_recItem,G_recLocation,i);
          END;
        END
      END;

      RecDefaultDimension_L.RESET;
      RecDefaultDimension_L.SETRANGE("Table ID",27);
      RecDefaultDimension_L.SETRANGE("No.",P_ItemJnl."Item No.");
      IF RecDefaultDimension_L.FINDSET THEN
        REPEAT
          // --> 20150217-WI00-CMC02.02.00-2763-DRN01
          IF P_ItemJnl."Entry Type" = P_ItemJnl."Entry Type"::Transfer THEN BEGIN
            IF Mutation_L THEN BEGIN
              IF RecDefaultDimension2_L.GET(27,RecNewItem_L."No.",0,RecDefaultDimension_L."Dimension Code") THEN
                CreatTempJnlDim(P_TemJnlDim,DATABASE::"Item Journal Line",P_ItemJnl."Journal Template Name",
                  P_ItemJnl."Journal Batch Name",P_ItemJnl."Line No.",RecDefaultDimension_L."Dimension Code",
                  RecDefaultDimension_L."Dimension Value Code",RecDefaultDimension2_L."Dimension Value Code")
            END ELSE
              CreatTempJnlDim(P_TemJnlDim,DATABASE::"Item Journal Line",P_ItemJnl."Journal Template Name",P_ItemJnl."Journal Batch Name"
      ,
                P_ItemJnl."Line No.",RecDefaultDimension_L."Dimension Code",RecDefaultDimension_L."Dimension Value Code",
                RecDefaultDimension_L."Dimension Value Code")
          END ELSE
          // <-- 20150217-WI00-CMC02.02.00-2763-DRN01
            CreatTempJnlDim(P_TemJnlDim,DATABASE::"Item Journal Line",P_ItemJnl."Journal Template Name",P_ItemJnl."Journal Batch Name",
              P_ItemJnl."Line No.",RecDefaultDimension_L."Dimension Code",RecDefaultDimension_L."Dimension Value Code",'');

          IF RecDefaultDimension_L."Dimension Code" = G_GLSetup."Global Dimension 1 Code" THEN BEGIN
            P_ItemJnl."Shortcut Dimension 1 Code" := RecDefaultDimension_L."Dimension Value Code";
            // --> 20150217-WI00-CMC02.02.00-2763-DRN01
            IF P_ItemJnl."Entry Type" = P_ItemJnl."Entry Type"::Transfer THEN
              IF Mutation_L THEN BEGIN
                IF RecDefaultDimension2_L.GET(27,RecNewItem_L."No.",0,RecDefaultDimension_L."Dimension Code") THEN
                  P_ItemJnl."New Shortcut Dimension 1 Code" := RecDefaultDimension2_L."Dimension Value Code";;
              END ELSE
                P_ItemJnl."New Shortcut Dimension 1 Code" := RecDefaultDimension_L."Dimension Value Code";;
            // <-- 20150217-WI00-CMC02.02.00-2763-DRN01
          END;
          IF RecDefaultDimension_L."Dimension Code" = G_GLSetup."Global Dimension 2 Code" THEN BEGIN
            P_ItemJnl."Shortcut Dimension 2 Code" := RecDefaultDimension_L."Dimension Value Code";
            // --> 20150217-WI00-CMC02.02.00-2763-DRN01
            IF P_ItemJnl."Entry Type" = P_ItemJnl."Entry Type"::Transfer THEN
              IF Mutation_L THEN BEGIN
                IF RecDefaultDimension2_L.GET(27,RecNewItem_L."No.",0,RecDefaultDimension_L."Dimension Code") THEN
                  P_ItemJnl."New Shortcut Dimension 2 Code" := RecDefaultDimension2_L."Dimension Value Code";;
              END ELSE
                P_ItemJnl."New Shortcut Dimension 2 Code" := RecDefaultDimension_L."Dimension Value Code";;
            // <-- 20150217-WI00-CMC02.02.00-2763-DRN01
          END
        UNTIL RecDefaultDimension_L.NEXT=0;
    END;
