    PROCEDURE UpdateContact@60030(RecContBusRel_P@60006 : Record 5054;xRecContBusRel_P@60007 : Record 5054;ContactNo_P@60001 : Code[20];ActionType_P@60000 : 'Insert,Delete,Rename,Init');
    VAR
      ContBusinessRelation_L@60002 : Record 5054;
      TmpContBusinessRelation_L@60003 : TEMPORARY Record 5054;
      TmpContact_L@60004 : TEMPORARY Record 5050;
      Cont_L@60005 : Record 5050;
    BEGIN
      WITH RecContBusRel_P DO BEGIN

        Cont_L.LOCKTABLE;
        Cont_L.GET(ContactNo_P);
        TmpContBusinessRelation_L.DELETEALL;

        TmpContact_L.INIT;
        TmpContact_L := Cont_L;
        TmpContact_L.INSERT;

        ContBusinessRelation_L.RESET;
        ContBusinessRelation_L.SETRANGE("Contact No.",ContactNo_P);
        IF ActionType_P = ActionType_P::Delete THEN
          ContBusinessRelation_L.SETFILTER("Business Relation Code",'<>%1',RecContBusRel_P."Business Relation Code");
        IF ActionType_P = ActionType_P::Rename THEN
          ContBusinessRelation_L.SETFILTER("Business Relation Code",'<>%1',xRecContBusRel_P."Business Relation Code");

        IF ContBusinessRelation_L.FINDFIRST THEN
          REPEAT
            TmpContBusinessRelation_L.INIT;
            TmpContBusinessRelation_L := ContBusinessRelation_L;
            TmpContBusinessRelation_L.INSERT;
          UNTIL ContBusinessRelation_L.NEXT = 0;

        IF (ActionType_P = ActionType_P::Insert) OR (ActionType_P = ActionType_P::Rename) THEN BEGIN
          TmpContBusinessRelation_L.INIT;
          TmpContBusinessRelation_L := RecContBusRel_P;
          TmpContBusinessRelation_L.INSERT;
        END;

        Cont_L."Is Sales Contact" := FALSE;
        Cont_L."Is Purchase Contact" := FALSE;
        Cont_L."Customer No." := '';
        Cont_L."Vendor No." := '';
        Cont_L."Bank Account No." := '';

        IF TmpContBusinessRelation_L.FINDFIRST THEN
          REPEAT
            IF TmpContBusinessRelation_L."Is Sales Contact" THEN
              Cont_L."Is Sales Contact" := TRUE;
            IF TmpContBusinessRelation_L."Is Purchase Contact" THEN
              Cont_L."Is Purchase Contact" := TRUE;
            IF (TmpContBusinessRelation_L."Link to Table" = TmpContBusinessRelation_L."Link to Table"::Customer) AND
               (TmpContBusinessRelation_L."No." <> '') THEN
              Cont_L."Customer No." := TmpContBusinessRelation_L."No.";
            IF (TmpContBusinessRelation_L."Link to Table" = TmpContBusinessRelation_L."Link to Table"::Vendor) AND
               (TmpContBusinessRelation_L."No." <> '') THEN
              Cont_L."Vendor No." := TmpContBusinessRelation_L."No.";
            IF (TmpContBusinessRelation_L."Link to Table" = TmpContBusinessRelation_L."Link to Table"::"Bank Account") AND
               (TmpContBusinessRelation_L."No." <> '') THEN
              Cont_L."Bank Account No." := TmpContBusinessRelation_L."No.";
          UNTIL TmpContBusinessRelation_L.NEXT = 0;

        Cont_L.MODIFY;

        Cont_L.OnModify(TmpContact_L);

      END;
    END;
