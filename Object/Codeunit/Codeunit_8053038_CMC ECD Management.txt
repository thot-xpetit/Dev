    PROCEDURE CalcRevSalesECDExpAmtToBalance@8053006(VAR ECDLedgerEntry_P@8053001 : Record 8053089;InvoicedQty_P@8053000 : Decimal);
    VAR
      ItemLedgerEntry_L@8053005 : Record 32;

      OldExpectedAmount_L@8053002 : ARRAY [3] OF Decimal;
    BEGIN
      // --> 20140805-APP33-CMC02.02.00-FDA01
      CLEAR(OldExpectedAmount_L);
      GLSetup.GET;

      WITH SalesECDLedgerEntry_L DO BEGIN
        // --> 20140912-APP33-CMC02.02.00-FDA01
        {

          OldExpectedQty_L := ItemLedgerEntry_L.Quantity;

        SETCURRENTKEY("Item Ledger Entry No.","Entry Type");
        SETRANGE("Item Ledger Entry No.",ECDLedgerEntry_P."Item Ledger Entry No.");
        SETRANGE("Entry Type","Entry Type"::"Direct Cost");
        SETFILTER("Document Type", '%1|%2', "Document Type"::"Sales Shipment", "Document Type"::"Sales Return Receipt");
        // --> 20170406-WI00-ADV01.01.00-26533-CJY01

            OldExpectedAmount_L[3] := OldExpectedAmount_L[3] + SalesECDLedgerEntry_L."ECD Unit Amount (Shipment)";

    END;
