    PROCEDURE FormatFileTest@1000000003(VAR RecInterfaceEnginesetup_P@1000000003 : Record 8061230;FileName_P@1000000000 : Text[205]) : Boolean;

        EXIT (FALSE);
      IF NOT EVALUATE(Minute_L, COPYSTR(FileName_L,cpt_L+14,2)) THEN
        EXIT (FALSE);
      IF NOT EVALUATE(Sec_L,COPYSTR(FileName_L,cpt_L+17,2)) THEN
        EXIT (FALSE);
      // --> 20190514-Wi00-ADV01.01.00-42431-MRU01
      //IF (Heure_L > 12)OR (Minute_L > 60) OR (Sec_L > 60) OR (COPYSTR(FileName_L,cpt_L+2,1) <> '_')
      IF (Heure_L > 24)OR (Minute_L > 60) OR (Sec_L > 60) OR (COPYSTR(FileName_L,cpt_L+2,1) <> '_')
      // <-- 20190514-Wi00-ADV01.01.00-42431-MRU01
                       OR (COPYSTR(FileName_L,cpt_L+5,1) <> '_') OR (COPYSTR(FileName_L,cpt_L+10,1)<> '_')
                       OR (COPYSTR(FileName_L,cpt_L+13,1)<> '_') OR (COPYSTR(FileName_L,cpt_L+16,1) <> '_')
                       OR (COPYSTR(FileName_L,cpt_L+19,1) <> '.')
                       OR ((UPPERCASE(COPYSTR(FileName_L,cpt_L+20)) <> UPPERCASE(RecInterfaceEnginesetup_P."File extension"))
                       OR ((UPPERCASE(Prefix_L)) <> UPPERCASE(RecInterfaceEnginesetup_P."File Prefix")))
                       OR (NOT EVALUATE(DateTime_L,FORMAT(DMY2DATE(Jour_L,Mois_L,Annee_L))))THEN
        EXIT (FALSE)
      ELSE
        EXIT (TRUE);
    END;

    PROCEDURE ExtractTextCsv@1000000008(Texte_P@1000000000 : Text[1024];VAR Position_P@1000000001 : Integer;Separateur_P@1000000002 : Text[1];LongueurMax_P@1000000005 : Integer;VAR Max_P@1000000006 : Boolean) Result_L : Text[1024];
    VAR
      PositionSep_L@1000000004 : Integer;
    BEGIN
      Max_G := FALSE;
      Result_L := COPYSTR (Texte_P, Position_P);
      PositionSep_L := STRPOS(Result_L, Separateur_P);
      IF PositionSep_L <> 0 THEN BEGIN
        IF PositionSep_L > LongueurMax_P+1 THEN BEGIN
          Max_G := TRUE;
          Result_L := COPYSTR(Result_L, 1, LongueurMax_P);
        END
        ELSE BEGIN
          Result_L := COPYSTR(Result_L, 1, PositionSep_L - 1);
        END;
        Position_P += PositionSep_L;
      END
      ELSE BEGIN
        IF STRLEN(Result_L) >LongueurMax_P THEN BEGIN
           Max_G := TRUE;
           Position_P := STRLEN(Texte_P) + 1;
        END;
      END;
    END;

    PROCEDURE FileContentControl@60003(VAR RecCMBInterfaceEnginesetup_P@60004 : Record 8061230);

            TROUV_G := FALSE;
            SeqBool_G := FALSE;
          END ELSE
            IF Ligne_G = 1  THEN BEGIN
              IF Type_G <> 'DEBUT' THEN
                TROUV_G := FALSE;
            END ELSE
              IF Ligne_G=2 THEN BEGIN
                IF Type_G<>'COM' THEN
                  TROUV_G := FALSE;
              END
              ELSE BEGIN
                IF Type_G = 'ENT' THEN
                  EnteteExist_G := TRUE
                ELSE
                  IF Type_G='FIN' THEN
                    Stop_G := TRUE;
              END
        END ELSE
          TROUV_G := FALSE;
        IF NOT TROUV_G AND NOT SeqBool_G THEN
          CduToolsFile_G.LogWriteErreur(COPYSTR(STRSUBSTNO(Text60011, Ligne_G),1, MAXSTRLEN(MsgErr_L)));
        IF (Ligne_G = 1) AND (NOT TROUV_G) THEN

        IF  Stop_G AND NOT EnteteExist_G  THEN
          CduToolsFile_G.LogWriteErreur(COPYSTR(STRSUBSTNO(Text60008, "File in use"),1, MAXSTRLEN(MsgErr_L)));
