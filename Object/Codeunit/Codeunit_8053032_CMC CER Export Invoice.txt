    PROCEDURE FormatLine@1120003(Data_P@1120004 : Variant;FieldLen_P@1120003 : Integer;PadChar_P@1120002 : Char;"Left/Right_P"@1120001 : 'Right,Left';EndOfText_P@1120000 : Char) : Text[250];
    VAR
      Data_L@1120008 : Text[250];
      Line_L@1120007 : Text[250];
      I_L@1120006 : Integer;
      NUL_L@1120005 : Char;
    BEGIN
      {ParamStres:
       Data_P       : Data
       FieldLen_P   : Longueur du champ  (Si=0, alors on prend la longueur de Data_P)
       PadChar_P    : CaractSre ajout'
       Left/Right_P : Cadrage  Gauche/Droit
       EndOfText_P  : CaractSre fin de champ}

      // Init
      //----------------------------------------------------------------------------------------------------------------------------------
      NUL_L  := 0;
      Data_L := FORMAT(Data_P);

      // --> 20130927-INT40/001-CMC02.02.00-MBE01 :
      IF FieldLen_P = 0 THEN
        FieldLen_P := STRLEN(Data_L);
      // <-- 20130927-INT40/001-CMC02.02.00-MBE01 :

      CASE EndOfText_P OF

      // No character end of text
      //----------------------------------------------------------------------------------------------------------------------------------
        NUL_L: BEGIN

      // same lenght
      //----------------------------------------------------------------------------------------------------------------------------------
          IF STRLEN(Data_L) = FieldLen_P THEN EXIT (Data_L);

      // Length higher
      //----------------------------------------------------------------------------------------------------------------------------------
          IF STRLEN(Data_L) > FieldLen_P THEN BEGIN
            CASE "Left/Right_P" OF
              "Left/Right_P"::Right:  EXIT(TruncStrRight(Data_L,FieldLen_P));
              "Left/Right_P"::Left:   EXIT(TruncStrLeft(Data_L, FieldLen_P));
            END;
          END;

      // Length lower
      //----------------------------------------------------------------------------------------------------------------------------------
          IF STRLEN(Data_L) < FieldLen_P THEN BEGIN
            CASE "Left/Right_P" OF
              "Left/Right_P"::Right: EXIT(PadStrRight(Data_L, FieldLen_P, PadChar_P));
              "Left/Right_P"::Left:  EXIT(PadStrLeft(Data_L, FieldLen_P, PadChar_P));
            END;
          END;
        END;

      // No character end of text
      //----------------------------------------------------------------------------------------------------------------------------------
        ELSE BEGIN

      // same lenght
      //----------------------------------------------------------------------------------------------------------------------------------
          IF STRLEN(Data_L) = FieldLen_P THEN EXIT (Data_L+FORMAT(EndOfText_P));

      // Length higher
      //----------------------------------------------------------------------------------------------------------------------------------
          IF STRLEN(Data_L) > FieldLen_P THEN BEGIN
            CASE "Left/Right_P" OF
              "Left/Right_P"::Right: EXIT (TruncStrRight(Data_L, FieldLen_P)+FORMAT(EndOfText_P));
              "Left/Right_P"::Left:  EXIT (TruncStrRight(Data_L, FieldLen_P)+FORMAT(EndOfText_P));
            END;
          END;

      // Length lower
      //----------------------------------------------------------------------------------------------------------------------------------
          IF STRLEN(Data_L) < FieldLen_P THEN BEGIN
            CASE "Left/Right_P" OF
              "Left/Right_P"::Right: EXIT (PadStrRight(Data_L, FieldLen_P, PadChar_P)+FORMAT(EndOfText_P));
              "Left/Right_P"::Left:  EXIT (PadStrLeft(Data_L, FieldLen_P, PadChar_P)+FORMAT(EndOfText_P));
            END;
          END;
        END;
      END;
    END;
