    PROCEDURE ImportPrixTexte@1000000001(FileName_P@1000000000 : Text[256]) Result_L : Text[256];

      "<-- 3LI LOCALS -->"@1000000001 : Integer;
      TableSasImport_L@1000000008 : Record 8087780;
      FileLogName_L@1000000003 : Text[256];
      FileImport_L@1000000004 : File;
      FileInStream_L@1000000005 : InStream;
      NbCarLu_L@1000000006 : Integer;
      TexteLu_L@1000000007 : Text[1024];
      Pos_L@1000000010 : Integer;
      Texte_L@1000000011 : Text[256];
      DateText_L@1000000012 : Text[10];
      EnregOK_L@1000000013 : Boolean;
    BEGIN
      //----------------------------------------------------------------------------------------------------------------------------------
      // Importation des prix de Revient RADAR : Copie vers SAS Import prix revient (8087780)
      //----------------------------------------------------------------------------------------------------------------------------------
      // Entr'es:
      //   - FileName_P : Nom du fichier PC . int'grer
      // Sorties:
      //   - Result_L   : Erreur Retourn'e, Vide si Ok
      //----------------------------------------------------------------------------------------------------------------------------------
      IF ((GetRadarParameters()) AND (TempRadarParameter_G."Import Cost Price")) THEN BEGIN
      //IF ((TableParameter_L.FINDFIRST) AND (TableParameter_L."Import Cost Price")) THEN BEGIN
        ToolsFile_G.LogInitFileExecute(FileName_P);
        IF NOT EXISTS(FileName_P) THEN BEGIN
          ToolsFile_G.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0002, FileName_P), 1, MAXSTRLEN(Result_L)));
        END ELSE BEGIN
          IF NOT FileImport_L.OPEN(FileName_P) THEN BEGIN
            ToolsFile_G.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0003, FileName_P, GETLASTERRORTEXT), 1, MAXSTRLEN(Result_L)));
            CLEARLASTERROR;
          END ELSE BEGIN
            FileImport_L.CREATEINSTREAM(FileInStream_L);
            WHILE NOT (FileInStream_L.EOS) DO BEGIN
              EnregOK_L := TRUE;
              NbCarLu_L := FileInStream_L.READTEXT(TexteLu_L);
              CLEAR(TableSasImport_L);
              Pos_L := 1;
              // N› Article
              TableSasImport_L."Item No" := ToolsFile_G.ExtractTextCsv(TexteLu_L, Pos_L, ';', 8);
              // Date d'but
              Texte_L := ToolsFile_G.ExtractTextCsv(TexteLu_L, Pos_L, ';', 8);
              DateText_L := COPYSTR(Texte_L, 5, 4) + '-' + COPYSTR(Texte_L, 3, 2) + '-' + COPYSTR(Texte_L, 1, 2);
              IF DateText_L = '0000-00-00' THEN BEGIN
                TableSasImport_L."Starting Date" := 0D;
              END ELSE BEGIN
                IF NOT EVALUATE(TableSasImport_L."Starting Date" , DateText_L, 9) THEN BEGIN
                  ToolsFile_G.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0006,
                    TableSasImport_L.FIELDCAPTION(TableSasImport_L."Starting Date"),
                    DateText_L, COPYSTR(TexteLu_L, 1, 12)), 1, MAXSTRLEN(Result_L)));
                END;
              END;
              // Date fin
              Texte_L := ToolsFile_G.ExtractTextCsv(TexteLu_L, Pos_L, ';', 8);
              DateText_L := COPYSTR(Texte_L, 5, 4) + '-' + COPYSTR(Texte_L, 3, 2) + '-' + COPYSTR(Texte_L, 1, 2);
              IF DateText_L = '0000-00-00' THEN BEGIN
                TableSasImport_L."Ending Date" := 0D;
              END ELSE BEGIN
                IF NOT EVALUATE(TableSasImport_L."Ending Date" , DateText_L, 9) THEN BEGIN
                  ToolsFile_G.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0006,
                    TableSasImport_L.FIELDCAPTION(TableSasImport_L."Ending Date"),
                    DateText_L, COPYSTR(TexteLu_L, 1, 12)), 1, MAXSTRLEN(Result_L)));
                END;
              END;
              // Activit'
              TableSasImport_L.Activity := ToolsFile_G.ExtractTextCsv(TexteLu_L, Pos_L, ';', 10);
              // Prix revient (NB: Vu avec Bertrand , prendre le prix tel quel, pas de test avec l'unit')
              Texte_L := ToolsFile_G.ExtractTextCsv(TexteLu_L, Pos_L, ';', 17);
              Texte_L := CONVERTSTR(Texte_L, '.', ',');
              IF Texte_L = '' THEN BEGIN
                EnregOK_L := FALSE;
              END ELSE BEGIN
                IF NOT EVALUATE(TableSasImport_L."Unit Cost" , Texte_L) THEN BEGIN
                  ToolsFile_G.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0006,
                    TableSasImport_L.FIELDCAPTION(TableSasImport_L."Unit Cost"),
                    Texte_L, COPYSTR(TexteLu_L, 1, 12)), 1, MAXSTRLEN(Result_L)));
                END;
              END;
              // Unit'
              TableSasImport_L."Unit of Measure Code" := ToolsFile_G.ExtractTextCsv(TexteLu_L, Pos_L, ';', 10);
              // Type de prix
              TableSasImport_L."Price Type" := ToolsFile_G.ExtractTextCsv(TexteLu_L, Pos_L, ';', 3);
              TableSasImport_L.HoroCreated := CURRENTDATETIME;
              TableSasImport_L."Integration date" := 0D;
              IF EnregOK_L THEN BEGIN
                IF NOT TableSasImport_L.INSERT(TRUE) THEN BEGIN
                  ToolsFile_G.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0007, FORMAT(TableSasImport_L."Record No"))
                  , 1, MAXSTRLEN(Result_L)));
                END;
              END;
            END;
            FileImport_L.CLOSE();
            TempRadarParameter_G."Last Date Import Cost Price" := CURRENTDATETIME;
            IF NOT TempRadarParameter_G.MODIFY(TRUE) THEN BEGIN
              ToolsFile_G.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0009,
                                                            TempRadarParameter_G.TABLECAPTION,COMPANYNAME),1,MAXSTRLEN(Result_L)));
            END;
            IF (NOT SaveRadarParameters()) THEN BEGIN
              ToolsFile_G.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0009,
                                                            TempRadarParameter_G.TABLECAPTION,COMPANYNAME),1,MAXSTRLEN(Result_L)));
            END;

          END;
        END;

        // Pour g'rer le ROLLBACK.
        FileLogName_L := ToolsFile_G.LogName;
        ToolsFile_G.LogClose();

        IF ToolsFile_G.LogErreurType() = 'E' THEN BEGIN
          ERROR('ERREUR ' + FileLogName_L);
        END ELSE BEGIN
          IF NOT ToolsFile_G.FileRenameSav(FileName_P, 'Sav', TRUE) THEN BEGIN
            ToolsFile_G.LogWriteErreur(STRSUBSTNO(Text0004, FileName_P));
          END;
          IF ToolsFile_G.LogErreurType() = 'I' THEN BEGIN
            MoteurInterface_G.InsertLog(CodeUnit_G, FileLogName_L);
          END ELSE BEGIN
            MoteurInterface_G.InsertOk(CodeUnit_G);
          END;
        END;
      END;
    END;

    PROCEDURE ImportPrixRecord@1000000002() result_L : Text[256];
    VAR
      "<-- 3LI LOCALS -->"@1000000000 : Integer;
      TableSasImport_L@1000000006 : Record 8087780;

      FileLogName_L@1000000002 : Text[256];

      Article_L@1000000008 : Code[20];
      Variant_L@1000000009 : Code[10];
      CodMag_L@1000000010 : Code[10];
      Creation_L@1000000004 : Boolean;
      Item@1000000011 : Record 27;
      UnitCostConverted@1000000012 : Decimal;
    BEGIN
      //----------------------------------------------------------------------------------------------------------------------------------
      // Importation des Prix Revie dans Nav : copie du SAS vers Table 8053229 (CMC Item Valuation)
      //----------------------------------------------------------------------------------------------------------------------------------
      // Entr'es:
      //
      // Sorties:
      //   - Result_L               : Erreur Retourn'e, Vide si Ok
      //----------------------------------------------------------------------------------------------------------------------------------
      IF (GetRadarParameters() AND TempRadarParameter_G."Import Cost Price") THEN BEGIN
      //IF ((TableParameter_L.FINDFIRST) AND (TableParameter_L."Import Cost Price")) THEN BEGIN
        TableSasImport_L.SETRANGE(TableSasImport_L."Integration date", 0D);
        IF TableSasImport_L.FINDSET THEN BEGIN
          REPEAT
            TraitOK_L := TRUE;
            Creation_L := FALSE;
            CLEAR(NoRef_L);
            CLEAR(Variant_L);
            CLEAR(CodMag_L);
            CLEAR(TablePrixRevient_L);

            // contr"le correspondance Article
            Article_L := MoteurInterface_G.GetReferentialCorrespondance(ItemTemplateCode_G, TRUE, TableSasImport_L."Item No", FALSE);
            IF Article_L = '' THEN BEGIN
              ToolsFile_G.LogWriteInfo(COPYSTR(STRSUBSTNO(Text0017, TableSasImport_L."Item No", TableSasImport_L."Record No")
              , 1, MAXSTRLEN(result_L)));
              TraitOK_L := FALSE;
            END ELSE BEGIN
              // --> 20181022--INT08-CMC02.02.00-ADVITAM-XPE
              TraitOK_L := CheckItem(Article_L, Item);
              // <-- 20181022--INT08-CMC02.02.00-ADVITAM-XPE
            END;

            IF (TableSasImport_L."Price Type" <> 'PRR') AND (TableSasImport_L."Price Type" <> 'PRP')
              AND (TableSasImport_L."Price Type" <> 'PRE') THEN BEGIN
              ToolsFile_G.LogWriteInfo(COPYSTR(STRSUBSTNO(Text0018, TableSasImport_L."Price Type", TableSasImport_L."Record No")
              , 1, MAXSTRLEN(result_L)));
              TraitOK_L := FALSE;
            END;

            IF TraitOK_L THEN BEGIN
              IF NOT TablePrixRevient_L.GET(TempRadarParameter_G."Inv. Adjmt Reason Code", Article_L, Variant_L, CodMag_L,
              TableSasImport_L."Starting Date", TableSasImport_L."Ending Date", TableSasImport_L.Activity, NoRef_L) THEN BEGIN
                TablePrixRevient_L."Inv. Adjmt Reason Code" := TempRadarParameter_G."Inv. Adjmt Reason Code";
                TablePrixRevient_L."Item No." := Article_L;
                TablePrixRevient_L."Variant Code" := '';
                TablePrixRevient_L."Location Code" := '';
                TablePrixRevient_L."Starting Date" := TableSasImport_L."Starting Date";
                TablePrixRevient_L."Ending Date" := TableSasImport_L."Ending Date";
                TablePrixRevient_L.Type := TableSasImport_L.Activity;
                TablePrixRevient_L."Reference No." := '';
                Creation_L := TRUE;
              END;
              // --> 20181022--INT08-CMC02.02.00-ADVITAM-XPE
              //TablePrixRevient_L.VALIDATE(TablePrixRevient_L."LCY Unit Cost", TableSasImport_L."Unit Cost");
              UnitCostConverted := ConvertUnitCost(Item, TableSasImport_L."Unit Cost");
              TablePrixRevient_L.VALIDATE(TablePrixRevient_L."LCY Unit Cost", UnitCostConverted);
              // <-- 20181022--INT08-CMC02.02.00-ADVITAM-XPE
              CASE TRUE OF
                TableSasImport_L."Price Type" = 'PRR': BEGIN
                  TablePrixRevient_L.VALIDATE(TablePrixRevient_L.Status, TablePrixRevient_L.Status::Actual);
                END;
                TableSasImport_L."Price Type" = 'PRP': BEGIN
                  TablePrixRevient_L.VALIDATE(TablePrixRevient_L.Status, TablePrixRevient_L.Status::Provisional);
                END;
                TableSasImport_L."Price Type" = 'PRE': BEGIN
                  TablePrixRevient_L.VALIDATE(TablePrixRevient_L.Status, TablePrixRevient_L.Status::Commitment);
                END;
              END;

              IF Creation_L THEN BEGIN
                IF NOT TablePrixRevient_L.INSERT(TRUE) THEN BEGIN
                  ToolsFile_G.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0011, TablePrixRevient_L.TABLECAPTION, TableSasImport_L."Record No")
                    , 1, MAXSTRLEN(result_L)));
                  TraitOK_L := FALSE;
                END;
              END ELSE BEGIN
                TablePrixRevient_L."Creation Date" := TODAY;
                TablePrixRevient_L."Creation User" := 'RADAR';
                IF NOT TablePrixRevient_L.MODIFY(TRUE) THEN BEGIN
                  ToolsFile_G.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0012, TablePrixRevient_L.TABLECAPTION, TableSasImport_L."Record No")
                    , 1, MAXSTRLEN(result_L)));
                  TraitOK_L := FALSE;
                END;
              END;
            END;

            IF TraitOK_L THEN BEGIN
              TableSasImport_L."Integration date" := TODAY;
              IF NOT TableSasImport_L.MODIFY(TRUE) THEN BEGIN
                ToolsFile_G.LogWriteErreur(COPYSTR(STRSUBSTNO(Text0005, TableSasImport_L."Record No"), 1, MAXSTRLEN(result_L)));
              END;
            END;
          UNTIL TableSasImport_L.NEXT = 0;
        END;
      END;

      // Pour g'rer le ROLLBACK.
      FileLogName_L := ToolsFile_G.LogName;
      ToolsFile_G.LogClose();

      IF ToolsFile_G.LogErreurType() = 'E' THEN BEGIN
        ERROR('ERREUR ' + FileLogName_L);
      END ELSE BEGIN
        IF ToolsFile_G.LogErreurType() = 'I' THEN BEGIN
          MoteurInterface_G.InsertLog(CodeUnit_G, FileLogName_L);
        END ELSE BEGIN
          MoteurInterface_G.InsertOk(CodeUnit_G);
        END;
      END;

      //<--20140107-Int07-CMC02.02.00-ADVITAM-GMX
    END;

    PROCEDURE ConvertUnitCost@1000000006(Item@1000000000 : Record 27;UnitCost@1000000001 : Decimal) ReturnValue : Decimal;
    VAR
      UOM@1000000002 : Record 204;
      ItemUOM@1000000003 : Record 5404;
    BEGIN
      // --> 20181022--INT08-CMC02.02.00-ADVITAM-XPE
      ReturnValue := UnitCost;
      ItemUOM.SETRANGE("Item No.", Item."No.");
      ItemUOM.SETRANGE(Code, Item."Sales Unit of Measure");
      IF (ItemUOM.FINDFIRST()) THEN BEGIN
        IF (UOM.GET(ItemUOM.Code)) THEN BEGIN
          CASE (UOM.Code) OF
            'T' : BEGIN
              ReturnValue := UnitCost;
            END;
            'KG' : BEGIN
              ReturnValue := (UnitCost / 1000);
            END;
            ELSE BEGIN
              IF (Item."Net Weight" <> 0) THEN BEGIN
                ReturnValue := (UnitCost / Item."Net Weight");
              END;
            END;
          END;
        END;
      END;
      // <-- 20181022--INT08-CMC02.02.00-ADVITAM-XPE
    END;
