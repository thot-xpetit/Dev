    PROCEDURE UpdateTrackingLines@8053003(ItemJnlLine_P@8053005 : Record 83);
    VAR
      ItemJnlTrackingLine_L@8053000 : Record 337;
      ItemJnlTrackingLine2_L@8053001 : Record 337;
      NewManufacturingDate_L@8053002 : Date;
      ItemTrackingMngt_L@8053003 : Codeunit 6500;
      EntriesExists_L@8053004 : Boolean;
    BEGIN
      WITH ItemJnlLine_P DO BEGIN
        ItemJnlTrackingLine_L.SETCURRENTKEY("Source ID","Source Ref. No.","Source Type","Source Subtype","Source Batch Name");
        ItemJnlTrackingLine_L.SETRANGE("Source Type",DATABASE::"Item Journal Line");
        ItemJnlTrackingLine_L.SETRANGE("Source ID","Journal Template Name");
        ItemJnlTrackingLine_L.SETRANGE("Source Batch Name","Journal Batch Name");
        ItemJnlTrackingLine_L.SETRANGE("Source Ref. No.","Line No.");
        ItemJnlTrackingLine_L.SETRANGE("Source Subtype",4);
        ItemJnlTrackingLine_L.SETRANGE("Lot No.",NewCodeWithoutTmpChar(ExistingLotNo));
        IF NOT ItemJnlTrackingLine_L.FINDFIRST THEN BEGIN
          ItemJnlTrackingLine_L.INIT;
          IF ItemJnlTrackingLine2_L.FINDLAST THEN
            ItemJnlTrackingLine_L."Entry No." := ItemJnlTrackingLine2_L."Entry No." + 1
          ELSE
            ItemJnlTrackingLine_L."Entry No." := 1;
          ItemJnlTrackingLine_L.INSERT(TRUE);
          ItemJnlTrackingLine_L."Reservation Status" := ItemJnlTrackingLine_L."Reservation Status"::Prospect;
          ItemJnlTrackingLine_L.VALIDATE("Item No.","Item No.");
          IF "Variant Code" <> '' THEN
            ItemJnlTrackingLine_L.VALIDATE("Variant Code","Variant Code");
          ItemJnlTrackingLine_L.VALIDATE("Location Code","Location Code");
          ItemJnlTrackingLine_L.VALIDATE("Quantity (Base)",-Quantity);
          ItemJnlTrackingLine_L."Creation Date" := TODAY;
          ItemJnlTrackingLine_L."Source Type" := DATABASE::"Item Journal Line";
          ItemJnlTrackingLine_L."Source ID" := "Journal Template Name";
          ItemJnlTrackingLine_L."Source Batch Name" := "Journal Batch Name";
          ItemJnlTrackingLine_L."Source Ref. No." := "Line No.";
          ItemJnlTrackingLine_L."Source Subtype" := 4;
          ItemJnlTrackingLine_L."Shipment Date" := TODAY;
          ItemJnlTrackingLine_L."Created By" := USERID;
          ItemJnlTrackingLine_L."Item Tracking" := ItemJnlTrackingLine_L."Item Tracking"::"Lot No.";
          ItemJnlTrackingLine_L."Lot No." := NewCodeWithoutTmpChar(ExistingLotNo);
          ItemJnlTrackingLine_L."Manufacturing Date" := ItemTrackingMngt_L.ExistingManufactDate(ItemJnlTrackingLine_L."Item No.",
                                                        ItemJnlTrackingLine_L."Variant Code",ItemJnlTrackingLine_L."Lot No.",'',
                                                        FALSE,EntriesExists_L);
          ItemJnlTrackingLine_L.MODIFY(TRUE);
        END;
      END;

      // New Lot No.
      IF NewCodeWithoutTmpChar(NewLotNo) = '' THEN
        ItemJnlTrackingLine_L.VALIDATE("New Lot No.",ItemJnlTrackingLine_L."Lot No.")
      ELSE
        ItemJnlTrackingLine_L.VALIDATE("New Lot No.",NewCodeWithoutTmpChar(NewLotNo));

      // New Manufacturing Date
