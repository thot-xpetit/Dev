    PROCEDURE GetUnitConvert@1000000013(ItemNo_P@1000000003 : Code[20];UnitOfMesure_P@1000000001 : Code[10];IndivBaseQty_P@1000000002 : Decimal) : Text[30];
    VAR
      ItemUnitOfMeasure_L@1000000000 : Record 5404;
      Result_L@1000000005 : Decimal;
    BEGIN
      Result_L := IndivBaseQty_P;
      IF (UnitOfMesure_P <> TxtUnit_L) AND (UnitOfMesure_P <> TxtUnit_Kg) THEN BEGIN
        ItemUnitOfMeasure_L.SETRANGE("Item No.",ItemNo_P);
        ItemUnitOfMeasure_L.SETFILTER(Code,'%1|%2',TxtUnit_Kg,TxtUnit_L);
        IF ItemUnitOfMeasure_L.FINDFIRST THEN BEGIN
          IF ItemUnitOfMeasure_L."Qty. per Unit of Measure" <> 0 THEN
            Result_L := ROUND(IndivBaseQty_P / ItemUnitOfMeasure_L."Qty. per Unit of Measure");
        END;
      END;
      //EXIT(FORMAT(Result_L));
      EXIT(FORMAT(Result_L,0,'<Integer><Decimals><Comma,.>'));
    END;
