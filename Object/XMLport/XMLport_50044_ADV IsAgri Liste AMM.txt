      ItemUnitOfMeasure_L@1000000000 : Record 5404;

      Result_L := IndivBaseQty_P;
      IF (UnitOfMesure_P <> TxtUnit_L) AND (UnitOfMesure_P <> TxtUnit_Kg) THEN BEGIN
        ItemUnitOfMeasure_L.SETRANGE("Item No.",ItemNo_P);
        ItemUnitOfMeasure_L.SETFILTER(Code,'%1|%2',TxtUnit_Kg,TxtUnit_L);

        END;
      END;
      EXIT(FORMAT(Result_L));
    END;
