    PROCEDURE ManageSalesCommentLine@1000000036();
    BEGIN



      IF SalesCommentLine.FINDSET THEN
      REPEAT

        // (R')initialise la ligne
        CLEAR(TableSalesCommentLine);
        TableSalesCommentLine.RESET;
        TableSalesCommentLine.INIT;

        IF
          (SalesHeader."No." <> '')
          AND TableSalesCommentLine.GET(
              SalesHeader."Document Type",
              SalesHeader."No.",
              SalesCommentLine."Document Line No.",
              SalesCommentLine."Line No.") THEN BEGIN
            TableSalesCommentLine.TRANSFERFIELDS(SalesCommentLine);
            TableSalesCommentLine.MODIFY(TRUE);
        END ELSE BEGIN
            TableSalesCommentLine.COPY(SalesCommentLine);
            // Map
            TableSalesCommentLine."Document Type"   := TableSalesHeader."Document Type";
            TableSalesCommentLine."No."             := TableSalesHeader."No.";
            TableSalesCommentLine.INSERT(TRUE);
        END;
      UNTIL SalesCommentLine.NEXT = 0;
    END;

    PROCEDURE ManageSalesLine@1000000037();
    BEGIN

      IF SalesLine.FINDSET THEN
      REPEAT

        // (R')initialise la ligne
        CLEAR(TableSalesLine);
        TableSalesLine.RESET;
        TableSalesLine.INIT;

        // R'cup'ration de l'article
        IF NOT TableItem.GET(SalesLine."No.")  THEN BEGIN
            ERROR('L''article n''existe pas.');
        END;

        // Test et conversion de la quantit' par rapport . l'unit'
        IF (TableItem."Base Unit of Measure" = 'T') AND (SalesLine."Unit of Measure Code" = 'KG') THEN BEGIN
           SalesLine.Quantity := SalesLine.Quantity / 1000;
        END;
        IF (TableItem."Base Unit of Measure" = 'KG') AND (SalesLine."Unit of Measure Code" = 'T') THEN BEGIN
           SalesLine.Quantity := SalesLine.Quantity * 1000;
        END;

        // Si unit' article 'U' et que l'on reÎoit unit' != 'U' => ERROR
        IF (TableItem."Base Unit of Measure" = 'U') AND (SalesLine."Unit of Measure Code" <> 'U') THEN BEGIN
           ERROR('L''unit' de l''article doit ^tre ''U''');
        END;

        // R'cup'ration de la ligne de commande pour mise . jour sinno cr'ation
        IF
          (SalesHeader."No." <> '')
          AND TableSalesLine.GET(SalesHeader."Document Type", SalesHeader."No.", SalesLine."Line No.") THEN BEGIN

            IF DEL = FORMAT(SalesLine."Line No.") THEN BEGIN
              TableSalesLine.DELETE(TRUE);
            END ELSE BEGIN
              //TableSalesLine.TRANSFERFIELDS(SalesLine);

              // Validation
              TableSalesLine.VALIDATE("Line No.",                    SalesLine."Line No.");
              TableSalesLine.VALIDATE(Type,                          SalesLine.Type);
              TableSalesLine.VALIDATE("No.",                         SalesLine."No.");
              TableSalesLine.VALIDATE(Quantity,                      SalesLine.Quantity);
              TableSalesLine.VALIDATE("Time Slot Code",              SalesLine."Time Slot Code");
              TableSalesLine.VALIDATE("Shipment Method Code",        SalesLine."Shipment Method Code");
              TableSalesLine.VALIDATE("Requested Delivery Date",     SalesLine."Requested Delivery Date");
              TableSalesLine.VALIDATE("Latest Req. Deliv. Date",     SalesLine."Latest Req. Deliv. Date");

              TableSalesLine.MODIFY(TRUE);
            END;
        END ELSE IF DEL = '' THEN BEGIN
            TableSalesLine.COPY(SalesLine);

            // Map
            //TableSalesLine."Document Type"   := TableSalesHeader."Document Type";
            //TableSalesLine."Document No."    := TableSalesHeader."No.";
            TableSalesLine.VALIDATE("Document Type",               TableSalesHeader."Document Type");
            TableSalesLine.VALIDATE("Document No.",                TableSalesHeader."No.");

            // Validation
            TableSalesLine.VALIDATE("Line No.",                    SalesLine."Line No.");
            TableSalesLine.VALIDATE(Type,                          SalesLine.Type);
            TableSalesLine.VALIDATE("No.",                         SalesLine."No.");
            TableSalesLine.VALIDATE(Quantity,                      SalesLine.Quantity);
            TableSalesLine.VALIDATE("Time Slot Code",              SalesLine."Time Slot Code");
            TableSalesLine.VALIDATE("Shipment Method Code",        SalesLine."Shipment Method Code");
            TableSalesLine.VALIDATE("Requested Delivery Date",     SalesLine."Requested Delivery Date");
            TableSalesLine.VALIDATE("Latest Req. Deliv. Date",     SalesLine."Latest Req. Deliv. Date");

            TableSalesLine.INSERT(TRUE);
        END;
      UNTIL SalesLine.NEXT = 0;
    END;
