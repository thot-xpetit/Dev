    PROCEDURE ReplaceText@1000000006(string_P@1000000003 : Text[250];findwhat_P@1000000002 : Text[250];ReplaceWith_P@1000000001 : Text[250]) newstring : Text[250];
    BEGIN
      IF (UPPERCASE(findwhat_P) <> UPPERCASE(ReplaceWith_P)) THEN BEGIN
        WHILE STRPOS(UPPERCASE(string_P), UPPERCASE(findwhat_P)) > 0 DO
        string_P:= DELSTR(string_P, STRPOS(UPPERCASE(string_P), UPPERCASE(findwhat_P))) + ReplaceWith_P +
          COPYSTR(string_P,STRPOS(UPPERCASE(string_P), UPPERCASE(findwhat_P)) + STRLEN(findwhat_P));
      END;
      newstring := string_P;
    END;

          (EngineSetup."Folder Import/export" <>
           (ReplaceText(EngineSetup."Folder Import/export",
                               ServerFileProduction,
                               ServerFileRecette) + LocalFolderSuffix)))
        THEN BEGIN
          EngineSetup."Folder Import/export":=ReplaceText(EngineSetup."Folder Import/export",
                                                          ServerFileProduction,
                                                          ServerFileRecette)
                                                        +LocalFolderSuffix;
          Compteur += 1;
          EngineSetup.MODIFY;
      END;

      IF ((EngineSetup."Folder Log Error" <> '' ) AND
          (EngineSetup."Folder Log Error" <>
          (ReplaceText(EngineSetup."Folder Log Error",
                       ServerFileProduction,
                       ServerFileRecette) + LocalFolderSuffix)))
          THEN BEGIN
        EngineSetup."Folder Log Error":=ReplaceText(EngineSetup."Folder Log Error",
                                                    ServerFileProduction,
                                                    ServerFileRecette)
                                        +LocalFolderSuffix;
        Compteur += 1;
        EngineSetup.MODIFY;
      END;
      IF ((EngineSetup."Network Foler" <> '') AND
          (EngineSetup."Network Foler" <>
          (ReplaceText(EngineSetup."Network Foler",
                       NetworkFolderProd,
                       NetworkFolderRec) + NetworkFolderSuffix))) THEN BEGIN
        IF (STRPOS(EngineSetup."Network Foler", EDIServerProd) <> 0) THEN BEGIN
          EngineSetup."Network Foler" := ReplaceText(EngineSetup."Network Foler", EDIServerProd, EDIServerRec);
          EngineSetup.MODIFY;
        END ELSE BEGIN
          EngineSetup."Network Foler" := ReplaceText(EngineSetup."Network Foler",
                                                     NetworkFolderProd,
                                                     NetworkFolderRec)
                                         + NetworkFolderSuffix;
          IF COPYSTR(EngineSetup."Connector Name",1,4) = 'WMS_' THEN
            EngineSetup."Network Foler" := CMC_Parser_L.ReplaceSubstring(EngineSetup."Network Foler",'Recette','Test');
          IF COPYSTR(EngineSetup."Connector Name",1,6) = 'EXTR_A' THEN BEGIN
            IF STRPOS(UPPERCASE(EngineSetup."Network Foler"),UPPERCASE('sinfo15')) <> 0 THEN
              EngineSetup."Network Foler" := INSSTR(EngineSetup."Network Foler",'b',
                                                    STRPOS(UPPERCASE(EngineSetup."Network Foler"),UPPERCASE('sinfo15')) + 7);
            IF (STRPOS(EngineSetup."Network Foler", ProductionKettleServer) <> 0) THEN BEGIN
              EngineSetup."Network Foler" := ReplaceText(EngineSetup."Network Foler",
                                                         ProductionKettleServer,
                                                         RecetteKettleServer);
            END;
          END;
          Compteur += 1;
          EngineSetup.MODIFY;
        END;
      END;
      IF ((EngineSetup."Folder FTP" <> '') AND
          (EngineSetup."Folder FTP" <>
          (ReplaceText(EngineSetup."Folder FTP",
                       FTPProduction,
                      FTPRecette) + FTPFolderSuffix))) THEN BEGIN
        EngineSetup."Folder FTP" := ReplaceText(EngineSetup."Folder FTP",
                                                FTPProduction,
                                                FTPRecette)
                                    +FTPFolderSuffix;
        Compteur += 1;
        EngineSetup.MODIFY;
      END;

      // Engine Setp ERESIS FTP / USER /PWD
      // navagri_neuville --> navagri_neuville_pp
      // *Nav113120 --> *Nav113120pp
      IF (EngineSetup."User FTP" = 'navagri_neuville') THEN BEGIN
        EngineSetup."User FTP" := 'navagri_neuville_pp';
        Compteur += 1;
        EngineSetup.MODIFY;
      END;

      IF (EngineSetup."Password FTP" = '*Nav113120') THEN BEGIN
        EngineSetup."Password FTP" := '*Nav113120pp';
        Compteur += 1;
        EngineSetup.MODIFY;
      END;


      IF (Email <> '') THEN BEGIN
        IF EngineSetup."Mail Notification Error"<>'' THEN BEGIN
          EngineSetup."Mail Notification Error" := Email;
          Compteur +=1 ;
          EngineSetup.MODIFY;
        END;
        IF EngineSetup."Mail Notification Information"<>'' THEN BEGIN
          EngineSetup."Mail Notification Information" := Email;
          Compteur += 1;
          EngineSetup.MODIFY;
        END;
      END;

      IF (ClearNodeExecution) THEN BEGIN
        EngineSetup."Node of execution" := '';
      END;
      IF (NOT UserReplica) THEN BEGIN
        EngineSetup."Use Replica Database" := FALSE;
      END;
      EngineSetup.MODIFY;
    END;

    BEGIN
    END.
  }
