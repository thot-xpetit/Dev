    PROCEDURE GetRecFilters@2();
    BEGIN
      IF GETFILTERS <> '' THEN BEGIN
        IF GETFILTER("Sales Type") <> '' THEN
          SalesTypeFilter_G := "Sales Type"
        ELSE
          SalesTypeFilter_G := SalesTypeFilter_G::None;

        IF GETFILTER(Type) <> '' THEN
          ItemTypeFilter_G := Type
        ELSE
          ItemTypeFilter_G := ItemTypeFilter_G::None;

        SalesCodeFilter_G := GETFILTER("Sales Code");
        CodeFilter_G := GETFILTER(Code);
        CurrencyCodeFilter_G := GETFILTER("Currency Code");
        EVALUATE(StartingDateFilter_G,GETFILTER("Starting Date"));
      END;
    END;

    PROCEDURE SetRecFilters@1();
    BEGIN
      CurrForm.SalesCodeFilterCtrl.ENABLED(TRUE);
      CurrForm.CodeFilterCtrl.ENABLED(TRUE);

      IF SalesTypeFilter_G <> SalesTypeFilter_G::None THEN
        SETRANGE("Sales Type",SalesTypeFilter_G)
      ELSE
        SETRANGE("Sales Type");

      IF SalesTypeFilter_G IN [SalesTypeFilter_G::"All Customers",SalesTypeFilter_G::None] THEN BEGIN
        CurrForm.SalesCodeFilterCtrl.ENABLED(FALSE);
        SalesCodeFilter_G := '';
      END;

      IF SalesCodeFilter_G <> '' THEN
        SETFILTER("Sales Code",SalesCodeFilter_G)
      ELSE
        SETRANGE("Sales Code");

      IF ItemTypeFilter_G <> ItemTypeFilter_G::None THEN
        SETRANGE(Type,ItemTypeFilter_G)
      ELSE
        SETRANGE(Type);

      IF ItemTypeFilter_G = ItemTypeFilter_G::None THEN BEGIN
        CurrForm.CodeFilterCtrl.ENABLED(FALSE);
        CodeFilter_G := '';
      END;

      IF CodeFilter_G <> '' THEN BEGIN
        SETFILTER(Code,CodeFilter_G);
      END ELSE
        SETRANGE(Code);

      IF CurrencyCodeFilter_G <> '' THEN BEGIN
        SETFILTER("Currency Code",CurrencyCodeFilter_G);
      END ELSE
        SETRANGE("Currency Code");

      IF StartingDateFilter_G <> '' THEN
        SETFILTER("Starting Date",StartingDateFilter_G)
      ELSE
        SETRANGE("Starting Date");

      IF SalesLevelDiscountFilter_G <> SalesLevelDiscountFilter_G::" " THEN BEGIN
        CASE SalesLevelDiscountFilter_G OF
           SalesLevelDiscountFilter_G::"1" : SETRANGE("Discount Level","Discount Level"::"1");
           SalesLevelDiscountFilter_G::"2" : SETRANGE("Discount Level","Discount Level"::"2");
           SalesLevelDiscountFilter_G::"3" : SETRANGE("Discount Level","Discount Level"::"3");
           SalesLevelDiscountFilter_G::"4" : SETRANGE("Discount Level","Discount Level"::"4");
           SalesLevelDiscountFilter_G::"5" : SETRANGE("Discount Level","Discount Level"::"5");
        END;
      END ELSE
        SETRANGE("Discount Level");

      CurrForm.UPDATE(FALSE);
    END;
