    PROCEDURE Load@1();
    VAR
      CustDocDistribEntry_L@60003 : Record 60005;
      ShptLine_L@60004 : Record 111;
      Columntxt@60002 : Text[60];
      y_L@60000 : Integer;
      c_L@60001 : Integer;
    BEGIN
      CLEAR(MatrixColumnCaptions);

      //set header columns
      c_L:=0;
      IF CustDocDistrib_G.FINDSET THEN BEGIN
        //IF (CustDocDistrib_G."Total %"<>100) AND
        //   (PageModeOpt=PageModeOpt::"%") THEN
        //  ERROR(Text012);
        REPEAT
          CustDocDistrib_G.CALCFIELDS("Distributed Customer Name");
          IF ShowCustName_G THEN
            Columntxt := CustDocDistrib_G."Distributed Customer Name"
          ELSE
            Columntxt := CustDocDistrib_G."Distributed Customer No.";
          c_L+=1;
          IF (PageModeOpt=PageModeOpt::Qty) THEN
            MatrixColumnCaptions[c_L]:=STRSUBSTNO(Text006,Columntxt)
          ELSE
            FOR y_L:=1 TO 2 DO
              CASE  y_L OF
                1: IF c_L=1 THEN
                     MatrixColumnCaptions[1]:=STRSUBSTNO(Text005,Columntxt)
                   ELSE
                     MatrixColumnCaptions[c_L*2-1]:=STRSUBSTNO(Text005,Columntxt);
                2: MatrixColumnCaptions[c_L*2]:=STRSUBSTNO(Text006,Columntxt);
              END;
        UNTIL CustDocDistrib_G.NEXT=0;
        c_L+=1;
        IF (PageModeOpt=PageModeOpt::Qty) THEN
          MatrixColumnCaptions[c_L]:=Text008
        ELSE BEGIN
          MatrixColumnCaptions[c_L*2-1]:=Text007;
          MatrixColumnCaptions[c_L*2]:=Text008;
        END;
      END;

      //set cells
      ShptLine_L:=Rec;
      IF FINDFIRST THEN
        REPEAT
          c_L:=0;
          IF CustDocDistrib_G.FINDFIRST THEN
            REPEAT
              c_L+=1;
              IF (PageModeOpt=PageModeOpt::Qty) THEN BEGIN
                IF NOT FindEntry(c_L,CustDocDistribEntry_L) THEN BEGIN
                  MatrixData[c_L]:=ROUND(Quantity*CustDocDistrib_G."Distribution %"/100,0.00001);
                  FieldUpdateEntry(c_L);
                END;
              END ELSE
                FOR y_L:=1 TO 2 DO
                  CASE  y_L OF
                    1: IF c_L=1 THEN BEGIN
                         IF NOT FindEntry(c_L,CustDocDistribEntry_L) THEN BEGIN
                           MatrixData[1]:=CustDocDistrib_G."Distribution %";
                           FieldUpdateEntry(1);
                         END;
                       END ELSE
                         IF NOT FindEntry(c_L*2-1,CustDocDistribEntry_L) THEN BEGIN
                           MatrixData[c_L*2-1]:=CustDocDistrib_G."Distribution %";
                           FieldUpdateEntry(c_L*2-1);
                         END;
                    2: IF NOT FindEntry(c_L*2,CustDocDistribEntry_L) THEN BEGIN
                         MatrixData[c_L*2]:=ROUND(Quantity*CustDocDistrib_G."Distribution %"/100,0.00001);
                         FieldUpdateEntry(c_L*2);
                       END;
                  END;
            UNTIL CustDocDistrib_G.NEXT=0;
        UNTIL NEXT=0;

      IF FINDFIRST AND (ShptLine_L."Line No."<>0) THEN
        WHILE "Line No." <> ShptLine_L."Line No." DO
          NEXT;
    END;
