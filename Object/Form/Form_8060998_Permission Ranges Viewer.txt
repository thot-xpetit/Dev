    LOCAL PROCEDURE ReadPermissionRanges@1();
    VAR
      "<< Locals >>"@1000000000 : Integer;
      AllObj@1000 : Record 2000000038;
      iObjectType@1001 : Integer;
      Total@1002 : Integer;
    BEGIN
      PermissionRangeView.RESET;
      PermissionRangeView.DELETEALL;

      Wnd.OPEN(TextWnd01 + TextWnd02 + TextWnd03);
      IF ObjectType <> ObjectType::All THEN
        PermissionRangesByType(ObjectType - 1)
      ELSE
        FOR iObjectType := ObjectType::TableData TO ObjectType::CodeUnit DO
          PermissionRangesByType(iObjectType - 1);
      Wnd.CLOSE();
    END;

    LOCAL PROCEDURE PermissionRangesByType@2(ObjectType@1000 : Integer);
    VAR
      "<< Locals >>"@1000000000 : Integer;
      Object@1001 : Record 2000000001;
      PermissionRange@1002 : Record 2000000044;
      NextObjectNo@1003 : Integer;
    BEGIN
      IF ObjectType = PermissionRange."Object Type"::TableData THEN BEGIN
        PermissionRanges4Data();
        EXIT;
      END;
      PermissionRange.RESET;
      PermissionRange.SETRANGE("Object Type",ObjectType);

      Object.SETRANGE(Type,ObjectType);
      IF PermissionRange.FIND('-') THEN BEGIN
        PermissionRangeView."Object Type" := PermissionRange."Object Type";
        PermissionRangeView.Index := 0;
        PermissionRangeView.INIT;
        Wnd.UPDATE(1,PermissionRange."Object Type");
        REPEAT
          Object.SETRANGE(ID,PermissionRangeView."To" + 1,PermissionRange.From - 1);
          IF Object.FIND('-') THEN BEGIN
            BuildNewRecord(Object.ID);
            WHILE Object.NEXT(1) <> 0 DO BEGIN
              IF Details OR ((PermissionRangeView."To" + 1) <> Object.ID) THEN
                BuildNewRecord(Object.ID)
              ELSE BEGIN
                PermissionRangeView."To" := Object.ID;
                PermissionRangeView.MODIFY;
              END;
            END;
          END;
          Object.SETRANGE(ID,PermissionRange.From,PermissionRange."To");
          PermissionRangeView.TRANSFERFIELDS(PermissionRange,FALSE);
          IF NOT Object.FIND('-') THEN BEGIN
            PermissionRangeView.Index += 1;
            PermissionRangeView.INSERT;
          END ELSE BEGIN
            NextObjectNo := PermissionRange.From;
            REPEAT
              IF NextObjectNo <> Object.ID THEN BEGIN
                PermissionRangeView.From := NextObjectNo;
                PermissionRangeView."To" := Object.ID - 1;
                PermissionRangeView.Index += 1;
                PermissionRangeView.INSERT;
              END;
              PermissionRangeView.From := Object.ID;
              PermissionRangeView."To" := Object.ID;
              PermissionRangeView.Index += 1;
              PermissionRangeView.INSERT;
              NextObjectNo := Object.ID + 1;
              Wnd.UPDATE(2,Object.ID);
            UNTIL Object.NEXT(1) = 0;
            IF NextObjectNo <= PermissionRange."To" THEN BEGIN
              PermissionRangeView.From := NextObjectNo;
              PermissionRangeView."To" := PermissionRange."To";
              PermissionRangeView.Index += 1;
              PermissionRangeView.INSERT;
            END;
          END;
        UNTIL PermissionRange.NEXT(1) = 0;
      END;
    END;

    LOCAL PROCEDURE PermissionRanges4Data@3();
    VAR
      "<< Locals >>"@1000000000 : Integer;
      Object@1000 : Record 2000000001;
      PermissionRange@1001 : Record 2000000044;
      NextObjectNo@1002 : Integer;
    BEGIN
      PermissionRange.RESET;
      PermissionRange.SETRANGE("Object Type",PermissionRange."Object Type"::TableData);

      LicensePermission.RESET;
      LicensePermission.SETRANGE("Object Type",LicensePermission."Object Type"::TableData);
      Object.SETRANGE(Type,Object.Type::Table);
      IF PermissionRange.FIND('-') THEN BEGIN
        PermissionRangeView."Object Type" := PermissionRange."Object Type";
        PermissionRangeView.Index := 0;
        PermissionRangeView.INIT;
        Wnd.UPDATE(1,PermissionRange."Object Type");
        REPEAT
          Object.SETRANGE(ID,PermissionRangeView."To" + 1,PermissionRange.From - 1);
          IF Object.FIND('-') THEN BEGIN
            LicensePermission.GET(LicensePermission."Object Type"::TableData,Object.ID);
            BuildNewRecord(Object.ID);
            WHILE Object.NEXT(1) <> 0 DO BEGIN
              LicensePermission.GET(LicensePermission."Object Type"::TableData,Object.ID);
              IF Details OR ((PermissionRangeView."To" + 1) <> Object.ID)
              OR (PermissionRangeView."Read Permission" <> LicensePermission."Read Permission")
              OR (PermissionRangeView."Insert Permission" <> LicensePermission."Insert Permission")
              OR (PermissionRangeView."Modify Permission" <> LicensePermission."Modify Permission")
              OR (PermissionRangeView."Delete Permission" <> LicensePermission."Delete Permission")
              OR (PermissionRangeView."Execute Permission" <> LicensePermission."Execute Permission") THEN
                BuildNewRecord(Object.ID)
              ELSE BEGIN
                PermissionRangeView."To" := Object.ID;
                PermissionRangeView.MODIFY;
              END;
            END;
          END;
          Object.SETRANGE(ID,PermissionRange.From,PermissionRange."To");
          PermissionRangeView.TRANSFERFIELDS(PermissionRange,FALSE);
          IF NOT Object.FIND('-') THEN BEGIN
            PermissionRangeView.Index += 1;
            PermissionRangeView.INSERT;
          END ELSE BEGIN
            NextObjectNo := PermissionRange.From;
            REPEAT
              IF NextObjectNo <> Object.ID THEN BEGIN
                PermissionRangeView.From := NextObjectNo;
                PermissionRangeView."To" := Object.ID - 1;
                PermissionRangeView.Index += 1;
                PermissionRangeView.INSERT;
              END;
              PermissionRangeView.From := Object.ID;
              PermissionRangeView."To" := Object.ID;
              PermissionRangeView.Index += 1;
              PermissionRangeView.INSERT;
              NextObjectNo := Object.ID + 1;
              Wnd.UPDATE(2,Object.ID);
            UNTIL Object.NEXT(1) = 0;
            IF NextObjectNo <= PermissionRange."To" THEN BEGIN
              PermissionRangeView.From := NextObjectNo;
              PermissionRangeView."To" := PermissionRange."To";
              PermissionRangeView.Index += 1;
              PermissionRangeView.INSERT;
            END;
          END;
        UNTIL PermissionRange.NEXT(1) = 0;
      END;
    END;

    LOCAL PROCEDURE GetLicenseInformation@5();
    VAR
      "<< Locals >>"@1000000001 : Integer;
      LicenseInformation@1000 : Record 2000000040;
      Delimiter@1001 : Integer;
      "<< 3LI Locals >>"@1000000002 : Integer;
      i_L@1000000000 : Integer;
    BEGIN
      // --> 3LI - 20050701 - DPU01
      // --> Avant
      {
      WITH LicenseInformation DO
        IF FIND('-') THEN BEGIN
          REPEAT
            IF COPYSTR(Text,1,1) = '*' THEN
              IF LicenseInfo <> '' THEN
                FIND('+')
              ELSE
                LicenseInfo := ' '
            ELSE IF LicenseInfo <> '' THEN BEGIN
              Delimiter := STRPOS(Text,':');
              IF Delimiter <> 0 THEN
                Text := COPYSTR(Text,Delimiter + 1);
              Text := DELCHR(Text,'<>',' ');
              LicenseInfo :=
                COPYSTR(STRSUBSTNO('%1 %2',LicenseInfo,DELCHR(Text,'<>',' ')),1,MAXSTRLEN(LicenseInfo));
            END;
          UNTIL NEXT(1) = 0;
          LicenseInfo := DELCHR(LicenseInfo,'<>',' ');
        END;
      }
      // --> AprSs
      WITH LicenseInformation DO
        IF FIND('-') THEN BEGIN
          REPEAT
            i_L += 1;
            IF i_L > 3 THEN BEGIN
              Delimiter := STRPOS(Texte,':');
              IF Delimiter <> 0 THEN
                Texte := COPYSTR(Texte,Delimiter + 1);
              Texte := DELCHR(Texte,'<>',' ');
              LicenseInfo_G[i_L - 3] := COPYSTR(DELCHR(Texte,'<>',' '),1,MAXSTRLEN(LicenseInfo_G[i_L - 3]));
            END;
          UNTIL (NEXT = 0) OR (i_L = 10);
        END;
      // --< 3LI - 20050701 - DPU01
    END;

    LOCAL PROCEDURE BuildNewRecord@6(ObjectID@1000 : Integer);
    BEGIN
      PermissionRangeView.INIT;
      PermissionRangeView.From := ObjectID;
      PermissionRangeView."To" := ObjectID;
      IF PermissionRangeView."Object Type" = PermissionRangeView."Object Type"::TableData THEN BEGIN
        PermissionRangeView."Read Permission" := LicensePermission."Read Permission";
        PermissionRangeView."Insert Permission" := LicensePermission."Insert Permission";
        PermissionRangeView."Modify Permission" := LicensePermission."Modify Permission";
        PermissionRangeView."Delete Permission" := LicensePermission."Delete Permission";
        PermissionRangeView."Execute Permission" := LicensePermission."Execute Permission";
      END;
      PermissionRangeView.Index += 1;
      PermissionRangeView.INSERT;
    END;
