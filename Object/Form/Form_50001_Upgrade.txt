    BEGIN
      IF NOT GET(COMPANYNAME) THEN
        INIT;
      I := 1;
      Phase.RESET;

      IF Phase.FIND('-') THEN

      UNTIL Phase.NEXT = 0;

        IF Libell'[I] <> '' THEN BEGIN

          Font_G[I]    := TRUE;

          Visible_G[I] := FALSE;
