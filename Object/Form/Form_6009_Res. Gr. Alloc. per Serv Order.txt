    VAR
      PeriodFormMgt@1000 : Codeunit 359;
      PeriodType@1001 : 'Day,Week,Month,Quarter,Year,Accounting Period';
      QtyType@1002 : 'Net Change,Balance at Date';

    LOCAL PROCEDURE SetDateFilter@1();
    BEGIN
      IF QtyType = QtyType::"Net Change" THEN
        IF CurrForm.Matrix.MatrixRec."D'but p'riode" = CurrForm.Matrix.MatrixRec."Fin p'riode" THEN
          SETRANGE("Date Filter",CurrForm.Matrix.MatrixRec."D'but p'riode")
        ELSE
          SETRANGE("Date Filter",CurrForm.Matrix.MatrixRec."D'but p'riode",CurrForm.Matrix.MatrixRec."Fin p'riode")
      ELSE
        SETRANGE("Date Filter",0D,CurrForm.Matrix.MatrixRec."Fin p'riode");
    END;

    BEGIN
