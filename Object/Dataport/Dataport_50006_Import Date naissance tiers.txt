OBJECT Dataport 50006 Import Date naissance tiers
{
  OBJECT-PROPERTIES
  {
    Date=10/05/16;
    Heure=15:22:05;
    Modifi'=Oui;
    Liste versions=;
  }
  PROPERTIES
  {
    FieldStartDelimiter=<">;
    FieldEndDelimiter=<">;
    FieldSeparator=[;];
  }
  DATAITEMS
  {
    { PROPERTIES
      {
        DataItemTable=Table2000000026;
        AutoSave=Non;
        AutoUpdate=Non;
        AutoReplace=Non;
        DataItemTableView=SORTING(Num'ro);
        OnPreDataItem=BEGIN
                        CLEAR(Cpt_Advitam_G);
                        CLEAR(Cpt_Uneal_G);
                        CLEAR(Cpt_Total_G);
                      END;

        OnAfterImportRecord=VAR
                              ImportOK_L@1000000000 : Boolean;
                              Closed_L@1000000001 : Boolean;
                            BEGIN
                              Cpt_Total_G += 1;
                              //V'rification de la soci't'
                              ImportOK_L:=FALSE;
                              CASE Code_Societe_G OF
                                'UNEAL' :    BEGIN
                                             IF COMPANYNAME='UNEAL' THEN ImportOK_L:=TRUE;
                                             END;
                                'GROUPE ADVITAM' :    BEGIN
                                             IF COMPANYNAME='GROUPE ADVITAM' THEN ImportOK_L:=TRUE;
                                             END;
                              END;

                              IF ImportOK_L =FALSE THEN CurrDataport.SKIP;

                              IF COMPANYNAME = 'GROUPE ADVITAM' THEN BEGIN
                                IF Tiers_MS_G.GET(Code_Tiers_G) THEN BEGIN
                                  IF Tiers_MS_G."Moral Or Natural Person" = Tiers_MS_G."Moral Or Natural Person"::Natural THEN BEGIN
                                    Closed_L := FALSE;
                                    IF ((Tiers_MS_G."Birth Date" <> 0D) AND (Tiers_MS_G."Birth Date" <> Date_Naissance_G)) OR
                                       ((Tiers_MS_G."INSEE Birth City Code" <> '') AND (Tiers_MS_G."INSEE Birth City Code" <> Lieu_Naissance_G) AND
                                       (Lieu_Naissance_G <> '00000')) THEN BEGIN
                                         MESSAGE('%1 : %2 - %3 , %4 - %5', Code_Tiers_G, Tiers_MS_G."Birth Date", Date_Naissance_G ,
                                            Tiers_MS_G."INSEE Birth City Code", Lieu_Naissance_G);
                                         CurrDataport.SKIP;
                                    END;
                                    IF Tiers_MS_G.Status = Tiers_MS_G.Status::Closed THEN BEGIN
                                      Closed_L := TRUE;
                                      Tiers_MS_G.Status := Tiers_MS_G.Status::Open;
                                    END;
                                    Tiers_MS_G.VALIDATE("Birth Date",Date_Naissance_G);
                                    IF Lieu_Naissance_G <> '00000' THEN BEGIN
                                      CodPost_G.SETCURRENTKEY(CodPost_G."INSEE No.");
                                      CodPost_G.SETRANGE(CodPost_G."INSEE No.", Lieu_Naissance_G);
                                      IF CodPost_G.COUNT = 0 THEN BEGIN
                                        MESSAGE('Code insee commune inexistant : %1 , Tiers : %2', Lieu_Naissance_G, Code_Tiers_G);
                                        CurrDataport.SKIP;
                                      END;
                                      IF CodPost_G.COUNT > 1 THEN BEGIN
                                        IF CodPost_G.FINDFIRST THEN BEGIN
                                          Tiers_MS_G."INSEE Birth City Code" := Lieu_Naissance_G;
                                          Tiers_MS_G."INSEE Birth City Description" := CodPost_G.City;
                                        END;
                                      END ELSE BEGIN
                                        Tiers_MS_G.VALIDATE("INSEE Birth City Code",Lieu_Naissance_G);
                                      END;
                                    END;
                                    IF Tiers_MS_G."INSEE Birth City Code" = '' THEN BEGIN
                                      CLEAR(Tiers_MS_G."INSEE Birth City Description");
                                    END;
                                    Tiers_MS_G.MODIFY(TRUE);
                                    IF Closed_L THEN BEGIN
                                      Tiers_MS_G.Status := Tiers_MS_G.Status::Closed;
                                      Tiers_MS_G.MODIFY();
                                    END;
                                    Cpt_Advitam_G += 1;
                                  END;
                                END ELSE BEGIN
                                  MESSAGE('Tiers inexistant : %1', Code_Tiers_G);
                                END;
                              END;

                              IF COMPANYNAME = 'UNEAL' THEN BEGIN
                                IF Contact_G.GET(Code_Tiers_G) THEN BEGIN
                                  IF Contact_G."Moral Or Natural Person" = Contact_G."Moral Or Natural Person"::Natural THEN BEGIN
                                    IF ((Contact_G."Birth Date" <> 0D) AND (Contact_G."Birth Date" <> Date_Naissance_G)) OR
                                       ((Contact_G."INSEE Birth City Code" <> '') AND (Contact_G."INSEE Birth City Code" <> Lieu_Naissance_G) AND
                                       (Lieu_Naissance_G <> '00000')) THEN BEGIN
                                         MESSAGE('%1 : %2 - %3 , %4 - %5', Code_Tiers_G, Contact_G."Birth Date", Date_Naissance_G ,
                                            Contact_G."INSEE Birth City Code", Lieu_Naissance_G);
                                         CurrDataport.SKIP;
                                    END;
                                    Contact_G.VALIDATE("Birth Date",Date_Naissance_G);
                                    IF Lieu_Naissance_G <> '00000' THEN BEGIN
                                      CodPost_G.SETCURRENTKEY(CodPost_G."INSEE No.");
                                      CodPost_G.SETRANGE(CodPost_G."INSEE No.", Lieu_Naissance_G);
                                      IF CodPost_G.COUNT = 0 THEN BEGIN
                                        MESSAGE('Code insee commune inexistant : %1 , Tiers : %2', Lieu_Naissance_G, Code_Tiers_G);
                                        CurrDataport.SKIP;
                                      END;
                                      IF CodPost_G.COUNT > 1 THEN BEGIN
                                        IF CodPost_G.FINDFIRST THEN BEGIN
                                          Contact_G."INSEE Birth City Code" := Lieu_Naissance_G;
                                          Contact_G."INSEE Birth City Description" := CodPost_G.City;
                                        END;
                                      END ELSE BEGIN
                                        Contact_G.VALIDATE("INSEE Birth City Code",Lieu_Naissance_G);
                                      END;
                                    END;
                                    IF Contact_G."INSEE Birth City Code" = '' THEN BEGIN
                                      CLEAR(Contact_G."INSEE Birth City Description");
                                    END;
                                    Contact_G.MODIFY(TRUE);
                                    Cpt_Uneal_G += 1;
                                  END;
                                END ELSE BEGIN
                                  MESSAGE('Tiers inexistant : %1', Code_Tiers_G);
                                END;
                              END;
                              CLEAR(Lieu_Naissance_G);
                              CLEAR(CodPost_G);
                            END;

        OnPostDataItem=BEGIN
                         MESSAGE('Enreg lus : %1', Cpt_Total_G);
                         MESSAGE('GR ADVITAM trait's : %1', Cpt_Advitam_G);
                         MESSAGE('UNEAL trait's : %1', Cpt_Uneal_G);
                       END;

      }
      FIELDS
      {
        {      ;     ;Code_Societe_G       }
        {      ;     ;Code_Tiers_G         }
        {      ;     ;Date_Naissance_G     }
        {      ;     ;Lieu_Naissance_G     }
      }
       }
  }

    PROPERTIES

      Width=9020;
      Height=3410;
    }
    CONTROLS
    {
    }
  }
  CODE
  {
    VAR
      Code_Societe_G@1000000000 : Text[20];
      Code_Tiers_G@1000000001 : Text[20];
      Date_Naissance_G@1000000002 : Date;
      Lieu_Naissance_G@1000000003 : Text[10];
      Tiers_MS_G@1000000004 : Record 8053185;
      Contact_G@1000000005 : Record 5050;
      Cpt_Advitam_G@1000000006 : Integer;
      Cpt_Uneal_G@1000000007 : Integer;
      Cpt_Total_G@1000000008 : Integer;
      CodPost_G@1000000009 : Record 225;

    BEGIN
    END.
