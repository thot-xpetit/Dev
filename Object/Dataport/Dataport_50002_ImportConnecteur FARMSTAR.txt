    PROCEDURE GetContactCode@1000000006(VAR ContactCode@1000000000 : Text[30]) ReturnValue : Boolean;
    BEGIN
      ReturnValue := FALSE;
      ContactCode := '';
      //The CodeTiers Must'nt be empty or null
      IF (CodeTiers <> '') THEN BEGIN
        // Is Technician or not (Column Value ZAM/ ZONE 15 : Technician = TC)
        IF UPPERCASE(Zone15) = 'TC' THEN BEGIN
          // Get the correspondance Code for Contact Code
          IF CMB_Interface_Correspondence_G.GET (CorrespondanceCode, IdTable, IdField, CodeTiers) THEN BEGIN
            ContactCode :=CMB_Interface_Correspondence_G."Interface Value";
            IF (SearchCtcFronOption) THEN BEGIN
              ContactCode := GetContactFromOption(ContactCode, 'EXTRANET', 'GENERALITES');
            END;
          END ELSE BEGIN
            ErrorMsg := STRSUBSTNO(TextErrSalesPerson, CodeTiers);
          END;
          ReturnValue := (ContactCode <> '');
          IF (NOT ReturnValue) THEN BEGIN
            ErrorMsg := TextErrEmptyCodeTiers;
          END;
          // <-- 20160310-DataPort01-ADV01.01.00_ADVITAM-KRE-002
        END ELSE BEGIN
          // Agri
          ContactCode := CodeTiers;
          ReturnValue := TRUE;
        END;
      END ELSE BEGIN
        ErrorMsg := TextErrEmptyCodeTiers;
      END;
    END;
