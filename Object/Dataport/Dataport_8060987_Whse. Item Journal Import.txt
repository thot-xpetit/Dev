    PROCEDURE SetTracking@1120000();
    VAR
      WhseItemTraking_L@1120000 : Record 6550;
      NextEntryNo_L@1120001 : Integer;
      SerialNoInfo_L@60001 : Record 6504;
      LotNoInfo_L@60002 : Record 6505;
      DummyDate_L@60003 : Date;
    BEGIN
      Item.GET(WhseItemJnlLine."Item No.");
      IF ItemTrackingCode_G.GET(Item."Item Tracking Code") THEN BEGIN

        WhseItemTraking_L.RESET;
        IF WhseItemTraking_L.FINDLAST THEN
          NextEntryNo_L := WhseItemTraking_L."Entry No." + 1
        ELSE
          NextEntryNo_L := 1;

        WhseItemTraking_L.INIT;
        WhseItemTraking_L."Entry No." := NextEntryNo_L;
        WhseItemTraking_L.VALIDATE("Item No.",WhseItemJnlLine."Item No.");
        WhseItemTraking_L.VALIDATE("Variant Code",WhseItemJnlLine."Variant Code");
        WhseItemTraking_L.VALIDATE("Quantity (Base)",WhseItemJnlLine.Quantity);
        WhseItemTraking_L."Source Type" := 7311;
        WhseItemTraking_L."Source Subtype" := 0;
        WhseItemTraking_L."Source Batch Name" := WhseItemJnlLine."Journal Template Name";
        WhseItemTraking_L."Source ID" := WhseItemJnlLine."Journal Batch Name";
        WhseItemTraking_L.VALIDATE("Source Ref. No.",WhseItemJnlLine."Line No.");
        WhseItemTraking_L.VALIDATE("Source Prod. Order Line",0);
        WhseItemTraking_L."Lot No." := InputText[15];
        WhseItemTraking_L."Serial No." := InputText[14];
        WhseItemTraking_L."Location Code" := WhseItemJnlLine."Location Code";
        WhseItemTraking_L.VALIDATE("Qty. to Handle (Base)",WhseItemJnlLine.Quantity);


        IF InputText[16] <> '' THEN BEGIN
          FormatFieldsMgt.Text2Date(DummyDate_L,InputText[16]);
          WhseItemTraking_L."Expiration Date" := DummyDate_L;
        END;

        WhseItemTraking_L.INSERT(TRUE);

        IF (WhseItemTraking_L."Serial No." <> '') AND
           (ItemTrackingCode_G."SN Info. Inbound Must Exist") THEN BEGIN
          SerialNoInfo_L.INIT;
          SerialNoInfo_L."Item No." := WhseItemTraking_L."Item No.";
          SerialNoInfo_L."Variant Code" := WhseItemTraking_L."Variant Code";
          SerialNoInfo_L."Serial No." := WhseItemTraking_L."Serial No.";
          SerialNoInfo_L.Description := WhseItemTraking_L."Serial No.";
          IF SerialNoInfo_L.INSERT THEN;
        END;

        IF (WhseItemTraking_L."Lot No." <> '') AND
           (ItemTrackingCode_G."Lot Info. Inbound Must Exist") THEN BEGIN
          LotNoInfo_L.INIT;
          LotNoInfo_L."Item No." := WhseItemTraking_L."Item No.";
          LotNoInfo_L."Variant Code" := WhseItemTraking_L."Variant Code";
          LotNoInfo_L."Lot No." := WhseItemTraking_L."Lot No.";
          LotNoInfo_L.Description := WhseItemTraking_L."Lot No.";
          IF LotNoInfo_L.INSERT THEN;
        END;

      END;
    END;
