    PROCEDURE CreatContactPers@1000000039(Prenom_P@1000000000 : Text[30];Nom_P@1000000001 : Text[30];CodeContactSoc_P@1000000002 : Code[20]) Res_CodeCont : Code[20];
    VAR
      Contact_L@1000000003 : Record 5050;
    BEGIN
      Contact_L.INIT;
      Contact_L."No.":=STRSUBSTNO('%1_%2',CodeContactSoc_P,1);
      Contact_L.Type:=Contact_L.Type::Person;
      Contact_L.VALIDATE("Company No.",CodeContactSoc_P);
      IF Contact_L.INSERT THEN;

      Contact_L.Name:=STRSUBSTNO('%1 %2',Prenom_P,Nom_P);
      Contact_L."Search Name":=Contact_L.Name;
      Contact_L."First Name":=Prenom_P;
      Contact_L.Surname:=Nom_P;

      Contact_L.MODIFY;

      Res_CodeCont:=Contact_L."No.";
    END;

      RecTiersMS_L@8053002 : Record 8053185;
