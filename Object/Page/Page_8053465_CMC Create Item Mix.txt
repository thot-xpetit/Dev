    PROCEDURE FillPossibleRootItemMixes@8053004(CurrentSaleslLine_L@8053001 : Record 37);
    VAR
      ItemMixRootItem_L@8053000 : Record 8053188;
      Item_L@8053002 : Record 27;
    BEGIN
      //------------------------------------------------------------------------------------------------
      // The function checks if the item from the current sales order line is a root item for an item mix and stores it in a
      // temporary table.The search is done in an hierarchy: item, species,sub-family,family,activity.
      // Input:
      //   - Sales Line rec
      // Output:
      //   -
      //------------------------------------------------------------------------------------------------


      //Item Level
      ItemMixRootItem_L.RESET;
      ItemMixRootItem_L.SETRANGE(Type,ItemMixRootItem_L.Type::Item);
      ItemMixRootItem_L.SETRANGE("No.",CurrentSaleslLine_L."No.");

      IF ItemMixRootItem_L.FINDSET THEN BEGIN
        REPEAT
          IF NOT TempItemMix_G.GET(ItemMixRootItem_L."Item Mix Code") THEN BEGIN
            TempItemMix_G.INIT;
            TempItemMix_G.VALIDATE(Code,ItemMixRootItem_L."Item Mix Code");
            TempItemMix_G.INSERT(TRUE);
          END;
        UNTIL ItemMixRootItem_L.NEXT =0;
        EXIT;
      END;

      Item_L.GET(CurrentSaleslLine_L."No.");

      //Species Level
      ItemMixRootItem_L.RESET;
      ItemMixRootItem_L.SETRANGE(Type,ItemMixRootItem_L.Type::Species);
      ItemMixRootItem_L.SETRANGE("No.",Item_L."Species Code");

      IF ItemMixRootItem_L.FINDSET THEN BEGIN
        REPEAT
          IF NOT TempItemMix_G.GET(ItemMixRootItem_L."Item Mix Code") THEN BEGIN
            TempItemMix_G.INIT;
            TempItemMix_G.VALIDATE(Code,ItemMixRootItem_L."Item Mix Code");
            TempItemMix_G.INSERT(TRUE);
          END;
        UNTIL ItemMixRootItem_L.NEXT =0;
        EXIT;
      END;

      //Sub-family Level
      ItemMixRootItem_L.RESET;
      ItemMixRootItem_L.SETRANGE(Type,ItemMixRootItem_L.Type::"Sub-family");
      ItemMixRootItem_L.SETRANGE("No.",Item_L."Sub-Family Code");

      IF ItemMixRootItem_L.FINDSET THEN BEGIN
        REPEAT
          IF NOT TempItemMix_G.GET(ItemMixRootItem_L."Item Mix Code") THEN BEGIN
            TempItemMix_G.INIT;
            TempItemMix_G.VALIDATE(Code,ItemMixRootItem_L."Item Mix Code");
            TempItemMix_G.INSERT(TRUE);
          END;
        UNTIL ItemMixRootItem_L.NEXT =0;
        EXIT;
      END;

      //Family Level
      ItemMixRootItem_L.RESET;
      ItemMixRootItem_L.SETRANGE(Type,ItemMixRootItem_L.Type::Family);
      ItemMixRootItem_L.SETRANGE("No.",Item_L."Family Code");

      IF ItemMixRootItem_L.FINDSET THEN BEGIN
        REPEAT
          IF NOT TempItemMix_G.GET(ItemMixRootItem_L."Item Mix Code") THEN BEGIN
            TempItemMix_G.INIT;
            TempItemMix_G.VALIDATE(Code,ItemMixRootItem_L."Item Mix Code");
            TempItemMix_G.INSERT(TRUE);
          END;
        UNTIL ItemMixRootItem_L.NEXT =0;
        EXIT;
      END;

      //Activity Level
      ItemMixRootItem_L.RESET;
      ItemMixRootItem_L.SETRANGE(Type,ItemMixRootItem_L.Type::Activity);
      ItemMixRootItem_L.SETRANGE("No.",Item_L."Activity Code");

      IF ItemMixRootItem_L.FINDSET THEN BEGIN
        REPEAT
          IF NOT TempItemMix_G.GET(ItemMixRootItem_L."Item Mix Code") THEN BEGIN
            TempItemMix_G.INIT;
            TempItemMix_G.VALIDATE(Code,ItemMixRootItem_L."Item Mix Code");
            TempItemMix_G.INSERT(TRUE);
          END;
        UNTIL ItemMixRootItem_L.NEXT =0;
        EXIT;
      END;
    END;

    PROCEDURE CalculateQuantities@8053011();
    BEGIN
      //------------------------------------------------------------------------------------------------
      // Calculates quantities for selected item mix and additives
      // Input:  -
      // Output: -
      //------------------------------------------------------------------------------------------------


      BaseQty_G := SalesLine_G.Quantity;
      RootItemQty_G := BaseQty_G;
      TotalQty_G  := BaseQty_G;
      // --> 20130610-Nut04-CMC02.02.00-FEC12
      UnitofMeasureCode_G := SalesLine_G."Unit of Measure Code";
      // <-- 20130610-Nut04-CMC02.02.00-FEC12

      // --> 20150610-NUT04-CMC02.02.00-3056-FEC01
      {BEFORE
      CalcRoundedRootItemQty;
      }
      // <-- 20150610-NUT04-CMC02.02.00-3056-FEC01

      TempPageMirrorTable_G.RESET;
      TempPageMirrorTable_G.SETRANGE(Selected,TRUE);

      IF TempPageMirrorTable_G.FINDSET THEN
        REPEAT
           IF TempPageMirrorTable_G.Type = TempPageMirrorTable_G.Type::Incorporation THEN
             RootItemQty_G := RootItemQty_G - TempPageMirrorTable_G."Quantity (base)";
           IF TempPageMirrorTable_G.Type = TempPageMirrorTable_G.Type::Addition THEN
             TotalQty_G := TotalQty_G + TempPageMirrorTable_G."Quantity (base)";
        UNTIL TempPageMirrorTable_G.NEXT = 0;

      // --> 20150610-NUT04-CMC02.02.00-3056-FEC02
      CalcRoundedRootItemQty;
      // <-- 20150610-NUT04-CMC02.02.00-3056-FEC02
    END;

    PROCEDURE CheckRootItemQtyIsValid@8053038(RootItemQty@8053000 : Decimal);
    VAR
      ItemMixRootItem_L@8053002 : Record 8053188;
      ItemMix_L@8053003 : Record 8053187;
      Item_L@8053001 : Record 27;
    BEGIN
      //------------------------------------------------------------------------------------------------
      // Checks if the root quantity is not smaller than a minumum quantity parametrized in
      // tables CMC Item MixRoot Item or CMC Item Mix
      // Input:  - root quantity
      // Output: -
      //------------------------------------------------------------------------------------------------

      ItemMix_L.RESET;
      ItemMix_L.GET("Item Mix Code");

      ItemMixRootItem_L.RESET;


      // --> 20130603-Nut04-CMC02.02.00-BSI01
      //Before
      {
      IF NOT ItemMixRootItem_L.GET("Item Mix Code",ItemMixRootItem_L.Type::Item) THEN BEGIN
        IF NOT ItemMixRootItem_L.GET("Item Mix Code",ItemMixRootItem_L.Type::Species) THEN
          IF NOT ItemMixRootItem_L.GET("Item Mix Code",ItemMixRootItem_L.Type::"Sub-family") THEN
            IF NOT ItemMixRootItem_L.GET("Item Mix Code",ItemMixRootItem_L.Type::Family) THEN
              IF NOT ItemMixRootItem_L.GET("Item Mix Code",ItemMixRootItem_L.Type::Activity) THEN BEGIN
                IF ItemMix_L."Mini. Root Item Rate" <>0 THEN
                  IF (RootItemQty < (ItemMix_L."Mini. Root Item Rate" * BaseQty_G / 100)) THEN  BEGIN
                    ERROR(Text002);
                  END;
                EXIT;
              END;
      END;
      }
      //After
      Item_L.GET(SalesLine_G."No.");

      IF NOT ItemMixRootItem_L.GET("Item Mix Code",ItemMixRootItem_L.Type::Item,Item_L."No.") THEN BEGIN
        IF NOT ItemMixRootItem_L.GET("Item Mix Code",ItemMixRootItem_L.Type::Species,Item_L."Species Code") THEN
          IF NOT ItemMixRootItem_L.GET("Item Mix Code",ItemMixRootItem_L.Type::"Sub-family",Item_L."Sub-Family Code") THEN
            IF NOT ItemMixRootItem_L.GET("Item Mix Code",ItemMixRootItem_L.Type::Family,Item_L."Family Code") THEN
              IF NOT ItemMixRootItem_L.GET("Item Mix Code",ItemMixRootItem_L.Type::Activity,Item_L."Activity Code") THEN BEGIN
                IF ItemMix_L."Mini. Root Item Rate" <>0 THEN
                  IF (RootItemQty < (ItemMix_L."Mini. Root Item Rate" * BaseQty_G / 100)) THEN  BEGIN
                    ERROR(Text002);
                  END;
                EXIT;
              END;
      END;
      // <-- 20130603-Nut04-CMC02.02.00-BSI01

      IF ItemMixRootItem_L."Mini. Root Item Rate" <> 0  THEN BEGIN
        IF (RootItemQty < (ItemMixRootItem_L."Mini. Root Item Rate" * BaseQty_G / 100)) THEN BEGIN
          ERROR(Text002);
        END;
      END ELSE BEGIN
        IF ItemMix_L."Mini. Root Item Rate" <>0 THEN
          IF (RootItemQty < (ItemMix_L."Mini. Root Item Rate" * BaseQty_G / 100)) THEN BEGIN
            ERROR(Text002);
          END;
      END;
    END;

    PROCEDURE InsertItemMixSalesLines@8053022();
    VAR
      DocItemMix_L@8053000 : Record 8053190;
      SalesLines_L@8053001 : Record 37;
      NoOfAdditives_L@8053002 : Integer;
      ToSalesLine_L@8053003 : Record 37;
      LineSpacing_L@8053004 : Integer;
      NextLineNo_L@8053005 : Integer;
    BEGIN
      //------------------------------------------------------------------------------------------------
      // Updates the original sales line with information regarding item mix and inserts secondary sales lines for each additive
      // Input:  -
      // Output: -
      //------------------------------------------------------------------------------------------------

      SalesLines_L.RESET;

      // --> 20181010-WI00-ADV01.01.00-35769-MRU01
      DocItemMix_L.RESET;
      DocItemMix_L.SETRANGE("Document Type",SalesLine_G."Document Type");
      DocItemMix_L.SETRANGE("Document No.",SalesLine_G."Document No.");
      DocItemMix_L.SETRANGE("Document Line No.",SalesLine_G."Line No.");
      // <-- 20181010-WI00-ADV01.01.00-35769-MRU01

      NoOfAdditives_L := DocItemMix_L.COUNT;

      ToSalesLine_L.RESET;
      ToSalesLine_L.SETRANGE("Document Type","Document Type");
      ToSalesLine_L.SETRANGE("Document No.","Document No.");
      ToSalesLine_L := SalesLine_G;

      IF ToSalesLine_L.FIND('>') THEN BEGIN
        LineSpacing_L := (ToSalesLine_L."Line No." - SalesLine_G."Line No.") DIV (1 + NoOfAdditives_L);
        IF LineSpacing_L = 0 THEN
          ERROR(Text004);
      END ELSE
        LineSpacing_L := 10000;

      NextLineNo_L := SalesLine_G."Line No.";

      DocItemMix_L.RESET;
      DocItemMix_L.SETRANGE("Document Type",SalesLine_G."Document Type");
      DocItemMix_L.SETRANGE("Document No.",SalesLine_G."Document No.");
      DocItemMix_L.SETRANGE("Document Line No.",SalesLine_G."Line No.");

      // --> 20130524-Nut04-CMC02.02.00-BSI01
      //Before
      {
      IF DocItemMix_L.FINDSET THEN BEGIN
          REPEAT
            NextLineNo_L := NextLineNo_L + LineSpacing_L;
            SalesLines_L.INIT;
            SalesLines_L.VALIDATE("Document Type",DocItemMix_L."Document Type");
            SalesLines_L.VALIDATE("Document No.",DocItemMix_L."Document No.");
            SalesLines_L.VALIDATE("Line No.",NextLineNo_L);
            SalesLines_L.INSERT(TRUE);
            SalesLines_L."No." := DocItemMix_L."Item No.";
            SalesLines_L.Type := SalesLines_L.Type::Item;
            SalesLines_L.Quantity := DocItemMix_L.Quantity;
            SalesLines_L."Location Code" := SalesLine_G."Location Code";
            SalesLines_L."Item Mix Root Line No." := SalesLine_G."Line No.";
            SalesLines_L."No." := DocItemMix_L."Item No.";
            SalesLines_L."Item Mix Code" := DocItemMix_L."Item Mix Code";
            SalesLines_L.MODIFY(FALSE);
          UNTIL DocItemMix_L.NEXT = 0;

        SalesLine_G.GET(SalesLine_G."Document Type", SalesLine_G."Document No.", SalesLine_G."Line No.");

        SalesLine_G."Item Mix Base Quantity" := SalesLine_G.Quantity;
        SalesLine_G.Quantity := RootItemQty_G;
        SalesLine_G."Item Mix Root Line No." := SalesLine_G."Line No.";
        SalesLine_G."Item Mix Code" := DocItemMix_L."Item Mix Code";
        SalesLine_G.MODIFY(FALSE);
      END;
      }
      //After
      IF DocItemMix_L.FINDSET THEN BEGIN
          REPEAT
            NextLineNo_L := NextLineNo_L + LineSpacing_L;
            SalesLines_L.INIT;
            SalesLines_L.VALIDATE("Document Type",DocItemMix_L."Document Type");
            SalesLines_L.VALIDATE("Document No.",DocItemMix_L."Document No.");
            SalesLines_L.VALIDATE("Line No.",NextLineNo_L);
            SalesLines_L.INSERT(TRUE);
            //-->20160608-Wi00-ADV01.01.00-3063-VMR01
            //BEFORE
            {

    PROCEDURE CopyExistingItemMix@8053001(SalesLine@8053000 : Record 37);
    VAR
      DocItemMix_L@8053001 : Record 8053190;
    BEGIN
      //------------------------------------------------------------------------------------------------
      // When the page is open, copies previous created item mixes to the temporary table of the page and also to the mirror table.
      // Input:  - sales line rec from the original sales order line
      // Output: -
      //------------------------------------------------------------------------------------------------

      DocItemMix_L.RESET;
      DocItemMix_L.SETRANGE("Document Type",SalesLine."Document Type");

      IF DocItemMix_L.FINDSET THEN BEGIN
        TempItemMix_G.INIT;
        TempItemMix_G.VALIDATE(Code,DocItemMix_L."Item Mix Code");
        TempItemMix_G.INSERT(FALSE);
        REPEAT
          Rec := DocItemMix_L;
          // --> 20140411-Nut04-CMC02.02.00-1974-FEC03
          CalcRoundedBaseQty;
          CalcRoundedQty;
          // --< 20140411-Nut04-CMC02.02.00-1974-FEC03

          TempPageMirrorTable_G := DocItemMix_L;

          TempPageMirrorTable_G.Quantity := Quantity;
          // --< 20140411-Nut04-CMC02.02.00-1974-FEC04
          TempPageMirrorTable_G.INSERT(FALSE);
        UNTIL DocItemMix_L.NEXT = 0;
      END;
    END;

    PROCEDURE ConvertUnitofMeasure@8053006() : Decimal;
    VAR
      CMCUnitsMeasure_L@1000000000 : Record 8053025;
      CMCUnitsMeasure2_L@1000000001 : Record 8053025;
      ItemUnitofMeasure_L@1000000002 : Record 5404;
      ItemUnitofMeasure2_L@1000000004 : Record 5404;
      Rate_L@1000000003 : Decimal;
    BEGIN
      IF UnitofMeasureCode_G <> "Unit of Measure Code" THEN BEGIN

        Rate_L := 0;

        CMCUnitsMeasure_L.RESET;
        CMCUnitsMeasure_L.SETRANGE(Code,UnitofMeasureCode_G);
        CMCUnitsMeasure_L.FINDFIRST;

        CMCUnitsMeasure2_L.RESET;
        CMCUnitsMeasure2_L.SETRANGE(Code, "Unit of Measure Code");
        IF CMCUnitsMeasure2_L.FINDFIRST THEN
          Rate_L := CMCUnitsMeasure2_L."Qty. per Unit of Measure" / CMCUnitsMeasure_L."Qty. per Unit of Measure"
        ELSE BEGIN
          ItemUnitofMeasure_L.RESET;
          ItemUnitofMeasure_L.SETRANGE("Item No.","Item No.");
          ItemUnitofMeasure_L.SETRANGE(Code,UnitofMeasureCode_G);
          IF NOT ItemUnitofMeasure_L.FINDFIRST THEN
            ERROR(Text005,UnitofMeasureCode_G,"Unit of Measure Code")
          ELSE BEGIN
            ItemUnitofMeasure2_L.RESET;
            ItemUnitofMeasure2_L.SETRANGE("Item No.","Item No.");
            ItemUnitofMeasure2_L.SETRANGE(Code,"Unit of Measure Code");
            ItemUnitofMeasure2_L.FINDFIRST;
            Rate_L := ItemUnitofMeasure2_L."Qty. per Unit of Measure" / ItemUnitofMeasure_L."Qty. per Unit of Measure";
          END;
        END;
        EXIT(Rate_L);
      END ELSE
        EXIT(1);
    END;
