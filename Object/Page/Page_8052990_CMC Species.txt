    PROCEDURE GetSelectionFilter@2() : Code[80];
    VAR
      Species@1000 : Record 8052988;
      FirstSpecies@1001 : Code[30];
      LastSpecies@1002 : Code[30];
      SelectionFilter@1003 : Code[250];
      SpeciesCount@1004 : Integer;
      More@1005 : Boolean;
    BEGIN
      CurrPage.SETSELECTIONFILTER(Species);
      SpeciesCount := Species.COUNT;
      IF SpeciesCount > 0 THEN BEGIN
        Species.FIND('-');
        WHILE SpeciesCount > 0 DO BEGIN
          SpeciesCount := SpeciesCount - 1;
          Species.MARKEDONLY(FALSE);
          FirstSpecies := Species.Code;
          LastSpecies := FirstSpecies;
          More := (SpeciesCount > 0);
          WHILE More DO
            IF Species.NEXT = 0 THEN
              More := FALSE
            ELSE
              IF NOT Species.MARK THEN
                More := FALSE
              ELSE BEGIN
                LastSpecies := Species.Code;
                SpeciesCount := SpeciesCount - 1;
                IF SpeciesCount = 0 THEN
                  More := FALSE;
              END;
          IF SelectionFilter <> '' THEN
            SelectionFilter := SelectionFilter + '|';
          IF FirstSpecies = LastSpecies THEN
            SelectionFilter := SelectionFilter + FirstSpecies
          ELSE
            SelectionFilter := SelectionFilter + FirstSpecies + '..' + LastSpecies;
          IF SpeciesCount > 0 THEN BEGIN
            Species.MARKEDONLY(TRUE);
            Species.NEXT;
          END;
        END;
      END;
      EXIT(SelectionFilter);
    END;

      CMCFamily_L@8053002 : Record 8052986;
      CMCSubFamily_L@8053003 : Record 8052987;

    BEGIN

      IF SubFamily_G <> '' THEN BEGIN
        CMCSubFamily_L.RESET;

        IF CMCSubFamily_L.FINDSET THEN

          CMCFamily_L.RESET;
          CMCFamily_L.SETFILTER(CMCFamily_L.Code,Family_G);
          IF CMCFamily_L.FINDSET THEN
            REPEAT
              CMCSubFamily_L.RESET;
              CMCSubFamily_L.SETFILTER(CMCSubFamily_L."Family Code",CMCFamily_L.Code);
              IF CMCSubFamily_L.FINDSET THEN
                REPEAT
                  IF CMCSubFamily_L.Code = CMcSpecies_L."Sub-Family Code" THEN BEGIN
                    IF NOT CMCSpeciesTMP_G.FINDFIRST THEN
                      EXIT(TRUE)
                    ELSE IF CMCSpeciesTMP_G.GET(SpeciesCode_P)THEN
                      EXIT(TRUE);
                  END;
                UNTIL CMCSubFamily_L.NEXT =0;
            UNTIL CMCFamily_L.NEXT =0;
