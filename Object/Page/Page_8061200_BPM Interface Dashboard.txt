    PROCEDURE CalculateProgression@8060982() RES_Result : Decimal;
    VAR
      IncomingMessage_L@8060985 : Record 8061188;
      TotalMessage_L@8060984 : Integer;
      TotalNewMessage_L@8060983 : Integer;
      TotalPendingMessage_L@8060982 : Integer;
    BEGIN
      RES_Result := 0;

      IncomingMessage_L.RESET;
      IncomingMessage_L.SETCURRENTKEY("Interface ID");
      IncomingMessage_L.SETRANGE("Interface ID", "Interface ID");
      TotalMessage_L := IncomingMessage_L.COUNT;

      IncomingMessage_L.SETRANGE(Status, IncomingMessage_L.Status::New);
      TotalNewMessage_L := IncomingMessage_L.COUNT;

      IncomingMessage_L.SETRANGE(Status, IncomingMessage_L.Status::Pending);
      TotalPendingMessage_L := IncomingMessage_L.COUNT;

      IF TotalPendingMessage_L = 0 THEN BEGIN
        RES_Result := 0;
      END ELSE BEGIN
        IF TotalNewMessage_L = 0 THEN BEGIN
          RES_Result := 10000
        END ELSE IF TotalMessage_L > 0 THEN BEGIN
          RES_Result := ROUND((1 - (TotalNewMessage_L / (TotalMessage_L -
                              (TotalMessage_L - TotalNewMessage_L - TotalPendingMessage_L))))*10000, 1, '>');
        END;
      END;

      EXIT(RES_Result);
    END;
