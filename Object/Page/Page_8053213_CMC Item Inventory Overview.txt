        UnitsMeasureCMC2_L.RESET;

      }

      UnitsMeasureCMC_L.RESET;
      UnitsMeasureCMC_L.SETRANGE(Code, Item_P."Base Unit of Measure");

      ItemOfUnit_L.SETRANGE(Code, ICPECategory_G."Unit of Measure Code");
