    PROCEDURE CheckUncheckLine@1180250004() : Boolean;

      IsInserted_L@1180250002 : Boolean;
    BEGIN
      TempPaymentLine_G.INIT;
      TempPaymentLine_G := Rec;
      IF NOT TempPaymentLine_G.INSERT THEN BEGIN

    END;
