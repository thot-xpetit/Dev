        CASE "Ledger type" OF
         "Ledger type"::Vendor:

           L_Rec_VendorLedger.GET("Original Ledger Entry No.");

              ("Remaining Amount"<>L_Rec_VendorLedger."Remaining Amount") OR
              (L_Rec_VendorLedger."Applies-to ID"<>"Applies-to ID") THEN
            EXIT(FALSE);
          END;
         "Ledger type"::Customer:
          BEGIN
           L_Rec_CustomerLedger.GET("Original Ledger Entry No.");
           L_Rec_CustomerLedger.CALCFIELDS(Amount,"Remaining Amount");
           IF (L_Rec_CustomerLedger.Open<>Open) OR
              (L_Rec_CustomerLedger."Remaining Amount"<>"Remaining Amount") OR
              (L_Rec_CustomerLedger."Applies-to ID"<>"Applies-to ID") THEN BEGIN
             EXIT(FALSE);
             END;
          END;
         "Ledger type"::"Debt Recognition":
          BEGIN
           L_Rec_Debt.GET("Document No.");
           L_Rec_Debt.CALCFIELDS(L_Rec_Debt."Amount reimbursed");
           IF (L_Rec_Debt."Species Code"<>"Species Code") OR
              ((L_Rec_Debt."Amount to payback"-L_Rec_Debt."Amount reimbursed")<>"Remaining Amount") THEN
             EXIT(FALSE);
          END;
        END;
       UNTIL NEXT=0;

      EXIT(TRUE);
    END;
