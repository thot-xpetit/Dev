        ELSE

      // --> 20171025-WI00-ADV01.01.00-3290-OLE
      MATIFSpeciesLink_L.SETRANGE(Status,MATIFSpeciesLink_L.Status::Active);
      // <-- 20171025-WI00-ADV01.01.00-3290-OLE
      IF MATIFSpeciesLink_L.FINDSET( FALSE , FALSE ) THEN
      BEGIN
        REPEAT

          // Insertion des 'ch'ances MATIF
          MATIFTerms_L.GET(MATIFSpeciesLink_L."MATIF Terms");
          // -- >20171025-WI00-ADV01.01.00-3290-OLE
          {
          // -- > 20160526-WI00-ADV01.01.00-3005-MMI01
          //BEFORE
          //IF MATIFTerms_L."Period Start">=Campaign_L."Starting Date" THEN BEGIN
          //AFTER
          IF (MATIFTerms_L."Period Start" >= Campaign_L."Starting Date") AND
            (MATIFTerms_L."Period End" <= Campaign_L."Ending Date") THEN
          BEGIN
          // <-- 20160526-WI00-ADV01.01.00-3005-MMI01
            Rec := MATIFTerms_L;
            INSERT;

            // Insertion du d'tail de l''ch'ance MATIF

          END;
          }

          Rec := MATIFTerms_L;
          INSERT;

        INIT;
        "MATIF Terms" := 'TOTAL';
        "Period Start":=01019999D;
        INSERT;

      END ELSE
        ERROR(Text8053004);
    END;

    PROCEDURE CalcValuesMatif@1000000001(VAR LedgerMATIFBuffer_P@1000000006 : Record 8053111);
    VAR
      CMCSalesContract_L@1000000000 : Record 8053012;
      NoLotSales_L@1000000001 : Integer;
      NoLotPurch_L@1000000002 : Integer;
      Position_L@1000000003 : ARRAY [5] OF Integer;
      Future_L@1000000004 : ARRAY [5] OF Integer;
      i_l@1000000005 : Integer;
      CMCMatifAcc_L@1000000008 : Record 8053112;
      CMCMatifContract_L@1000000011 : Record 8053065;
    BEGIN
      // --> 20161222-WI00-ADV01.01.00-3186-JCD01
      WITH LedgerMATIFBuffer_P DO BEGIN
        // this function will calculate values
        "Open Position 1" := 0;
        "Open Position 2" := 0;
        "Open Position 3" := 0;
        "Open Position 4" := 0;
        "Open Position 5" := 0;
        "Future AA" := 0;
        "Account Open Position Choice" := "Open Position";
        "Future AA 1" := 0;
        "Future AA 2" := 0;
        "Future AA 3" := 0;
        "Future AA 4" := 0;
        "Future AA 5" := 0;
        "Future Account AA Choice" := "Future AA";

        IF "Contract Type" <> "Contract Type"::Sales THEN
          EXIT;

        IF "Line No Distrib" <> 0 THEN
         EXIT;

        IF CMCSetup_G."MATIF Weight per Lot" = 0 THEN
          CMCSetup_G.GET;

        CLEAR(CMCMatifAcc_L);
        CMCMatifAcc_L.SETCURRENTKEY("Order Column on MATIF Screens");
        CMCMatifAcc_L.SETFILTER("Order Column on MATIF Screens",'<>%1',0);
        IF CompteNo_G <> '' THEN
          CMCMatifAcc_L.SETFILTER("Account No.",CompteNo_G)
        ELSE
          CMCMatifAcc_L.SETRANGE("Account No.");
        IF NOT CMCMatifAcc_L.FINDFIRST THEN
          EXIT;

        FOR i_l := 1 TO 5 DO BEGIN
          Position_L[i_l] := 0;
          Future_L[i_l] := 0;
        END;

        IF CMCSalesContract_L.GET("Contract No.") THEN;
        "Future AA" := CMCSalesContract_L."Lot Number";

        CASE CMCSalesContract_L."Contract Type" OF
          CMCSalesContract_L."Contract Type"::"Sales Bonus": BEGIN
            REPEAT
              NoLotSales_L := 0;
              NoLotPurch_L := 0;

              CLEAR(CMCMatifContract_L);
              CMCMatifContract_L.SETRANGE("Sales Contract Link","Contract No.");
              CMCMatifContract_L.SETFILTER(Way,'%1|%2',CMCMatifContract_L.Way::Sales,CMCMatifContract_L.Way::Purchase);
              CMCMatifContract_L.SETRANGE("Account No.",CMCMatifAcc_L."Account No.");
              IF CMCMatifContract_L.FINDFIRST THEN
                REPEAT
                  IF CMCMatifContract_L.Way = CMCMatifContract_L.Way::Sales THEN
                    NoLotSales_L := NoLotSales_L + CMCMatifContract_L."Lot Number"
                  ELSE
                    NoLotPurch_L := NoLotPurch_L + CMCMatifContract_L."Lot Number";
                UNTIL CMCMatifContract_L.NEXT = 0;
              Position_L[CMCMatifAcc_L."Order Column on MATIF Screens"] := NoLotPurch_L - NoLotSales_L;
              Future_L[CMCMatifAcc_L."Order Column on MATIF Screens"] := NoLotPurch_L - NoLotSales_L;
              IF Position_L[CMCMatifAcc_L."Order Column on MATIF Screens"] < 0 THEN
                Position_L[CMCMatifAcc_L."Order Column on MATIF Screens"] := 0;
              IF Future_L[CMCMatifAcc_L."Order Column on MATIF Screens"] > 0 THEN
                Future_L[CMCMatifAcc_L."Order Column on MATIF Screens"] := 0;
              "Future AA" := "Future AA" - NoLotPurch_L;

            UNTIL CMCMatifAcc_L.NEXT = 0;
            "Future AA" := -"Future AA";
          END;
          CMCSalesContract_L."Contract Type"::"Purch. Bonus": BEGIN
            REPEAT
              NoLotSales_L := 0;
              NoLotPurch_L := 0;

              CLEAR(CMCMatifContract_L);
              CMCMatifContract_L.SETRANGE("Sales Contract Link","Contract No.");
              CMCMatifContract_L.SETFILTER(Way,'%1|%2',CMCMatifContract_L.Way::Sales,CMCMatifContract_L.Way::Purchase);
              CMCMatifContract_L.SETRANGE("Account No.",CMCMatifAcc_L."Account No.");
              IF CMCMatifContract_L.FINDFIRST THEN
                REPEAT
                  IF CMCMatifContract_L.Way = CMCMatifContract_L.Way::Sales THEN
                    NoLotSales_L := NoLotSales_L + CMCMatifContract_L."Lot Number"
                  ELSE
                    NoLotPurch_L := NoLotPurch_L + CMCMatifContract_L."Lot Number";
                UNTIL CMCMatifContract_L.NEXT = 0;
              Position_L[CMCMatifAcc_L."Order Column on MATIF Screens"] := NoLotPurch_L - NoLotSales_L;
              Future_L[CMCMatifAcc_L."Order Column on MATIF Screens"] := NoLotPurch_L - NoLotSales_L;
              IF Position_L[CMCMatifAcc_L."Order Column on MATIF Screens"] > 0 THEN
                Position_L[CMCMatifAcc_L."Order Column on MATIF Screens"] := 0;
              IF Future_L[CMCMatifAcc_L."Order Column on MATIF Screens"] < 0 THEN
                Future_L[CMCMatifAcc_L."Order Column on MATIF Screens"] := 0;
              "Future AA" := "Future AA" - NoLotSales_L;
            UNTIL CMCMatifAcc_L.NEXT = 0;

          END;
        END;
        "Open Position 1" := Position_L[1];
        "Open Position 2" := Position_L[2];
        "Open Position 3" := Position_L[3];
        "Open Position 4" := Position_L[4];
        "Open Position 5" := Position_L[5];
        "Future AA 1" := Future_L[1];
        "Future AA 2" := Future_L[2];
        "Future AA 3" := Future_L[3];
        "Future AA 4" := Future_L[4];
        "Future AA 5" := Future_L[5];

        "Account Open Position Choice" := "Open Position" - "Open Position 1" - "Open Position 2" - "Open Position 3"
                     - "Open Position 4" - "Open Position 5";

        "Future Account AA Choice" := "Future AA" - "Future AA 1" - "Future AA 2" - "Future AA 3" - "Future AA 4" - "Future AA 5";
        MODIFY;
      END;
        // <-- 20161222-WI00-ADV01.01.00-3186-JCD01
    END;
