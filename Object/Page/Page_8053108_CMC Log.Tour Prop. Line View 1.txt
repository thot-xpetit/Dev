        REPEAT

          //----------------------------------------------------------------------------------

          //----------------------------------------------------------------------------------

          //----------------------------------------------------------------------------------
          IF L_CurrDeliveryZone <> L_recOutstandingLineSource_TMP."Delivery Area Code" THEN BEGIN
            L_CurrDeliveryZone := L_recOutstandingLineSource_TMP."Delivery Area Code";

            L_recOutstandingLineSource.INSERT;

            L_SeqNo := L_SeqNo + 1;
          END;

          //----------------------------------------------------------------------------------
          // Insertion d'une ligne de d'tail
          //----------------------------------------------------------------------------------
          CLEAR(L_recOutstandingLineSource);
          L_recOutstandingLineSource.INIT;
          L_recOutstandingLineSource.TRANSFERFIELDS(L_recOutstandingLineSource_TMP, FALSE);
          L_recOutstandingLineSource."Source Document Type" := L_recOutstandingLineSource_TMP."Source Document Type";
          L_recOutstandingLineSource."Source Document No." := L_recOutstandingLineSource_TMP."Source Document No.";
