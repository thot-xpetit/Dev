    VAR
      Campaign_L@8053001 : Record 8052991;

      RecPostedReceiptsLine_L@8053002 : Record 8053104;

        CampaignNo_L:=SalesCollectionContract_G."Campaign No.";

        UNTIL NEXT=0;
      Rec:=RecPostedReceiptsLine_L;

      // --> 20150319-CER16-CMC02.02.00-2815-BSI01

      IF (PreContract_G=FALSE) AND (TestQuantityContract_G=TRUE) THEN
      after}
      IF (PreContract_G=FALSE) THEN
      // <-- 20150319-CER16-CMC02.02.00-2815-BSI01
      // --> 20141204-CER03-TER01.01.00-RMZ01
      // Before
      //OutStandingQuantity_G:=QuantityContract_L-AffectedContractQty_G-QtyNotPosted_G;
      // After
      OutStandingQuantity_G := ROUND(QuantityContract_L * (1 + (CollectionContract_G."Threshold Percentage" / 100)),0.001)
                  -AffectedContractQty_G-QtyNotPosted_G;
      // <-- 20141204-CER03-TER01.01.00-RMZ01

      // --> 20150319-CER16-CMC02.02.00-2815-BSI01
      IF TestQuantityContract_G = FALSE THEN
        IsQuantityToAffect_G := FALSE
      ELSE BEGIN
      // <-- 20150319-CER16-CMC02.02.00-2815-BSI01
        IF (OutStandingQuantity_G = 0) AND (PreContract_G=FALSE) AND (TestQuantityContract_G=TRUE) THEN
          IsQuantityToAffect_G := FALSE
        ELSE BEGIN
          // On calcul la quantit' restante sur contrat avec la tol'rance param'tr'e sur la campagne
          Campaign_L.GET(CampaignNo_L);
          IF (PreContract_G=FALSE) AND (TestQuantityContract_G=TRUE) THEN
            Excess_L := ( ( OutStandingQuantity_G ) / QuantityContract_L )*100 ;
          IF Excess_L < 0 THEN
            IsQuantityToAffect_G := FALSE;
        END;
      END;

      IF TestQuantityContract_G=FALSE THEN
      IsQuantityToAffect_G := TRUE
    END;
