    PROCEDURE MarkSelection@8053000();
    VAR
      ACPLedgerEntry_L@8053001 : Record 60122;
      MarkValue_L@8053000 : Boolean;
    BEGIN
      //TotalACPNetUnitPrice_G := 0;

      CurrPage.SETSELECTIONFILTER(ACPLedgerEntry_L);

      IF ACPLedgerEntry_L.FIND('-') THEN BEGIN
        // On complSte le marquage si dans la s'lection il y a une ligne non marqu'e
        MarkValue_L := TRUE;
        GET(ACPLedgerEntry_L."Entry No.");
        MarkValue_L := MarkValue_L AND MARK;
        WHILE ((ACPLedgerEntry_L.NEXT <> 0) AND (MarkValue_L)) DO BEGIN
          GET(ACPLedgerEntry_L."Entry No.");

          REPEAT
            IF GET(ACPLedgerEntry_L."Entry No.") THEN

          UNTIL ACPLedgerEntry_L.NEXT = 0;
        END;

      END ELSE
        MARK(NOT MARK);

    END;
