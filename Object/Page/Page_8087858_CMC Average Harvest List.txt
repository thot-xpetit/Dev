                      RefVal_R += ABS(RecIndiceBonisRef_L."Add Unit Price")*RecPostedReceiptLine_L."Brut Quantity";

                  IF RefactionType_P = RefactionType_G::Averaged THEN

            END
              ELSE BEGIN
                IF RefactionType_P = RefactionType_P::Invoiced THEN

                  ELSE
                    RefVal_R += ABS(RecIndiceBonisRef_L."Add Unit Price")*RecPostedReceiptLine_L."Brut Quantity";
                  // <-- 20160629-WI00-ADV01.01.00-3037-SDE
                IF RefactionType_P = RefactionType_G::Averaged THEN
                  RefVal_R := ABS(RecIndiceBonisRef_L."Add Unit Price");
          END;
          // <-- 20150403-CER41/001-CMC02.02.00-2707-BSI01
        END;
      UNTIL RecPostedReceiptLine_L.NEXT = 0;
      // <-- 20140610-CER41-CMC02.02.00-2132-BSI01

      // --> 20160629-WI00-ADV01.01.00-3037-SDE
      IF (RefactionType_P = RefactionType_P::Invoiced) AND (HarvestQty_L <> 0) THEN
        RefVal_R := RefVal_R/HarvestQty_L;
      // <-- 20160629-WI00-ADV01.01.00-3037-SDE
    END;
