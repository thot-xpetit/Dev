    LOCAL PROCEDURE LoadCandidates@15000320();
    BEGIN
      // Same combinations as automatic reconciliation, using public searchReceiptLine.

        SicapaInvoiceHdr."Vendor No.",
        SicapaInvoiceLine."Source Document No.",
        SicapaInvoiceLine."Source Document Line No.",
        SicapaInvoiceLine."Quantite facture",
        SicapaInvoiceLine."Coefficient Unit");

        SicapaInvoiceLine."Coefficient Unit");

      AddSearchResult(
        '',
        SicapaInvoiceLine."Source Document No.",
        SicapaInvoiceLine."Source Document Line No.",
        SicapaInvoiceLine."Quantite facture",
        SicapaInvoiceLine."Coefficient Unit");

      AddSearchResult(
        '',
        SicapaInvoiceLine."Source Document No.",
        SicapaInvoiceLine."Source Document Line No.",
        0,
        1);

      AddSearchResult(
        SicapaInvoiceHdr."Vendor No.",
        '',
        0,
        SicapaInvoiceLine."Quantite facture",
        SicapaInvoiceLine."Coefficient Unit");

      AddSearchResult(
        '',
        '',
        0,
        SicapaInvoiceLine."Quantite facture",
        SicapaInvoiceLine."Coefficient Unit");
    END;

    LOCAL PROCEDURE AddSearchResult@15000330(VendorNo@15000331 : Code[20];OrderNo@15000332 : Code[20];OrderLineNo@15000333 : Integer;QtyToReceive@15000334 : Decimal;QtyFactor@15000335 : Decimal);
    VAR
      PurchRcptLine@15002000 : Record 121;
      MultiLine@15002001 : Boolean;
    BEGIN
      IF SicapaMgt.searchReceiptLine(
        SicapaInvoiceHdr,
        SicapaInvoiceLine,
        VendorNo,
        OrderNo,
        OrderLineNo,
        SicapaInvoiceLine."Item No. NavAgri",
        SicapaInvoiceLine."Variant Code NavAgri",
        SicapaInvoiceLine."Location Code",

        IF PurchRcptLine.FINDSET(FALSE,FALSE) THEN
          REPEAT
            AddCandidate(PurchRcptLine);
          UNTIL PurchRcptLine.NEXT = 0;
      END;
    END;

      PurchRcptLine@15002000 : Record 121;
    BEGIN

      SicapaReconcInvLine.SETRANGE("Destinataire Type",SicapaInvoiceLine."Destinataire Type");
      SicapaReconcInvLine.SETRANGE(Destinataire,SicapaInvoiceLine.Destinataire);
      SicapaReconcInvLine.SETRANGE("Type Facture",SicapaInvoiceLine."Type Facture");
      SicapaReconcInvLine.SETRANGE("Num Facture",SicapaInvoiceLine."Num Facture");
      SicapaReconcInvLine.SETRANGE("Num Ligne",SicapaInvoiceLine."Num Ligne");

      IF SicapaReconcInvLine.FINDSET(FALSE,FALSE) THEN
        REPEAT

            SicapaReconcInvLine."Reconc. Doc. Line No.")

            AddCandidate(PurchRcptLine);

    END;

    LOCAL PROCEDURE SetLineVars@15000360();
    BEGIN
      CLEAR(Reconciled);
      CLEAR(SourceUnitPrice);
      CLEAR(LineAmountExclVAT);
      CLEAR(LineAmountInclVAT);
      CLEAR(MatchLevel);
      CLEAR(WarningText);

      IF Rec."Document No." = '' THEN
        EXIT;
      Reconciled := SicapaMgt.IsPurchRcptLineReconciled(SicapaInvoiceLine,Rec);
      SourceUnitPrice := CalcSourceUnitPrice(Rec);
      LineAmountExclVAT := CalcLineAmountExclVAT(Rec);
      LineAmountInclVAT := CalcLineAmountInclVAT(Rec);
      MatchLevel := GetMatchLevelForDisplay(Rec);
      WarningText := GetWarningTextForDisplay(Rec);
    END;

    LOCAL PROCEDURE CalcUnitCostChildAmount@15000365() UnitCostAmount : Decimal;
    VAR
      ChildLine@15000366 : Record 8087768;
    BEGIN
      UnitCostAmount := 0;
      ChildLine.RESET;
      ChildLine.SETRANGE("Emetteur Type",         SicapaInvoiceLine."Emetteur Type");
      ChildLine.SETRANGE(Emetteur,                SicapaInvoiceLine.Emetteur);
      ChildLine.SETRANGE("Destinataire Type",      SicapaInvoiceLine."Destinataire Type");
      ChildLine.SETRANGE(Destinataire,             SicapaInvoiceLine.Destinataire);
      ChildLine.SETRANGE("Type Facture",           SicapaInvoiceLine."Type Facture");
      ChildLine.SETRANGE("Num Facture",            SicapaInvoiceLine."Num Facture");
      ChildLine.SETRANGE("Invoice Attached Line No.", SicapaInvoiceLine."Num Ligne");
      ChildLine.SETRANGE("Invoice Line Applied Type",
                         ChildLine."Invoice Line Applied Type"::UnitCost);
      IF ChildLine.FINDSET THEN
        REPEAT
          UnitCostAmount += ChildLine.Montant;
        UNTIL ChildLine.NEXT = 0;
    END;

    LOCAL PROCEDURE RefreshSummary@15000370();
    BEGIN
      CLEAR(InvoiceQty);
      CLEAR(ReconciledQty);
      CLEAR(QtyGap);
      CLEAR(InvoiceAmount);
      CLEAR(ReconciledAmount);
      CLEAR(AmountGap);
      CLEAR(SourceAvgUnitPrice);
      CLEAR(ReconciliationCount);

      InvoiceQty := SicapaInvoiceLine."Quantite facture";
      // InvoiceAmount = montant article + montant des lignes UnitCost liees
      // (le BR inclut les deux, donc la comparaison doit inclure les deux)
      InvoiceAmount := SicapaInvoiceLine.Montant + CalcUnitCostChildAmount;

      SicapaReconcInvLine.RESET;
      SicapaReconcInvLine.SETRANGE("Emetteur Type",SicapaInvoiceLine."Emetteur Type");
      SicapaReconcInvLine.SETRANGE(Emetteur,SicapaInvoiceLine.Emetteur);
      SicapaReconcInvLine.SETRANGE("Destinataire Type",SicapaInvoiceLine."Destinataire Type");
      SicapaReconcInvLine.SETRANGE(Destinataire,SicapaInvoiceLine.Destinataire);
      SicapaReconcInvLine.SETRANGE("Type Facture",SicapaInvoiceLine."Type Facture");
      SicapaReconcInvLine.SETRANGE("Num Facture",SicapaInvoiceLine."Num Facture");
      SicapaReconcInvLine.SETRANGE("Num Ligne",SicapaInvoiceLine."Num Ligne");
      // Uniquement les BR (PostingDoc) contribuent au rapprochement
      // Les Child (UnitCost, Charge...) ne sont pas des lignes BR
      SicapaReconcInvLine.SETRANGE("Reconc. Doc. Type",SicapaReconcInvLine."Reconc. Doc. Type"::PostingDoc);

      IF SicapaReconcInvLine.FINDSET(FALSE,FALSE) THEN
        REPEAT
          ReconciliationCount += 1;
          ReconciledQty += SicapaReconcInvLine."Reconc. Qty. Not Inv.";
          ReconciledAmount +=
            ROUND(
              SicapaReconcInvLine."Reconc. Source Unit Price" *
              SicapaReconcInvLine."Reconc. Qty. Not Inv.");
        UNTIL SicapaReconcInvLine.NEXT = 0;

      QtyGap := InvoiceQty - ReconciledQty;
      AmountGap := InvoiceAmount - ReconciledAmount;

      IF ReconciledQty <> 0 THEN
        SourceAvgUnitPrice := ROUND(ReconciledAmount / ReconciledQty);
    END;

    LOCAL PROCEDURE CalcSourceUnitPrice@15000380(PurchRcptLine@15000381 : Record 121) : Decimal;
    BEGIN
      EXIT(
        ROUND(
          PurchRcptLine."Unit Cost (LCY)" *
          (1 - (PurchRcptLine."Line Discount %" / 100))));
    END;

    LOCAL PROCEDURE CalcLineAmountExclVAT@15000390(PurchRcptLine@15000391 : Record 121) : Decimal;
    BEGIN
      EXIT(ROUND(CalcSourceUnitPrice(PurchRcptLine) * PurchRcptLine."Qty. Rcd. Not Invoiced"));
    END;

    LOCAL PROCEDURE CalcLineAmountInclVAT@15000400(PurchRcptLine@15000401 : Record 121) : Decimal;
    VAR
      AmountExclVAT@15002000 : Decimal;
    BEGIN
      AmountExclVAT := CalcLineAmountExclVAT(PurchRcptLine);
      EXIT(ROUND(AmountExclVAT * (1 + (PurchRcptLine."VAT %" / 100))));
    END;

    LOCAL PROCEDURE GetMatchLevelForDisplay@15000410(PurchRcptLine@15000411 : Record 121) : Code[10];
    BEGIN
      IF
        (PurchRcptLine."Buy-from Vendor No." = SicapaInvoiceHdr."Vendor No.") AND
        (PurchRcptLine."Order No." = SicapaInvoiceLine."Source Document No.") AND
        (PurchRcptLine."Order Line No." = SicapaInvoiceLine."Source Document Line No.") AND
        (PurchRcptLine."No." = SicapaInvoiceLine."Item No. NavAgri") AND
        ((SicapaInvoiceLine."Variant Code NavAgri" = '') OR
         (PurchRcptLine."Variant Code" = SicapaInvoiceLine."Variant Code NavAgri")) AND
        ((SicapaInvoiceLine."Location Code" = '') OR
         (PurchRcptLine."Location Code" = SicapaInvoiceLine."Location Code"))
      THEN
        EXIT('RPRR1');

      IF
        (PurchRcptLine."Order No." = SicapaInvoiceLine."Source Document No.") AND
        (PurchRcptLine."Order Line No." = SicapaInvoiceLine."Source Document Line No.") AND
        (PurchRcptLine."No." = SicapaInvoiceLine."Item No. NavAgri") AND
        ((SicapaInvoiceLine."Variant Code NavAgri" = '') OR
         (PurchRcptLine."Variant Code" = SicapaInvoiceLine."Variant Code NavAgri")) AND
        ((SicapaInvoiceLine."Location Code" = '') OR
         (PurchRcptLine."Location Code" = SicapaInvoiceLine."Location Code"))
      THEN
        EXIT('RPRR2');

      IF
        (PurchRcptLine."Buy-from Vendor No." = SicapaInvoiceHdr."Vendor No.") AND
        (PurchRcptLine."No." = SicapaInvoiceLine."Item No. NavAgri") AND
        ((SicapaInvoiceLine."Variant Code NavAgri" = '') OR
         (PurchRcptLine."Variant Code" = SicapaInvoiceLine."Variant Code NavAgri")) AND
        ((SicapaInvoiceLine."Location Code" = '') OR
         (PurchRcptLine."Location Code" = SicapaInvoiceLine."Location Code"))
      THEN
        EXIT('RPRR3');

      IF
        (PurchRcptLine."No." = SicapaInvoiceLine."Item No. NavAgri") AND
        ((SicapaInvoiceLine."Variant Code NavAgri" = '') OR
         (PurchRcptLine."Variant Code" = SicapaInvoiceLine."Variant Code NavAgri")) AND
        ((SicapaInvoiceLine."Location Code" = '') OR
         (PurchRcptLine."Location Code" = SicapaInvoiceLine."Location Code"))
      THEN
        EXIT('RPRR4');

      EXIT('MANUAL');
    END;

    LOCAL PROCEDURE GetWarningTextForDisplay@15000420(PurchRcptLine@15000421 : Record 121) WarningText : Text[250];
    VAR
      SourcePrice@15002000 : Decimal;
    BEGIN
      WarningText := '';

      IF PurchRcptLine."VAT %" <> SicapaInvoiceLine."Taux Tva" THEN
        WarningText := AddWarningText(WarningText,'TVA');

      SourcePrice := CalcSourceUnitPrice(PurchRcptLine);
      IF ABS(SourcePrice - SicapaInvoiceLine."Prix net") > 0.01 THEN
        WarningText := AddWarningText(WarningText,'Prix');

      IF (SicapaInvoiceLine."Source Document No." <> '') AND
         (PurchRcptLine."Order No." <> SicapaInvoiceLine."Source Document No.") THEN
        WarningText := AddWarningText(WarningText,'Commande');

      EXIT(WarningText);
    END;
