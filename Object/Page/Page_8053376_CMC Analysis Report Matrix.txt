    PROCEDURE EvaluateExpression@8053000(Expression_P@8053001 : Text[80];AnalysisLine_P@8053002 : Record 8053361;SetColFilter_P@8053016 : Boolean) : Decimal;
    VAR
      Result_L@8053003 : Decimal;
      Parantheses_L@8053004 : Integer;
      Operator_L@8053005 : Char;
      LeftOperand_L@8053014 : Text[80];
      RightOperand_L@8053013 : Text[80];
      LeftResult_L@8053012 : Decimal;
      RightResult_L@8053011 : Decimal;
      i_L@8053010 : Integer;
      IsExpression_L@8053009 : Boolean;
      IsFilter_L@8053008 : Boolean;
      Operators_L@8053007 : Text[8];
      OperatorNo_L@8053006 : Integer;
      AnalysisLineID_L@8053015 : Integer;
      AnalysisLine_L@8053000 : Record 8053361;
    BEGIN
      // --> 20130418-FINA12-CMC02.02.00-WSA01
      Result_L :=0;
      Expression_P := DELCHR(Expression_P,'<>',' ');
      IF STRLEN(Expression_P) > 0 THEN
       BEGIN
        Parantheses_L := 0;
        IsExpression_L := FALSE;
        Operators_L := '+-*/^%';
        OperatorNo_L := 1;
        REPEAT
          i_L := STRLEN(Expression_P);
          REPEAT
            IF Expression_P[i_L] = '(' THEN
              Parantheses_L := Parantheses_L + 1
            ELSE
              IF Expression_P[i_L] = ')' THEN
                Parantheses_L := Parantheses_L - 1;
            IF (Parantheses_L = 0) AND (Expression_P[i_L] = Operators_L[OperatorNo_L]) THEN
              IsExpression_L := TRUE
            ELSE
              i_L := i_L - 1;
          UNTIL IsExpression_L OR (i_L <= 0);
          IF NOT IsExpression_L THEN
            OperatorNo_L := OperatorNo_L + 1;
        UNTIL (OperatorNo_L > STRLEN(Operators_L)) OR IsExpression_L;
        IF IsExpression_L THEN
         BEGIN
          IF i_L > 1 THEN
            LeftOperand_L := COPYSTR(Expression_P,1,i_L - 1)
          ELSE
            LeftOperand_L := '';
          IF i_L < STRLEN(Expression_P) THEN
            RightOperand_L := COPYSTR(Expression_P,i_L + 1)
          ELSE
            RightOperand_L := '';
          Operator_L := Expression_P[i_L];
          LeftResult_L := EvaluateExpression(LeftOperand_L,AnalysisLine_P,SetColFilter_P);
          RightResult_L := EvaluateExpression(RightOperand_L,AnalysisLine_P,SetColFilter_P);
          CASE Operator_L OF
            '^':
              Result_L := POWER(LeftResult_L,RightResult_L);
            '%' :
              IF RightResult_L = 0 THEN BEGIN
                Result_L := 0;
              END ELSE
                Result_L := 100 * LeftResult_L / RightResult_L;
            '*':
              Result_L := LeftResult_L * RightResult_L;
            '/':
              IF RightResult_L = 0 THEN BEGIN
                Result_L := 0;
              END ELSE
                Result_L := LeftResult_L / RightResult_L;
            '+':
              Result_L := LeftResult_L + RightResult_L;
            '-':
              Result_L := LeftResult_L - RightResult_L;
          END;
         END ELSE
          IF (Expression_P[1] = '(') AND (Expression_P[STRLEN(Expression_P)] = ')') THEN
            Result_L := EvaluateExpression(COPYSTR(Expression_P,2,STRLEN(Expression_P) - 2),AnalysisLine_P,SetColFilter_P)
          ELSE BEGIN
            IsFilter_L :=
               (STRPOS(Expression_P,'..') +
                STRPOS(Expression_P,'|') +
                STRPOS(Expression_P,'<') +
                STRPOS(Expression_P,'>') +
                STRPOS(Expression_P,'&') +
                STRPOS(Expression_P,'=') > 0);
            IF (STRLEN(Expression_P) > 10) AND (NOT IsFilter_L) THEN
              EVALUATE(Result_L,Expression_P)
            ELSE
              BEGIN
               AnalysisLine_L.RESET;
               AnalysisLine_L.SETRANGE(Code,Code);
               AnalysisLine_L.SETFILTER(AnalysisLine_L."Row Ref. No.",Expression_P);
               AnalysisLineID_L := AnalysisLine_L.Line;
               IF AnalysisLine_L.FIND('-') THEN
                 REPEAT
                   IF AnalysisLine_L.Line <> AnalysisLineID_L THEN
                     Result_L := Result_L + CalcCellValue(AnalysisLine_L,SetColFilter_P);
                 UNTIL AnalysisLine_L.NEXT = 0
                ELSE
                 IF IsFilter_L OR (NOT EVALUATE(Result_L,Expression_P)) THEN
                   ERROR(Text8053003);
              END;
          END;
      END;
      EXIT(Result_L);
      // <-- 20130418-FINA12-CMC02.02.00-WSA01
    END;
