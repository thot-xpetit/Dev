    PROCEDURE CalculateAlertDate@60007(VendorNo_P@60003 : Code[20];SpecieCode_P@60002 : Code[20]) AlertDate_Res : Date;
    VAR
      RecCollContract_L@60001 : Record 8052983;
    BEGIN
      AlertDate_Res := 0D;
      IF VendorNo_P = '' THEN
        ERROR (Text0001,'Code fournisseur');
      IF SpecieCode_P = '' THEN
        ERROR (Text0001,'EspSce');
      CLEAR(RecCollContract_L);
      RecCollContract_L.SETRANGE("Vendor No.",VendorNo_G);
      RecCollContract_L.SETRANGE("Species Code",SpecieCode_G);
      RecCollContract_L.SETFILTER(Status,'<>%1',RecCollContract_L.Status::Finished);
      IF RecCollContract_L.FINDSET THEN
      REPEAT
        IF RecCollContract_L."Warning Date" > AlertDate_Res THEN
          AlertDate_Res := RecCollContract_L."Warning Date";
      UNTIL RecCollContract_L.NEXT = 0;
      EXIT(AlertDate_Res);
    END;

    PROCEDURE NbrOfTrucksOnContracts@60002() TrucksNumber_Res : Integer;
    VAR
      RecCollContract_L@60000 : Record 8052983;
      TotQtyNonReceived_L@60002 : Decimal;
      Page_CollContract_L@60005 : Page 8053139;
      RecCollContract2_L@60006 : Record 8052983;
      RecShipType_L@60001 : Record 8053264;
    BEGIN
      TrucksNumber_Res := 0;
      TotQtyNonReceived_L := 0;

      CLEAR(RecCollContract_L);
      RecCollContract_L.SETRANGE(Type,RecCollContract_L.Type::Contract);
      RecCollContract_L.SETRANGE(Status,RecCollContract_L.Status::Locked);
      RecCollContract_L.SETRANGE("Vendor No.",VendorNo_G);
      RecCollContract_L.SETRANGE("Campaign No.",CampainNo_G);
      RecCollContract_L.SETRANGE("Species Code",SpecieCode_G);
      RecCollContract_L.SETRANGE("Warning Date",0D,AlertDate_G);
      IF RecCollContract_L.FINDSET THEN
      REPEAT
        RecCollContract_L.CALCFIELDS("Quantity Received");
        TotQtyNonReceived_L += RecCollContract_L.Quantity - RecCollContract_L."Quantity Received";
      UNTIL RecCollContract_L.NEXT = 0;

      RecShipType_L.GET(ShippingType_G);

      IF RecShipType_L.Capacity <> 0 THEN
        TrucksNumber_Res := ROUND((TotQtyNonReceived_L / RecShipType_L.Capacity),1,'>');
    END;

    PROCEDURE NbrOfTrucksOnCharterLines@60004() TrucksNumber_Res : Integer;
    VAR
      RecCharterLine_L@60001 : Record 8053265;
      RecCharter_L@60002 : Record 60013;
      RecLocation_L@60003 : Record 14;
      RecItem_L@60000 : Record 27;
    BEGIN
      TrucksNumber_Res := 0;
      CLEAR(RecCharter_L);
      RecCharter_L.SETRANGE("Ending Date",0D,AlertDate_G);
      IF RecCharter_L.FINDSET THEN
      REPEAT
        CLEAR(RecCharterLine_L);
        WITH RecCharterLine_L DO BEGIN
          SETRANGE("Charter No",RecCharter_L."No.");
          SETRANGE("From Type","From Type"::Vendor);
          SETRANGE("From Code",VendorNo_G);
          SETRANGE(Status,Status::Open);
          IF FINDSET THEN
          REPEAT
            RecItem_L.GET("Item No.");
            IF RecItem_L."Species Code" = SpecieCode_G THEN BEGIN
              IF ("From Type"= "From Type"::Vendor) AND ("Destination Type"= "Destination Type"::Customer) THEN
                TrucksNumber_Res += 1;
              IF "Destination Type"= "Destination Type"::Location THEN BEGIN
                IF RecLocation_L.GET("Destination Code") THEN
                  IF RecLocation_L.Port = RecLocation_L.Port::"Maritime " THEN
                    TrucksNumber_Res += 1;
              END;
            END;
          UNTIL NEXT = 0;
        END;
      UNTIL RecCharter_L.NEXT = 0;
      EXIT(TrucksNumber_Res);
    END;
