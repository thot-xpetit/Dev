      "<< 3LI LOCALS >>"@8053007 : Integer;
      L_pgeLocationsWarehouseList@8053006 : Page 7347;
      L_recLocation@8053005 : Record 14;

      recLocation_L@8053002 : Record 14;
      Count_L@8053001 : Integer;
      cduWMSManag_L@8053000 : Codeunit 7302;
      GetSourceDocOutbound_L@8053008 : Codeunit 5752;
    BEGIN
      // --> 20141014-Log07-CMC02.02.00-FHS01
      recLocationTMP_L.DELETEALL;
      L_recLine.RESET;
      L_recLine.SETRANGE("Document No.", "No.");
      IF L_recLine.FINDSET(FALSE, FALSE) THEN
        REPEAT
          IF NOT(recLocationTMP_L.GET(L_recLine."Transfer-from Code")) THEN BEGIN
            IF L_recLocation.GET(L_recLine."Transfer-from Code") THEN BEGIN
              IF L_recLocation."Require Shipment" = TRUE THEN BEGIN
                IF cduWMSManag_L.LocationIsAllowed(L_recLine."Transfer-from Code") THEN BEGIN
                  recLocationTMP_L.INIT;
                  recLocationTMP_L.Code := L_recLine."Transfer-from Code";
                  recLocationTMP_L.INSERT;
                END;
              END;
            END;
          END;
        UNTIL L_recLine.NEXT = 0;

      Count_L := recLocationTMP_L.COUNT;

      IF Count_L = 0 THEN
        MESSAGE(Text8053001);
      IF Count_L >= 1 THEN BEGIN

        L_pgeLocationsWarehouseList.DeleteLimitLocation;
        IF recLocationTMP_L.FINDSET(FALSE, FALSE) THEN
          REPEAT
            L_pgeLocationsWarehouseList.LimitLocation(recLocationTMP_L.Code);
          UNTIL recLocationTMP_L.NEXT = 0;

        L_pgeLocationsWarehouseList.LOOKUPMODE := TRUE;
        IF L_pgeLocationsWarehouseList.RUNMODAL = ACTION::LookupOK THEN BEGIN
          L_pgeLocationsWarehouseList.GETRECORD(L_recLocation);

          L_recLine.RESET;
          L_recLine.SETRANGE("Document No.", "No.");
          L_recLine.SETRANGE("Transfer-from Code", L_recLocation.Code);
          IF L_recLine.ISEMPTY THEN
            MESSAGE(Text8053000, L_recLocation.Code)
          ELSE BEGIN
            GetSourceDocOutbound_L.SetLocationFilter(L_recLocation.Code);
            GetSourceDocOutbound_L.CreateFromOutbndTransferOrder(Rec);

            IF NOT FIND('=><') THEN
              INIT;
          END;
        END;
      END;
      // <-- 20141014-Log07-CMC02.02.00-FHS01
    END;
