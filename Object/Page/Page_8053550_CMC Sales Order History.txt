    PROCEDURE CreatePageEntries@8053009();
    VAR
      SH_L@8053002 : Record 36;
      SL_L@8053003 : Record 37;
      SHA_L@8053004 : Record 5107;
      SLA_L@8053005 : Record 5108;
      CMCSetup_L@8053001 : Record 8052982;
      VersionNo_L@8053000 : Integer;
    BEGIN

      //------------------------------------------------------------------------------------------------
      // The function fills the temporary table of this page with lines of the sales orders or archived sales orders filtered
      // using the parameters set by function SetPageParameters.
      // The priority is: if an order exist, the system uses the order, else it uses the last archived version of the order
      // to create the buffer.
      // Input:
      //   -
      // Output:
      //   -
      //------------------------------------------------------------------------------------------------


      RESET;
      DELETEALL;

      TmpSHA_G.DELETEALL(FALSE);
      VersionNo_L := 0;

      CMCSetup_L.FINDFIRST;

      SH_L.RESET;
      SH_L.SETRANGE("Document Type",DocumentType_G);
      SH_L.SETRANGE("Sell-to Customer No.",SellToCustomer_G);
      IF DocCategoryCodeFilter_G <> '' THEN
        SH_L.SETFILTER("Doc. Category Code",DocCategoryCodeFilter_G);
      SH_L.SETFILTER("Order Date",'>%1',CALCDATE(CMCSetup_L."Order History Date Calculation",WORKDATE));

      IF SH_L.FINDSET THEN
        REPEAT
          IF (IsCopyLineActionEnabled_G = FALSE) OR (SH_L."No." <> DocumentNo_G)  THEN BEGIN
            TmpSHA_G.TRANSFERFIELDS(SH_L);
            TmpSHA_G.INSERT(TRUE);
          END;
        UNTIL SH_L.NEXT =0;

      SHA_L.RESET;
      SHA_L.SETRANGE("Document Type",DocumentType_G);
      SHA_L.SETRANGE("Sell-to Customer No.",SellToCustomer_G);
      SHA_L.SETFILTER("Doc. Category Code",DocCategoryCodeFilter_G);
      SHA_L.SETFILTER("Order Date",'>%1',CALCDATE(CMCSetup_L."Order History Date Calculation",WORKDATE));

      IF SHA_L.FINDSET THEN
        REPEAT
          IF NOT DocumentAlreadyInserted(SHA_L) THEN
            IF IsLastVersion(SHA_L) THEN BEGIN
              TmpSHA_G.TRANSFERFIELDS(SHA_L);
              TmpSHA_G.INSERT(TRUE);
            END;
        UNTIL SHA_L.NEXT =0;

      TmpSHA_G.RESET;
      TmpSHA_G.ASCENDING(FALSE);
      IF TmpSHA_G.FIND('-') THEN
        REPEAT
          IF TmpSHA_G."Version No." = 0 THEN BEGIN // Sales Order
            SL_L.RESET;
            SL_L.SETRANGE("Document Type",TmpSHA_G."Document Type");
            SL_L.SETRANGE("Document No.",TmpSHA_G."No.");
              IF SL_L.FINDSET THEN BEGIN
                VersionNo_L += 1;
                REPEAT
                  Rec.TRANSFERFIELDS(SL_L);
                  Rec.VALIDATE("Version No.",VersionNo_L);
                  Rec.INSERT(TRUE);
                UNTIL SL_L.NEXT = 0;
              END;
          END ELSE BEGIN // Sales Order Archived
            SLA_L.RESET;
            SLA_L.SETRANGE("Document Type",TmpSHA_G."Document Type");
            SLA_L.SETRANGE("Document No.",TmpSHA_G."No.");
            SLA_L.SETRANGE("Version No.",TmpSHA_G."Version No.");
              IF SLA_L.FINDSET THEN BEGIN
                VersionNo_L += 1;
                REPEAT
                  Rec.TRANSFERFIELDS(SLA_L);
                  Rec.VALIDATE("Version No.",VersionNo_L);
                  Rec.INSERT(TRUE);
                UNTIL SLA_L.NEXT = 0;
              END;
          END;
        UNTIL TmpSHA_G.NEXT = 0;

      Rec.RESET;
      Rec.SETCURRENTKEY("Document Type","Version No.","Line No.");
      Rec.ASCENDING(TRUE);

      // --> 20131028-Nut09-CMC02.02.00-0981-BSI01
      IF Rec.FINDFIRST THEN
        CurrPage.SETRECORD(Rec);
      // <-- 20131028-Nut09-CMC02.02.00-0981-BSI01
    END;

    PROCEDURE DocumentAlreadyInserted@8053034(SHA_P@8053000 : Record 5107) : Boolean;
    BEGIN
      //------------------------------------------------------------------------------------------------
      // The function checks if a sales order wasn't already inserted for the input archived document.
      // This check is done, because if we have the sales order, any archived version of this sales order will
      // not be taken in consideration.
      // Input:
      //   - SHA_P - archived sales document to be checked
      // Output:
      //   -
      //------------------------------------------------------------------------------------------------

      TmpSHA_G.RESET;
      TmpSHA_G.SETRANGE("Document Type",SHA_P."Document Type");
      TmpSHA_G.SETRANGE("No.",SHA_P."No.");

      IF TmpSHA_G.ISEMPTY THEN BEGIN
        TmpSHA_G.RESET;
        EXIT(FALSE)
      END ELSE BEGIN
        TmpSHA_G.RESET;
        EXIT(TRUE);
      END;
    END;

    PROCEDURE IsLastVersion@8053040(SHA_P@8053000 : Record 5107) : Boolean;
    VAR
      SHA_L@8053001 : Record 5107;
    BEGIN
      //------------------------------------------------------------------------------------------------
      // The function checks if the input archived document is the last archived version of the sales order.
      // Input:
      //   - SHA_P - archived sales document to be checked
      // Output:
      //   -
      //------------------------------------------------------------------------------------------------

      SHA_L.RESET;
      SHA_L.SETRANGE("Document Type",SHA_P."Document Type");
      SHA_L.SETRANGE("No.",SHA_P."No.");

      IF SHA_L.FINDSET THEN
        REPEAT
          IF SHA_L."Version No." <> SHA_P."Version No." THEN BEGIN  //exclude current line
            IF SHA_L."Version No." > SHA_P."Version No." THEN BEGIN
              EXIT(FALSE);
            END;
          END;
        UNTIL SHA_L.NEXT = 0;
      EXIT(TRUE);
    END;

    PROCEDURE InsertSelectedArchSalesLines@8053013(VAR SalesLineArch_P@8053001 : Record 5108);
    VAR
      SalesLine_L@8053000 : Record 37;
      LineNo_L@8053002 : Integer;
      SalesHeader_L@8053003 : Record 36;
      recSalesLineArch_Tmp_L@8053004 : TEMPORARY Record 5108;
      ToInsert_L@8053005 : Integer;
      recSalesLineArch_L@8053006 : Record 5108;
      RootMixLineNo_L@8053007 : Integer;
      recSalesLineOpen_L@8053008 : Record 37;
    BEGIN
      //------------------------------------------------------------------------------------------------
      // The function is used to copy in the sales order subform the history lines selected by the user from the current page.
      // Input:
      //   - SalesLineArch_P - archived sales line to be copied
      // Output:
      //   -
      //------------------------------------------------------------------------------------------------

      SalesLine_L.RESET;
      SalesHeader_L.GET(DocumentType_G,DocumentNo_G);

      IF SalesLineArch_P.FINDSET THEN
        REPEAT
          // --> 20131031-Nut09-CMC02.02.00-0981-BSI01
          IF ((SalesLineArch_P."CVO Item Attached Line" = 0)
            AND (SalesLineArch_P."RPD Item Attached Line" =0)
              AND (SalesLineArch_P."CVO Corn Item Attached Line" = 0)) THEN BEGIN
          // <-- 20131031-Nut09-CMC02.02.00-0981-BSI01

            //--> 20140407-Nut04-CMC02.02.00-1956-FHS01
            ToInsert_L := 0;
            //Recherche : est-ce que la ligne est d'j. ins'r'e
            recSalesLineArch_Tmp_L.RESET;
            recSalesLineArch_Tmp_L.SETRANGE("Document Type", SalesLineArch_P."Document Type");
            recSalesLineArch_Tmp_L.SETRANGE("Document No.", SalesLineArch_P."Document No.");
            recSalesLineArch_Tmp_L.SETRANGE("Line No.", SalesLineArch_P."Line No.");
            IF NOT(recSalesLineArch_Tmp_L.FINDLAST) THEN BEGIN
              IF (SalesLineArch_P."Item Mix Root Line No." <> 0) AND
                 (SalesLineArch_P."Item Mix Root Line No." <> SalesLineArch_P."Line No.") THEN BEGIN
                //Recherche : est-ce que la ligne principale est d'j. ins'r'e
                recSalesLineArch_Tmp_L.SETRANGE("Line No.", SalesLineArch_P."Item Mix Root Line No.");
                IF NOT(recSalesLineArch_Tmp_L.FINDLAST) THEN BEGIN
                  //On se positionne sur la ligne . ins'rer
                  IF recSalesLineArch_L.GET(SalesLineArch_P."Document Type", SalesLineArch_P."Document No.",
                                            SalesLineArch_P."Doc. No. Occurrence", SalesLineArch_P."Version No.",
                                            SalesLineArch_P."Item Mix Root Line No.") THEN BEGIN
                    ToInsert_L := 2;
                  END
                  ELSE BEGIN
                    //On se positionne sur la ligne . ins'rer
                    IF recSalesLineOpen_L.GET(SalesLineArch_P."Document Type", SalesLineArch_P."Document No.",
                                              SalesLineArch_P."Item Mix Root Line No.") THEN BEGIN
                      ToInsert_L := 3;
                    END;
                  END;
                END;
              END
              ELSE BEGIN
                IF recSalesLineOpen_L.GET(SalesLineArch_P."Document Type", SalesLineArch_P."Document No.",
                                          SalesLineArch_P."Item Mix Root Line No.") THEN BEGIN
                  ToInsert_L := 3;
                END
                ELSE BEGIN
                  ToInsert_L := 1;
                END;
              END;
            END;

            IF ToInsert_L > 0 THEN BEGIN
            //<-- 20140407-Nut04-CMC02.02.00-1956-FHS01

              LineNo_L := GetLastLineNo;
              SalesLine_L.VALIDATE("Document Type", SalesHeader_L."Document Type");
              SalesLine_L.VALIDATE("Document No.", SalesHeader_L."No.");
              SalesLine_L.VALIDATE("Line No.", LineNo_L + 10000);
              SalesLine_L.INSERT(TRUE);
              //--> 20140407-Nut04-CMC02.02.00-1956-FHS01
              //BEFORE
              {
              SalesLine_L.VALIDATE(Type,SalesLineArch_P.Type);
              SalesLine_L.VALIDATE("No.",SalesLineArch_P."No.");
              SalesLine_L.VALIDATE("Unit of Measure Code",SalesLineArch_P."Unit of Measure Code");
              SalesLine_L.VALIDATE(Quantity,SalesLineArch_P.Quantity);

              IF SalesLineArch_P."Location Code" = SalesHeader_L."Location Code" THEN BEGIN
                IF SalesLineArch_P."Ship-to Code" <> '' THEN
                  SalesLine_L.VALIDATE("Ship-to Code",SalesLineArch_P."Ship-to Code");
                IF SalesLineArch_P."Unloading Place Code" <> '' THEN
                  SalesLine_L.VALIDATE("Unloading Place Code",SalesLineArch_P."Unloading Place Code");
                IF SalesLineArch_P."Vehicle Size Code" <> '' THEN
                  SalesLine_L.VALIDATE("Vehicle Size Code",SalesLineArch_P."Vehicle Size Code");
                IF SalesLineArch_P."Unloading Method Code" <> '' THEN
                  SalesLine_L.VALIDATE("Unloading Method Code",SalesLineArch_P."Unloading Method Code");
                IF SalesLineArch_P."Time Slot Code" <> '' THEN
                  SalesLine_L.VALIDATE("Time Slot Code",SalesLineArch_P."Time Slot Code");
                SalesLine_L.VALIDATE("Call Ship-to Contact",SalesLineArch_P."Call Ship-to Contact");
              END;
              }
              //AFTER
              IF ToInsert_L = 1 THEN BEGIN
                SetSalesLineFromArchived(SalesLine_L, SalesLineArch_P, SalesHeader_L);
                IF SalesLineArch_P."Item Mix Root Line No." = SalesLineArch_P."Line No." THEN BEGIN
                  SalesLine_L."Item Mix Code" := SalesLineArch_P."Item Mix Code";
                  //--> 20140410-Nut04-CMC02.02.00-1956-FHS02
                  SalesLine_L."Item Mix Base Quantity" := SalesLineArch_P."Item Mix Base Quantity";
                  //<-- 20140410-Nut04-CMC02.02.00-1956-FHS02
                  SalesLine_L."Item Mix Root Line No." := SalesLine_L."Line No.";
                END;
              END;
              IF ToInsert_L = 2 THEN BEGIN
                SetSalesLineFromArchived(SalesLine_L, recSalesLineArch_L, SalesHeader_L);
                IF recSalesLineArch_L."Item Mix Root Line No." = recSalesLineArch_L."Line No." THEN BEGIN
                  SalesLine_L."Item Mix Code" := recSalesLineArch_L."Item Mix Code";
                  //--> 20140410-Nut04-CMC02.02.00-1956-FHS02
                  SalesLine_L."Item Mix Base Quantity" := recSalesLineArch_L."Item Mix Base Quantity";
                  //<-- 20140410-Nut04-CMC02.02.00-1956-FHS02
                  SalesLine_L."Item Mix Root Line No." := SalesLine_L."Line No.";
                END;
              END;
              IF ToInsert_L = 3 THEN BEGIN
                SetSalesLineFromOpen(SalesLine_L, recSalesLineOpen_L, SalesHeader_L);
                IF recSalesLineOpen_L."Item Mix Root Line No." = recSalesLineOpen_L."Line No." THEN BEGIN
                  SalesLine_L."Item Mix Code" := recSalesLineOpen_L."Item Mix Code";
                  //--> 20140410-Nut04-CMC02.02.00-1956-FHS02
                  SalesLine_L."Item Mix Base Quantity" := recSalesLineOpen_L."Item Mix Base Quantity";
                  //<-- 20140410-Nut04-CMC02.02.00-1956-FHS02
                  SalesLine_L."Item Mix Root Line No." := SalesLine_L."Line No.";
                END;
              END;
              //<-- 20140407-Nut04-CMC02.02.00-1956-FHS01

              SalesLine_L.MODIFY(TRUE);

              //--> 20140407-Nut04-CMC02.02.00-1956-FHS01
              IF ToInsert_L = 1 THEN BEGIN
                InsertMixLine(SalesLineArch_P, SalesHeader_L, SalesLine_L."Line No.");
                InsertMixPercent(SalesLine_L, SalesLineArch_P);
              END;
              IF ToInsert_L = 2 THEN BEGIN
                InsertMixLine(recSalesLineArch_L, SalesHeader_L, SalesLine_L."Line No.");
                InsertMixPercent(SalesLine_L, recSalesLineArch_L);
              END;
              IF ToInsert_L = 3 THEN BEGIN
                InsertMixLineFromOpen(recSalesLineOpen_L, SalesHeader_L, SalesLine_L."Line No.");
                InsertMixPercentFromOpen(SalesLine_L, recSalesLineOpen_L);
              END;
              //<-- 20140407-Nut04-CMC02.02.00-1956-FHS01
              //On sauvegarde le fait que l'on vient d'ins'rer une ligne
              recSalesLineArch_Tmp_L.INIT;
              IF ToInsert_L = 1 THEN BEGIN
                recSalesLineArch_Tmp_L.TRANSFERFIELDS(SalesLineArch_P, TRUE);
              END;
              IF ToInsert_L = 2 THEN BEGIN
                recSalesLineArch_Tmp_L.TRANSFERFIELDS(recSalesLineArch_L, TRUE);
              END;
              IF ToInsert_L = 3 THEN BEGIN
                recSalesLineArch_Tmp_L.TRANSFERFIELDS(recSalesLineOpen_L, TRUE);
              END;
              recSalesLineArch_Tmp_L.INSERT;
              //<-- 20140407-Nut04-CMC02.02.00-1956-FHS01
            END;
            //<-- 20140407-Nut04-CMC02.02.00-1956-FHS01

          // --> 20131031-Nut09-CMC02.02.00-0981-BSI01
          END;
          // <-- 20131031-Nut09-CMC02.02.00-0981-BSI01
        UNTIL SalesLineArch_P.NEXT = 0;
    END;

    PROCEDURE SetSalesLineFromArchived@8053002(VAR recSalesLine_Var_P@8053000 : Record 37;VAR recArchivedSalesLine_Var_P@8053001 : Record 5108;recSalesHeader_P@8053002 : Record 36);
    BEGIN
      //----------------------------------------------------------------------------------------------------------------------------------
      // met . jour la ligne de commande de vente en fonction des donn'es de la ligne de commande de vente archiv'e
      //----------------------------------------------------------------------------------------------------------------------------------
      //  Entr'es
      //    - recSalesLine_Var_P : Ligne . mettre . jour
      //    - recArchivedSalesLine_Var_P : Ligne contenant les donn'es . recopier
      //    - recSalesHeader_P : Ent^te de commande de la ligne . mettre . jour
      //  Sorties
      //----------------------------------------------------------------------------------------------------------------------------------
      //--> 20140407-Nut04-CMC02.02.00-1956-FHS01

      recSalesLine_Var_P.VALIDATE("Unit of Measure Code", recArchivedSalesLine_Var_P."Unit of Measure Code");
      recSalesLine_Var_P.VALIDATE(Quantity, recArchivedSalesLine_Var_P.Quantity);

      IF recArchivedSalesLine_Var_P."Location Code" = recSalesHeader_P."Location Code" THEN BEGIN
        IF recArchivedSalesLine_Var_P."Ship-to Code" <> '' THEN
          recSalesLine_Var_P.VALIDATE("Ship-to Code", recArchivedSalesLine_Var_P."Ship-to Code");
        IF recArchivedSalesLine_Var_P."Unloading Place Code" <> '' THEN
          recSalesLine_Var_P.VALIDATE("Unloading Place Code", recArchivedSalesLine_Var_P."Unloading Place Code");
        IF recArchivedSalesLine_Var_P."Vehicle Size Code" <> '' THEN
          recSalesLine_Var_P.VALIDATE("Vehicle Size Code", recArchivedSalesLine_Var_P."Vehicle Size Code");
        IF recArchivedSalesLine_Var_P."Unloading Method Code" <> '' THEN
          recSalesLine_Var_P.VALIDATE("Unloading Method Code", recArchivedSalesLine_Var_P."Unloading Method Code");
        IF recArchivedSalesLine_Var_P."Time Slot Code" <> '' THEN
          recSalesLine_Var_P.VALIDATE("Time Slot Code", recArchivedSalesLine_Var_P."Time Slot Code");
        recSalesLine_Var_P.VALIDATE("Call Ship-to Contact", recArchivedSalesLine_Var_P."Call Ship-to Contact");
      END;
      //<-- 20140407-Nut04-CMC02.02.00-1956-FHS01
    END;

    PROCEDURE InsertMixLine@8053011(VAR recArchivedSalesLine_Var_P@8053004 : Record 5108;recSalesHeader_P@8053000 : Record 36;LineNo_P@8053005 : Integer);
    VAR
      recSalesLine_L@8053003 : Record 37;
      recSalesLineArch_L@8053001 : Record 5108;
      LineNo_L@8053002 : Integer;
    BEGIN
      //----------------------------------------------------------------------------------------------------------------------------------
      // InsSre les lignes associ'es
      //----------------------------------------------------------------------------------------------------------------------------------
      //  Entr'es
      //    - recArchivedSalesLine_P : Ligne . mettre . jour
      //    - recSalesHeader_P : Ent^te de commande de la ligne . mettre . jour
      //    - LineNo_P : num'ro de ligne
      //  Sorties
      //----------------------------------------------------------------------------------------------------------------------------------
      //--> 20140407-Nut04-CMC02.02.00-1956-FHS01
      LineNo_L := LineNo_P;
      recSalesLineArch_L.RESET;
      recSalesLineArch_L.SETRANGE("Document Type", recArchivedSalesLine_Var_P."Document Type");
      recSalesLineArch_L.SETRANGE("Document No.", recArchivedSalesLine_Var_P."Document No.");
      recSalesLineArch_L.SETRANGE("Doc. No. Occurrence", recArchivedSalesLine_Var_P."Doc. No. Occurrence");
      recSalesLineArch_L.SETRANGE("Version No.", recArchivedSalesLine_Var_P."Version No.");
      recSalesLineArch_L.SETFILTER("Line No.", '<>%1', recArchivedSalesLine_Var_P."Line No.");
      recSalesLineArch_L.SETRANGE("Item Mix Root Line No.", recArchivedSalesLine_Var_P."Line No.");
      IF recSalesLineArch_L.FINDSET(FALSE, FALSE) THEN BEGIN
        REPEAT
          LineNo_L := LineNo_L + 10000;
          recSalesLine_L.INIT;
          recSalesLine_L.VALIDATE("Document Type", recSalesHeader_P."Document Type");
          recSalesLine_L.VALIDATE("Document No.", recSalesHeader_P."No.");
          recSalesLine_L.VALIDATE("Line No.", LineNo_L);
          recSalesLine_L.INSERT(TRUE);

          SetSalesLineFromArchived(recSalesLine_L, recSalesLineArch_L, recSalesHeader_P);
          recSalesLine_L."Item Mix Code" := recSalesLineArch_L."Item Mix Code";
          recSalesLine_L."Item Mix Root Line No." := LineNo_P;
          recSalesLine_L.MODIFY(TRUE);

          InsertMixPercent(recSalesLine_L, recSalesLineArch_L);
        UNTIL recSalesLineArch_L.NEXT = 0;
      END;
      //<-- 20140407-Nut04-CMC02.02.00-1956-FHS01
    END;
