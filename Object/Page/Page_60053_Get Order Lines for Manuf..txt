    PROCEDURE LookupSalesOrder@60002(OnlyLookup_P@60001 : Boolean);
    VAR
      DocNo_L@60000 : Code[20];
    BEGIN
      // --> 20141002-APP28-TER01.01.00-RMZ01
      // This function will filter the sales line with mix items
      // Input
      //    - OnlyLookup_P
      // Output
      CLEAR(Item_G);
      IF ItemNo_G <> '' THEN
        Item_G.SETRANGE("No.",ItemNo_G);
      Item_G.SETFILTER("Mix BOM No.",'<>%1','');
      Item_G.FINDSET(FALSE,FALSE);
      CLEAR(SalesHeader_G);
      SalesHeader_G.FILTERGROUP := 2;
      FoundItem_G := FALSE;
      REPEAT
        CLEAR(SalesLine_G);
        SalesLine_G.SETRANGE("Document Type",SalesLine_G."Document Type"::Order);
        IF NOT OnlyLookup_P THEN BEGIN
          SalesLine_G.SETRANGE("Document Type",SalesHeader_G."Document Type"::Order);
          SalesLine_G.SETRANGE("No.",OrderNo_G);
        END;
        IF CustomerNo_G <> '' THEN
          SalesLine_G.SETRANGE("Sell-to Customer No.",CustomerNo_G);
        SalesLine_G.SETRANGE(Type,SalesLine_G.Type::Item);
        IF ItemNo_G <> '' THEN
          SalesLine_G.SETRANGE("No.",ItemNo_G)
        ELSE
          SalesLine_G.SETFILTER("No.",'<>%1','');
        SalesLine_G.SETRANGE("Extract Manufacturing",FALSE);
        SalesLine_G.SETRANGE("Create Mix",FALSE);
        DocNo_L := '';
        IF SalesLine_G.FINDSET(FALSE,FALSE) THEN
          REPEAT
            Item_G.GET(SalesLine_G."No.");
            IF (Item_G."Mix BOM No." <> '') AND (SalesLine_G."Quantity Manufactured" < SalesLine_G.Quantity)
              AND (MixAllowedByLoc_G.VerifyLocationAllowed(Item_G."Mix BOM No.",SalesLine_G."Location Code")) THEN BEGIN
              IF SalesLine_G."Document No." <> DocNo_L THEN BEGIN
                SalesHeader_G.GET(SalesLine_G."Document Type",SalesLine_G."Document No.");
                SalesHeader_G.MARK := TRUE;
                DocNo_L := SalesLine_G."Document No.";
                FoundItem_G := TRUE;
              END;
            END;
          UNTIL SalesLine_G.NEXT = 0;

      UNTIL Item_G.NEXT = 0;
      SalesHeader_G.MARKEDONLY(TRUE);

      IF NOT OnlyLookup_P THEN
        EXIT;

      CLEAR(SalesOrderList_G);
      SalesOrderList_G.SETTABLEVIEW(SalesHeader_G);
      SalesOrderList_G.LOOKUPMODE := TRUE;
      IF SalesOrderList_G.RUNMODAL = ACTION::LookupOK THEN BEGIN
        SalesOrderList_G.GETRECORD(SalesHeader_G);
        OrderNo_G := SalesHeader_G."No.";
      END;
      // <-- 20141002-APP28-TER01.01.00-RMZ01
    END;

    PROCEDURE FillBOMJournal@60000();
    VAR
      LineNo_G@60000 : Integer;
    BEGIN
      // Input
      // Output
      IF NOT CONFIRM(Text60001,FALSE) THEN
        EXIT;

      CLEAR(SalesLine_G);
      SalesLine_G.SETRANGE("Document Type",SalesLine_G."Document Type"::Order);
      IF OrderNo_G <> '' THEN
        SalesLine_G.SETRANGE("Document No.",OrderNo_G);
      IF CustomerNo_G <> '' THEN
        SalesLine_G.SETRANGE("Sell-to Customer No.",CustomerNo_G);
      SalesLine_G.SETRANGE(Type,SalesLine_G.Type::Item);
      IF ItemNo_G <> '' THEN
        SalesLine_G.SETRANGE("No.",ItemNo_G)
      ELSE
        SalesLine_G.SETFILTER("No.",'<>%1','');

      SalesLine_G.SETRANGE("Extract Manufacturing",FALSE);
      SalesLine_G.SETRANGE("Create Mix",FALSE);

      LineNo_G := 0;
      CLEAR(BOMJournalLine_G);
      BOMJournalLine_G.SETRANGE("Journal Template Name",BOMTemplate_G);
      BOMJournalLine_G.SETRANGE("Journal Batch Name",BOMBatch_G);
      IF BOMJournalLine_G.FINDLAST THEN
        LineNo_G := BOMJournalLine_G."Line No." + 10000
      ELSE
        LineNo_G := 10000;

      Counter_G := 0;
      IF SalesLine_G.FINDFIRST THEN
        REPEAT
          Item_G.GET(SalesLine_G."No.");
          IF (Item_G."Mix BOM No." <> '') AND (SalesLine_G."Quantity Manufactured" < SalesLine_G.Quantity)
            AND (MixAllowedByLoc_G.VerifyLocationAllowed(Item_G."Mix BOM No.",SalesLine_G."Location Code")) THEN BEGIN
            CLEAR(BOMJournalLine_G);
            BOMJournalLine_G."Journal Template Name" := BOMTemplate_G;
            BOMJournalLine_G."Journal Batch Name" := BOMBatch_G;
            BOMJournalLine_G."Line No." := LineNo_G;
            BOMJournalLine_G.VALIDATE("Posting Date",WORKDATE);
            BOMJournalLine_G.SetUpNewLine(BOMJournalLine_G);
            BOMJournalLine_G.VALIDATE("Item No.",SalesLine_G."No.");
            BOMJournalLine_G.SkipShowMix(TRUE);
            BOMJournalLine_G."Sales Order No." := SalesLine_G."Document No.";
            BOMJournalLine_G."Sales Order Line No." := SalesLine_G."Line No.";
            BOMJournalLine_G.VALIDATE("Location Code",SalesLine_G."Location Code");
            BOMJournalLine_G.VALIDATE(Quantity,SalesLine_G.Quantity - SalesLine_G."Quantity Manufactured");
            BOMJournalLine_G.VALIDATE("Unit of Measure Code",SalesLine_G."Unit of Measure Code");
            BOMJournalLine_G.INSERT;
            LineNo_G := LineNo_G + 10000;
            Counter_G := Counter_G + 1;
          END;
        UNTIL SalesLine_G.NEXT = 0;

      MESSAGE(STRSUBSTNO(Text60002,Counter_G));
      CurrPage.CLOSE;
    END;
