    PROCEDURE CalcLines@1000000001();
    VAR
      Flag_L@1000000000 : Boolean;
      HierarchyLocType_L@1000000001 : Record 60160;
    BEGIN

      IF FiltreActivit'_G + FiltreFamille_G + FiltreSousFam_G + FiltreEspece_G = '' THEN
        ERROR(Text60001);

      IF FiltreMagasin_G = '' THEN
        ERROR(Text60002);

      NbLine_G :=0;
      DELETEALL;
      ItemTmp_G.DELETEALL;
      CLEAR(Item_G);
      Item_G.SETCURRENTKEY("Activity Code","Family Code","Sub-Family Code","Species Code");
      CLEAR(Location_G);
      CLEAR(MaxStockAuthorized_G);
      ItemVariantTmp_G.DELETEALL;
      CLEAR(ItemVariant_G);


      IF FiltreActivit'_G <>'' THEN BEGIN
        IF STRLEN(FiltreActivit'_G) > MAXSTRLEN(CMCActivity_G.Code) THEN
          Item_G.SETFILTER("Activity Code",FiltreActivit'_G)
        ELSE IF CMCActivity_G.GET(FiltreActivit'_G) THEN
          Item_G.SETRANGE("Activity Code",FiltreActivit'_G)
        ELSE
          Item_G.SETFILTER("Activity Code",FiltreActivit'_G);
      END;
      IF FiltreFamille_G <>'' THEN BEGIN
        IF STRLEN(FiltreFamille_G) > MAXSTRLEN(CMCFamily_G.Code) THEN
          Item_G.SETFILTER("Family Code",FiltreFamille_G)
        ELSE IF CMCFamily_G.GET(FiltreFamille_G) THEN
          Item_G.SETRANGE("Family Code",FiltreFamille_G)
        ELSE
          Item_G.SETFILTER("Family Code",FiltreFamille_G);
      END;
      IF FiltreSousFam_G <>'' THEN BEGIN
        IF STRLEN(FiltreSousFam_G) > MAXSTRLEN(CMCSubFamily_G.Code) THEN
          Item_G.SETFILTER("Sub-Family Code",FiltreSousFam_G)
        ELSE IF CMCSubFamily_G.GET(FiltreSousFam_G) THEN
          Item_G.SETRANGE("Sub-Family Code",FiltreSousFam_G)
        ELSE
          Item_G.SETFILTER("Sub-Family Code",FiltreSousFam_G);
      END;
      IF FiltreEspece_G <>'' THEN BEGIN
        IF STRLEN(FiltreEspece_G) > MAXSTRLEN(CMCSpecies_G.Code) THEN
          Item_G.SETFILTER("Species Code",FiltreEspece_G)
        ELSE IF CMCSpecies_G.GET(FiltreEspece_G) THEN
          Item_G.SETRANGE("Species Code",FiltreEspece_G)
        ELSE
          Item_G.SETFILTER("Species Code",FiltreEspece_G);
      END;
      IF FiltreArticle_G <>'' THEN BEGIN
          Item_G.SETFILTER("No.",FiltreArticle_G);
      END;

      IF NOT Item_G.FINDSET THEN
        ERROR(Text60003);

      Location_G.GET(FiltreMagasin_G);

      REPEAT
        ItemTmp_G."No." := Item_G."No.";
        ItemTmp_G."Species Code" := Item_G."Species Code";
        ItemTmp_G."Sub-Family Code" := Item_G."Sub-Family Code";
        ItemTmp_G."Family Code" := Item_G."Family Code";
        ItemTmp_G."Activity Code" := Item_G."Activity Code";
        ItemTmp_G."Packaging Type" := Item_G."Packaging Type";
        ItemTmp_G.INSERT;
      UNTIL Item_G.NEXT =0;

      IF FiltreVariant_G <>'' THEN BEGIN
        ItemVariant_G.SETFILTER(Code,FiltreVariant_G);
        IF ItemVariant_G.FINDSET THEN BEGIN
          REPEAT
            ItemVariantTmp_G."Item No." := ItemVariant_G."Item No.";
            ItemVariantTmp_G.Code := ItemVariant_G.Code;
            ItemVariantTmp_G.INSERT;
          UNTIL ItemVariant_G.NEXT = 0;
        END ELSE
          FiltreVariant_G := '';
      END;

      //MESSAGE(FORMAT(ItemTmp_G.COUNT));
      //MESSAGE(Item_G.GETFILTERS);
      //MESSAGE(FORMAT("Authorized Quantity") + ' ' + FORMAT(NbLine_G));

      LocationType_G.GET(Location_G."Location Type");
      //MaxStockAuthorized_G.SETRANGE("Location Type",LocationType_G.Code);
      MaxStockAuthorized_G.SETFILTER("Location Code",STRSUBSTNO('%1|''''',Location_G.Code)) ;
       MaxStockAuthorized_G.SETRANGE("Start Date",0D,WORKDATE);
      IF NOT MaxStockAuthorized_G.FINDSET THEN
        ERROR(Text60000);

      //MESSAGE(FORMAT(MaxStockAuthorized_G.COUNT));

      REPEAT
        Flag_L := FALSE;
        IF (MaxStockAuthorized_G."End Date" >= WORKDATE) OR (MaxStockAuthorized_G."End Date" = 0D) THEN BEGIN
          IF ItemTmp_G.GET(MaxStockAuthorized_G."Item No.") THEN BEGIN
            IF FiltreVariant_G <> '' THEN BEGIN
              IF NOT ItemVariantTmp_G.GET(MaxStockAuthorized_G."Item No.",MaxStockAuthorized_G."Variant Code") THEN
                Flag_L := TRUE;
            END;
            IF NOT Flag_L THEN
              IF NOT HierarchyLocType_L.GET(HierarchyLocType_L.Type::Species,ItemTmp_G."Species Code",
                  Location_G."Location Type",ItemTmp_G."Packaging Type") THEN
                IF NOT HierarchyLocType_L.GET(HierarchyLocType_L.Type::SubFamily,ItemTmp_G."Sub-Family Code",
                    Location_G."Location Type",ItemTmp_G."Packaging Type") THEN
                  IF NOT HierarchyLocType_L.GET(HierarchyLocType_L.Type::Family,ItemTmp_G."Family Code",
                      Location_G."Location Type",ItemTmp_G."Packaging Type") THEN
                    IF NOT HierarchyLocType_L.GET(HierarchyLocType_L.Type::Activity,ItemTmp_G."Activity Code",
                        Location_G."Location Type",ItemTmp_G."Packaging Type") THEN
                       IF NOT HierarchyLocType_L.GET(HierarchyLocType_L.Type::Species,ItemTmp_G."Species Code",
                           Location_G."Location Type",'') THEN
                         IF NOT HierarchyLocType_L.GET(HierarchyLocType_L.Type::SubFamily,ItemTmp_G."Sub-Family Code",
                             Location_G."Location Type",'') THEN
                           IF NOT HierarchyLocType_L.GET(HierarchyLocType_L.Type::Family,ItemTmp_G."Family Code",
                               Location_G."Location Type",'') THEN
                             IF NOT HierarchyLocType_L.GET(HierarchyLocType_L.Type::Activity,ItemTmp_G."Activity Code",
                                 Location_G."Location Type",'') THEN
                               Flag_L := TRUE;
            IF NOT Flag_L THEN
              IF HierarchyLocType_L."Transfer Prohibited" = HierarchyLocType_L."Transfer Prohibited"::Yes THEN
                Flag_L := TRUE;

            IF NOT Flag_L THEN
              CalcLocation;
          END;
        END;
      UNTIL MaxStockAuthorized_G.NEXT = 0;

      RESET;

      IF NOT FINDSET THEN
        ERROR(Text60000);
    END;

    PROCEDURE CalcLocation@1000000005();
    VAR
      CMCLocationType_L@1000000006 : Record 8053242;
      CMCLocationTypeOK_L@1000000009 : Boolean;
      CMCPackagingType_L@1000000011 : Record 8053152;
      Qty_L@1000000013 : Decimal;
      Stock_L@1000000014 : Decimal;
      SalesLine_L@1000000016 : Record 37;
      Item_L@1000000001 : Record 27;
      ItemVariant_L@1000000000 : Record 5401;
    BEGIN
      IF (MaxStockAuthorized_G."Location Code" <> Location_G.Code)
        AND (MaxStockAuthorized_G."Location Code" <> '') THEN
        EXIT;


      Item_L.GET(MaxStockAuthorized_G."Item No.");
      Item_L.SETRANGE("Drop Shipment Filter",FALSE);

      Qty_L := MaxStockAuthorized_G."Authorized Quantity";

      SalesLine_L.SETCURRENTKEY("Document Type",Type,"No.","Variant Code","Drop Shipment","Build Kit","Location Code");
      SalesLine_L.SETRANGE("Document Type",SalesLine_L."Document Type"::"Return Order");
      SalesLine_L.SETRANGE(Type,SalesLine_L.Type::Item);
      SalesLine_L.SETRANGE("No.",Item_L."No.");

      IF MaxStockAuthorized_G."Variant Code" <> '' THEN BEGIN
        Item_L.SETRANGE("Variant Filter",MaxStockAuthorized_G."Variant Code");
        SalesLine_L.SETRANGE("Variant Code",MaxStockAuthorized_G."Variant Code");
      END;

      Item_L.SETRANGE("Location Filter",Location_G.Code);
      SalesLine_L.SETRANGE("Location Code",Location_G.Code);
      Item_L.CALCFIELDS(Inventory,"Qty. on Purch. Order","Trans. Ord. Receipt (Qty.)");
      Stock_L += Item_L.Inventory + Item_L."Qty. on Purch. Order" + Item_L."Trans. Ord. Receipt (Qty.)";
      IF SalesLine_L.FINDSET THEN BEGIN
        SalesLine_L.CALCSUMS("Outstanding Qty. (Base)");
        Stock_L -= SalesLine_L."Outstanding Qty. (Base)";
      END;

      IF GET(MaxStockAuthorized_G.Type::Species,Item_L."Species Code",MaxStockAuthorized_G."Location Type",
            MaxStockAuthorized_G."Packaging Type",MaxStockAuthorized_G."Item No.",
            MaxStockAuthorized_G."Variant Code",Location_G.Code,0D) THEN BEGIN
        IF MaxStockAuthorized_G."Location Code" = Location_G.Code THEN
          DELETE
        ELSE
          EXIT;
      END;

      INIT;
      Type := MaxStockAuthorized_G.Type::Species;
      "Type No." := Item_L."Species Code";
      "Location Type" := MaxStockAuthorized_G."Location Type";
      "Packaging Type" := MaxStockAuthorized_G."Packaging Type";
      "Item No." := MaxStockAuthorized_G."Item No.";
      "Variant Code" := MaxStockAuthorized_G."Variant Code";
      // -- > 20241018-WI00-ADV01.01.00-3662-IJI
      IF (MaxStockAuthorized_G."Variant Code" <> '') AND (MaxStockAuthorized_G."Item No." <> '') THEN BEGIN
        ItemVariant_L.GET(MaxStockAuthorized_G."Item No.",MaxStockAuthorized_G."Variant Code");
        "Variant Description" :=  ItemVariant_L.Description;
      END;
      // <--  20241018-WI00-ADV01.01.00-3662-IJI
      "Location Code" := Location_G.Code;
      Location := Location_G.Name;
      Item := Item_L.Description;
      Unity := Item_L."Base Unit of Measure";
      "Authorized Quantity" := Qty_L;
      "Real Quantity" := Stock_L;
      IF "Authorized Quantity" <> 0 THEN BEGIN
        Step := (Stock_L*100) / "Authorized Quantity";
      END ELSE
        Step := 0;
      NbLine_G +=1;

      FormatIndentTmp;
    END;
