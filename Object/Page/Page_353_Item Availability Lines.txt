    LOCAL PROCEDURE CalculateNeed@5();
    BEGIN
      SetItemFilter;
      Item.CALCFIELDS(
        "Qty. on Purch. Order",
        "Qty. on Sales Order",
        "Qty. on Kit Sales Lines",
        // <-- 20110601-Wi00-CMC01.04
        "Qty. on Kit Sales Lines 2",
        // --> 20110601-Wi00-CMC01.04
        "Qty. on Service Order",
        "Qty. on Blanket Purch. Order",
        "Qty. on Blanket Sales Order",
        "Qty. on Blank. Kit Sales Lines",
        Inventory,
        "Net Change",
        "Scheduled Receipt (Qty.)",
        "Scheduled Need (Qty.)",
        "Planned Order Receipt (Qty.)",
        "FP Order Receipt (Qty.)",
        "Rel. Order Receipt (Qty.)",
        "Planned Order Release (Qty.)",
        "Purch. Req. Receipt (Qty.)",
        "Planning Issues (Qty.)",
        "Purch. Req. Release (Qty.)");
      IF Item.GETFILTER("Location Filter") <> '' THEN
        Item.CALCFIELDS(
          "Trans. Ord. Shipment (Qty.)",
          "Qty. in Transit",
          "Trans. Ord. Receipt (Qty.)");

      GrossRequirement :=
        Item."Qty. on Sales Order" +
        Item."Qty. on Kit Sales Lines" +
        Item."Qty. on Service Order" +
        Item."Scheduled Need (Qty.)" +
        Item."Trans. Ord. Shipment (Qty.)" +
        Item."Planning Issues (Qty.)" +
        Item.CalcPurchReturn;
      PlannedOrderReceipt :=
        Item."Planned Order Receipt (Qty.)" +
        Item."Purch. Req. Receipt (Qty.)";
      ScheduledReceipt :=
        Item."FP Order Receipt (Qty.)" +
        Item."Rel. Order Receipt (Qty.)" +
        Item."Qty. on Purch. Order" +
        Item."Qty. in Transit" +
        Item."Trans. Ord. Receipt (Qty.)" +
        Item.CalcSalesReturn;
      PlannedOrderReleases :=
        Item."Planned Order Release (Qty.)" +
        Item."Purch. Req. Release (Qty.)";
    END;
