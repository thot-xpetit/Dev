    PROCEDURE ApplyFilters@60003();
    BEGIN
      IF LocationCodeFilter_G = '' THEN

    PROCEDURE GetTempLocation@60002();
    VAR
      WhseEmployee_L@60001 : Record 7301;
      Location_L@60000 : Record 14;
    BEGIN
      TmpLocation_G.RESET;
      TmpLocation_G.DELETEALL;

      IF AllLocationAccess_G THEN BEGIN
        WhseEmployee_L.RESET;
        WhseEmployee_L.SETCURRENTKEY(Default);
        WhseEmployee_L.SETRANGE(Default,TRUE);
        WhseEmployee_L.SETRANGE("User ID",USERID);
        IF WhseEmployee_L.FINDSET THEN
          DefaultLocation_G := WhseEmployee_L."Location Code";
        EXIT;
      END;

      WhseEmployee_L.RESET;
      WhseEmployee_L.SETRANGE("User ID",USERID);
      IF WhseEmployee_L.FINDSET THEN
        REPEAT
          IF WhseEmployee_L."Location Code" <> '' THEN BEGIN
            Location_L.GET(WhseEmployee_L."Location Code");
            TmpLocation_G.INIT;
            TmpLocation_G.Code := WhseEmployee_L."Location Code";
            TmpLocation_G.Name := Location_L.Name;
            TmpLocation_G.INSERT;
            IF WhseEmployee_L.Default THEN
              DefaultLocation_G := WhseEmployee_L."Location Code";
          END;
        UNTIL WhseEmployee_L.NEXT = 0;
    END;

      END;
    END;
