    PROCEDURE CreateDocumentCategoryFilter@8053001() : Text[1024];
    VAR
      UserSetup_L@8053001 : Record 91;
      DocCategUser_L@8053000 : Record 8061333;
      DocCategFilter_L@8053002 : Text[1024];
    BEGIN
      // --> 20131010-Nut09-CMC02.02.00-BSI01
      //------------------------------------------------------------------------------------------------
      // The function creates a text filter concatenating the document categories codes founf first a user level and then at group level.
      // Output:
      //   - the document category filter
      //------------------------------------------------------------------------------------------------

      DocCategFilter_L:= '';
      UserSetup_L.GET(USERID);

      //search at user level
      DocCategUser_L.RESET;
      DocCategUser_L.SETRANGE(Code,UserSetup_L."User ID");
      DocCategUser_L.SETRANGE("Sales/Purchase",DocCategUser_L."Sales/Purchase"::Sales);
      DocCategUser_L.SETRANGE("Document Type",DocCategUser_L."Document Type"::Order);
      DocCategUser_L.SETRANGE(Type,DocCategUser_L.Type::User);
      IF DocCategUser_L.FINDSET THEN
        REPEAT
          DocCategFilter_L += DocCategUser_L."Doc. Category Code" + '|';
        UNTIL DocCategUser_L.NEXT = 0;

      // search at group level
      IF UserSetup_L."User Groupe Code" <> '' THEN BEGIN
        DocCategUser_L.SETRANGE(Code,UserSetup_L."User Groupe Code");
        DocCategUser_L.SETRANGE(Type,DocCategUser_L.Type::"User Group");
        IF DocCategUser_L.FINDSET THEN
          REPEAT
            DocCategFilter_L += DocCategUser_L."Doc. Category Code" + '|';
          UNTIL DocCategUser_L.NEXT = 0;
      END;

      IF DocCategFilter_L <> '' THEN
        DocCategFilter_L := DELSTR(DocCategFilter_L,STRLEN(DocCategFilter_L),1);

      EXIT(DocCategFilter_L);
      // <-- 20131010-Nut09-CMC02.02.00-BSI01
    END;
