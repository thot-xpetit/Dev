    PROCEDURE UpdateQtyAssgnt@2();
    VAR
      AcpAssgntPurch_L@1000 : Record 60130;
    BEGIN
      SalesLine2.CALCFIELDS("Qty. to Assign Acp","Qty. Assigned Acp");
      AssignableQty := SalesLine2."Qty. to Assign Acp";
      GetSalesHeader(SalesLine2);
      AssgntAmount := ROUND(SalesLine2."Line Amount",Currency_G."Amount Rounding Precision");
      AcpAssgntPurch_L.RESET;
      AcpAssgntPurch_L.SETCURRENTKEY("Document Type","Document No.","Document Line No.");
      AcpAssgntPurch_L.SETRANGE("Document Type","Document Type");
      AcpAssgntPurch_L.SETRANGE("Document No.","Document No.");
      AcpAssgntPurch_L.SETRANGE("Document Line No.","Document Line No.");
      AcpAssgntPurch_L.CALCSUMS("Qty. to Assign","Amount to Assign");
      TotalQtyToAssign := AcpAssgntPurch_L."Qty. to Assign";
      TotalAmountToAssign := AcpAssgntPurch_L."Amount to Assign";

      RemQtyToAssign := AssignableQty - TotalQtyToAssign;
      RemAmountToAssign := AssgntAmount - TotalAmountToAssign;
    END;

    LOCAL PROCEDURE UpdateQty@1();
    BEGIN
      CASE "Applies-to Doc. Type" OF
        "Applies-to Doc. Type"::Order,"Applies-to Doc. Type"::Invoice:
          BEGIN
            PurchLine_L.GET("Applies-to Doc. Type","Applies-to Doc. No.","Applies-to Doc. Line No.");
            QtyToReceiveBase := PurchLine_L."Qty. to Receive (Base)";
            QtyReceivedBase := PurchLine_L."Qty. Received (Base)";
            QtyToShipBase := 0;
            QtyShippedBase := 0;
          END;
        "Applies-to Doc. Type"::"Return Order","Applies-to Doc. Type"::"Credit Memo":
          BEGIN
            PurchLine_L.GET("Applies-to Doc. Type","Applies-to Doc. No.","Applies-to Doc. Line No.");
            QtyToReceiveBase := 0;
            QtyReceivedBase := 0;
            QtyToShipBase := PurchLine_L."Return Qty. to Ship (Base)";
            QtyShippedBase := PurchLine_L."Return Qty. Shipped (Base)";
          END;
        "Applies-to Doc. Type"::Receipt:
          BEGIN
            PurchRcptLine.GET("Applies-to Doc. No.","Applies-to Doc. Line No.");
            QtyToReceiveBase := 0;
            QtyReceivedBase := PurchRcptLine."Quantity (Base)";
            QtyToShipBase := 0;
            QtyShippedBase := 0;
          END;
        "Applies-to Doc. Type"::"Return Shipment":
          BEGIN
            ReturnShptLine.GET("Applies-to Doc. No.","Applies-to Doc. Line No.");
            QtyToReceiveBase := 0;
            QtyReceivedBase := 0;
            QtyToShipBase := 0;
            QtyShippedBase := ReturnShptLine."Quantity (Base)";
          END;
        "Applies-to Doc. Type"::"Transfer Receipt":
          BEGIN
            TransferRcptLine.GET("Applies-to Doc. No.","Applies-to Doc. Line No.");
