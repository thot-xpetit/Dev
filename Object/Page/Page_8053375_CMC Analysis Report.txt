    PROCEDURE CreateCaptionSet@29(RecRef_P@8053005 : Record 8053364;Step_P@8053004 : 'First,Previous,Same,Next';MaximumNoOfCaptions_P@8053003 : Integer;VAR PrimaryKeyFirstCaptionInCurrSe@8053002 : Text[1024];VAR CaptionSet_P@8053001 : ARRAY [32] OF Text[1024];VAR CaptionRange_P@8053000 : Text[1024]) : Integer;

      CASE Step_P OF
        Step_P::First:
          BEGIN
            IF (ColumnOption_G = ColumnOption_G::Period) AND (DateFilter_G = '') THEN
              FindRec(ColumnOption_G,RecRef_P,'=><')
          ELSE
            IF NOT FindRec(ColumnOption_G,RecRef_P,'-') THEN
              EXIT;
          END;
        Step_P::Previous:
          BEGIN
            RecRef_P.SETPOSITION(PrimaryKeyFirstCaptionInCurrSe);
            IF ColumnOption_G = ColumnOption_G::Period THEN
              EVALUATE(RecRef_P."Period Start",RecRef_P.Code);
            IF NOT FindRec(ColumnOption_G,RecRef_P,'=') THEN
              EXIT;
            NextRec(ColumnOption_G,RecRef_P,-MaximumNoOfCaptions_P);
          END;
        Step_P::Same:
          BEGIN
            RecRef_P.SETPOSITION(PrimaryKeyFirstCaptionInCurrSe);
            IF ColumnOption_G = ColumnOption_G::Period THEN
              EVALUATE(RecRef_P."Period Start",RecRef_P.Code);
            IF NOT FindRec(ColumnOption_G,RecRef_P,'=') THEN
              EXIT;
          END;
        Step_P::Next:
          BEGIN
            RecRef_P.SETPOSITION(PrimaryKeyFirstCaptionInCurrSe);
            IF ColumnOption_G = ColumnOption_G::Period THEN
              EVALUATE(RecRef_P."Period Start",RecRef_P.Code);
            IF NOT FindRec(ColumnOption_G,RecRef_P,'=') THEN
              EXIT;
            IF NOT (NextRec(ColumnOption_G,RecRef_P,MaximumNoOfCaptions_P) = MaximumNoOfCaptions_P) THEN BEGIN
              RecRef_P.SETPOSITION(PrimaryKeyFirstCaptionInCurrSe);
              FindRec(ColumnOption_G,RecRef_P,'=');
            END;
          END;
      END;

      PrimaryKeyFirstCaptionInCurrSe := RecRef_P.GETPOSITION;

      REPEAT
        CurrentCaptionOrdinal_L := CurrentCaptionOrdinal_L + 1;
        IF ShowColumnName_G THEN
          CaptionSet_P[CurrentCaptionOrdinal_L] := RecRef_P.Description
        ELSE
          CaptionSet_P[CurrentCaptionOrdinal_L] := RecRef_P.Code;
      UNTIL (CurrentCaptionOrdinal_L = MaximumNoOfCaptions_P) OR (NextRec(ColumnOption_G,RecRef_P,1) <> 1);

      IF CurrentCaptionOrdinal_L = 1 THEN
        CaptionRange_P := CaptionSet_P[1]
      ELSE
        CaptionRange_P := CaptionSet_P[1] + '..' + CaptionSet_P[CurrentCaptionOrdinal_L];
    END;

    LOCAL PROCEDURE FindPeriod@2(SearchText_P@8053000 : Code[10]);

      IF DateFilter_G <> '' THEN BEGIN

          PeriodFormMgt_L.FindDate('+',Calendar_L,PeriodType_G::Day);

      AnalysisReportEntry_G.SETRANGE(Date,Calendar_L."D'but p'riode",Calendar_L."Fin p'riode");

      InternalDateFilter_G := AnalysisReportEntry_G.GETFILTER(Date);

    END;
