    PROCEDURE CalcAmountApplied@1120002();
    VAR
      GLEntry_L@1120000 : Record 17;
    BEGIN
      ApplnCode_G := Letter;
      DebitApplied_G := 0;
      CreditApplied_G :=0;

      IF Letter <> '' THEN BEGIN
        GLEntry_L.RESET;
        GLEntry_L.SETCURRENTKEY("G/L Account No.",Letter);
        GLEntry_L.SETRANGE("G/L Account No.","G/L Account No.");
        GLEntry_L.SETRANGE(Letter,Letter);
        IF GLEntry_L.FIND('-') THEN
          REPEAT
            DebitApplied_G := DebitApplied_G + GLEntry_L."Debit Amount";
            CreditApplied_G := CreditApplied_G + GLEntry_L."Credit Amount";
          UNTIL GLEntry_L.NEXT = 0;
      END ELSE BEGIN
        DebitApplied_G := "Debit Amount";
        CreditApplied_G := "Credit Amount";
      END;
    END;

    PROCEDURE CalcAmountInProgress@60001();
    VAR
      GLEntry_L@60001 : Record 17;
    BEGIN
      DebitInProgress_G := 0;
      CreditInProgress_G := 0;

      GLEntry_L.RESET;
      GLEntry_L.SETCURRENTKEY("G/L Account No.","Applies-to ID");
      GLEntry_L.SETRANGE("G/L Account No.","G/L Account No.");
      GLEntry_L.SETRANGE("Applies-to ID",GetEntryApplID);
      IF GLEntry_L.FIND('-') THEN BEGIN
        REPEAT
          DebitInProgress_G := DebitInProgress_G + GLEntry_L."Debit Amount";
          CreditInProgress_G := CreditInProgress_G + GLEntry_L."Credit Amount";
        UNTIL GLEntry_L.NEXT = 0;
      END;
    END;

    PROCEDURE UpdateFilters@60002();
    BEGIN
      IF GLAccFilter_G <> '' THEN
        SETFILTER("G/L Account No.",GLAccFilter_G)
      ELSE
        SETRANGE("G/L Account No.");

      CASE AppliedEntryFilter_G OF
        AppliedEntryFilter_G :: All         : SETRANGE(Applied);
        AppliedEntryFilter_G :: Applied     : SETRANGE(Applied,TRUE);
        AppliedEntryFilter_G :: "Not Applied" : SETRANGE(Applied,FALSE);
      END;
    END;

    PROCEDURE UpdateAmountFilter@60003();
    VAR
      DummyTxt@60000 : Text[250];
    BEGIN
      IF AmountFilter_G <> '' THEN BEGIN
        RankingType_G := RankingType_G::Amount;
        UpdateRanking;
        SETFILTER(Amount,AmountFilter_G);
        IF GETRANGEMIN(Amount) = GETRANGEMAX(Amount) THEN
          DummyTxt := STRSUBSTNO('%1|-%1',ABS(GETRANGEMIN(Amount)))
        ELSE
          DummyTxt := STRSUBSTNO('-%2..-%1|%1..%2',ABS(GETRANGEMIN(Amount)),ABS(GETRANGEMAX(Amount)));
        SETFILTER(Amount,DummyTxt);
      END ELSE
        SETRANGE(Amount);
    END;

    END;
