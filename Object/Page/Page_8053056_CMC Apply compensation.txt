    PROCEDURE CalcApplnRemainingAmount@4(Amount@1000 : Decimal) : Decimal;
    VAR
      ApplnRemainingAmount@1001 : Decimal;
    BEGIN
      ValidExchRate := TRUE;
      IF ApplnCurrencyCode = "Currency Code" THEN
        EXIT(Amount)
      ELSE BEGIN
        IF ApplnDate = 0D THEN
          ApplnDate := "Posting Date";
        ApplnRemainingAmount :=
          CurrExchRate.ApplnExchangeAmtFCYToFCY(
            ApplnDate,"Currency Code",ApplnCurrencyCode,Amount,ValidExchRate);
        EXIT(ApplnRemainingAmount);
      END;
    END;

    PROCEDURE CalcApplnAmounttoApply@11(AmounttoApply@1000 : Decimal) : Decimal;
    VAR
      ApplnAmountToApply@1001 : Decimal;
    BEGIN
      ValidExchRate := TRUE;

      IF ApplnCurrencyCode = "Currency Code" THEN
        EXIT(AmounttoApply)
      ELSE BEGIN
        IF ApplnDate = 0D THEN
          ApplnDate := "Posting Date";
        ApplnAmountToApply :=
          CurrExchRate.ApplnExchangeAmtFCYToFCY(
            ApplnDate,"Currency Code",ApplnCurrencyCode,AmounttoApply,ValidExchRate);
        EXIT(ApplnAmountToApply);
      END;
    END;

    PROCEDURE FindAmountRounding@6();
    BEGIN
      IF ApplnCurrencyCode = '' THEN BEGIN
        Currency.INIT;
        Currency.Code := '';
        Currency.InitRoundingPrecision;
      END ELSE
        IF ApplnCurrencyCode <> Currency.Code THEN
          Currency.GET(ApplnCurrencyCode);

      AmountRoundingPrecision := Currency."Amount Rounding Precision";
    END;

    PROCEDURE CheckRounding@3();
    BEGIN
      ApplnRounding := 0;

      IF ApplnCurrencyCode = '' THEN
        ApplnRoundingPrecision := GLSetup."Appln. Rounding Precision"
      ELSE BEGIN
        IF ApplnCurrencyCode <> "Currency Code" THEN
          Currency.GET(ApplnCurrencyCode);
        ApplnRoundingPrecision := Currency."Appln. Rounding Precision";
      END;

      IF (ABS((AppliedAmount - PmtDiscAmount) + ApplyingAmount) <= ApplnRoundingPrecision) AND DifferentCurrenciesInAppln THEN
        ApplnRounding := -((AppliedAmount - PmtDiscAmount) + ApplyingAmount);
    END;

    PROCEDURE GetVendLedgEntry@5(VAR VendLedgEntry@1000 : Record 8053020);
    BEGIN
      VendLedgEntry := Rec;
    END;

    PROCEDURE FindApplyingEntry@12();
    BEGIN
        VendEntryApplID := USERID;
        IF VendEntryApplID = '' THEN
          VendEntryApplID := '***';

        IF VendLedgEntry.GET("Original Ledger Entry No.") THEN BEGIN
          VendLedgEntry.CALCFIELDS(Amount,"Remaining Amount");
          ApplyingVendLedgEntry := VendLedgEntry;
          SETCURRENTKEY("Entry No.");
          SETFILTER("Original Ledger Entry No.",'<> %1',VendLedgEntry."Entry No.");
          ApplyingAmount := VendLedgEntry."Remaining Amount";
          ApplnDate := VendLedgEntry."Posting Date";
          ApplnCurrencyCode := VendLedgEntry."Currency Code";
        END;
    END;

    PROCEDURE UpdateRealLedger@8053001(VAR P_CMCSettOffLedger@8053000 : TEMPORARY Record 8053020);

      L_Rec_VendLedger@8053001 : Record 25;
      L_Rec_CustLedger@8053002 : Record 21;
      L_Rec_CMCDebt@8053003 : Record 8053017;
    BEGIN
      //L_Rec_VendLedger.LOCKTABLE;
      //L_Rec_CustLedger.LOCKTABLE;
      //L_Rec_CMCDebt.LOCKTABLE;
      P_CMCSettOffLedger.FINDFIRST;
      REPEAT
       IF P_CMCSettOffLedger."SettOff Amount to apply"<>P_CMCSettOffLedger."Amount to Apply" THEN
        BEGIN
         CASE P_CMCSettOffLedger."Ledger type" OF
           P_CMCSettOffLedger."Ledger type"::Vendor:
            BEGIN
             L_Rec_VendLedger.GET(P_CMCSettOffLedger."Original Ledger Entry No.");

            END;
           P_CMCSettOffLedger."Ledger type"::Customer:
            BEGIN
             L_Rec_CustLedger.GET(P_CMCSettOffLedger."Original Ledger Entry No.");
             L_Rec_CustLedger.TESTFIELD(L_Rec_CustLedger.Open,TRUE);
             L_Rec_CustLedger."Amount to Apply" := P_CMCSettOffLedger."Amount to Apply";
             L_Rec_CustLedger."SettOff Applies-to ID" := P_CMCSettOffLedger."SettOff Applies-to ID";
             L_Rec_CustLedger.MODIFY;
            END;
           P_CMCSettOffLedger."Ledger type"::Vendor:

    PROCEDURE CheckOriginalLedger@8053000() : Boolean;
    VAR
      L_Rec_VendorLedger@8053000 : Record 25;
      L_Rec_CustomerLedger@8053001 : Record 21;
      L_Rec_Debt@8053002 : Record 8053017;
    BEGIN
      L_Rec_VendorLedger.GET(G_REC_CMCSettOffToApply."Original Ledger Entry No.");
      L_Rec_VendorLedger.CALCFIELDS(L_Rec_VendorLedger."Remaining Amount",L_Rec_VendorLedger.Amount);
      IF (G_REC_CMCSettOffToApply.Open<>L_Rec_VendorLedger.Open) OR
         (G_REC_CMCSettOffToApply."Remaining Amount"<>L_Rec_VendorLedger."Remaining Amount") OR
         (L_Rec_VendorLedger."Applies-to ID"<>G_REC_CMCSettOffToApply."Applies-to ID") OR
         (G_REC_CMCSettOffToApply."SettOff Applies-to ID"<>L_Rec_VendorLedger."SettOff Applies-to ID") THEN

        EXIT(FALSE);

      RESET;
      IF FIND('-') THEN
       REPEAT
        CASE "Ledger type" OF
         "Ledger type"::Vendor:
          BEGIN
           L_Rec_VendorLedger.GET("Original Ledger Entry No.");
           L_Rec_VendorLedger.CALCFIELDS("Remaining Amount");
           IF (Open<>L_Rec_VendorLedger.Open) OR
              ("Remaining Amount"<>L_Rec_VendorLedger."Remaining Amount") OR
              (L_Rec_VendorLedger."Applies-to ID"<>"Applies-to ID") THEN
            EXIT(FALSE);
          END;
         "Ledger type"::Customer:
          BEGIN
           L_Rec_CustomerLedger.GET("Original Ledger Entry No.");
           L_Rec_CustomerLedger.CALCFIELDS(Amount,"Remaining Amount");
           IF (L_Rec_CustomerLedger.Open<>Open) OR
              (L_Rec_CustomerLedger."Remaining Amount"<>"Remaining Amount") OR
              (L_Rec_CustomerLedger."Applies-to ID"<>"Applies-to ID") THEN
             EXIT(FALSE);
          END;
         "Ledger type"::"Debt Recognition":
          BEGIN
           L_Rec_Debt.GET("Document No.");
           L_Rec_Debt.CALCFIELDS(L_Rec_Debt."Amount reimbursed");
           IF (L_Rec_Debt."Species Code"<>"Species Code") OR
              ((L_Rec_Debt."Amount to payback"-L_Rec_Debt."Amount reimbursed")<>"Remaining Amount") THEN
             EXIT(FALSE);
          END;
        END;
       UNTIL NEXT=0;

      EXIT(TRUE);
    END;
