      CurrentRec@60000 : Text[40];
      LastRecord@60001 : Text[40];

      SurplusContract_G.RESET;
      SurplusContract_G.SETCURRENTKEY("Vendor No.","Item No.","Receipt No.","Receipt Line No.","Include on Contract","Is Header");
      SurplusContract_G.SETRANGE("Vendor No.","Vendor No.");

      IF (SurplusContract_G."Line No." = "Line No.") AND "Is Header" THEN BEGIN
        VendorIndent_G := 0;
        SurplusContract_G.SETRANGE("Is Header",FALSE);

        VendorIndent_G := 1;
