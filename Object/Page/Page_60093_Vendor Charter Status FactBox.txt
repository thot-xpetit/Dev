    PROCEDURE GetCampainNo@8053006(P_CampainType@8053000 : Code[10];P_StartingDate@8053001 : Date;P_EndingDate@8053002 : Date) Res_CampainNo : Code[10];
    VAR
      L_recCampaign@8053003 : Record 8052991;

      L_recCampaign.SETCURRENTKEY("Type Code","Starting Date","Ending Date");

      EXIT(Res_CampainNo);
    END;

    PROCEDURE NbrOfTrucksOnContracts@60001(RecCharterLines_P@60001 : Record 8053265;OpenPage_L@60004 : Boolean) TrucksNumber_Res : Integer;
    VAR
      RecCollContract_L@60000 : Record 8052983;
      TotQtyNonReceived_L@60002 : Decimal;
      RecCharter_L@60003 : Record 60013;
      Page_CollContract_L@60005 : Page 8053139;
      RecCollContract2_L@60006 : Record 8052983;
    BEGIN
      TrucksNumber_Res := 0;
      TotQtyNonReceived_L := 0;
      RecCharter_L.GET(RecCharterLines_P."Charter No");

      CLEAR(RecCollContract_L);
      RecCollContract_L.SETRANGE(Type,RecCollContract_L.Type::Contract);
      RecCollContract_L.SETRANGE(Status,RecCollContract_L.Status::Locked);
      RecCollContract_L.SETRANGE("Vendor No.",RecCharterLines_P."From Code");
      RecCollContract_L.SETRANGE("Campaign No.",GetCampainNo('',WORKDATE,WORKDATE));
      RecCollContract_L.SETRANGE("Species Code",ExitSpecieCode(RecCharterLines_P));
      RecCollContract_L.SETRANGE("Warning Date",0D,ExitAlertDate(RecCharterLines_P));
      IF RecCollContract_L.FINDSET THEN
      REPEAT
        RecCollContract_L.CALCFIELDS("Quantity Received");
        TotQtyNonReceived_L += RecCollContract_L.Quantity - RecCollContract_L."Quantity Received";
        IF RecCollContract_L.Quantity > RecCollContract_L."Quantity Received" THEN BEGIN
          RecCollContract2_L.GET(RecCollContract_L."No.");
          RecCollContract2_L.MARK(TRUE);
        END;
      UNTIL RecCollContract_L.NEXT = 0;
      IF RecCharter_L."Shipping Type Capacity" <> 0 THEN
        TrucksNumber_Res := ROUND((TotQtyNonReceived_L / RecCharter_L."Shipping Type Capacity"),1,'>');

      IF OpenPage_L THEN BEGIN
        RecCollContract2_L.MARKEDONLY(TRUE);
        CLEAR(Page_CollContract_L);
        Page_CollContract_L.SETTABLEVIEW(RecCollContract2_L);
        Page_CollContract_L.RUN;
      END;
      CLEAR(RecCollContract2_L);
    END;

    PROCEDURE NbrOfTrucksOnCharterLines@60004(RecCharterLines_P@60000 : Record 8053265) TrucksNumber_Res : Integer;
    VAR
      RecCharterLine_L@60001 : Record 8053265;
      RecCharter_L@60002 : Record 60013;
      RecLocation_L@60003 : Record 14;
      RecItem_L@60004 : Record 27;
      RecItemParam_L@60005 : Record 27;
    BEGIN
      TrucksNumber_Res := 0;
      RecItemParam_L.GET(RecCharterLines_P."Item No.");
      CLEAR(RecCharter_L);
      RecCharter_L.SETRANGE("Ending Date",0D,ExitAlertDate(RecCharterLines_P));
      IF RecCharter_L.FINDSET THEN
      REPEAT
        CLEAR(RecCharterLine_L);
        WITH RecCharterLine_L DO BEGIN
          SETRANGE("Charter No",RecCharter_L."No.");
          SETRANGE("From Type","From Type"::Vendor);
          SETRANGE("From Code",RecCharterLines_P."From Code");
          SETRANGE(Status,Status::Open);
          IF FINDSET THEN
          REPEAT
            RecItem_L.GET("Item No.");
            IF RecItem_L."Species Code" = RecItemParam_L."Species Code" THEN BEGIN
              IF ("From Type"= "From Type"::Vendor) AND ("Destination Type"= "Destination Type"::Customer) THEN
                TrucksNumber_Res += 1;
              IF "Destination Type"= "Destination Type"::Location THEN BEGIN
                IF RecLocation_L.GET("Destination Code") THEN
                  IF RecLocation_L.Port = RecLocation_L.Port::"Maritime " THEN
                    TrucksNumber_Res += 1;
              END;
            END;
          UNTIL NEXT = 0;
        END;
      UNTIL RecCharter_L.NEXT = 0;
      EXIT(TrucksNumber_Res);
    END;
