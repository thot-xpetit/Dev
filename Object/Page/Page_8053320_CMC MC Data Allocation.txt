              FieldRef_L := RecRef_G.FIELD(5050);

                ThirdMsIsCompany_L := TRUE;

                END;
              END ELSE BEGIN
                "Allocate vendor"  := FALSE;
                "Allocate customer":= FALSE;
                Allocate :=  MCDataRelation_G.GET(Code,MCDataRelation_G.Type::"Third party",FieldRef_G.VALUE);
              END;
            END;
          END;
        UNTIL RecRef_G.NEXT = 0;
      END ELSE BEGIN
        Allocate := FALSE;
        "Allocate vendor"  := FALSE;
        "Allocate customer":= FALSE;
        FieldRef_G := RecRef_G.FIELDINDEX(1);
        CASE TableNo_G OF
          DATABASE::"CMC MC G/L Account":
            Allocate :=  MCDataRelation_G.GET(Code,MCDataRelation_G.Type::"G/L Account",FieldRef_G.VALUE);
          DATABASE::"CMC MC Item":

            Allocate := MCDataRelation_G.GET(Code,MCDataRelation_G.Type::"Production Header",FieldRef_G.VALUE);
          DATABASE::"CMC MC Third party": BEGIN
            FieldRef_L := RecRef_G.FIELD(5050);

              ThirdMsIsCompany_L := TRUE;
              Allocate := FALSE;
              IF MCDataRelation_G.GET(Code,MCDataRelation_G.Type::"Third party",FieldRef_G.VALUE) THEN BEGIN
                "Allocate vendor"  := MCDataRelation_G."Is vendor";
                "Allocate customer":= MCDataRelation_G."Is Customer";
              END ELSE BEGIN
                "Allocate vendor"  := FALSE;
                "Allocate customer":= FALSE;
              END;
            END ELSE BEGIN
              "Allocate vendor"  := FALSE;
              "Allocate customer":= FALSE;
              Allocate := MCDataRelation_G.GET(Code,MCDataRelation_G.Type::"Third party",FieldRef_G.VALUE) ;
            END;
            //ajoute les propositions envoy'es par le connecteur CRM
            IF NOT "Allocate vendor" OR  NOT "Allocate customer"  THEN BEGIN
              IF cduSynchroCRM_L.ModifyProposition(RecRef_G, Code, "Company Name") THEN BEGIN

        ToggleCompanySelect;
