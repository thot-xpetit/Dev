    PROCEDURE SetVendApplId@10();
    VAR
      Balance_L@8053000 : Decimal;
      RestraintAmount_L@8053001 : Decimal;
      RestraintEntry_L@8053002 : Record 8053374;
    BEGIN
      CLEAR(Balance_L);
      Balance_L := AppliedAmount;
      LOCKTABLE;
        IF "SettOff Applies-to ID" <> '' THEN
          VendEntryApplID := ''
        ELSE
          VendEntryApplID := "IC Partner Code";

         TESTFIELD(Open,TRUE);

      CLEAR(RestraintAmount_L);
          IF ("SettOff Applies-to ID" <> '')THEN  // on supprime la lettre
            BEGIN
             "SettOff Applies-to ID" := '';
             AppliedAmount -= "Amount to Apply";
              "Setting Off Remain" += "Amount to Apply";
             "Amount to Apply" := 0;
             CMCSettOffToApplyTMP_G.GET("Entry No.");
             CMCSettOffToApplyTMP_G."Amount to Apply" := 0;
             CMCSettOffToApplyTMP_G.MODIFY;

            END ELSE //on lettre
              BEGIN
               "SettOff Applies-to ID" := VendEntryApplID;
               {
               IF "Ledger type" = "Ledger type"::Restraint THEN
                BEGIN
                  RestraintEntry_L.GET("Original Ledger Entry No.");
                  CASE RestraintEntry_L."Tiers Type" OF
                    RestraintEntry_L."Tiers Type"::Client:
                      RestraintAmount_L := - "Remaining Amount";
                    RestraintEntry_L."Tiers Type"::Fournisseur:
                      RestraintAmount_L := "Remaining Amount";
                    RestraintEntry_L."Tiers Type"::"0":
                      EXIT;
                  END;
                  IF (Balance_L*RestraintAmount_L) >0 THEN
                    EXIT;
                  IF (((Balance_L*RestraintAmount_L)<0) = (((Balance_L+RestraintAmount_L)*Balance_L)<0))THEN
                   BEGIN
                    "Amount to Apply":= ABS(Balance_L);
                    AppliedAmount += "Amount to Apply";
                   END
                  ELSE BEGIN
                    "Amount to Apply" := "Remaining Amount"-"Other SettOff Amount";
                    AppliedAmount += "Amount to Apply";
                  END;
                END ELSE BEGIN
                  "Amount to Apply" := "Remaining Amount"-"Other SettOff Amount";
                  AppliedAmount += "Amount to Apply";
                 END;
                 }

                CurTotAmt_G := 0;
                CMCSettOffToApplyTMP_G.RESET;
                IF CMCSettOffToApplyTMP_G.FINDSET THEN
                  REPEAT
                    CurTotAmt_G += CMCSettOffToApplyTMP_G."Amount to Apply"
                  UNTIL CMCSettOffToApplyTMP_G.NEXT = 0 ;

                IF (CurTotAmt_G + "Setting Off Remain") >= "SettOff Amount to apply" THEN BEGIN
                  "Amount to Apply" +=  "SettOff Amount to apply" - CurTotAmt_G;
                  "Setting Off Remain" := "Setting Off Remain" - ("SettOff Amount to apply" - CurTotAmt_G) ;
                  CMCSettOffToApplyTMP_G.GET("Entry No.");
                  CMCSettOffToApplyTMP_G."Amount to Apply" := "Amount to Apply";
                  CMCSettOffToApplyTMP_G.MODIFY;
                END ELSE BEGIN
                  "Amount to Apply" += "Setting Off Remain";
                  "Setting Off Remain" := 0;
                  CMCSettOffToApplyTMP_G.GET("Entry No.");
                  CMCSettOffToApplyTMP_G."Amount to Apply" := "Amount to Apply";
                  CMCSettOffToApplyTMP_G.MODIFY;
                END;
              END;
          MODIFY;
    END;

    PROCEDURE CalcApplnRemainingAmount@4(Amount@1000 : Decimal) : Decimal;
    VAR
      ApplnRemainingAmount@1001 : Decimal;
    BEGIN
      ValidExchRate := TRUE;
      IF ApplnCurrencyCode = "Currency Code" THEN
        EXIT(Amount)
      ELSE BEGIN
        IF ApplnDate = 0D THEN
          ApplnDate := "Posting Date";
        ApplnRemainingAmount :=
          CurrExchRate.ApplnExchangeAmtFCYToFCY(
            ApplnDate,"Currency Code",ApplnCurrencyCode,Amount,ValidExchRate);
        EXIT(ApplnRemainingAmount);
      END;
    END;

    PROCEDURE CalcApplnAmounttoApply@11(AmounttoApply@1000 : Decimal) : Decimal;
    VAR
      ApplnAmountToApply@1001 : Decimal;
    BEGIN
      ValidExchRate := TRUE;

      IF ApplnCurrencyCode = "Currency Code" THEN
        EXIT(AmounttoApply)
      ELSE BEGIN
        IF ApplnDate = 0D THEN
          ApplnDate := "Posting Date";
        ApplnAmountToApply :=
          CurrExchRate.ApplnExchangeAmtFCYToFCY(
            ApplnDate,"Currency Code",ApplnCurrencyCode,AmounttoApply,ValidExchRate);
        EXIT(ApplnAmountToApply);
      END;
    END;

    PROCEDURE FindAmountRounding@6();
    BEGIN
      IF ApplnCurrencyCode = '' THEN BEGIN
        Currency.INIT;
        Currency.Code := '';
        Currency.InitRoundingPrecision;
      END ELSE
        IF ApplnCurrencyCode <> Currency.Code THEN
          Currency.GET(ApplnCurrencyCode);

      AmountRoundingPrecision := Currency."Amount Rounding Precision";
    END;

    PROCEDURE CheckRounding@3();
    BEGIN
      ApplnRounding := 0;

      IF ApplnCurrencyCode = '' THEN
        ApplnRoundingPrecision := GLSetup."Appln. Rounding Precision"
      ELSE BEGIN
        IF ApplnCurrencyCode <> "Currency Code" THEN
          Currency.GET(ApplnCurrencyCode);
        ApplnRoundingPrecision := Currency."Appln. Rounding Precision";
      END;

      IF (ABS((AppliedAmount - PmtDiscAmount) + ApplyingAmount) <= ApplnRoundingPrecision) AND DifferentCurrenciesInAppln THEN
        ApplnRounding := -((AppliedAmount - PmtDiscAmount) + ApplyingAmount);
    END;

    PROCEDURE UpdateRealLedger@8053001(VAR P_CMCSettOffLedger@8053000 : TEMPORARY Record 8053020);
    VAR
      L_Rec_VendLedger@8053001 : Record 25;
      L_Rec_CustLedger@8053002 : Record 21;
      RestraintEntry_L@8053003 : Record 8053374;
      DtldVendLedgEntry_L@8053005 : Record 380;
      DtldCustLedgEntry_L@8053004 : Record 379;
    BEGIN
      P_CMCSettOffLedger.FINDFIRST;
      REPEAT
       IF P_CMCSettOffLedger."SettOff Amount to apply"<>P_CMCSettOffLedger."Amount to Apply" THEN
        BEGIN
         CASE P_CMCSettOffLedger."Ledger type" OF
           P_CMCSettOffLedger."Ledger type"::Vendor:
            BEGIN
             L_Rec_VendLedger.GET(P_CMCSettOffLedger."Original Ledger Entry No.");
             L_Rec_VendLedger.TESTFIELD(L_Rec_VendLedger.Open,TRUE);
             L_Rec_VendLedger."Amount to Apply" := P_CMCSettOffLedger."Amount to Apply";
             L_Rec_VendLedger."SettOff Applies-to ID" := P_CMCSettOffLedger."SettOff Applies-to ID";

             L_Rec_VendLedger.MODIFY;
             DtldVendLedgEntry_L.RESET;
             DtldVendLedgEntry_L.SETRANGE("Vendor Ledger Entry No.",L_Rec_VendLedger."Entry No.");
             DtldVendLedgEntry_L.MODIFYALL("SettOff Applies-to ID",P_CMCSettOffLedger."SettOff Applies-to ID");

            END;
           P_CMCSettOffLedger."Ledger type"::Customer:
            BEGIN
             L_Rec_CustLedger.GET(P_CMCSettOffLedger."Original Ledger Entry No.");
             L_Rec_CustLedger.TESTFIELD(L_Rec_CustLedger.Open,TRUE);
             L_Rec_CustLedger."Amount to Apply" := P_CMCSettOffLedger."Amount to Apply";
             L_Rec_CustLedger."SettOff Applies-to ID" := P_CMCSettOffLedger."SettOff Applies-to ID";
             L_Rec_CustLedger."Setting Off not Allowed" := P_CMCSettOffLedger."Setting Off not Allowed";
             L_Rec_CustLedger.MODIFY;
             DtldCustLedgEntry_L.RESET;
             DtldCustLedgEntry_L.SETRANGE("Cust. Ledger Entry No.",L_Rec_CustLedger."Entry No.");
             DtldCustLedgEntry_L.MODIFYALL("SettOff Applies-to ID",P_CMCSettOffLedger."SettOff Applies-to ID");
            END;
           P_CMCSettOffLedger."Ledger type"::Restraint:
            BEGIN
             RestraintEntry_L.GET(P_CMCSettOffLedger."Original Ledger Entry No.");
             RestraintEntry_L."Amount to Apply" := P_CMCSettOffLedger."Amount to Apply";
             RestraintEntry_L."SettOff Applies-to ID" := P_CMCSettOffLedger."SettOff Applies-to ID";
             RestraintEntry_L.MODIFY;

            END;
          END;
        END;
      UNTIL P_CMCSettOffLedger.NEXT=0;
      P_CMCSettOffLedger.DELETEALL;
    END;
