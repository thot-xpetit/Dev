    PROCEDURE MarkSelection@8053000();
    VAR
      ECDLedgerEntry_L@8053001 : Record 8053088;
      MarkValue_L@8053000 : Boolean;
    BEGIN
      TotalECDNetUnitPrice_G := 0;

      CurrPage.SETSELECTIONFILTER(ECDLedgerEntry_L);

      IF ECDLedgerEntry_L.FIND('-') THEN BEGIN
        // On complSte le marquage si dans la s'lection il y a une ligne non marqu'e
        MarkValue_L := TRUE;
        GET(ECDLedgerEntry_L."Entry No.");
        MarkValue_L := MarkValue_L AND MARK;
        WHILE ((ECDLedgerEntry_L.NEXT <> 0) AND (MarkValue_L)) DO BEGIN
          GET(ECDLedgerEntry_L."Entry No.");
          MarkValue_L := MarkValue_L AND MARK;
        END;

        IF ECDLedgerEntry_L.FIND('-') THEN BEGIN
          REPEAT
            IF GET(ECDLedgerEntry_L."Entry No.") THEN
              MARK(NOT MarkValue_L);
          UNTIL ECDLedgerEntry_L.NEXT = 0;
        END;

      END ELSE
        MARK(NOT MARK);

      UpdateCumulValue;
    END;
