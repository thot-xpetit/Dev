    LOCAL PROCEDURE AnalyzeData@1000000004();
    VAR
      TempExcelBuf@1000 : TEMPORARY Record 370;
      HeaderRowNo@1003 : Integer;
      OldRowNo@1006 : Integer;
      Window@1000000004 : Dialog;
      RecNo@1000000003 : Integer;
      TotalRecNo@1000000002 : Integer;
      RowNo@1000000001 : Integer;
      ColumnNo@1000000000 : Integer;
    BEGIN
      Window.OPEN(
        Text000 +
        '@1@@@@@@@@@@@@@@@@@@@@@@@@@\');
      Window.UPDATE(1,0);
      TotalRecNo := TempExcelBuffer.COUNT;
      RecNo := 0;

      IF TempExcelBuffer.FIND('-') THEN BEGIN
        REPEAT
          RecNo := RecNo + 1;
          Window.UPDATE(1,ROUND(RecNo / TotalRecNo * 10000,1));
          CASE TRUE OF
            (STRPOS(TempExcelBuffer."Cell Value as Text",ItemToExport.FIELDCAPTION("No.")) <> 0):
              BEGIN
                IF HeaderRowNo = 0 THEN BEGIN
                  HeaderRowNo := TempExcelBuffer."Row No.";
                  TempExcelBuf := TempExcelBuffer;
                  TempExcelBuf.Comment := ItemToExport.FIELDCAPTION("No.");
                  TempExcelBuf.INSERT;
                END ELSE BEGIN
                  IF (TempExcelBuffer."Cell Value as Text" = ItemToExport.FIELDCAPTION("No.")) THEN BEGIN
                    ERROR(STRSUBSTNO(TErr001,ItemToExport.FIELDCAPTION("No.")));
                  END;
                END;
              END;

            TempExcelBuffer."Row No." = HeaderRowNo:
              BEGIN
                TempExcelBuf := TempExcelBuffer;
                TempExcelBuf.Comment := TempExcelBuffer."Cell Value as Text";
                TempExcelBuf.INSERT;
              END;

            (TempExcelBuffer."Row No." > HeaderRowNo) AND (HeaderRowNo > 0):
              BEGIN
                IF TempExcelBuffer."Row No." <> OldRowNo THEN BEGIN
                  OldRowNo := TempExcelBuffer."Row No.";
                  ClearVariable();
                END;

                TempExcelBuf.SETRANGE("Column No.",TempExcelBuffer."Column No.");
                IF TempExcelBuf.FIND('-') THEN BEGIN
                  CASE TempExcelBuf.Comment OF
                    ItemToExport.FIELDCAPTION("No."):
                      BEGIN
                        ItemNo := TempExcelBuffer."Cell Value as Text";
                      END;
                    ItemToExport.FIELDCAPTION(Description):
                      BEGIN
                        ItemDescription := TempExcelBuffer."Cell Value as Text";
                      END;
                    ItemToExport.FIELDCAPTION("Blocked For Sales"): BEGIN
                      EVALUATE(ItemBlocked, TempExcelBuffer."Cell Value as Text");
                      UpdateItem();
                    END;
                  END;
                END;
              END;
          END;
        UNTIL TempExcelBuffer.NEXT = 0;
      END;

      Window.CLOSE;
    END;

    PROCEDURE ClearVariable@1000000008();

      ItemBlocked := FALSE;
    END;

        END;
      END;
    END;

    BEGIN
    END.
