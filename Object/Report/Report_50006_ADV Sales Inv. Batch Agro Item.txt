    PROCEDURE VerifyDimAnalogy@3() : Boolean;
    BEGIN
      IF NOT G_recFromDocDim.FIND('-') THEN BEGIN
        G_recFromDocDim.INIT;
        G_recFromDocDim."Dimension Code" := '';
      END;

      G_recToDocDim.SETRANGE("Table ID",DATABASE::"Sales Header");
      G_recToDocDim.SETRANGE("Document Type",G_recSalesHeader."Document Type");
      G_recToDocDim.SETRANGE("Document No.",G_recSalesHeader."No.");
      IF NOT G_recToDocDim.FIND('-') THEN BEGIN
        G_recToDocDim.INIT;
        G_recToDocDim."Dimension Code" := '';
      END;

      IF G_recFromDocDim.FIND('-') THEN
        REPEAT
          IF (G_recFromDocDim."Dimension Code" <> G_recToDocDim."Dimension Code") OR
             (G_recFromDocDim."Dimension Value Code" <> G_recToDocDim."Dimension Value Code")
          THEN
            EXIT(FALSE);
        UNTIL (G_recFromDocDim.NEXT = 0) AND (G_recToDocDim.NEXT = 0);

      EXIT(TRUE);
    END;
