    PROCEDURE SetUpNewLineLoc@9();

      CapJnlTemplate_L@8061300 : Record 8053354;
      CapJnlBatch_L@8061301 : Record 8053355;
      CapJnlLine_L@8061302 : Record 8053356;
    BEGIN
      CapJnlLine_G.VALIDATE("Capital Journal Template",BatchRec_G."Capital Journal Template");
      CapJnlLine_G.VALIDATE("Capital Batch Name",BatchRec_G.Name);
      CapJnlLine_L.SETRANGE("Capital Journal Template",BatchRec_G."Capital Journal Template");
      CapJnlLine_L.SETRANGE("Capital Batch Name",BatchRec_G.Name);

      IF CapJnlLine_L.FINDLAST THEN BEGIN
        LastLineNo_G := CapJnlLine_L."Line No." + 10000;
        IF CapJnlLine_L."Document No." <> '' THEN BEGIN
          CapJnlLine_G."Document No." := INCSTR(CapJnlLine_L."Document No.");
          IF CapJnlLine_G."Document No." = '' THEN
            CapJnlLine_G."Document No." := CapJnlLine_L."Document No.";
        END;
      END ELSE BEGIN
        LastLineNo_G := 10000;
        IF DocumentNo_G <> '' THEN
          CapJnlLine_G."Document No." := DocumentNo_G
        ELSE
          CapJnlLine_G."Document No." := NoSeriesMgt_L.TryGetNextNo(BatchRec_G."No. Series",WORKDATE);
      END;

      IF CapJnlLine_G."Document No." = '' THEN
        ERROR(Text60006);

      CapJnlLine_G."Capital Journal Template" := BatchRec_G."Capital Journal Template";
      CapJnlLine_G."Capital Batch Name" := BatchRec_G.Name;
      CapJnlLine_G."Line No." := LastLineNo_G;
      CapJnlLine_G."Source Code" := CapJnlTemplate_G."Source Code";
      CapJnlLine_G."Posting No. Series" := CapJnlBatch_G."Posting No. Series";
      CapJnlLine_G."G/L Posting No. Series" := CapJnlBatch_G."G/L Posting No. Series";
      CapJnlLine_G."Bal. Account Type" := CapJnlBatch_G."Bal. Account Type";
      IF CapJnlBatch_G."Bal. Account No." <> '' THEN
        CapJnlLine_G.VALIDATE("Bal. Account No.",CapJnlBatch_G."Bal. Account No.");
      CapJnlLine_G.VALIDATE("Posting Date",PostingDate_G);
    END;

    PROCEDURE CheckPartTyp@1000000001(PartTypeFrom_P@1000000000 : Code[10];PartTypeTo_P@1000000001 : Code[10];ClosingDate_P@1000000004 : Date) Value_R : Decimal;
    VAR
      PartTypeDetail_L@1000000002 : Record 8053352;
      PartTypeDetail2_L@1000000003 : Record 8053352;
    BEGIN
      //Les types de part doivent avoir la m^me valeur

      PartTypeDetail_L.RESET;
      PartTypeDetail_L.ASCENDING(FALSE);
      PartTypeDetail_L.SETFILTER("Effective Date",'..%1',ClosingDate_P);
      PartTypeDetail_L.SETRANGE("Part Code",PartTypeFrom_P);

        ERROR(Text60005,PartTypeFrom_P,PartTypeTo_P);

      PartTypeDetail_L.TESTFIELD("Unit Value");

      Value_R := PartTypeDetail_L."Unit Value";
    END;
