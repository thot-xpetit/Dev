    PROCEDURE CalcRendement@8053003();
    VAR
      Shipped_L@8053000 : Decimal;
      recPostReceiptLine_L@8053001 : Record 8053104;
      recDetails_L@8053002 : Record 8053210;
      recContrat_L@8053003 : Record 8052983;
      recTypeContrat_L@8053004 : Record 8053096;
    BEGIN
      Shipped_L := 0;
      // --> 20150914-Wi00-ADV01.01.00-2230-FHS01
      //BEFORE
      {
      recDetails_L.RESET;
      recDetails_L.SETRANGE("Contract No.", "CMC Frame Contract Line"."Contract No.");
      recDetails_L.SETRANGE("Contract Line No.", "CMC Frame Contract Line"."Line No.");
      IF recDetails_L.FINDSET(FALSE, FALSE) THEN
        REPEAT
          IF recContrat_L.GET(recDetails_L."Collection Contract No.") THEN BEGIN
            recTypeContrat_L.GET(recContrat_L."Contract Type");
            IF (recTypeContrat_L."Allow In Frame Contract" = TRUE) OR (recTypeContrat_L."Qte PF on CCC" = TRUE) THEN BEGIN
              recPostReceiptLine_L.RESET;
              // --> 20150605-WI00-CMC02.02.00-3045-FBE01
              //recPostReceiptLine_L.SETCURRENTKEY("Contract No.",Invoiced,"Canceled Line","Cancellation line");
              recPostReceiptLine_L.SETCURRENTKEY("Contract No.",Invoiced,"Canceled Line","Cancellation line","VAT Bus. Posting Group");
              // <-- 20150605-WI00-CMC02.02.00-3045-FBE01
              recPostReceiptLine_L.SETRANGE("Contract No.", recDetails_L."Collection Contract No.");
              recPostReceiptLine_L.SETRANGE("Canceled Line", FALSE);
              recPostReceiptLine_L.SETRANGE("Cancellation line", FALSE);
              IF recPostReceiptLine_L.FINDSET(FALSE, FALSE) THEN
                REPEAT
                  Shipped_L += recPostReceiptLine_L.Quantity;
                UNTIL recPostReceiptLine_L.NEXT = 0;

            END;
          END;
        UNTIL recDetails_L.NEXT = 0;
      }
      //AFTER
      recRendementTmp_G.RESET;
      recRendementTmp_G.SETRANGE("Contract No.", "CMC Frame Contract Line"."Contract No.");
      recRendementTmp_G.SETRANGE("Line No.", "CMC Frame Contract Line"."Line No.");
      IF recRendementTmp_G.FINDSET(FALSE, FALSE) THEN
        REPEAT
          Shipped_L += recRendementTmp_G."Quantity (T)";
        UNTIL recRendementTmp_G.NEXT = 0;
      // <-- 20150914-Wi00-ADV01.01.00-2230-FHS01
      //Somme des livraisons = ne pas reprendre les types de contrat " autoriser en contrat cadre = non " et " quantit' PF sur CCC=non " )
      //Formule du rendement Somme des livraisons / (surface - surface autoconsomm'e)
      IF ("CMC Frame Contract Line"."Area (Ha)" - "CMC Frame Contract Line"."Autoconsumed Surface (Ha)") = 0 THEN
        Rendement_G := 0
      ELSE
        Rendement_G := Shipped_L / ("CMC Frame Contract Line"."Area (Ha)" - "CMC Frame Contract Line"."Autoconsumed Surface (Ha)");
    END;
