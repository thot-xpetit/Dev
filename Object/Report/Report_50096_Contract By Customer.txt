    PROCEDURE GetTC@1000000001(Ctc@1000000000 : Record 5050) ReturnValue : Boolean;
    VAR
      ContactSalesPerson@1000000001 : Record 8053160;
    BEGIN
      ReturnValue := TRUE;
      ContactSalesPerson.SETRANGE("Contact Code", Ctc."No.");
      ContactSalesPerson.SETRANGE("Prod. Activity Code", 'PV');
      ContactSalesPerson.SETRANGE("Sales Network Code", 'PV');
      IF (SalesPersonCode <> '') THEN BEGIN
        ContactSalesPerson.SETFILTER("Salesperson Code", '%1', SalesPersonCode);
      END;
      IF (NOT ContactSalesPerson.FINDFIRST()) THEN BEGIN
        TC.INIT();
        ReturnValue := FALSE;
      END ELSE BEGIN
        IF (NOT TC.GET(ContactSalesPerson."Salesperson Code")) THEN BEGIN
          TC.INIT();
          ReturnValue := FALSE;
        END;
      END;
    END;

    PROCEDURE GetFiltersText@1000000000();
    BEGIN
      Filters := '';

        Filters := Filters + Customer.GETFILTERS() +  ' ';
      END;
      IF (Agriculteur.GETFILTERS() <> '') THEN BEGIN
        Filters := Filters + Agriculteur.GETFILTERS() +  ' ';
      END;
      IF ("CMC Sup. Frame Contract Header".GETFILTERS() <> '') THEN BEGIN
        Filters := Filters + "CMC Sup. Frame Contract Header".GETFILTERS() +  ' ';
      END;
      IF (Filters <> '') THEN BEGIN
        Filters := STRSUBSTNO(TFilters, Filters);
      END;
    END;

    PROCEDURE CheckFilters@1000000007();
    VAR
      ErrMsg@1000000000 : Text[1024];
    BEGIN
      ErrMsg := '';

        ErrMsg := ErrMsg + STRSUBSTNO(TErrFilter, "CMC Sup. Frame Contract Header".FIELDCAPTION("Frame Contract Type")) + '\'

        ERROR(ErrMsg);
      END;
    END;
