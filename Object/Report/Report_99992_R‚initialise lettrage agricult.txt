              recCustLedgerEntry_L.GET(CVLedgerEntry_L."Entry No.");

            CVLedgerEntry_L."Customer / Vendor"::Vendor:

              recVendorLedgerEntry_L.GET(CVLedgerEntry_L."Entry No.");
              recLedgerEntryTmp_L[1].Letter:=recVendorLedgerEntry_L.Letter;
            END;
          END;
          recLedgerEntryTmp_L[1].INSERT;
        UNTIL CVLedgerEntry_L.NEXT = 0;

        REPEAT

            recLedgerEntryTmp_L[1].SETRANGE(Letter,recLedgerEntryTmp_L[1].Letter);
            recLedgerEntryTmp_L[1].SETRANGE("Setting Off",TRUE);
            IF recLedgerEntryTmp_L[1].ISEMPTY THEN BEGIN // pas de compensation
              recLedgerEntryTmp_L[1].SETRANGE("Setting Off");
              recLedgerEntryTmp_L[1].MODIFYALL("CV Entry Letter",Letter_L);
              recLedgerEntryTmp_L[1].FINDLAST;
              recLedgerEntryTmp_L[1].SETFILTER(Letter,'<>%1','');

              recLedgerEntryTmp_L[1].FINDSET(FALSE,FALSE);
              REPEAT  // pour chaque ecr compens client
                recLedgerEntryTmp_L[2].RESET; // on cherche la compensation four
                recLedgerEntryTmp_L[2].SETRANGE("Customer / Vendor",recLedgerEntryTmp_L[2]."Customer / Vendor"::Vendor);
                recLedgerEntryTmp_L[2].SETRANGE("Cust./Vend. No.",ContactNo_P);

                      REPEAT
                        recLedgerEntryTmp_L[3]."CV Entry Letter":=Letter_L;
                        recLedgerEntryTmp_L[3].MODIFY;
                        IF (recLedgerEntryTmp_L[3]."Setting Off") AND (recLedgerEntryTmp_L[3]."Entry No."<>
                          recLedgerEntryTmp_L[2]."Entry No.")
                        THEN BEGIN
                          recLedgerEntryTmp_L[4].RESET;
                          recLedgerEntryTmp_L[4].SETRANGE("Customer / Vendor",recLedgerEntryTmp_L[4]."Customer / Vendor"::Customer);
                          recLedgerEntryTmp_L[4].SETRANGE("Cust./Vend. No.",ContactNo_P);
                          recLedgerEntryTmp_L[4].SETRANGE("Transaction No.",recLedgerEntryTmp_L[3]."Transaction No."-1);
                          recLedgerEntryTmp_L[4].SETRANGE("Document No.",recLedgerEntryTmp_L[3]."Document No." );
                          IF recLedgerEntryTmp_L[3].FINDSET THEN BEGIN
                            recLedgerEntryTmp_L[4].SETRANGE("Transaction No.");
                            recLedgerEntryTmp_L[4].SETRANGE("Document No.");
                            recLedgerEntryTmp_L[4].SETRANGE(Letter,recLedgerEntryTmp_L[4].Letter);
                            recLedgerEntryTmp_L[4].MODIFYALL("CV Entry Letter",Letter_L);
                          END;
                        END;
                      UNTIL recLedgerEntryTmp_L[3].NEXT=0;
                  UNTIL recLedgerEntryTmp_L[2].NEXT=0

                  recLedgerEntryTmp_L[2].SETRANGE("Transaction No.",recLedgerEntryTmp_L[1]."Transaction No." + 1);
                  recLedgerEntryTmp_L[2].SETRANGE("Document No.",recLedgerEntryTmp_L[1]."Document No.");

                    REPEAT
                      recLedgerEntryTmp_L[3].RESET;
                      recLedgerEntryTmp_L[3].SETCURRENTKEY("Customer / Vendor","Cust./Vend. No.",Letter);
                      recLedgerEntryTmp_L[3].SETRANGE("Customer / Vendor",recLedgerEntryTmp_L[3]."Customer / Vendor"::Vendor);
                      recLedgerEntryTmp_L[3].SETRANGE("Cust./Vend. No.",ContactNo_P);
                      recLedgerEntryTmp_L[3].SETRANGE(Letter,recLedgerEntryTmp_L[2].Letter);
                      IF recLedgerEntryTmp_L[3].FINDSET(FALSE,FALSE) THEN
                        REPEAT
                          recLedgerEntryTmp_L[3]."CV Entry Letter":=Letter_L;
                          recLedgerEntryTmp_L[3].MODIFY;
                          IF (recLedgerEntryTmp_L[3]."Setting Off") AND (recLedgerEntryTmp_L[3]."Entry No."<>
                            recLedgerEntryTmp_L[2]."Entry No.")
                          THEN BEGIN
                            recLedgerEntryTmp_L[4].RESET;
                            recLedgerEntryTmp_L[4].SETRANGE("Customer / Vendor",recLedgerEntryTmp_L[4]."Customer / Vendor"::Customer);
                            recLedgerEntryTmp_L[4].SETRANGE("Cust./Vend. No.",ContactNo_P);
                            recLedgerEntryTmp_L[4].SETRANGE("Transaction No.",recLedgerEntryTmp_L[3]."Transaction No."-1);

                            IF recLedgerEntryTmp_L[4].FINDSET(FALSE,FALSE) THEN
                              REPEAT
                                recLedgerEntryTmp_L[5].RESET;
                                recLedgerEntryTmp_L[5].SETCURRENTKEY("Customer / Vendor","Cust./Vend. No.",Letter);
                                recLedgerEntryTmp_L[5].SETRANGE("Customer / Vendor",
                                  recLedgerEntryTmp_L[3]."Customer / Vendor"::Customer);

                                        recLedgerEntryTmp_L[6].SETRANGE("Document No.");

                              UNTIL recLedgerEntryTmp_L[4].NEXT=0;

              UNTIL recLedgerEntryTmp_L[1].NEXT=0; //dernier ecr compen client

              recLedgerEntryTmp_L[1].MODIFYALL("CV Entry Letter",Letter_L); // maj des ecr compen client

              recLedgerEntryTmp_L[1].FINDLAST;
              recLedgerEntryTmp_L[1].SETFILTER(Letter,'<>%1','');
            END;
          cuGLEntryApplication_G.NextLetter(Letter_L);
          END;

      recLedgerEntryTmp_L[1].RESET;
