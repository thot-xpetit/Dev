    LOCAL PROCEDURE AnalyzeData@2();
    VAR
      TempExcelBuf@1000 : TEMPORARY Record 370;
      HeaderRowNo@1003 : Integer;
      OldRowNo@1006 : Integer;
    BEGIN
      Window.OPEN(
        Text007 +
        '@1@@@@@@@@@@@@@@@@@@@@@@@@@\');
      Window.UPDATE(1,0);
      TotalRecNo := ExcelBuf.COUNT;
      RecNo := 0;

      IF ExcelBuf.FIND('-') THEN BEGIN
        REPEAT
          RecNo := RecNo + 1;
          Window.UPDATE(1,ROUND(RecNo / TotalRecNo * 10000,1));
          CASE TRUE OF
            (ExcelBuf."Cell Value as Text"=Text009):
              BEGIN
                IF HeaderRowNo = 0 THEN BEGIN
                  HeaderRowNo := ExcelBuf."Row No.";
                  TempExcelBuf := ExcelBuf;
                  TempExcelBuf.Comment := Text009;
                  TempExcelBuf.INSERT;
                END ELSE
                  ERROR(Text011);
              END;

            ExcelBuf."Row No." = HeaderRowNo:
              BEGIN
                TempExcelBuf := ExcelBuf;
                TempExcelBuf.Comment := ExcelBuf."Cell Value as Text";
                TempExcelBuf.INSERT;
              END;

            (ExcelBuf."Row No." > HeaderRowNo) AND (HeaderRowNo > 0):
              BEGIN
                IF ExcelBuf."Row No." <> OldRowNo THEN BEGIN
                  UpdateData();
                  OldRowNo := ExcelBuf."Row No.";
                  ClearVariable();
                END;

                TempExcelBuf.SETRANGE("Column No.",ExcelBuf."Column No.");
                IF TempExcelBuf.FIND('-') THEN BEGIN
                  CASE TempExcelBuf.Comment OF
                    Text013: BEGIN
                      Code := ExcelBuf."Cell Value as Text";
                    END;
                    Text014: BEGIN
                      Type := ExcelBuf."Cell Value as Text";
                      IF (Type = 'Activit'') THEN BEGIN
                        TypeActivity := TypeActivity::Activity;
                      END;
                      IF (Type = 'Famille') THEN BEGIN
                        TypeActivity := TypeActivity::Family;
                      END;
                      IF (Type = 'Sous-famille') THEN BEGIN
                        TypeActivity := TypeActivity::Subfamily;
                      END;
                      IF (Type = 'EspSce') THEN BEGIN
                        TypeActivity := TypeActivity::Species;
                      END;
                      IF (Type = 'Article') THEN BEGIN
                        TypeActivity := TypeActivity::Item;
                      END;
                    END;
                    Text015:
                      No := ExcelBuf."Cell Value as Text";
                    Text016:
                      BEGIN
                      CodeVariante := ExcelBuf."Cell Value as Text";
                      END;
                    Text017:
                      CodeLivraison := ExcelBuf."Cell Value as Text";
                    Text018:
                      CodeConditionnement := ExcelBuf."Cell Value as Text";
                    Text019:
                      CodeScenario := ExcelBuf."Cell Value as Text";
                    Text021:
                      ProcedureAchat := ExcelBuf."Cell Value as Text";
                    Text022:
                      CodeMagasin := ExcelBuf."Cell Value as Text";
                    Text023: BEGIN
                      LivraisonDirecte := ExcelBuf."Cell Value as Text";
                      IF (LivraisonDirecte='Non') THEN BEGIN
                        DropShipment := FALSE;
                      END ELSE BEGIN
                        DropShipment := TRUE;
                      END;
                    END;
                    Text024: BEGIN
                      CommandeSpeciale := ExcelBuf."Cell Value as Text";
                      IF (CommandeSpeciale='Non') THEN BEGIN
                        SpecialOrder := FALSE;
                      END ELSE BEGIN
                        SpecialOrder := TRUE;
                      END;
                    END;
                    Text025: BEGIN
                      Flexibilite := ExcelBuf."Cell Value as Text";
                      IF (Flexibilite = 'Ilimit'') THEN BEGIN
                        PurchasingFlexibility := PurchasingFlexibility::Unlimited;
                      END;
                      IF (Flexibilite = 'Aucune') THEN BEGIN
                        PurchasingFlexibility := PurchasingFlexibility::None;
                      END;

                    END;
                    Text026: BEGIN
                      TypeLogistique := ExcelBuf."Cell Value as Text";
                      IF (TypeLogistique = 'Tous les magasins') THEN BEGIN
                        LogisticType := LogisticType::"All Location";
                      END;
                      IF (TypeLogistique = 'Magasin') THEN BEGIN
                        LogisticType := LogisticType::Location;
                      END;
                      IF (TypeLogistique = 'Zone logistique') THEN BEGIN
                        LogisticType := LogisticType::"Logistique Area";
                      END;
                    END;
                    Text027: BEGIN
                      IDClassement := ExcelBuf."Cell Value as Text";
                    END;
                    Text028: BEGIN
                      Indentation := ExcelBuf."Cell Value as Text";
                    END;
                  END;
                END;
              END;
          END;
        UNTIL ExcelBuf.NEXT = 0;
        UpdateData();
      END;


      Window.CLOSE;
    END;
