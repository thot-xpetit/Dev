        OnAfterGetRecord=BEGIN
                           IF "Actual Shipping Agent Code" = '' THEN
                             IF RecOriginTransferLine_G.GET("Document No.","Derived From Line No.") THEN
                               // Si la ligne d'origine contient le transporteur r'el, on l'affecte . la ligne
                               IF RecOriginTransferLine_G."Actual Shipping Agent Code" <> '' THEN BEGIN
                                 "Actual Shipping Agent Code" := RecOriginTransferLine_G."Actual Shipping Agent Code";
                                 NbModifies_G += 1;
                                 MODIFY;
                                 COMMIT;
                               END ELSE BEGIN
                                 // si la ligne d'origine ne contient pas non plus de code transporteur reel, on recupSre le premiSre ligne
                                 // avec un code transporteur r'el renseign', et on modifie la ligne de reception
                                 // mais aussi la ligne . laquelle elle est associ'e
                                 RecOriginTransferLine_G.SETRANGE("Document No.","Document No.");
                                 RecOriginTransferLine_G.SETFILTER("Actual Shipping Agent Code",'<>%1','0');
                                 IF RecOriginTransferLine_G.FINDSET THEN BEGIN
                                   ActualShippingCode_G := RecOriginTransferLine_G."Actual Shipping Agent Code";
                                   "Actual Shipping Agent Code" := ActualShippingCode_G;
                                   RecOriginTransferLine_G.GET("Document No.","Derived From Line No.");
                                   RecOriginTransferLine_G."Actual Shipping Agent Code" := ActualShippingCode_G;
                                   RecOriginTransferLine_G.MODIFY;
                                   NbModifies_G += 1;
                                   MODIFY;
                                   COMMIT;
                                 END;
                               END;
                         END;

        DataItemLink=Document No.=FIELD(No.);
