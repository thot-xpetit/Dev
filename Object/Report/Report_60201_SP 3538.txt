    PROCEDURE CertiPhytoExist@1000000032(GUID_P@1000000000 : GUID;VAR CMCContactAccreditation_P@1000000002 : Record 8053314) : Boolean;
    VAR
      CMC_CRM_GUID2_L@1000000001 : Record 8053609;
      CodeNav_L@1000000003 : Text[200];

      ContactType_L@1000000005 : 'Company,Person';
      ContactNo_L@1000000006 : Code[20];
      AccreditCode_L@1000000007 : Code[20];

      EndDate_L@1000000009 : Date;
      Separator_L@1000000010 : Text[4];
      Decoupe_L@1000000011 : Text[200];
      JJ_L@1000000012 : Integer;
      MM_L@1000000013 : Integer;
      AAAA_L@1000000014 : Integer;
      StartDateText_L@1000000015 : Text[10];
      EndDateText_L@1000000016 : Text[10];
    BEGIN
      // --> 20201221-WI00-ADV01.01.00-3504-CJY01
      CMC_CRM_GUID2_L.RESET;
      CMC_CRM_GUID2_L.SETRANGE("Type Data",CMC_CRM_GUID2_L."Type Data"::Certiphyto);
      CMC_CRM_GUID2_L.SETRANGE(GUID,GUID_P);
      IF CMC_CRM_GUID2_L.FINDFIRST THEN BEGIN
        // on essaie de charger le record avec l'enregistrement certi phyto
        CLEAR(CuCrm_L);
        CodeNav_L := CuCrm_L.GetCodeFromGuid2(Text60010,CuCrm_L.CleanGuidToCRM(GUID_P),FALSE);
        Decoupe_L := CodeNav_L;
        Separator_L := CuCrm_L.GetSeparator();
        //on r'cupSre les 'l'ments de la cl'
        //le type est obligatoirement "personne"
        ContactType_L := ContactType_L::Person;
        Decoupe_L := COPYSTR(Decoupe_L,STRPOS(Decoupe_L,Separator_L)+1);
        ContactNo_L := COPYSTR(Decoupe_L,1,STRPOS(Decoupe_L,Separator_L)-1);
        Decoupe_L := COPYSTR(Decoupe_L,STRPOS(Decoupe_L,Separator_L)+1);
        AccreditCode_L := COPYSTR(Decoupe_L,1,STRPOS(Decoupe_L,Separator_L)-1);
        Decoupe_L := COPYSTR(Decoupe_L,STRPOS(Decoupe_L,Separator_L)+1);
        StartDateText_L := COPYSTR(Decoupe_L,1,STRPOS(Decoupe_L,Separator_L)-1);
        IF StartDateText_L <> '' THEN BEGIN
          EVALUATE(JJ_L,COPYSTR(StartDateText_L,1,2));
          EVALUATE(MM_L,COPYSTR(StartDateText_L,4,2));
          EVALUATE(AAAA_L,COPYSTR(StartDateText_L,7,4));
          StartDate_L := DMY2DATE(JJ_L,MM_L,AAAA_L);
        END
        ELSE BEGIN
          StartDate_L := 0D;
        END;

        Decoupe_L := COPYSTR(Decoupe_L,STRPOS(Decoupe_L,Separator_L)+1);
        EndDateText_L := Decoupe_L;

        IF EndDateText_L <> '' THEN BEGIN

          EVALUATE(JJ_L,COPYSTR(EndDateText_L,1,2));
          EVALUATE(MM_L,COPYSTR(EndDateText_L,4,2));
          EVALUATE(AAAA_L,COPYSTR(EndDateText_L,7,4));
          EndDate_L := DMY2DATE(JJ_L,MM_L,AAAA_L);
        END
        ELSE BEGIN
          EndDate_L := 0D;
        END;
        //on cherche . charger le record agr'ment avec l'enregistrement
        IF CMCContactAccreditation_P.GET(ContactType_L,ContactNo_L,AccreditCode_L,StartDate_L,EndDate_L) THEN BEGIN
           EXIT(TRUE);
        END
        ELSE BEGIN
           EXIT(FALSE);
        END;
      END;
      EXIT(FALSE);
      // <-- 20201221-WI00-ADV01.01.00-3504-CJY01
    END;
