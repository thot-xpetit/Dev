    LOCAL PROCEDURE CreateTempShipmentLine@8053000() : Integer;
    VAR
      NextLineNo_L@8053003 : Integer;
      RecLocation_L@8053000 : Record 14;
    BEGIN
      // creation of invoice line from sales contract deliveries
      NextLineNo_L := GetNextInvLineNo(RecSalesHeaderTmp_G);
      RecSalesLineTmp_G.INIT;
      RecSalesLineTmp_G."Document Type" := RecSalesLineTmp_G."Document Type" :: Invoice;
      RecSalesLineTmp_G."Document No." := RecSalesHeaderTmp_G."No.";
      RecSalesLineTmp_G."Line No." := NextLineNo_L;
      IF "CMC Posted Delivery"."Delivery Type" = "CMC Posted Delivery"."Delivery Type"::Delivery THEN
        RecSalesLineTmp_G.Description:=STRSUBSTNO(Text8053000,"CMC Posted Delivery"."No.","CMC Posted Delivery"."Customer Delivery Date"
      )
      ELSE
        RecSalesLineTmp_G.Description:=STRSUBSTNO(Text60002,"CMC Posted Delivery"."No.","CMC Posted Delivery"."Customer Delivery Date");
      IF "CMC Posted Delivery".Application <> '' THEN
        RecSalesLineTmp_G."Description 2":= STRSUBSTNO(Text8053012, "CMC Posted Delivery".Application);
      RecSalesLineTmp_G."Posted Delivery No." := "CMC Posted Delivery"."No.";
      RecSalesLineTmp_G.INSERT();

      CreateTempCommentLine;

      NextLineNo_L := GetNextInvLineNo(RecSalesHeaderTmp_G);
      RecSalesLineTmp_G.INIT;
      RecSalesLineTmp_G."Document Type" := RecSalesLineTmp_G."Document Type" :: Invoice;
      RecSalesLineTmp_G."Document No." := RecSalesHeaderTmp_G."No.";
      RecSalesLineTmp_G."Line No." := NextLineNo_L;
      RecSalesLineTmp_G."Sell-to Customer No.":= RecSalesHeaderTmp_G."Sell-to Customer No.";
      RecSalesLineTmp_G.Type := RecSalesLineTmp_G.Type :: Item;
      RecSalesLineTmp_G."No." := "CMC Posted Delivery"."Item No.";
      RecSalesLineTmp_G."Location Code" := "CMC Posted Delivery"."Location Code";
      RecSalesLineTmp_G.Quantity := CMCPostedDeliveryQty_G;
      RecSalesLineTmp_G."Unit Price" := "CMC Posted Delivery"."Unit Price" - "CMC Posted Delivery"."Shipping Unit Price";
      RecSalesLineTmp_G."Line Amount" := RecSalesLineTmp_G.Quantity * RecSalesLineTmp_G."Unit Price";
      RecSalesLineTmp_G."Posted Delivery No." := "CMC Posted Delivery"."No.";
      RecSalesLineTmp_G."Unloading Method Code" := "CMC Posted Delivery"."Unloading Method Code";
      IF "CMC Posted Delivery"."Delivery Type" = "CMC Posted Delivery"."Delivery Type"::Delivery THEN BEGIN
        RecSalesLineTmp_G."Total Net Weight" := CMCPostedDeliveryNetWeight_G;
        RecSalesLineTmp_G."Total Recalc. Standard Weight" := CMCPostedDeliveryStandWeight_G;

        RecSalesLineTmp_G."Total Recalc. Standard Weight" := - CMCPostedDeliveryStandWeight_G;

      RecSalesLineTmp_G.INSERT();

      EXIT(RecSalesLineTmp_G."Line No." );
    END;

    LOCAL PROCEDURE CreateTempRaiseLine@8053003(ShipmentLineNo_P@8053003 : Integer);

      // creation of Price raise invoice line

    END;
