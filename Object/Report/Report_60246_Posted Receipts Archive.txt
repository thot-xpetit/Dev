    PROCEDURE IsArchivable@1000000003(PostedReceipts_P@1000000000 : Record 8052995) ToArchive_R : Boolean;
    VAR
      PostedReceiptsLine_L@1000000001 : Record 8053104;
      NbrPostedReceiptsLine_L@1000000002 : Integer;
      NbrPostedInvoiceReceiptLine_L@1000000003 : Integer;
      NbrPostedCanceledReceiptLine_L@1000000004 : Integer;

      NbrFinishedCollectionContra_L@1000000006 : Integer;
      PostedReceiptLine_L@1000000007 : Record 8052997;
      NbrComplementPrice_L@1000000008 : Integer;
      NbrComplementCanceledPrice_L@1000000009 : Integer;
      NbrComplementInvoicePrice_L@1000000010 : Integer;
      NbrPostedCancellLine_L@1000000011 : Integer;
    BEGIN
      WITH PostedReceipts_P DO BEGIN
           // Initialisation des variables
           ToArchive_R := FALSE;
           NbrPostedReceiptsLine_L        :=  0;
           NbrPostedInvoiceReceiptLine_L  :=  0;
           NbrPostedCanceledReceiptLine_L :=  0;
           NbrFinishedCollectionContra_L  :=  0;
           NbrComplementPrice_L           :=  0;
           // --> 20260226-WI00-ADV01.01.00-3596-CJY
           NbrPostedCancellLine_L := 0;
           // <-- 20260226-WI00-ADV01.01.00-3596-CJY

           // PRESTATIONS
           PostedReceiptLine_L.RESET;
           PostedReceiptLine_L.SETRANGE("Receipt Type" , "Receipt Type");
           PostedReceiptLine_L.SETRANGE("Posted Receipt No.","No.");
           // --> 20260128-WI00-ADV01.01.00-3596-CJY
           PostedReceiptLine_L.SETFILTER(PostedReceiptLine_L.Quantity,'<>%1',0);
           PostedReceiptLine_L.SETFILTER(PostedReceiptLine_L."Unit Price",'<>%1',0);
           // <-- 20260128-WI00-ADV01.01.00-3596-CJY
           NbrComplementPrice_L :=  PostedReceiptLine_L.COUNT;
           PostedReceiptLine_L.SETRANGE(Canceled,TRUE);
           NbrComplementCanceledPrice_L :=   PostedReceiptLine_L.COUNT;
           PostedReceiptLine_L.SETRANGE(Canceled,FALSE);
           PostedReceiptLine_L.SETRANGE(Invoiced,TRUE);
           NbrComplementInvoicePrice_L := PostedReceiptLine_L.COUNT;

           IF NbrComplementPrice_L <> 0 THEN BEGIN
              IF ((NbrComplementCanceledPrice_L + NbrComplementInvoicePrice_L) <> NbrComplementPrice_L) THEN EXIT;
           END;

           PostedReceiptsLine_L.RESET;
           PostedReceiptsLine_L.SETRANGE("Receipt Type", "Receipt Type");
           PostedReceiptsLine_L.SETRANGE("No.","No.");
           NbrPostedReceiptsLine_L :=  PostedReceiptsLine_L.COUNT;

           // --> 20260226-WI00-ADV01.01.00-3596-CJY
           PostedReceiptsLine_L.SETRANGE("Cancellation line",FALSE);
           // <-- 20260226-WI00-ADV01.01.00-3596-CJY
           PostedReceiptsLine_L.SETRANGE("Canceled Line",FALSE);
           PostedReceiptsLine_L.SETRANGE(Invoiced,TRUE);
           NbrPostedInvoiceReceiptLine_L :=  PostedReceiptsLine_L.COUNT;

          // --> 20260226-WI00-ADV01.01.00-3596-CJY
          PostedReceiptsLine_L.SETRANGE(Invoiced);
          PostedReceiptsLine_L.SETRANGE("Cancellation line",TRUE);

          NbrPostedCancellLine_L :=  PostedReceiptsLine_L.COUNT;
          PostedReceiptsLine_L.SETRANGE("Cancellation line");
          // <-- 20260226-WI00-ADV01.01.00-3596-CJY


           // --> 20260128-WI00-ADV01.01.00-3596-CJY
           {BEFORE
           PostedReceiptsLine_L.SETRANGE(Invoiced,FALSE);
           AFTER}
           PostedReceiptsLine_L.SETRANGE(Invoiced);
           // <-- 20260128-WI00-ADV01.01.00-3596-CJY
           PostedReceiptsLine_L.SETRANGE("Canceled Line",TRUE);


           NbrPostedCanceledReceiptLine_L :=  PostedReceiptsLine_L.COUNT;

           // --> 20260226-WI00-ADV01.01.00-3596-CJY
           {BEFORE
           IF NbrPostedReceiptsLine_L  = (NbrPostedInvoiceReceiptLine_L + NbrPostedCanceledReceiptLine_L) THEN BEGIN
           AFTER}
           IF NbrPostedReceiptsLine_L  = (NbrPostedInvoiceReceiptLine_L + NbrPostedCanceledReceiptLine_L +
            NbrPostedCancellLine_L ) THEN BEGIN
           // <-- 20260226-WI00-ADV01.01.00-3596-CJY
              PostedReceiptsLine_L.SETRANGE(Invoiced,TRUE);
              PostedReceiptsLine_L.SETRANGE("Canceled Line",FALSE);
              // --> 20260226-WI00-ADV01.01.00-3596-CJY
              PostedReceiptsLine_L.SETRANGE("Cancellation line",FALSE);
              // <-- 20260226-WI00-ADV01.01.00-3596-CJY
              IF PostedReceiptsLine_L.FINDSET THEN REPEAT
                 CollectionContract_L.RESET;
                 CollectionContract_L.COPYFILTERS("CMC Collection Contract");
                 IF "CMC Collection Contract".GETFILTER("No.") = '' THEN
                     CollectionContract_L.SETRANGE("No.", PostedReceiptsLine_L."Contract No.");
                 IF CollectionContract_L.FINDSET THEN REPEAT
                   IF PostedReceiptsLine_L."Contract No." = CollectionContract_L."No." THEN
                      NbrFinishedCollectionContra_L += 1;
                UNTIL CollectionContract_L.NEXT = 0;
              UNTIL PostedReceiptsLine_L.NEXT = 0;
              IF (NbrFinishedCollectionContra_L =  NbrPostedInvoiceReceiptLine_L)
                 AND (NbrFinishedCollectionContra_L <> 0 ) THEN ToArchive_R := TRUE;
           END;
      END;
    END;
