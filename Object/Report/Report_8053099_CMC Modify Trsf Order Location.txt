    PROCEDURE CheckTransfer@1000000000(TransferHeader_P@1000000001 : Record 5740);
    VAR
      TransferLine_L@1000000000 : Record 5741;
      WhseRcptLine_L@1000000002 : Record 7317;
    BEGIN
      TransferHeader_P.TESTFIELD(Status,TransferHeader_P.Status::Open);

      TransferLine_L.RESET;
      TransferLine_L.SETRANGE("Document No.",TransferHeader_P."No.");
      IF TransferLine_L.FINDSET THEN
        REPEAT

          IF TransferLine_L."Quantity Received" <> 0 THEN
            ERROR(Text1000000020);

          WhseRcptLine_L.RESET;
          WhseRcptLine_L.SETRANGE("Source Type",DATABASE::"Transfer Line");
          WhseRcptLine_L.SETRANGE("Source Subtype",1);
          WhseRcptLine_L.SETRANGE("Source No.",TransferHeader_P."No.");
          IF WhseRcptLine_L.FINDSET THEN
            ERROR(Text1000000000);

        UNTIL TransferLine_L.NEXT = 0;
    END;

    PROCEDURE ValidateDocNo@1000000007();
    BEGIN
      IF TransferNo_G <> '' THEN BEGIN
        TransferHeader_G.GET(TransferNo_G);
        CheckTransfer(TransferHeader_G);
        TrsfFromLocationCode_G := TransferHeader_G."Transfer-from Code";
        TrsfToLocationCode_G := TransferHeader_G."Transfer-to Code";
        GetLocationNames;
      END ELSE
        ClearLocationInfo;
    END;

    PROCEDURE GetLocationNames@1000000009();
    VAR
      Location_L@1000000000 : Record 14;
    BEGIN
      IF TrsfFromLocationCode_G <> '' THEN BEGIN
        Location_L.GET(TrsfFromLocationCode_G);
        TrsfFromLocationName_G := Location_L.Name;
      END ELSE
        TrsfFromLocationName_G := '';

      IF TrsfToLocationCode_G <> '' THEN BEGIN
        Location_L.GET(TrsfToLocationCode_G);
        TrsfToLocationName_G := Location_L.Name;
      END ELSE
        TrsfToLocationName_G := '';

      Location_L.TESTFIELD("Use As In-Transit",FALSE);
    END;

    PROCEDURE ClearLocationInfo@1000000008();
    BEGIN
      TrsfFromLocationCode_G := '';
      TrsfFromLocationName_G := '';
      TrsfToLocationCode_G := '';
      TrsfToLocationName_G := '';
    END;
