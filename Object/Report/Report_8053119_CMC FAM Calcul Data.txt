    PROCEDURE GetDepartment@8053004(P_ItemLedgerEntry@8053000 : Record 32) Res_Depart : Code[10];
    VAR
      RecVendor_L@8053002 : Record 23;
      RecCustomer_L@8053001 : Record 18;
    BEGIN
      // renvoi le n› d'partement de l'acheteur ou le client
      CASE P_ItemLedgerEntry."Source Type" OF
          P_ItemLedgerEntry."Source Type"::Vendor:
            BEGIN
              RecVendor_L.GET(P_ItemLedgerEntry."Source No.");
              IF (RecVendor_L."Country/Region Code" <> 'FR') AND (RecVendor_L."Country/Region Code" <> '') THEN
                Res_Depart:=RecVendor_L."Country/Region Code"
              ELSE
                Res_Depart:= COPYSTR(RecVendor_L."Post Code",1,2);
            END;

          P_ItemLedgerEntry."Source Type"::Customer:
            BEGIN
              IF P_ItemLedgerEntry.Area<>'' THEN
               Res_Depart:= P_ItemLedgerEntry.Area
              ELSE BEGIN
                RecCustomer_L.GET(P_ItemLedgerEntry."Source No.");
                IF (RecCustomer_L."Country/Region Code" <> 'FR') AND (RecCustomer_L."Country/Region Code" <> '') THEN
                  Res_Depart:=RecCustomer_L."Country/Region Code"
                ELSE
                  Res_Depart:= COPYSTR(RecCustomer_L."Post Code",1,2);
              END;
              // --> 20150826-CER19-CMC02.02.00-2109-SDE01
              IF (P_ItemLedgerEntry."Entry/Exit Point"<>'') AND (P_ItemLedgerEntry."Entry/Exit Point"<>'FR')THEN
              Res_Depart:= COPYSTR(P_ItemLedgerEntry."Entry/Exit Point",1,2);

              // <-- 20150826-CER19-CMC02.02.00-2109-SDE01
            END;

      END;

      EXIT(Res_Depart)
    END;

    PROCEDURE GetCustInfo@8053000(P_ItemLedgerEntry@8053000 : Record 32) Res_CustInfo : Text[30];
    VAR
      RecCustomer_L@8053001 : Record 18;
      cduCMCCVLedgerMgt_L@8053004 : Codeunit 8053377;
      RecContact_L@8053003 : Record 5050;
      RecSpeciesSetup_L@8053002 : Record 8053323;
      RecItem_L@8053005 : Record 27;
    BEGIN
      //Renvoie l'info :
      // IAA : industrie agro
      // E  : eleveur
      // B  : orge de brasserie

      Res_CustInfo:='';
      //IF P_ItemLedgerEntry."Source Type"=P_ItemLedgerEntry."Source Type"::Customer
      IF P_ItemLedgerEntry."Movement Type" IN[P_ItemLedgerEntry."Movement Type"::S10,
                                              P_ItemLedgerEntry."Movement Type"::S11]
      THEN BEGIN
          RecCustomer_L.GET(P_ItemLedgerEntry."Source No.");
          IF RecCustomer_L."Food Industry"
           THEN Res_CustInfo:='IAA'; //dont industrie agro alimentaire

          IF RecContact_L.GET(cduCMCCVLedgerMgt_L.GetContNoFromCustVentNo(P_ItemLedgerEntry."Source No.",0)) AND
            (RecContact_L."Contact Type"=RecContact_L."Contact Type"::Farm)
          THEN BEGIN
            IF Res_CustInfo<>'' THEN
            Res_CustInfo:=STRSUBSTNO('%1|%2',Res_CustInfo,'E') //dont eleveur
            ELSE
            Res_CustInfo:=STRSUBSTNO('%1','E')
          END;

          IF RecItem_L.GET(P_ItemLedgerEntry."Item No.") AND
             RecSpeciesSetup_L.GET(RecItem_L."Species Code") AND
             (RecSpeciesSetup_L."Rich Product"=RecItem_L."No.")
          THEN BEGIN
            IF Res_CustInfo<>'' THEN
            Res_CustInfo:=STRSUBSTNO('%1|%2',Res_CustInfo,'B') //dont orge de Brasserie
            ELSE
            Res_CustInfo:=STRSUBSTNO('%1','B')

          END;

      END;
      EXIT(Res_CustInfo);
    END;

    PROCEDURE GetLocation@8053002(P_ItemLedgerEntry@8053000 : Record 32) Res_Location : Code[10];
    VAR
      RecLocation_L@1000000000 : Record 14;
    BEGIN
      // renvoi le magasin
      Res_Location:=P_ItemLedgerEntry."Location Code";

      IF P_ItemLedgerEntry."Independent Transfer" THEN BEGIN
        CASE P_ItemLedgerEntry."Movement Type" OF
          P_ItemLedgerEntry."Movement Type"::S05 : // si tranfert sortie, les 2 'critures ont le mag de sortie
            // --> 20160429-CER19-ADV01.01.00-2902-FBE
            IF P_ItemLedgerEntry."Location Code" <> P_ItemLedgerEntry."Indep. Transfer-From Location" THEN BEGIN
              RecLocation_L.GET( P_ItemLedgerEntry."Location Code");
              IF NOT RecLocation_L."Compensation Location" THEN
                Res_Location:=P_ItemLedgerEntry."Indep. Transfer-From Location" ;
            END ELSE
            // <-- 20160429-CER19-ADV01.01.00-2902-FBE
            Res_Location:=P_ItemLedgerEntry."Indep. Transfer-From Location" ;
          P_ItemLedgerEntry."Movement Type"::E05 : // si tranfert entr'e, le mag du tranfert = mag origine
            IF P_ItemLedgerEntry."Location Code"<>P_ItemLedgerEntry."Indep. Transfer-To Location"
            THEN
              Res_Location:=P_ItemLedgerEntry."Indep. Transfer-From Location"

        END;
      END;
      EXIT(Res_Location)
    END;

    PROCEDURE GetPositiveSign@8053007(P_ItemLedgerEntry@8053000 : Record 32) Res_PositivSign : Boolean;
    VAR
      RecItemLedgerEntry_L@8053001 : Record 32;
    BEGIN
      // renvoi le signe

      Res_PositivSign:=P_ItemLedgerEntry.Quantity>0;

      //******Gestion des exceptions*********** //

      // cas des annulation pure et annulation pour mise en achat
      IF (P_ItemLedgerEntry."Movement Type"=P_ItemLedgerEntry."Movement Type"::E03) AND
        (P_ItemLedgerEntry.Quantity<0) THEN
      BEGIN
        Res_PositivSign:=TRUE;  // cas des annulations pures
        RecItemLedgerEntry_L.RESET; // recherche des cas d'incorporation
        RecItemLedgerEntry_L:=P_ItemLedgerEntry;
        // --> 20170811-WI00-ADV01.01.00-28807-SDE
        RecItemLedgerEntry_L.SETCURRENTKEY("Document No.");
        RecItemLedgerEntry_L.SETRANGE("Document No.",P_ItemLedgerEntry."Document No.");
        // <-- 20170811-WI00-ADV01.01.00-28807-SDE
        IF (RecItemLedgerEntry_L.NEXT<>0) AND // prochaine 'criture
          (RecItemLedgerEntry_L."Movement Type"=RecItemLedgerEntry_L."Movement Type"::E01)
          AND (RecItemLedgerEntry_L.Quantity > 0 ) //<-- 20171213-Wi00-ADV01.01.00-30202-JLT01
          THEN Res_PositivSign:=FALSE;

      END;

      // --> 20150826-CER19-CMC02.02.00-2108-SDE01
      //cas ses annulations de reprise et annulaion pour mise en achat
      IF (P_ItemLedgerEntry."Movement Type"=P_ItemLedgerEntry."Movement Type"::S03) AND
        (P_ItemLedgerEntry.Quantity>0) THEN
      BEGIN
        Res_PositivSign:=FALSE;  // cas des annulations pures
        RecItemLedgerEntry_L.RESET; // recherche des cas d'incorporation
        RecItemLedgerEntry_L:=P_ItemLedgerEntry;
        // --> 20170811-WI00-ADV01.01.00-28807-SDE
        RecItemLedgerEntry_L.SETCURRENTKEY("Document No.");
        RecItemLedgerEntry_L.SETRANGE("Document No.",P_ItemLedgerEntry."Document No.");
        // <-- 20170811-WI00-ADV01.01.00-28807-SDE

        IF (RecItemLedgerEntry_L.NEXT<>0) AND // prochaine 'criture
          (RecItemLedgerEntry_L."Movement Type"=RecItemLedgerEntry_L."Movement Type"::E01)
          THEN Res_PositivSign:=TRUE;

      END;

      // <-- 20150826-CER19-CMC02.02.00-2108-SDE01

      // --> 20151106-CER19-ADV01.01.00-2236-SDE01
      // cas des reprises et annulation de reprises
      IF (P_ItemLedgerEntry."Movement Type"=P_ItemLedgerEntry."Movement Type"::E01) AND
      (RecPostedReceipts_G.GET(RecPostedReceipts_G."Receipt Type"::Resumption,P_ItemLedgerEntry."Document No."))
       THEN Res_PositivSign:=P_ItemLedgerEntry.Quantity<0;

      // <-- 20151106-CER19-ADV01.01.00-2236-SDE01


      // Cas de l'incorporation TAF

        // 1› 'criture (aprSs le E05)  diff'rencier annulation pure de l'annulation pour incorpo
      IF (P_ItemLedgerEntry."Movement Type"=P_ItemLedgerEntry."Movement Type"::E04) AND
        (P_ItemLedgerEntry.Quantity<0)  THEN
      BEGIN
        RecItemLedgerEntry_L.RESET;
        Res_PositivSign:=TRUE; // cas des annulations pures
        RecItemLedgerEntry_L:=P_ItemLedgerEntry; // cas des incorporation
        IF (RecItemLedgerEntry_L.NEXT(-1)<>0) AND //  'criture pr'c'dente
          (RecItemLedgerEntry_L."Movement Type"=P_ItemLedgerEntry."Movement Type"::E05) AND // type E05
          (RecItemLedgerEntry_L."Document No."=P_ItemLedgerEntry."Document No.") AND // meme doc
          (RecItemLedgerEntry_L.Quantity>0) // qu' >0
          THEN Res_PositivSign:=FALSE;
      END;

        // 2› ecriture , . mettre dans le meme signe que la 1› 'criture
      IF (P_ItemLedgerEntry."Movement Type"=P_ItemLedgerEntry."Movement Type"::E04) AND // si TAF
      (P_ItemLedgerEntry.Quantity>0) THEN BEGIN
        RecItemLedgerEntry_L.RESET;
        RecItemLedgerEntry_L:=P_ItemLedgerEntry;
        IF (RecItemLedgerEntry_L.NEXT(-1)<>0) AND // Pr'c'dente 'critre
          (RecItemLedgerEntry_L."Movement Type"=RecItemLedgerEntry_L."Movement Type"::E04) AND
          (P_ItemLedgerEntry."Document No."=RecItemLedgerEntry_L."Document No.") AND
          (RecItemLedgerEntry_L.Quantity<0)
        THEN Res_PositivSign:=FALSE;
      END;


        // cas du retour TAF
      IF (P_ItemLedgerEntry."Movement Type"=P_ItemLedgerEntry."Movement Type"::E04) AND // si TAF
      (P_ItemLedgerEntry.Quantity>0) THEN BEGIN
        RecItemLedgerEntry_L.RESET;
        RecItemLedgerEntry_L:=P_ItemLedgerEntry;
        IF (RecItemLedgerEntry_L.NEXT(-1)<>0) AND // Pr'c'dente 'critre
          (RecItemLedgerEntry_L."Movement Type"=RecItemLedgerEntry_L."Movement Type"::E05) AND
          (P_ItemLedgerEntry."Document No."=RecItemLedgerEntry_L."Document No.") AND
          (ABS(P_ItemLedgerEntry.Quantity) = ABS(RecItemLedgerEntry_L.Quantity)) // meme valeur absolue
        THEN Res_PositivSign:=FALSE;
      END;


      EXIT(Res_PositivSign)
    END;
