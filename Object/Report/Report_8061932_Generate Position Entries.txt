                           // Si l''criture ne dispose pas de dates, alors on ne la traite pas.
                           IF ("Position Starting Date" = 0D) OR ("Position Ending Date" = 0D) THEN
                             CurrReport.SKIP;

                           // V'rification si une 'criture d'en-cours n'existe pas pour l''criture.
                           PositionEntry_G.RESET;
                           PositionEntry_G.SETCURRENTKEY("Position Source","G/L Entry No.","Position Date");

                           IF NOT PositionEntry_G.ISEMPTY THEN
                             CurrReport.SKIP;

                           // Calculs du montant factur' . la date de situation
                           IF "G/L Entry"."Posting Date" <= PositionDate_G THEN BEGIN
                             InvoicedAmount_G := "G/L Entry".Amount;
                             InAdvance_G := TRUE;
                           END ELSE BEGIN
                             InvoicedAmount_G := 0;
                             InAdvance_G := FALSE;
                           END;

                           // Calculs du montant r'alis' . la date de situation
                           TotalPeriodNo_G := "Position Ending Date" - "Position Starting Date" +1;
                           CASE TRUE OF
                             PositionDate_G < "Position Starting Date" : RealizedPeriodNo_G := 0;
                             PositionDate_G > "Position Ending Date" :   RealizedPeriodNo_G := TotalPeriodNo_G;
                             ELSE
                               RealizedPeriodNo_G := PositionDate_G - "Position Starting Date" + 1;
                           END;
                           UnrealPeriodNo_G := TotalPeriodNo_G - RealizedPeriodNo_G;

                           IF (TotalPeriodNo_G <> 0) THEN
                             RealizedPercent_G := RealizedPeriodNo_G / TotalPeriodNo_G;
                           UnrealizedPercent_G := 1 - RealizedPercent_G;

                           RealizedAmount_G := ROUND((Amount * RealizedPercent_G),0.01);
                           PositionAmount_G := InvoicedAmount_G - RealizedAmount_G;

                           CLEAR(TransferDatas_G);
                           TransferDatas_G[1] := TotalPeriodNo_G;
                           TransferDatas_G[2] := RealizedPeriodNo_G;
                           TransferDatas_G[3] := UnrealPeriodNo_G;
                           TransferDatas_G[4] := RealizedPercent_G * 100;
                           TransferDatas_G[5] := UnrealizedPercent_G * 100;
                           TransferDatas_G[6] := Amount;
                           TransferDatas_G[7] := InvoicedAmount_G;
                           TransferDatas_G[8] := RealizedAmount_G;

                           PositionMngt_G.InsertGLPosEntry("G/L Entry",PositionDate_G,PositionAmount_G,TransferDatas_G,InAdvance_G);
                         END;
