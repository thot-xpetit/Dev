    PROCEDURE UpdateContact@1000000022();
    VAR
      Contact@1000000000 : Record 5050;
      Vendor@1000000001 : Record 23;
      Customer@1000000002 : Record 18;
      Location@1000000003 : Record 14;
    BEGIN
      IF (IDCustomer <> '') THEN BEGIN
        IF (Contact.GET(IDCustomer)) THEN BEGIN
          {
          IF (Vendor.GET(Contact."Vendor No.")) THEN BEGIN
            IF ((ZoneGeo <> '') AND (ZoneGeo <> Vendor."Localization Area Code")) THEN BEGIN
              Vendor.VALIDATE("Localization Area Code", ZoneGeo);
              Vendor.MODIFY();
            END;
          END;
          IF (Location.GET(SiteCode)) THEN BEGIN
            IF ((SiteCode <> '') AND (SiteCode <> Contact."Location Code")) THEN BEGIN
              Contact.VALIDATE("Location Code", SiteCode);
              Contact.MODIFY();
            END;
            IF (Vendor.GET(Contact."Vendor No.")) THEN BEGIN
              IF ((SiteCode <> '') AND (SiteCode <> Vendor."Location Code")) THEN BEGIN
                Vendor.VALIDATE("Location Code", SiteCode);
                Vendor.MODIFY();
              END;
            END;
            IF (Customer.GET(Contact."Customer No.")) THEN BEGIN
              IF ((SiteCode <> '') AND (SiteCode <> Customer."Location Code")) THEN BEGIN
                Customer.VALIDATE("Location Code", SiteCode);
                Customer.MODIFY();
              END;
            END;
          END;
          }
          IF (NewCommCode <> '') THEN BEGIN
            Contact.VALIDATE("Farmer List 3", NewCommCode);
            Contact.MODIFY();
          END;
        END;
      END;
    END;
