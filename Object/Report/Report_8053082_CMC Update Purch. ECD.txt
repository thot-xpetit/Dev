    PROCEDURE InsertNewDiscount@8053009(PrevFinalPrice_P@8053001 : Decimal) : Decimal;
    VAR
      CMCPurchDiscValue_L@8053000 : Record 8053248;
    BEGIN
      CASE ValuationType_G OF
        ValuationType_G::"New ECD Amount": BEGIN
          ValueEntryDiscForUpdate_G.VALIDATE("Discount Amount",ModificationValue_G);
        END;
        ValuationType_G::"New ECD Percentage": BEGIN
          ValueEntryDiscForUpdate_G.VALIDATE("Discount %",ModificationValue_G);
          ValueEntryDiscForUpdate_G.VALIDATE("Discount Amount",
                        ROUND(ValueEntryDiscForUpdate_G."Base Value" * ModificationValue_G/100, 0.01, '='));
        END;
        ValuationType_G::"Change ECD amount": BEGIN
          ERROR(Text030);
        END;
        ValuationType_G::"Apply Purch. Discount": BEGIN
          IF GetPurchDiscount(CMCPurchDiscValue_L) THEN BEGIN
            CASE PurchDiscount_G."Valuation Type" OF
              PurchDiscount_G."Valuation Type"::Percentage: BEGIN
                ValueEntryDiscForUpdate_G.VALIDATE("Discount %",CMCPurchDiscValue_L."Discount Value");
                ValueEntryDiscForUpdate_G.VALIDATE("Discount Amount",
                              ROUND(ValueEntryDiscForUpdate_G."Base Value" * CMCPurchDiscValue_L."Discount Value"/100, 0.01, '='));
              END;
              PurchDiscount_G."Valuation Type"::"Unit Amount": BEGIN
                ValueEntryDiscForUpdate_G.VALIDATE("Discount Amount",CMCPurchDiscValue_L."Discount Value");
              END;
            END;
          END ELSE
            CurrReport.SKIP;
        END;
      END;

      ValueEntryDiscForUpdate_G.VALIDATE("Final Price",
                  PrevFinalPrice_P - ValueEntryDiscForUpdate_G."Discount Amount");


      // --> 20151110-APP33-ADV01.01.00-2025-CJY01
      //On d'sactive le controle d'erreur suivant
      //que l'on remplace, en dessous, par une non-'criture quand on se retrouve dans ce cas
      {BEFORE
      IF ValueEntryDiscForUpdate_G."Final Price" < 0 THEN
        ERROR(Text004,ValueEntryDiscForUpdate_G."Value Entry No.");
      AFTER}
      // <-- 20151110-APP33-ADV01.01.00-2025-CJY01
      ValueEntryDiscForUpdate_G."ECD Vendor No." := VendorNo_G;
      // --> 20151110-APP33-ADV01.01.00-2025-CJY01

      // <-- 20151110-APP33-ADV01.01.00-2025-CJY01

        IF PrintToExcel_G THEN
          MakeExcelDataBody3;
      // --> 20151110-APP33-ADV01.01.00-2025-CJY01
      END;

    PROCEDURE UpdateDiscount@8053004(VAR ValueEntryDisc_P@8053000 : Record 8053252;ValuationType_P@8053001 : 'New ECD Amount,New ECD Percentage,Change ECD amount,Apply Purch. Discount';ModificationValue_P@8053002 : Decimal;PrevFinalPrice_P@8053003 : Decimal) : Decimal;
    VAR
      CMCPurchDiscValue_L@8053004 : Record 8053248;
    BEGIN
      CASE ValuationType_P OF
        ValuationType_P::"New ECD Amount" : BEGIN
          IF ValueEntryDisc_P."Discount Code" <> CMCSetup_G."Except. Purch. ECD Disc. Code" THEN BEGIN
            ValueEntryDisc_P."Discount Amount" := ModificationValue_P;
            ValueEntryDisc_P."Discount %" := 0;
            SetFinalPrice(ValueEntryDisc_P,PrevFinalPrice_P);
          END ELSE BEGIN
            IF UpdateType_G <> UpdateType_G::"Unapply Budgeted Unit Cost" THEN BEGIN
              IF ValueEntryDisc_P."Discount Amount" <> 0 THEN BEGIN
                IF ValueEntryDisc_P."Base Value" > ValueEntryDisc_P."Final Price" THEN
                  ValueEntryDisc_P."Discount Amount" := ValueEntryDisc_P."Base Value" - ValueEntryDisc_P."Final Price"
                ELSE
                  IF ValueEntryDisc_P."Base Value" < ValueEntryDisc_P."Final Price" THEN
                    ERROR(Text8053018,"Value Entry"."Entry No.")
                  ELSE BEGIN
                    ValueEntryDisc_P."Discount %" := 0;
                    ValueEntryDisc_P."Discount Amount" := 0;
                  END;
              END ELSE BEGIN
                ValueEntryDisc_P."Discount Amount" := ModificationValue_P;
                SetFinalPrice(ValueEntryDisc_P,PrevFinalPrice_P);
              END;
            END ELSE BEGIN
              ValueEntryDisc_P."Discount %" := 0;
              ValueEntryDisc_P."Discount Amount" := 0;
              ValueEntryDisc_P."Final Price" := ValueEntryDisc_P."Base Value";
            END;
          END;
        END;
        ValuationType_P::"New ECD Percentage" : BEGIN
          ValueEntryDisc_P."Discount Amount" :=
                                   ROUND(ValueEntryDisc_P."Base Value" * ModificationValue_P / 100, 0.01, '=');
          ValueEntryDisc_P."Discount %" := ModificationValue_P;
          SetFinalPrice(ValueEntryDisc_P,PrevFinalPrice_P);
        END;
        ValuationType_P::"Change ECD amount" : BEGIN
          ValueEntryDisc_P."Discount Amount" := ValueEntryDisc_P."Discount Amount" + ModificationValue_P;
          SetFinalPrice(ValueEntryDisc_P,PrevFinalPrice_P);
        END;
        ValuationType_P::"Apply Purch. Discount": BEGIN
          GetPurchDiscount(CMCPurchDiscValue_L);
          CASE CMCPurchDiscValue_L."Valuation Type" OF
            CMCPurchDiscValue_L."Valuation Type" ::Percentage : BEGIN
              ValueEntryDisc_P."Discount %" := CMCPurchDiscValue_L."Discount Value";
              ValueEntryDisc_P."Discount Amount" :=
                                 ROUND(ValueEntryDisc_P."Base Value" * CMCPurchDiscValue_L."Discount Value"/100, 0.01, '=');
            END;
            CMCPurchDiscValue_L."Valuation Type" ::"Unit Amount" : BEGIN
              ValueEntryDisc_P."Discount Amount" := CMCPurchDiscValue_L."Discount Value";
            END;
          END;
          SetFinalPrice(ValueEntryDisc_P,PrevFinalPrice_P);
        END;
      END;

      ValueEntryDisc_P.MODIFY(TRUE);

      EXIT(ValueEntryDisc_P."Final Price");
    END;

    PROCEDURE UpdateRemainingValueEntryDisc@8053007(ValueEntryDisc_P@8053001 : Record 8053252;ValueEntry_P@8053004 : Record 5802;PrevFinalPrice_P@8053005 : Decimal) : Decimal;
    VAR
      ValueEntryDisc_L@8053000 : Record 8053252;
      Found_L@8053002 : Boolean;
      PreviousFinalPrice_L@8053003 : Decimal;
      PurchDiscount_L@1000000000 : Record 8053247;
    BEGIN
      PurchECDLedgEntryTmp_G.RESET;
      PurchECDLedgEntryTmp_G.DELETEALL;

      Found_L := FALSE;

      ValueEntryDisc_L.SETCURRENTKEY("Value Entry No.",Rank);
      ValueEntryDisc_L.SETRANGE("Value Entry No.","Value Entry"."Entry No.");

      IF ValueEntryDisc_L.FINDSET THEN
        REPEAT
          IF ValueEntryDisc_L."Discount Code" = ValueEntryDisc_P."Discount Code" THEN BEGIN
            Found_L := TRUE;
            PreviousFinalPrice_L := PrevFinalPrice_P;
            StoreBeforeECDLedgerEntry(PurchECDLedgEntryTmp_G,ValueEntryDisc_L,ValueEntry_P);

            IF Found_L THEN BEGIN
              ValueEntryDisc_L."Base Value" := GetDiscountBaseValue(ValueEntryDisc_L,PreviousFinalPrice_L);
