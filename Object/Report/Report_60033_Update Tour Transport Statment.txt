    PROCEDURE FinishTranspPurchOrderTour@60023();
    VAR
      RecPostedWhseShptLine_L@60007 : Record 7323;
      RecCharterTourne_L@60004 : Record 8053265;
      TransportOrderNo_L@60001 : Code[20];
      RecCharterTourne2_L@60000 : Record 8053265;
      AllowedPostingDate_L@1000000000 : Date;
      PostingDate_L@1000000001 : Date;
      TransferShipmentLine_L@1000000002 : Record 5745;
      TransferReceiptLine_L@1000000003 : Record 5747;
    BEGIN
      AllowedPostingDate_L := 0D;
      PostingDate_L := 0D;
      TransportOrderNo_L := '';
      RecCharterTourne_L.GET("Posted Whse. Shipment Header"."Shipping Order No.");
      IF DateNotAllowed("Posted Whse. Shipment Header"."Posting Date",AllowedPostingDate_L) THEN
        PostingDate_L := AllowedPostingDate_L
      ELSE PostingDate_L := "Posted Whse. Shipment Header"."Posting Date";
      TransportOrderNo_L :=CUCharterTransMgmt_G.CreateTransportPurchOrder(RecCharterTourne_L,RecCharterTourne_L.Quantity,PostingDate_L);
      IF TransportOrderNo_L <> '' THEN BEGIN
          "Posted Whse. Shipment Header"."Transport Order No." := TransportOrderNo_L;
          "Posted Whse. Shipment Header".MODIFY;
          CLEAR(RecPostedWhseShptLine_L);
          RecPostedWhseShptLine_L.SETRANGE("No.","Posted Whse. Shipment Header"."No.");
          IF RecPostedWhseShptLine_L.FINDSET THEN
          REPEAT
            RecPostedWhseShptLine_L."Transport Order No." := TransportOrderNo_L;
            RecPostedWhseShptLine_L.MODIFY;
            // --> 20170606-WI00-ADV01.01.00-23779-CJY01
            IF RecPostedWhseShptLine_L."Source Document" = RecPostedWhseShptLine_L."Source Document"::"Outbound Transfer" THEN BEGIN
              //on r'cupSre le numero OT original
              //mise . jour ligne exp' transfert
              TransferShipmentLine_L.RESET;
              TransferShipmentLine_L.SETRANGE("Transfer Order No.",RecPostedWhseShptLine_L."Source No.");
              TransferShipmentLine_L.SETRANGE("Line No.",RecPostedWhseShptLine_L."Source Line No.");
              IF TransferShipmentLine_L.FINDSET THEN BEGIN
                TransferShipmentLine_L."Transport Order No." := TransportOrderNo_L;
                TransferShipmentLine_L.MODIFY;
                //mise . jour du cot' reception transfert
                TransferReceiptLine_L.RESET;
                TransferReceiptLine_L.SETRANGE("Transfer Order No.",RecPostedWhseShptLine_L."Source No.");
                TransferReceiptLine_L.SETRANGE("Line No.",RecPostedWhseShptLine_L."Source Line No.");
                IF TransferReceiptLine_L.FINDSET THEN REPEAT
                  TransferReceiptLine_L."Transport Order No." := TransportOrderNo_L;
                  TransferReceiptLine_L.MODIFY;
                UNTIL TransferReceiptLine_L.NEXT = 0;
              END;
             END;
            // <-- 20170606-WI00-ADV01.01.00-23779-CJY01
          UNTIL RecPostedWhseShptLine_L.NEXT = 0;
        RecCharterTourne2_L.GET("Posted Whse. Shipment Header"."Shipping Order No.");
        RecCharterTourne2_L."Number of Achieved Units" := RecCharterTourne2_L."Number of Achieved Units" + RecCharterTourne2_L.Quantity;
        RecCharterTourne2_L."Realized Quantity" := RecCharterTourne2_L."Realized Quantity" + RecCharterTourne2_L.Quantity;
        RecCharterTourne2_L.MODIFY;
        CUCharterTransMgmt_G.TerminateCharter(RecCharterTourne2_L."Charter No");
      END;
    END;

    PROCEDURE DateNotAllowed@1(PostingDate@1000 : Date;VAR AllowPostingFrom@1000000000 : Date) : Boolean;
    VAR
      AllowPostingTo@1000000001 : Date;
      UserSetup@1000000002 : Record 91;
      GLSetup@1000000003 : Record 98;
    BEGIN
      IF (AllowPostingFrom = 0D) AND (AllowPostingTo = 0D) THEN BEGIN
       IF USERID <> '' THEN
         IF UserSetup.GET(USERID) THEN BEGIN
           AllowPostingFrom := UserSetup."Allow Posting From";
           AllowPostingTo := UserSetup."Allow Posting To";
         END;
         IF (AllowPostingFrom = 0D) AND (AllowPostingTo = 0D) THEN BEGIN
          GLSetup.GET;
          AllowPostingFrom := GLSetup."Allow Posting From";
          AllowPostingTo := GLSetup."Allow Posting To";
         END;
         IF AllowPostingTo = 0D THEN
           AllowPostingTo := 31129999D;
         END;
      EXIT((PostingDate < AllowPostingFrom) OR (PostingDate > AllowPostingTo));
    END;
