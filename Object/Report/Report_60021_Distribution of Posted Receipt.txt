    PROCEDURE MakeDistributionLine@60006();
    VAR
      "<-- 3LI TEXT CONSTANTS -->"@60001 : TextConst;
      Text60000@60000 : TextConst 'ENU=There is not enough space to insert extended text lines.;FRA=Il n''y a pas suffisamment de place pour ins'rer des lignes texte 'tendu.';
      "<< -- 3LI LOCALS -->>"@60002 : Integer;
      PtdReceiptLine_L@60008 : Record 8053104;
      ParamRepartition_L@60007 : Record 60008;
      PtdReceiptLine2_L@60006 : Record 8053104;
      LineSpacing_L@60005 : Integer;
      NextLineNo_L@60004 : Integer;
      OriginalQty_L@60003 : Decimal;
      FirstLine_L@60009 : Boolean;
      PtdReceiptLineUpd_L@1000000000 : Record 8053104;
      Vendor_L@1000000001 : Record 23;
      DistributedQty_L@1000000002 : Decimal;
    BEGIN
      // This function will create lines of distribution for each CMC Posted Receipt Line for the criteria
      CLEAR(ParamRepartition_L);
      ParamRepartition_L.SETRANGE("Campaign No.","CMC Posted Receipts"."Campaign No.");
      ParamRepartition_L.SETRANGE("Starting Date",StartDate_G);
      ParamRepartition_L.SETRANGE("Vendor No.","CMC Posted Receipts"."Vendor No.");
      ParamRepartition_L.SETRANGE("Item Code","CMC Posted Receipts"."Item No.");
      IF ParamRepartition_L.FINDSET(FALSE, FALSE) THEN BEGIN
        PtdReceiptLine_L.INIT;
        PtdReceiptLine_L.RESET;
        PtdReceiptLine_L.SETCURRENTKEY("Receipt Type","No.");
        PtdReceiptLine_L.SETRANGE("Receipt Type","CMC Posted Receipts"."Receipt Type");
        PtdReceiptLine_L.SETRANGE("No.","CMC Posted Receipts"."No.");
        PtdReceiptLine_L.SETFILTER("Contract No.",'%1','');
        PtdReceiptLine_L.SETRANGE(Invoiced,FALSE);
        PtdReceiptLine_L.SETRANGE("Canceled Line",FALSE);
        PtdReceiptLine_L.SETRANGE("Cancellation line",FALSE);
        PtdReceiptLine_L.SETRANGE("Attached to Line No.",0);
        IF PtdReceiptLine_L.FINDSET(FALSE, FALSE) THEN BEGIN
          IF PtdReceiptLine_L."Distributed Vendor No." = PtdReceiptLine_L."Vendor No." THEN BEGIN
            ToRelease_G := FALSE;
            CLEAR(CMCStatusMgt_G);
          IF "CMC Posted Receipts".Status = "CMC Posted Receipts".Status::Released THEN BEGIN
              IF CallByICFunction_G THEN BEGIN
                "CMC Posted Receipts".Status := "CMC Posted Receipts".Status::Open;
                "CMC Posted Receipts".MODIFY;
              END ELSE
            CMCStatusMgt_G.OpenPostedReceipt("CMC Posted Receipts");
            ToRelease_G := TRUE;
          END;
            REPEAT
              PtdReceiptLineUpd_L := PtdReceiptLine_L;
              ParamRepartition_L.FINDSET(FALSE, FALSE); // Because of the loop lines it is necessary to find again
              OriginalQty_L := PtdReceiptLine_L.Quantity;
              PtdReceiptLineUpd_L.VALIDATE("Distributed Vendor No.",ParamRepartition_L."Distributed Vendor No.");
              // --> 20151116-CER15-TER01.01.00-2455-FBE01
              PtdReceiptLineUpd_L.Quantity := ROUND(PtdReceiptLineUpd_L.Quantity * ParamRepartition_L."Distribution %" / 100,
              //  G_GeneralLedgerSetup."Unit-Amount Rounding Precision");
                UnitAmountRoundingPrecision_G);
              DistributedQty_L := PtdReceiptLineUpd_L.Quantity  ;
              // <-- 20151116-CER15-TER01.01.00-2455-FBE01
              PtdReceiptLineUpd_L."Quantity To Invoice" := PtdReceiptLineUpd_L.Quantity;
              PtdReceiptLineUpd_L.MODIFY;

              PtdReceiptLine2_L.RESET;
              PtdReceiptLine2_L.SETRANGE("Receipt Type","CMC Posted Receipts"."Receipt Type");
              PtdReceiptLine2_L.SETRANGE("No.","CMC Posted Receipts"."No.");
              PtdReceiptLine2_L.SETFILTER("Line No.", '>%1', PtdReceiptLine_L."Line No.");
              IF PtdReceiptLine2_L.FINDFIRST THEN BEGIN
                LineSpacing_L := (PtdReceiptLine2_L."Line No." - PtdReceiptLine_L."Line No.") DIV (1 + ParamRepartition_L.COUNT);
                IF LineSpacing_L = 0 THEN
                  ERROR(Text60000);
              END ELSE
                LineSpacing_L := 10000;
              NextLineNo_L := PtdReceiptLine_L."Line No." + LineSpacing_L;

              FirstLine_L := TRUE;
              REPEAT
                IF NOT FirstLine_L THEN BEGIN
                  PtdReceiptLine2_L.INIT;
                  PtdReceiptLine2_L.TRANSFERFIELDS(PtdReceiptLineUpd_L);
                  PtdReceiptLine2_L."Item Ledger Entry No." := 0;
                  PtdReceiptLine2_L."Canceled Item Ledg. Entry No." := 0;
                  PtdReceiptLine2_L."Line No." := NextLineNo_L;
                  PtdReceiptLine2_L.VALIDATE("Distributed Vendor No.",ParamRepartition_L."Distributed Vendor No.");
                  // --> 20151116-CER15-TER01.01.00-2455-FBE01
                  PtdReceiptLine2_L.Quantity := ROUND(OriginalQty_L * ParamRepartition_L."Distribution %" / 100,
                  //  G_GeneralLedgerSetup."Unit-Amount Rounding Precision");
                    UnitAmountRoundingPrecision_G);
                  DistributedQty_L += PtdReceiptLine2_L.Quantity ;
                  // <-- 20151116-CER15-TER01.01.00-2455-FBE01
                  PtdReceiptLine2_L."Attached to Line No." := PtdReceiptLine_L."Line No.";
                  PtdReceiptLine2_L."Quantity To Invoice" := PtdReceiptLine2_L.Quantity;
                  PtdReceiptLine2_L."Difference Line" := PtdReceiptLine_L."Difference Line";
                  PtdReceiptLine2_L.INSERT;
                  NextLineNo_L := NextLineNo_L + LineSpacing_L;
                END ELSE
                  FirstLine_L := FALSE;
              UNTIL ParamRepartition_L.NEXT =0;
              // --> 20151116-CER15-TER01.01.00-2455-FBE01
              IF DistributedQty_L <> OriginalQty_L THEN BEGIN
                PtdReceiptLineUpd_L.Quantity := PtdReceiptLineUpd_L.Quantity + OriginalQty_L - DistributedQty_L;
                PtdReceiptLineUpd_L."Quantity To Invoice" := PtdReceiptLineUpd_L.Quantity;
                PtdReceiptLineUpd_L.MODIFY;
              END;
              // <-- 20151116-CER15-TER01.01.00-2455-FBE01
            UNTIL PtdReceiptLine_L.NEXT = 0;

            CMCStatusMgt_G.SetSkipCalibration(TRUE);
            CMCStatusMgt_G.ReleasePostedReceipt("CMC Posted Receipts");
            CMCStatusMgt_G.SetSkipCalibration(FALSE);
            "CMC Posted Receipts"."Receipt Distributed" := TRUE;
            "CMC Posted Receipts".MODIFY;
          END;
        END;
      END;
    END;
