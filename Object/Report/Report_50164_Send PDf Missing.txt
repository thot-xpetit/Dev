    LOCAL PROCEDURE AnalyzeData@2();
    VAR
      TempExcelBuf@1000 : TEMPORARY Record 370;
      HeaderRowNo@1003 : Integer;
      OldRowNo@1006 : Integer;
    BEGIN
      Window.OPEN(
        Text007 +
        '@1@@@@@@@@@@@@@@@@@@@@@@@@@\');
      Window.UPDATE(1,0);
      TotalRecNo := ExcelBuf.COUNT;
      RecNo := 0;

      Initialize();

      IF ExcelBuf.FIND('-') THEN BEGIN
        REPEAT
          RecNo := RecNo + 1;
          Window.UPDATE(1,ROUND(RecNo / TotalRecNo * 10000,1));
          CASE TRUE OF
            (STRPOS(ExcelBuf."Cell Value as Text",Text009) <> 0):
              BEGIN
                IF HeaderRowNo = 0 THEN BEGIN
                  HeaderRowNo := ExcelBuf."Row No.";
                  TempExcelBuf := ExcelBuf;
                  TempExcelBuf.Comment := Text009;
                  TempExcelBuf.INSERT;
                END ELSE
                  ERROR(Text011);
              END;

            ExcelBuf."Row No." = HeaderRowNo:
              BEGIN
                TempExcelBuf := ExcelBuf;
                TempExcelBuf.Comment := ExcelBuf."Cell Value as Text";
                TempExcelBuf.INSERT;
              END;

            (ExcelBuf."Row No." > HeaderRowNo) AND (HeaderRowNo > 0):
              BEGIN
                IF ExcelBuf."Row No." <> OldRowNo THEN BEGIN
                  OldRowNo := ExcelBuf."Row No.";
                  ClearVariable();
                END;

                TempExcelBuf.SETRANGE("Column No.",ExcelBuf."Column No.");
                IF TempExcelBuf.FIND('-') THEN BEGIN
                  CASE TempExcelBuf.Comment OF
                    Text010:
                      BEGIN
                        EVALUATE(LinkID, ExcelBuf."Cell Value as Text");
                      END;
                    Text015:
                      Desc := ExcelBuf."Cell Value as Text";
                    Text016: BEGIN
                      url1 := ExcelBuf."Cell Value as Text";
                      SendFile();
                    END;
                  END;
                END;
              END;
          END;
        UNTIL ExcelBuf.NEXT = 0;
      END;
      Finalize();

      Window.CLOSE;
    END;

    PROCEDURE GetYears@1000000004(Radical@1000000000 : Text[255];VAR Year1@1000000001 : Text[4];VAR Year2@1000000002 : Text[4]);
    VAR
      Position@1000000003 : Integer;
    BEGIN
      Year1 := '';
      Year2 := '';
      // on inverse la chaine entier afin de retrouver les dates en debut de chaine
      Radical := ReverseString(Radical);
      // on decoupe la chaine pour recuperer uniquement le dossier "Date"
      Position := STRPOS(Radical, '\');
      Radical := COPYSTR(Radical, Position + 1);
      Position := STRPOS(Radical, '\');
      Radical := COPYSTR(Radical, 1, Position - 1);
      // On peut maintenant recuperer les deux dates
      Year2 := ReverseString(COPYSTR(Radical, 1, 4));
      Year1 := ReverseString(COPYSTR(Radical, 5));
    END;

    PROCEDURE ReverseString@1000000003(StringToReverse@1000000000 : Text[255]) ReturnValue : Text[255];
    VAR
      index@1000000001 : Integer;
    BEGIN
      ReturnValue := '';
      index := STRLEN(StringToReverse);
      REPEAT
        ReturnValue := ReturnValue + FORMAT(StringToReverse[index]);
        index := index - 1;
      UNTIL (index = 0)
    END;
