                           // Calcul du stock . la date d''quilibrage pour chaque d'p"t hors transit
                           NegStock_G := FALSE;
                           IF RecLocation_G.FINDSET THEN
                             REPEAT
                               //Calcul du stock global du d'p"t
                               RecItemLdgEntry_G.SETRANGE("Item No.","No.");
                               RecItemLdgEntry_G.SETRANGE("Location Code",RecLocation_G.Code);
                               RecItemLdgEntry_G.SETRANGE("Posting Date",0D,BalancingDate_G);
                               RecItemLdgEntry_G.CALCSUMS(Quantity);
                               NegStock_G := RecItemLdgEntry_G.Quantity < 0;
                               IF RecItemLdgEntry_G.Quantity >= 0 THEN BEGIN
                                 //Calcul du stock soci't' non lettr' du d'p"t
                                 RecItemLdgEntry_G.SETRANGE("Vendor Storage",FALSE);
                                 RecItemLdgEntry_G.CALCSUMS("Remaining Quantity");
                                 //Buffer 1 -> D'p"t ayant un stock soci't' > 0
                                 IF RecItemLdgEntry_G."Remaining Quantity" > 0 THEN BEGIN
                                   RecBalCerealInvBuffer1_G."Location Code" := RecLocation_G.Code;
                                   RecBalCerealInvBuffer1_G."Cereal Inventory" := RecItemLdgEntry_G."Remaining Quantity";
                                   RecBalCerealInvBuffer1_G.INSERT;
                                 //Buffer 2 -> D'p"t ayant un stock soci't' < 0
                                 END ELSE IF RecItemLdgEntry_G."Remaining Quantity" < 0 THEN BEGIN
                                   RecBalCerealInvBuffer2_G."Location Code" := RecLocation_G.Code;
                                   RecBalCerealInvBuffer2_G."Cereal Inventory" := RecItemLdgEntry_G."Remaining Quantity";
                                   RecBalCerealInvBuffer2_G.INSERT;
                                 END;
                               END;
                               RecItemLdgEntry_G.SETRANGE("Vendor Storage");
                             UNTIL (RecLocation_G.NEXT = 0); //OR NegStock_G;

                           //Si le stock d'un d'p"t est n'gatif , on affiche l'article en fin de traitement
                           IF NegStock_G THEN BEGIN
                             RecItemBuffer_G := Item;
                             RecItemBuffer_G.INSERT;
                           END;
                           //Si aucun d'p"t n'a de stock soci't' n'gatif, on ne traite pas l'article
                           //IF NegStock_G OR RecBalCerealInvBuffer2_G.ISEMPTY THEN
                           IF RecBalCerealInvBuffer2_G.ISEMPTY THEN
                             CurrReport.SKIP;

                           IF RecBalCerealInvBuffer1_G.FINDSET THEN
                             REPEAT
                               IF RecBalCerealInvBuffer2_G.FINDSET THEN
                                 REPEAT
                                   //Le stock soci't' du d'p"t Buffer 1 couvre entiSrement le stock soci't' du d'p"t Buffer 2
                                   IF RecBalCerealInvBuffer1_G."Cereal Inventory" >= ABS(RecBalCerealInvBuffer2_G."Cereal Inventory") THEN BEGIN
                                     //Transfert de stock soci't' d'p"t Buffer 1 -> d'p"t Buffer 2
                                     PostTransfer(ABS(RecBalCerealInvBuffer2_G."Cereal Inventory"));
                                     //On d'duit du d'p"t Buffer 1 le stock soci't' du d'p"t Buffer 2
                                     RecBalCerealInvBuffer1_G."Cereal Inventory" += RecBalCerealInvBuffer2_G."Cereal Inventory";
                                     RecBalCerealInvBuffer1_G.MODIFY;
                                     //Suppression du d'p"t Buffer 2
                                     RecBalCerealInvBuffer2_G.DELETE;
                                   //Le stock soci't' du d'p"t Buffer 1 couvre partiellement le stock soci't' du d'p"t Buffer 2
                                   END ELSE BEGIN
                                     //Transfert de stock soci't' d'p"t Buffer 1 -> d'p"t Buffer 2
                                     PostTransfer(RecBalCerealInvBuffer1_G."Cereal Inventory");
                                     //On d'duit du d'p"t Buffer 2 la quantit' transf'r'e
                                     RecBalCerealInvBuffer2_G."Cereal Inventory" += RecBalCerealInvBuffer1_G."Cereal Inventory";
                                     RecBalCerealInvBuffer2_G.MODIFY;
                                     //On d'duit du d'p"t Buffer 1 le stock soci't' du d'p"t Buffer 2
                                     RecBalCerealInvBuffer1_G."Cereal Inventory" := 0;
                                     RecBalCerealInvBuffer1_G.MODIFY;
                                   END;
                                 UNTIL (RecBalCerealInvBuffer2_G.NEXT = 0) OR (RecBalCerealInvBuffer1_G."Cereal Inventory" = 0);
                             UNTIL RecBalCerealInvBuffer1_G.NEXT = 0;
                         END;

        OnPostDataItem=BEGIN

                         COMMIT;
