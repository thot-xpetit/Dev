    PROCEDURE UpdateDialog@8053000();
    VAR
      LastTreatment_G@8053003 : Integer;
      LastCustomer_G@8053002 : Integer;
      LastTemplate_G@8053001 : Integer;
    BEGIN
      IF CountTreatment_G <> 0 THEN BEGIN
        IF ABS(ROUND((LastTreatment_G - NoTreatment_G) / CountTreatment_G * 9999, 1, '=')) >= 99 THEN BEGIN
          Dialog_G.UPDATE(1, ROUND((NoTreatment_G / CountTreatment_G) * 9999, 1, '='));
          LastTreatment_G := NoTreatment_G;
        END;
      END ELSE BEGIN
        Dialog_G.UPDATE(1, 0);
        LastTreatment_G := NoTreatment_G;
      END;

      IF CountCustomer_G <> 0 THEN BEGIN
        IF ABS(ROUND((LastCustomer_G - NoCustomer_G) / CountCustomer_G * 9999, 1, '=')) >= 99 THEN BEGIN
          Dialog_G.UPDATE(2, ROUND((NoCustomer_G / CountCustomer_G) * 9999, 1, '='));
          LastCustomer_G := NoCustomer_G;
        END;
      END ELSE BEGIN
        Dialog_G.UPDATE(2, 0);
        LastCustomer_G := NoCustomer_G;
      END;

      IF CountTemplate_G <> 0 THEN BEGIN
        IF ABS(ROUND((LastTemplate_G - NoTemplate_G) / CountTemplate_G * 9999, 1, '=')) >= 99 THEN BEGIN
          Dialog_G.UPDATE(3, ROUND((NoTemplate_G / CountTemplate_G) * 9999, 1, '='));
          LastTemplate_G := NoTemplate_G;
        END;
      END ELSE BEGIN
        Dialog_G.UPDATE(3, 0);
        LastTemplate_G := NoTemplate_G;
      END;
    END;

    PROCEDURE GenerateContract@8053011();
    VAR
      recContract_L@8053000 : Record 8052983;
      cduStatusMgt_L@8053001 : Codeunit 8052985;
    BEGIN
      recListContractByFarmerTmp_G.RESET;
      recListContractByFarmerTmp_G.DELETEALL;

      recContractTemp_G.RESET;
      IF recContractTemp_G.FINDSET(FALSE, FALSE) THEN
        REPEAT
          //Cr'ation du contrat
          CLEAR(recContract_L);
          recContract_L.INIT;
          recContract_L.Type := recContract_L.Type::Contract;
          recContract_L.Description := Designation_G;
          recContract_L.INSERT(TRUE);
          recContract_L.VALIDATE("Campaign No.", recContractTemp_G."Campaign No.");
          recContract_L.VALIDATE("Contract Type", recContractTemp_G."Contract Type");
          recContract_L.VALIDATE("Vendor No.", diCustomer."No.");
          recContract_L.VALIDATE("Salesperson Code", recContractTemp_G."Salesperson Code");
          recContract_L.VALIDATE("Purchaser Code", recContractTemp_G."Purchaser Code");
          recContract_L.VALIDATE("Species Code", recContractTemp_G."Species Code");
          recContract_L.VALIDATE("Item No.", recContractTemp_G."Item No.");
          recContract_L.VALIDATE("Ship Starting Date", recContractTemp_G."Ship Starting Date");
          recContract_L.VALIDATE("Ship Ending Date", recContractTemp_G."Ship Ending Date");
          recContract_L.VALIDATE(Quantity, recContractTemp_G.Quantity);
          recContract_L.VALIDATE("Boni/Refaction Less", recContractTemp_G."Boni/Refaction Less");
          recContract_L.VALIDATE("Bonis Refaction Code", recContractTemp_G."Bonis Refaction Code");
          recContract_L.VALIDATE("Default Location Code", recContractTemp_G."Default Location Code");
          recContract_L.VALIDATE("Unit Price", recContractTemp_G."Unit Price");
          recContract_L.VALIDATE("Deposit Fee", recContractTemp_G."Deposit Fee");
          recContract_L.VALIDATE("Engagment Bonus", recContractTemp_G."Engagment Bonus");
          IF recContract_L."Default Location Code" <> '' THEN
            recContract_L.VALIDATE("Approach Bonus", recContractTemp_G."Approach Bonus");
          recContract_L.VALIDATE("Price Supplement", recContractTemp_G."Price Supplement");
          recContract_L.VALIDATE("Bonus Due Code", recContractTemp_G."Bonus Due Code");
          recContract_L.MODIFY(TRUE);

          //v'rouillage
          cduStatusMgt_L.FinishContract(recContract_L);

          //Remplissage de la liste des contrat de l'agriculteur
          recListContractByFarmerTmp_G.INIT;
          recListContractByFarmerTmp_G.TRANSFERFIELDS(recContract_L, TRUE);
          recListContractByFarmerTmp_G.Description := recContractTemp_G."No.";
          recListContractByFarmerTmp_G.INSERT;
        UNTIL recContractTemp_G.NEXT = 0;
    END;
