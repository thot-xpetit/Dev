    PROCEDURE CalcTotalVAT@8060989();
    VAR
      VatPositionLine_L@8060982 : Integer;
      VATLCYExist_L@8060983 : Boolean;
    BEGIN
      {//Permet de calculer la total TVA

      // --> 3LI-20110318-8060990/XXX-CMB6.00.01.01-JDL01
      //Calcul du total
      FOR VatPositionLine_L := 1 TO ARRAYLEN(BaseTva_G) - 1 DO BEGIN
        BaseTva_G[ARRAYLEN(BaseTva_G)] += BaseTva_G[VatPositionLine_L];
        MtTVA_G[ARRAYLEN(BaseTva_G)] += MtTVA_G[VatPositionLine_L];
        BaseTvaLCY_G[ARRAYLEN(BaseTva_G)] += BaseTvaLCY_G[VatPositionLine_L];
        MtTVALCY_G[ARRAYLEN(BaseTva_G)] += MtTVALCY_G[VatPositionLine_L];
      END;


      // Gestion en-t^te et total TVA (d'signation)
      IF (BaseTva_G[ARRAYLEN(BaseTva_G)] = 0) THEN BEGIN
        VATIdentifier_G[ARRAYLEN(BaseTva_G)] := ''
      END ELSE BEGIN
        VATIdentifier_G[ARRAYLEN(BaseTva_G)] := Text8060993;
        DescEntTVA_G[1] := VATAmountLine.FIELDCAPTION("VAT Identifier");
        DescEntTVA_G[3] := VATAmountLine.FIELDCAPTION("VAT %");
        IF (MtTVALCY_G[ARRAYLEN(BaseTva_G)] <> 0) OR (BaseTvaLCY_G[ARRAYLEN(BaseTva_G)] <> 0) THEN BEGIN
          //D'tail TVA LCY demand'
          DescEntTVA_G[2] := STRSUBSTNO('%1 %2',VATAmountLine.FIELDCAPTION("VAT Base"),Text8060996);
          DescEntTVA_G[4] := STRSUBSTNO('%1 %2',VATAmountLine.FIELDCAPTION("VAT Amount"),Text8060996);
          VATLCYExist_L := TRUE;
        END ELSE BEGIN
          DescEntTVA_G[2] := VATAmountLine.FIELDCAPTION("VAT Base");
          DescEntTVA_G[4] := VATAmountLine.FIELDCAPTION("VAT Amount");
          VATLCYExist_L := FALSE;
        END;
        // Gestion des 2 valeurs TVA dans une colonne
        FOR VatPositionLine_L := 1 TO ARRAYLEN(BaseTva_G) DO BEGIN
          IF BaseTva_G[VatPositionLine_L] <> 0 THEN BEGIN
            IF VATLCYExist_L THEN BEGIN
              BaseTvaAsText_G[VatPositionLine_L] := STRSUBSTNO('%1 (%2)',BaseTva_G[VatPositionLine_L],BaseTvaLCY_G[VatPositionLine_L]);
              MtTVAAsText_G[VatPositionLine_L] := STRSUBSTNO('%1 (%2)',MtTVA_G[VatPositionLine_L],MtTVALCY_G[VatPositionLine_L]);
            END ELSE BEGIN
              BaseTvaAsText_G[VatPositionLine_L] := STRSUBSTNO('%1',BaseTva_G[VatPositionLine_L]);
              MtTVAAsText_G[VatPositionLine_L] := STRSUBSTNO('%1',MtTVA_G[VatPositionLine_L]);
            END;
          END;
        END;
      END;
      // <-- 3LI-20110318-8060990/XXX-CMB6.00.01.01-JDL01
      }
    END;
