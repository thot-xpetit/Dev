                             // Code repris de la table 38 - Doc. Category Code - OnLookup()
                             CLEAR(DocCategoryFilter_L);
                             CMBDocCategoryUser_L.SETRANGE(Type, CMBDocCategoryUser_L.Type::User);
                             CMBDocCategoryUser_L.SETRANGE(Code, USERID);
                             CMBDocCategoryUser_L.SETRANGE("Sales/Purchase", CMBDocCategoryUser_L."Sales/Purchase"::Purchase);
                             CMBDocCategoryUser_L.SETRANGE("Document Type", CMBDocCategoryUser_L."Document Type"::"Blanket Order");
                             IF CMBDocCategoryUser_L.FINDSET THEN REPEAT
                               IF DocCategoryFilter_L = '' THEN
                                 DocCategoryFilter_L := CMBDocCategoryUser_L."Doc. Category Code"
                               ELSE
                               DocCategoryFilter_L := DocCategoryFilter_L + '|' + CMBDocCategoryUser_L."Doc. Category Code";
                             UNTIL CMBDocCategoryUser_L.NEXT = 0;

                             IF DocCategoryFilter_L = '' THEN
                               IF UserSetup_L.GET(USERID) THEN
                               BEGIN
                                 CMBDocCategoryUser_L.SETRANGE(Type, CMBDocCategoryUser_L.Type::"User Group");
                                 CMBDocCategoryUser_L.SETRANGE(Code, UserSetup_L."User Groupe Code");
                                 IF CMBDocCategoryUser_L.FINDSET THEN REPEAT
                                   IF DocCategoryFilter_L = '' THEN
                                     DocCategoryFilter_L := CMBDocCategoryUser_L."Doc. Category Code"
                                   ELSE
                                   DocCategoryFilter_L := DocCategoryFilter_L + '|' + CMBDocCategoryUser_L."Doc. Category Code";
                                 UNTIL CMBDocCategoryUser_L.NEXT = 0;
                               END;

                             CMBDocCategory_L.RESET;
                             CMBDocCategory_L.FILTERGROUP(2);
                             CMBDocCategory_L.SETRANGE("Sales/Purchase", CMBDocCategory_L."Sales/Purchase"::Purchase);
                             CMBDocCategory_L.SETRANGE("Document Type",CMBDocCategoryUser_L."Document Type"::"Blanket Order");
                             CMBDocCategory_L.SETFILTER(Code, DocCategoryFilter_L);
                             CMBDocCategory_L.FILTERGROUP(0);
                             CMBDocCategoryPage_L.SETTABLEVIEW(CMBDocCategory_L);
                             CMBDocCategoryPage_L.EDITABLE(FALSE);
                             CMBDocCategoryPage_L.LOOKUPMODE(TRUE);
                             IF CMBDocCategoryPage_L.RUNMODAL = ACTION::LookupOK THEN BEGIN
                               CMBDocCategoryPage_L.GETRECORD(CMBDocCategory_L);
                               NewDocCategoryCode_G := CMBDocCategory_L.Code;
                             END;
                           END;
                            }
