    PROCEDURE ReadingPurchaseContract@60001();
    VAR
      "<<-- 3LI LOCALS -->"@60004 : Integer;
      PurchInvLine_L@60003 : Record 123;
      PurchCrMemoLine_L@60002 : Record 125;
      IgnoreLine_L@60001 : Boolean;
      LineNo_L@60000 : Integer;
      Price_L@60005 : Decimal;
      CollectionType_P@60006 : Integer;
      QtyContract_L@1000000000 : Decimal;
      Prix_tmp_L@1000000001 : Decimal;
      CMCSupplementDetail_L@1000000002 : Record 60010;
    BEGIN
      // Inserting Lines from CMC Collection Contract
      CLEAR(CMCCollContract_G);

      CMCCollContract_G.SETCURRENTKEY(Type,"Campaign No.","Species Code","Item No.",Status,"Contract Type");
      CMCCollContract_G.SETRANGE(Type,CMCCollContract_G.Type::Contract);
      CMCCollContract_G.SETRANGE("Campaign No.",CampaignNo_G);
      TotalCount_G := CMCCollContract_G.COUNT;
      Counter_G := 0;

      IF CMCCollContract_G.FINDSET THEN  REPEAT
          Counter_G := Counter_G + 1;
          Window_G.UPDATE(1,ROUND(Counter_G / TotalCount_G * 10000, 1));

          IgnoreLine_L := FALSE;

          CLEAR(GroupingCereales_G);
          GroupingCereales_G.SETRANGE("Campaign No.",CMCCollContract_G."Campaign No.");
          GroupingCereales_G.SETRANGE("Species Code",CMCCollContract_G."Species Code");
          GroupingCereales_G.SETRANGE("Item No.",CMCCollContract_G."Item No.");
          IF GroupCode_G <> '' THEN BEGIN
            GroupingCereales_G.SETRANGE("Grouping Cereal Code",GroupCode_G);
            IF NOT GroupingCereales_G.FINDFIRST THEN IgnoreLine_L := TRUE;
          END ELSE BEGIN
            IF GroupingCereales_G.FINDFIRST THEN;
          END;

          IF PurchContractType_G.GET(CMCCollContract_G."Contract Type") THEN;
          IF Option_G = Option_G::"Calculation Average Price" THEN BEGIN
            IF PurchContractType_G."Price Type" <> PurchContractType_G."Price Type"::Prepayment THEN IgnoreLine_L := TRUE;
          END ELSE BEGIN
            IF PurchContractType_G."Price Type" <> PurchContractType_G."Price Type"::"Contract price" THEN IgnoreLine_L := TRUE;
          END;

          IF NOT IgnoreLine_L THEN BEGIN

            CMCCollContract_G.CALCFIELDS("Quantity Received");

            InsertLine(GroupingCereales_G."Grouping Cereal Code",CMCCollContract_G."Quantity Received",
                                0,0,0,0,0,0,0);

            // lire les r'ceptions
            CLEAR(CMCPtdReceiptLine_G);
            CMCPtdReceiptLine_G.SETCURRENTKEY("Contract No.","Variety Code");
            CMCPtdReceiptLine_G.SETRANGE("Contract No.",CMCCollContract_G."No.");
            CMCPtdReceiptLine_G.SETFILTER(CMCPtdReceiptLine_G."Collection Type",'%1|%2',
              CMCPtdReceiptLine_G."Collection Type"::Storage,CMCPtdReceiptLine_G."Collection Type"::Purchase);
            CMCPtdReceiptLine_G.SETRANGE(CMCPtdReceiptLine_G."Canceled Line",FALSE);
            CMCPtdReceiptLine_G.SETRANGE(CMCPtdReceiptLine_G."Cancellation line",FALSE);

            IF CMCPtdReceiptLine_G.FINDFIRST THEN REPEAT
              CMCPtdReceiptHdr_G.GET(CMCPtdReceiptLine_G."Receipt Type",CMCPtdReceiptLine_G."No.");
              IF CMCPtdReceiptLine_G.Invoiced THEN BEGIN
                CASE CMCPtdReceiptLine_G."Posted Purch. Document Type" OF

                  CMCPtdReceiptLine_G."Posted Purch. Document Type"::"Posted invoice": BEGIN
                    IF PurchInvLine_L.GET(CMCPtdReceiptLine_G."Posted Purch. Document No",
                                    CMCPtdReceiptLine_G."Posted Purch. Doc. Line No") THEN;
                    Price_L := PurchInvLine_L."Direct Unit Cost";
                  END;

                  CMCPtdReceiptLine_G."Posted Purch. Document Type"::"Posted Credit Memo": BEGIN
                    IF PurchCrMemoLine_L.GET(CMCPtdReceiptLine_G."Posted Purch. Document No",
                                    CMCPtdReceiptLine_G."Posted Purch. Doc. Line No") THEN;
                    Price_L := PurchCrMemoLine_L."Direct Unit Cost";
                  END;

                END;
              END ELSE
                Price_L := GetUnitPriceFromReceiptLine(CMCPtdReceiptHdr_G,CMCPtdReceiptLine_G);
              // --> 20161207-CER77-TER01.01.00-3162-OLE01 : MODCODE
              { BEFORE
              InsertLine(GroupingCereales_G."Grouping Cereal Code",0,0,0,0,
                        CMCPtdReceiptLine_G.Quantity * Price_L+F_Complement(),
                        0,0,0);
              AFTER }
              InsertLine(GroupingCereales_G."Grouping Cereal Code",0,0,0,0,
                        CMCPtdReceiptLine_G.Quantity * (Price_L+F_Complement()),
                        0,0,0);
              // <-- 20161207-CER77-TER01.01.00-3162-OLE01 : MODCODE

            UNTIL CMCPtdReceiptLine_G.NEXT = 0;

            // to get the not received
            IF CMCCollContract_G.Quantity - CMCCollContract_G."Quantity Received" > 0 THEN BEGIN

              IF CMCCollContract_G.Status <> CMCCollContract_G.Status::Finished THEN BEGIN
                CLEAR(CMCSupplementDetail_L);
                CMCSupplementDetail_L.SETRANGE(Code,CMCCollContract_G."Supplement Code");
                CMCSupplementDetail_L.SETRANGE(Campaign,CMCCollContract_G."Campaign No.");
                CMCSupplementDetail_L.SETFILTER("Starting Date",'%1|<=%2',0D,CMCCollContract_G."Ship Ending Date");
                IF CMCSupplementDetail_L.FINDLAST THEN
                  CMCCollContract_G.Supplement := CMCSupplementDetail_L."Supplement Value";
                InsertLine(GroupingCereales_G."Grouping Cereal Code",(CMCCollContract_G.Quantity - CMCCollContract_G."Quantity Received"),
                          0,0,0,0,
                          (CMCCollContract_G.Quantity - CMCCollContract_G."Quantity Received")
                              * (CMCCollContract_G."Unit Price" + CMCCollContract_G.Supplement +
                              CMCCollContract_G."Transport Cost" + CMCCollContract_G."Engagment Bonus"),
                          0,0);
              END;
            END;

          END;
        UNTIL CMCCollContract_G.NEXT = 0;
    END;

    PROCEDURE ReadingSalesContract@60009();
    VAR
      "<<-- 3LI LOCALS -->"@60001 : Integer;
      RecAvailableCalcDtl_L@60003 : Record 60047;
      CMCSalesContractDist_L@60009 : Record 60044;
      CMCMATIFContractTmp_L@60011 : TEMPORARY Record 8053065;
      AvailableCalcDet_L@60013 : Record 60047;
      AvailableCalcDetTemp_L@60014 : TEMPORARY Record 60047;
      PurchInvLine_L@1000000000 : Record 123;
      IgnoreLine_L@60000 : Boolean;
      LineNo_L@60002 : Integer;
      Qty_L@60004 : Decimal;
      CalcRefPrice_L@60006 : Decimal;
      Qty2_L@60007 : Decimal;
      Price_L@60008 : Decimal;
      NextLoop_L@60010 : Boolean;
      RemainingQty_L@60012 : Decimal;
      LotPurch_L@1000000001 : Decimal;
      LotSales_L@1000000002 : Decimal;
    BEGIN
      // Input
      // Output
      // This function will read the CMC Sales Contract filtered by the filters of the request page and will fill the table
      // 60047

      CLEAR(SalesContract_G);

      SalesContract_G.SETCURRENTKEY("Campaign No.","Commercial Type");

      SalesContract_G.SETRANGE("Campaign No.",CampaignNo_G);
      IF CommercialType_G <> '' THEN   SalesContract_G.SETRANGE("Commercial Type",CommercialType_G);
      TotalCount_G := SalesContract_G.COUNT;
      Counter_G := 0;

      IF SalesContract_G.FINDFIRST THEN REPEAT

        SalesContract_G.CalcOpenPosition;
        Counter_G := Counter_G + 1;
        Window_G.UPDATE(2,ROUND(Counter_G / TotalCount_G * 10000, 1));

        IgnoreLine_L := FALSE;
        CLEAR(GroupingCereales_G);
        GroupingCereales_G.SETRANGE("Campaign No.",SalesContract_G."Campaign No.");
        GroupingCereales_G.SETRANGE("Species Code",SalesContract_G."Species Code");
        GroupingCereales_G.SETRANGE("Item No.",SalesContract_G."Item No.");

        IF GroupCode_G <> '' THEN BEGIN
          GroupingCereales_G.SETRANGE("Grouping Cereal Code",GroupCode_G);
          IF NOT GroupingCereales_G.FINDFIRST THEN IgnoreLine_L := TRUE;
        END ELSE BEGIN
          IF GroupingCereales_G.FINDFIRST THEN;
        END;

        SalesContract_G.CALCFIELDS(Distributed,"Calculated Reference Price","Quantity Received");

        CLEAR(CMCCommTypeSales_G);
        IF CMCCommTypeSales_G.GET(SalesContract_G."Commercial Type") THEN;

         IF Option_G = Option_G::"Calculation Average Price" THEN BEGIN
          IF CMCCommTypeSales_G."Price Type" <> CMCCommTypeSales_G."Price Type"::Advance THEN IgnoreLine_L := TRUE;
        END ELSE BEGIN
          IF CMCCommTypeSales_G."Price Type" <> CMCCommTypeSales_G."Price Type"::"Contract price" THEN IgnoreLine_L := TRUE;
        END;

        IF NOT IgnoreLine_L THEN BEGIN
          IF SalesContract_G.Distributed THEN BEGIN
            InsertDistributionLine;
          END ELSE BEGIN
            CASE SalesContract_G."Contract Type" OF
              // Vente . Prime et Achat . Prime
              SalesContract_G."Contract Type"::"Firm Sales Price",SalesContract_G."Contract Type"::"Firm Purch. Price": BEGIN

                IF SalesContract_G.Status <> SalesContract_G.Status::Closed THEN
                  Price_L := SalesContract_G."Reference Price"
                ELSE
                  Price_L := SalesContract_G."Calculated Reference Price";
                IF SalesContract_G."Contract Type" =SalesContract_G."Contract Type"::"Firm Sales Price" THEN
                  InsertLine(GroupingCereales_G."Grouping Cereal Code",0,SalesContract_G.Quantity,
                        0,0,0,0,SalesContract_G.Quantity * Price_L,
                        SalesContract_G.Quantity)
                ELSE
                  InsertLine(GroupingCereales_G."Grouping Cereal Code",0,-SalesContract_G.Quantity,
                        0,0,0,0,-SalesContract_G.Quantity * Price_L,
                        -SalesContract_G.Quantity)
              END;

              // Contrats . prime
              // =================
              SalesContract_G."Contract Type"::"Sales Bonus",SalesContract_G."Contract Type"::"Purch. Bonus": BEGIN
                //  gestion de l'estimation du non r'alis'
                IF SalesContract_G.Bonus <> '' THEN BEGIN
                  SalesContract_G.CalcBonusPrime;
                  SalesContract_G.MODIFY;
                END;

                IF SalesContract_G.Status = SalesContract_G.Status::Closed THEN BEGIN
                  Price_L := SalesContract_G."Calculated Reference Price";
                  Qty_L :=SalesContract_G.Quantity; //<-- 20161123-Wi00-TER-23259-JLT01
                END ELSE BEGIN
                  Price_L := SalesContract_G."Reference Price";
                  // --> 20161124-Wi00-TER-23259-OLE01
                  {BEFORE
                  Qty_L := (SalesContract_G."Lot Number" - SalesContract_G."Open Position")* CMCSetup_G."MATIF Weight per Lot";
                  AFTER}
                  Qty_L := (SalesContract_G."Lot Number" - ABS(SalesContract_G."Open Position"))* CMCSetup_G."MATIF Weight per Lot";
                  // <-- 20161124-Wi00-TER-23259-OLE01
                  //<-- 20161123-Wi00-TER-23259-JLT01
                  END;
                IF SalesContract_G."Contract Type"=SalesContract_G."Contract Type"::"Sales Bonus" THEN
                  InsertLine(GroupingCereales_G."Grouping Cereal Code",0,
                          Qty_L,0,0,0,0,
                          Qty_L*Price_L+ SalesContract_G."Estimated Profit and Loss", // estimation prime
                          Qty_L)
                ELSE
                  InsertLine(GroupingCereales_G."Grouping Cereal Code",0,
                          -Qty_L,0,0,0,0,
                          -Qty_L*Price_L+ SalesContract_G."Estimated Profit and Loss", // estimation prime
                          -Qty_L);
              END;

            END;

            //primes long / short
            IF SalesContract_G."Contract Type" IN [SalesContract_G."Contract Type"::"Purch. Bonus",
                                                   SalesContract_G."Contract Type"::"Sales Bonus"] THEN BEGIN

              InsertLine(GroupingCereales_G."Grouping Cereal Code",0,0,0,
                        CMCSetup_G."MATIF Weight per Lot" * (-SalesContract_G."Open Position") ,
                        0,0,0,0);
            END;

          END;
        END;

      UNTIL SalesContract_G.NEXT = 0;
    END;

        REPEAT
