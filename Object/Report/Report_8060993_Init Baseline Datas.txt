    PROCEDURE InitGLEntry@8060982();
    VAR
      GLEntry_L@8060982 : Record 17;
      DebitAmount_L@8060983 : Decimal;
      CreditAmount_L@8060984 : Decimal;
      GLEntry2_L@8060985 : Record 17;
      UpdateGLEntry_L@8060986 : Codeunit 8060993;
    BEGIN
      GLEntry_L.RESET;

      CurrentRec_G := 0;
      TotalRec_G := GLEntry_L.COUNT;

      IF GLEntry_L.FINDSET THEN
        REPEAT

          CurrentRec_G := CurrentRec_G + 1;
          UpdateWindow(2,CurrentRec_G,TotalRec_G);

          IF GLEntry_L.Letter <> '' THEN BEGIN

            DebitAmount_L := 0;
            CreditAmount_L := 0;

            GLEntry2_L.RESET;
            GLEntry2_L.SETCURRENTKEY("G/L Account No.",Letter);
            GLEntry2_L.SETRANGE("G/L Account No.",GLEntry_L."G/L Account No.");
            GLEntry2_L.SETRANGE(Letter,GLEntry_L.Letter);
            IF GLEntry2_L.FINDSET THEN
              REPEAT
                DebitAmount_L := DebitAmount_L + GLEntry2_L."Debit Amount";
                CreditAmount_L := CreditAmount_L - GLEntry2_L."Credit Amount";
              UNTIL GLEntry2_L.NEXT = 0;

            UpdateGLEntry_L.UpdateGLApplied(GLEntry_L,DebitAmount_L,CreditAmount_L);
            GLEntry_L.MODIFY;

          END;

        UNTIL GLEntry_L.NEXT = 0;
    END;

    PROCEDURE InitContactCustVendNo@8060989();
    VAR
      ContBusinessRelation_L@8060982 : Record 5054;
      Contact_L@8060983 : Record 5050;
      BusinessRelation_L@8060984 : Record 5053;
    BEGIN
      // --> 20111118-8060990/046-CMB6.00.10.04-JDL01
      ContBusinessRelation_L.RESET;

      CurrentRec_G := 0;
      TotalRec_G := ContBusinessRelation_L.COUNT;

      IF NOT ContBusinessRelation_L.ISEMPTY THEN BEGIN
        REPEAT

          CurrentRec_G := CurrentRec_G + 1;
          UpdateWindow(2,CurrentRec_G,TotalRec_G);


          IF (ContBusinessRelation_L."Link to Table" = ContBusinessRelation_L."Link to Table"::Customer) AND
             (ContBusinessRelation_L."No." <> '')
          THEN BEGIN
            IF Contact_L.GET(ContBusinessRelation_L."Contact No.") THEN BEGIN
              Contact_L."Customer No." := ContBusinessRelation_L."No.";
              Contact_L."Is Sales Contact" := TRUE;
              Contact_L.MODIFY;
            END;
          END;

          IF (ContBusinessRelation_L."Link to Table" = ContBusinessRelation_L."Link to Table"::Vendor) AND
             (ContBusinessRelation_L."No." <> '')
          THEN BEGIN
            IF Contact_L.GET(ContBusinessRelation_L."Contact No.") THEN BEGIN
              Contact_L."Vendor No." := ContBusinessRelation_L."No.";
              Contact_L."Is Purchase Contact" := TRUE;
              Contact_L.MODIFY;
            END;
          END;

          IF (ContBusinessRelation_L."Link to Table" = ContBusinessRelation_L."Link to Table"::"Bank Account") AND
             (ContBusinessRelation_L."No." <> '')
          THEN BEGIN
            IF Contact_L.GET(ContBusinessRelation_L."Contact No.") THEN BEGIN
              Contact_L."Bank Account No." := ContBusinessRelation_L."No.";
              Contact_L.MODIFY;
            END;
          END;

          //Mise . jour
          IF BusinessRelation_L.GET(ContBusinessRelation_L."Business Relation Code") THEN BEGIN
            ContBusinessRelation_L."Is Sales Contact" := BusinessRelation_L."Is Sales Contact";
            ContBusinessRelation_L."Is Purchase Contact" := BusinessRelation_L."Is Purchase Contact";
            ContBusinessRelation_L.MODIFY;
          END;
        UNTIL ContBusinessRelation_L.NEXT = 0;
      END;
