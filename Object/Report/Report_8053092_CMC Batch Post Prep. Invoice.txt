    PROCEDURE ProcessDepositAccount@8053005();
    VAR
      CurrBalanceLCY_L@8053000 : Decimal;
      ContNo_L@8053001 : Code[20];
      Contact_L@8053002 : Record 5050;
      recContactPayMethod_L@8053022 : Record 8053255;
      recPaymentmethod_L@8053021 : Record 289;
      recSalesHeader_L@8053017 : Record 36;
      OldPaymentMethod_L@8053016 : Code[10];
    BEGIN
      IF TmpCust_G.FINDSET THEN BEGIN
        REPEAT

          ContNo_L := CVLedgerMgt_G.GetContNoFromCustVentNo(TmpCust_G."No.",0);
          Contact_L.GET(ContNo_L);
          Contact_L.CALCFIELDS("Balance (LCY)");
          CurrBalanceLCY_L := -Contact_L."Balance (LCY)";

          TmpNewSalesHeader_G.RESET;
          TmpNewSalesHeader_G.SETRANGE("Document Type",TmpNewSalesHeader_G."Document Type"::"Credit Memo");
          TmpNewSalesHeader_G.SETRANGE("Bill-to Customer No.",TmpCust_G."No.");
          IF TmpNewSalesHeader_G.FINDSET THEN
            REPEAT
              CurrBalanceLCY_L := CurrBalanceLCY_L + TmpNewSalesHeader_G."Prepayment %";
            UNTIL TmpNewSalesHeader_G.NEXT = 0;

          TmpNewSalesHeader_G.SETRANGE("Document Type",TmpNewSalesHeader_G."Document Type"::Invoice);
          IF TmpNewSalesHeader_G.FINDSET THEN
            REPEAT
              IF TmpNewSalesHeader_G."Prepayment %" <= CurrBalanceLCY_L THEN BEGIN
                //Parcours des documents cr''
                IF recSalesHeader_L.GET(TmpNewSalesHeader_G."Document Type"::Invoice, TmpNewSalesHeader_G."No.") THEN BEGIN
                  IF recSalesHeader_L.Status <> recSalesHeader_L.Status::Released THEN BEGIN
                    OldPaymentMethod_L := recSalesHeader_L."Payment Method Code";
                    recContactPayMethod_L.RESET;
                    recContactPayMethod_L.SETRANGE("Contact No.", ContNo_L);
                    IF recContactPayMethod_L.FINDSET(FALSE, FALSE) THEN BEGIN
                      REPEAT
                        recPaymentmethod_L.GET(recContactPayMethod_L."Payment Method Code");
                      UNTIL (recContactPayMethod_L.NEXT=0) OR (recPaymentmethod_L."Depositor Account");
                      IF recPaymentmethod_L."Depositor Account" THEN BEGIN
                        //Le traitement ne fonctionne que si le client . un compte d'posant
                        recSalesHeader_L.SetSystemCreatedPayment(TRUE);
                        recSalesHeader_L.VALIDATE("Payment Method Code",recPaymentmethod_L.Code);
                        recSalesHeader_L.MODIFY;
                        CurrBalanceLCY_L := CurrBalanceLCY_L - TmpNewSalesHeader_G."Prepayment %";
                      END;
                    END;
                  END;
                END;
              END;
            UNTIL TmpNewSalesHeader_G.NEXT = 0;

        UNTIL TmpCust_G.NEXT = 0;

      END;
    END;
