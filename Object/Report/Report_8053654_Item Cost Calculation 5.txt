    PROCEDURE CalculateCostAmt@10(VAR Item@1000 : Record 27;LedgerEntryNo_P@1000000004 : Integer) : Decimal;
    VAR
      ValueEntry@1001 : Record 5802;
      CostActual@1000000000 : Decimal;
      CostExpected@1000000001 : Decimal;
      BooToManage_L@1000000002 : Boolean;
      ValueEntryTest_L@1000000003 : Record 5802;
    BEGIN
      // --> 20170418-Wi00-ADV01.01.00-3182-CJY01
      IF NOT BooExcludePurchInvoice THEN BEGIN
        WITH ValueEntry DO BEGIN
          SetFilters(ValueEntry,Item);
          CALCSUMS("Cost Amount (Actual)","Cost Amount (Expected)");
          EXIT("Cost Amount (Actual)" + "Cost Amount (Expected)");
        END;
      END
      ELSE BEGIN
        CostActual := 0;
        CostExpected := 0;
        WITH ValueEntry DO BEGIN
          FctSetFiltersOnLedEntryDate(ValueEntry,Item,LedgerEntryNo_P);

          IF ValueEntry.FINDSET THEN REPEAT
            BooToManage_L := TRUE;
            IF (("Document Type" = "Document Type"::"Purchase Invoice") OR
                ("Document Type" = "Document Type"::"Purchase Credit Memo") OR
                ("Document Type" = "Document Type"::"Purchase Receipt") OR
                ("Document Type" = "Document Type"::"Purchase Return Shipment"))
                THEN BEGIN
                  ValueEntryTest_L.SETRANGE("Entry No.","Entry No.");
                  ValueEntryTest_L.SETFILTER("Posting Date",Item.GETFILTER("Date Filter"));
                  IF ValueEntryTest_L.ISEMPTY THEN
                    BooToManage_L := FALSE;
            END;
            IF BooToManage_L THEN BEGIN
              CostActual := CostActual + "Cost Amount (Actual)";
              CostExpected := CostExpected + "Cost Amount (Expected)";
            END;
          UNTIL ValueEntry.NEXT = 0;
          EXIT(CostActual + CostExpected);
        END;

      END;
      // <-- 20170418-Wi00-ADV01.01.00-3182-CJY01
    END;
