    BEGIN
      MbrJnlLine_G."Capital Journal Template" := CurrTemplateName_G;
      MbrJnlLine_G."Capital Batch Name" := CurrWorksheetName_G;
      MbrJnlLine_G."Line No." := LastLineNo_G+10000;
      MbrJnlLine_G."Posting Date" := PostingDate_G;
      MbrJnlLine_G."Movement Type" := MvtType_G;

      MbrJnlLine_G."Campaign No." := CampagneNo_G;
      MbrJnlLine_G."Closing Date" := ArretDate_G;

        BEGIN
         CapJnlBatch_L.GET(CurrTemplateName_G,CurrWorksheetName_G);
         CapJnlBatch_L.TESTFIELD(CapJnlBatch_L."No. Series");
         MbrJnlLine_G."Document No." := NoSeries_G.GetNextNo(CapJnlBatch_L."No. Series",PostingDate_G,FALSE);
        END ELSE
          MbrJnlLine_G."Document No." := DocNo_G;
      MbrJnlLine_G."Account Type" := MbrJnlLine_G."Account Type"::Adh'rent;
      MbrJnlLine_G."Account No." := Contact_P."No.";
      MbrJnlLine_G.Description := Contact_P.Name;
      MbrJnlLine_G."Posting Description" := STRSUBSTNO('%1 %2',Contact_P."No.",MbrJnlLine_G."Action Message");
      MbrJnlLine_G."No. Series" := CapJnlBatch_L."No. Series";
      IF CapJnlBatch_L."Posting No. Series" ='' THEN
        MbrJnlLine_G."Posting No. Series" := CapJnlBatch_L."No. Series"
      ELSE
        MbrJnlLine_G."Posting No. Series" := CapJnlBatch_L."Posting No. Series";
      MbrJnlLine_G."G/L Posting No. Series" := CapJnlBatch_L."G/L Posting No. Series";
      LoadAmount(MbrJnlLine_G);
      IF TxtDescription_P <> '' THEN BEGIN
        MbrJnlLine_G."Accept Action Message" := FALSE;
        MbrJnlLine_G.Description := TxtDescription_P;
      END ELSE
        MbrJnlLine_G."Accept Action Message" := TRUE;
      MbrJnlLine_G.INSERT;
      LastLineNo_G := LastLineNo_G+10000;
    END;
