    LOCAL PROCEDURE AnalyzeData@2();
    VAR
      TempExcelBuf@1000 : TEMPORARY Record 370;
      HeaderRowNo@1003 : Integer;
      OldRowNo@1006 : Integer;
    BEGIN
      Window.OPEN(
        Text007 +
        '@1@@@@@@@@@@@@@@@@@@@@@@@@@\');
      Window.UPDATE(1,0);
      TotalRecNo := ExcelBuf.COUNT;
      RecNo := 0;

      IF ExcelBuf.FIND('-') THEN BEGIN
        REPEAT
          RecNo := RecNo + 1;
          Window.UPDATE(1,ROUND(RecNo / TotalRecNo * 10000,1));
          CASE TRUE OF
            (STRPOS(ExcelBuf."Cell Value as Text",Text009) <> 0):
              BEGIN
                IF HeaderRowNo = 0 THEN BEGIN
                  HeaderRowNo := ExcelBuf."Row No.";
                  TempExcelBuf := ExcelBuf;
                  TempExcelBuf.Comment := Text009;
                  TempExcelBuf.INSERT;
                END ELSE
                  ERROR(Text011);
              END;

            ExcelBuf."Row No." = HeaderRowNo:
              BEGIN
                TempExcelBuf := ExcelBuf;
                TempExcelBuf.Comment := ExcelBuf."Cell Value as Text";
                TempExcelBuf.INSERT;
              END;

            (ExcelBuf."Row No." > HeaderRowNo) AND (HeaderRowNo > 0):
              BEGIN
                IF ExcelBuf."Row No." <> OldRowNo THEN BEGIN
                  OldRowNo := ExcelBuf."Row No.";
                  ClearVariable();
                END;

                TempExcelBuf.SETRANGE("Column No.",ExcelBuf."Column No.");
                IF TempExcelBuf.FIND('-') THEN BEGIN
                  CASE TempExcelBuf.Comment OF
                    Text010: BEGIN
                      ContactNo := ExcelBuf."Cell Value as Text";
                    END;
                    Text015: BEGIN
                      ContactName := ExcelBuf."Cell Value as Text";
                    END;
                    Text016: BEGIN
                      EVALUATE(PartsNumber,FormatData(ExcelBuf."Cell Value as Text"));
                    END;
                    Text017: BEGIN
                      EVALUATE(ValUnit, FormatData(ExcelBuf."Cell Value as Text"));
                    END;
                    Text018: BEGIN
                      EVALUATE(Amount, FormatData(ExcelBuf."Cell Value as Text"));
                      CreateJnlLine();
                    END;
                  END;
                END;
              END;
          END;
        UNTIL ExcelBuf.NEXT = 0;
      END;


      Window.CLOSE;
    END;

      END;
    END;

    PROCEDURE CreateJnlLine@1000000011();
    VAR
      Cust@1000000000 : Record 18;
      CapitalJnlLine@1000000001 : Record 8053356;
      Erreur_L@1000000002 : Boolean;
      CapitalJournalLineLast_L@1000000003 : Record 8053356;
    BEGIN
      Erreur_L := FALSE;

      CLEAR(CapitalJnlLine);
      CLEAR(CapitalJournalLineLast_L);
      CapitalJnlLine.SETRANGE(CapitalJnlLine."Capital Batch Name", CapitalJnlBatchName);
      CapitalJnlLine.SETRANGE(CapitalJnlLine."Account No.", ContactNo);

      IF CapitalJnlLine.FINDFIRST THEN BEGIN
        ERROR(Text0004, CapitalJnlLine."Account No.", CapitalJnlLine.TABLECAPTION);
      END;

      // <-- 20220706-CLA117789-XPE1
      CheckMember(ContactNo, Erreur_L);
      // --> 20220706-CLA117789-XPE1

      IF NOT Erreur_L THEN BEGIN
        //Cr'ation du no de document
        CapitalJnlLine.VALIDATE("Capital Journal Template", CapitalJnlTemplate);
        CapitalJnlLine.VALIDATE("Capital Batch Name", CapitalJnlBatchName);
        // Recherche du dernier enregistrement
        CapitalJournalLineLast_L.SETRANGE("Capital Batch Name", CapitalJnlBatchName);

        IF NOT CapitalJournalLineLast_L.FINDLAST THEN BEGIN
          CapitalJournalLineLast_L.VALIDATE("Movement Type", MvtType);
        END;
        COMMIT;
        CapitalJnlLine.SetUpNewLine(CapitalJournalLineLast_L, TRUE);

        // --> 20220706-CLA117789-XPE1
        CapitalJnlLine.VALIDATE("Posting Date", PostingDate);
        CapitalJnlLine.VALIDATE("Due Date", DueDate);
        // <-- 20220706-CLA117789-XPE1
        CapitalJnlLine.VALIDATE("Part Type", PartType);
        CapitalJnlLine.VALIDATE("Account No.", ContactNo);

        CLEAR(BalAccName_G);
        CapJnlManagement_G.GetAccounts(CapitalJnlLine, AccName_G, BalAccName_G);

        CapitalJnlLine.VALIDATE("Line No.", CapitalJournalLineLast_L."Line No." + 10000);

        CapitalJnlLine.VALIDATE(Quantity, PartsNumber);

        CapitalJnlLine."Posting Description" := 'Retenue Capital sur Prime Fid'lit'';

        IF NOT CapitalJnlLine.INSERT(TRUE) THEN BEGIN
          Erreur_L := TRUE;
          ERROR(Text0002, CapitalJnlLine."Account No.", CapitalJnlLine.TABLECAPTION);
          //ROLLBACK
          ASSERTERROR ERROR('');
        END ELSE BEGIN
          COMMIT;
        END;
      END;
    END;
