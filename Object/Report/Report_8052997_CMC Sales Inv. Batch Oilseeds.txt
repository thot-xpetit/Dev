    PROCEDURE CreateInvoiceHeader@8053002();
    VAR
      L_recItem@8053000 : Record 27;
      L_InvoiceText@8053001 : Text[250];
      LineAmount_L@60000 : Decimal;
    BEGIN
      // creation of invoice header
      G_RecSalesHeader.INIT;
      CLEAR(G_RecSalesHeader);

      //IF NOT G_Prepay THEN BEGIN
      intCountSalesInvoice_G += 1;

      CLEAR(LineAmount_L);
      recSalesHeaderTmp_G.SETRANGE("No. Printed",intCountSalesInvoice_G);
      IF NOT recSalesHeaderTmp_G.FINDFIRST THEN
        LineAmount_L := recDeliveryTmp_G.Quantity * (recDeliveryTmp_G."Unit Price" - "CMC Sales Contract"."Shipping Unit Price")
      ELSE BEGIN
        recSalesLineTmp_G.RESET;
        recSalesLineTmp_G.SETRANGE("Document Type",recSalesHeaderTmp_G."Document Type");
        recSalesLineTmp_G.SETRANGE("Document No.",recSalesHeaderTmp_G."No.");
        recSalesLineTmp_G.SETFILTER(Type, '<>%1', recSalesLineTmp_G.Type::" ");
        IF recSalesLineTmp_G.FINDSET(FALSE, FALSE) THEN
          REPEAT
            LineAmount_L += recSalesLineTmp_G."Line Amount";
          UNTIL recSalesLineTmp_G.NEXT = 0;
      END;

      IF LineAmount_L >= 0 THEN BEGIN
        G_RecSalesHeader.VALIDATE("Document Type",G_RecSalesHeader."Document Type"::Invoice);
      END ELSE BEGIN
        G_RecSalesHeader.VALIDATE("Document Type",G_RecSalesHeader."Document Type"::"Credit Memo");
        recSalesLineTmp_G.RESET;
        recSalesLineTmp_G.SETRANGE("Document Type", recSalesHeaderTmp_G."Document Type");
        recSalesLineTmp_G.SETRANGE("Document No.", recSalesHeaderTmp_G."No.");
        IF recSalesLineTmp_G.FINDSET(FALSE, FALSE) THEN
          REPEAT
            recSalesLineTmp_G."Unit Price" := - recSalesLineTmp_G."Unit Price";
            recSalesLineTmp_G.MODIFY;
          UNTIL recSalesLineTmp_G.NEXT = 0;
      END;
      G_RecSalesHeader."No.":='';
      G_RecSalesHeader.INSERT(TRUE);
      G_RecSalesHeader.VALIDATE("Posting Date",G_PostingDateReq);
      G_RecSalesHeader.SetHideCollectifChoice(TRUE);

      G_RecSalesHeader.VALIDATE("Document Date",G_DocDateReq);
      IF G_Prepay THEN
       G_RecSalesHeader.VALIDATE("Invoice Type",G_RecSalesHeader."Invoice Type" :: "PrePay Oilseeds")
      ELSE
       G_RecSalesHeader.VALIDATE("Invoice Type",G_RecSalesHeader."Invoice Type" :: "Invoice Oilseeds");
      G_RecSalesHeader.VALIDATE("CMC Campaign No.","CMC Sales Contract"."Campaign No.");
      G_RecSalesHeader.VALIDATE("Sell-to Customer No.",Customer."No.");
      G_RecSalesHeader.VALIDATE("Sales Contract No.","CMC Sales Contract"."No.");
      G_RecSalesHeader."Your Reference":="CMC Sales Contract"."Customer Reference";
      G_RecSalesHeader.VALIDATE("Activity Code", G_Item."Activity Code");

      IF NOT G_Prepay THEN
        G_RecSalesHeader."Discount Rate" := "CMC Sales Contract"."Discount Rate";

      G_RecSalesHeader."Grouping Code" := GroupingCode_G;

      G_RecSalesHeader.MODIFY(TRUE);

      IF "CMC Sales Contract"."Invoice Text" <> '' THEN BEGIN
        L_InvoiceText := "CMC Sales Contract"."Invoice Text";
        REPEAT
          G_RecSalesLine.INIT;

    PROCEDURE CreateItemChargeAssgnt@8053009(P_SalesLine@8053000 : Record 37);

      L_ItemChargeAssgntSales@8053002 : Record 5809;
      L_Currency@8053003 : Record 4;
    BEGIN
      // create item charge assignment for current sales line
      WITH P_SalesLine DO
       BEGIN
        TESTFIELD(Type,Type::"Charge (Item)");
        TESTFIELD("No.");
        TESTFIELD(Quantity);

        L_ItemChargeAssgntSales.RESET;

           L_Currency.GET(L_SalesHeader."Currency Code");
          IF ("Inv. Discount Amount" = 0) AND
             ("Line Discount Amount" = 0) AND
             (NOT L_SalesHeader."Prices Including VAT")

            L_ItemChargeAssgntSales."Unit Cost" := "Unit Price"

                  ("Line Amount" - "Inv. Discount Amount") / Quantity,
                  L_Currency."Unit-Amount Rounding Precision");
        END;

        L_ItemChargeAssgntSales.VALIDATE("Qty. to Assign", Quantity);
        IF NOT L_ItemChargeAssgntSales.INSERT(TRUE) THEN
         L_ItemChargeAssgntSales.MODIFY;

       END;
    END;
