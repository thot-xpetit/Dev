    LOCAL PROCEDURE AnalyzeData@1000000004();
    VAR
      TempExcelBuf@1000 : TEMPORARY Record 370;
      HeaderRowNo@1003 : Integer;
      OldRowNo@1006 : Integer;
      Window@1000000004 : Dialog;
      RecNo@1000000003 : Integer;
      TotalRecNo@1000000002 : Integer;
      RowNo@1000000001 : Integer;
      ColumnNo@1000000000 : Integer;
    BEGIN
      Window.OPEN(
        Text000 +
        '@1@@@@@@@@@@@@@@@@@@@@@@@@@\');
      Window.UPDATE(1,0);
      TotalRecNo := TempExcelBuffer.COUNT;
      RecNo := 0;

      IF TempExcelBuffer.FIND('-') THEN BEGIN
        REPEAT
          RecNo := RecNo + 1;
          Window.UPDATE(1,ROUND(RecNo / TotalRecNo * 10000,1));
          CASE TRUE OF
            (STRPOS(TempExcelBuffer."Cell Value as Text",PostCodeToImport.FIELDCAPTION(Code)) <> 0):
              BEGIN
                IF HeaderRowNo = 0 THEN BEGIN
                  HeaderRowNo := TempExcelBuffer."Row No.";
                  TempExcelBuf := TempExcelBuffer;
                  TempExcelBuf.Comment := PostCodeToImport.FIELDCAPTION(Code);
                  TempExcelBuf.INSERT;
                END ELSE BEGIN
                  IF (TempExcelBuffer."Cell Value as Text" = PostCodeToImport.FIELDCAPTION(Code)) THEN BEGIN
                    ERROR(STRSUBSTNO(TErr001,PostCodeToImport.FIELDCAPTION(Code)));
                  END;
                END;
              END;

            TempExcelBuffer."Row No." = HeaderRowNo:
              BEGIN
                TempExcelBuf := TempExcelBuffer;
                TempExcelBuf.Comment := TempExcelBuffer."Cell Value as Text";
                TempExcelBuf.INSERT;
              END;

            (TempExcelBuffer."Row No." > HeaderRowNo) AND (HeaderRowNo > 0):
              BEGIN
                IF TempExcelBuffer."Row No." <> OldRowNo THEN BEGIN
                  OldRowNo := TempExcelBuffer."Row No.";
                  ClearVariable();
                END;

                TempExcelBuf.SETRANGE("Column No.",TempExcelBuffer."Column No.");
                IF TempExcelBuf.FIND('-') THEN BEGIN
                  CASE TempExcelBuf.Comment OF
                    PostCodeToImport.FIELDCAPTION(Code):
                      BEGIN
                        PostCodeNo := TempExcelBuffer."Cell Value as Text";
                      END;
                    PostCodeToImport.FIELDCAPTION(City):
                      BEGIN
                        PostCodeCity := TempExcelBuffer."Cell Value as Text";
                      END;
                    PostCodeToImport.FIELDCAPTION("INSEE No."): BEGIN
                      PostCodeInsee := TempExcelBuffer."Cell Value as Text";
                      UpdatePostCode();
                    END;
                  END;
                END;
              END;
          END;
        UNTIL TempExcelBuffer.NEXT = 0;
      END;

      Window.CLOSE;
    END;

    PROCEDURE UpdatePostCode@1000000010();
    VAR
      PostCode@1000000000 : Record 225;
    BEGIN
      PostCode.SETRANGE(Code, PostCodeNo);
      PostCode.SETRANGE(City, PostCodeCity);
      IF (PostCode.FINDFIRST()) THEN BEGIN
        PostCode.VALIDATE("INSEE No.", PostCodeInsee);
        PostCode.MODIFY(TRUE);
      END ELSE BEGIN
        PostCode.INIT();
        PostCode.VALIDATE(Code, PostCodeNo);
        PostCode.VALIDATE(City, PostCodeCity);
        PostCode.VALIDATE("INSEE No.", PostCodeInsee);
        PostCode.INSERT(TRUE);
      END;
    END;
