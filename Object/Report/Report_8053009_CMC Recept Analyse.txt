    PROCEDURE CheckAnalyse@8053002() res_AnOK : Boolean;
    VAR
      L_recPostedAnalyse@8053000 : Record 8052996;
    BEGIN
      res_AnOK:=TRUE;

      IF (G_Analyse1<>'') AND (G_Signe1>0)
      THEN BEGIN
       IF NOT L_recPostedAnalyse.GET("CMC Posted Receipts"."Receipt Type","CMC Posted Receipts"."No.",G_Analyse1)
       THEN res_AnOK:=FALSE
       ELSE BEGIN
        CASE G_Signe1 OF
          G_Signe1 :: "<": IF L_recPostedAnalyse.Value < G_Value1 THEN res_AnOK := FALSE;
          G_Signe1 :: "<=": IF L_recPostedAnalyse.Value <= G_Value1 THEN res_AnOK := FALSE;
          G_Signe1 :: ">": IF L_recPostedAnalyse.Value > G_Value1 THEN res_AnOK := FALSE;
          G_Signe1 :: ">=": IF L_recPostedAnalyse.Value >= G_Value1 THEN res_AnOK := FALSE;
        END;
       END;
      END;

      IF ((res_AnOK=FALSE) AND (G_EtOu1=G_EtOu1::"AND")) OR (res_AnOK=TRUE) AND (G_EtOu1=G_EtOu1::"OR")  THEN
      BEGIN
       IF G_EtOu1=G_EtOu1::"AND" THEN res_AnOK:=TRUE;
       IF NOT L_recPostedAnalyse.GET("CMC Posted Receipts"."Receipt Type","CMC Posted Receipts"."No.",G_Analyse2)
       THEN res_AnOK:=FALSE
       ELSE BEGIN
        CASE G_Signe2 OF
          G_Signe2 :: "<": IF L_recPostedAnalyse.Value < G_Value2 THEN res_AnOK := FALSE;
          G_Signe2 :: "<=": IF L_recPostedAnalyse.Value <= G_Value2 THEN res_AnOK := FALSE;

          G_Signe2 :: ">=": IF L_recPostedAnalyse.Value >= G_Value2 THEN res_AnOK := FALSE;
        END;
       END;
      END;
      EXIT(res_AnOK);
