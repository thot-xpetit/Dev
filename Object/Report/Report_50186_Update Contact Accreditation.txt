    LOCAL PROCEDURE AnalyzeData@2();
    VAR
      TempExcelBuf@1000 : TEMPORARY Record 370;
      HeaderRowNo@1003 : Integer;
      OldRowNo@1006 : Integer;
      TextTemp@1000000000 : Text[80];
    BEGIN
      Window.OPEN(
        Text007 +
        '@1@@@@@@@@@@@@@@@@@@@@@@@@@\');
      Window.UPDATE(1,0);
      TotalRecNo := ExcelBuf.COUNT;
      RecNo := 0;
      ClearVariable();
      IF ExcelBuf.FIND('-') THEN BEGIN
        REPEAT
          RecNo := RecNo + 1;
          Window.UPDATE(1,ROUND(RecNo / TotalRecNo * 10000,1));
          CASE TRUE OF
            (STRPOS(ExcelBuf."Cell Value as Text",Text009) <> 0):
              BEGIN
                IF HeaderRowNo = 0 THEN BEGIN
                  HeaderRowNo := ExcelBuf."Row No.";
                  TempExcelBuf := ExcelBuf;
                  TempExcelBuf.Comment := Text009;
                  TempExcelBuf.INSERT;
                END ELSE
                  ERROR(Text011);
              END;

            ExcelBuf."Row No." = HeaderRowNo:
              BEGIN
                TempExcelBuf := ExcelBuf;
                TempExcelBuf.Comment := ExcelBuf."Cell Value as Text";
                TempExcelBuf.INSERT;
              END;

            (ExcelBuf."Row No." > HeaderRowNo) AND (HeaderRowNo > 0):
              BEGIN
                IF ExcelBuf."Row No." <> OldRowNo THEN BEGIN
                  UpdateAccreditation();
                  OldRowNo := ExcelBuf."Row No.";
                  ClearVariable();
                END;

                TempExcelBuf.SETRANGE("Column No.",ExcelBuf."Column No.");
                IF TempExcelBuf.FIND('-') THEN BEGIN
                  CASE TempExcelBuf.Comment OF
                    Text010:
                      BEGIN
                        TextTemp := ExcelBuf."Cell Value as Text";
                        IF ((TextTemp = 'Company') OR (TextTemp = 'Soci't'')) THEN BEGIN
                          ContactType := ContactType::Company;
                        END;
                        IF ((TextTemp = 'Person') OR (TextTemp = 'Personne')) THEN BEGIN
                          ContactType := ContactType::Person;
                        END;
                      END;
                    Text013: BEGIN
                      ContactNo := ExcelBuf."Cell Value as Text";
                    END;
                    Text014: BEGIN
                      ContactName := ExcelBuf."Cell Value as Text";
                    END;
                    Text015: BEGIN
                      AccreditationCode := ExcelBuf."Cell Value as Text";
                    END;
                    Text016: BEGIN
                      ContactCompanyNo := ExcelBuf."Cell Value as Text";
                    END;
                    Text017: BEGIN
                      ContactCompanyName := ExcelBuf."Cell Value as Text";
                    END;
                    Text018: BEGIN
                      AccreditationNo := ExcelBuf."Cell Value as Text";
                    END;
                    Text019: BEGIN
                      TextTemp := ExcelBuf."Cell Value as Text";
                      IF (TextTemp = '') THEN BEGIN
                        BeginDate := 0D;
                      END ELSE BEGIN
                        IF (NOT EVALUATE(BeginDate, FORMAT(TextTemp))) THEN BEGIN
                          BeginDate := 0D;
                        END;
                      END;
                    END;
                    Text020: BEGIN
                      TextTemp := ExcelBuf."Cell Value as Text";
                      IF (TextTemp = '') THEN BEGIN
                        EndDate := 0D;
                      END ELSE BEGIN
                        IF (NOT EVALUATE(EndDate, FORMAT(TextTemp))) THEN BEGIN
                          EndDate := 0D;
                        END;
                      END;

                    END;
                    Text021: BEGIN
                      AuthorisationCode := ExcelBuf."Cell Value as Text";
                    END;
                    Text022: BEGIN
                      AccreditationType := ExcelBuf."Cell Value as Text";
                    END;
                    Text023: BEGIN
                      Comments := ExcelBuf."Cell Value as Text";
                    END;
                  END;
                END;
              END;
          END;
        UNTIL ExcelBuf.NEXT = 0;
      END;
      UpdateAccreditation();

      Window.CLOSE;
    END;
