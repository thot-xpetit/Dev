    PROCEDURE CheckDimUsed@8061736(Position_P@8061732 : Integer);
    BEGIN
      IF Dimension_G[Position_P].Code = '' THEN
        EXIT;

      FOR i_G:=1 TO 8 DO BEGIN
        IF i_G <> Position_P THEN
          IF Dimension_G[i_G].Code = Dimension_G[Position_P].Code THEN
            ERROR(Text8061732,i_G);
      END;
    END;

    PROCEDURE InitDimValue@8061735();
    BEGIN
      FOR i_G:=1 TO 8 DO BEGIN
        CLEAR(DimValue_G[i_G]);
        IF Dimension_G[i_G].Code <> '' THEN BEGIN
          DimValue_G[i_G].RESET;
          DimValue_G[i_G].SETRANGE("Dimension Code",Dimension_G[i_G].Code);
        END;
      END;
    END;

    PROCEDURE ProcessDimVal@8061734(Position_P@8061732 : Integer);
    BEGIN
      IF Position_P = 1 THEN BEGIN
        CLEAR(TmpDimAllocTemplate_G);
        REPEAT

          TmpDefaultDim2_G."Dimension Code" := DimValue_G[1]."Dimension Code";
          TmpDefaultDim2_G."Dimension Value Code" := DimValue_G[1].Code;
          TmpDefaultDim2_G.INSERT;
        UNTIL DimValue_G[1].NEXT = 0;
      END ELSE BEGIN
        IF Dimension_G[i_G].Code <> '' THEN BEGIN
          ClearTemp(TmpDimAllocTemplate_G,TmpDefaultDim_G);
          IF DimValue_G[Position_P].FINDFIRST THEN
            REPEAT
              TransferDimAllocTemp(TmpDimAllocTemplate2_G,TmpDefaultDim2_G,
                                   TmpDimAllocTemplate_G,TmpDefaultDim_G,DimValue_G[Position_P]);
            UNTIL DimValue_G[Position_P].NEXT = 0;
          ClearTemp(TmpDimAllocTemplate2_G,TmpDefaultDim2_G);
          TransferDimAllocTemp(TmpDimAllocTemplate_G,TmpDefaultDim_G,
                               TmpDimAllocTemplate2_G,TmpDefaultDim2_G,TmpDimValue_G);
        END;
      END;
    END;
