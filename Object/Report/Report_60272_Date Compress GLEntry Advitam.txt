        InitNewEntry(NewGLEntry);

          GLEntry.GET(GLEntryNo);
          SummarizeEntry(NewGLEntry,GLEntry);

          Found := DimBufMgt.NextDimEntryNo(DimEntryNo,GLEntryNo);
