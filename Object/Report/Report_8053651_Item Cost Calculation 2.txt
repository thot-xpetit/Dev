    PROCEDURE CreateItemCostValueLine@8053038(ItemVariantCode_P@8053000 : Code[10];BlanketOrderNo_P@8053003 : Code[20]);
    VAR
      ItemCostValue_L@8053001 : Record 8053287;
      DocumentType_L@8053002 : 'Quote,Order,Invoice,Credit Memo,Blanket Order,Return Order';
    BEGIN

      //------------------------------------------------------------------------------------------------
      // The function inserts or modifies the "Item Cost Calc. Value" line for item variant and purchase contract pear with the
      // medium cost calculated.
      // Input:
      //   -  ItemVariantCode_P
      //   -  BlanketOrderNo_P - purchase contract number
      //------------------------------------------------------------------------------------------------

      TotalQuantity_G := 0;
      TotalAmount_G := 0;

      // --> 20130108-Gen17/001-CMC02.02.00-1354-BSI01
      //BEFORE
      //CalculForDebutStock(ItemVariantCode_P);
      //CalculForValueEntry(ItemVariantCode_P);
      //AFTER
      CalculForDebutStock(ItemVariantCode_P,BlanketOrderNo_P);
      CalculForValueEntry(ItemVariantCode_P,BlanketOrderNo_P);
      // <-- 20130108-Gen17/001-CMC02.02.00-1354-BSI01


      IF TotalQuantity_G <> 0 THEN BEGIN
        FinalUnitCost_G := TotalAmount_G / TotalQuantity_G;
        CLEAR(ItemCostValue_L);
        ItemCostValue_L.VALIDATE("Item Cost Calc. Code",ItemCostCalcCode_G);
        ItemCostValue_L.VALIDATE("Item Cost Calc. Date",ItemCostCalcDate_G);
        ItemCostValue_L.VALIDATE("Item No.",Item."No.");
        ItemCostValue_L.VALIDATE("Variant Code",ItemVariantCode_P);
        // --> 20130108-Gen17/001-CMC02.02.00-1354-BSI01
        IF BlanketOrderNo_P <> '' THEN
          ItemCostValue_L.VALIDATE("Purch. Blanket Order No.",BlanketOrderNo_P);
        // <-- 20130108-Gen17/001-CMC02.02.00-1354-BSI01

        IF ItemCostValue_L.INSERT(TRUE) THEN;
        ItemCostValue_L.VALIDATE("LCY Unit Cost",FinalUnitCost_G );
        ItemCostValue_L.VALIDATE("Starting Date",StartingDate_G);
        ItemCostValue_L.VALIDATE("Ending Date",EndingDate_G);
        ItemCostValue_L.MODIFY(TRUE);
      END;
    END;

    PROCEDURE CalculForDebutStock@8053029(ItemVariantCode_P@8053001 : Code[10];BlanketOrderNo_P@8053004 : Code[20]);
    VAR
      ValueEntry_L@8053000 : Record 5802;
      StockQty_L@8053002 : Decimal;
      StockAmount_L@8053003 : Decimal;
    BEGIN
      //------------------------------------------------------------------------------------------------
      // The function calculates total quantity in stock of the item and variant to the date entered in the
      // request page (InitialStockCalcDate_G)  on all store and the amount that coresponds to this stock.
      // Input:
      //   -  ItemVariantCode_P
      //   -  BlanketOrderNo_P - purchase contract number
      //------------------------------------------------------------------------------------------------


      StockQty_L := 0;
      StockAmount_L :=0;

      ValueEntry_L.RESET;
      ValueEntry_L.SETCURRENTKEY("Item No.","Posting Date","Item Ledger Entry Type",
        "Entry Type","Variance Type","Item Charge No.","Location Code","Variant Code","Campaign No.");

      ValueEntry_L.SETRANGE("Item No.",Item."No.");
      ValueEntry_L.SETRANGE("Variant Code",ItemVariantCode_P);
      ValueEntry_L.SETFILTER("Posting Date",'<=%1',StartingDate_G);

      // --> 20130108-Gen17/001-CMC02.02.00-1354-BSI01
      IF BlanketOrderNo_P <> '' THEN
        ValueEntry_L.SETRANGE("Purch. Blanket Order No.",BlanketOrderNo_P);
      // <-- 20130108-Gen17/001-CMC02.02.00-1354-BSI01

      // --> 20130108-Gen17/001-CMC02.02.00-1354-BSI01
      //BEFORE

      {
      // --> 20131223-Gen17/001-CMC02.02.00-1324-BSI01
      //BEFORE
      {
      ValueEntry_L.CALCSUMS("Item Ledger Entry Quantity");
      ValueEntry_L.CALCSUMS("Cost Amount (Expected)");
      }
      //AFTER
      ValueEntry_L.CALCSUMS("Item Ledger Entry Quantity","Cost Amount (Expected)","Cost Amount (Actual)");
      // <-- 20131223-Gen17/001-CMC02.02.00-1324-BSI01


      StockQty_L := ValueEntry_L."Item Ledger Entry Quantity";
      // --> 20131223-Gen17/001-CMC02.02.00-1324-BSI01
      //BEFORE
      //StockAmount_L := ValueEntry_L."Cost Amount (Expected)";
      //AFTER
      StockAmount_L := ValueEntry_L."Cost Amount (Expected)" + ValueEntry_L."Cost Amount (Actual)";
      // <-- 20131223-Gen17/001-CMC02.02.00-1324-BSI01
      }

      //AFTER
      IF ValueEntry_L.FINDSET THEN
        REPEAT
          IF BlanketOrderNo_P <> '' THEN BEGIN
            StockQty_L += ValueEntry_L."Item Ledger Entry Quantity";
            StockAmount_L += ValueEntry_L."Cost Amount (Expected)" + ValueEntry_L."Cost Amount (Actual)";
          END ELSE
            IF ((ValueEntry_L."Purch. Blanket Order No." = '') OR
              ((ValueEntry_L."Purch. Blanket Order No." <> '') AND NOT IsSpecificBlanketOrder(ValueEntry_L."Purch. Blanket Order No.")))
                THEN BEGIN
              StockQty_L += ValueEntry_L."Item Ledger Entry Quantity";
              StockAmount_L += ValueEntry_L."Cost Amount (Expected)" + ValueEntry_L."Cost Amount (Actual)";
            END;
        UNTIL ValueEntry_L.NEXT = 0;

      // <-- 20130108-Gen17/001-CMC02.02.00-1354-BSI01


      TotalQuantity_G += StockQty_L;
      TotalAmount_G += StockAmount_L;
    END;

    PROCEDURE CalculForValueEntry@8053019(ItemVariantCode_P@8053001 : Code[10];BlanketOrderNo_P@8053004 : Code[20]);
    VAR
      ValueEntry_L@8053000 : Record 5802;
      Qty_L@8053002 : Decimal;
      Amount_L@8053003 : Decimal;
    BEGIN

      //------------------------------------------------------------------------------------------------
      // The function calculates total quantity and amount for "Montant facture, transport, frais annexe and RFC vers'es"
      // for the item ,variant and purchase contract.
      // Input:
      //   -  ItemVariantCode_P
      //   -  BlanketOrderNo_P - purchase contract number
      //------------------------------------------------------------------------------------------------

      Qty_L := 0;
      Amount_L :=0;

      ValueEntry_L.RESET;
      ValueEntry_L.SETCURRENTKEY("Item No.","Posting Date","Item Ledger Entry Type",
        "Entry Type","Variance Type","Item Charge No.","Location Code","Variant Code","Campaign No.");

      ValueEntry_L.SETRANGE("Item No.",Item."No.");

      ValueEntry_L.SETRANGE("Item Ledger Entry Type",ValueEntry_L."Item Ledger Entry Type"::Purchase);
      ValueEntry_L.SETFILTER("Document Type",'%1|%2',ValueEntry_L."Document Type"::"Purchase Invoice",
        ValueEntry_L."Document Type"::"Purchase Credit Memo");
      ValueEntry_L.SETRANGE("Variant Code",ItemVariantCode_P);

      ValueEntry_L.SETFILTER("Posting Date",'%1..%2',StartingDate_G+1,EndingDate_G);

      // --> 20130108-Gen17/001-CMC02.02.00-1354-BSI01
      IF BlanketOrderNo_P <> '' THEN
        ValueEntry_L.SETRANGE("Purch. Blanket Order No.",BlanketOrderNo_P);
      // <-- 20130108-Gen17/001-CMC02.02.00-1354-BSI01

      IF ValueEntry_L.FINDSET THEN
        REPEAT
          // --> 20130108-Gen17/001-CMC02.02.00-1354-BSI01
          IF BlanketOrderNo_P <> '' THEN BEGIN
          // <-- 20130108-Gen17/001-CMC02.02.00-1354-BSI01
            Qty_L += ValueEntry_L."Invoiced Quantity";
            Amount_L += ValueEntry_L."Purchase Amount (Actual)";
          // --> 20130108-Gen17/001-CMC02.02.00-1354-BSI01
          END ELSE
            IF ((ValueEntry_L."Purch. Blanket Order No." = '') OR
              ((ValueEntry_L."Purch. Blanket Order No." <> '') AND NOT IsSpecificBlanketOrder(ValueEntry_L."Purch. Blanket Order No.")))
                THEN BEGIN
              Qty_L += ValueEntry_L."Invoiced Quantity";
              Amount_L += ValueEntry_L."Purchase Amount (Actual)";
            END;
          // <-- 20130108-Gen17/001-CMC02.02.00-1354-BSI01
        UNTIL ValueEntry_L.NEXT = 0;

      TotalQuantity_G += Qty_L;
      TotalAmount_G += Amount_L;
    END;
