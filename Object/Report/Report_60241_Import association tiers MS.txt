    LOCAL PROCEDURE AnalyzeData@2();
    VAR
      TempExcelBuf@1000 : TEMPORARY Record 370;

      TimePos@1009 : Integer;
    BEGIN
      Window.OPEN(
        Text007 +
        '@1@@@@@@@@@@@@@@@@@@@@@@@@@\');
      Window.UPDATE(1,0);
      TotalRecNo := ExcelBuf.COUNT;
      RecNo := 0;

      gIntNbIntegre := 0;
      gIntErr := 0;


      IntLine_G := 0;


      IF ExcelBuf.FIND('-') THEN BEGIN
        REPEAT

          RecNo := RecNo + 1;
          Window.UPDATE(1,ROUND(RecNo / TotalRecNo * 10000,1));

          IF HeaderRowNo = 0 THEN
            HeaderRowNo := ExcelBuf."Row No."
          ELSE
            IF HeaderRowNo <> ExcelBuf."Row No." THEN BEGIN
              IF HeaderRowNo > 1 THEN
                fNulAjoutEnreg(HeaderRowNo);
              CLEAR(TabTxt_G);
              HeaderRowNo := ExcelBuf."Row No.";
            END;

          IF ExcelBuf."Column No." IN [1..100] THEN
            TabTxt_G[ExcelBuf."Column No."] := COPYSTR(ExcelBuf."Cell Value as Text",1,250);

        UNTIL ExcelBuf.NEXT = 0;
        fNulAjoutEnreg(HeaderRowNo);
      END;


      IF gIntNbIntegre > 0 THEN BEGIN
        IF gIntErr > 0 THEN
          MESSAGE(Text003,'',0,gIntErr)
        ELSE
          MESSAGE(Text004,'',gIntNbIntegre);
      END ELSE BEGIN
        IF gIntErr > 0 THEN
          MESSAGE(Text002,'',gIntErr);
      END;

      Window.CLOSE;
    END;
