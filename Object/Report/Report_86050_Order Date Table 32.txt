    PROCEDURE FctFindOrderDate@1000000000(OrderNo_P@1000000000 : Code[20]) : Date;
    VAR
      PurchHeader_L@1000000001 : Record 38;
      PurchHeaderArchive_L@1000000002 : Record 5109;
    BEGIN
      //Cas 1 : la commande est trouvable
      IF "Item Ledger Entry"."Document Type" = "Item Ledger Entry"."Document Type"::"Purchase Receipt" THEN BEGIN
        IF PurchHeader_L.GET(PurchHeader_L."Document Type"::Order,OrderNo_P) THEN
           EXIT(PurchHeader_L."Order Date");
      END
      ELSE BEGIN
        IF PurchHeader_L.GET(PurchHeader_L."Document Type"::"Return Order",OrderNo_P) THEN
           EXIT(PurchHeader_L."Order Date");
      END;
      //Cas 2 : sinon on recherche l'archive
      PurchHeaderArchive_L.RESET;
      IF "Item Ledger Entry"."Document Type" = "Item Ledger Entry"."Document Type"::"Purchase Receipt" THEN
        PurchHeaderArchive_L.SETRANGE("Document Type",PurchHeaderArchive_L."Document Type"::Order)
      ELSE
        PurchHeaderArchive_L.SETRANGE("Document Type",PurchHeaderArchive_L."Document Type"::"Return Order");
      PurchHeaderArchive_L.SETRANGE("No.",OrderNo_P);
      IF PurchHeaderArchive_L.FINDLAST THEN
        EXIT(PurchHeaderArchive_L."Order Date");

      EXIT(0D);
    END;

    BEGIN
    END.
