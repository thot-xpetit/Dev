    PROCEDURE CalculRefaction@1000000004(PostedReceiptsLine_P@1000000000 : Record 8053104) : Decimal;
    VAR
      Item_L@1000000003 : Record 27;
      RecPostedReceipts_L@1000000002 : Record 8052995;
      RecCollectionContract_L@1000000001 : Record 8052983;
      recPostedReceiptAnalysis_L@1000000010 : Record 8052996;
      BonisRefaction_L@1000000008 : Record 8053009;
      recContractQuality_L@1000000007 : Record 8052984;
      BonisRefCode_L@1000000006 : Code[10];
      AnalysisCharacteristic_L@1000000005 : Record 8053092;
      BonisRefactionExceptions_L@1000000004 : Record 8053098;
      recIndiceBonisRefaction_L@1000000011 : Record 8053010;
      recIndiceBonisConditions_L@1000000009 : Record 8053149;
      recPostedReceiptAnalysis2_L@1000000012 : Record 8052996;
      OutFilter_L@1000000017 : Boolean;
      ApplyPOL_L@1000000016 : Boolean;
      ApplyBonisRefaction_L@1000000015 : Boolean;
      AnalysisOK_L@1000000014 : Boolean;
      Stop_L@1000000013 : Boolean;
      ReceiptAnalysisValue_L@1000000018 : Decimal;
      DateAux_L@1000000019 : Date;
      IndiceBonisRefactionAux_L@1000000020 : Record 8053010;
      PriceToExit_L@1000000021 : Decimal;
    BEGIN
      CMCSetup_G.GET;
      Item_L.GET(PostedReceiptsLine_P."Item Contract No.");
      RecPostedReceipts_L.GET(PostedReceiptsLine_P."Receipt Type",PostedReceiptsLine_P."No.");
      RecCollectionContract_L.GET(PostedReceiptsLine_P."Contract No.");


      //D'but : Pr'paration filtre pour la boucle
      recIndiceBonisConditions_L.SETRANGE("Campain No.", PostedReceiptsLine_P."Campaign No.");

      recPostedReceiptAnalysis2_L.RESET;
      recPostedReceiptAnalysis2_L.SETCURRENTKEY("Receipt Type", "Receipt No.", "Analysis Code");
      recPostedReceiptAnalysis2_L.SETRANGE("Receipt Type", RecPostedReceipts_L."Receipt Type");
      recPostedReceiptAnalysis2_L.SETRANGE("Receipt No.", RecPostedReceipts_L."No.");
      recPostedReceiptAnalysis2_L.SETFILTER("Averaged Value",'<>%1', 0);
      //Fin : Pr'paration filtre pour la boucle

      //// R'faction analyse
      recPostedReceiptAnalysis_L.RESET;
      recPostedReceiptAnalysis_L.SETCURRENTKEY("Receipt Type","Receipt No.");
      recPostedReceiptAnalysis_L.SETRANGE("Receipt Type",RecPostedReceipts_L."Receipt Type");
      recPostedReceiptAnalysis_L.SETRANGE("Receipt No.", RecPostedReceipts_L."No.");
      recPostedReceiptAnalysis_L.SETFILTER("Averaged Value",'<>%1', 0);
      IF recPostedReceiptAnalysis_L.FINDSET(FALSE, FALSE) THEN
        REPEAT
          OutFilter_L := FALSE;
          IF BonisRefactionExceptions_L.GET(RecCollectionContract_L."Campaign No.",RecCollectionContract_L."No.",
            recPostedReceiptAnalysis_L."Analysis Code")
          THEN
            BonisRefCode_L:=BonisRefactionExceptions_L."Bonis Refaction Code"
          ELSE BEGIN
            // --> CER17-TER01.01.00-RMZ01
            IF PostedReceiptsLine_P."Bonis Refaction Code" <> '' THEN
              BonisRefCode_L:= PostedReceiptsLine_P."Bonis Refaction Code"
            ELSE BEGIN
            // <-- CER17-TER01.01.00-RMZ01
              IF RecCollectionContract_L."Bonis Refaction Code" <> '' THEN
                BonisRefCode_L := RecCollectionContract_L."Bonis Refaction Code"
              ELSE
                BonisRefCode_L := Item_L."Bonis Refaction Code";
            END;
          END;

              IF RecCollectionContract_L."Bonis Refaction Code" <> '' THEN
                BonisRefCode_L := RecCollectionContract_L."Bonis Refaction Code"
              ELSE
                BonisRefCode_L := Item_L."Bonis Refaction Code";


          //BonisRefCode_L:=
          IF BonisRefaction_L.GET(RecPostedReceipts_L."Campaign No.",BonisRefCode_L,recPostedReceiptAnalysis_L."Analysis Code")
          THEN BEGIN
            // --> 20140210-CER37-TER01.01.00-PDC01
            IF BonisRefaction_L."Average Calculation" AND
                (recPostedReceiptAnalysis_L."Averaged Value"<>0) AND
                (CMCSetup_G."Average Impact Stock"=CMCSetup_G."Average Impact Stock"::"Invoice on average no stock impact") THEN
              ReceiptAnalysisValue_L:=recPostedReceiptAnalysis_L."Averaged Value"
            ELSE
              // --> 20150325-CER15-TER01.01.00-RMZ01
              // Before
              //ReceiptAnalysisValue_L:=recPostedReceiptAnalysis_L.Value;
              // After
              ReceiptAnalysisValue_L:= recPostedReceiptAnalysis_L.Value - recPostedReceiptAnalysis_L.Tolerance;
              // <-- 20150325-CER15-TER01.01.00-RMZ01
            // --> 20140210-CER37-TER01.01.00-PDC01

            ReceiptAnalysisValue_L:=recPostedReceiptAnalysis_L."Averaged Value";

            IF RecPostedReceipts_L."Collection Type" <> RecPostedReceipts_L."Collection Type"::"Storage For Job" THEN
              IF (BonisRefaction_L."Average Calculation") AND (recPostedReceiptAnalysis_L.Value > BonisRefaction_L."Lower Value") AND
                (recPostedReceiptAnalysis_L.Value < BonisRefaction_L."Upper Value") AND
                (CMCSetup_G."Average Impact Stock"= CMCSetup_G."Average Impact Stock"::"On Invoice")
              THEN BEGIN
                //TMPAverageReceiptAnalysis_P.INIT;
                //TMPAverageReceiptAnalysis_P.TRANSFERFIELDS(recPostedReceiptAnalysis_L);
                //TMPAverageReceiptAnalysis_P."Line No." := "Line No.";
                //TMPAverageReceiptAnalysis_P.INSERT;
              END;

            IF BonisRefaction_L."Invoice Type" = BonisRefaction_L."Invoice Type"::"D'duction montant Unitaire" THEN BEGIN
              recIndiceBonisRefaction_L.RESET;
              // --> 20140908-CER38-TER01.01.00-ENS01
              //L_CMCIndiceBonisRefaction.SETCURRENTKEY("Campain No.", "Bonis Refaction Code", "Analysis Code");
              recIndiceBonisRefaction_L.SETCURRENTKEY(
                "Campain No.","Bonis Refaction Code","Analysis Code","Range Mini","Region Code",Date);
              // <-- 20140908-CER38-TER01.01.00-ENS01
              recIndiceBonisRefaction_L.SETRANGE("Campain No.", RecPostedReceipts_L."Campaign No.");
              recIndiceBonisRefaction_L.SETRANGE("Bonis Refaction Code", BonisRefCode_L);
              recIndiceBonisRefaction_L.SETRANGE("Analysis Code", recPostedReceiptAnalysis_L."Analysis Code");
              // --> 20140908-CER38-TER01.01.00-ENS01
              recIndiceBonisRefaction_L.SETRANGE("Region Code",RecCollectionContract_L."Region Code");
              // --> 20141014-CER33-TER01.01.00-ENS01
              //L_CMCIndiceBonisRefaction.SETRANGE(Date,"Shipment Date");
              // <-- 20141014-CER33-TER01.01.00-ENS01

              //on teste s'il n'existe pas de lignes d'indice pour le code region commerciale
              IF recIndiceBonisRefaction_L.COUNT = 0 THEN BEGIN
              //on r'initialise les filtres, on prend la r'gion commerciale vide.
                recIndiceBonisRefaction_L.RESET;
                // --> 20140916-CER38-TER01.01.00-FBE01
                //L_CMCIndiceBonisRefaction.SETCURRENTKEY("Campain No.", "Bonis Refaction Code", "Analysis Code","Region Code");
                recIndiceBonisRefaction_L.SETCURRENTKEY(
                  "Campain No.","Bonis Refaction Code","Analysis Code","Range Mini","Region Code",Date);
                // <-- 20140916-CER38-TER01.01.00-FBE01
                recIndiceBonisRefaction_L.SETRANGE("Campain No.", RecPostedReceipts_L."Campaign No.");
                recIndiceBonisRefaction_L.SETRANGE("Bonis Refaction Code", BonisRefCode_L);
                recIndiceBonisRefaction_L.SETRANGE("Analysis Code",recPostedReceiptAnalysis_L."Analysis Code");
                recIndiceBonisRefaction_L.SETRANGE("Region Code",'');
              END;
              // <-- 20140908-CER38-TER01.01.00-ENS01

              //si exception dans le contrat
              IF recContractQuality_L.GET(RecPostedReceipts_L."Contract No.", recPostedReceiptAnalysis_L."Analysis Code") THEN BEGIN
                IF (recContractQuality_L."B-R Exception" > 0) AND (recContractQuality_L."Departure Value" <> 0) THEN BEGIN
                  IF recContractQuality_L."B-R Exception" = recContractQuality_L."B-R Exception"::Boni THEN
                    recIndiceBonisRefaction_L.SETFILTER("Add Unit Price", '>0');
                  IF recContractQuality_L."B-R Exception" = recContractQuality_L."B-R Exception"::Refaction THEN
                    recIndiceBonisRefaction_L.SETFILTER("Add Unit Price", '<0');
                  IF recContractQuality_L."Option Value Type" = recContractQuality_L."Option Value Type"::Floor THEN
                    // --> 20140210-CER37-TER01.01.00-PDC01
                    //IF recPostedReceiptAnalysis_L.Value < recContractQuality_L."Departure Value" THEN  //OLD
                    IF ReceiptAnalysisValue_L < recContractQuality_L."Departure Value" THEN
                    // <-- 20140210-CER37-TER01.01.00-PDC01
                      OutFilter_L:=TRUE;
                  IF recContractQuality_L."Option Value Type" = recContractQuality_L."Option Value Type"::Ceiling THEN
                    // --> 20140210-CER37-TER01.01.00-PDC01
                    //IF L_CMCPostedReceiptAnalysis.Value > L_recContractQuality."Departure Value" THEN  //OLD
                    IF ReceiptAnalysisValue_L > recContractQuality_L."Departure Value" THEN
                    // <-- 20140210-CER37-TER01.01.00-PDC01
                      OutFilter_L:=TRUE;
                END;
              END;

              //D'but : Pr'paration filtre pour la boucle
              recIndiceBonisConditions_L.SETRANGE("Bonis Refaction Code", BonisRefCode_L);
              recIndiceBonisConditions_L.SETRANGE("Bonis Refaction Analysis Code", recPostedReceiptAnalysis_L."Analysis Code");
              //Fin : Pr'paration filtre pour la boucle
              // --> 20140210-CER37-TER01.01.00-PDC01
              //recIndiceBonisRefaction_L.SETFILTER("Range Mini", '<=%1', recPostedReceiptAnalysis_L.Value);//OLD
              recIndiceBonisRefaction_L.SETFILTER("Range Mini",'<=%1', ReceiptAnalysisValue_L);
              // <-- 20140210-CER37-TER01.01.00-PDC01
              // --> 20141014-CER33-TER01.01.00-ENS01
              DateAux_L := 0D;
              IndiceBonisRefactionAux_L.RESET;
              IndiceBonisRefactionAux_L.COPYFILTERS(recIndiceBonisRefaction_L);
              IF IndiceBonisRefactionAux_L.FIND('-') THEN
                REPEAT
                  DateAux_L :=IndiceBonisRefactionAux_L.Date
                UNTIL (IndiceBonisRefactionAux_L.NEXT = 0) OR ((IndiceBonisRefactionAux_L.Date > PostedReceiptsLine_P."Shipment Date")
                      // --> 20150527-CER33-TER01.01.00-442-BSI01
                      AND (IndiceBonisRefactionAux_L."Range Mini" <= ReceiptAnalysisValue_L)
                      AND (IndiceBonisRefactionAux_L."Range Max" >= ReceiptAnalysisValue_L));
                      // <-- 20150527-CER33-TER01.01.00-442-BSI01
              recIndiceBonisRefaction_L.SETRANGE(Date,DateAux_L);
              // <-- 20141014-CER33-TER01.01.00-ENS01

              ApplyBonisRefaction_L := FALSE;
              IF recIndiceBonisRefaction_L.FINDSET(FALSE, FALSE) THEN BEGIN
                REPEAT
                  // --> 20140210-CER37-TER01.01.00-PDC01
                  //ApplyBonisRefaction_L := recIndiceBonisRefaction_L."Range Max" >= recPostedReceiptAnalysis_L.Value;//OLD
                  ApplyBonisRefaction_L := recIndiceBonisRefaction_L."Range Max" >= ReceiptAnalysisValue_L;
                  // <-- 20140210-CER37-TER01.01.00-PDC01
                  recIndiceBonisConditions_L.SETRANGE("Range Mini", recIndiceBonisRefaction_L."Range Mini");
                  AnalysisOK_L := FALSE;
                  IF recIndiceBonisConditions_L.FINDSET(FALSE, FALSE) THEN
                    REPEAT
                      recPostedReceiptAnalysis2_L.SETRANGE("Analysis Code", recIndiceBonisConditions_L."Analysis Code");
                      recPostedReceiptAnalysis2_L.FINDFIRST;

                      AnalysisOK_L := FALSE;
                      CASE recIndiceBonisConditions_L.Sign OF
                        recIndiceBonisConditions_L.Sign::"=":
                          AnalysisOK_L := recPostedReceiptAnalysis2_L.Value = recIndiceBonisConditions_L."Analysis Value";
                        recIndiceBonisConditions_L.Sign::">":
                          AnalysisOK_L := recPostedReceiptAnalysis2_L.Value > recIndiceBonisConditions_L."Analysis Value";
                        recIndiceBonisConditions_L.Sign::">=":
                          AnalysisOK_L := recPostedReceiptAnalysis2_L.Value >= recIndiceBonisConditions_L."Analysis Value";
                        recIndiceBonisConditions_L.Sign::"<":
                          AnalysisOK_L := recPostedReceiptAnalysis2_L.Value < recIndiceBonisConditions_L."Analysis Value";
                        recIndiceBonisConditions_L.Sign::"<=":
                          AnalysisOK_L := recPostedReceiptAnalysis2_L.Value <= recIndiceBonisConditions_L."Analysis Value";
                      END;

                      CASE recIndiceBonisConditions_L."Logical Operator" OF
                        recIndiceBonisConditions_L."Logical Operator"::"OR":
                          ApplyBonisRefaction_L := ApplyBonisRefaction_L OR AnalysisOK_L;
                        recIndiceBonisConditions_L."Logical Operator"::"AND":
                          ApplyBonisRefaction_L := ApplyBonisRefaction_L AND AnalysisOK_L;
                      END;
                    UNTIL recIndiceBonisConditions_L.NEXT = 0;

                  Stop_L := FALSE;
                  IF ApplyBonisRefaction_L THEN
                    Stop_L := TRUE
                  ELSE
                    IF recIndiceBonisRefaction_L.NEXT = 0 THEN
                      Stop_L := TRUE;
                UNTIL Stop_L;
              END;

              IF (ApplyBonisRefaction_L = TRUE) AND (OutFilter_L = FALSE) AND (recIndiceBonisRefaction_L."Add Unit Price" <> 0)
              THEN BEGIN
                PriceToExit_L:=PriceToExit_L+recIndiceBonisRefaction_L."Add Unit Price";
              END;

            END;
          END;
        UNTIL recPostedReceiptAnalysis_L.NEXT = 0;

      EXIT(PriceToExit_L);
    END;

    PROCEDURE CalcuInitallRefaction@1000000000(PostedReceiptsLine_P@1000000000 : Record 8053104) : Decimal;
    VAR
      PurchLinevariatPrice_L@1000000001 : Record 8053005;
      PurchInvLine_L@1000000002 : Record 123;
      recPostedReceiptAnalysis_L@1000000004 : Record 8052996;
      PriceToExit_L@1000000003 : Decimal;
    BEGIN

      recPostedReceiptAnalysis_L.RESET;
      recPostedReceiptAnalysis_L.SETCURRENTKEY("Receipt Type","Receipt No.");
      recPostedReceiptAnalysis_L.SETRANGE("Receipt Type",PostedReceiptsLine_P."Receipt Type");
      recPostedReceiptAnalysis_L.SETRANGE("Receipt No.", PostedReceiptsLine_P."No.");
      recPostedReceiptAnalysis_L.SETFILTER("Averaged Value",'<>%1', 0);
      IF recPostedReceiptAnalysis_L.FINDSET(FALSE, FALSE) THEN
      REPEAT
        PurchLinevariatPrice_L.RESET;
        PurchLinevariatPrice_L.SETRANGE("Document Type",PurchLinevariatPrice_L."Document Type"::"Posted invoice");
        PurchLinevariatPrice_L.SETFILTER("Document Type",'%1|%2',PurchLinevariatPrice_L."Document Type"::"Posted invoice",
        PurchLinevariatPrice_L."Document Type"::"Posted Credit Memo");
        PurchLinevariatPrice_L.SETRANGE("Document No.",PostedReceiptsLine_P."Posted Purch. Document No");
        PurchLinevariatPrice_L.SETRANGE("Purch Line No.",PostedReceiptsLine_P."Posted Purch. Doc. Line No");
        PurchLinevariatPrice_L.SETRANGE("Analysis Code",recPostedReceiptAnalysis_L."Analysis Code");
        IF PurchLinevariatPrice_L.FINDFIRST THEN
          PriceToExit_L := PriceToExit_L + PurchLinevariatPrice_L."Unit Price Variation";
      UNTIL recPostedReceiptAnalysis_L.NEXT=0;

      EXIT(PriceToExit_L);
    END;

    BEGIN
    {
