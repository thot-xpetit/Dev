    PROCEDURE EnableControls@8053003();
    BEGIN
      IF NOT ISSERVICETIER THEN BEGIN
        RequestOptionsForm.Suffix.EDITABLE(LotNoType_G = LotNoType_G::"Item No. + Suffix");

      END ELSE BEGIN

    PROCEDURE SetTrackingType@1000000006();
    BEGIN
      WITH ItemTrackingCode_G DO BEGIN
        IF "Lot Specific Tracking" THEN BEGIN
          TrackingType_G := TrackingType_G::"In & Out";
        END ELSE BEGIN
          // Testing Inbound Tracking Setup
          IF "Lot Purchase Inbound Tracking" OR "Lot Sales Inbound Tracking" OR
             "Lot Pos. Adjmt. Inb. Tracking" OR "Lot Neg. Adjmt. Inb. Tracking" OR
             "Lot Manuf. Inbound Tracking" THEN
              IF TrackingType_G IN [TrackingType_G::Out,TrackingType_G::"In & Out"] THEN
                TrackingType_G := TrackingType_G::"In & Out"
              ELSE
                TrackingType_G := TrackingType_G::"In";

          // Testing Outbound Tracking Setup
          IF "Lot Purchase Outbound Tracking" OR "Lot Sales Outbound Tracking" OR
             "Lot Pos. Adjmt. Outb. Tracking" OR "Lot Neg. Adjmt. Outb. Tracking" OR
             "Lot Manuf. Outbound Tracking" THEN
              IF TrackingType_G IN [TrackingType_G::"In",TrackingType_G::"In & Out"] THEN
                TrackingType_G := TrackingType_G::"In & Out"
              ELSE
                TrackingType_G := TrackingType_G::Out;
        END;
      END;
    END;
