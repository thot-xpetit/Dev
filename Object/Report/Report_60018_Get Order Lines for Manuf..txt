    PROCEDURE FillBOMJournal@60000();
    VAR
      LineNo_G@60000 : Integer;
    BEGIN
      // --> 20141015-APP28-TER01.01.00-RMZ01
      // This function will create lines on the table "BOM Journal Line"

      //IF NOT CONFIRM(Text60001,FALSE) THEN
      //  EXIT;

      LineNo_G := 0;
      CLEAR(BOMJournalLine_G);
      BOMJournalLine_G.SETRANGE("Journal Template Name",BOMTemplate_G);
      BOMJournalLine_G.SETRANGE("Journal Batch Name",BOMBatch_G);
      IF BOMJournalLine_G.FINDLAST THEN
        LineNo_G := BOMJournalLine_G."Line No." + 10000
      ELSE
        LineNo_G := 10000;


      Item_G.GET("Sales Line"."No.");
      //IF (Item_G."Mix BOM No." <> '') AND ("Sales Line"."Quantity Manufactured" < "Sales Line".Quantity)
      //  AND (MixAllowedByLoc_G.VerifyLocationAllowed(Item_G."Mix BOM No.","Sales Line"."Location Code")) THEN BEGIN
        CLEAR(BOMJournalLine_G);
        BOMJournalLine_G."Journal Template Name" := BOMTemplate_G;
        BOMJournalLine_G."Journal Batch Name" := BOMBatch_G;
        BOMJournalLine_G."Line No." := LineNo_G;
        BOMJournalLine_G.VALIDATE("Posting Date",WORKDATE);
        BOMJournalLine_G.SetUpNewLine(BOMJournalLine_G);
        BOMJournalLine_G.VALIDATE("Item No.","Sales Line"."No.");
        BOMJournalLine_G.SkipShowMix(TRUE);
        BOMJournalLine_G."Sales Order No." := "Sales Line"."Document No.";
        BOMJournalLine_G."Sales Order Line No." := "Sales Line"."Line No.";
        BOMJournalLine_G."Sales Order Quantity" := "Sales Line".Quantity - "Sales Line"."Quantity Manufactured";
        // --> 20141105-APP28-TER01.01.00-PDC01
        //BOMJournalLine_G.VALIDATE("Location Code","Sales Line"."Location Code");
        BOMJournalLine_G.VALIDATE("Location Code","Sales Line"."Location Code Prod. Mix");
        // <-- 20141105-APP28-TER01.01.00-PDC01
        BOMJournalLine_G.VALIDATE(Quantity,"Sales Line".Quantity - "Sales Line"."Quantity Manufactured");
        BOMJournalLine_G.VALIDATE("Unit of Measure Code","Sales Line"."Unit of Measure Code");
        BOMJournalLine_G.VALIDATE("Variant Code","Sales Line"."Variant Code");
        BOMJournalLine_G.INSERT;
        // --> 20141105-APP28-TER01.01.00-PDC01
        BOMJournalLine_G.VALIDATE(Quantity,0);
        BOMJournalLine_G.MODIFY;
        // <-- 20141105-APP28-TER01.01.00-PDC01

        LineNo_G := LineNo_G + 10000;
        Counter_G := Counter_G + 1;
      //END;

      "Sales Line"."Extract Manufacturing" := TRUE;
      "Sales Line".MODIFY;
    END;
