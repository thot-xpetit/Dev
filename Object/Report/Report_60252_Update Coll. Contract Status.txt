    PROCEDURE OpenWindow@1000000001();
    BEGIN
      IF (GUIALLOWED) THEN BEGIN
        Window.OPEN(TProgress + '\' + TContractNo);
        NoOfRecs := "CMC Collection Contract".COUNT;
        CurrRec := 0;
        newprogress := 0;
      END;
    END;

    PROCEDURE CloseWindow@1000000002();
    BEGIN
      IF (GUIALLOWED) THEN BEGIN
        Window.CLOSE();
      END;
    END;

    PROCEDURE UpdateWindow@1000000003(index@1000000000 : Integer;data@1000000001 : Text[50]);
    BEGIN
      IF (GUIALLOWED) THEN BEGIN
        IF (index = 1) THEN BEGIN
          CurrRec := CurrRec + 1;
          IF (NoOfRecs <= 100) THEN BEGIN
            newprogress := (CurrRec / NoOfRecs * 10000) DIV 1;
          END ELSE BEGIN
            IF (CurrRec MOD(NoOfRecs DIV 100) = 0) THEN BEGIN
              newprogress := (CurrRec / NoOfRecs * 10000) DIV 1;
            END;
          END;
          Window.UPDATE(index, newprogress);
        END ELSE BEGIN
          Window.UPDATE(index, data);
        END;
      END;
    END;

    END.
