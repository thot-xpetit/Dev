    PROCEDURE IsRecepOK@1000000003(RecPostedReceiptsLine_P@1000000000 : Record 8053104) Res_recpOK : Boolean;
    VAR
      RecPostedReceipts_L@1000000001 : Record 8052995;
      CMCManagement_L@1000000003 : Codeunit 8052982;
      CMCSpeciesSetup_L@1000000002 : Record 8053323;
    BEGIN
      WITH RecPostedReceipts_L DO BEGIN
        RESET;

        CASE RecPostedReceiptsLine_P."Receipt Type" OF
          RecPostedReceiptsLine_P."Receipt Type"::Receipt: RecPostedReceipts_L.COPYFILTERS(CMCPostedReceipts);
          RecPostedReceiptsLine_P."Receipt Type"::Resumption: RecPostedReceipts_L.COPYFILTERS(CMCPostedResump)
        END;

        RecPostedReceipts_L."Receipt Type":=RecPostedReceiptsLine_P."Receipt Type";
        RecPostedReceipts_L."No.":=RecPostedReceiptsLine_P."No.";
        IF NOT RecPostedReceipts_L.FIND('=')
          THEN EXIT(FALSE);


        GET(RecPostedReceiptsLine_P."Receipt Type",RecPostedReceiptsLine_P."No.");
        IF (Canceled = TRUE) OR
        (Invoiced = TRUE) OR
        ("Does Not Affect On Contract"= TRUE) OR
        ("Not Usable IC"= TRUE)
        THEN EXIT(FALSE);

        IF NOT IncludeOpenDoc_G AND (Status=Status::Open)
        THEN EXIT(FALSE);

        IF ("IC Partner Code" <> '') AND
          ("IC Partner Code" <> CompanyInfo_G."IC Partner Code") AND
          ("IC Direction" = "IC Direction"::Outgoing) THEN
           EXIT(FALSE);

        IF ("CMC Purch. Contract Type"."Ctct Date Ctrl. on Autom. Aff."=
           "CMC Purch. Contract Type"."Ctct Date Ctrl. on Autom. Aff."::Controlled) AND
          NOT ("Posting Date" IN
            ["CMC Collection Contract"."Starting Date".."CMC Collection Contract"."Ending Date"] )
          THEN
            EXIT(FALSE);

        IF ("CMC Purch. Contract Type"."Shp. Date Ctrl. on Autom. Aff." =
          "CMC Purch. Contract Type"."Shp. Date Ctrl. on Autom. Aff."::Controlled) AND
          NOT ("Shipment Date" IN
            ["CMC Collection Contract"."Ship Starting Date".."CMC Collection Contract"."Ship Ending Date"] )
          THEN
              EXIT(FALSE);

        // --> 20160626-CER06-TER01.01.00-2827-NED
        "CMC Collection Contract".GetPriorityQuality("Contract Priority Quality"."Priority Code");
        // <-- 20160626-CER06-TER01.01.00-2827-NED
        IF NOT "CMC Collection Contract".isCheckPostedAnalysis("No.","CMC Collection Contract"."No.")
          THEN EXIT(FALSE);

      // --> 20151012-CER06-TER01.01.00-2309-SDE01
        IF CheckAnalysContract = FALSE
          THEN EXIT(FALSE);
      // <-- 20151012-CER06-TER01.01.00-2309-SDE01

        IF RecPostedReceiptsLine_P."Receipt Type"=RecPostedReceiptsLine_P."Receipt Type"::Receipt
        THEN BEGIN

        IF NOT Calibration THEN BEGIN
          CLEAR(CMCManagement_L);
          IF CMCManagement_L.IsCalibration(RecPostedReceipts_L) THEN
            EXIT(FALSE);
        END;


          CASE RecPostedReceiptsLine_P."Product Type" OF
            RecPostedReceiptsLine_P."Product Type"::" ",
            RecPostedReceiptsLine_P."Product Type"::"Brut Product":
            BEGIN
              IF NOT CMCSetup_G."Ignore Item Affec. Rec/Contrat" THEN
                IF "Item No." <> "CMC Collection Contract"."Item No."
                  THEN EXIT(FALSE);
            END;
            RecPostedReceiptsLine_P."Product Type"::"Rich Product":
            BEGIN
              CMCSpeciesSetup_L.GET("CMC Collection Contract"."Species Code");
              IF CMCSpeciesSetup_L."Rich Product" <> "CMC Collection Contract"."Item No." THEN
                EXIT(FALSE);
            END;
            RecPostedReceiptsLine_P."Product Type"::"Poor Product":
            BEGIN
              CMCSpeciesSetup_L.GET("CMC Collection Contract"."Species Code");
              IF CMCSpeciesSetup_L."Poor Product" <> "CMC Collection Contract"."Item No." THEN
               EXIT(FALSE);
            END;
            RecPostedReceiptsLine_P."Product Type"::"Declassed Product":
            BEGIN
              CMCSpeciesSetup_L.GET("CMC Collection Contract"."Species Code");
              IF CMCSpeciesSetup_L."Declassed Product" <> "CMC Collection Contract"."Item No." THEN
               EXIT(FALSE);
            END;
          END;
        END;

      END;

      EXIT(TRUE);
    END;
