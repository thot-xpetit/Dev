    PROCEDURE EcritureComptable@1000000006();
    VAR
      GLEntry_L@1000000000 : Record 17;
      TotAmountGLEntry_L@1000000003 : Decimal;
      Coef_L@1000000004 : Decimal;
      Number_L@1000000005 : Integer;
      AcountPrevue_L@1000000001 : Code[20];
    BEGIN
      CLEAR(AnaFluxTresoTmp_G);
      AnaFluxTresoTmp_G.DELETEALL;
      TotAmountGLEntry_L := 0;


      GLEntry_L.RESET;
      GLEntry_L.SETCURRENTKEY("Document No.","Posting Date");
      GLEntry_L.SETRANGE("Document No.",BankLedgerEntry."Document No.");
      GLEntry_L.SETRANGE("Posting Date",BankLedgerEntry."Posting Date");
      GLEntry_L.SETFILTER(GLEntry_L."Source Type",'<>%1',GLEntry_L."Source Type"::"Bank Account");
      //GLEntry_L.SETFILTER("Entry No.",'<>%1',BankLedgerEntry."Entry No.");
      IF GLEntry_L.FIND('-') THEN BEGIN
        REPEAT
          CLEAR(AnaFluxTresoTmp_G);

          AnaFluxTresoTmp_G."Entry No." := Number_L;
          AnaFluxTresoTmp_G."Entry No. GL" := GLEntry_L."Entry No.";

          AnaFluxTresoTmp_G."Credit Amount (LCY)" := GLEntry_L."Debit Amount";
          AnaFluxTresoTmp_G."Amount (LCY)" := -GLEntry_L.Amount;
          AnaFluxTresoTmp_G."Source No." := GLEntry_L."Source No.";
          AnaFluxTresoTmp_G."Source Type" := GLEntry_L."Source Type";
          AnaFluxTresoTmp_G.Description := GLEntry_L.Description;
          AnaFluxTresoTmp_G."Global Dimension 2 Code" := GLEntry_L."Global Dimension 2 Code";
          IF GLEntry_L."System-Created Entry" THEN BEGIN
            CASE GLEntry_L."Source Type" OF
            GLEntry_L."Source Type"::Customer :
              EcritureClientLettrage(AnaFluxTresoTmp_G);
            GLEntry_L."Source Type"::Vendor :
              EcritureFouLettrage(AnaFluxTresoTmp_G);
            GLEntry_L."Source Type"::"Fixed Asset" :
              AnaFluxTresoTmp_G.Nature := 'NON TRAITE';
            GLEntry_L."Source Type"::" " :
              IF COPYSTR(GLEntry_L."G/L Account No.",1,3) = '445' THEN
                AnaFluxTresoTmp_G.AccountNoCP := AcountPrevue_L //TVA
              ELSE
                AnaFluxTresoTmp_G.Nature := GLEntry_L."G/L Account No.";
            ELSE
              AnaFluxTresoTmp_G.VALIDATE(AccountNoCP,GLEntry_L."G/L Account No.");
            END;
          END ELSE BEGIN
            AnaFluxTresoTmp_G.VALIDATE(AccountNoCP,GLEntry_L."G/L Account No.");
          END;
          TotAmountGLEntry_L += AnaFluxTresoTmp_G."Amount (LCY)";
          AnaFluxTresoTmp_G.INSERT(TRUE);
          AcountPrevue_L := GLEntry_L."G/L Account No.";
        UNTIL GLEntry_L.NEXT = 0;

        Coef_L := TotAmountGLEntry_L / BankLedgerEntry."Amount (LCY)";
        InsertFlux(Coef_L);
      END ELSE BEGIN
        GLEntry_L.SETRANGE(GLEntry_L."Source Type",GLEntry_L."Source Type"::"Bank Account");
        GLEntry_L.SETRANGE("System-Created Entry",FALSE);
        IF GLEntry_L.ISEMPTY THEN BEGIN
          GLEntry_L.SETRANGE("System-Created Entry");
          GLEntry_L.SETFILTER("Entry No.",'<>%1',BankLedgerEntry."Entry No.");
        END;
        IF GLEntry_L.FIND('-') THEN BEGIN
          AnaFluxTreso_G.AccountNo := GLEntry_L."G/L Account No.";
          AnaFluxTreso_G."Source No." := GLEntry_L."Source No.";
          AnaFluxTreso_G."Source Type" := GLEntry_L."Source Type";
          AnaFluxTreso_G.Description := GLEntry_L.Description;
          AnaFluxTreso_G."Global Dimension 2 Code" := GLEntry_L."Global Dimension 2 Code";
          AnaFluxTreso_G.VALIDATE(AccountNoCP,GLEntry_L."G/L Account No.");
        END;
        AnaFluxTreso_G."Debit Amount (LCY)" := BankLedgerEntry."Debit Amount (LCY)";
        AnaFluxTreso_G."Credit Amount (LCY)" := BankLedgerEntry."Credit Amount (LCY)";

    PROCEDURE EcritureClientLettrage@1000000000(VAR AnaFluxTreso_P@1000000002 : TEMPORARY Record 60073) AccountNoCP_R : Code[20];
    VAR
      CustLedgerEntry_L@1000000001 : Record 21;
      CustLELettrage_L@1000000000 : Record 21;
      GLEntry_L@1000000003 : Record 17;
    BEGIN
      // Recherche de l''criture client
      CustLedgerEntry_L.RESET;
      CustLedgerEntry_L.SETCURRENTKEY("Source Code","Posting Date","Document No.");
      CustLedgerEntry_L.SETRANGE("Source Code",AnaFluxTreso_P."Source Code");
      CustLedgerEntry_L.SETRANGE("Posting Date",AnaFluxTreso_P."Posting Date");
      CustLedgerEntry_L.SETRANGE("Document No.",AnaFluxTreso_P."Document No.");
      CustLedgerEntry_L.SETRANGE("Customer No.",AnaFluxTreso_P."Source No.");
      IF CustLedgerEntry_L.ISEMPTY THEN
        IF AnaFluxTreso_P."Document No. Line BOR" <> '' THEN
          CustLedgerEntry_L.SETRANGE("Document No.",AnaFluxTreso_P."Document No. Line BOR")
        ELSE
          CustLedgerEntry_L.SETRANGE("Source Code");
      IF CustLedgerEntry_L.FIND('-') THEN
        AnaFluxTreso_P."Entry No. Tiers" := CustLedgerEntry_L."Entry No.";
        IF CustLedgerEntry_L.Open = TRUE THEN
          AnaFluxTreso_P.Nature := 'NON LETTRE'
        ELSE BEGIN
          //Recherche du lettrage
          CustLELettrage_L.SETCURRENTKEY("Customer No.",Letter);
          CustLELettrage_L.SETRANGE("Customer No.",AnaFluxTreso_P."Source No.");
          CustLELettrage_L.SETRANGE(Letter,CustLedgerEntry_L.Letter);
          CustLELettrage_L.SETFILTER("Entry No.",'<>%1',CustLedgerEntry_L."Entry No.");
          CustLELettrage_L.SETRANGE("Document Type",CustLELettrage_L."Document Type"::Invoice);
          CustLELettrage_L.SETFILTER(Description,'<>*AGIOS*&<>*INTERETS DE RETARD*');
          IF CustLELettrage_L.ISEMPTY THEN BEGIN
            CustLELettrage_L.SETRANGE(Description);
            CustLELettrage_L.SETRANGE("Document Type",CustLELettrage_L."Document Type"::"Credit Memo");
            IF CustLELettrage_L.ISEMPTY THEN
              CustLELettrage_L.SETRANGE("Document Type");
          END;
          IF CustLELettrage_L.FIND('+') THEN BEGIN
            GLEntry_L.RESET;
            GLEntry_L.SETCURRENTKEY("Document No.","Posting Date");
            GLEntry_L.SETRANGE("Document No.",CustLELettrage_L."Document No.");
            GLEntry_L.SETRANGE("Posting Date",CustLELettrage_L."Posting Date");
            GLEntry_L.SETRANGE("Gen. Posting Type",GLEntry_L."Gen. Posting Type"::Purchase,GLEntry_L."Gen. Posting Type"::Sale);
            GLEntry_L.SETRANGE("G/L Account No.",'6','8');
            IF GLEntry_L.ISEMPTY THEN BEGIN
                GLEntry_L.SETRANGE("Gen. Posting Type");
              IF GLEntry_L.ISEMPTY THEN BEGIN
                GLEntry_L.SETRANGE("G/L Account No.");
                GLEntry_L.SETFILTER("Source Type",'<>%1',GLEntry_L."Source Type"::"Bank Account");
                IF GLEntry_L.ISEMPTY THEN
                  GLEntry_L.SETRANGE("Source Type");
              END;
            END;
            IF GLEntry_L.FIND('-') THEN BEGIN
              AnaFluxTreso_P.VALIDATE(AccountNoCP,GLEntry_L."G/L Account No.");
              AnaFluxTreso_P."Global Dimension 2 Code" := GLEntry_L."Global Dimension 2 Code";
            END ELSE BEGIN
              AnaFluxTreso_P.VALIDATE(AccountNoCP,CustLELettrage_L."GLAccountNo.");
            END;
          END;
        END;
    END;

            PmtLine_L.SETRANGE("Due Date");
        END;

          AnaFluxTresoTmp_G."Credit Amount (LCY)" := PmtLine_L."Debit Amount";

            AnaFluxTresoTmp_G."Source No." := PmtLine_L."Account No.";
          IF PmtLine_L."G/L Description" <> '' THEN
            AnaFluxTresoTmp_G.Description := PmtLine_L."G/L Description";
          CASE PmtLine_L."Account Type" OF
          PmtLine_L."Account Type"::Customer :
            EcritureClientLettrage(AnaFluxTresoTmp_G);
          PmtLine_L."Account Type"::Vendor :

          PmtLine_L."Account Type"::"G/L Account" :
            BEGIN
              AnaFluxTreso_G.AccountNo := PmtLine_L."Account No.";
              AnaFluxTresoTmp_G.AccountNoCP := PmtLine_L."Account No.";
            END;
          END;
          TotAmountGLEntry_L += AnaFluxTresoTmp_G."Amount (LCY)";

    VAR
      AnaFluxTreso_L@1000000000 : Record 60073;
      Number_L@1000000002 : Integer;
      Amount_L@1000000003 : Decimal;
      Increment_L@1000000004 : Integer;
    BEGIN
      Number_L := AnaFluxTresoTmp_G.COUNT;
      Increment_L := 0;
      Amount_L := 0;
      IF AnaFluxTresoTmp_G.FINDFIRST THEN REPEAT
        Increment_L += 1;
        AnaFluxTreso_L.COPY(AnaFluxTresoTmp_G);
        AnaFluxTreso_L."Entry No." := 0;
        IF (Coef_P <> 1) AND (Coef_P <> 0) THEN BEGIN
          AnaFluxTreso_L."Debit Amount (LCY)" := ROUND(AnaFluxTresoTmp_G."Debit Amount (LCY)"/Coef_P,0.01);
          AnaFluxTreso_L."Credit Amount (LCY)" := ROUND(AnaFluxTresoTmp_G."Credit Amount (LCY)"/Coef_P,0.01);
          AnaFluxTreso_L."Amount (LCY)":= ROUND(AnaFluxTresoTmp_G."Amount (LCY)"/Coef_P,0.01);
        END;
        IF Coef_P = 0 THEN BEGIN
          AnaFluxTreso_L.Nature := '0';
        END;
        IF Number_L = Increment_L THEN BEGIN
          AnaFluxTreso_L."Amount (LCY)":= BankLedgerEntry."Amount (LCY)" - Amount_L;
          IF AnaFluxTreso_L."Amount (LCY)" > 0 THEN BEGIN
            AnaFluxTreso_L."Debit Amount (LCY)" := AnaFluxTreso_L."Amount (LCY)";
            AnaFluxTreso_L."Credit Amount (LCY)" := 0;
          END ELSE BEGIN
            AnaFluxTreso_L."Debit Amount (LCY)" := 0;
            AnaFluxTreso_L."Credit Amount (LCY)" := ABS(AnaFluxTreso_L."Amount (LCY)");
          END;
        END;
        Amount_L += AnaFluxTreso_L."Amount (LCY)";
        AnaFluxTreso_L.INSERT(TRUE);
      UNTIL AnaFluxTresoTmp_G.NEXT = 0;
      {
      IF Amount_L <> BankLedgerEntry."Amount (LCY)" THEN
        ERROR('test %1',BankLedgerEntry."Document No.");
