    PROCEDURE InsertSalesForecastEntry@8053004(LocationCode_P@8053000 : Code[10]);
    VAR
      SalesPerson_L@8053005 : Record 13;
      MCItem_L@8053004 : Record 8053079;
      ItemVariant_L@8053003 : Record 5401;
      Item_L@8053002 : Record 27;
      SalesForecastEntry_L@8053001 : Record 8053204;
      ValueEntry_L@8053006 : Record 5802;
      SalesAmountActual_L@8053007 : Decimal;
      InvoicedQuantity_L@8053008 : Decimal;
      CMCPlanting_L@8053009 : Record 8053258;
      StopSearch_L@1000000000 : Boolean;
      ProductStrategy_L@8053010 : Record 8053184;
      Authorized_L@8053011 : Boolean;
      ItemDescription_L@8053012 : Text[50];
    BEGIN
      // Fonction d'insertion de l''criture pr'vision

      Item_L.GET("CMC Sales Forecast Item"."Item No.");

      SalesForecastEntry_G.INIT;
      SalesForecastEntry_G."Entry No." := EntryNo;
      EntryNo+=1;
      SalesForecastEntry_G."Sales Forecast Code" := "CMC Sales Forecast".Code;
      SalesForecastEntry_G."Campaing No." := "CMC Sales Forecast"."Campaign No.";
      SalesForecastEntry_G."Recipient Company Code" := CompanyFilter_G;
      SalesForecastEntry_G."Salesperson Code" := SalespersonFilter_G;
      SalesForecastEntry_G."Item No." := "CMC Sales Forecast Item"."Item No.";
      SalesForecastEntry_G."Variant No." := "CMC Sales Forecast Item"."Item Variant Code";
      MCItem_L.GET("CMC Sales Forecast Item"."Item No.");
      SalesForecastEntry_G.Description := MCItem_L.Description;

      ItemDescription_L := SalesForecastEntry_G.Description;

      IF "CMC Sales Forecast Item"."Item Variant Code" <> '' THEN BEGIN
        ItemVariant_L.GET("CMC Sales Forecast Item"."Item No.","CMC Sales Forecast Item"."Item Variant Code");
        SalesForecastEntry_G.Description := ItemVariant_L.Description;
        SalesForecastEntry_G."Hectare Dose" := ItemVariant_L."Hectare Dose";
      END ELSE
        SalesForecastEntry_G."Hectare Dose" := Item_L."Hectare Dose";

      SalesForecastEntry_G."Unit Of Measure Code" := MCItem_L."Sales Unit of Measure";
      SalesForecastEntry_G."Prev. Campain Item Qty" := SalesForecastEntry_G.GetPreviousCampaignForecast(
                                                       "CMC Sales Forecast Item"."Item No.",
                                                       SalespersonFilter_G,
                                                       "CMC Sales Forecast"."Campaign No.",
                                                       "CMC Sales Forecast Item"."Item Variant Code",
                                                       LocationCode_P,
                                                       // --> 20150428-APP02-CMC02.02.00-2927-FHS01
                                                       "CMC Sales Forecast"."Not Compute Variant On Item"
                                                       // <-- 20150428-APP02-CMC02.02.00-2927-FHS01
                                                       );

      SalesForecastEntry_G."Cur. Campaign Item Qty" :=
                  SalesForecastEntry_G.GetCurrCampaignForecast("CMC Sales Forecast Item"."Item No.",
                                                               SalespersonFilter_G,
                                                               "CMC Sales Forecast"."Campaign No.",
                                                               "CMC Sales Forecast Item"."Item Variant Code",
                                                               LocationCode_P,
                                                               "CMC Sales Forecast"."Location Incl. Starting Date",
                                                               "CMC Sales Forecast"."Location Incl. Ending Date"
                                                               );

      // Gestion de l'insertion ou non de la pr'vision selon la strat'gie ou des mouvements historiques.
      Authorized_L := TRUE;
      IF "CMC Sales Forecast Item"."Item Variant Code" = '' THEN BEGIN
        IF Item_L."Product Strategy Code" <> '' THEN BEGIN
          ProductStrategy_L.GET(Item_L."Product Strategy Code");
          Authorized_L := ProductStrategy_L."Authorized Input Forecast";
        END;
      END ELSE BEGIN
        ItemVariant_L.GET("CMC Sales Forecast Item"."Item No.","CMC Sales Forecast Item"."Item Variant Code");
        IF ItemVariant_L."Product Strategy Code" <> '' THEN BEGIN
          ProductStrategy_L.GET(ItemVariant_L."Product Strategy Code");
          Authorized_L := ProductStrategy_L."Authorized Input Forecast";
        END;
      END;

      IF (SalesForecastEntry_G."Prev. Campain Item Qty" = 0) AND
            (SalesForecastEntry_G."Cur. Campaign Item Qty" = 0) THEN
        IF NOT Authorized_L THEN BEGIN
          EntryNo -= 1;
          EXIT;
       END;
      IF SalesForecastEntry_G."Hectare Dose" <> 0 THEN
        SalesForecastEntry_G."Prev. Campain Item Hect." := SalesForecastEntry_G."Prev. Campain Item Qty" /
                                                           SalesForecastEntry_G."Hectare Dose"
      ELSE
        SalesForecastEntry_G."Prev. Campain Item Hect." := 0;
      SalesForecastEntry_G."Prev. Campain Prev. Qty" := GetPrevCampainSF;

      // Alimentation des informations commerciales
      IF SalesPerson_L.GET(SalespersonFilter_G) THEN BEGIN
        IF (SalesPerson_L."Team Code" <> '') THEN
          SalesForecastEntry_G."Team Code" := SalesPerson_L."Team Code";
         SalesForecastEntry_G."Activity Production Code" := SalesPerson_L."Activity Production Code";
         SalesForecastEntry_G."Sales Network Code"       := SalesPerson_L."Sales Network Code";
         SalesForecastEntry_G."Sales Area Code"          := SalesPerson_L."Sales Area Code";
         SalesForecastEntry_G."Sales Zone Code"          := SalesPerson_L."Sales Zone Code"
      END;

      SalesForecastEntry_G."Variety Code" := Item_L."Variety Code";
      SalesForecastEntry_G."Species Code" := Item_L."Species Code";
      SalesForecastEntry_G."Entry Type" := EntryType_G;
      SalesForecastEntry_G."Location Code" := LocationCode_P;

      // Alimentation du prix moyen
      ValueEntry_L.SETCURRENTKEY("Item No.","Posting Date","Item Ledger Entry Type","Entry Type",
                                 "Variance Type","Item Charge No.","Location Code","Variant Code");
      ValueEntry_L.SETRANGE("Item No.", "CMC Sales Forecast Item"."Item No.");
      ValueEntry_L.SETRANGE("Campaign No.", "CMC Sales Forecast"."Campaign No.");
      SalesAmountActual_L := 0;
      InvoicedQuantity_L := 0;
      IF ValueEntry_L.FINDFIRST THEN
        REPEAT
          ValueEntry_L.CALCSUMS("Sales Amount (Actual)", "Invoiced Quantity");
          SalesAmountActual_L += ValueEntry_L."Sales Amount (Actual)";
          InvoicedQuantity_L += ValueEntry_L."Invoiced Quantity";
        UNTIL ValueEntry_L.NEXT=0;
      IF InvoicedQuantity_L <> 0 THEN
        SalesForecastEntry_G."Average Unit Price" := ABS(ROUND(SalesAmountActual_L / InvoicedQuantity_L, 0.01, '='));

      // Par d'faut, la quantit' en pr'vision est 'gale . la quantit' sur la campagne en cours.
      IF (SalesForecastEntry_G."Cur. Campaign Item Qty" <> 0) THEN
        SalesForecastEntry_G.Quantity := SalesForecastEntry_G."Cur. Campaign Item Qty";

      // Conversion par la dose hectare.
      IF SalesForecastEntry_G."Hectare Dose" <> 0 THEN BEGIN
        SalesForecastEntry_G."Qty (Hectare)" := SalesForecastEntry_G.Quantity / SalesForecastEntry_G."Hectare Dose";
        SalesForecastEntry_G."Cur. Campaign Hectare" := SalesForecastEntry_G."Cur. Campaign Item Qty" /
                                                        SalesForecastEntry_G."Hectare Dose";
      END;

      // Calcul de divers taux
      IF SalesForecastEntry_G.Quantity <> 0 THEN
        SalesForecastEntry_G."Actual Qty / Forec. Qty %"   := ROUND(SalesForecastEntry_G."Cur. Campaign Item Qty"*100 /
                                                                    SalesForecastEntry_G.Quantity, 0.01, '=');
      IF SalesForecastEntry_G."Prev. Campain Item Qty" <> 0 THEN
        SalesForecastEntry_G."ForecQty/PrevCampActualQty%" := ROUND(SalesForecastEntry_G.Quantity*100 /
                                                                    SalesForecastEntry_G."Prev. Campain Item Qty", 0.01, '=');
      IF SalesForecastEntry_G."Prev. Campain Item Hect." <> 0 THEN
        SalesForecastEntry_G."ForecHec/PrevCampActualHec%" := ROUND(SalesForecastEntry_G."Qty (Hectare)"*100 /
                                                                    SalesForecastEntry_G."Prev. Campain Item Hect.", 0.01, '=');

      SalesForecastEntry_G."Prev. Campaign Sales Amount" := SalesForecastEntry_G."Prev. Campain Item Qty" *
                                                            SalesForecastEntry_G."Average Unit Price";

      SalesForecastEntry_G."Forecasted Amount"           := SalesForecastEntry_G.Quantity * SalesForecastEntry_G."Average Unit Price";

      // Alimentation des mises en terre
      CMCPlanting_L.SETRANGE("Campaign No.", "CMC Sales Forecast"."Campaign No.");
      CMCPlanting_L.SETRANGE("Activity Production Code", SalesForecastEntry_G."Activity Production Code");
      CMCPlanting_L.SETRANGE(Type, CMCPlanting_L.Type::Item);
      CMCPlanting_L.SETRANGE("No.", SalesForecastEntry_G."Item No.");
      StopSearch_L := FALSE;

      IF SalesForecastEntry_G."Salesperson Code" <> '' THEN BEGIN
        CMCPlanting_L.SETRANGE("Salesperson Code", SalesForecastEntry_G."Salesperson Code");
        IF CMCPlanting_L.FINDFIRST THEN
        BEGIN
          REPEAT
            SalesForecastEntry_G."Planting Quantity" += CMCPlanting_L.Quantity;
          UNTIL CMCPlanting_L.NEXT=0;
          StopSearch_L := TRUE;
        END;
      END;

      IF (NOT StopSearch_L) AND (SalesForecastEntry_G."Sales Zone Code" <> '') THEN BEGIN
        CMCPlanting_L.SETFILTER("Salesperson Code",'%1','');
        CMCPlanting_L.SETRANGE("Sales Zone Code", SalesForecastEntry_G."Sales Zone Code");
        IF CMCPlanting_L.FINDFIRST THEN
        BEGIN
          REPEAT
            SalesForecastEntry_G."Planting Quantity" += CMCPlanting_L.Quantity;
          UNTIL CMCPlanting_L.NEXT=0;
          StopSearch_L := TRUE;
        END;
      END;

      IF (NOT StopSearch_L) AND (SalesForecastEntry_G."Sales Area Code" <> '') THEN BEGIN
        CMCPlanting_L.SETFILTER("Salesperson Code",'%1','');
        CMCPlanting_L.SETFILTER("Sales Zone Code",'%1','');
        CMCPlanting_L.SETRANGE("Sales Area Code", SalesForecastEntry_G."Sales Area Code");
        IF CMCPlanting_L.FINDFIRST THEN
          REPEAT
            SalesForecastEntry_G."Planting Quantity" += CMCPlanting_L.Quantity;
          UNTIL CMCPlanting_L.NEXT=0;
      END;
      // --> 20140725-APP20-TER01.01.00-FHS01
      SalesForecastEntry_G."Tarif N-1" := GetTarifN1(SalesForecastEntry_G, SalesForecastFilter_G);
      // <-- 20140725-APP20-TER01.01.00-FHS01

      // Alimentation des niveaux dans l''criture
      SalesForecastEntry_G."Level 1 Value" := SetLevelValue(Item_L, "CMC Sales Forecast"."Level Break 1");
      SalesForecastEntry_G."Level 2 Value" := SetLevelValue(Item_L, "CMC Sales Forecast"."Level Break 2");
      SalesForecastEntry_G."Level 3 Value" := SetLevelValue(Item_L, "CMC Sales Forecast"."Level Break 3");
      SalesForecastEntry_G."Level 4 Value" := SetLevelValue(Item_L, "CMC Sales Forecast"."Level Break 4");
      SalesForecastEntry_G."Line Level" := 0;
      SalesForecastEntry_G."Level Type" := SalesForecastEntry_G."Level Type"::Item;
      SalesForecastEntry_G."Level No."  := SalesForecastEntry_G."Item No.";
      SalesForecastEntry_G.Indent := 0;

      // Calcul du rang
      IF "CMC Sales Forecast"."Use Location" THEN
        SalesForecastEntry_G.Ranking := CalculateRanking2(LocationCode_P,
                                                       SalesForecastEntry_G."Level 1 Value", SalesForecastEntry_G."Level 2 Value",
                                                       SalesForecastEntry_G."Level 3 Value", SalesForecastEntry_G."Level 4 Value",
                                                       ItemDescription_L,SalesForecastEntry_G.Description,SalesForecastEntry_G.Indent)
      ELSE
        SalesForecastEntry_G.Ranking := CalculateRanking(LocationCode_P,
                                                       SalesForecastEntry_G."Level 1 Value", SalesForecastEntry_G."Level 2 Value",
                                                       SalesForecastEntry_G."Level 3 Value", SalesForecastEntry_G."Level 4 Value",
                                                       SalesForecastEntry_G.Description) ;

      // Insertion
      SalesForecastEntry_G.INSERT(TRUE);

      // Mise . jour de la table des unit's fertilisantes

          "Prev. Campain Prev. Qty" += CMCSalesForecastEntry_P."Prev. Campain Prev. Qty";
          "Quota Quantity"          += CMCSalesForecastEntry_P."Quota Quantity";
          "Cur. Campaign Item Qty"  += CMCSalesForecastEntry_P."Cur. Campaign Item Qty";

          END;
        END;


        IF CMCSalesForecastEntry_P."Hectare Dose" <> 0 THEN BEGIN

          "Cur. Campaign Hectare" := "Cur. Campaign Item Qty" / CMCSalesForecastEntry_P."Hectare Dose";
        END;

        IF Quantity <> 0 THEN
          "Actual Qty / Forec. Qty %" := ROUND("Cur. Campaign Item Qty"*100 / Quantity, 0.01, '=');

        IF "Prev. Campain Item Qty" <> 0  THEN
          "ForecQty/PrevCampActualQty%" := ROUND(Quantity*100 / "Prev. Campain Item Qty", 0.01, '=');

        IF "Prev. Campain Item Hect." <> 0 THEN

        "Prev. Campaign Sales Amount" := "Prev. Campain Item Qty" * "Average Unit Price";

        "Forecasted Amount" := Quantity * "Average Unit Price";
        MODIFY;
      END;

      MaintainFertilQtySFEntryTotal(CMCSalesForecastEntry_L, EntryNo_P, Quantity_P, QuantityUpdated_P);
    END;
