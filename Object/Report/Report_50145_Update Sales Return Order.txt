    PROCEDURE UpdateQuantity@1000000002(SalesLine@1000000000 : Record 37);
    VAR
      SalesLineUpdated@1000000001 : Record 37;
      xRecQty@1000000002 : Decimal;
      xRecQtyBase@1000000003 : Decimal;
    BEGIN
      IF (SalesLineUpdated.GET(SalesLine."Document Type",
                               SalesLine."Document No.",
                               SalesLine."Line No.")) THEN BEGIN
        xRecQty := SalesLineUpdated.Quantity;
        xRecQtyBase := SalesLineUpdated."Quantity (Base)";

        IF (SalesLineUpdated.Quantity > SalesLineUpdated."Return Qty. Received") THEN BEGIN
          SalesLineUpdated.Quantity := SalesLineUpdated."Return Qty. Received";
          SalesLineUpdated."Quantity (Base)" := CalcBaseQty(SalesLineUpdated.Quantity, SalesLineUpdated);
          IF (xRecQty <> SalesLineUpdated.Quantity) OR
             (xRecQtyBase <> SalesLineUpdated."Quantity (Base)") THEN BEGIN
            SalesLineUpdated.InitOutstanding;
            IF SalesLineUpdated."Document Type" IN [SalesLineUpdated."Document Type"::"Return Order",
                                                    SalesLineUpdated."Document Type"::"Credit Memo"] THEN
              SalesLineUpdated.InitQtyToReceive
           ELSE
              SalesLineUpdated.InitQtyToShip;
          END ELSE BEGIN
            IF SalesLineUpdated."Build Kit" THEN
              SalesLineUpdated.UpdateKitSalesLines;
          END;
          SalesLineUpdated.VALIDATE("Analytical Quantity", SalesLineUpdated.Quantity);
          SalesLineUpdated.MODIFY;
        END;
      END;
    END;
