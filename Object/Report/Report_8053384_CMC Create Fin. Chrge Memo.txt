    PROCEDURE CalcBalanceAtDate@8053001(ContactNo_L@8053003 : Code[20];GLAccount_L@8053002 : Code[20];StartingDate_L@8053001 : Date) : Decimal;

        // --> 20161115-WI00-ADV01.01.00-22925-CJY01

        }

        CVCollBalance_L.SETFILTER("Date Filter",'..%1',StartingDate_L);
        CVCollBalance_L.CALCFIELDS("Balance Due (LCY)");

          CMCCVLedgerEntry_L.RESET;
          CMCCVLedgerEntry_L.SETRANGE("Contact No.",ContactNo_L);
          CMCCVLedgerEntry_L.SETRANGE("GLAccountNo.",GLAccount_L);

      END ELSE
        EXIT(0);
    END;
