    PROCEDURE SetAnalysisValue@1000000004(ReceiptsType@1000000002 : Option;ReceiptsNo@1000000003 : Code[20];AnalysisCode@1000000000 : Code[10];AnalysisValue@1000000001 : Decimal);
    VAR
      ChangeAnalysis@1000000004 : Record 8052996;
    BEGIN
      ChangeAnalysis.SETRANGE("Receipt Type", ReceiptsType);
      ChangeAnalysis.SETRANGE("Receipt No.", ReceiptsNo);
      ChangeAnalysis.SETRANGE("Analysis Code",AnalysisCode);
      IF (ChangeAnalysis.FINDFIRST()) THEN BEGIN

          ChangeAnalysis.Value := AnalysisValue;
          ChangeAnalysis.MODIFY(TRUE);
        END;
      END;
    END;

    PROCEDURE GetAnalysis@1000000005(PostedReceipts@1000000000 : Record 8052995);
    VAR
      Analysis@1000000001 : Record 8052996;
    BEGIN
      AnalysisPSOK := TRUE;
      AnalysisHUMIDITYOK := TRUE;
      AnaLysisPROTEINOK := TRUE;
      Analysis.SETRANGE("Receipt Type", PostedReceipts."Receipt Type");
      Analysis.SETRANGE("Receipt No.", PostedReceipts."No.");
      // Get PS Analysis
      Analysis.SETRANGE("Analysis Code", PSAnalysis);
      IF (Analysis.FINDFIRST()) THEN BEGIN
        PSValue := Analysis.Value;
        IF (COMPANYNAME = 'UNEAL') THEN BEGIN
          IF (Analysis.Value < MinimalPSValue) THEN BEGIN
            AnalysisPSOK := FALSE;
          END;
        END ELSE BEGIN
          IF (Analysis.Value >= MinimalPSValue) THEN BEGIN
            AnalysisPSOK := FALSE;
          END;

        END;
      END;
      IF (COMPANYNAME = 'UNEAL') THEN BEGIN
        // Get HUMIDITY Analysis
        Analysis.SETRANGE("Analysis Code", HUMIDITYAnalysis);
        IF (Analysis.FINDFIRST()) THEN BEGIN
          HumdityValue := Analysis.Value;
          IF (Analysis.Value > MaximumHUMIDITYValue) THEN BEGIN
            AnalysisHUMIDITYOK := FALSE;
          END;
        END;
        // Get PROTEIN Analysis
        Analysis.SETRANGE("Analysis Code", PROTEINAnalysis);
        IF (Analysis.FINDFIRST()) THEN BEGIN
          ProteinValue := Analysis.Value;
          IF (Analysis.Value < MinimumPROTEINValue) THEN BEGIN
            AnaLysisPROTEINOK := FALSE;
          END;
        END;
      END;
    END;

    PROCEDURE PreProcess@1000000000();
    VAR
      Analysis@1000000000 : Record 8052996;
      UpdateAnalysis@1000000001 : Record 8052996;
    BEGIN
      IF (Reset) THEN BEGIN
        Analysis.SETRANGE("Analysis Code", AnalysisCode);
        Analysis.SETRANGE(Value, AnalysisValue);
        IF (Analysis.FINDFIRST()) THEN BEGIN
          REPEAT
            UpdateAnalysis.SETRANGE("Receipt Type", Analysis."Receipt Type");
            UpdateAnalysis.SETRANGE("Receipt No.", Analysis."Receipt No.");
            UpdateAnalysis.SETRANGE("Analysis Code", Analysis."Analysis Code");
            IF (UpdateAnalysis.FINDFIRST()) THEN BEGIN
              UpdateAnalysis.Value := 0;
              UpdateAnalysis.MODIFY();
            END;
          UNTIL (Analysis.NEXT() = 0);
        END;
      END;
    END;

    BEGIN
