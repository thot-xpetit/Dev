    PROCEDURE UpdateData@1000000004(Contract@1000000000 : Record 8052983);
    VAR
      UpdateContract@1000000001 : Record 8052983;
      Vendor@1000000002 : Record 23;
      Contact@1000000003 : Record 5050;
      Loc@1000000004 : Record 14;
    BEGIN
      CurrRec += 1;
      IF (NoOfRecs <= 100) THEN BEGIN
        UpdateWindow((CurrRec / NoOfRecs * 10000) DIV 1);
      END ELSE BEGIN
        IF (CurrRec MOD (NoOfRecs DIV 100) = 0) THEN BEGIN
          UpdateWindow((CurrRec / NoOfRecs * 10000) DIV 1);
        END;
      END;

      // get the contract
      IF (UpdateContract.GET(Contract."No.")) THEN BEGIN
        IF (Vendor.GET(UpdateContract."Vendor No.")) THEN BEGIN
          Contact.SETRANGE(Contact."Contact Type", Contact."Contact Type"::Farm);
          Contact.SETRANGE("Vendor No.", Vendor."No.");
          IF (Contact.FINDFIRST()) THEN BEGIN
            IF (Loc.GET(Vendor."Location Code")) THEN BEGIN
              IF (UpdateContract."Region Code" <> Loc."Region Code") THEN BEGIN
                UpdateContract."Region Code" := Loc."Region Code";
                UpdateContract.MODIFY();
              END;
            END;
          END
        END;
      END;
    END;
