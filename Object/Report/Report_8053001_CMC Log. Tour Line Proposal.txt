    PROCEDURE GetSourceDocLine@21(SourceSubType@1011 : ' ,Sales Order,Sales Return Order,Purchase Order,Purchase Return Order,Transfert,Delivery';SourceNo@1012 : Code[20];SourceLineNo@1013 : Integer) Res_DoOK : Boolean;
    VAR
      SalesLine@1001 : Record 37;
      PurchLine@1002 : Record 39;
      TransLine@1003 : Record 5741;
      ProdOrderComp@1005 : Record 5407;
      SalesLinesLookup@1007 : Form 516;
      PurchLinesLookup@1008 : Form 518;
      TransLinesLookup@1009 : Form 5749;
      ProdOrderCompLookup@1010 : Form 5407;
    BEGIN
      Res_DoOK:=FALSE;
      CASE SourceSubType OF
        SourceSubType::"Sales Order":
          BEGIN
            IF SalesLine.GET(SalesLine."Document Type"::Order,SourceNo,SourceLineNo)
            THEN
              Res_DoOK:=TRUE;
          END;
        SourceSubType::"Sales Return Order":
          BEGIN
            IF SalesLine.GET(SalesLine."Document Type"::"Return Order",SourceNo,SourceLineNo)
            THEN
              Res_DoOK:=TRUE;
          END;

        SourceSubType::"Purchase Order":
          BEGIN
            IF PurchLine.GET(SalesLine."Document Type"::Order,SourceNo,SourceLineNo)
            THEN
              Res_DoOK:=TRUE;
          END;

        SourceSubType::"Purchase Return Order":
          BEGIN
            IF PurchLine.GET(SalesLine."Document Type"::"Return Order",SourceNo,SourceLineNo)
            THEN
              Res_DoOK:=TRUE;
          END;

        SourceSubType::Transfert:
          BEGIN
            IF TransLine.GET(SourceNo,SourceLineNo)
            THEN
              Res_DoOK:=TRUE;
          END;

      END;

      EXIT( Res_DoOK);
    END;
