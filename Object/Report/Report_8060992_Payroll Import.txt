    PROCEDURE GenericVariableDataSplit@1000000001();
    BEGIN
      SplitBigInput(BigInputText_G,';',14);

      FormatFieldsMgt_G.Text2Decimal(LineCreditAmount_G,InputText_G[5]);

      FOR i_G := 1 TO 8 DO
        LineDimValue_G[i_G] := InputText_G[i_G+6];
    END;

    PROCEDURE GenericFixedDataSplit@1180250004();

      LineAccountType_G := 0;
      LineAccountNo_G := COPYSTR(BigInputText_G,12,8);

      LineDebitAmount_G := 0;
      LineCreditAmount_G := 0;
      FormatFieldsMgt_G.Text2Decimal(LineDebitAmount_G,COPYSTR(BigInputText_G,20,10));
      FormatFieldsMgt_G.Text2Decimal(LineCreditAmount_G,COPYSTR(BigInputText_G,30,10));

      LineDescription_G := FormatFieldsMgt_G.Ansi2Ascii(COPYSTR(BigInputText_G,41,12));

    END;

    PROCEDURE SAGEDataSplit@1000000008();
    BEGIN
      FormatFieldsMgt_G.Text2Date(LinePostingDate_G,COPYSTR(BigInputText_G,4,6));
      LineAccountType_G := 0;
      LineAccountNo_G := COPYSTR(BigInputText_G,12,8);

      LineDebitAmount_G := 0;
      LineCreditAmount_G := 0;

      IF COPYSTR(BigInputText_G,86,1) = 'D' THEN BEGIN
        FormatFieldsMgt_G.Text2Decimal(LineDebitAmount_G,COPYSTR(BigInputText_G,87,20));
        FormatFieldsMgt_G.Text2Decimal(LineCreditAmount_G,'');

        FormatFieldsMgt_G.Text2Decimal(LineDebitAmount_G,'');

      LineDescription_G := FormatFieldsMgt_G.Ansi2Ascii(COPYSTR(BigInputText_G,41,12));
      LineDimValue_G[1] := COPYSTR(BigInputText_G,28,4);
    END;

    PROCEDURE CCMXRHPDataSplit@1000000010();
    BEGIN
      SplitBigInput(BigInputText_G,';',14);

      FOR i_G := 1 TO 20 DO
        DeleteEndBlank(InputText_G[i_G]);

      FormatFieldsMgt_G.Text2Date(LinePostingDate_G,InputText_G[1]);
      FormatFieldsMgt_G.Text2Option(LineAccountType_G,ConvertLineAccountType(InputText_G[2]));
      LineAccountNo_G := InputText_G[3];
      FormatFieldsMgt_G.Text2Decimal(LineDebitAmount_G,InputText_G[4]);
      FormatFieldsMgt_G.Text2Decimal(LineCreditAmount_G,InputText_G[5]);

          DeleteChar(InputText_G[i_G+6],'/');

        END;
    END;

    PROCEDURE CheckValues@1000000003() : Boolean;
    VAR
      GLAccount_L@1000000000 : Record 15;
    BEGIN
      // --> V'rification de l'existence du compte g'n'ral.
      IF NOT GLAccount_L.GET(LineAccountNo_G) THEN BEGIN
        InsertError(STRSUBSTNO(GLAccNotExistErr,LineAccountNo_G,FileLineNo_G),FALSE);
        EXIT(FALSE)
      END;

      // --> V'rification de l'analytique

      IF ((GLAccount_L."Income/Balance" = GLAccount_L."Income/Balance"::"Income Statement") AND
           SetupDimOnBalanceAccOnly_G) OR ( NOT SetupDimOnBalanceAccOnly_G) THEN BEGIN
        FOR i_G := 1 TO 8 DO
          IF LineDimValue_G[i_G] <> '' THEN
            IF NOT DimValue_G.GET(DimCode_G[i_G],LineDimValue_G[i_G]) THEN BEGIN
              InsertError(STRSUBSTNO(DimValueNotExistErr,DimCode_G[i_G],LineDimValue_G[i_G],FileLineNo_G),FALSE);
              EXIT(FALSE)
            END;
      END;

      EXIT(TRUE);
    END;

    PROCEDURE InsertJnlLine@1000000002();
    BEGIN
      IF (LineDebitAmount_G - LineCreditAmount_G) <> 0 THEN BEGIN

        // Insertion de la ligne comptabilit'
        NextLineNo_G += 10000;

        WITH GenJnlLine_G DO BEGIN

          INIT;
          "Journal Template Name" := JnlTemplName_G;
          "Journal Batch Name" := JnlBatchName_G;
          "Line No." := NextLineNo_G;
          "Document No." := DocNum_G;
          "Source Code" := GenJnlTemplate_G."Source Code";
          VALIDATE("Account Type",LineAccountType_G);
          VALIDATE("Account No.",LineAccountNo_G);
          VALIDATE("Gen. Bus. Posting Group",'');
          VALIDATE("Gen. Prod. Posting Group",'');
          VALIDATE("Gen. Posting Type","Gen. Posting Type"::" ");
          VALIDATE("VAT Bus. Posting Group",'');
          VALIDATE("VAT Prod. Posting Group",'');
          VALIDATE("Posting Date",LinePostingDate_G);
          Description := LineDescription_G;
          VALIDATE(Amount,LineDebitAmount_G - LineCreditAmount_G);
          INSERT(TRUE);

          // Insertion de l'analytique.
          FOR i_G := 1 TO 8 DO
            IF LineDimValue_G[i_G] <> '' THEN
              InsertJnlLineDim(GenJnlLine_G,DimCode_G[i_G],LineDimValue_G[i_G]);

        END;

      END;
    END;
