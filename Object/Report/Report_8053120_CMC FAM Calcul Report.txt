    PROCEDURE CreateCulumnLoop@8053009(P_RecFAMReporColumn@8053000 : Record 8053491);

      CASE P_RecFAMReporColumn."Loop Field" OF

          END;

        P_RecFAMReporColumn."Loop Field"::"Item Group":
          BEGIN
            RecFAMTemplateItemGroupLink_G.RESET;
            RecFAMTemplateItemGroupLink_G.SETCURRENTKEY("Report Code",Order);
