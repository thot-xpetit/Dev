      AnalysisValue_L@1000000002 : Decimal;
      HarvestWeight_L@1000000003 : Decimal;

    BEGIN
      // --> 20160629-WI00-ADV01.01.00-3037-SDE
      RecPostedReceiptAnalysisTMP_G.DELETEALL;
      // <-- 20160629-WI00-ADV01.01.00-3037-SDE
      RecPostedReceiptAnalysisTMP_G.RESET;
      RecPostedReceiptAnalysisTMP_G.SETRANGE("Analysis Code",AnalysisCode_P);

        SumNetWeight_L := 0;

        IF RecSetup_G."Average Impact Stock" = RecSetup_G."Average Impact Stock"::"Invoice on average no stock impact" THEN
          RecTempPostedReceipts2_G.SETRANGE(Invoiced, FALSE);
        IF RecTempPostedReceipts2_G.FIND('-') THEN
          REPEAT
            // --> 20160629-WI00-ADV01.01.00-3037-SDE
            HarvestWeight_L := 0;
            // <-- 20160629-WI00-ADV01.01.00-3037-SDE
            IF RecPostedReceiptAnalysis_L.GET(RecPostedReceiptAnalysis_L."Receipt Type"::Receipt,
               RecTempPostedReceipts2_G."No.",AnalysisCode_P) AND (RecPostedReceiptAnalysis_L.Value<>0) THEN BEGIN
                CLEAR(RecPostedReceiptsLine_L);
                RecPostedReceiptsLine_L.SETCURRENTKEY("Receipt Type","No.","Resump No.","Canceled Line","Cancellation line",Invoiced);
                RecPostedReceiptsLine_L.SETRANGE("Receipt Type",RecPostedReceiptsLine_L."Receipt Type"::Receipt);
                RecPostedReceiptsLine_L.SETRANGE("No.",RecTempPostedReceipts2_G."No.");
                RecPostedReceiptsLine_L.SETRANGE("Canceled Line",FALSE);
                RecPostedReceiptsLine_L.SETRANGE("Cancellation line",FALSE);
                RecPostedReceiptsLine_L.SETRANGE("Resump No.",'');
                IF RecPostedReceiptsLine_L.FINDSET THEN
                  REPEAT
                    // --> 20160629-WI00-ADV01.01.00-3037-SDE
                    {
                    SumNetWeight_L += RecPostedReceiptsLine_L.Quantity;
                    AnalysisValue_L += (RecPostedReceiptAnalysis_L.Value * RecPostedReceiptsLine_L.Quantity);
                    }
                    IF RecSetup_G."Average Impact Stock" = RecSetup_G."Average Impact Stock"::"Invoice on average no stock impact"
                    THEN
                    BEGIN
                      SumNetWeight_L += RecPostedReceiptsLine_L.Quantity;
                      AnalysisValue_L += (RecPostedReceiptAnalysis_L.Value * RecPostedReceiptsLine_L.Quantity);
                      HarvestWeight_L += RecPostedReceiptsLine_L.Quantity;
                    END ELSE
                    BEGIN
                      SumNetWeight_L += RecPostedReceiptsLine_L."Brut Quantity";
                      AnalysisValue_L += (RecPostedReceiptAnalysis_L.Value * RecPostedReceiptsLine_L."Brut Quantity");
                      HarvestWeight_L += RecPostedReceiptsLine_L."Brut Quantity";
                    END;
                    // <-- 20160629-WI00-ADV01.01.00-3037-SDE
                  UNTIL RecPostedReceiptsLine_L.NEXT=0;

                // --> 20160629-WI00-ADV01.01.00-3037-SDE
                PostedReceipt_L.GET(RecTempPostedReceipts2_G."Receipt Type",RecTempPostedReceipts2_G."No.");
                PostedReceipt_L."Harvest weight" := HarvestWeight_L;
                // <-- 20160629-WI00-ADV01.01.00-3037-SDE
                PostedReceipt_L.MODIFY;
            END;
          UNTIL RecTempPostedReceipts2_G.NEXT = 0;

          RecPostedReceiptAnalysisTMP_G.RESET;
          RecPostedReceiptAnalysisTMP_G.INIT;
          RecPostedReceiptAnalysisTMP_G."Analysis Code":=AnalysisCode_P;
          IF SumNetWeight_L>0 THEN
            RecPostedReceiptAnalysisTMP_G."Averaged Value":=ROUND(AnalysisValue_L/SumNetWeight_L,0.01)
          ELSE
            RecPostedReceiptAnalysisTMP_G."Averaged Value":=0;
          RecPostedReceiptAnalysisTMP_G.INSERT;
      END;
      AverageValue_Res:=RecPostedReceiptAnalysisTMP_G."Averaged Value";
    END;
