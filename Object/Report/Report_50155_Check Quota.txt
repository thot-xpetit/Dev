    PROCEDURE CalculateActualQty@8053014(SalesQuota_P@8053003 : Record 8053260);
    VAR
      DetSalesLine_L@8053004 : Record 8053475;
      ActualQty_L@8053005 : Decimal;
      ItemTransEntry_L@8053007 : Record 8053336;
      Salesperson_L@8053006 : Record 13;
      Item_L@8053008 : Record 27;
      CMCSalesArea_L@1000000000 : Record 8053156;
      UOMConversionRate_L@1000000001 : Decimal;
    BEGIN
      ActualQty_L := 0;

      // Detailed Sales Line

      WITH DetSalesLine_L DO BEGIN
        RESET;
        SETCURRENTKEY("Document Type",Type,"Prod. Activity Code","CMC Campaign No.","Order Date","No.");
        //-->20150928-WI00-ADV01.01.00-2211-DGL01
        SETRANGE("Document Type","Document Type"::Order);
        //<--20150928-WI00-ADV01.01.00-2211-DGL01
        SETRANGE(Type,Type::Item);
        SETRANGE("Prod. Activity Code",SalesQuota_P."Production Activity Code");
        SETRANGE("CMC Campaign No.",SalesQuota_P."Campaign No.");
        SETFILTER("Order Date",'>=%1&<=%2',SalesQuota_P."Starting Date",SalesQuota_P."Ending Date");
        CASE SalesQuota_P.Type OF
          SalesQuota_P.Type::Item :
          BEGIN
            SETRANGE("No.",SalesQuota_P."No.");
            IF SalesQuota_P."Variant Code" <> '' THEN
              SETRANGE("Variant Code",SalesQuota_P."Variant Code");
          END;
          SalesQuota_P.Type::Variety : SETRANGE("Variety Code",SalesQuota_P."No.");
          SalesQuota_P.Type::Submarket : SETRANGE("Submarket Code",SalesQuota_P."No.");
        END;
        CASE SalesQuota_P."Sales Type" OF
          SalesQuota_P."Sales Type"::Salesperson:
            SETRANGE("Salesperson Code",SalesQuota_P."Salesperson Code");
          SalesQuota_P."Sales Type"::"Sales Zone":
            SETRANGE("Sales Zone Code",SalesQuota_P."Sales Zone Code");
          SalesQuota_P."Sales Type"::"Sales Area":
            SETRANGE("Sales Area Code",SalesQuota_P."Sales Area Code");
        END;
        //-->20150928-WI00-ADV01.01.00-2211-DGL01
        IF SalesQuota_P."Doc. Category Code" <> '' THEN
          SETRANGE("Doc. Category Code",SalesQuota_P."Doc. Category Code");
        //<--20150928-WI00-ADV01.01.00-2211-DGL01
        //-->20160623-Wi00-ADV01.01.00-2209-VMR01
        //SETRANGE(DetSalesLine_L."Excluded from society quota",SalesQuota_P."Excluded from society quota");
        CMCSalesArea_L.SETRANGE(Code,SalesQuota_P."Sales Area Code");
        CMCSalesArea_L.SETRANGE("Excluded from society quota",TRUE);
        IF CMCSalesArea_L.FINDFIRST THEN
          SETRANGE("Excluded from society quota",TRUE)
        ELSE
          SETRANGE("Excluded from society quota",FALSE);
        //<--//-->20160623-Wi00-ADV01.01.00-2209-VMR01
        // --> 20190612-WI00-ADV01.01.00-3425-CJY01
        IF SalesQuota_P.Type <> SalesQuota_P.Type::Variety THEN BEGIN
        CALCSUMS("Outstanding Quantity");
        ActualQty_L := "Outstanding Quantity";
        MakeExcelDataBody(SalesQuota_P, DetSalesLine_L, ItemTransEntry_L, TRUE,
                              (DetSalesLine_L."Outstanding Quantity"));

        END
        ELSE BEGIN
          //remplacement CALCSUMS par parcours ligne . ligne + gestion unit' de prev vari't'
          IF DetSalesLine_L.FINDSET(FALSE,FALSE) THEN REPEAT
            UOMConversionRate_L := ConvertVarietyUnit(DetSalesLine_L."Variety Code",DetSalesLine_L."No.");
            ActualQty_L := ActualQty_L + (DetSalesLine_L."Outstanding Quantity"/UOMConversionRate_L);
            MakeExcelDataBody(SalesQuota_P, DetSalesLine_L, ItemTransEntry_L, TRUE,
                              (DetSalesLine_L."Outstanding Quantity"/UOMConversionRate_L));
          UNTIL DetSalesLine_L.NEXT = 0;
        END;
        // <-- 20190612-WI00-ADV01.01.00-3425-CJY01


      END;

      // Item Transaction Entry

      WITH ItemTransEntry_L DO BEGIN
        RESET;
        SETCURRENTKEY("Entry Type","Campaign No.","CMC Production Activity Code","Order Date","Item No.");
        SETRANGE("Entry Type","Entry Type"::Sale);
        SETFILTER("Campaign No.",SalesQuota_P."Campaign No.");
        SETRANGE("CMC Production Activity Code",SalesQuota_P."Production Activity Code");
        SETFILTER("Order Date",'>=%1&<=%2',SalesQuota_P."Starting Date",SalesQuota_P."Ending Date");
        //-->20150928-WI00-ADV01.01.00-2211-DGL01
        SETRANGE("Document Type","Document Type"::"Sales Shipment");
        IF SalesQuota_P."Doc. Category Code" <> '' THEN
          SETRANGE("Doc. Category Code",SalesQuota_P."Doc. Category Code");
        //<--20150928-WI00-ADV01.01.00-2211-DGL01
        CASE SalesQuota_P.Type OF
          SalesQuota_P.Type::Item:
          BEGIN
            SETRANGE("Item No.",SalesQuota_P."No.");
            IF SalesQuota_P."Variant Code" <> '' THEN
              SETRANGE("Variant Code",SalesQuota_P."Variant Code");
          END;
          SalesQuota_P.Type::Variety: SETFILTER("CMC Variety Code",SalesQuota_P."No.");
          SalesQuota_P.Type::Submarket: SETFILTER("Submarket Code",SalesQuota_P."No.");
        END;
        CASE SalesQuota_P."Sales Type" OF
          SalesQuota_P."Sales Type"::Salesperson:
            SETFILTER("Salesperson Code",SalesQuota_P."Sales No.");
          SalesQuota_P."Sales Type"::"Sales Zone":
            SETFILTER("CMC Sales Zone Code",SalesQuota_P."Sales No.");
          SalesQuota_P."Sales Type"::"Sales Area":
            SETFILTER("CMC Sales Area Code",SalesQuota_P."Sales No.");
        END;
        //-->20160928-WI00-ADV01.01.00-21985-VMR01
        CLEAR(CMCSalesArea_L);
        CMCSalesArea_L.SETRANGE(Code,SalesQuota_P."Sales Area Code");
        CMCSalesArea_L.SETRANGE("Excluded from society quota",TRUE);
        IF CMCSalesArea_L.FINDFIRST THEN
          SETRANGE("Excluded from society quota",TRUE)
        ELSE
          SETRANGE("Excluded from society quota",FALSE);
        //-->20160928-WI00-ADV01.01.00-21985-VMR01
        // --> 20190612-WI00-ADV01.01.00-3425-CJY01
        IF SalesQuota_P.Type <> SalesQuota_P.Type::Variety THEN BEGIN
        CALCSUMS(Quantity);
        ActualQty_L -= Quantity;
            MakeExcelDataBody(SalesQuota_P, DetSalesLine_L, ItemTransEntry_L, FALSE,
                              (ItemTransEntry_L.Quantity) * -1);

        END
        ELSE BEGIN
          IF ItemTransEntry_L.FINDSET(FALSE,FALSE) THEN REPEAT
            UOMConversionRate_L := ConvertVarietyUnit(ItemTransEntry_L."CMC Variety Code",ItemTransEntry_L."Item No.");
            ActualQty_L := ActualQty_L - (ItemTransEntry_L.Quantity/UOMConversionRate_L);
            MakeExcelDataBody(SalesQuota_P, DetSalesLine_L, ItemTransEntry_L, FALSE,
                              (ItemTransEntry_L.Quantity/UOMConversionRate_L) * -1);

          UNTIL ItemTransEntry_L.NEXT = 0;

      END;
      END;
    END;

    PROCEDURE ConvertVarietyUnit@1000000006(VarietyCode_P@1000000001 : Code[10];ItemCode_P@1000000003 : Code[20]) : Decimal;
    VAR
      Variety_L@1000000000 : Record 8053070;
      MCItem_L@1000000002 : Record 8053079;
      RepoProposeSalesForEntries_L@1000000004 : Report 8053200;
    BEGIN
      IF NOT Variety_L.GET(VarietyCode_P) THEN
        Variety_L.INIT;
      MCItem_L.GET(ItemCode_P);
      IF MCItem_L."Sales Unit of Measure" = Variety_L."Sales Forecast Unit Of Measure" THEN
        EXIT(1);

      CLEAR(RepoProposeSalesForEntries_L);
      EXIT(RepoProposeSalesForEntries_L.ConvertUnitofMeasure(MCItem_L."Sales Unit of Measure",Variety_L."Sales Forecast Unit Of Measure"
      ,
        ItemCode_P));
      //ConvertUnitofMeasure(CMCSalesForecastEntry_P."Unit Of Measure Code",
        //                     Variety_L."Sales Forecast Unit Of Measure",CMCSalesForecastEntry_P."Item No.");
    END;

    BEGIN
