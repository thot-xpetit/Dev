    PROCEDURE UpdateDialog@8053000();
    VAR
      LastTreatment_G@8053003 : Integer;
      LastCustomer_G@8053002 : Integer;
      LastTemplate_G@8053001 : Integer;
    BEGIN
      IF CountTreatment_G <> 0 THEN BEGIN
        IF ABS(ROUND((LastTreatment_G - NoTreatment_G) / CountTreatment_G * 9999, 1, '=')) >= 99 THEN BEGIN
          Dialog_G.UPDATE(1, ROUND((NoTreatment_G / CountTreatment_G) * 9999, 1, '='));
          LastTreatment_G := NoTreatment_G;
        END;
      END ELSE BEGIN
        Dialog_G.UPDATE(1, 0);
        LastTreatment_G := NoTreatment_G;
      END;

      IF CountCustomer_G <> 0 THEN BEGIN
        IF ABS(ROUND((LastCustomer_G - NoCustomer_G) / CountCustomer_G * 9999, 1, '=')) >= 99 THEN BEGIN
          Dialog_G.UPDATE(2, ROUND((NoCustomer_G / CountCustomer_G) * 9999, 1, '='));
          LastCustomer_G := NoCustomer_G;
        END;
      END ELSE BEGIN
        Dialog_G.UPDATE(2, 0);
        LastCustomer_G := NoCustomer_G;
      END;

      IF CountTemplate_G <> 0 THEN BEGIN
        IF ABS(ROUND((LastTemplate_G - NoTemplate_G) / CountTemplate_G * 9999, 1, '=')) >= 99 THEN BEGIN
          Dialog_G.UPDATE(3, ROUND((NoTemplate_G / CountTemplate_G) * 9999, 1, '='));
          LastTemplate_G := NoTemplate_G;
        END;
      END ELSE BEGIN
        Dialog_G.UPDATE(3, 0);
        LastTemplate_G := NoTemplate_G;
      END;
    END;
