      RecCollectionContract_L@1000000001 : Record 8052983;
      RecPostedReceiptsLine_L@1000000002 : Record 8053104;
      PurchContractType_L@1000000003 : Record 8053096;
      QtdOnContract_L@1000000004 : Decimal;
      RemainingQty_L@1000000005 : Decimal;
      LineNo_L@1000000006 : Integer;
      SkipReceipt_L@1000000007 : Boolean;

      Windows1_G.OPEN('Affectation Contrat n› #1################');
      PurchContractType_L.SETCURRENTKEY(Priority);
      PurchContractType_L.ASCENDING(TRUE);
      PurchContractType_L.SETRANGE("Enable Automatic Assignment",TRUE);
      //PurchContractType_L.SETRANGE("Mandatory Quantity",FALSE);
      IF (CollectionContractFilters_G.GETFILTER("Contract Type") <> '') THEN
        PurchContractType_L.SETFILTER("Contract Type",CollectionContractFilters_G.GETFILTER("Contract Type"));
      IF PurchContractType_L.FINDSET THEN
      REPEAT
        tempContractPriorityQuality_G.RESET;
        IF SortingByDueDate_G THEN
          tempContractPriorityQuality_G.SETCURRENTKEY("Contract Due Date","Contract No.","Priority Code")
        ELSE
          tempContractPriorityQuality_G.SETCURRENTKEY("Contract Type",Priority);
        tempContractPriorityQuality_G.ASCENDING(TRUE);
        tempContractPriorityQuality_G.SETRANGE("Contract Type",PurchContractType_L."Contract Type");
        IF (CollectionContractFilters_G.GETFILTER("No.") <> '') THEN
          tempContractPriorityQuality_G.SETFILTER("Contract No.",CollectionContractFilters_G.GETFILTER("No."));
        //MESSAGE('%1',tempContractPriorityQuality_G.COUNT); /////
        IF tempContractPriorityQuality_G.FINDSET THEN
        REPEAT
        CLEAR(RecCollectionContract_L);
        IF SortingByDueDate_G THEN
          RecCollectionContract_L.SETCURRENTKEY(Priority,"Due Date","Starting Date","Creation Date","No.")
        ELSE
          RecCollectionContract_L.SETCURRENTKEY(Priority,"Starting Date","Creation Date","No.");
        RecCollectionContract_L.ASCENDING(TRUE);
        //RecCollectionContract_L.COPYFILTERS("CMC Collection Contract");
        RecCollectionContract_L.COPYFILTERS(CollectionContractFilters_G);
        RecCollectionContract_L.SETRANGE(Status,RecCollectionContract_L.Status::Locked);
        RecCollectionContract_L.SETRANGE(Type,RecCollectionContract_L.Type::Contract);
        IF (RecCollectionContract_L.GETFILTER("Contract Type") = '') THEN
          RecCollectionContract_L.SETFILTER("Contract Type",tempContractPriorityQuality_G."Contract Type");
        IF (RecCollectionContract_L.GETFILTER("No.") = '') THEN
          RecCollectionContract_L.SETFILTER("No.",tempContractPriorityQuality_G."Contract No.");
        IF RecCollectionContract_L.FINDSET THEN
        REPEAT
          Windows1_G.UPDATE(1,RecCollectionContract_L."No.");

          //bons de reprises
          // NE PAS TRAITER LES BONS DE REPRISES
          // --> 20240902-WI00-ADV01.01.00-3664-SAN
           { Before
              RecPostedReceiptsLine_L.RESET;
              RecPostedReceiptsLine_L.SETCURRENTKEY("Receipt Type","Campaign No.","Distributed Vendor No.","Species Code","Contract No."
                ,"Canceled Line","Cancellation line","Resump No.","Collection Type");
              RecPostedReceiptsLine_L.ASCENDING(TRUE);
              RecPostedReceiptsLine_L.SETRANGE("Receipt Type",RecPostedReceiptsLine_L."Receipt Type"::Resumption);
              RecPostedReceiptsLine_L.SETRANGE("Canceled Line",FALSE);
              RecPostedReceiptsLine_L.SETRANGE("Cancellation line",FALSE);
              RecPostedReceiptsLine_L.SETRANGE("Distributed Vendor No.",RecCollectionContract_L."Vendor No.");
              RecPostedReceiptsLine_L.SETRANGE("Campaign No.",RecCollectionContract_L."Campaign No.");
              RecPostedReceiptsLine_L.SETRANGE("Species Code",RecCollectionContract_L."Species Code");
              RecPostedReceiptsLine_L.SETFILTER(Quantity,'<>0');
              RecPostedReceiptsLine_L.SETFILTER("Contract No.",'%1','');
              RecPostedReceiptsLine_L.SETFILTER("Collection Type",'<>%1',CMCReceiptsLine_G."Collection Type"::Sales);
              IF RecPostedReceiptsLine_L.FINDSET THEN
              REPEAT
                IF IsThisRecepOK(RecPostedReceiptsLine_L,PurchContractType_L,RecCollectionContract_L,
                  tempContractPriorityQuality_G) THEN BEGIN

                  CMCReceiptsLineUpd_G.RESET;
                  CMCReceiptsLineUpd_G.GET(RecPostedReceiptsLine_L."Receipt Type",RecPostedReceiptsLine_L."No.",
                    RecPostedReceiptsLine_L."Line No.");

                  CMCPostedReceiptsReceipt_G.GET(RecPostedReceiptsLine_L."Receipt Type",RecPostedReceiptsLine_L."No.");

                  ToRelease_G := FALSE;
                  IF CMCPostedReceiptsReceipt_G.Status = CMCPostedReceiptsReceipt_G.Status::Released THEN BEGIN
                    CMCStatusMgt_G.OpenPostedReceipt(CMCPostedReceiptsReceipt_G);
                    ToRelease_G := TRUE;
                  END;


                  CMCReceiptsLineUpd_G.VALIDATE("Contract No.",RecCollectionContract_L."No.");
                  CMCReceiptsLineUpd_G.MODIFY;

                  HasChanges_G := TRUE;

                  IF ToRelease_G THEN BEGIN
                     CLEAR(CMCStatusMgt_G);
                    CMCStatusMgt_G.ReleasePostedReceipt(CMCPostedReceiptsReceipt_G);
                    COMMIT;
                  END;
                END;

              UNTIL RecPostedReceiptsLine_L.NEXT=0;
          AFTER}
          // <--> 20240902-WI00-ADV01.01.00-3664-SAN

          RecPostedReceipts_L.RESET;
          RecPostedReceipts_L.SETCURRENTKEY("Vendor No.","Species Code","Item No.","Contract No.","Collection Type","Posting Date");
          RecPostedReceipts_L.ASCENDING(TRUE);
          RecPostedReceipts_L.COPYFILTERS(PostedReceiptsFilters_G);
          RecPostedReceipts_L.SETRANGE("Receipt Type",RecPostedReceipts_L."Receipt Type"::Receipt);
          RecPostedReceipts_L.SETRANGE(Canceled,FALSE);
          RecPostedReceipts_L.SETRANGE("Vendor No.",RecCollectionContract_L."Vendor No.");
          RecPostedReceipts_L.SETRANGE("Campaign No.",RecCollectionContract_L."Campaign No.");
          RecPostedReceipts_L.SETRANGE("Species Code",RecCollectionContract_L."Species Code");
          IF RecPostedReceipts_L.FINDSET THEN
          REPEAT
              //bons des r'ceptions
              RecPostedReceiptsLine_L.RESET;
              RecPostedReceiptsLine_L.SETCURRENTKEY("Receipt Type","Campaign No.","Distributed Vendor No.","Species Code","Contract No.",
                "Canceled Line","Cancellation line","Resump No.","Collection Type");
              RecPostedReceiptsLine_L.ASCENDING(TRUE);
              RecPostedReceiptsLine_L.SETRANGE("Receipt Type",RecPostedReceiptsLine_L."Receipt Type"::Receipt);
              RecPostedReceiptsLine_L.SETFILTER("No.",RecPostedReceipts_L."No."); //
              RecPostedReceiptsLine_L.SETRANGE("Canceled Line",FALSE);
              RecPostedReceiptsLine_L.SETRANGE("Cancellation line",FALSE);
              RecPostedReceiptsLine_L.SETRANGE("Distributed Vendor No.",RecCollectionContract_L."Vendor No.");
              RecPostedReceiptsLine_L.SETRANGE("Campaign No.",RecCollectionContract_L."Campaign No.");
              RecPostedReceiptsLine_L.SETRANGE("Species Code",RecCollectionContract_L."Species Code");
              RecPostedReceiptsLine_L.SETFILTER(Quantity,'<>0');
              RecPostedReceiptsLine_L.SETFILTER("Contract No.",'%1','');
              // --> 20240902-WI00-ADV01.01.00-3664-SAN
              RecPostedReceiptsLine_L.SETRANGE("Resump No.",'');
              // <-- 20240902-WI00-ADV01.01.00-3664-SAN
              IF RecPostedReceiptsLine_L.FINDSET THEN
              REPEAT
                //MESSAGE(RecPostedReceiptsLine_L."No.");
                IF IsThisRecepOK(RecPostedReceiptsLine_L,PurchContractType_L,RecCollectionContract_L,
                  tempContractPriorityQuality_G) THEN BEGIN

                  RecCollectionContract_L.CALCFIELDS("Qty On Post. Recept Line");
                  QtdOnContract_L := RecCollectionContract_L.Quantity + ROUND(RecCollectionContract_L.Quantity *
                          (RecCollectionContract_L."Threshold Percentage" / 100),0.00001);
                  RemainingQty_L := QtdOnContract_L - RecCollectionContract_L."Qty On Post. Recept Line";



                  SkipReceipt_L := FALSE;
                  {IF NOT Overtaking2ndTime_G THEN BEGIN
                    IF PurchContractType_L."Mandatory Quantity" THEN BEGIN
                      IF RemainingQty_L <= 0 THEN
                        SkipReceipt_L := TRUE;
                     END ELSE BEGIN
                    IF RemainingQty_L <= 0 THEN
                      RemainingQty_L := RecPostedReceiptsLine_L.Quantity;
                    END;
                  END;}

                  //MESSAGE('%1 ,, RemainingQty_L :%2 , Skip:%3',RecPostedReceiptsLine_L."No.",RemainingQty_L,SkipReceipt_L);

                  IF (RemainingQty_L > 0) AND (SkipReceipt_L = FALSE) THEN BEGIN
                    //CurrReport.BREAK;

                  CMCReceiptsLineUpd_G.RESET;
                  CMCReceiptsLineUpd_G.GET(RecPostedReceiptsLine_L."Receipt Type",RecPostedReceiptsLine_L."No.",
                    RecPostedReceiptsLine_L."Line No.");

                  CMCPostedReceiptsReceipt_G.GET(RecPostedReceiptsLine_L."Receipt Type",RecPostedReceiptsLine_L."No.");

                  ToRelease_G := FALSE;
                  IF CMCPostedReceiptsReceipt_G.Status = CMCPostedReceiptsReceipt_G.Status::Released THEN BEGIN
                    CMCStatusMgt_G.OpenPostedReceipt(CMCPostedReceiptsReceipt_G);
                    ToRelease_G := TRUE;
                  END;


                  IF (CMCReceiptsLineUpd_G.Quantity <= RemainingQty_L) THEN BEGIN
                    CMCReceiptsLineUpd_G.VALIDATE("Contract No.",RecCollectionContract_L."No.");
                    CMCReceiptsLineUpd_G.MODIFY;
                  END ELSE BEGIN
                    CMCReceiptsLineUpd_G.VALIDATE("Contract No.",RecCollectionContract_L."No.");
                    CMCReceiptsLineUpd_G.Quantity := RemainingQty_L;
                    CMCReceiptsLineUpd_G.MODIFY;

                    CMCReceiptsLineUpd_G.RESET;
                    CMCReceiptsLineUpd_G.SETRANGE("Receipt Type",RecPostedReceiptsLine_L."Receipt Type");
                    CMCReceiptsLineUpd_G.SETRANGE("No.",RecPostedReceiptsLine_L."No.");
                    IF CMCReceiptsLineUpd_G.FIND('+') THEN
                      LineNo_L := CMCReceiptsLineUpd_G."Line No." + 10000
                    ELSE
                      LineNo_L := 10000;

                     CMCReceiptsLineUpd_G.INIT;
                     CMCReceiptsLineUpd_G.TRANSFERFIELDS(RecPostedReceiptsLine_L);
                     CMCReceiptsLineUpd_G."Line No." := LineNo_L;
                     CMCReceiptsLineUpd_G.INSERT(TRUE);
                     CMCReceiptsLineUpd_G.VALIDATE("Contract No.",'');
                     CMCReceiptsLineUpd_G.Quantity := RecPostedReceiptsLine_L.Quantity - RemainingQty_L;
                     CMCReceiptsLineUpd_G."Item Ledger Entry No." := 0;
                     CMCReceiptsLineUpd_G.MODIFY(TRUE);
                  END;
                  HasChanges_G := TRUE;


                  IF ToRelease_G THEN BEGIN
                     CLEAR(CMCStatusMgt_G);
                    CMCStatusMgt_G.ReleasePostedReceipt(CMCPostedReceiptsReceipt_G);
                    COMMIT;
                  END;
                  END;
                END;
              UNTIL RecPostedReceiptsLine_L.NEXT=0;
            UNTIL RecPostedReceipts_L.NEXT =0;

        UNTIL RecCollectionContract_L.NEXT=0;
        UNTIL tempContractPriorityQuality_G.NEXT=0;
      UNTIL PurchContractType_L.NEXT=0;
      Windows1_G.CLOSE;
    END;

        RESET;

        CASE RecPostedReceiptsLine_P."Receipt Type" OF
          RecPostedReceiptsLine_P."Receipt Type"::Receipt: RecPostedReceipts_L.COPYFILTERS(CMCPostedReceipts);
          RecPostedReceiptsLine_P."Receipt Type"::Resumption: RecPostedReceipts_L.COPYFILTERS(CMCPostedResump)
