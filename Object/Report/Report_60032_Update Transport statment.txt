      RecCustDelArea_L@1000000009 : Record 60018;
      RecShippingAgent_L@60000 : Record 291;
    BEGIN
      WITH RecCollectionReceipt_P DO BEGIN
        CLEAR("Area code");
        CLEAR(Km);
        CLEAR("Transport Unit Cost");
        IF ("Transport Type" <> "Transport Type"::Company) THEN//OR ("Shipping Order No." <> '') THEN
          EXIT;
        IF ("Location Code" = '') OR ("Vendor No." = '') OR ("Shipping Agent Code" = '') OR ("Shipping Vendor Code" = '') THEN
          EXIT;
        // --> 20150420-APP27-TER01.01.00-362-BSI01
        RecShippingAgent_L.GET("Shipping Agent Code");
        IF RecShippingAgent_L."Do not generate transp. costs" THEN
          EXIT;
        // <-- 20150420-APP27-TER01.01.00-362-BSI01
        // --> 20140929-APP27-TER01.01.00-BSI01
        CLEAR(RecCustDelArea_L);
        RecCustDelArea_L.SETRANGE("Customer No.","Vendor No.");
        RecCustDelArea_L.SETRANGE("Vendor No.","Shipping Vendor Code");
        IF RecCustDelArea_L.FIND('-') THEN BEGIN
          "Grid Type" := "Grid Type"::Area;
          "Area code" := RecCustDelArea_L."Delivery Area";
        END ELSE BEGIN
          RecCustDelArea_L.SETRANGE("Vendor No.",'');
          IF RecCustDelArea_L.FIND('-') THEN BEGIN
            "Grid Type" := "Grid Type"::Area;
            "Area code" := RecCustDelArea_L."Delivery Area";
          END ELSE BEGIN
            "Grid Type" := "Grid Type"::KM;
            Km := GetDistance(1,"Vendor No.",3,"Location Code");
          END;
        END;
        //D'termination du tarif transport (C'r'ale)
        CASE "Grid Type" OF
          "Grid Type"::Area : BEGIN
            CLEAR(RecTranspGridHead_L);
            RecTranspGridHead_L.SETCURRENTKEY("Transporter Code","Grid Type");
            RecTranspGridHead_L.SETRANGE("Transporter Code","Shipping Agent Code");
            RecTranspGridHead_L.SETRANGE("Grid Type",RecTranspGridHead_L."Grid Type"::Area);
            IF RecTranspGridHead_L.FIND('-') THEN BEGIN
              CLEAR(RecTranspGridLine_L);
              RecTranspGridLine_L.SETRANGE("Grid No.",RecTranspGridHead_L."Grid No.");
              RecTranspGridLine_L.SETRANGE("Grid Type",RecTranspGridHead_L."Grid Type");
              RecTranspGridLine_L.SETRANGE("Area Code","Area code");
              RecTranspGridLine_L.SETRANGE("From-Location Code","Location Code");
              // --> 20150226-APP27-TER01.01.00-270-BSI01
              RecTranspGridLine_L.SETFILTER("Starting Date",'<=%1',"Shipment Date");
              RecTranspGridLine_L.SETFILTER("Ending Date",'>=%1',"Shipment Date");
              // <-- 20150226-APP27-TER01.01.00-270-BSI01
              IF RecTranspGridLine_L.FIND('-') THEN
                VALIDATE("Transport Unit Cost",RecTranspGridLine_L."Euro/Ton");
            END ELSE BEGIN
              RecTranspGridHead_L.SETRANGE("Transporter Code",'');
              IF RecTranspGridHead_L.FIND('-') THEN BEGIN
                CLEAR(RecTranspGridLine_L);
                RecTranspGridLine_L.SETRANGE("Grid No.",RecTranspGridHead_L."Grid No.");
                RecTranspGridLine_L.SETRANGE("Grid Type",RecTranspGridHead_L."Grid Type");
                RecTranspGridLine_L.SETRANGE("Area Code","Area code");
                RecTranspGridLine_L.SETRANGE("From-Location Code","Location Code");
                // --> 20150226-APP27-TER01.01.00-270-BSI01
                RecTranspGridLine_L.SETFILTER("Starting Date",'<=%1',"Shipment Date");
                RecTranspGridLine_L.SETFILTER("Ending Date",'>=%1',"Shipment Date");
                // <-- 20150226-APP27-TER01.01.00-270-BSI01
                IF RecTranspGridLine_L.FIND('-') THEN
                  VALIDATE("Transport Unit Cost",RecTranspGridLine_L."Euro/Ton");
              END;
            END;
          END;

          "Grid Type"::KM : BEGIN
            CLEAR(RecTranspGridHead_L);
            RecTranspGridHead_L.SETCURRENTKEY("Transporter Code","Grid Type");
            RecTranspGridHead_L.SETRANGE("Transporter Code","Shipping Agent Code");
            RecTranspGridHead_L.SETRANGE("Grid Type",RecTranspGridHead_L."Grid Type"::Km);
            IF RecTranspGridHead_L.FIND('-') THEN BEGIN
              CLEAR(RecTranspGridLine_L);
              RecTranspGridLine_L.SETRANGE("Grid No.",RecTranspGridHead_L."Grid No.");
              RecTranspGridLine_L.SETRANGE("Grid Type",RecTranspGridHead_L."Grid Type");
              RecTranspGridLine_L.SETFILTER("Number Of Km",'>=%1',Km);
              RecTranspGridLine_L.SETRANGE("From-Location Code","Location Code");
              // --> 20150226-APP27-TER01.01.00-270-BSI01
              RecTranspGridLine_L.SETFILTER("Starting Date",'<=%1',"Shipment Date");
              RecTranspGridLine_L.SETFILTER("Ending Date",'>=%1',"Shipment Date");
              // <-- 20150226-APP27-TER01.01.00-270-BSI01
              IF RecTranspGridLine_L.FIND('-') THEN BEGIN
                IF Km <> 0 THEN
                  VALIDATE("Transport Unit Cost",RecTranspGridLine_L."Euro/Ton");
              END ELSE BEGIN
                RecTranspGridLine_L.SETRANGE("From-Location Code",'');
                IF RecTranspGridLine_L.FIND('-') THEN BEGIN
                  IF Km <> 0 THEN
                    VALIDATE("Transport Unit Cost",RecTranspGridLine_L."Euro/Ton");
                END;
              END;
            END ELSE BEGIN
              RecTranspGridHead_L.SETRANGE("Transporter Code",'');
              IF RecTranspGridHead_L.FIND('-') THEN BEGIN
                CLEAR(RecTranspGridLine_L);
                RecTranspGridLine_L.SETRANGE("Grid No.",RecTranspGridHead_L."Grid No.");
                RecTranspGridLine_L.SETRANGE("Grid Type",RecTranspGridHead_L."Grid Type");
                RecTranspGridLine_L.SETFILTER("Number Of Km",'>=%1',Km);
                RecTranspGridLine_L.SETRANGE("From-Location Code","Location Code");
                // --> 20150226-APP27-TER01.01.00-270-BSI01
                RecTranspGridLine_L.SETFILTER("Starting Date",'<=%1',"Shipment Date");
                RecTranspGridLine_L.SETFILTER("Ending Date",'>=%1',"Shipment Date");
                // <-- 20150226-APP27-TER01.01.00-270-BSI01
                IF RecTranspGridLine_L.FIND('-') THEN BEGIN
                  IF Km <> 0 THEN
                    VALIDATE("Transport Unit Cost",RecTranspGridLine_L."Euro/Ton");

                  RecTranspGridLine_L.SETRANGE("From-Location Code",'');

    PROCEDURE CalcTranspPriceForDelivery@1000000024(VAR RecPostedDelivery_P@1000000000 : Record 8053014);
    VAR
      RecLocation_L@1000000001 : Record 14;
      RecCustomer_L@1000000002 : Record 18;
      RecDestPostCode_L@1000000003 : Record 225;
      RecLocationPostCode_L@1000000004 : Record 225;
      RecDistance_L@1000000005 : Record 60007;
      RecTranspGridHead_L@1000000006 : Record 60011;
      RecTranspGridLine_L@1000000007 : Record 60012;
      Text00001_L@1000000008 : TextConst 'FRA=Aucun Km nest renseign' pour cet itin'raire dans la table distancier. Veuillez le renseigner avant de valider votre bon de r'ception.';
      RecCustDelArea_L@1000000009 : Record 60018;
      RecDestLocation_L@1000000010 : Record 14;
      RecShippingAgent_L@60000 : Record 291;
    BEGIN
      WITH RecPostedDelivery_P DO BEGIN
        CLEAR("Area Code");
        CLEAR(Km);
        CLEAR("Transport Unit Cost");
        IF ("Shipping Type" <> "Shipping Type"::Company) THEN
          EXIT;
        IF ("Location Code" = '') OR ("Shipping Agent Code" = '') OR ("Shipping Vendor Code" = '') THEN
          EXIT;
        RecShippingAgent_L.GET("Shipping Agent Code");
        IF RecShippingAgent_L."Do not generate transp. costs" THEN
          EXIT;
        CASE "Independent Transfer" OF
          TRUE : IF "Transfer-to Code" = '' THEN EXIT;
          FALSE : IF "Customer No." = '' THEN EXIT;
        END;
        CASE "Independent Transfer" OF
          FALSE : BEGIN
            CLEAR(RecCustDelArea_L);
            RecCustDelArea_L.SETRANGE("Customer No.","Customer No.");
            RecCustDelArea_L.SETRANGE("Vendor No.","Shipping Vendor Code");
            IF RecCustDelArea_L.FIND('-') THEN BEGIN
              "Grid Type" := "Grid Type"::Area;
              "Area Code" := RecCustDelArea_L."Delivery Area";
            END ELSE BEGIN
              RecCustDelArea_L.SETRANGE("Vendor No.",'');
              IF RecCustDelArea_L.FIND('-') THEN BEGIN
                "Grid Type" := "Grid Type"::Area;
                "Area Code" := RecCustDelArea_L."Delivery Area";
              END ELSE BEGIN
                "Grid Type" := "Grid Type"::Km;
                Km := GetDistance(3,"Location Code",2,"Customer No.");
              END;
            END;
          END; // END FALSE
          TRUE : BEGIN
            "Grid Type" := "Grid Type"::Km;
            Km := GetDistance(3,"Location Code",3,"Transfer-to Code");
          END; // END TRUE
        END; // END CASE IND TRANSFERT
        //D'termination du tarif transport (C'r'ale)
        CASE "Grid Type" OF
          "Grid Type"::Area : BEGIN
            CLEAR(RecTranspGridHead_L);
            RecTranspGridHead_L.SETCURRENTKEY("Transporter Code","Grid Type");
