    PROCEDURE ReceptLineExist@1000000003(ReturnSalesLine@1000000000 : Record 37) returnvalue : Boolean;
    VAR
      SourceOrderLine@1000000001 : Record 37;
      SalesShipmentLine@1000000002 : Record 111;
    BEGIN
      returnvalue := FALSE;
      // Get the Source ORder Line
      IF (SourceOrderLine.GET(SourceOrderLine."Document Type"::Order,
                              ReturnSalesLine."Source Order No.",
                              ReturnSalesLine."Source Order Line No.")) THEN BEGIN
        // Get the last BL which posting date < at posting date parameter
        SalesShipmentLine.SETFILTER("Posting Date", '<%1' , ReturnPostingDatefilter);
        SalesShipmentLine.SETRANGE("Order No.", SourceOrderLine."Document No.");
        SalesShipmentLine.SETRANGE("Order Line No.", SourceOrderLine."Line No.");
        SalesShipmentLine.SETRANGE(Quantity, ReturnSalesLine.Quantity);
        IF (SalesShipmentLine.FINDLAST()) THEN BEGIN
          VatRateShipment := SalesShipmentLine."VAT %";
          ShipmentNo := SalesShipmentLine."Document No.";
          returnvalue := TRUE;
        END ELSE BEGIN
          SalesShipmentLine.SETRANGE(Quantity);
          SalesShipmentLine.SETFILTER(Quantity, '<>%1', 0);
          IF (SalesShipmentLine.FINDLAST()) THEN BEGIN
            VatRateShipment := SalesShipmentLine."VAT %";
            ShipmentNo := SalesShipmentLine."Document No.";
            returnvalue := TRUE;
          END;
        END;
      END;
    END;

        END;
      END;
    END;
