    PROCEDURE GetDataFromItem@1000000003(Item@1000000000 : Record 27);
    BEGIN
      CMCActivity.SETRANGE(Code, Item."Activity Code");
      IF (CMCActivity.FINDFIRST()) THEN BEGIN
        CMCFamily.SETRANGE(Code, Item."Family Code");
        CMCFamily.SETRANGE("Activity Code", CMCActivity.Code);
        IF (CMCFamily.FINDFIRST()) THEN BEGIN
          CMCSubFamily.SETRANGE("Family Code", CMCFamily.Code);
          CMCSubFamily.SETRANGE(Code, Item."Sub-Family Code");
          IF (NOT CMCSubFamily.FINDFIRST()) THEN BEGIN
            CMCSubFamily.INIT();
          END;
        END ELSE BEGIN
          CMCFamily.INIT;
          CMCSubFamily.INIT();
        END;
      END ELSE BEGIN
        CMCActivity.INIT;
        CMCFamily.INIT;
        CMCSubFamily.INIT();
      END;
    END;
