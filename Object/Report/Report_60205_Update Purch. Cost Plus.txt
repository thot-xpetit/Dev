    PROCEDURE CostPlusTOUpdate@1000000003();
    BEGIN
      // --> 20200617-WI00-ADV01.01.00-3446-SCR01

      // Mise . jour des lignes Cost Plus

      CLEAR(TbPurchCostPlusLedgerEntry_G);
      TbPurchCostPlusLedgerEntry_G.SETRANGE("Item Ledger Entry Type",TbValueEntry_G."Item Ledger Entry Type");
      TbPurchCostPlusLedgerEntry_G.SETRANGE("Item Ledger Entry No.",TbValueEntry_G."Item Ledger Entry No.");
      TbPurchCostPlusLedgerEntry_G.SETRANGE("Value Entry No.",TbValueEntry_G."Entry No.");
      TbPurchCostPlusLedgerEntry_G.SETRANGE("Cost Plus Code" ,CostPlusCode_G);
      TbPurchCostPlusLedgerEntry_G.SETRANGE(Canceled,FALSE);
      TbPurchCostPlusLedgerEntry_G.SETRANGE("Cost Pl. No. final doc. purch.",0);
      IF TbPurchCostPlusLedgerEntry_G.FINDSET THEN BEGIN

        // Mise . jour Value Entry Cost Plus
        TbValueEntryCPLUS_G.GET(TbValueEntry_G."Entry No.",CostPlusCode_G);
        TempValueEntryCPLUS_G.INIT;
        TempValueEntryCPLUS_G.TRANSFERFIELDS(TbValueEntryCPLUS_G, TRUE);
        TempValueEntryCPLUS_G.INSERT(TRUE);

        IF CostPlusType_G = CostPlusType_G::Pourcentage THEN BEGIN
           TempValueEntryCPLUS_G.VALIDATE("Cost Plus %",CostPlusPct_G);
           TempValueEntryCPLUS_G."Cost Plus Amount" := 0;
           IF CostPlusPct_G > 0 THEN
            TempValueEntryCPLUS_G.VALIDATE("Cost Plus Amount",
                                        ROUND(TempValueEntryCPLUS_G."Base Value" * CostPlusPct_G /100,
                                        UnitAmtRndPrecision_G,'='));
        END;
        IF CostPlusType_G = CostPlusType_G::"Montant unitaire" THEN BEGIN
            TempValueEntryCPLUS_G."Cost Plus %" := 0;
            TempValueEntryCPLUS_G.VALIDATE("Cost Plus Amount",CostPlusValue_G);
        END;
        TempValueEntryCPLUS_G.MODIFY(TRUE);

        // Mise . jour Cost Plus Ledgr Entry
        UpdateCostPlus;

      END;

      // <-- 20200617-WI00-ADV01.01.00-3446-SCR01
    END;

    PROCEDURE CalcRevPurchCPlusExpAmtToBal@1000000000();
    VAR
      EntryNo_L@1000000002 : Integer;
      OldExpectedQty_L@1000000001 : Decimal;
      OldExpectedAmount_L@1000000000 : ARRAY [3] OF Decimal;
    BEGIN
      // --> 2020061-WI00-ADV01.01.00-3446-SCR01

      EntryNo_L := TempCostPlusLedgerEntry_G."Entry No.";
      CLEAR(TempCostPlusLedgerEntry_G);
      WITH TempCostPlusLedgerEntry_G DO BEGIN
        SETCURRENTKEY("Item Ledger Entry No.","Entry Type");
        SETRANGE("Item Ledger Entry No.",TbValueEntry_G."Item Ledger Entry No.");
        SETRANGE("Entry Type","Entry Type"::"Direct Cost");
        SETFILTER("Document Type", '%1|%2', "Document Type"::"Purchase Receipt", "Document Type"::"Purchase Return Shipment");
        SETRANGE("Item Charge No.",'');
        SETRANGE("Cost Plus Code",CostPlusCode_G);
        IF FINDSET THEN
          REPEAT
            OldExpectedQty_L := OldExpectedQty_L + "Valued Quantity";
            OldExpectedAmount_L[2] := OldExpectedAmount_L[2] + "Cost Plus Amount (Receipt)";
            OldExpectedAmount_L[3] := OldExpectedAmount_L[3] + "C. Plus Unit Amount (Receipt)";
          UNTIL NEXT = 0;
       END;

       CLEAR(TempCostPlusLedgerEntry_G);
       TempCostPlusLedgerEntry_G.GET(EntryNo_L);
       IF OldExpectedQty_L <> 0 THEN BEGIN
          TempCostPlusLedgerEntry_G."Cost Plus Amount (Receipt)" :=
            -ROUND((TbValueEntry_G."Invoiced Quantity" * OldExpectedAmount_L[2]) / OldExpectedQty_L, AmtRndPrecision_G);

          TempCostPlusLedgerEntry_G."C. Plus Unit Amount (Receipt)" :=
            -ROUND(( TbValueEntry_G."Invoiced Quantity" * OldExpectedAmount_L[3]) / OldExpectedQty_L, UnitAmtRndPrecision_G);
          TempCostPlusLedgerEntry_G.MODIFY(TRUE);
       END;

      // <-- 2020061-WI00-ADV01.01.00-3446-SCR01
    END;
