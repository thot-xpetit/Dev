    PROCEDURE CreateTransferReconciliation@1000000000();
    VAR
      Ecart_L@1000000000 : Decimal;
      FromLocation_L@1000000002 : Record 14;
      ToLocation_L@1000000003 : Record 14;
      SharedToSilo_L@1000000004 : Boolean;
      SharedFromSilo_L@1000000005 : Boolean;
      cduICInboxOutboxMgt_L@1000000006 : Codeunit 427;
      CellsLocation_L@1000000007 : Record 8053031;
    BEGIN
      Ecart_L := 0;

      TempReconciliationTransfer_G.INIT;
      TempReconciliationTransfer_G."Entry No." := 1;
      TempReconciliationTransfer_G."Date (Out)" := PostedDelivery_G."Sales Date";
      TempReconciliationTransfer_G."Document No. (Out)" := PostedDelivery_G."No.";
      TempReconciliationTransfer_G."Species Code (Out)" := PostedDelivery_G."Species Code";
      TempReconciliationTransfer_G."Item No. (Out)"    := PostedDelivery_G."Item No.";
      TempReconciliationTransfer_G."From-Location Code (Out)" := PostedDelivery_G."Location Code";
      TempReconciliationTransfer_G."To-Location Code (Out)" := PostedDelivery_G."Transfer-to Code";
      TempReconciliationTransfer_G."Net Weight (Out)" := PostedDelivery_G."Net Weight";
      TempReconciliationTransfer_G."Standard Weight (Out)" := PostedDelivery_G."Standard Weight";
      TempReconciliationTransfer_G."Manual Transf. No. (Out)" := PostedDelivery_G."Manual Transfer No.";
      TempReconciliationTransfer_G."Campaign No. (Out)" := PostedDelivery_G."Campaign No.";
      TempReconciliationTransfer_G."IC Partner Code (Out)" := PostedDelivery_G."IC Partner Code";
      TempReconciliationTransfer_G."Shipping Agent Code (Out)" := PostedDelivery_G."Shipping Agent Code";

      TempReconciliationTransfer_G."Document No. (In)"      := "CMC Posted Receipts"."No.";
      TempReconciliationTransfer_G."Species Code (In)"      := "CMC Posted Receipts"."Species Code";
      TempReconciliationTransfer_G."Item No. (In)"         := "CMC Posted Receipts"."Item No.";
      TempReconciliationTransfer_G."From-Location Code (In)":= "CMC Posted Receipts"."Transfer-from Code";
      TempReconciliationTransfer_G."To-Location Code (In)"  := "CMC Posted Receipts"."Location Code";
      TempReconciliationTransfer_G."Net Weight (In)"        := "CMC Posted Receipts"."Net Weight";
      TempReconciliationTransfer_G."Standard Weight (In)"   := "CMC Posted Receipts"."Standard Weight";
      TempReconciliationTransfer_G."Manual Transf. No. (In)":= "CMC Posted Receipts"."Manual Transfer No.";
      TempReconciliationTransfer_G."Campaign No. (In)"      := "CMC Posted Receipts"."Campaign No.";
      TempReconciliationTransfer_G."IC Partner Code (In)" := "CMC Posted Receipts"."IC Partner Code";
      TempReconciliationTransfer_G."Shipping Agent Code (In)" := "CMC Posted Receipts"."Shipping Agent Code";

      Ecart_L := TempReconciliationTransfer_G."Net Weight (In)" - TempReconciliationTransfer_G."Net Weight (Out)";
      IF (Ecart_L >= - CMCSetup_G."Independent Transfer Tolerance") AND
         (Ecart_L <=   CMCSetup_G."Independent Transfer Tolerance") THEN BEGIN
        TempReconciliationTransfer_G."Standard Gap Tolerated" :=
          TempReconciliationTransfer_G."Standard Weight (In)" - TempReconciliationTransfer_G."Standard Weight (Out)";
        TempReconciliationTransfer_G."Net Gap Tolerated" :=
          TempReconciliationTransfer_G."Net Weight (In)" - TempReconciliationTransfer_G."Net Weight (Out)";
      END ELSE BEGIN
       TempReconciliationTransfer_G."Standard Gap Out Tolerance" :=
         TempReconciliationTransfer_G."Standard Weight (In)" - TempReconciliationTransfer_G."Standard Weight (Out)";
       TempReconciliationTransfer_G."Net Gap Out Tolerance" :=
         TempReconciliationTransfer_G."Net Weight (In)" - TempReconciliationTransfer_G."Net Weight (Out)";
      END;

      IF (TempReconciliationTransfer_G."Manual Transf. No. (Out)" = TempReconciliationTransfer_G."Manual Transf. No. (In)") AND
        TempReconciliationTransfer_G.SpeciesReconcilied(TempReconciliationTransfer_G."Species Code (Out)",
          TempReconciliationTransfer_G."Species Code (In)") AND
        (TempReconciliationTransfer_G."From-Location Code (Out)" = TempReconciliationTransfer_G."From-Location Code (In)") AND
        //(TempReconciliationTransfer_G."To-Location Code (Out)" = TempReconciliationTransfer_G."To-Location Code (In)") AND
        (TempReconciliationTransfer_G."Campaign No. (Out)"=TempReconciliationTransfer_G."Campaign No. (In)")
      THEN BEGIN
        IF CMCSetup_G."Shipping Code On Reconc Transf" AND
          (TempReconciliationTransfer_G."Shipping Agent Code (Out)" <> TempReconciliationTransfer_G."Shipping Agent Code (In)")
        THEN
          CurrReport.SKIP;
        FromLocation_L.GET(TempReconciliationTransfer_G."From-Location Code (Out)");
        ToLocation_L.GET(TempReconciliationTransfer_G."To-Location Code (In)");
        SharedToSilo_L := cduICInboxOutboxMgt_L.CompanyManageSharedSilo(ToLocation_L);
        SharedFromSilo_L := cduICInboxOutboxMgt_L.CompanyManageSharedSilo(FromLocation_L);

        // --> 20160516-WI00-ADV01.01.00-2885-BGS0
        {BEFORE
        IF NOT ((FromLocation_L."Compensation Location" OR
          FromLocation_L."Shared Silo" AND NOT SharedFromSilo_L) AND
          (ToLocation_L."Shared Silo" AND NOT SharedToSilo_L))
        THEN BEGIN
        AFTER}
        // <-- 20160516-WI00-ADV01.01.00-2885-BGS0
          IF (Ecart_L >= - CMCSetup_G."Independent Transfer Tolerance") AND   // si dans la tol'rance
            (Ecart_L <= CMCSetup_G."Independent Transfer Tolerance")
          THEN BEGIN
            TempReconciliationTransfer_G.Reconciled := TRUE;
            TempReconciliationTransfer_G."Automatic Reconciliation" := TRUE;

            CASE CMCSetup_G."Difference On" OF
              CMCSetup_G."Difference On"::"Transfer-From Location" :
                BEGIN
                  IF FromLocation_L."Compensation Location" OR
                    FromLocation_L."Shared Silo" AND NOT SharedFromSilo_L
                  THEN BEGIN
                    TempReconciliationTransfer_G."Gap Location Code" := TempReconciliationTransfer_G."To-Location Code (In)";
                  END ELSE
                    TempReconciliationTransfer_G."Gap Location Code" := TempReconciliationTransfer_G."From-Location Code (Out)";
                  CLEAR(CellsLocation_L);
                  CellsLocation_L.SETRANGE("Location Code",TempReconciliationTransfer_G."Gap Location Code");
                  CellsLocation_L.SETRANGE("Ajustment Cell",TRUE);
                  IF CellsLocation_L.FINDFIRST THEN BEGIN
                    TempReconciliationTransfer_G."Gap Cell Code" := CellsLocation_L."Cell Code";
                  END ELSE BEGIN
                    IF FromLocation_L."Compensation Location" OR
                      FromLocation_L."Shared Silo" AND NOT SharedFromSilo_L
                    THEN BEGIN
                      TempReconciliationTransfer_G."Gap Cell Code" := "CMC Posted Receipts"."Cell Code";
                    END ELSE
                      TempReconciliationTransfer_G."Gap Cell Code" := PostedDelivery_G."Cell Code";
                  END;
                  IF (TempReconciliationTransfer_G."IC Partner Code (Out)" <> '') AND (FromLocation_L."Shared Silo") THEN BEGIN
                      TempReconciliationTransfer_G."Gap Location Code"  := TempReconciliationTransfer_G."From-Location Code (Out)";
                      CLEAR(CellsLocation_L);
                      TempReconciliationTransfer_G."Gap Cell Code"      :=
                        CellsLocation_L.GetUncontroledCell(TempReconciliationTransfer_G."From-Location Code (Out)");
                    END;
                END;
              CMCSetup_G."Difference On"::"Transfer-To Location" :
                BEGIN
                  IF ToLocation_L."Compensation Location" OR
                    ToLocation_L."Shared Silo" AND NOT SharedToSilo_L
                  THEN BEGIN
                    TempReconciliationTransfer_G."Gap Location Code" := TempReconciliationTransfer_G."From-Location Code (Out)"
                  END ELSE
                    TempReconciliationTransfer_G."Gap Location Code" := TempReconciliationTransfer_G."To-Location Code (In)";
                  CLEAR(CellsLocation_L);
                  CellsLocation_L.SETRANGE("Location Code",TempReconciliationTransfer_G."Gap Location Code");
                  CellsLocation_L.SETRANGE("Ajustment Cell",TRUE);
                  IF CellsLocation_L.FINDFIRST THEN BEGIN
                    TempReconciliationTransfer_G."Gap Cell Code" := CellsLocation_L."Cell Code";
