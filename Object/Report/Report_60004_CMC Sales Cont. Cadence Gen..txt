    PROCEDURE ReturnTotalOfPer@1000000002(StartDate@1000000001 : Date;EndDate@1000000004 : Date) NumTotalPeriodos : Decimal;
    VAR
      i@1000000003 : Integer;
      j@1000000002 : Integer;
      MesInicial@1000000000 : Integer;
    BEGIN
      // 20140901-CER20-TER01.01.00-RMZ01
      //----------------------------------------------------------------------------------------------------------
      // This function will return the number of months between two dates
      //----------------------------------------------------------------------------------------------------------
      // Input:
      //   - StartDate   : starting date
      // Outputs:
      //   - EndDate     : ending date
      //----------------------------------------------------------------------------------------------------------
      IF (StartDate = 0D) OR (EndDate = 0D) THEN
        EXIT;
      NumTotalPeriodos := 0;
      MesInicial := DATE2DMY(StartDate,2);
      IF (DATE2DMY(EndDate,2) = DATE2DMY(StartDate,2)) AND (DATE2DMY(StartDate,3) <> DATE2DMY(EndDate,3)) THEN
        EndDate := CALCDATE('<-1M>',EndDate);
      FOR i := DATE2DMY(StartDate,3) TO DATE2DMY(EndDate,3) DO BEGIN

        IF i <> DATE2DMY(EndDate,3) THEN BEGIN
          FOR j := MesInicial TO 12 DO
            NumTotalPeriodos := NumTotalPeriodos + 1;
        END
        ELSE BEGIN
          FOR j := MesInicial TO DATE2DMY(EndDate,2) DO
            NumTotalPeriodos := NumTotalPeriodos + 1;
        END;
        MesInicial := 1;
      END;
      IF NumTotalPeriodos = 0 THEN
        NumTotalPeriodos := 1;
      EXIT(NumTotalPeriodos);
    END;

    BEGIN

    }
    END.
