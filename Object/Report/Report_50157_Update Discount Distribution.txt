    BEGIN

        Text007 +
        '@1@@@@@@@@@@@@@@@@@@@@@@@@@\');
      Window.UPDATE(1,0);
      TotalRecNo := ExcelBuf.COUNT;
      RecNo := 0;

      IF ExcelBuf.FIND('-') THEN BEGIN
        OnBeforeProcess(TypeProcess);
        REPEAT
          RecNo := RecNo + 1;
          Window.UPDATE(1,ROUND(RecNo / TotalRecNo * 10000,1));
          CASE(TypeProcess) OF
            TypeProcess::Distribution : BEGIN
              ProcessDistribution(TempExcelBuf, HeaderRowNo, OldRowNo);

            TypeProcess::Payment : BEGIN
              ProcessPayment(TempExcelBuf, HeaderRowNo, OldRowNo);
            END;
          END;
        UNTIL ExcelBuf.NEXT = 0;
      END;
      ExcelBuf.DELETEALL;
      Window.CLOSE;
    END;
