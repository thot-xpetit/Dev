            G_NewSettEntryNo+=1;

            CMCSettOffLedEntTmp_G."Amount to Apply" := CMCSettOffLedEntTmp_G.Amount;

            CMCSettOffLedEntTmp_G."Credit Amount" := CVLedgEntry_L."Credit Amount";
            CMCSettOffLedEntTmp_G."Debit Amount (LCY)" := CVLedgEntry_L."Debit Amount (LCY)";
            CMCSettOffLedEntTmp_G."Credit Amount (LCY)" := CVLedgEntry_L."Credit Amount (LCY)";

      //      SettingOffRemain_L := GetRemainComp(0, CVLedgEntry_L."Entry No.");
      //      CMCSettOffLedEntTmp_G."Setting Off Remain" := SettingOffRemain_L;

      //MESSAGE('0:%1:%2', EcrComp_L."Entry No.", EcrComp_L.Amount);

        REPEAT
          CASE EcrComp_L."Entry Type" OF
            EcrComp_L."Entry Type"::Vendor, EcrComp_L."Entry Type"::Customer: BEGIN

              CMCSettOffLedEntTmp_G.SETRANGE("Original Ledger Entry No.", EcrComp_L."Entry No.");
              IF EcrComp_L."Entry Type" = EcrComp_L."Entry Type"::Customer THEN BEGIN
                CMCSettOffLedEntTmp_G.SETRANGE("Ledger type", CMCSettOffLedEntTmp_G."Ledger type"::Customer);

              END;

              IF CMCSettOffLedEntTmp_G.ISEMPTY THEN BEGIN

                  G_NewSettEntryNo+=1;
                  CMCSettOffLedEntTmp_G."Posting Date" := WORKDATE;
                  CMCSettOffLedEntTmp_G."Document Type" := CMCSettOffLedEntTmp_G."Document Type"::" ";
                  CMCSettOffLedEntTmp_G."Document No." := CMCRestraintEntry_L."Document No.";
                  IF CMCRestraint_L.GET(CMCRestraintEntry_L."Restraint Code") THEN BEGIN
                    CMCSettOffLedEntTmp_G.Description := CMCRestraint_L.Description;

                  CMCSettOffLedEntTmp_G."Original Ledger Entry No." := CMCRestraintEntry_L."Entry No.";
                  CMCSettOffLedEntTmp_G."Remaining Amount" := CMCRestraintEntry_L."Remaining Amount";
                  CMCSettOffLedEntTmp_G.Amount := CMCRestraintEntry_L.Amount;

                  CMCSettOffLedEntTmp_G."SettOff Amount to apply" := "CMC Seq. Comp. Journal LineDoc"."Vendor Balance";
                  CMCSettOffLedEntTmp_G."IC Partner Code" := "CMC Seq. Comp. Journal LineDoc"."Document No.";

      //            SettingOffRemain_L := GetRemainComp1(3, CMCRestraintEntry_L."Entry No.");

                  CMCSettOffLedEntTmp_G."Amount to Apply" := EcrComp_L.Amount ;
                  CMCSettOffLedEntTmp_G."SettOff Applies-to ID" := "CMC Seq. Comp. Journal LineDoc"."Document No.";

                  CMCSettOffLedEntTmp_G."Setting Off Remain" := SettingOffRemain_L;
