    PROCEDURE UpdateProgressBar@1000000006(index@1000000000 : Integer;Value@1000000001 : Integer);
    VAR
      ProgressValue@1000000002 : Decimal;
    BEGIN
      IF (NbSalesHeaderRecord <= 100) THEN BEGIN
        ProgressValue := ((Value / NbSalesHeaderRecord * 10000) DIV 1);
      END ELSE BEGIN
        IF ((Value MOD (NbSalesHeaderRecord DIV 100)) = 0) THEN BEGIN
          ProgressValue := ((Value / NbSalesHeaderRecord * 10000) DIV 1);
        END;
      END;
      IF (GUIALLOWED) THEN BEGIN
        IF (ProgressValue <> 0) THEN BEGIN
          Window.UPDATE(index, ProgressValue);
        END;
      END;
    END;
