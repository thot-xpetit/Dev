    LOCAL PROCEDURE AnalyzeData@2();
    VAR
      TempExcelBuf@1000 : TEMPORARY Record 370;
      HeaderRowNo@1003 : Integer;
      OldRowNo@1006 : Integer;
      valuecell@1000000000 : Text[30];
    BEGIN
      Window.OPEN(
        Text007 +
        '@1@@@@@@@@@@@@@@@@@@@@@@@@@\');
      Window.UPDATE(1,0);
      TotalRecNo := ExcelBuf.COUNT;
      RecNo := 0;

      IF ExcelBuf.FIND('-') THEN BEGIN
        REPEAT
          RecNo := RecNo + 1;
          Window.UPDATE(1,ROUND(RecNo / TotalRecNo * 10000,1));
          CASE TRUE OF
            (STRPOS(ExcelBuf."Cell Value as Text",Text009) <> 0):
              BEGIN
                IF HeaderRowNo = 0 THEN BEGIN
                  HeaderRowNo := ExcelBuf."Row No.";
                  TempExcelBuf := ExcelBuf;
                  TempExcelBuf.Comment := Text009;
                  TempExcelBuf.INSERT;
                END ELSE
                  ERROR(Text111);
              END;

            ExcelBuf."Row No." = HeaderRowNo:
              BEGIN
                TempExcelBuf := ExcelBuf;
                TempExcelBuf.Comment := ExcelBuf."Cell Value as Text";
                TempExcelBuf.INSERT;
              END;

            (ExcelBuf."Row No." > HeaderRowNo) AND (HeaderRowNo > 0):
              BEGIN
                IF ExcelBuf."Row No." <> OldRowNo THEN BEGIN
                  OldRowNo := ExcelBuf."Row No.";
                  ClearVariable();
                END;

                TempExcelBuf.SETRANGE("Column No.",ExcelBuf."Column No.");
                IF TempExcelBuf.FIND('-') THEN BEGIN
                  CASE TempExcelBuf.Comment OF
                    Text010: BEGIN
                      CrmGuidText := '{' + ExcelBuf."Cell Value as Text" + '}';

                      EVALUATE(CrmGuid,CrmGuidText);
                    END;
                    Text011: BEGIN
                      CustomerNo := ExcelBuf."Cell Value as Text";
                      UpdateContact();
                    END;
                  END;
                END;
              END;
          END;
        UNTIL ExcelBuf.NEXT = 0;
      END;


      Window.CLOSE;
    END;

    PROCEDURE UpdateContact@1000000011();
    VAR
      CmcCrmGuid@1000000000 : Record 8053601;
      Ctc@1000000001 : Record 5050;
    BEGIN
      // check CMC CRM Guid Table if don't exist then create
      // Get the contact and update GUI Dynmics Crm Field et set to true the CRM Dynamics Update field
      IF ((CustomerNo <> '') AND (NOT ISNULLGUID(CrmGuid))) THEN BEGIN
        Ctc.SETRANGE(Ctc."Customer No.", CustomerNo);
        Ctc.SETRANGE(Type, Ctc.Type::Company);
        IF (Ctc.FINDFIRST()) THEN BEGIN
          CmcCrmGuid.SETRANGE("Type Data", CmcCrmGuid."Type Data"::Account);
          CmcCrmGuid.SETRANGE("Code Nav", Ctc."No.");

            CmcCrmGuid.INIT();
            CmcCrmGuid."Type Data" := CmcCrmGuid."Type Data"::Account;
            CmcCrmGuid."Code Nav" := Ctc."No.";
            CmcCrmGuid.GUID := CrmGuid;
            CmcCrmGuid.INSERT();
          END ELSE BEGIN
            CmcCrmGuid.GUID := CrmGuid;
            CmcCrmGuid.MODIFY();
          END;
          Ctc."Dynamics CRM Id" := CrmGuid;
          Ctc."Contact CRM" := TRUE;
          Ctc.MODIFY();
        END;
      END;
    END;
