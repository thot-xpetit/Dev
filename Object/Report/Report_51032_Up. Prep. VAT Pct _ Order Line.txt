    PROCEDURE GetHeaderSetCurrency@1000000002(SalesOrderLine_P@1000000000 : Record 37);
    BEGIN
      SalesHeader_G.GET(SalesOrderLine_P."Document Type",SalesOrderLine_P."Document No.");
      IF SalesHeader_G."Currency Code" = '' THEN
        Currency_G.InitRoundingPrecision
      ELSE BEGIN
        SalesHeader_G.TESTFIELD("Currency Factor");
        Currency_G.GET(SalesHeader_G."Currency Code");
        Currency_G.TESTFIELD("Amount Rounding Precision");
      END;
    END;
