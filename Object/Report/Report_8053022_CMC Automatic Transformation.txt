  DATAITEMS
  {
    { PROPERTIES
      {
        DataItemTable=Table8052983;
        DataItemTableView=SORTING(Type)
                          ORDER(Ascending)
                          WHERE(Type=CONST(Contract),
                                Vendor No.=FILTER(<>''));
        OnPreDataItem=BEGIN
                        "CMC Collection Contract".SETFILTER("Purchase Date Limit",'(<>%1)&(<=%2)',0D,StartDate_G);

                        DlgWin_G.OPEN('#1###############################' + '\' +
                                           '@2@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@' + '\');
                        DlgWin_G.UPDATE(1, Text8053003);
                        TotalIter_G := "CMC Collection Contract".COUNT;
                        Iter_G := 0;
                      END;

        OnAfterGetRecord=VAR
                           ItemLedgerEntry_L@8053000 : Record 32;
                           PostedReceipts_L@8053007 : Record 8052995;
                           PostedReceipts2_L@8053006 : Record 8052995;
                           StatusMgtcdu_L@8053005 : Codeunit 8052985;
                           TotalIter_L@8053003 : Integer;
                           Iter_L@8053002 : Integer;
                           QteLetTest_L@8053001 : Decimal;
                           CMCManagementcdu_L@8053009 : Codeunit 8052982;
                           CMCCollectionContract_L@8053004 : Record 8052983;
                         BEGIN
                           DlgWin_G.UPDATE(2, ROUND(Iter_G / TotalIter_G * 10000,1));
                           Iter_G := Iter_G + 1;

                           ItemLedgerEntry_L.RESET;
                           ItemLedgerEntry_L.SETCURRENTKEY("Movement Type","Collection Contract No.");
                           ItemLedgerEntry_L.SETRANGE("Movement Type",ItemLedgerEntry_L."Movement Type"::S03);
                           ItemLedgerEntry_L.SETRANGE("Collection Contract No.","CMC Collection Contract"."No.");
                           IF ItemLedgerEntry_L.FINDFIRST THEN
                            CurrReport.SKIP; // On ne prend pas les contrats avec des reprises

                           PostedReceipts_L.RESET;
                           PostedReceipts_L.SETCURRENTKEY("Contract No.");
                           PostedReceipts_L.SETRANGE("Contract No.","CMC Collection Contract"."No.");
                           PostedReceipts_L.SETRANGE("Collection Type", PostedReceipts_L."Collection Type"::Storage);
                           PostedReceipts_L.SETFILTER("Receipt Type",'%1|%2',PostedReceipts_L."Receipt Type"::Receipt,
                                                                                PostedReceipts_L."Receipt Type"::Resumption);
                           PostedReceipts_L.SETRANGE(Canceled,FALSE);
                           PostedReceipts_L.SETRANGE(Invoiced,FALSE);
                           PostedReceipts_L.SETFILTER("Quantity To Invoice",'<>%1',0);

                           QteLetTest_L:=0;
                           IF PostedReceipts_L.FINDSET(FALSE, FALSE) THEN
                             REPEAT
                               QteLetTest_L := QteLetTest_L +PostedReceipts_L."Quantity To Lettred";
                             UNTIL PostedReceipts_L.NEXT=0;

                           IF QteLetTest_L <> 0 THEN
                            CurrReport.SKIP;

                           PostedReceipts_L.RESET;
                           PostedReceipts_L.SETCURRENTKEY("Contract No.");
                           PostedReceipts_L.SETRANGE("Contract No.","CMC Collection Contract"."No.");
                           PostedReceipts_L.SETRANGE("Collection Type", PostedReceipts_L."Collection Type"::Storage);
                           PostedReceipts_L.SETFILTER("Receipt Type",'%1|%2',PostedReceipts_L."Receipt Type"::Receipt,
                                                                                PostedReceipts_L."Receipt Type"::Resumption);
                           PostedReceipts_L.SETRANGE(Canceled,FALSE);
                           PostedReceipts_L.SETRANGE(Invoiced,FALSE);
                           PostedReceipts_L.SETFILTER("Quantity To Invoice",'<>%1',0);

                           IF PostedReceipts_L.FINDSET(TRUE, FALSE) THEN BEGIN

                               REPEAT
                                 CMCManagementcdu_L.LettrageMiseAchat(PostedReceipts_L);
                                 PostedReceipts2_L.GET(PostedReceipts_L."Receipt Type", PostedReceipts_L."No.");
                                 IF ( PostedReceipts2_L."Receipt Type" = PostedReceipts2_L."Receipt Type"::Receipt ) AND
                                    ( PostedReceipts2_L."Quantity To Invoice" - PostedReceipts2_L."Quantity To Lettred" > 0 )
                                 THEN BEGIN
                                   PostedReceipts2_L.VALIDATE(Status, PostedReceipts2_L.Status::Open);
                                   PostedReceipts2_L.VALIDATE("Collection Type", PostedReceipts_L."Collection Type");
                                   PostedReceipts2_L.VALIDATE("Purchase Date", PostedReceipts_L."Purchase Date Selection");
                                   PostedReceipts2_L.VALIDATE("Contract No.", PostedReceipts_L."Contract No.");
                                   PostedReceipts2_L.VALIDATE("Unit Price", PostedReceipts_L."Unit Price");
                                   PostedReceipts2_L."Quantity To Invoice":=
                                     PostedReceipts2_L."Quantity To Invoice" - PostedReceipts_L."Quantity To Lettred";
                                   PostedReceipts2_L."Collection Type" := PostedReceipts2_L."Collection Type"::Purchase;
                                   PostedReceipts2_L.MODIFY(TRUE);
                                   StatusMgtcdu_L.PurchasingPostedReceipt(PostedReceipts2_L, PostedReceipts_L."Purchase Date");
                                 END ELSE BEGIN
                                   PostedReceipts2_L."Quantity To Invoice":=
                                     PostedReceipts2_L."Quantity To Invoice"-PostedReceipts_L."Quantity To Lettred";
                                   PostedReceipts2_L."Collection Type" := PostedReceipts2_L."Collection Type"::Purchase;
                                   PostedReceipts2_L.MODIFY;
                                 END;

                               UNTIL PostedReceipts_L.NEXT = 0;
                               CMCCollectionContract_L.GET("CMC Collection Contract"."No.");
                               StatusMgtcdu_L.OpenContract(CMCCollectionContract_L);
                               CMCCollectionContract_L."Collection Type" := CMCCollectionContract_L."Collection Type"::Purchase;
                               CMCCollectionContract_L."Transformation Date" := TODAY;
                               CMCCollectionContract_L.MODIFY;
                               StatusMgtcdu_L.LockContract(CMCCollectionContract_L);
                            END;
                         END;

        OnPostDataItem=BEGIN
                         DlgWin_G.CLOSE;
                         MESSAGE(Text8053001);
                       END;

        ReqFilterFields=No.;
