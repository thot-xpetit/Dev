    LOCAL PROCEDURE AnalyzeData@2();
    VAR
      TempExcelBuf@1000 : TEMPORARY Record 370;
      HeaderRowNo@1003 : Integer;
      OldRowNo@1006 : Integer;
      StrDate@1000000000 : Text[30];
    BEGIN
      Window.OPEN(
        Text007 +
        '@1@@@@@@@@@@@@@@@@@@@@@@@@@\');
      Window.UPDATE(1,0);
      TotalRecNo := ExcelBuf.COUNT;
      RecNo := 0;

      IF ExcelBuf.FIND('-') THEN BEGIN
        REPEAT
          RecNo := RecNo + 1;
          Window.UPDATE(1,ROUND(RecNo / TotalRecNo * 10000,1));
          CASE TRUE OF
            (STRPOS(ExcelBuf."Cell Value as Text",Text008) <> 0):
              BEGIN
                IF HeaderRowNo = 0 THEN BEGIN
                  HeaderRowNo := ExcelBuf."Row No.";
                  TempExcelBuf := ExcelBuf;
                  TempExcelBuf.Comment := Text008;
                  TempExcelBuf.INSERT;
                END ELSE
                  ERROR(Text011);
              END;

            ExcelBuf."Row No." = HeaderRowNo:
              BEGIN
                TempExcelBuf := ExcelBuf;
                TempExcelBuf.Comment := ExcelBuf."Cell Value as Text";
                TempExcelBuf.INSERT;
              END;

            (ExcelBuf."Row No." > HeaderRowNo) AND (HeaderRowNo > 0):
              BEGIN
                IF ExcelBuf."Row No." <> OldRowNo THEN BEGIN
                  OldRowNo := ExcelBuf."Row No.";
                  ClearVariable();
                END;

                TempExcelBuf.SETRANGE("Column No.",ExcelBuf."Column No.");
                IF TempExcelBuf.FIND('-') THEN BEGIN
                  CASE TempExcelBuf.Comment OF
                    Text008:
                      BEGIN
                        //EVALUATE(EntryNo, ExcelBuf."Cell Value as Text");
                        Lib_TypeMouvement := ExcelBuf."Cell Value as Text";
                        IF (Lib_TypeMouvement = 'Devis') THEN BEGIN
                          DocType := DocType::Quote;
                        END;
                        IF (Lib_TypeMouvement = 'Commande') THEN BEGIN
                          DocType := DocType::Order;
                        END;
                        IF (Lib_TypeMouvement = 'Facture') THEN BEGIN
                          DocType := DocType::Invoice;
                        END;
                        IF (Lib_TypeMouvement = 'Avoir') THEN BEGIN
                          DocType := DocType::"Credit Memo";
                        END;
                        IF (Lib_TypeMouvement = 'Commande ouverte') THEN BEGIN
                          DocType := DocType::"Blanket Order";
                        END;
                        IF (Lib_TypeMouvement = 'Retour') THEN BEGIN
                          DocType := DocType::"Return Order";
                        END;
                      END;
                    Text009: BEGIN
                      No_Document := ExcelBuf."Cell Value as Text";
                    END;
                    Text010: BEGIN
                      NoContact := ExcelBuf."Cell Value as Text";
                    END;
                    Text011: BEGIN
                      FamilyCode := ExcelBuf."Cell Value as Text";
                    END;
                    Text012: BEGIN
                      FamilyName := ExcelBuf."Cell Value as Text";
                    END;
                    Text013: BEGIN
                      ItemNo := ExcelBuf."Cell Value as Text";
                    END;
                    Text014: BEGIN
                      VariantCode := ExcelBuf."Cell Value as Text";
                    END;
                    Text015: BEGIN
                      ItemDesc := ExcelBuf."Cell Value as Text";
                    END;
                    Text016: BEGIN
                      VariantDesc := ExcelBuf."Cell Value as Text";
                    END;
                    Text017: BEGIN
                    END;
                    Text018: BEGIN
                      StrDate := ExcelBuf."Cell Value as Text";
                      {
                      IF (STRLEN(StrDate) > 10) THEN BEGIN
                        StrDate := COPYSTR(StrDate,1, 10);
                      END;
                      IF (EVALUATE(LatestReqDelivDate, StrDate)) THEN BEGIN
                      }
                      IF (GetDate(StrDate, LatestReqDelivDate)) THEN BEGIN
                        UpdateSalesDocument();
                      END ELSE BEGIN
                        ERROR(STRSUBSTNO('PB Convertion Date : [%1]', StrDate));
                      END;
                    END;
                  END;
                END;
              END;

          END;
        UNTIL ExcelBuf.NEXT = 0;

    PROCEDURE UpdateSalesDocument@1000000009();
    VAR
      SalesHeader@1000000000 : Record 36;
      IsReleased@1000000001 : Boolean;
      ReleaseSalesDoc@1000000002 : Codeunit 414;
      ChangeDate@1000000003 : Boolean;
    BEGIN
      SalesHeader.SETRANGE("Document Type", DocType);
      SalesHeader.SETRANGE("No.", No_Document);
      IF (SalesHeader.FINDFIRST()) THEN BEGIN
        IsReleased := SalesHeader.Status =SalesHeader.Status::Released;
        ChangeDate :=  SalesHeader."Latest Req. Deliv. Date" <> LatestReqDelivDate;
        IF (ChangeDate) THEN BEGIN
          IF (IsReleased) THEN BEGIN
            ReleaseSalesDoc.Reopen(SalesHeader);
            SalesHeader.RESET();
            SalesHeader.SETRANGE("Document Type", DocType);
            SalesHeader.SETRANGE("No.", No_Document);
            IF (SalesHeader.FINDFIRST()) THEN BEGIN
            END;
          END;
          SalesHeader.VALIDATE("Latest Req. Deliv. Date", LatestReqDelivDate);
          SalesHeader.MODIFY(TRUE);

          IF (IsReleased) THEN BEGIN
            ReleaseSalesDoc.PerformManualRelease(SalesHeader);
          END;
          COMMIT();
        END;
      END;
    END;

    PROCEDURE GetDate@1000000000(StrDate@1000000000 : Text[30];VAR NewDate@1000000001 : Date) ReturnValue : Boolean;
    VAR
      Pos@1000000002 : Integer;

      ReturnValue := TRUE;
      NewDate := 0D;
      Pos := STRPOS(StrDate, ' ');
      IF (Pos <> 0) THEN BEGIN
        StrDate := COPYSTR(StrDate, 1, Pos - 1);
      END;
      IF (NOT EVALUATE(NewDate, StrDate)) THEN BEGIN
        ReturnValue := FALSE;
      END;
    END;
