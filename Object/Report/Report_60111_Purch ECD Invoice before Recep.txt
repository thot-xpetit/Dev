    PROCEDURE CalcRevPurchECDExpAmtToBalance@8053011(VAR ECDLedgerEntry_P@8053001 : Record 8053088;InvoicedQty_P@8053000 : Decimal);
    VAR
      ItemLedgerEntry_L@8053005 : Record 32;
      PurchECDLedgerEntry_L@8053004 : Record 8053088;
      OldExpectedQty_L@8053003 : Decimal;
      OldExpectedAmount_L@8053002 : ARRAY [3] OF Decimal;
    BEGIN
      // --> 20140805-APP33-CMC02.02.00-FDA01
      CLEAR(OldExpectedAmount_L);
      GLSetup.GET;

      WITH PurchECDLedgerEntry_L DO BEGIN
        SETCURRENTKEY("Item Ledger Entry No.","Entry Type");
        SETRANGE("Item Ledger Entry No.",ECDLedgerEntry_P."Item Ledger Entry No.");
        SETRANGE("Entry Type","Entry Type"::"Direct Cost");
        SETFILTER("Document Type", '%1|%2', "Document Type"::"Purchase Receipt", "Document Type"::"Purchase Return Shipment");
        //SETFILTER("Entry No.",'<%1',ECDLedgerEntry_P."Entry No.");
        SETRANGE("Item Charge No.",'');
        SETRANGE("Discount Code",ECDLedgerEntry_P."Discount Code");
        IF FINDSET THEN
          REPEAT
            OldExpectedQty_L := OldExpectedQty_L + "Valued Quantity";
            OldExpectedAmount_L[1] := OldExpectedAmount_L[1] + PurchECDLedgerEntry_L."Net Amount (Receipt)";
            OldExpectedAmount_L[2] := OldExpectedAmount_L[2] + PurchECDLedgerEntry_L."ECD Amount (Receipt)";
            OldExpectedAmount_L[3] := OldExpectedAmount_L[3] + PurchECDLedgerEntry_L."ECD Unit Amount (Receipt)";
          UNTIL NEXT = 0;

        IF OldExpectedQty_L <> 0 THEN BEGIN
          ECDLedgerEntry_P."Net Amount (Receipt)" :=
            -ROUND((InvoicedQty_P * OldExpectedAmount_L[1]) / OldExpectedQty_L, GLSetup."Amount Rounding Precision");

          ECDLedgerEntry_P."ECD Amount (Receipt)" :=
            -ROUND((InvoicedQty_P * OldExpectedAmount_L[2]) / OldExpectedQty_L, GLSetup."Amount Rounding Precision");

          ECDLedgerEntry_P."ECD Unit Amount (Receipt)" :=
            -ROUND((InvoicedQty_P * OldExpectedAmount_L[3]) / OldExpectedQty_L, GLSetup."Amount Rounding Precision");
        END;
      END;

    END;
