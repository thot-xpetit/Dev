      CerealsInvSch_L.RESET;

              END;

            IF DueDateFound_L THEN

        ("CMC Collection Contract"."Due Date" < "CMC Posted Receipt Line".Date) THEN BEGIN
        IF CerealsInvSch_L.FINDSET THEN BEGIN
          DueDateFound_L := FALSE;
          REPEAT
            IF "CMC Posted Receipt Line".Date <= CerealsInvSch_L."Collect Ending Date" THEN BEGIN
              DueDateFound_L := TRUE;
              DueDate_L := CerealsInvSch_L."Due Date";
            END;
          UNTIL (CerealsInvSch_L.NEXT = 0) OR DueDateFound_L;
          IF DueDateFound_L THEN
            InvoiceDueDate_L := DueDate_L;
        END;

      // <-- 20131126-Cer15-CMC02.02.00-BSI01
      END ELSE BEGIN
        IF CerealsInvSch_L.FINDSET THEN BEGIN
          DueDateFound_L := FALSE;
          REPEAT

              DueDateFound_L := TRUE;
              DueDate_L := CerealsInvSch_L."Due Date";
            END;
          UNTIL (CerealsInvSch_L.NEXT = 0) OR DueDateFound_L;
          IF DueDateFound_L THEN

        END;
      END;

      EXIT(InvoiceDueDate_L);

    END;
