    PROCEDURE CreatePrepCreditMemoLines@8053006(DocNo_P@8053004 : Code[20];GLAccountNo_P@8053002 : Code[20]);
    VAR
      SalesLine_L@8053003 : Record 37;
      SLPrepBuffer_L@8053006 : Record 8053522;
      PrepLogEntry_L@8053001 : Record 8053510;
      SalesHeader_L@8053000 : Record 36;
    BEGIN
      SalesHeader_L.GET(SalesHeader_L."Document Type"::"Credit Memo",DocNo_P);
      CLEAR(PrepLogEntry_L);
      PrepLogEntry_L.SETCURRENTKEY("Bill-to Customer No.","Main / Detailed","G/L Account No.","CMC Campaign");
      PrepLogEntry_L.SETRANGE("Bill-to Customer No.",Customer."No.");
      PrepLogEntry_L.SETRANGE("Main / Detailed",PrepLogEntry_L."Main / Detailed"::Detailed);
      PrepLogEntry_L.SETRANGE("G/L Account No.",GLAccountNo_P);
      PrepLogEntry_L.SETRANGE("CMC Campaign",Campagne_G);
      PrepLogEntry_L.CALCSUMS(Amount);
      IF PrepLogEntry_L.Amount <> 0 THEN BEGIN
        CLEAR(SalesLine_L);
        SalesLine_L.SETRANGE("Document Type",SalesLine_L."Document Type"::"Credit Memo");
        SalesLine_L.SETRANGE("Document No.",SalesHeader_L."No.");
        SalesLine_L.SETRANGE(Type,SalesLine_L.Type::"G/L Account");
        SalesLine_L.SETRANGE("No.",GLAccountNo_P);
        IF NOT SalesLine_L.FIND('-') THEN BEGIN
          SalesLine_L.INIT;
          SalesLine_L."Document Type" := SalesLine_L."Document Type"::"Credit Memo";
          SalesLine_L."Document No." := SalesHeader_L."No.";
          SalesLine_L."Line No." := IncrementLineNo(SalesLine_L);
          SalesLine_L."Sell-to Customer No." := SalesHeader_L."Sell-to Customer No.";
          SalesLine_L.Type := SalesLine_L.Type::"G/L Account";
          SalesLine_L.VALIDATE("No.",GLAccountNo_P);
          SalesLine_L.VALIDATE(Quantity,1);
          SalesLine_L.VALIDATE("Unit Price",PrepLogEntry_L.Amount);
          SalesLine_L."CMC Prepayment Line Type" := SalesLine_L."CMC Prepayment Line Type"::Prepayment;
          SalesLine_L."CMC Campaign No." := Campagne_G;
          SalesLine_L.INSERT(TRUE);

          // Insertion de l'enregistrement temporaire
          SLPrepBuffer_L.INIT;
          SLPrepBuffer_L."Document Type" := SalesLine_L."Document Type";
          SLPrepBuffer_L."Document No." := SalesLine_L."Document No.";
          SLPrepBuffer_L."Document Line No." := SalesLine_L."Line No.";
          SLPrepBuffer_L."Customer No." := SalesLine_L."Bill-to Customer No.";
          SLPrepBuffer_L."G/L Account No." := SalesLine_L."No.";
          SLPrepBuffer_L."Prepayment Amount" := - PrepLogEntry_L.Amount;
          SLPrepBuffer_L.Quantity := SalesLine_L.Quantity;
          SLPrepBuffer_L.Type := SalesLine_L.Type;
          SLPrepBuffer_L."No." := SalesLine_L."No.";
          SLPrepBuffer_L.Description := SalesLine_L.Description;
          SLPrepBuffer_L."Description 2" := SalesLine_L."Description 2";
          SLPrepBuffer_L."Entry Type" := SLPrepBuffer_L."Entry Type"::Prepayment;
          SLPrepBuffer_L."CMC Campaign" := Campagne_G;
          SLPrepBuffer_L.INSERT;
        END;
      END;
    END;
