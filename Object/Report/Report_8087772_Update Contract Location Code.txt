    PROCEDURE ProgressWindow@1000000004();
    BEGIN
      IF (GUIALLOWED) THEN BEGIN
        CurrRec += 1;
        IF NoOfRecs <= 100 THEN
          Window.UPDATE(1,(CurrRec / NoOfRecs * 10000) DIV 1)
        ELSE IF CurrRec MOD (NoOfRecs DIV 100) = 0 THEN
          Window.UPDATE(1,(CurrRec / NoOfRecs * 10000) DIV 1);
      END;
    END;

    PROCEDURE UpdateContract@1000000009(Contract@1000000000 : Record 8052983);
    VAR
      vendor@1000000001 : Record 23;
      Contact@1000000002 : Record 5050;
      UpdateCtct@1000000003 : Record 8052983;
      LocationCode@1000000004 : Code[10];
    BEGIN
      IF (UpdateCtct.GET(Contract."No.")) THEN BEGIN
        // Get the vendor
        IF (vendor.GET(UpdateCtct."Vendor No.")) THEN BEGIN
          LocationCode := vendor."Location Code";
          IF (LocationCode = '') THEN BEGIN
            IF (Contact.GET(vendor."No.")) THEN BEGIN
              LocationCode := Contact."Location Code";
            END;
          END;
          IF ((UpdateCtct."Default Location Code" = '') AND (LocationCode <> '')) THEN BEGIN
            UpdateCtct."Default Location Code" := Contact."Location Code";
            UpdateCtct.MODIFY();
          END;
        END;
      END;
    END;

    BEGIN
    {
      // If the locartion code exist in the vendor then this is prior than default locztion code founded into contact
    }
