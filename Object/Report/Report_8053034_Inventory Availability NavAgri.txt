      DateFinP2@1000000012 : Date;
      Text64000@1000000013 : TextConst 'FRA=Vous devez renseigner toutes les dates';
      Text64001@1000000014 : TextConst 'FRA=Les dates de fin ne doivent pas ^tre ant'rieures aux date de d'but';
      CurrDate@1000000015 : Text[30];

    PROCEDURE CalcNeed@1(Item@1000 : Record 27;LocationFilter@1001 : Text[250];VariantFilter@1002 : Text[250];ReorderPoint@1003 : Decimal);
    BEGIN
      WITH TabItem DO BEGIN

        RESET;
        GET(Item."No.");
        COPYFILTERS(Item);

        SETFILTER("Location Filter",LocationFilter);
        SETFILTER("Variant Filter",VariantFilter);

        //SETRANGE("Date Filter",0D,WORKDATE);
        CALCFIELDS(
          "Qty. on Purch. Order",
          "Planning Receipt (Qty.)",
          "Scheduled Receipt (Qty.)",
          "Purch. Req. Receipt (Qty.)",
          "Qty. in Transit",
          "Trans. Ord. Receipt (Qty.)",
          "Reserved Qty. on Inventory",
          "Qty. on Sales Order",
          "Qty. on Purch. Order");


          BackOrderQty :=
          "Qty. on Purch. Order" + "Scheduled Receipt (Qty.)" +
          "Qty. in Transit" + "Trans. Ord. Receipt (Qty.)" +
          "Planning Receipt (Qty.)" + "Purch. Req. Receipt (Qty.)";

        //SETRANGE("Date Filter",0D,31129999D);
        //GrossRequirement := AvailToPromise.CalcGrossRequirement(Item) + "Reserved Qty. on Inventory";
        //ScheduledReceipt := AvailToPromise.CalcScheduledReceipt(Item);

        CALCFIELDS(
          Inventory,
          "Planning Receipt (Qty.)",
          "Planned Order Receipt (Qty.)",
          "Purch. Req. Receipt (Qty.)");

        PlannedOrderReceipt :=
          "Planning Receipt (Qty.)" +
          "Planned Order Receipt (Qty.)" +
          "Purch. Req. Receipt (Qty.)";

        ProjAvailBalance :=
          Inventory +
          PlannedOrderReceipt +
          ScheduledReceipt -
          GrossRequirement;

        InvtReorder := ProjAvailBalance < ReorderPoint;

        TabItemCY.RESET;
        TabItemCY.GET(Item."No.");
        TabItemCY.COPYFILTERS(TabItem);
        TabItemCY.SETRANGE("Date Filter",DateDebP1,DateFinP1);
        TabItemCY.SETFILTER("Location Filter",LocationFilter);
        TabItemCY.CALCFIELDS("Ship (Qty.)");

        TabItemLY.RESET;
        TabItemLY.GET(Item."No.");
        TabItemLY.COPYFILTERS(TabItem);
        TabItemLY.SETRANGE("Date Filter",DateDebP2,DateFinP2);
        TabItemLY.SETFILTER("Location Filter",LocationFilter);
        TabItemLY.CALCFIELDS("Ship (Qty.)");

      END;
    END;
