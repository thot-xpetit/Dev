    PROCEDURE FindDoc@60000(FileName@60008 : Text[1000];IDNo@60011 : Integer;IDType@60012 : 'TableData,Table,Form,Report,Dataport,Codeunit');
    VAR
      F@60007 : File;
      DWORD@60006 : Text[4];
      IsEnd@60005 : Boolean;
      DocLine@60004 : Text[1000];
      IsDoc@60003 : Boolean;
      Chr@60002 : Char;
      IsOnlyZero@60001 : Boolean;
      WasZero@60000 : Boolean;
      i@60013 : Integer;
    BEGIN
      CLEAR(DocLine);
      IsDoc := FALSE;
      F.TEXTMODE := FALSE;
      F.OPEN(FileName);      //Open file with binary data for object (exported blob)
      IsEnd:=FALSE;

      WHILE NOT IsEnd DO BEGIN     //end of file or end of documentation
        WasZero := IsOnlyZero;
        IsOnlyZero := TRUE;
        FOR i:= 1 TO 4 DO BEGIN    //read 4 bytes, if 00 00 00 00 then IsOnlyZero is true
          IsEnd:= F.READ(Chr)=0;
          DWORD[i] := Chr;
          IsOnlyZero := IsOnlyZero AND (DWORD[i]=0);
        END;
        IF WasZero AND (DWORD[1]=115) AND (DWORD[2]=101) THEN   //  00 00 00 00 73 65  - begin of documentation
          IsDoc := TRUE
        ELSE
        IF ((DWORD[1]=24) OR (DWORD[1]=23)) AND (DWORD[2]=101) THEN BEGIN // 18 65 ?? ?? or 17 65 ?? ?? is end of doc... if was doc
          IF IsDoc THEN BEGIN
            InsertDocLine(DocLine);   //insert new line into table (or somewhere)
            DocLine := '';                                //read new line
            IF DWORD[1]=23 THEN                           // 17 65 ?? ?? is end of all documentation, end loop
              IsEnd := TRUE;
          END;
          IsDoc := FALSE;
          IsOnlyZero := TRUE;
        END ELSE
        IF IsDoc THEN                        //is in doc, read text into buffer
          DocLine := DocLine + DWORD;
      END;
      F.CLOSE;
    END;

    PROCEDURE DelBefAftBlank@60003(Input@60000 : Text[100];IsDelBefore@60001 : Boolean;IsDelAfter@60002 : Boolean) : Text[100];

        IF COPYSTR(Input,1,1) = ' ' THEN
          REPEAT
            Input := DELSTR(Input,1,1);
          UNTIL COPYSTR(Input,1,1) <> ' ';

      IF IsDelAfter THEN
        IF COPYSTR(Input,STRLEN(Input),1) = ' ' THEN
          REPEAT
            Input := DELSTR(Input,STRLEN(Input),1);
          UNTIL COPYSTR(Input,STRLEN(Input),1) <> ' ';

      EXIT(Input);
    END;
