    PROCEDURE SetGroupAccountCorrectif@8053007(VAR SalesHeader_P@1000000000 : Record 36);
    VAR
      Cust_L@8053001 : Record 18;
      GroupAccCode_L@8053000 : Code[10];
      SalesLine_L@1000000001 : Record 37;
    BEGIN
      CLEAR(GroupAccountMgt_G);
      Launched_G := FALSE;
      IF SalesHeader_P.Status = SalesHeader_P.Status::Released THEN BEGIN
        Launched_G := TRUE;

      ExclusionLiv_G := FALSE;
      IF ((SalesHeader_P."Shipment Method Code" = 'LIV') AND (SalesHeader_P."Doc. Category Code" = 'ENLEV')) THEN
        ExclusionLiv_G := TRUE;

      IF SalesHeader_P."Bill-to Customer No." = '' THEN
        EXIT;


      SalesHeader_P.SetHideValidationDialog(TRUE);
      GroupAccCode_L := '';

      IF GUIALLOWED AND (NOT HideCollectifChoice_G) AND GroupAccountMgt_G.SalesIsGroupAccUpdateAllowed(SalesHeader_P) THEN BEGIN
        GroupAccCode_L := GroupAccountMgt_G.LookupGroupAccount(0,SalesHeader_P."Bill-to Customer No.");
          IF Option_G = Option_G::correction THEN BEGIN
            //si commande lanc', on rouvre
            IF GroupAccCode_L <> '' THEN BEGIN
             IF ExclusionLiv_G = FALSE THEN BEGIN
                IF Launched_G THEN
                  ReleaseSalesDocument_G.Reopen(SalesHeader_P);

                SalesHeader_P.VALIDATE("Group Account Code",GroupAccCode_L);
                SalesHeader_P.MODIFY;
             END;

            END;
          END;
        END;
      END;

      Cust_L.GET(SalesHeader_P."Bill-to Customer No.");
      IF (GroupAccCode_L = '') AND (Cust_L."Group Account Code" <> '') THEN BEGIN
        GroupAccCode_L := Cust_L."Group Account Code";
        IF Option_G = Option_G::correction THEN BEGIN
          //si commande lanc', on rouvre
          IF GroupAccCode_L <> '' THEN BEGIN
           IF ExclusionLiv_G = FALSE THEN BEGIN
            IF ((Launched_G = TRUE) AND (SalesHeader_P.Status = SalesHeader_P.Status::Released)) THEN
              ReleaseSalesDocument_G.Reopen(SalesHeader_P);
            SalesHeader_P.VALIDATE("Group Account Code",Cust_L."Group Account Code");
            SalesHeader_P.MODIFY;


           END;
          END;
        END;
      END;

      CollectifCorrection_G := GroupAccCode_L;

      IF Option_G = Option_G::correction THEN BEGIN
        IF ((Launched_G) AND (SalesHeader_P.Status <> SalesHeader_P.Status::Released)) THEN BEGIN
          //on ne peut relancer une commande que s'il existe quelque chose . lancer
          //sinon, on a un message d'erreur
          //LigneExiste_L := FALSE;
          IF ExclusionLiv_G = FALSE THEN BEGIN
            SalesLine_L.RESET;
            SalesLine_L.SETRANGE("Document Type",SalesHeader_P."Document Type");
            SalesLine_L.SETRANGE("Document No.",SalesHeader_P."No.");
            IF NOT SalesLine_L.ISEMPTY THEN
              ReleaseSalesDocument_G.PerformManualRelease(SalesHeader_P);
          END;
        END;
      END;
    END;

    BEGIN
    END.
