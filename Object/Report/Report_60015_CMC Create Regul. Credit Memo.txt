    PROCEDURE GetAxeLieu@60001() AxeCode_Res : Code[10];
    VAR
      RecCollContract_L@60000 : Record 8052983;
      RecPostedReceiptsLine_L@60001 : Record 8053104;
      TmpLocation_L@60002 : TEMPORARY Record 14;
      RecLocation_L@60003 : Record 14;
    BEGIN
      // --> 20150603-CER148-TER01.01.00-302-BSI01
      AxeCode_Res := '';
      TmpLocation_L.DELETEALL;
      IF RecCollContract_L.GET(ContractNo_G) THEN BEGIN
        CLEAR(RecPostedReceiptsLine_L);
        RecPostedReceiptsLine_L.SETRANGE("Receipt Type",RecPostedReceiptsLine_L."Receipt Type"::Receipt);
        RecPostedReceiptsLine_L.SETRANGE("Contract No.",ContractNo_G);
        RecPostedReceiptsLine_L.SETFILTER("Location Code",'<>%1','');
        IF RecPostedReceiptsLine_L.FIND('-') THEN
        REPEAT
          TmpLocation_L.INIT;
          TmpLocation_L.Code := RecPostedReceiptsLine_L."Location Code";
          TmpLocation_L."Cathedral Concret Cells Numb" := 1;
          IF NOT TmpLocation_L.INSERT THEN BEGIN
            TmpLocation_L.GET(RecPostedReceiptsLine_L."Location Code");
            TmpLocation_L."Cathedral Concret Cells Numb" += 1;
            TmpLocation_L.MODIFY;
          END;
        UNTIL RecPostedReceiptsLine_L.NEXT = 0;

        TmpLocation_L.SETCURRENTKEY("Cathedral Concret Cells Numb");
        IF TmpLocation_L.FINDLAST THEN BEGIN
          RecLocation_L.GET(TmpLocation_L.Code);
          AxeCode_Res := RecLocation_L."Analytic Section";
        END;
      END;
    END;
