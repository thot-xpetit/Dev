    LOCAL PROCEDURE AnalyzeData@2();
    VAR
      TempExcelBuf@1000 : TEMPORARY Record 370;
      BudgetBuf@1001 : Record 371;
      TempBudgetBuf@1002 : TEMPORARY Record 371;
      HeaderRowNo@1003 : Integer;
      CountDim@1004 : Integer;
      TestDate@1005 : Date;
      OldRowNo@1006 : Integer;
      DimRowNo@1007 : Integer;
      DimCode3@1008 : Code[20];
      TimePos@1009 : Integer;
    BEGIN
      Window.OPEN(
        Text007 +
        '@1@@@@@@@@@@@@@@@@@@@@@@@@@\');
      Window.UPDATE(1,0);
      TotalRecNo := ExcelBuf.COUNT;
      RecNo := 0;

      gIntNbIntegre := 0;
      gIntErr := 0;

      GenJournalLine_G.RESET;
      GenJournalLine_G.SETFILTER("Journal Template Name",CodGeneralJournalTemplate_G);
      GenJournalLine_G.SETFILTER("Journal Batch Name",CodGeneralJournal_G);
      IF NOT GenJournalLine_G.ISEMPTY THEN
        ERROR(STRSUBSTNO(Txt60006,CodGeneralJournalTemplate_G,CodGeneralJournal_G));

      GenJournalTemplate_G.RESET;
      GenJournalTemplate_G.GET(CodGeneralJournalTemplate_G);

      IntLine_G := 0;
      DecTotalAmount_G := 0;


      IF ExcelBuf.FIND('-') THEN BEGIN
        REPEAT

          RecNo := RecNo + 1;
          Window.UPDATE(1,ROUND(RecNo / TotalRecNo * 10000,1));

          IF HeaderRowNo = 0 THEN
            HeaderRowNo := ExcelBuf."Row No."
          ELSE
            IF HeaderRowNo <> ExcelBuf."Row No." THEN BEGIN
              IF HeaderRowNo > 1 THEN
                fNulAjoutEnreg(HeaderRowNo);
              CLEAR(TabTxt_G);
              HeaderRowNo := ExcelBuf."Row No.";
            END;

          IF ExcelBuf."Column No." IN [1..100] THEN
            TabTxt_G[ExcelBuf."Column No."] := COPYSTR(ExcelBuf."Cell Value as Text",1,250);

        UNTIL ExcelBuf.NEXT = 0;
        fNulAjoutEnreg(HeaderRowNo);
      END;


      IF gIntNbIntegre > 0 THEN BEGIN
        IF gIntErr > 0 THEN
          MESSAGE(Text003,'',0,gIntErr)
        ELSE
          MESSAGE(Text004,'',gIntNbIntegre);
      END ELSE BEGIN
        IF gIntErr > 0 THEN
          MESSAGE(Text002,'',gIntErr);
      END;

      Window.CLOSE;
    END;
