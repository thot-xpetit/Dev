        IF gIntErr > 0 THEN

        ELSE
          MESSAGE(Text004,'',gIntNbIntegre);
      END ELSE BEGIN
        IF gIntErr > 0 THEN
          MESSAGE(Text002,'',gIntErr);
      END;


      Window.CLOSE;
    END;
