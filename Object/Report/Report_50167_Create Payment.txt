    BEGIN

      TotalRecNo := ExcelBuf.COUNT;
      RecNo := 0;

      IF ExcelBuf.FIND('-') THEN BEGIN

      Window.CLOSE;
    END;

    PROCEDURE ProcessPayment@1000000012(VAR TempExcelBuf@1000000002 : TEMPORARY Record 370;VAR HeaderRowNo@1000000001 : Integer;VAR OldRowNo@1000000000 : Integer);
    BEGIN
      CASE TRUE OF
        (ExcelBuf."Cell Value as Text" = Text009):
        BEGIN
          IF HeaderRowNo = 0 THEN BEGIN
            HeaderRowNo := ExcelBuf."Row No.";
            TempExcelBuf := ExcelBuf;
            TempExcelBuf.Comment := Text009;
            TempExcelBuf.INSERT;
            CreateHeader := TRUE;
          END ELSE
            ERROR(STRSUBSTNO(Text011, Text009));
          END;

          ExcelBuf."Row No." = HeaderRowNo:
          BEGIN
            TempExcelBuf := ExcelBuf;
            TempExcelBuf.Comment := ExcelBuf."Cell Value as Text";
            TempExcelBuf.INSERT;
          END;

          (ExcelBuf."Row No." > HeaderRowNo) AND (HeaderRowNo > 0):
          BEGIN
            CreatePaymentHeader();
            IF ExcelBuf."Row No." <> OldRowNo THEN BEGIN
              OldRowNo := ExcelBuf."Row No.";
              ClearVariable();
            END;

            TempExcelBuf.SETRANGE("Column No.",ExcelBuf."Column No.");
            IF TempExcelBuf.FIND('-') THEN BEGIN
              CASE TempExcelBuf.Comment OF
                Text009: BEGIN
                  //EVALUATE(NoCCA, ExcelBuf."Cell Value as Text");
                  VendorCode := ExcelBuf."Cell Value as Text";
                END;
                Text010: BEGIN
                  VendorName := ExcelBuf."Cell Value as Text";
                END;
                Text012: BEGIN
                  BankCode := ExcelBuf."Cell Value as Text";
                END;
                Text013: BEGIN
                  EVALUATE(BankAmount, ExcelBuf."Cell Value as Text");
                END;
                Text014: BEGIN
                  EVALUATE(ValueDate, ExcelBuf."Cell Value as Text");
                END;
                Text015: BEGIN
                  //ContactNo := ExcelBuf."Cell Value as Text";

                END;
                Text016: BEGIN
                  //CommitmentCode := ExcelBuf."Cell Value as Text";
                  EVALUATE(Amount, ExcelBuf."Cell Value as Text");
                END;
                Text017 : BEGIN
                  EVALUATE(CalculAmount, ExcelBuf."Cell Value as Text");
                END;
                Text018 : BEGIN

            END;

    END;

    PROCEDURE UpdatePayment@1000000026();
    VAR
      PaymentLine@1000000000 : Record 10866;
      LastPaymentLine@1000000001 : Record 10866;
      LineNo@1000000002 : Integer;
    BEGIN
      IF (CheckDataLine()) THEN BEGIN
        LastPaymentLine.SETRANGE("No.", PaymentHeader."No.");
        IF (LastPaymentLine.FINDLAST()) THEN BEGIN
          LineNo := LastPaymentLine."Line No." + 10000;
        END ELSE BEGIN
          LineNo := 10000;
        END;
        PaymentLine.INIT();
        PaymentLine.VALIDATE("No.", PaymentHeader."No.");
        PaymentLine."Line No." := LineNo;
        PaymentLine.SetUpNewLine(LastPaymentLine, TRUE);
        PaymentLine.VALIDATE("Account Type", PaymentLine."Account Type"::Vendor);
        PaymentLine.VALIDATE("Account No.", VendorCode);
        PaymentLine.VALIDATE("Due Date",DT2DATE(ValueDate));
        PaymentLine.VALIDATE(Amount, ROUND(CalculAmount,0.01,'='));
        PaymentLine.INSERT(TRUE);
      END;
    END;

    PROCEDURE CreatePaymentHeader@1000000013();
    VAR
      Process@1000000000 : Record 10860;
      NoSeriesMgt@1000000001 : Codeunit 396;
    BEGIN
      IF (CreateHeader) THEN BEGIN
        Process.GET(PaymentClassCode);
        Process.TESTFIELD("Header No. Series");
        PaymentHeader.INIT();
        NoSeriesMgt.InitSeries(Process."Header No. Series",PaymentHeader."No. Series",0D,PaymentHeader."No.",PaymentHeader."No. Series");
        PaymentHeader.VALIDATE("Payment Class",Process.Code);
        // --> 20111101-FINA03-CMC02.01.00-BRA01
        PaymentHeader."Multi Bank Management" := Process."Multi Bank Management";
        // <-- 20111101-FINA03-CMC02.01.00-BRA01
        // --> 20130719-8060990/100-CMB6.00.10.09-BRA01
        PaymentHeader.Reference := Process."Default Reference";
        // <-- 20130719-8060990/100-CMB6.00.10.09-BRA01
        IF (PaymentMethod <> '') THEN BEGIN
          PaymentHeader.VALIDATE("Payment Method Code", PaymentMethod);
        END;
        PaymentHeader.VALIDATE("Posting Date", PostingDate);
        PaymentHeader.VALIDATE("Document Date", DocumentDate);
        PaymentHeader.VALIDATE(Reference, Mention);
        PaymentHeader.INSERT(TRUE);
        CreateHeader := FALSE;
      END;
    END;

    PROCEDURE CheckDataLine@1000000017() ReturnValue : Boolean;
    VAR
      Vendor@1000000000 : Record 23;
    BEGIN
      ReturnValue := FALSE;
      IF ((VendorCode <> '') AND (CalculAmount <> 0)) THEN BEGIN
        // PAD with '0' Character if le lenght of the code is < 5
        IF (STRLEN(VendorCode) < 5) THEN BEGIN
          REPEAT
            VendorCode := '0' + VendorCode;
          UNTIL (STRLEN(VendorCode) = 5);
        END;
        IF (Vendor.GET(VendorCode)) THEN BEGIN
          IF (Vendor.Blocked <> Vendor.Blocked::" ") THEN BEGIN
            AddLogEvent(STRSUBSTNO(TErrVendorBlocked,VendorCode, Vendor.Blocked::" "));
          END ELSE BEGIN
            ReturnValue := TRUE;
          END;
        END ELSE BEGIN
          AddLogEvent(STRSUBSTNO(TErrVendorNoExist,VendorCode));
        END;

      END;
    END;
