    PROCEDURE CreateItemChargeAssgnt@5801(P_SalesLine@8053000 : Record 37);
    VAR
      L_AssignItemChargeSales@8053001 : Codeunit 5807;
      L_SalesHeader@8053002 : Record 36;
      L_ItemChargeAssgntSales@8053003 : Record 5809;
      L_Currency@8053004 : Record 4;
    BEGIN
      // create item charge assignment for current sales line
      WITH P_SalesLine DO
       BEGIN
        TESTFIELD(Type,Type::"Charge (Item)");
        TESTFIELD("No.");
        TESTFIELD(Quantity);

        L_ItemChargeAssgntSales.RESET;
        L_ItemChargeAssgntSales.SETRANGE("Document Type","Document Type");
        L_ItemChargeAssgntSales.SETRANGE("Document No.","Document No.");
        L_ItemChargeAssgntSales.SETRANGE("Document Line No.","Line No.");
        L_ItemChargeAssgntSales.SETRANGE("Item Charge No.","No.");
        IF NOT L_ItemChargeAssgntSales.FINDLAST THEN BEGIN
          L_ItemChargeAssgntSales."Document Type" := "Document Type";
          L_ItemChargeAssgntSales."Document No." := "Document No.";
          L_ItemChargeAssgntSales."Document Line No." := "Line No.";
          L_ItemChargeAssgntSales."Line No." := 10000;
          L_ItemChargeAssgntSales."Item Charge No." := "No.";
          L_ItemChargeAssgntSales."Item No." := G_RecSalesLine."No.";
          L_ItemChargeAssgntSales.Description := G_RecSalesLine.Description;
          L_ItemChargeAssgntSales."Applies-to Doc. Type" := "Document Type";
          L_ItemChargeAssgntSales."Applies-to Doc. No." := "Document No.";
          L_ItemChargeAssgntSales."Applies-to Doc. Line No." := "CMC Sales Line No.";
          L_SalesHeader.GET("Document Type","Document No.");
          IF L_SalesHeader."Currency Code" <> '' THEN
           L_Currency.GET(L_SalesHeader."Currency Code");
          IF ("Inv. Discount Amount" = 0) AND
             ("Line Discount Amount" = 0) AND
             (NOT L_SalesHeader."Prices Including VAT")
          THEN
            L_ItemChargeAssgntSales."Unit Cost" := "Unit Price"
          ELSE
            IF L_SalesHeader."Prices Including VAT" THEN
              L_ItemChargeAssgntSales."Unit Cost" :=
                ROUND(
                  ("Line Amount" - "Inv. Discount Amount") / Quantity / (1 + "VAT %" / 100),
                  L_Currency."Unit-Amount Rounding Precision")
            ELSE
              L_ItemChargeAssgntSales."Unit Cost" :=
                ROUND(
                  ("Line Amount" - "Inv. Discount Amount") / Quantity,
                  L_Currency."Unit-Amount Rounding Precision");
        END;

        L_ItemChargeAssgntSales.VALIDATE("Qty. to Assign", Quantity);
        IF NOT L_ItemChargeAssgntSales.INSERT(TRUE) THEN
         L_ItemChargeAssgntSales.MODIFY;

       END;
    END;
