    PROCEDURE ACPToCreate@1000000000();
    BEGIN
      // --> 20200610-WI00-ADV01.01.00-3446-SCR01

      // Appliquer les mises . jour ACP

      CLEAR(TbPurchACPLedgerEntry_G);
      TbPurchACPLedgerEntry_G.SETRANGE("Item Ledger Entry Type",TbValueEntry_G."Item Ledger Entry Type");
      TbPurchACPLedgerEntry_G.SETRANGE("Item Ledger Entry No.",TbValueEntry_G."Item Ledger Entry No.");
      TbPurchACPLedgerEntry_G.SETRANGE("Value Entry No.",TbValueEntry_G."Entry No.");
      TbPurchACPLedgerEntry_G.SETRANGE("ACP Code" ,ACPCode_G);
      IF NOT TbPurchACPLedgerEntry_G.FINDSET THEN
        CreateACP
      ELSE BEGIN
        // Recherche de la nouvelle valeur ACP
        FindNewValue;
        // Mise . jour ACP Ledger Entry
        updateACP;
      END;

      // <-- 20200610-WI00-ADV01.01.00-3446-SCR01
    END;

    PROCEDURE ACPToUpdate@1000000002();
    BEGIN
      // --> 20200610-WI00-ADV01.01.00-3446-SCR01

      // Mise . jour des lignes ACP

      CLEAR(TbPurchACPLedgerEntry_G);
      TbPurchACPLedgerEntry_G.SETRANGE("Item Ledger Entry Type",TbValueEntry_G."Item Ledger Entry Type");
      TbPurchACPLedgerEntry_G.SETRANGE("Item Ledger Entry No.",TbValueEntry_G."Item Ledger Entry No.");
      TbPurchACPLedgerEntry_G.SETRANGE("Value Entry No.",TbValueEntry_G."Entry No.");
      TbPurchACPLedgerEntry_G.SETRANGE("ACP Code" ,ACPCode_G);
      TbPurchACPLedgerEntry_G.SETRANGE(Canceled,FALSE);
      TbPurchACPLedgerEntry_G.SETRANGE("ACP  No. final doc. purch.",0);
      IF TbPurchACPLedgerEntry_G.FINDSET THEN BEGIN

        // Mise . jour Value Entry ACP
        TbValueEntryACP_G.GET(TbValueEntry_G."Entry No.",ACPCode_G);
        TempValueEntryACP_G.INIT;
        TempValueEntryACP_G.TRANSFERFIELDS(TbValueEntryACP_G, TRUE);
        TempValueEntryACP_G.INSERT(TRUE);

        IF ACPType_G = ACPType_G::Pourcentage THEN BEGIN
           TempValueEntryACP_G.VALIDATE("Acp %",ACPPct_G);
           TempValueEntryACP_G."Acp Amount" := 0;
           IF ACPPct_G > 0 THEN
            TempValueEntryACP_G.VALIDATE("Acp Amount",
                                        ROUND(TempValueEntryACP_G."Base Value" * ACPPct_G /100,
                                        UnitAmtRndPrecision_G,'='));
        END;
        IF ACPType_G = ACPType_G::"Montant unitaire" THEN BEGIN
            TempValueEntryACP_G."Acp %" := 0;
            TempValueEntryACP_G.VALIDATE("Acp Amount",ACPValue_G);
        END;
        TempValueEntryACP_G.MODIFY(TRUE);

        // Mise . jour ACP Ledgr Entry
        updateACP;

      END;

      // <-- 20200610-WI00-ADV01.01.00-3446-SCR01
    END;

          // R'ceptiion, Retour

            TempACPLedgerEntry_G."ACP Amount (Receipt)" := 0;

            TempACPLedgerEntry_G.INSERT(TRUE);

            // Vers Excell Apres
            MakeExcelDataBody2(TempACPLedgerEntry_G);

    PROCEDURE CalcRevPurchACPExpAmtToBalance@1000000011();
    VAR
      EntryNo_L@1000000002 : Integer;
      OldExpectedQty_L@1000000001 : Decimal;
      OldExpectedAmount_L@1000000000 : ARRAY [3] OF Decimal;
    BEGIN
      // --> 20200610-WI00-ADV01.01.00-3446-SCR01

      EntryNo_L := TempACPLedgerEntry_G."Entry No.";
      CLEAR(TempACPLedgerEntry_G);
      WITH TempACPLedgerEntry_G DO BEGIN
        SETCURRENTKEY("Item Ledger Entry No.","Entry Type");
        SETRANGE("Item Ledger Entry No.",TbValueEntry_G."Item Ledger Entry No.");
        SETRANGE("Entry Type","Entry Type"::"Direct Cost");
        SETFILTER("Document Type", '%1|%2', "Document Type"::"Purchase Receipt", "Document Type"::"Purchase Return Shipment");
        SETRANGE("Resource No.",'');
        SETRANGE("ACP Code",ACPCode_G);
        IF FINDSET THEN
          REPEAT
            OldExpectedQty_L := OldExpectedQty_L + "Valued Quantity";
            OldExpectedAmount_L[2] := OldExpectedAmount_L[2] + "ACP Amount (Receipt)";
            OldExpectedAmount_L[3] := OldExpectedAmount_L[3] + "ACP Unit Amount (Receipt)";
          UNTIL NEXT = 0;
       END;
       CLEAR(TempACPLedgerEntry_G);
       TempACPLedgerEntry_G.GET(EntryNo_L);
       IF OldExpectedQty_L <> 0 THEN BEGIN
          TempACPLedgerEntry_G."ACP Amount (Receipt)" :=
            -ROUND((TbValueEntry_G."Invoiced Quantity" * OldExpectedAmount_L[2]) / OldExpectedQty_L, AmtRndPrecision_G);

          TempACPLedgerEntry_G."ACP Unit Amount (Receipt)" :=
            -ROUND(( TbValueEntry_G."Invoiced Quantity" * OldExpectedAmount_L[3]) / OldExpectedQty_L, UnitAmtRndPrecision_G);
          TempACPLedgerEntry_G.MODIFY(TRUE);
       END;

      // <-- 20200610-WI00-ADV01.01.00-3446-SCR01
    END;
