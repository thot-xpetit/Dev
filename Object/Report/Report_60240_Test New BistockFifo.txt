    PROCEDURE ProcessItemVariant@1000000002(ItemVariant_P@1000000000 : Record 5401);
    VAR
      AverageCost_L@1000000001 : Decimal;
      AverageCostACY_L@1000000002 : Decimal;
      ECDAmount_L@1000000003 : Decimal;
    BEGIN
      TotalQty_G := 0;
      TotalValue_G := 0;
      TotalEcd_G :=0;

      Item.SETFILTER("Date Filter",'..%1',InventoryDate_G);
      IF ItemVariant_P.Code <> '' THEN
        Item.SETRANGE("Variant Filter",ItemVariant_P.Code)
      ELSE
        Item.SETRANGE("Variant Filter");

      Item.CALCFIELDS(Inventory);
      IF Item.Inventory <> 0 THEN BEGIN
        CalcValueInventory(ItemVariant_P,Item.Inventory,TotalValue_G);
        ECDAmount_L := GetECDAmount(Item.Inventory,ItemVariant_P);
        InsertItemCostValueAndEcd(ItemVariant_P,ROUND((TotalValue_G - ECDAmount_L)/ Item.Inventory,0.01),Item.Inventory,
        ROUND(ECDAmount_L / Item.Inventory,0.01));
        IF ECDAmount_L <> 0 THEN
          InsertItemCostValueComp('RFC',ItemVariant_P,ROUND(ECDAmount_L / Item.Inventory,0.01));

      END;

      //gestion des 'tapes
      IF Item.Inventory <> 0 THEN BEGIN
        FctInsertStep(1,Text60001,ItemVariant_P,0,Item.Inventory,TotalValue_G +
          ECDAmount_L,'');
        IF ECDAmount_L <> 0 THEN
          FctInsertStep(2,Text60002,ItemVariant_P,0,0,-ECDAmount_L,'');

        //

      END;
    END;

    PROCEDURE CalcValueEntryItemVariant@1000000000(ItemVariant_P@1000000000 : Record 5401;Qty_P@1000000004 : Decimal) : Decimal;
    VAR
      ValueEntry_L@1000000001 : Record 5802;
      BooEndLoop_L@1000000002 : Boolean;
      RemainingQty_L@1000000003 : Decimal;
      ValuedQty_L@1000000005 : Decimal;
      CumulCost_L@1000000006 : Decimal;
      EntryNo_L@1000000007 : Integer;
    BEGIN
      //on doit atteindre la Quantit' en parcourant les achat du plus r'cent au plus ancien
      RemainingQty_L := Qty_P;
      ValueEntry_L.SETCURRENTKEY("Item No.","Variant Code","Posting Date","Document Type","Drop Shipment");
      ValueEntry_L.ASCENDING(FALSE);

      ValueEntry_L.SETRANGE("Item No.",ItemVariant_P."Item No.");
      ValueEntry_L.SETRANGE("Variant Code",ItemVariant_P.Code);
      ValueEntry_L.SETRANGE("Item Ledger Entry Type",ValueEntry_L."Item Ledger Entry Type"::Purchase);
      ValueEntry_L.SETFILTER("Posting Date",'..%1',InventoryDate_G);
      ValueEntry_L.SETRANGE("Item Charge No.",'');
      IF ValueEntry_L.FIND('-') THEN REPEAT
            IF ValueEntry_L."Valued Quantity" >= RemainingQty_L THEN BEGIN
              ValuedQty_L := RemainingQty_L;
              BooEndLoop_L := TRUE;
              CumulCost_L := CumulCost_L + (ValuedQty_L * RetunrUnitCost(ValueEntry_L));
            END ELSE BEGIN
              ValuedQty_L := ValueEntry_L."Valued Quantity";
              RemainingQty_L := RemainingQty_L - ValueEntry_L."Valued Quantity";
              CumulCost_L := CumulCost_L + (ValueEntry_L."Cost Amount (Expected)" + ValueEntry_L."Cost Amount (Actual)");
            END;
            EntryNo_L := ValueEntry_L."Entry No.";
      UNTIL (ValueEntry_L.NEXT = 0) OR (BooEndLoop_L);
      EXIT(CumulCost_L);
    END;

    PROCEDURE RetunrUnitCost@1000000013(ValueEntry_P@1000000000 : Record 5802) : Decimal;
    BEGIN
      IF ValueEntry_P."Valued Quantity" <> 0 THEN BEGIN
          EXIT((ValueEntry_P."Cost Amount (Expected)" + ValueEntry_P."Cost Amount (Actual)") / ValueEntry_P."Valued Quantity");
      END;
      EXIT(0);
    END;

      //on doit atteindre la Quantit' en parcourant les achat du plus r'cent au plus ancien

      ItemLedgerEntry_L.SETRANGE("Item No.",ItemVariant_P."Item No.");

      IF BooNegativeAjustement_G THEN
      // <-- 20211117-WI00-ADV01.01.00-SP3499-CJY01
        ItemLedgerEntry_L.SETFILTER("Entry Type",'%1|%2|%3|%4',ItemLedgerEntry_L."Entry Type"::Purchase,
          ItemLedgerEntry_L."Entry Type"::"Positive Adjmt.",ItemLedgerEntry_L."Entry Type"::Output,
            ItemLedgerEntry_L."Entry Type"::"Negative Adjmt.")
      // --> 20211117-WI00-ADV01.01.00-SP3499-CJY01
      ELSE
      // <-- 20211117-WI00-ADV01.01.00-SP3499-CJY01
        ItemLedgerEntry_L.SETFILTER("Entry Type",'%1|%2|%3',ItemLedgerEntry_L."Entry Type"::Purchase,
          ItemLedgerEntry_L."Entry Type"::"Positive Adjmt.",ItemLedgerEntry_L."Entry Type"::Output);

      ItemLedgerEntry_L.SETFILTER("Posting Date",'..%1',InventoryDate_G);

      IF ItemLedgerEntry_L.FIND('-') THEN REPEAT

            ValuePart_L:= 0;

            // --> 20220104-WI00-ADV01.01.00-SP3499-CJY02
            {BEFORE
            //on ne traite pas les : "achat n'gatif"
            IF NOT ((ItemLedgerEntry_L."Entry Type" = ItemLedgerEntry_L."Entry Type"::Purchase) AND
              (ItemLedgerEntry_L.Quantity <0)) THEN BEGIN
            AFTER}
            BooToManage_L := TRUE;
            IF (ItemLedgerEntry_L."Entry Type" = ItemLedgerEntry_L."Entry Type"::Purchase) AND
              (ItemLedgerEntry_L.Quantity <0) THEN
                IF BooNegativePurchInventory_G = FALSE THEN
                  BooToManage_L := FALSE;

            IF BooToManage_L THEN BEGIN
            // <-- 20220104-WI00-ADV01.01.00-SP3499-CJY02

              CostInter_L := 0;
              IF ItemLedgerEntry_L.Quantity >= RemainingQty_L THEN BEGIN
                ValuedQty_L := RemainingQty_L;
                BooEndLoop_L := TRUE;
                CostInter_L := ValuedQty_L * (CalcOnValueEntry(ItemLedgerEntry_L,ItemCostCalcDateDate_G,ValuedQty_L)/
                                              ItemLedgerEntry_L.Quantity);
               CumulCost_L := CumulCost_L + CostInter_L;
              END ELSE BEGIN
                ValuedQty_L := ItemLedgerEntry_L.Quantity;
                RemainingQty_L := RemainingQty_L - ItemLedgerEntry_L.Quantity;
                CostInter_L := CalcOnValueEntry(ItemLedgerEntry_L,ItemCostCalcDateDate_G,ValuedQty_L);
                CumulCost_L := CumulCost_L + CostInter_L;
              END;
               IF BooDetailGenerated_G THEN

                ItemCostCalcStepsDetailed_L.FctWriteStep1_FIFO(TextEcrArticleAdvitam,
                  ItemLedgerEntry_L."Document No.",ItemLedgerEntry_L."Document Line No.",
                  ItemLedgerEntry_L."Entry No.",ValuedQty_L,0,0,
                  ItemCostCalcCodeCode_G,ItemCostCalcDateDate_G,'',ItemLedgerEntry_L."Entry No.",1,
                  CostInter_L);

            END;
      UNTIL (ItemLedgerEntry_L.NEXT = 0) OR (BooEndLoop_L);
      CumulCost_P := CumulCost_L;
    END;

                  ValuedQty_L := RemainingQty_L;
                  EndLoop_L := TRUE;

                      BooSkipECDReceipt_L := FALSE;
                      ValuedQtyToInvoice_L := PurchInvValueEntry_L."Valued Quantity";
                      RemainingQtyToInvoice_L := RemainingQtyToInvoice_L - PurchInvValueEntry_L."Invoiced Quantity";
                    END;
                    ValuedQty_L := RemainingQtyToInvoice_L;
                    //Calcul sur les RFC factur's
                    PurchECDLedgerEntry_L.RESET;
                    PurchECDLedgerEntry_L.SETCURRENTKEY("Value Entry No.");
                    PurchECDLedgerEntry_L.SETRANGE("Value Entry No.",PurchInvValueEntry_L."Entry No.");
                    PurchECDLedgerEntry_L.SETFILTER("Date Filter",'..%1',InventoryDate_G);
                    IF PurchECDLedgerEntry_L.FINDSET THEN
                      BEGIN
                        REPEAT
                          PurchECDLedgerEntry_L.CALCFIELDS("Det. ECD Amount (Invoice)");
                          ECDValue_L += ValuedQtyToInvoice_L *
                            (PurchECDLedgerEntry_L."Det. ECD Amount (Invoice)" / PurchECDLedgerEntry_L."Valued Quantity");
                          IF BooDetailGenerated_G THEN

                            ItemCostCalcStepsDetailed_L.FctWriteStep2_FIFO(TextRfcProrataFac,
                              PurchECDLedgerEntry_L."Document No.",PurchECDLedgerEntry_L."Document Line No.",
                              PurchECDLedgerEntry_L."Item Ledger Entry No.",ValuedQtyToInvoice_L,0,0,
                              ItemCostCalcCodeCode_G,ItemCostCalcDateDate_G,'',PurchECDLedgerEntry_L."Entry No.",2,
                              PurchECDLedgerEntry_L."Det. ECD Amount (Invoice)" / PurchECDLedgerEntry_L."Valued Quantity");


                        UNTIL (PurchECDLedgerEntry_L.NEXT = 0);
                      END
                      ELSE BEGIN
                          //'criture du d'tail quand pas de RFC sur la document trouv'
                        IF BooDetailGenerated_G THEN
                          ItemCostCalcStepsDetailed_L.FctWriteStep2SansRFC_FIFO(TextNoRfcProrataFac,
                            PurchInvValueEntry_L."Document No.",PurchInvValueEntry_L."Document Line No.",
                            PurchInvValueEntry_L."Item Ledger Entry No.",ValuedQtyToInvoice_L,0,0,
                            ItemCostCalcCodeCode_G,ItemCostCalcDateDate_G,'',PurchInvValueEntry_L."Entry No.",2,
                            0,PurchInvValueEntry_L);

                      END;
                  UNTIL (PurchInvValueEntry_L.NEXT(-1) = 0) OR BooSkipECDReceipt_L;
                END ELSE
                  RemainingQtyToInvoice_L := 0;

                    IF (("Document Type" = "Document Type"::"Purchase Receipt") OR
                      ("Document Type" = "Document Type"::"Purchase Return Shipment")) THEN
                IF NOT BooSkipECDReceipt_L THEN BEGIN
                  //Calcul sur les RFC r'ception => Calcul s'effectue aussi si facturation partielle (au prorata des qt's restantes)
                  PurchECDLedgerEntry_L.RESET;
                  PurchECDLedgerEntry_L.SETCURRENTKEY("Value Entry No.");
                  PurchECDLedgerEntry_L.SETRANGE("Value Entry No.","Entry No.");
                  PurchECDLedgerEntry_L.SETFILTER("Date Filter",'..%1',InventoryDate_G);
                  IF PurchECDLedgerEntry_L.FINDSET THEN
                    REPEAT
                      PurchECDLedgerEntry_L.CALCFIELDS("Det. ECD Amount (Receipt)");
                      ECDValue_L += ValuedQty_L *
                        (PurchECDLedgerEntry_L."Det. ECD Amount (Receipt)" / PurchECDLedgerEntry_L."Valued Quantity");

                        IF BooDetailGenerated_G THEN
                          ItemCostCalcStepsDetailed_L.FctWriteStep2_FIFO(TextRfcProrataRec,
                            PurchECDLedgerEntry_L."Document No.",PurchECDLedgerEntry_L."Document Line No.",
                            PurchECDLedgerEntry_L."Item Ledger Entry No.",ValuedQty_L,0,0,
                            ItemCostCalcCodeCode_G,ItemCostCalcDateDate_G,'',PurchECDLedgerEntry_L."Entry No.",2,
                            PurchECDLedgerEntry_L."Det. ECD Amount (Receipt)" / PurchECDLedgerEntry_L."Valued Quantity");

                    UNTIL PurchECDLedgerEntry_L.NEXT = 0;
                END;
              END;
            // --> 20220104-WI00-ADV01.01.00-SP3499-CJY02
            END;
            // <-- 20220104-WI00-ADV01.01.00-SP3499-CJY02

          UNTIL (NEXT(-1) = 0) OR EndLoop_L;
        EXIT(ECDValue_L);

      END;
    END;
