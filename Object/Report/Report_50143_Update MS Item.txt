    LOCAL PROCEDURE AnalyzeData@2();
    VAR
      TempExcelBuf@1000 : TEMPORARY Record 370;
      HeaderRowNo@1003 : Integer;
      OldRowNo@1006 : Integer;
    BEGIN
      Window.OPEN(
        Text007 +
        '@1@@@@@@@@@@@@@@@@@@@@@@@@@\');
      Window.UPDATE(1,0);
      TotalRecNo := ExcelBuf.COUNT;
      RecNo := 0;
      OldRowNo := -1;

      IF ExcelBuf.FIND('-') THEN BEGIN
        REPEAT
          RecNo := RecNo + 1;
          Window.UPDATE(1,ROUND(RecNo / TotalRecNo * 10000,1));
          CASE TRUE OF
            //(STRPOS(ExcelBuf."Cell Value as Text",Text009) <> 0):
            (ExcelBuf."Cell Value as Text" = Text009) :
              BEGIN
                IF HeaderRowNo = 0 THEN BEGIN
                  HeaderRowNo := ExcelBuf."Row No.";
                  TempExcelBuf := ExcelBuf;
                  TempExcelBuf.Comment := Text009;
                  TempExcelBuf.INSERT;
                END ELSE
                  ERROR(Text011);
              END;

            ExcelBuf."Row No." = HeaderRowNo:
              BEGIN
                TempExcelBuf := ExcelBuf;
                TempExcelBuf.Comment := ExcelBuf."Cell Value as Text";
                TempExcelBuf.INSERT;
              END;

            (ExcelBuf."Row No." > HeaderRowNo) AND (HeaderRowNo > 0):
              BEGIN
                IF ExcelBuf."Row No." <> OldRowNo THEN BEGIN
                  IF (OldRowNo <> -1) THEN BEGIN
                    UpdateMS();
                  END;
                  OldRowNo := ExcelBuf."Row No.";
                  ClearVariable();
                END;

                TempExcelBuf.SETRANGE("Column No.",ExcelBuf."Column No.");
                IF TempExcelBuf.FIND('-') THEN BEGIN
                  CASE TempExcelBuf.Comment OF
                    Text010:
                      BEGIN
                        //EVALUATE(EntryNo, ExcelBuf."Cell Value as Text");
                        ItemNo := GetValue(ExcelBuf."Cell Value as Text");
                      END;
                    Text013: BEGIN
                      Description2 := GetValue(ExcelBuf."Cell Value as Text");
                      //UpdateMSItem();
                    END;
                    Text029 : BEGIN
                      VariantCode  := GetValue(ExcelBuf."Cell Value as Text");
                    END;
                    Text030 : BEGIN
                      VariantDesc2 := GetValue(ExcelBuf."Cell Value as Text");
                    END;
                    Text031 : BEGIN
                      KitBomNo := GetValue(ExcelBuf."Cell Value as Text");
                      //UpdateMS();
                    END;
                  END;
                END;
              END;
          END;
        UNTIL ExcelBuf.NEXT = 0;
      END;


      Window.CLOSE;
      {
      TempExcelBuf.RESET;
      TempExcelBuf.SETRANGE(Comment,Text010);
      IF NOT TempExcelBuf.FIND('-') THEN
        ERROR(Text025);
      TempExcelBuf.SETRANGE(Comment,Text014);
      IF NOT TempExcelBuf.FIND('-') THEN
        ERROR(Text026);
      }
    END;

    PROCEDURE UpdateMSItem@1000000011();
    VAR
      MSItem@1000000001 : Record 8053079;
      blocked@1000000000 : Boolean;
      Published@1000000002 : Boolean;
    BEGIN
      blocked := FALSE;
      Published := FALSE;
      IF ((ItemNo <> '') AND ((Description2 <> '') OR (KitBomNo <> '')) AND (VariantCode='')) THEN BEGIN
        IF (GetMSItem(MSItem, ItemNo)) THEN BEGIN
          blocked := (MSItem.Status=MSItem.Status::Closed);
          IF (blocked) THEN BEGIN
            OpenItem(MSItem);
            GetMSItem(MSItem, ItemNo);
          END;
          IF ((MSItem."Description 2" <> Description2) AND (Description2 <> '')) THEN BEGIN
            MSItem.VALIDATE("Description 2", Description2);
            MSItem.MODIFY(FALSE);
            Published := TRUE;
          END;
          IF ((MSItem."Kit BOM No." <> KitBomNo) AND (KitBomNo <> '')) THEN BEGIN
            MSItem."Kit BOM No." := KitBomNo;
            MSItem.MODIFY(FALSE);
            Published := TRUE;
          END;
          IF ((MSItem."Kit Disassembly BOM No." <> KitBomNo) AND (KitBomNo <> '')) THEN BEGIN
            MSItem."Kit Disassembly BOM No." := KitBomNo;
            MSItem.MODIFY(FALSE);
            Published := TRUE;
          END;


          IF (blocked) THEN BEGIN
            CloseItem(MSItem);
            GetMSItem(MSItem, ItemNo);
          END;
          IF (Published) THEN BEGIN
            PublishItem(MSItem);
          END;
        END;

      END;
    END;
