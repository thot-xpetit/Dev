    PROCEDURE GetGoodDate@8053015(P_NumSeq@8053001 : Integer;P_Date@8053000 : Date) Res_Date : Date;
    VAR
      L_AllowPostingTo@8053003 : Date;
      L_ItemLegEnt@8053002 : Record 32;
      Text8053009@8053004 : TextConst 'ENU=Receipt Finalisation;FRA=La date de travail n''est pas dans la p'riode AgriMer autoris'e.';
    BEGIN
      G_CMCSetup.GET;
      Res_Date:=P_Date;
      L_AllowPostingTo:=G_CMCSetup."ONIC Allow Posting To";
      IF G_CMCSetup."ONIC Allow Posting To" = 0D THEN
           L_AllowPostingTo:= 31129999D;

      IF L_ItemLegEnt.GET(P_NumSeq) THEN BEGIN
        IF (L_ItemLegEnt."Posting Date" >= G_CMCSetup."ONIC Allow Posting From") AND (L_ItemLegEnt."Posting Date"<= L_AllowPostingTo)
        THEN
          Res_Date:=L_ItemLegEnt."Posting Date";
      END;

      IF Res_Date=P_Date THEN BEGIN
        IF (P_Date < G_CMCSetup."ONIC Allow Posting From") OR (P_Date> L_AllowPostingTo) THEN
          ERROR(Text8053009)
      END;

      EXIT(Res_Date);
    END;
